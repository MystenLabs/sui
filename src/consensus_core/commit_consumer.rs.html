13/// Arguments from commit consumer to this consensus instance.
14#[derive(Clone)]
15pub struct CommitConsumerArgs {
16    pub(crate) replay_mode: ReplayMode,
17    pub(crate) commit_sender: UnboundedSender<CommittedSubDag>,
18    monitor: Arc<CommitConsumerMonitor>,
19}
20
21#[derive(Clone, Copy, Debug)]
22pub(crate) enum ReplayMode {
23    /// Replay from a consumer-supplied watermark without acknowledgement pacing.
24    FromWatermark {
25        /// Commit indices start at one. Replay starts at this index plus one;
26        /// zero requests the entire epoch.
27        replay_after_commit_index: CommitIndex,
28        /// Consumer-supplied target for startup readiness.
29        consumer_last_processed_commit_index: CommitIndex,
30    },
31    /// Replay the entire epoch, discovering the target from consensus storage.
32    /// Only the finalized prefix is paced by application acknowledgements;
33    /// the unfinalized tail and live consensus keep their existing flow control.
34    FullEpochPaced,
35}
36
37impl CommitConsumerArgs {
38    /// Requests replay after `replay_after_commit_index`, using the consumer's
39    /// last processed index as the recovery target. This mode does not wait for
40    /// application acknowledgements during startup.
41    /// Commit indices start at one; use zero to replay from the first commit.
42    pub fn new(
43        replay_after_commit_index: CommitIndex,
44        consumer_last_processed_commit_index: CommitIndex,
45    ) -> (Self, UnboundedReceiver<CommittedSubDag>) {
46        Self::new_with_replay_mode(ReplayMode::FromWatermark {
47            replay_after_commit_index,
48            consumer_last_processed_commit_index,
49        })
50    }
51
52    /// Replays the epoch from its first commit into a consumer with no durable
53    /// execution watermark. Consensus owns storage and delivers historical and
54    /// live commits over the returned stream.
55    ///
56    /// This is an integration API for external consensus consumers that rebuild
57    /// application state on restart. DO NOT REMOVE this entry point solely because it
58    /// has no production callers in this workspace. Changes to this contract
59    /// must account for downstream consumers.
60    ///
61    /// Consensus discovers the startup target from its store: the last stored
62    /// commit when transaction voting is disabled, or the last finalized commit
63    /// when voting is enabled. It waits for application acknowledgements between
64    /// recovery batches. An unfinalized tail goes through normal finalization
65    /// without waiting for application at startup, since it may need live consensus.
66    /// Consequently, pacing bounds only the finalized prefix: if voting is enabled
67    /// and no commits are finalized, the entire history is queued without pacing.
68    ///
69    /// The consumer must run before awaiting [`crate::ConsensusAuthority::start`]
70    /// and call [`CommitConsumerMonitor::set_highest_handled_commit`] only after
71    /// applying each commit. Waiting for startup before consuming would deadlock
72    /// recovery. Use [`Self::monitor`] to observe readiness and committed-head
73    /// progress without accessing storage.
74    /// Startup panics if the consumer closes while recovery is waiting for an
75    /// application acknowledgement.
76    pub fn new_with_full_replay() -> (Self, UnboundedReceiver<CommittedSubDag>) {
77        Self::new_with_replay_mode(ReplayMode::FullEpochPaced)
78    }
79
80    fn new_with_replay_mode(replay_mode: ReplayMode) -> (Self, UnboundedReceiver<CommittedSubDag>) {
81        let (replay_after_commit_index, replay_target) = match replay_mode {
82            ReplayMode::FromWatermark {
83                replay_after_commit_index,
84                consumer_last_processed_commit_index,
85            } => (
86                replay_after_commit_index,
87                Some(consumer_last_processed_commit_index),
88            ),
89            ReplayMode::FullEpochPaced => (0, None),
90        };
91        let (commit_sender, commit_receiver) = unbounded_channel("consensus_commit_output");
92        let monitor = Arc::new(CommitConsumerMonitor::new(
93            replay_after_commit_index,
94            replay_target,
95        ));
96        (
97            Self {
98                replay_mode,
99                commit_sender,
100                monitor,
101            },
102            commit_receiver,
103        )
104    }
105
106    pub fn monitor(&self) -> Arc<CommitConsumerMonitor> {
107        self.monitor.clone()
108    }
109}

111/// Process-local committed head reported by consensus, independently of application progress.
112#[derive(Clone, Copy, Debug, PartialEq, Eq)]
113pub struct CommitProgress {
114    /// Latest committed index reported by consensus, independently of consumer progress.
115    /// Recovery reports the stored head before replay, rather than each historical
116    /// commit as it is delivered. Live commits advance this value afterwards.
117    pub highest_committed_index: CommitIndex,
118    /// Leader round associated with [`Self::highest_committed_index`], including
119    /// the stored head during replay rather than the consumer's replay position.
120    pub highest_committed_round: Round,
121}
122
123/// Tracks application progress, startup readiness, and the consensus head.
124///
125/// External consumers observe the committed head, handled commits, and replay
126/// target separately to monitor catch-up without reading consensus storage.
127/// DO NOT REMOVE these observation APIs solely because they have no callers in
128/// this workspace. They support the external integration described by
129/// [`CommitConsumerArgs::new_with_full_replay`].
130pub struct CommitConsumerMonitor {
131    replay_target: SetOnce<CommitIndex>,
132    committed: watch::Sender<CommitProgress>,
133    handled: watch::Sender<CommitIndex>,
134}
135
136impl CommitConsumerMonitor {
137    pub(crate) fn new(
138        replay_after_commit_index: CommitIndex,
139        replay_target: Option<CommitIndex>,
140    ) -> Self {
141        Self {
142            replay_target: SetOnce::new_with(replay_target),
143            committed: watch::Sender::new(CommitProgress {
144                highest_committed_index: 0,
145                highest_committed_round: 0,
146            }),
147            handled: watch::Sender::new(replay_after_commit_index),
148        }
149    }
150
151    /// Returns the consensus head, independently of consumer acknowledgements.
152    pub fn progress(&self) -> CommitProgress {
153        *self.committed.borrow()
154    }
155
156    /// Subscribes to consensus head changes only. To observe application progress
157    /// and target discovery, also use [`Self::subscribe_handled_commit`] and
158    /// [`Self::wait_for_replay_target`].
159    pub fn subscribe_progress(&self) -> watch::Receiver<CommitProgress> {
160        self.committed.subscribe()
161    }
162
163    /// Returns the fixed startup target. With [`CommitConsumerArgs::new_with_full_replay`],
164    /// `None` means storage has not been inspected and `Some(0)` means there is no
165    /// finalized startup work. With [`CommitConsumerArgs::new`], the consumer-supplied
166    /// last processed index is available immediately.
167    pub fn replay_target(&self) -> Option<CommitIndex> {
168        self.replay_target.get().copied()
169    }
170
171    /// Waits for target discovery, without waiting for the consumer to apply it.
172    pub async fn wait_for_replay_target(&self) -> CommitIndex {
173        *self.replay_target.wait().await
174    }
175
176    pub(crate) fn set_replay_target(&self, target: CommitIndex) {
177        self.replay_target
178            .set(target)
179            .expect("startup replay target must be set only once");
180    }
181
182    pub(crate) fn report_committed(&self, index: CommitIndex, round: Round) {
183        self.committed.send_if_modified(|progress| {
184            if index <= progress.highest_committed_index {
185                return false;
186            }
187            progress.highest_committed_index = index;
188            progress.highest_committed_round = round;
189            true
190        });
191    }
192
193    pub fn highest_handled_commit(&self) -> CommitIndex {
194        *self.handled.borrow()
195    }
196
197    /// Subscribes to consumer acknowledgements, independently of consensus progress.
198    pub fn subscribe_handled_commit(&self) -> watch::Receiver<CommitIndex> {
199        self.handled.subscribe()
200    }
201
202    /// Acknowledge application, after the consumer has finished processing the commit.
203    pub fn set_highest_handled_commit(&self, highest_handled_commit: CommitIndex) {
204        debug!("Highest handled commit set to {}", highest_handled_commit);
205        self.handled.send_replace(highest_handled_commit);
206    }
207
208    pub(crate) async fn wait_for_handled(&self, target: CommitIndex) {
209        let mut handled = self.subscribe_handled_commit();
210        loop {
211            if *handled.borrow_and_update() >= target {
212                return;
213            }
214            handled.changed().await.unwrap();
215        }
216    }
217
218    /// Waits until the consumer has applied the startup replay target, including
219    /// target discovery for full replay. An undiscovered target is not an empty
220    /// database.
221    pub async fn replay_to_consumer_last_processed_commit_complete(&self) {
222        let target = self.wait_for_replay_target().await;
223        self.wait_for_handled(target).await;
224    }
225}

236    #[tokio::test]
237    async fn test_commit_consumer_monitor() {
238        let monitor = CommitConsumerMonitor::new(0, Some(10));
239        assert_eq!(monitor.highest_handled_commit(), 0);
240        monitor.set_highest_handled_commit(100);
241        assert_eq!(monitor.highest_handled_commit(), 100);
242        monitor
243            .replay_to_consumer_last_processed_commit_complete()
244            .await;
245    }

247    #[tokio::test]
248    async fn full_replay_waits_for_storage_and_for_the_consumer() {
249        let (args, _receiver) = CommitConsumerArgs::new_with_full_replay();
250        let monitor = args.monitor();
251        assert!(
252            timeout(
253                Duration::from_millis(20),
254                monitor.replay_to_consumer_last_processed_commit_complete()
255            )
256            .await
257            .is_err()
258        );
259        monitor.set_replay_target(5);
260        monitor.set_highest_handled_commit(4);
261        assert!(
262            timeout(
263                Duration::from_millis(20),
264                monitor.replay_to_consumer_last_processed_commit_complete()
265            )
266            .await
267            .is_err()
268        );
269        monitor.set_highest_handled_commit(5);
270        timeout(
271            Duration::from_secs(1),
272            monitor.replay_to_consumer_last_processed_commit_complete(),
273        )
274        .await
275        .unwrap();
276    }

279    #[case(0)]

281    #[tokio::test]
282    async fn acknowledged_full_replay_waits_for_target_discovery(#[case] target: CommitIndex) {
283        let (args, _receiver) = CommitConsumerArgs::new_with_full_replay();
284        let monitor = args.monitor();
285        assert_eq!(monitor.replay_target(), None);
286        let ready = monitor.replay_to_consumer_last_processed_commit_complete();
287        tokio::pin!(ready);
288        assert!(futures::poll!(&mut ready).is_pending());
289
290        monitor.set_highest_handled_commit(target);
291        assert!(futures::poll!(&mut ready).is_pending());
292        monitor.set_replay_target(target);
293        timeout(Duration::from_secs(1), ready).await.unwrap();
294    }

302    #[should_panic(expected = "startup replay target must be set only once")]
303    async fn replay_target_cannot_be_set_twice(
304        #[case] full_replay: bool,
305        #[case] target: CommitIndex,
306    ) {
307        let (args, _receiver) = if full_replay {
308            CommitConsumerArgs::new_with_full_replay()
309        } else {
310            CommitConsumerArgs::new(0, target)
311        };
312        let monitor = args.monitor();
313        if full_replay {
314            monitor.set_replay_target(target);
315        }
316        assert_eq!(monitor.replay_target(), Some(target));
317        assert_eq!(monitor.wait_for_replay_target().await, target);
318        monitor.set_replay_target(target);
319    }

321    #[tokio::test]
322    async fn committed_handled_and_target_notifications_are_independent() {
323        let (args, _receiver) = CommitConsumerArgs::new_with_full_replay();
324        let monitor = args.monitor();
325        let mut committed = monitor.subscribe_progress();
326        let mut handled = monitor.subscribe_handled_commit();
327        let target = monitor.wait_for_replay_target();
328        tokio::pin!(target);
329        assert!(futures::poll!(&mut target).is_pending());
330
331        monitor.report_committed(20, 50);
332        timeout(Duration::from_secs(1), committed.changed())
333            .await
334            .unwrap()
335            .unwrap();
336        assert_eq!(
337            *committed.borrow_and_update(),
338            CommitProgress {
339                highest_committed_index: 20,
340                highest_committed_round: 50,
341            }
342        );
343        assert!(!handled.has_changed().unwrap());
344        assert!(futures::poll!(&mut target).is_pending());
345
346        monitor.set_highest_handled_commit(2);
347        timeout(Duration::from_secs(1), handled.changed())
348            .await
349            .unwrap()
350            .unwrap();
351        assert_eq!(*handled.borrow_and_update(), 2);
352        assert!(!committed.has_changed().unwrap());
353        assert!(futures::poll!(&mut target).is_pending());
354
355        monitor.set_replay_target(5);
356        assert_eq!(timeout(Duration::from_secs(1), target).await.unwrap(), 5);
357        assert!(!committed.has_changed().unwrap());
358        assert!(!handled.has_changed().unwrap());
359        assert_eq!(monitor.replay_target(), Some(5));
360        assert_eq!(monitor.highest_handled_commit(), 2);
361
362        monitor.report_committed(19, 49);
363        monitor.report_committed(20, 50);
364        assert!(!committed.has_changed().unwrap());
365        assert_eq!(monitor.progress(), *committed.borrow());
366    }

370        let monitor = CommitConsumerMonitor::new(0, Some(0));
371        monitor.report_committed(20, 50);
372        monitor.set_highest_handled_commit(2);
