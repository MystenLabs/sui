30/// FlexCommitter supports committing multiple and varying number of leaders per round,
31/// based on DagState and the schedule for next commit leaders.
32pub(crate) struct FlexCommitter {
33    context: Arc<Context>,
34    dag_state: Arc<RwLock<DagState>>,
35    slot_decider: LeaderSlotDecider,
36    pending_commit_state: PendingCommitState,
37}
38
39impl FlexCommitter {
40    pub(crate) fn new(context: Arc<Context>, dag_state: Arc<RwLock<DagState>>) -> Self {
41        let slot_decider = LeaderSlotDecider::new(context.clone(), dag_state.clone());
42        Self {
43            context,
44            dag_state,
45            slot_decider,
46            pending_commit_state: PendingCommitState::default(),
47        }
48    }

50    /// Attempts to create a commit based on the current DagState.
51    pub(crate) fn try_commit(
52        &mut self,
53        next_commit_leaders: NextCommitLeaderSchedule,
54    ) -> Option<(TrustedCommit, CommittedSubDag)> {
55        self.maybe_refresh_pending_commit_state(next_commit_leaders);
56
57        self.try_direct_commit();
58        if let Some(commit_leader_round) = self.find_commit_leader_round() {
59            return self.build_commit(commit_leader_round);
60        }
61
62        self.try_indirect_commit();
63        if let Some(commit_leader_round) = self.find_commit_leader_round() {
64            return self.build_commit(commit_leader_round);
65        }
66
67        None
68    }
69
70    fn maybe_refresh_pending_commit_state(
71        &mut self,
72        next_commit_leaders: NextCommitLeaderSchedule,
73    ) {
74        let current = &self.pending_commit_state.next_commit_leaders;
75        if current.next_commit_index == next_commit_leaders.next_commit_index {
76            assert_eq!(
77                current.min_next_leader_round, next_commit_leaders.min_next_leader_round,
78                "minimum leader round must not change at the same commit index"
79            );
80            assert_eq!(
81                current.allowed_leaders, next_commit_leaders.allowed_leaders,
82                "leader schedule must not change at the same commit index"
83            );
84            return;
85        }
86        let current_commit_index = current.next_commit_index;
87        let current_min_next_leader_round = current.min_next_leader_round;
88        let leader_schedule_changed =
89            current.allowed_leaders != next_commit_leaders.allowed_leaders;
90        assert!(
91            current_commit_index < next_commit_leaders.next_commit_index,
92            "next_commit_index should only move forward: {} vs {}",
93            current_commit_index,
94            next_commit_leaders.next_commit_index
95        );
96        assert!(
97            current_min_next_leader_round < next_commit_leaders.min_next_leader_round,
98            "min_next_leader_round should only move forward: {} vs {}",
99            current_min_next_leader_round,
100            next_commit_leaders.min_next_leader_round
101        );
102
103        if leader_schedule_changed {
104            tracing::debug!(
105                "Resetting pending commit state for a leader schedule change: old_commit_index={}, new_commit_index={}, min_next_leader_round={}, num_leaders={}",
106                current_commit_index,
107                next_commit_leaders.next_commit_index,
108                next_commit_leaders.min_next_leader_round,
109                next_commit_leaders.num_leaders(),
110            );
111            self.pending_commit_state = PendingCommitState {
112                next_commit_leaders,
113                rounds: VecDeque::new(),
114            };
115            return;
116        }
117
118        let min_next_leader_round = next_commit_leaders.min_next_leader_round;
119        while self
120            .pending_commit_state
121            .rounds
122            .front()
123            .is_some_and(|state| state.round < min_next_leader_round)
124        {
125            self.pending_commit_state.rounds.pop_front();
126        }
127        if let Some(first) = self.pending_commit_state.rounds.front() {
128            assert_eq!(first.round, min_next_leader_round);
129        }
130        tracing::debug!(
131            "Advancing pending commit state: old_commit_index={}, new_commit_index={}, min_next_leader_round={}, retained_rounds={}",
132            current_commit_index,
133            next_commit_leaders.next_commit_index,
134            min_next_leader_round,
135            self.pending_commit_state.rounds.len(),
136        );
137        self.pending_commit_state.next_commit_leaders = next_commit_leaders;
138    }

140    /// Runs the direct commit rule on every leader slot pending to be decided.
141    fn try_direct_commit(&mut self) {
142        let min_next_leader_round = self
143            .pending_commit_state
144            .next_commit_leaders
145            .min_next_leader_round;
146        let highest_accepted_round = self.dag_state.read().highest_accepted_round();
147
148        for leader_round in min_next_leader_round..highest_accepted_round {
149            let round_state = self
150                .pending_commit_state
151                .get_or_create_round_state(leader_round);
152            let slots_to_decide = round_state
153                .leader_slots
154                .iter()
155                .filter_map(|s| {
156                    if matches!(s.leader_status, LeaderStatus::Undecided(_)) {
157                        Some(s.slot)
158                    } else {
159                        None
160                    }
161                })
162                .collect::<Vec<_>>();
163            for slot in slots_to_decide {
164                let status = self.slot_decider.try_direct_decide(slot);
165                if status.is_decided() {
166                    tracing::debug!("Direct decision for slot {slot}: {status}");
167                }
168                round_state.update_slot_decision(slot, status, Decision::Direct);
169            }
170        }
171    }
172
173    fn try_indirect_commit(&mut self) {
174        let min_next_leader_round = self
175            .pending_commit_state
176            .next_commit_leaders
177            .min_next_leader_round;
178        let highest_accepted_round = self.dag_state.read().highest_accepted_round();
179        // Decide leaders from highest to lowest round, because lower rounds need anchor blocks in
180        // higher rounds to be decided first.
181        for round in (min_next_leader_round
182            ..=(highest_accepted_round.saturating_sub(INDIRECT_COMMIT_DEPTH)))
183            .rev()
184        {
185            let Some(anchor_block) = self.find_anchor_block(round + INDIRECT_COMMIT_DEPTH) else {
186                // Skip when there is no anchor block for this round yet.
187                continue;
188            };
189            self.decide_with_anchor_block(anchor_block.clone(), round);
190        }
191    }
192
193    fn find_anchor_block(&self, start_round: Round) -> Option<VerifiedBlock> {
194        let start_index = start_round
195            - self
196                .pending_commit_state
197                .next_commit_leaders
198                .min_next_leader_round;
199        for index in start_index as usize..self.pending_commit_state.rounds.len() {
200            let round_state = &self.pending_commit_state.rounds[index];
201            for slot in &round_state.leader_slots {
202                match &slot.leader_status {
203                    // First committed block becomes the anchor block, when there is no undecided slot before it.
204                    // Every validator will select the same anchor block.
205                    // This rule of choosing anchor block minimizes the chance of encountering undecided slots before it.
206                    LeaderStatus::Commit(block) => return Some(block.clone()),
207                    // There cannot be an anchor block after an undecided.
208                    LeaderStatus::Undecided(_) => return None,
209                    // Continue searching for an anchor block after skipped slots.
210                    LeaderStatus::Skip(_) => {}
211                }
212            }
213        }
214        None
215    }

217    // Finds leader slots on `decision_round` and calls indirect commit rule with `anchor_block`
218    // to decide on their statuses.
219    fn decide_with_anchor_block(&mut self, anchor_block: VerifiedBlock, decision_round: Round) {
220        let round_state = self
221            .pending_commit_state
222            .get_or_create_round_state(decision_round);
223        if round_state.undecided_slots.is_empty() {
224            // Skip indirect commit when the round has already been fully decided.
225            return;
226        }
227
228        let slots: Vec<Slot> = round_state.leader_slots.iter().map(|s| s.slot).collect();
229        let statuses = self.slot_decider.try_indirect_decide(&anchor_block, &slots);
230        let anchor_ref = anchor_block.reference();
231        for (slot, status) in slots.into_iter().zip_eq(statuses) {
232            // Only log slots the indirect rule actually decides. Slots already decided
233            // by the direct rule are re-evaluated (and asserted consistent) but their
234            // decision is kept, so they should not appear as indirect decisions.
235            if round_state.undecided_slots.contains(&slot) {
236                tracing::debug!(
237                    "Indirect decision for slot {slot} using anchor {anchor_ref}: {status}"
238                );
239            }
240            round_state.update_slot_decision(slot, status, Decision::Indirect);
241        }
242    }

244    // The next commit leader round must have all slots decided in this and earlier rounds,
245    // and at least one slot in this round committed.
246    fn find_commit_leader_round(&self) -> Option<Round> {
247        for round_state in &self.pending_commit_state.rounds {
248            if !round_state.undecided_slots.is_empty() {
249                // Found undecided slot, so this and later rounds cannot be committed yet.
250                return None;
251            }
252            if round_state.num_committed > 0 {
253                // Found a fully decided round with at least one committed leader, so it can be committed.
254                return Some(round_state.round);
255            }
256            // This round is fully decided but has no committed leader, so it is skipped.
257        }
258        None
259    }

261    /// Reports the leader decisions of every round through `commit_leader_round`.
262    ///
263    /// Each decision is reported one time: the next `maybe_refresh_pending_commit_state`
264    /// sets `min_next_leader_round` to `commit_leader_round + 1`, so exactly these rounds
265    /// are dropped from the pending state before the next commit is built.
266    ///
267    /// This does not count every decision the committer makes. A decision is counted only
268    /// when a commit covers its round, so these stay unreported: rounds that are decided
269    /// when the node stops committing, and rounds dropped because a certified commit moved
270    /// `min_next_leader_round` past them. This is accepted, because commits cover almost
271    /// all leader decisions, and tracking the remainder needs bookkeeping that this metric
272    /// does not justify.
273    fn report_decided_prefix_metrics(&self, commit_leader_round: Round) {
274        for round_state in &self.pending_commit_state.rounds {
275            if round_state.round > commit_leader_round {
276                break;
277            }
278            for leader_slot in &round_state.leader_slots {
279                let (authority, outcome) = match &leader_slot.leader_status {
280                    LeaderStatus::Commit(block) => (block.author(), "commit"),
281                    LeaderStatus::Skip(slot) => (slot.authority, "skip"),
282                    LeaderStatus::Undecided(slot) => {
283                        panic!("Unexpected undecided slot {}", slot)
284                    }
285                };
286                let decision = match leader_slot
287                    .decision
288                    .expect("Decided slot must have a decision")
289                {
290                    Decision::Direct => "direct",
291                    Decision::Indirect => "indirect",
292                    Decision::Certified => "certified",
293                };
294                let commit_type = format!("{decision}-{outcome}");
295                let leader_host = &self.context.committee.authority(authority).hostname;
296                self.context
297                    .metrics
298                    .node_metrics
299                    .committed_leaders_total
300                    .with_label_values(&[leader_host, &commit_type])
301                    .inc();
302            }
303        }
304    }

306    /// Builds a single commit with leaders from `commit_leader_round`.
307    ///
308    /// Traversal starts from all committed leaders in `commit_leader_round`,
309    /// read from `pending_commit_state`. Other committed blocks are selected
310    /// via DFS from the leaders. Only uncommitted blocks above GC round are selected.
311    ///
312    /// Committed blocks are ordered deterministically: by round ascending, then
313    /// by `hash(seed || block_digest)` within a round. The last block of the
314    /// sorted leader round becomes the commit's named `leader`.
315    fn build_commit(
316        &mut self,
317        commit_leader_round: Round,
318    ) -> Option<(TrustedCommit, CommittedSubDag)> {
319        let committed_leaders = {
320            let round_state = self
321                .pending_commit_state
322                .get_round_state(commit_leader_round);
323            let mut committed_leaders = Vec::new();
324            for s in &round_state.leader_slots {
325                match &s.leader_status {
326                    LeaderStatus::Commit(block) => {
327                        committed_leaders.push(block.clone());
328                    }
329                    LeaderStatus::Skip(_) => {}
330                    LeaderStatus::Undecided(slot) => panic!("Unexpected undecided slot {}", slot),
331                }
332            }
333            committed_leaders
334        };
335        assert!(!committed_leaders.is_empty(), "No committed leaders found");
336        self.report_decided_prefix_metrics(commit_leader_round);
337
338        let mut dag_state = self.dag_state.write();
339        let last_commit_digest = dag_state.last_commit_digest();
340        let gc_round = dag_state.gc_round();
341
342        // DFS from all committed leaders. For each block, select its uncommitted
343        // ancestors above gc_round, mark them committed, and continue until the
344        // frontier is empty.
345        for leader in &committed_leaders {
346            assert!(
347                leader.round() > gc_round,
348                "Leader block {:?} at or below gc_round {gc_round} should not be committed",
349                leader.reference()
350            );
351            assert!(
352                dag_state.set_committed(&leader.reference()),
353                "Leader block {:?} attempted to be committed twice",
354                leader.reference()
355            );
356        }
357        let mut to_visit: Vec<VerifiedBlock> = committed_leaders.clone();
358        let mut to_commit: Vec<VerifiedBlock> = Vec::new();
359        while let Some(block) = to_visit.pop() {
360            to_commit.push(block.clone());
361            for ancestor_ref in block.ancestors().iter().copied() {
362                if ancestor_ref.round <= gc_round || !dag_state.set_committed(&ancestor_ref) {
363                    continue;
364                }
365                to_visit.push(dag_state.get_block(&ancestor_ref).unwrap());
366            }
367        }
368        drop(dag_state);
369
370        // Sort the committed blocks deterministically, first by round ascending,
371        // then by deterministic hash specific to this commit and each block.
372        let seed = compute_sort_seed(&committed_leaders);
373        sort_committed_blocks(&mut to_commit, &seed);
374
375        // Named leader = the committed leader that ended up last among the
376        // committed leaders in the sorted sub-dag.
377        let leader_ref = to_commit
378            .last()
379            .map(|b| b.reference())
380            .expect("At least one committed leader must be in to_commit");
381        assert_eq!(leader_ref.round, commit_leader_round);
382
383        // Compute deterministic commit timestamp from leader parents.
384        let timestamp_ms =
385            calculate_commit_timestamp(&self.context, &self.dag_state.read(), &committed_leaders);
386
387        let commit = Commit::new(
388            self.pending_commit_state
389                .next_commit_leaders
390                .next_commit_index,
391            last_commit_digest,
392            timestamp_ms,
393            leader_ref,
394            to_commit.iter().map(|b| b.reference()).collect(),
395        );
396        let serialized = commit
397            .serialize()
398            .unwrap_or_else(|e| panic!("Failed to serialize commit: {e}"));
399        let trusted_commit = TrustedCommit::new_trusted(commit, serialized);
400        let sub_dag = CommittedSubDag::new(
401            leader_ref,
402            to_commit,
403            timestamp_ms,
404            trusted_commit.reference(),
405        );
406
407        tracing::debug!(
408            "Built commit {} for leader round {} with {} committed leaders and {} blocks. Scheduled min next leader round {}, num allowed leaders {}",
409            trusted_commit.reference(),
410            commit_leader_round,
411            committed_leaders.len(),
412            sub_dag.blocks.len(),
413            self.pending_commit_state
414                .next_commit_leaders
415                .min_next_leader_round,
416            self.pending_commit_state
417                .next_commit_leaders
418                .allowed_leaders
419                .len(),
420        );
421
422        Some((trusted_commit, sub_dag))
423    }

425    /// Builds a CommittedSubDag for a certified commit synced from a peer, and
426    /// marks every committed block in DagState. The commit is already
427    /// quorum-certified, so the commit rule is skipped here — unlike local
428    /// commits, which are produced by `build_commit`.
429    pub(crate) fn handle_certified_commit(&self, commit: &TrustedCommit) -> CommittedSubDag {
430        tracing::debug!(
431            "Handling certified commit {} with leader {} and {} blocks",
432            commit.reference(),
433            commit.leader(),
434            commit.blocks().len()
435        );
436        let mut dag_state = self.dag_state.write();
437        for block_ref in commit.blocks() {
438            assert!(
439                dag_state.set_committed(block_ref),
440                "Block {:?} attempted to be committed twice",
441                block_ref
442            );
443        }
444        let blocks: Vec<VerifiedBlock> = dag_state
445            .get_blocks(commit.blocks())
446            .into_iter()
447            .map(|b| b.expect("All blocks referenced by a trusted commit must be in DagState"))
448            .collect();
449        drop(dag_state);
450
451        // Every block in the commit's leader round is a committed leader (same-round
452        // blocks cannot be ancestors of each other, so they can only appear in the
453        // commit as leaders). Report them like `build_commit` does for local
454        // decisions, so `committed_leaders_total` covers the commit sync path too.
455        for block in &blocks {
456            if block.round() == commit.leader().round {
457                let leader_host = &self.context.committee.authority(block.author()).hostname;
458                self.context
459                    .metrics
460                    .node_metrics
461                    .committed_leaders_total
462                    .with_label_values(&[leader_host, "certified-commit"])
463                    .inc();
464            }
465        }
466
467        let mut subdag = CommittedSubDag::new(
468            commit.leader(),
469            blocks,
470            commit.timestamp_ms(),
471            commit.reference(),
472        );
473        subdag.decided_with_local_blocks = false;
474        subdag
475    }

478/// Tracks intermediate state for future commits.
479///
480/// An advanced commit removes the old round prefix. The remaining decisions stay valid while the
481/// ordered leader schedule is unchanged. A schedule change resets all round state.
482#[derive(Clone, Debug, Default)]
483struct PendingCommitState {
484    // Leader schedule for generating the upcoming commit.
485    next_commit_leaders: NextCommitLeaderSchedule,
486    // Intermediate state per round.
487    rounds: VecDeque<RoundState>,
488}
489
490impl PendingCommitState {
491    /// Returns the round' state and panics if it doesn't exist.
492    /// Use `get_or_create_round_state` to create it if needed.
493    fn get_round_state(&mut self, round: Round) -> &mut RoundState {
494        let index = round
495            .checked_sub(self.next_commit_leaders.min_next_leader_round)
496            .unwrap() as usize;
497        self.rounds.get_mut(index).unwrap()
498    }
499
500    /// Populate state up to round, if they don't exist yet. Then return the round's state.
501    fn get_or_create_round_state(&mut self, round: Round) -> &mut RoundState {
502        let add_from_round = self
503            .rounds
504            .back()
505            .map(|state| state.round + 1)
506            .unwrap_or(self.next_commit_leaders.min_next_leader_round);
507        // No-op if the round's state exists.
508        for r in add_from_round..=round {
509            let mut seed_bytes = [0u8; 32];
510            seed_bytes[28..].copy_from_slice(&r.to_le_bytes());
511            let mut rng = StdRng::from_seed(seed_bytes);
512            let mut leaders = self.next_commit_leaders.allowed_leaders.clone();
513            leaders.shuffle(&mut rng);
514            let leader_slots: Vec<LeaderSlot> = leaders
515                .into_iter()
516                .map(|leader| {
517                    let slot = Slot::new(r, leader);
518                    LeaderSlot {
519                        slot,
520                        leader_status: LeaderStatus::Undecided(slot),
521                        decision: None,
522                    }
523                })
524                .collect();
525            let undecided_slots: BTreeSet<Slot> = leader_slots.iter().map(|s| s.slot).collect();
526            self.rounds.push_back(RoundState {
527                round: r,
528                leader_slots,
529                undecided_slots,
530                num_committed: 0,
531            });
532        }
533
534        self.get_round_state(round)
535    }
536}

538/// Per-round intermediate state computed with commit rule and current leader schedule
539#[derive(Clone, Debug)]
540struct RoundState {
541    round: Round,
542    // Leader slots selected in the round, in order.
543    leader_slots: Vec<LeaderSlot>,
544    // Leader slots that haven't decided.
545    undecided_slots: BTreeSet<Slot>,
546    // Number of leader slots voted to be committed.
547    // They may not appear in an actual commit as leaders though.
548    num_committed: usize,
549}
550
551impl RoundState {
552    fn update_slot_decision(&mut self, slot: Slot, new_status: LeaderStatus, decision: Decision) {
553        let leader_slot = self
554            .leader_slots
555            .iter_mut()
556            .find(|s| s.slot == slot)
557            .unwrap_or_else(|| panic!("Slot {} must be in round {}", slot, self.round));
558        if matches!(new_status, LeaderStatus::Undecided(_)) {
559            if matches!(leader_slot.leader_status, LeaderStatus::Undecided(_)) {
560                // No-op if the status is still undecided.
561                return;
562            } else {
563                // Undeciding a previously decided slot should not happen.
564                panic!(
565                    "Cannot undecide slot {}: {:?} vs {:?}",
566                    slot, leader_slot.leader_status, new_status
567                );
568            }
569        }
570        if self.undecided_slots.remove(&slot) {
571            if matches!(new_status, LeaderStatus::Commit(_)) {
572                self.num_committed += 1;
573            }
574            leader_slot.leader_status = new_status;
575            // Record the rule that first decided this slot. A slot decided by the direct
576            // rule may be re-evaluated by the indirect rule in the same round, but the
577            // initial decision is kept (the else branch leaves `decision` untouched).
578            leader_slot.decision = Some(decision);
579        } else {
580            assert_eq!(
581                leader_slot.leader_status, new_status,
582                "Cannot update slot {} decision: {:?} vs {:?}",
583                slot, leader_slot.leader_status, new_status
584            );
585        }
586    }
587}

589#[derive(Clone, Debug)]
590struct LeaderSlot {
591    slot: Slot,
592    leader_status: LeaderStatus,
593    // Which commit rule decided the slot (direct or indirect), or None while undecided.
594    decision: Option<Decision>,
595}

597/// From a deterministic array of leaders, compute a deterministic digest used as seed for sort.
598fn compute_sort_seed(committed_leaders: &[VerifiedBlock]) -> [u8; DIGEST_LENGTH] {
599    let mut hasher = DefaultHashFunction::new();
600    for leader in committed_leaders {
601        hasher.update(leader.digest().as_ref());
602    }
603    hasher.finalize().into()
604}
605
606/// Per-block sort key:
607/// - Primary part: block round.
608/// - Secondary part: hash of seed and block digest.
609fn block_sort_key(
610    seed: &[u8; DIGEST_LENGTH],
611    block_ref: &BlockRef,
612) -> (Round, [u8; DIGEST_LENGTH]) {
613    let mut hasher = DefaultHashFunction::new();
614    hasher.update(seed);
615    hasher.update(block_ref.digest);
616    (block_ref.round, hasher.finalize().into())
617}
618
619fn sort_committed_blocks(blocks: &mut [VerifiedBlock], seed: &[u8; DIGEST_LENGTH]) {
620    blocks.sort_by_cached_key(|b| block_sort_key(seed, &b.reference()));
621}

623/// Takes the union of each committed leader's parent-round ancestors,
624/// and returns the median timestamp of parent blocks weighted by stake.
625/// Monotonicity is enforced against `last_commit_timestamp_ms`.
626fn calculate_commit_timestamp(
627    context: &Context,
628    dag_state: &DagState,
629    committed_leaders: &[VerifiedBlock],
630) -> BlockTimestampMs {
631    let leader_round = committed_leaders[0].round();
632    debug_assert!(committed_leaders.iter().all(|b| b.round() == leader_round));
633
634    // When there are enough leader stake, use leaders to compute the commit timestamp.
635    let total_leader_stake = committed_leaders
636        .iter()
637        .map(|b| context.committee.stake(b.author()))
638        .sum::<Stake>();
639    if total_leader_stake >= context.committee.certification_threshold() {
640        let ts = crate::linearizer::median_timestamp_by_stake(context, committed_leaders)
641            .unwrap_or_else(|e| panic!("Cannot compute commit timestamp: {e}"));
642        return ts.max(dag_state.last_commit_timestamp_ms());
643    }
644
645    let mut parent_refs: BTreeMap<AuthorityIndex, BlockRef> = BTreeMap::new();
646    for leader in committed_leaders {
647        for ancestor in leader.ancestors() {
648            if ancestor.round + 1 == leader_round {
649                parent_refs.insert(ancestor.author, *ancestor);
650            }
651        }
652    }
653    let parent_refs: Vec<BlockRef> = parent_refs.values().cloned().collect();
654    let block_opts = dag_state.get_blocks(&parent_refs);
655    let blocks = block_opts
656        .iter()
657        .map(|b| b.as_ref().expect("Parent block must be in dag state"));
658    let ts = crate::linearizer::median_timestamp_by_stake(context, blocks)
659        .unwrap_or_else(|e| panic!("Cannot compute commit timestamp: {e}"));
660    ts.max(dag_state.last_commit_timestamp_ms())
661}
