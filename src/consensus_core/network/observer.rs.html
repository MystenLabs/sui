70#[derive(Clone, prost::Message)]
71pub(crate) struct FetchBlocksRequest {
72    #[prost(bytes = "vec", repeated, tag = "1")]
73    block_refs: Vec<Vec<u8>>,
74    // The round per authority after which blocks should be fetched. The vector represents the round
75    // for each authority and its length should be the same as the committee size.
76    // When this field is non-empty, additional ancestors of the requested blocks can be fetched.
77    #[prost(uint32, repeated, tag = "2")]
78    fetch_after_rounds: Vec<Round>,
79    // When true, missing ancestors of the requested blocks will be fetched as well.
80    // When false, additional blocks are fetched depth-first from the requested block authorities.
81    // This field is only meaningful when fetch_after_rounds is non-empty.
82    #[prost(bool, tag = "3")]
83    fetch_missing_ancestors: bool,
84}

121struct ChannelPool {
122    context: Arc<Context>,
123    // Size is limited by known authorities in the committee.
124    channels: RwLock<BTreeMap<PeerId, Channel>>,
125    // The observer peers pool that this node is allowed to connect to.
126    observer_peers: BTreeMap<NetworkPublicKey, Multiaddr>,
127}
128
129impl ChannelPool {
130    fn new(context: Arc<Context>) -> Self {
131        // Only allow to connect to peers that are within this pool.
132        let mut observer_peers = BTreeMap::new();
133        for peer in &context.parameters.observer.peers {
134            observer_peers.insert(peer.public_key.clone(), peer.address.clone());
135        }
136        Self {
137            context,
138            channels: RwLock::new(BTreeMap::new()),
139            observer_peers,
140        }
141    }
142
143    pub(crate) async fn get_channel(
144        &self,
145        network_keypair: NetworkKeyPair,
146        peer: PeerId,
147        timeout: Duration,
148    ) -> ConsensusResult<Channel> {
149        {
150            let channels = self.channels.read();
151            if let Some(channel) = channels.get(&peer) {
152                return Ok(channel.clone());
153            }
154        }
155
156        // Find the network key of the peer.
157        let peer_network_key = match peer.clone() {
158            PeerId::Validator(authority) => self
159                .context
160                .committee
161                .authority(authority)
162                .network_key
163                .clone(),
164            PeerId::Observer(node_id) => (*node_id).clone(),
165        };
166
167        // Check if the peer is in the observer peers pool. If not return an error.
168        let peer_address = self.observer_peers.get(&peer_network_key).ok_or_else(|| {
169            ConsensusError::NetworkConfig(format!("Peer not in observer peers pool: {:?}", peer))
170        })?;
171
172        let address = to_host_port_str(peer_address).map_err(|e| {
173            ConsensusError::NetworkConfig(format!("Cannot convert address to host:port: {e:?}"))
174        })?;
175        let address = format!("https://{address}");
176        let config = &self.context.parameters.tonic;
177        let buffer_size = config.connection_buffer_size;
178        let client_tls_config = sui_tls::create_rustls_client_config(
179            peer_network_key.into_inner().clone(),
180            certificate_server_name(&self.context),
181            Some(network_keypair.private_key().into_inner()),
182        );
183        let endpoint = tonic_rustls::Channel::from_shared(address.clone())
184            .map_err(|e| ConsensusError::NetworkConfig(format!("invalid URI '{address}': {e}")))?
185            .connect_timeout(timeout)
186            .initial_connection_window_size(Some(buffer_size as u32))
187            .initial_stream_window_size(Some(buffer_size as u32 / 2))
188            .keep_alive_while_idle(true)
189            .keep_alive_timeout(config.keepalive_interval)
190            .http2_keep_alive_interval(config.keepalive_interval)
191            // tcp keepalive is probably unnecessary and is unsupported by msim.
192            .user_agent("mysticeti")
193            .unwrap()
194            .tls_config(client_tls_config)
195            .unwrap();
196
197        let deadline = tokio::time::Instant::now() + timeout;
198        let channel = loop {
199            trace!("Connecting to endpoint at {address}");
200            match endpoint.connect().await {
201                Ok(channel) => break channel,
202                Err(e) => {
203                    debug!("Failed to connect to endpoint at {address}: {e:?}");
204                    if tokio::time::Instant::now() >= deadline {
205                        return Err(ConsensusError::NetworkClientConnection(format!(
206                            "Timed out connecting to endpoint at {address}: {e:?}"
207                        )));
208                    }
209                    tokio::time::sleep(Duration::from_secs(1)).await;
210                }
211            }
212        };
213        trace!("Connected to {address}");
214
215        let channel = tower::ServiceBuilder::new()
216            .layer(CallbackLayer::new(MetricsCallbackMaker::new(
217                self.context.metrics.network_metrics.outbound.clone(),
218                self.context.parameters.tonic.excessive_message_size,
219            )))
220            .map_request(rebox_request as ReboxRequestFn)
221            .layer(
222                TraceLayer::new_for_grpc()
223                    .make_span_with(DefaultMakeSpan::new().level(tracing::Level::TRACE))
224                    .on_failure(DefaultOnFailure::new().level(tracing::Level::DEBUG)),
225            )
226            .service(channel);
227
228        let mut channels = self.channels.write();
229        // There should not be many concurrent attempts at connecting to the same peer.
230        let channel = channels.entry(peer).or_insert(channel);
231        Ok(channel.clone())
232    }
233}

235impl TonicObserverClient {
236    pub(crate) fn new(context: Arc<Context>, network_keypair: NetworkKeyPair) -> Self {
237        Self {
238            context: context.clone(),
239            network_keypair,
240            channel_pool: Arc::new(ChannelPool::new(context)),
241        }
242    }
243
244    async fn get_client(
245        &self,
246        peer: PeerId,
247        timeout: Duration,
248    ) -> ConsensusResult<super::tonic_gen::observer_service_client::ObserverServiceClient<Channel>>
249    {
250        use tonic::codec::CompressionEncoding;
251
252        let message_size_limit = self.context.parameters.tonic.message_size_limit;
253        let channel_pool = self.channel_pool.clone();
254        let network_keypair = self.network_keypair.clone();
255        let channel = channel_pool
256            .get_channel(network_keypair, peer, timeout)
257            .await?;
258        let client = super::tonic_gen::observer_service_client::ObserverServiceClient::new(channel)
259            .max_encoding_message_size(message_size_limit)
260            .max_decoding_message_size(message_size_limit)
261            .send_compressed(CompressionEncoding::Zstd)
262            .accept_compressed(CompressionEncoding::Zstd);
263        Ok(client)
264    }
265}

267#[async_trait]
268impl ObserverNetworkClient for TonicObserverClient {
269    async fn stream_blocks(
270        &self,
271        peer: PeerId,
272        highest_round_per_authority: Vec<Round>,
273        filter: BlockStreamFilter,
274        timeout: Duration,
275    ) -> ConsensusResult<ObserverBlockStream> {
276        let mut client = self.get_client(peer.clone(), timeout).await?;
277
278        let authors = match filter {
279            BlockStreamFilter::All => vec![],
280            BlockStreamFilter::Authors(authors) => authors
281                .into_iter()
282                .map(|author| author.value_u32())
283                .collect(),
284        };
285        let request = Request::new(BlockStreamRequest {
286            highest_round_per_authority,
287            authors,
288        });
289        let response = client
290            .stream_blocks(request)
291            .await
292            .map_err(|e| ConsensusError::NetworkRequest(format!("stream_blocks failed: {e:?}")))?;
293        let stream = response
294            .into_inner()
295            .take_while(|b| futures::future::ready(b.is_ok()))
296            .filter_map(move |b| {
297                let peer_cloned = peer.clone();
298                async move {
299                    match b {
300                        Ok(response) => Some(super::ObserverStreamItem {
301                            blocks: response.blocks,
302                            auxiliary_data: response.auxiliary_data.unwrap_or_default(),
303                        }),
304                        Err(e) => {
305                            debug!("Network error received from {:?}: {e:?}", peer_cloned);
306                            None
307                        }
308                    }
309                }
310            });
311        Ok(Box::pin(stream))
312    }
313
314    async fn fetch_blocks(
315        &self,
316        peer: PeerId,
317        block_refs: Vec<BlockRef>,
318        fetch_after_rounds: Vec<Round>,
319        fetch_missing_ancestors: bool,
320        timeout: Duration,
321    ) -> ConsensusResult<Vec<Bytes>> {
322        let mut client = self.get_client(peer, timeout).await?;
323        let max_allowed_bytes =
324            max_fetch_blocks_response_bytes(&self.context, &block_refs, &fetch_after_rounds);
325        let mut request = Request::new(FetchBlocksRequest {
326            block_refs: block_refs
327                .iter()
328                .filter_map(|r| match bcs::to_bytes(r) {
329                    Ok(serialized) => Some(serialized),
330                    Err(e) => {
331                        debug!("Failed to serialize block ref {:?}: {e:?}", r);
332                        None
333                    }
334                })
335                .collect(),
336            fetch_after_rounds,
337            fetch_missing_ancestors,
338        });
339        request.set_timeout(timeout);
340
341        let mut stream = client
342            .fetch_blocks(request)
343            .await
344            .map_err(|e| {
345                if e.code() == tonic::Code::DeadlineExceeded {
346                    ConsensusError::NetworkRequestTimeout(format!("fetch_blocks failed: {e:?}"))
347                } else {
348                    ConsensusError::NetworkRequest(format!("fetch_blocks failed: {e:?}"))
349                }
350            })?
351            .into_inner();
352
353        let mut blocks = vec![];
354        let mut total_fetched_bytes = 0;
355        loop {
356            match stream.message().await {
357                Ok(Some(response)) => {
358                    for b in &response.blocks {
359                        total_fetched_bytes += b.len();
360                    }
361                    blocks.extend(response.blocks);
362                    if total_fetched_bytes > max_allowed_bytes {
363                        info!(
364                            "fetch_blocks() fetched bytes exceeded limit: {} > {}, terminating stream.",
365                            total_fetched_bytes, max_allowed_bytes,
366                        );
367                        break;
368                    }
369                }
370                Ok(None) => {
371                    break;
372                }
373                Err(e) => {
374                    if blocks.is_empty() {
375                        if e.code() == tonic::Code::DeadlineExceeded {
376                            return Err(ConsensusError::NetworkRequestTimeout(format!(
377                                "fetch_blocks failed mid-stream: {e:?}"
378                            )));
379                        }
380                        return Err(ConsensusError::NetworkRequest(format!(
381                            "fetch_blocks failed mid-stream: {e:?}"
382                        )));
383                    } else {
384                        warn!("fetch_blocks failed mid-stream: {e:?}");
385                        break;
386                    }
387                }
388            }
389        }
390        Ok(blocks)
391    }
392
393    async fn fetch_commits(
394        &self,
395        peer: PeerId,
396        commit_range: CommitRange,
397        timeout: Duration,
398    ) -> ConsensusResult<(Vec<Bytes>, Vec<Bytes>)> {
399        let mut client = self.get_client(peer, timeout).await?;
400        let mut request = Request::new(FetchCommitsRequest {
401            start: commit_range.start(),
402            end: commit_range.end(),
403        });
404        request.set_timeout(timeout);
405        let response = client
406            .fetch_commits(request)
407            .await
408            .map_err(|e| ConsensusError::NetworkRequest(format!("fetch_commits failed: {e:?}")))?;
409        let response = response.into_inner();
410        Ok((response.commits, response.certifier_blocks))
411    }
412}

414/// Proxies Observer Tonic requests to ObserverNetworkService.
415/// Extracts peer NodeId from TLS certificates and delegates to the service layer.
416pub(crate) struct ObserverServiceProxy<S: ObserverNetworkService> {
417    context: Arc<Context>,
418    // ObserverServiceProxy is cloned into per-connection server tasks, which complete on the
419    // network's schedule during graceful shutdown, and can briefly outlive the node if it is
420    // dropped without stop(). Hold the service weakly so lingering connections cannot extend
421    // the life of the observer service and its state; requests racing shutdown fail with
422    // `unavailable` instead.
423    service: Weak<S>,
424}
425
426impl<S: ObserverNetworkService> ObserverServiceProxy<S> {
427    pub(crate) fn new(context: Arc<Context>, service: Arc<S>) -> Self {
428        Self {
429            context,
430            service: Arc::downgrade(&service),
431        }
432    }
433
434    /// Converts the wire-level author indices into a validated filter. Empty means no
435    /// filtering, which keeps clients predating the field wire-compatible.
436    fn block_stream_filter(&self, authors: Vec<u32>) -> Result<BlockStreamFilter, tonic::Status> {
437        if authors.is_empty() {
438            return Ok(BlockStreamFilter::All);
439        }
440        let committee = &self.context.committee;
441        let authors = authors
442            .into_iter()
443            .map(|index| {
444                committee
445                    .to_authority_index(index as usize)
446                    .ok_or_else(|| {
447                        tonic::Status::invalid_argument(format!(
448                            "Unknown authority index {index} in block stream filter, committee size is {}",
449                            committee.size()
450                        ))
451                    })
452            })
453            .collect::<Result<BTreeSet<_>, _>>()?;
454        Ok(BlockStreamFilter::Authors(authors))
455    }
456
457    fn service(&self) -> Result<Arc<S>, tonic::Status> {
458        self.service
459            .upgrade()
460            .ok_or_else(|| tonic::Status::unavailable("Consensus authority is shutting down"))
461    }
462}

464#[async_trait]
465impl<S: ObserverNetworkService> ObserverService for ObserverServiceProxy<S> {
466    type StreamBlocksStream =
467        Pin<Box<dyn Stream<Item = Result<BlockStreamResponse, tonic::Status>> + Send>>;
468
469    async fn stream_blocks(
470        &self,
471        request: Request<BlockStreamRequest>,
472    ) -> Result<Response<Self::StreamBlocksStream>, tonic::Status> {
473        let peer_id = request
474            .extensions()
475            .get::<ObserverPeerInfo>()
476            .map(|info| info.public_key.clone())
477            .ok_or_else(|| {
478                tonic::Status::unauthenticated(
479                    "Observer peer info not found in request. TLS authentication required.",
480                )
481            })?;
482
483        let request = request.into_inner();
484        let filter = self.block_stream_filter(request.authors)?;
485
486        let block_stream = self
487            .service()?
488            .handle_stream_blocks(peer_id, request.highest_round_per_authority, filter)
489            .await
490            .map_err(|e| tonic::Status::internal(format!("{e:?}")))?;
491
492        let response_stream = block_stream.map(|item| {
493            Ok(BlockStreamResponse {
494                blocks: item.blocks,
495                auxiliary_data: if item.auxiliary_data.randomness_signatures.is_empty() {
496                    None
497                } else {
498                    Some(item.auxiliary_data)
499                },
500            })
501        });
502
503        Ok(Response::new(Box::pin(response_stream)))
504    }
505
506    type FetchBlocksStream = Iter<std::vec::IntoIter<Result<FetchBlocksResponse, tonic::Status>>>;
507
508    async fn fetch_blocks(
509        &self,
510        request: Request<FetchBlocksRequest>,
511    ) -> Result<Response<Self::FetchBlocksStream>, tonic::Status> {
512        let peer_id = request
513            .extensions()
514            .get::<ObserverPeerInfo>()
515            .map(|info| info.public_key.clone())
516            .ok_or_else(|| {
517                tonic::Status::unauthenticated(
518                    "Observer peer info not found in request. TLS authentication required.",
519                )
520            })?;
521        let inner = request.into_inner();
522        let block_refs = inner
523            .block_refs
524            .into_iter()
525            .filter_map(|serialized| match bcs::from_bytes(&serialized) {
526                Ok(r) => Some(r),
527                Err(e) => {
528                    debug!("Failed to deserialize block ref {:?}: {e:?}", serialized);
529                    None
530                }
531            })
532            .collect();
533        let fetch_after_rounds = inner.fetch_after_rounds;
534        let fetch_missing_ancestors = inner.fetch_missing_ancestors;
535        let blocks = self
536            .service()?
537            .handle_fetch_blocks(
538                peer_id,
539                block_refs,
540                fetch_after_rounds,
541                fetch_missing_ancestors,
542            )
543            .await
544            .map_err(|e| tonic::Status::internal(format!("{e:?}")))?;
545        let responses: std::vec::IntoIter<Result<FetchBlocksResponse, tonic::Status>> =
546            chunk_blocks(blocks, MAX_FETCH_RESPONSE_BYTES)
547                .into_iter()
548                .map(|blocks| Ok(FetchBlocksResponse { blocks }))
549                .collect::<Vec<_>>()
550                .into_iter();
551        let stream = tokio_stream::iter(responses);
552        Ok(Response::new(stream))
553    }
554
555    async fn fetch_commits(
556        &self,
557        request: Request<FetchCommitsRequest>,
558    ) -> Result<Response<FetchCommitsResponse>, tonic::Status> {
559        let peer_id = request
560            .extensions()
561            .get::<ObserverPeerInfo>()
562            .map(|info| info.public_key.clone())
563            .ok_or_else(|| {
564                tonic::Status::unauthenticated(
565                    "Observer peer info not found in request. TLS authentication required.",
566                )
567            })?;
568        let request = request.into_inner();
569        let (commits, certifier_blocks) = self
570            .service()?
571            .handle_fetch_commits(peer_id, (request.start..=request.end).into())
572            .await
573            .map_err(|e| tonic::Status::internal(format!("{e:?}")))?;
574        let commits = commits
575            .into_iter()
576            .map(|c| c.serialized().clone())
577            .collect();
578        let certifier_blocks = certifier_blocks
579            .into_iter()
580            .map(|b| b.serialized().clone())
581            .collect();
582        Ok(Response::new(FetchCommitsResponse {
583            commits,
584            certifier_blocks,
585        }))
586    }
587}

687    /// The proxy validates the wire-level author indices before they reach the service.
688    #[tokio::test]
689    async fn observer_stream_blocks_filter_validation() {
690        use super::{BlockStreamRequest, ObserverPeerInfo, ObserverServiceProxy};
691        use crate::network::tonic_gen::observer_service_server::ObserverService as _;
692        use consensus_config::AuthorityIndex;
693
694        let (context, keys) = Context::new_for_test(4);
695        let context = Arc::new(context);
696        let service = Arc::new(Mutex::new(TestService::new()));
697        let proxy = ObserverServiceProxy::new(context.clone(), service.clone());
698        let peer_info = ObserverPeerInfo {
699            public_key: keys[0].0.public(),
700        };
701        let new_request = |authors: Vec<u32>| {
702            let mut request = tonic::Request::new(BlockStreamRequest {
703                highest_round_per_authority: vec![0u32; 4],
704                authors,
705            });
706            request.extensions_mut().insert(peer_info.clone());
707            request
708        };
709
710        // An authority index outside the committee is rejected without reaching the service.
711        let Err(status) = proxy.stream_blocks(new_request(vec![0, 10])).await else {
712            panic!("expected invalid_argument for unknown authority index");
713        };
714        assert_eq!(status.code(), tonic::Code::InvalidArgument);
715        assert!(service.lock().handle_stream_blocks.is_empty());
716
717        // Valid indices, including duplicates, are converted into a set of authorities.
718        proxy
719            .stream_blocks(new_request(vec![2, 1, 2]))
720            .await
721            .unwrap();
722        let (_, filter) = service.lock().handle_stream_blocks.pop().unwrap();
723        assert_eq!(
724            filter,
725            BlockStreamFilter::Authors(
726                [1, 2]
727                    .into_iter()
728                    .map(AuthorityIndex::new_for_test)
729                    .collect()
730            )
731        );
732
733        // An empty list means no filtering.
734        proxy.stream_blocks(new_request(vec![])).await.unwrap();
735        let (_, filter) = service.lock().handle_stream_blocks.pop().unwrap();
736        assert_eq!(filter, BlockStreamFilter::All);
737    }
