40/// Serves observer requests from observer or validator peers. It is the server-side
41/// counterpart to `ObserverNetworkClient`.

88    async fn handle_block(&self, peer: PeerId, block: Bytes) -> ConsensusResult<()> {
89        fail_point_async!("consensus-rpc-response");
90
91        // TODO: dedup block verifications, here and with fetched blocks.
92        let signed_block: SignedBlock =
93            bcs::from_bytes(&block).map_err(ConsensusError::MalformedBlock)?;
94
95        // Create owned strings for observer peer names to avoid borrowing issues
96        let observer_name;
97        let peer_name = match &peer {
98            PeerId::Validator(authority) => self
99                .context
100                .committee
101                .authority(*authority)
102                .hostname
103                .as_str(),
104            PeerId::Observer(node_id) => {
105                observer_name = format!("{:?}", node_id);
106                observer_name.as_str()
107            }
108        };
109
110        // Reject blocks failing parsing and validations.
111        // Of Observer nodes we don't care about the transaction votes.
112        let (verified_block, _reject_txn_votes) = self
113            .block_verifier
114            .verify_and_vote(signed_block, block)
115            .tap_err(|e| {
116                self.context
117                    .metrics
118                    .node_metrics
119                    .invalid_blocks
120                    .with_label_values(&[peer_name, "handle_send_block", e.name()])
121                    .inc();
122                tracing::info!("Invalid block from {}: {}", peer.clone(), e);
123            })?;
124
125        let block_author_hostname = &self
126            .context
127            .committee
128            .authority(verified_block.author())
129            .hostname;
130        let block_ref = verified_block.reference();
131        tracing::debug!("Received block {} via send block.", block_ref);
132
133        self.context
134            .metrics
135            .node_metrics
136            .verified_blocks
137            .with_label_values(&[block_author_hostname])
138            .inc();
139
140        let now = self.context.clock.timestamp_utc_ms();
141        let forward_time_drift =
142            Duration::from_millis(verified_block.timestamp_ms().saturating_sub(now));
143
144        self.context
145            .metrics
146            .node_metrics
147            .block_timestamp_drift_ms
148            .with_label_values(&[block_author_hostname.as_str(), "handle_send_block"])
149            .inc_by(forward_time_drift.as_millis() as u64);
150
151        // Observe the block for the commit votes. When local commit is lagging too much,
152        // commit sync loop will trigger fetching.
153        self.commit_vote_monitor.observe_block(&verified_block);
154
155        // Reject blocks when local commit index is lagging too far from quorum commit index,
156        // to avoid the memory overhead from suspended blocks.
157        //
158        // IMPORTANT: this must be done after observing votes from the block, otherwise
159        // observed quorum commit will no longer progress.
160        //
161        // Since the main issue with too many suspended blocks is memory usage not CPU,
162        // it is ok to reject after block verifications instead of before.
163        let last_commit_index = self.dag_state.read().last_commit_index();
164        let quorum_commit_index = self.commit_vote_monitor.quorum_commit_index();
165        if is_commit_lagging(
166            self.context.as_ref(),
167            last_commit_index,
168            quorum_commit_index,
169        ) {
170            self.context
171                .metrics
172                .node_metrics
173                .rejected_blocks
174                .with_label_values(&["commit_lagging"])
175                .inc();
176            tracing::debug!(
177                "Block {:?} is rejected because last commit index is lagging quorum commit index too much ({} < {})",
178                block_ref,
179                last_commit_index,
180                quorum_commit_index,
181            );
182            return Err(ConsensusError::BlockRejected {
183                block_ref,
184                reason: format!(
185                    "Last commit index is lagging quorum commit index too much ({} < {})",
186                    last_commit_index, quorum_commit_index,
187                ),
188            });
189        }
190
191        // Add the block to the transaction vote tracker. No "own" votes are recorded for observer nodes.
192        if self.context.protocol_config.transaction_voting_enabled() {
193            self.transaction_vote_tracker
194                .add_voted_blocks(vec![(verified_block.clone(), vec![])]);
195        }
196
197        // Send the block to Core to try accepting it into the DAG.
198        let missing_ancestors = self
199            .core_dispatcher
200            .add_blocks(vec![verified_block.clone()])
201            .await
202            .map_err(|_| ConsensusError::Shutdown)?;
203
204        // Schedule fetching missing ancestors from this peer in the background.
205        if !missing_ancestors.is_empty() {
206            self.context
207                .metrics
208                .node_metrics
209                .handler_received_block_missing_ancestors
210                .with_label_values(&[block_author_hostname])
211                .inc_by(missing_ancestors.len() as u64);
212
213            let synchronizer = self.synchronizer.clone();
214            mysten_metrics::spawn_monitored_task!(async move {
215                // This does not wait for the fetch request to complete.
216                // It only waits for synchronizer to queue the request to a peer.
217                // When this fails, it usually means the queue is full.
218                // The fetch will retry from other peers via live and periodic syncs.
219                if let Err(err) = synchronizer.fetch_blocks(missing_ancestors, peer).await {
220                    tracing::debug!("Failed to fetch missing ancestors via synchronizer: {err}");
221                }
222            });
223        }
224
225        Ok(())
226    }

228    async fn handle_stream_blocks(
229        &self,
230        peer: NodeId,
231        highest_round_per_authority: Vec<Round>,
232    ) -> ConsensusResult<ObserverBlockStream> {
233        if highest_round_per_authority.len() != self.context.committee.size() {
234            return Err(ConsensusError::InvalidSizeOfHighestAcceptedRounds(
235                highest_round_per_authority.len(),
236                self.context.committee.size(),
237            ));
238        }
239
240        // Subscribe before snapshotting past blocks below. This can duplicate
241        // a block in both the subscription stream and snapshot, which is fine.
242        // Otherwise, it is possible to miss a block if it is broadcasted after snapshotting
243        // but before subscribing.
244        let broadcast_rx = self.rx_accepted_block_broadcast.resubscribe();
245
246        // Collect all accepted blocks from DagState that the observer hasn't yet seen,
247        // sorted by round for consistent ordering.
248        let past_blocks = {
249            let dag_state = self.dag_state.read();
250            let mut past_blocks = Vec::new();
251
252            for (authority, _) in self.context.committee.authorities() {
253                // Saturate so an out-of-range round from the peer cannot wrap to 0 and
254                // replay the entire block cache.
255                let from_round = highest_round_per_authority[authority.value()].saturating_add(1);
256                past_blocks.extend(dag_state.get_cached_blocks(authority, from_round));
257            }
258
259            past_blocks.sort_unstable_by_key(|b| b.round());
260            past_blocks
261        };
262
263        let past_stream =
264            stream::iter(
265                past_blocks
266                    .into_iter()
267                    .map(move |block| ObserverStreamItem {
268                        blocks: vec![block.serialized().clone()],
269                        auxiliary_data: Default::default(),
270                    }),
271            );
272
273        const MAX_BLOCKS_PER_POLL: usize = 20;
274        let broadcast_stream = BroadcastStream::<VerifiedBlock>::new(
275            PeerId::Observer(Box::new(peer)),
276            broadcast_rx,
277            MAX_BLOCKS_PER_POLL,
278            self.subscription_counter.clone(),
279        );
280        // When quorum-gated release is disabled, serve blocks straight off the broadcast as
281        // soon as they are accepted, without the buffering task in between.
282        let live_block_stream = if self.context.parameters.observer.quorum_release {
283            Either::Left(quorum_gated_accepted_block_stream(
284                self.context.clone(),
285                broadcast_stream,
286                MAX_BLOCKS_PER_POLL,
287            ))
288        } else {
289            Either::Right(broadcast_stream)
290        }
291        .map(|blocks| ObserverStreamItem {
292            blocks: blocks
293                .into_iter()
294                .map(|block| block.serialized().clone())
295                .collect(),
296            auxiliary_data: Default::default(),
297        });
298
299        let block_stream = past_stream.chain(live_block_stream);
300
301        // Merge randomness signature broadcast into the block stream when a handler is available.
302        if let Some(handler) = &self.randomness_signature_handler {
303            const MAX_SIGNATURES_PER_POLL: usize = 20;
304            let sig_stream = BroadcastStream::new_untracked(
305                handler.subscribe_randomness_signatures(),
306                MAX_SIGNATURES_PER_POLL,
307            )
308            .map(|sigs| ObserverStreamItem {
309                blocks: vec![],
310                auxiliary_data: AuxiliaryData {
311                    randomness_signatures: sigs,
312                },
313            });
314            Ok(Box::pin(futures::stream::select(block_stream, sig_stream)))
315        } else {
316            Ok(Box::pin(block_stream))
317        }
318    }

347fn quorum_gated_accepted_block_stream(
348    context: Arc<Context>,
349    mut source: BroadcastStream<VerifiedBlock>,
350    max_blocks_per_item: usize,
351) -> ReceiverStream<Vec<VerifiedBlock>> {
352    let release_timeout = context.parameters.leader_timeout;
353    let (tx, rx) = mpsc::channel(max_blocks_per_item);
354
355    mysten_metrics::spawn_monitored_task!(async move {
356        let mut state = AcceptedBlockReleaseState::new(context, release_timeout);
357
358        loop {
359            let released = tokio::select! {
360                blocks = source.next() => {
361                    let Some(blocks) = blocks else {
362                        break;
363                    };
364                    state.accept_blocks(blocks)
365                }
366                _ = tx.closed() => {
367                    return;
368                }
369                _ = tokio::time::sleep_until(state.next_timeout()) => {
370                    state.release_timed_out_rounds()
371                }
372            };
373
374            if !send_released_blocks(&tx, released, max_blocks_per_item).await {
375                return;
376            }
377        }
378
379        let released = state.release_all();
380        let _ = send_released_blocks(&tx, released, max_blocks_per_item).await;
381    });
382
383    ReceiverStream::new(rx)
384}
385
386async fn send_released_blocks(
387    tx: &mpsc::Sender<Vec<VerifiedBlock>>,
388    released: Vec<VerifiedBlock>,
389    max_blocks_per_item: usize,
390) -> bool {
391    for blocks in released.chunks(max_blocks_per_item) {
392        if tx.send(blocks.to_vec()).await.is_err() {
393            return false;
394        }
395    }
396    true
397}

399struct BufferedRound {
400    first_buffered_at: Instant,
401    blocks: Vec<VerifiedBlock>,
402    // Aggregates the stake of block authors buffered on this stream only. Blocks dropped by
403    // broadcast lag or accepted before subscription are not counted, so a round may miss
404    // quorum here even when the validator has it — those rounds fall back to the release
405    // timeout.
406    quorum: StakeAggregator<QuorumThreshold>,
407}
408
409impl BufferedRound {
410    /// Buffers the block and counts its author's stake towards the round quorum.
411    /// Returns true when the quorum has been reached.
412    fn add_block(&mut self, block: VerifiedBlock, committee: &Committee) -> bool {
413        let reached_quorum = self.quorum.add(block.author(), committee);
414        self.blocks.push(block);
415        reached_quorum
416    }
417}
418
419impl Default for BufferedRound {
420    fn default() -> Self {
421        Self {
422            first_buffered_at: Instant::now(),
423            blocks: Vec::new(),
424            quorum: StakeAggregator::new(),
425        }
426    }
427}

429struct AcceptedBlockReleaseState {
430    context: Arc<Context>,
431    release_timeout: Duration,
432    buffered_rounds: BTreeMap<Round, BufferedRound>,
433    // Highest round released so far. Releasing a round (on quorum or timeout) also releases
434    // all buffered rounds below it, so released rounds form a contiguous prefix and blocks
435    // at or below this watermark pass through without buffering.
436    last_released_round: Round,
437}
438
439impl AcceptedBlockReleaseState {
440    fn new(context: Arc<Context>, release_timeout: Duration) -> Self {
441        Self {
442            context,
443            release_timeout,
444            buffered_rounds: BTreeMap::new(),
445            last_released_round: 0,
446        }
447    }
448
449    fn accept_blocks(&mut self, mut blocks: Vec<VerifiedBlock>) -> Vec<VerifiedBlock> {
450        // Process blocks in ascending order, so released blocks come out in increasing round
451        // order without re-sorting the (potentially much larger) output.
452        sort_blocks(&mut blocks);
453
454        let mut immediately_release = Vec::new();
455        for block in blocks {
456            let round = block.round();
457            if round <= self.last_released_round {
458                immediately_release.push(block);
459                continue;
460            }
461
462            let reached_quorum = self
463                .buffered_rounds
464                .entry(round)
465                .or_default()
466                .add_block(block, &self.context.committee);
467
468            if reached_quorum {
469                immediately_release.extend(self.release_rounds_up_to(round));
470            }
471        }
472        immediately_release
473    }
474
475    fn release_timed_out_rounds(&mut self) -> Vec<VerifiedBlock> {
476        let now = Instant::now();
477        let max_timed_out_round = self
478            .buffered_rounds
479            .iter()
480            .filter_map(|(round, buffered)| {
481                (now.duration_since(buffered.first_buffered_at) >= self.release_timeout)
482                    .then_some(*round)
483            })
484            .max();
485
486        match max_timed_out_round {
487            Some(round) => self.release_rounds_up_to(round),
488            None => Vec::new(),
489        }
490    }
491
492    fn release_all(&mut self) -> Vec<VerifiedBlock> {
493        match self.buffered_rounds.keys().next_back().copied() {
494            Some(max_round) => self.release_rounds_up_to(max_round),
495            None => Vec::new(),
496        }
497    }
498
499    fn next_timeout(&self) -> tokio::time::Instant {
500        // Effectively no timeout when nothing is buffered - the select loop re-evaluates on
501        // every incoming batch anyway.
502        const NO_BUFFERED_ROUNDS_TIMEOUT: Duration = Duration::from_secs(3600);
503
504        let next_timeout = self
505            .buffered_rounds
506            .values()
507            .map(|buffered| buffered.first_buffered_at + self.release_timeout)
508            .min()
509            .unwrap_or_else(|| Instant::now() + NO_BUFFERED_ROUNDS_TIMEOUT);
510        tokio::time::Instant::from_std(next_timeout)
511    }
512
513    /// Releases all buffered rounds at or below `round`. A quorum (or timeout) at a round
514    /// implies the rounds below it are settled, so there is no reason to keep them buffered.
515    fn release_rounds_up_to(&mut self, round: Round) -> Vec<VerifiedBlock> {
516        self.last_released_round = self.last_released_round.max(round);
517        let remaining_rounds = self.buffered_rounds.split_off(&round.saturating_add(1));
518        let released_rounds = std::mem::replace(&mut self.buffered_rounds, remaining_rounds);
519        released_rounds
520            .into_values()
521            .flat_map(|buffered| buffered.blocks)
522            .collect()
523    }
524}
525
526fn sort_blocks(blocks: &mut [VerifiedBlock]) {
527    blocks.sort_unstable_by_key(|block| block.reference());
528}

658

660        assert!(
661            state
662                .accept_blocks(vec![block1.clone(), block2.clone()])

664        );
665        assert_eq!(state.buffered_rounds.len(), 1);
666        assert_eq!(state.buffered_rounds[&1].blocks.len(), 2);
667
668        // Adding a third block forms a quorum, so all the buffered blocks are released and
669        // the round's buffer is cleaned up.
670        let released = state.accept_blocks(vec![block3.clone()]);
671        assert_eq!(
672            released
673                .iter()
674                .map(|block| block.reference())
675                .collect::<Vec<_>>(),
676            vec![block1.reference(), block2.reference(), block3.reference()]
677        );
678        assert!(state.buffered_rounds.is_empty());
679        assert_eq!(state.last_released_round, 1);
680
681        // Adding a forth block gets immediately released, as the round already reached quorum and released earlier.
682        let released = state.accept_blocks(vec![block4.clone()]);
683        assert_eq!(released.len(), 1);
684        assert_eq!(released[0].reference(), block4.reference());
685        assert!(state.buffered_rounds.is_empty());
