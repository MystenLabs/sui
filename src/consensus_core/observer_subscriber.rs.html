30/// Number of commit-sync batches of lag below which a suspended subscription is resumed.
31/// The subscription is suspended at `COMMIT_LAG_MULTIPLIER` (5) batches of lag, where the
32/// observer service starts rejecting streamed blocks anyway; the band between the two
33/// thresholds is hysteresis preventing rapid suspend/resume flapping.
34///
35/// The value must stay within `1..=COMMIT_LAG_MULTIPLIER`:
36/// - `>= 1` is load-bearing for liveness: commit sync never fetches partial batches, so
37///   it can leave up to `commit_sync_batch_size - 1` commits of residual lag that only
38///   streamed blocks can close. With `0`, the resume condition (zero lag) would be
39///   unreachable and the subscription would stay suspended forever.
40/// - `<= COMMIT_LAG_MULTIPLIER`, or the subscription resumes while still commit-lagging
41///   and the in-stream lag check immediately drops it again, churning reconnects until
42///   commit sync shrinks the lag below the suspension threshold.
43const RESUBSCRIBE_LAG_BATCHES: u32 = 1;
44
45/// How often a suspended subscription re-evaluates commit lag.
46const SUSPENSION_CHECK_INTERVAL: Duration = Duration::from_secs(1);
47
48/// ObserverSubscriber manages block stream subscriptions to peers (validators or other observers),
49/// taking care of retrying when subscription streams break. Blocks returned from peers are sent
50/// to the observer service for processing. The `ObserverSubscriber` can only subscribe to one peer at a time.
51///
52/// While the local commit index lags the quorum commit index too much, streamed blocks
53/// would be rejected by the observer service and re-fetched later via commit sync, so the
54/// subscription drops its stream and holds off reconnecting until commit sync has nearly
55/// caught up, saving the bandwidth and verification on both ends. While suspended, the quorum
56/// commit index stays effectively frozen: commit sync only fetches up to the already-known
57/// quorum commit index, so its observed commit votes cannot push it further. Instead, each
58/// resubscription refreshes the quorum commit index from the freshly streamed blocks, and
59/// if the node is still too far behind, the subscription is suspended again — so a deep
60/// catch-up proceeds as a bounded cycle of suspend, catch up, briefly resubscribe, re-suspend.
61pub(crate) struct ObserverSubscriber<C: ObserverNetworkClient, S: ObserverNetworkService> {
62    context: Arc<Context>,
63    network_client: Arc<C>,
64    observer_service: Arc<S>,
65    commit_vote_monitor: Arc<CommitVoteMonitor>,
66    dag_state: Arc<parking_lot::RwLock<DagState>>,
67    subscriptions: Mutex<Subscriptions>,
68    randomness_signature_handler: Option<Arc<dyn RandomnessSignatureHandler>>,
69}
70
71/// The current and retired subscriptions are tracked under a single lock, so replacing the
72/// current subscription and retiring the previous one is atomic w.r.t. concurrent subscribe()
73/// and stop() calls.
74struct Subscriptions {
75    current: Option<ObserverSubscription>,
76    // Retain replaced subscription tasks so stop() can await them and propagate panics.
77    retired: Vec<ObserverSubscription>,
78}
79
80struct ObserverSubscription {
81    // Signals the subscription task to stop cooperatively, instead of aborting it. Aborting
82    // would drop the block handler JoinSet owned by the task, which aborts the handlers without
83    // awaiting them: they could briefly keep running (holding strong references to the observer
84    // service) and any panic recorded in the set would be discarded. On this signal the task
85    // stops streaming, then cancels the remaining block handlers and awaits their termination,
86    // so join() returns only after no block handler is running and recorded handler panics
87    // have propagated.
88    shutdown_sender: Option<oneshot::Sender<()>>,
89    subscription_task: JoinHandle<()>,
90}
91
92impl ObserverSubscription {
93    fn request_shutdown(&mut self) {
94        if let Some(sender) = self.shutdown_sender.take() {
95            let _ = sender.send(());
96        }
97    }
98
99    async fn join(self) {
100        join_and_propagate_panic(self.subscription_task).await;
101    }
102}

105    pub(crate) fn new(
106        context: Arc<Context>,
107        network_client: Arc<C>,
108        observer_service: Arc<S>,
109        commit_vote_monitor: Arc<CommitVoteMonitor>,
110        dag_state: Arc<parking_lot::RwLock<DagState>>,
111        randomness_signature_handler: Option<Arc<dyn RandomnessSignatureHandler>>,
112    ) -> Self {
113        Self {
114            context,
115            network_client,
116            observer_service,
117            commit_vote_monitor,
118            dag_state,
119            subscriptions: Mutex::new(Subscriptions {
120                current: None,
121                retired: Vec::new(),
122            }),
123            randomness_signature_handler,
124        }

127    /// Subscribe to a peer (validator or observer) to receive block streams. The `ObserverSubscriber` can only subscribe to one peer at a time.
128    /// The method will stop the existing subscription (if any) and start a new one.
129    pub(crate) fn subscribe(&self, peer: PeerId) {
130        let context = self.context.clone();
131        let network_client = self.network_client.clone();
132        // ObserverSubscriber already holds these resources strongly. Give subscription tasks weak
133        // references so they do not become additional owners during shutdown.
134        let observer_service = Arc::downgrade(&self.observer_service);
135        let commit_vote_monitor = self.commit_vote_monitor.clone();
136        let dag_state = Arc::downgrade(&self.dag_state);
137        let randomness_signature_handler = self.randomness_signature_handler.clone();
138
139        let (shutdown_sender, shutdown_receiver) = oneshot::channel();
140        let subscription_task = spawn_monitored_task!(Self::subscription_loop(
141            context,
142            network_client,
143            observer_service,
144            commit_vote_monitor,
145            dag_state,
146            peer,
147            randomness_signature_handler,
148            shutdown_receiver,
149        ));
150
151        let mut subscriptions = self.subscriptions.lock();
152        // Reap retired subscriptions that have finished, so the list stays bounded under
153        // repeated resubscriptions.
154        subscriptions
155            .retired
156            .retain_mut(|subscription| !reap_finished_task(&mut subscription.subscription_task));
157        let previous_subscription = subscriptions.current.replace(ObserverSubscription {
158            shutdown_sender: Some(shutdown_sender),
159            subscription_task,
160        });
161        if let Some(mut previous_subscription) = previous_subscription {
162            previous_subscription.request_shutdown();
163            subscriptions.retired.push(previous_subscription);
164        }
165    }
166
167    /// Stop the active subscription (if any).
168    pub(crate) async fn stop(&self) {
169        let mut subscriptions = {
170            let mut subscriptions = self.subscriptions.lock();
171            let mut all = std::mem::take(&mut subscriptions.retired);
172            all.extend(subscriptions.current.take());
173            all
174        };
175        for subscription in &mut subscriptions {
176            subscription.request_shutdown();
177        }
178        for subscription in subscriptions {
179            subscription.join().await;
180        }
181    }

183    async fn subscription_loop(
184        context: Arc<Context>,
185        network_client: Arc<C>,
186        observer_service: Weak<S>,
187        commit_vote_monitor: Arc<CommitVoteMonitor>,
188        dag_state: Weak<parking_lot::RwLock<DagState>>,
189        peer: PeerId,
190        randomness_signature_handler: Option<Arc<dyn RandomnessSignatureHandler>>,
191        mut shutdown_receiver: oneshot::Receiver<()>,
192    ) {
193        let mut tasks = JoinSet::new();
194        {
195            // The `subscription` future borrows `tasks` mutably to spawn block handlers, so it
196            // is scoped to this block. When shutdown is signaled, the select below returns and
197            // `subscription` is dropped at the end of the block. This stops block streaming, but
198            // the current task (subscription_loop) and the block handlers already spawned into
199            // `tasks` keep running. Dropping `subscription` also ends the `&mut tasks` borrow,
200            // so the block handlers in `tasks` can be awaited below.
201            let subscription = Self::run_subscription(
202                context.clone(),
203                network_client,
204                observer_service,
205                commit_vote_monitor,
206                dag_state,
207                peer,
208                randomness_signature_handler,
209                &mut tasks,
210            );
211            tokio::pin!(subscription);
212            tokio::select! {
213                _ = &mut shutdown_receiver => {}
214                _ = &mut subscription => {}
215            }
216        }
217
218        shutdown_join_set(&mut tasks).await;
219
220        // The task may have been torn down mid-suspension (peer switch or stop), leaving
221        // the gauge at 1 with no task left to clear it. Reset it here; if a replacement
222        // task is genuinely suspended, it re-asserts 1 within SUSPENSION_CHECK_INTERVAL.
223        context
224            .metrics
225            .node_metrics
226            .observer_subscription_suspended
227            .set(0);
228    }
229
230    async fn run_subscription(
231        context: Arc<Context>,
232        network_client: Arc<C>,
233        observer_service: Weak<S>,
234        commit_vote_monitor: Arc<CommitVoteMonitor>,
235        dag_state: Weak<parking_lot::RwLock<DagState>>,
236        peer: PeerId,
237        randomness_signature_handler: Option<Arc<dyn RandomnessSignatureHandler>>,
238        tasks: &mut JoinSet<()>,
239    ) {
240        const IMMEDIATE_RETRIES: i64 = 3;
241        const MIN_TIMEOUT: Duration = Duration::from_millis(500);
242        let mut backoff = mysten_common::backoff::ExponentialBackoff::new(
243            Duration::from_millis(100),
244            Duration::from_secs(10),
245        );
246        let mut retries: i64 = 0;
247
248        'subscription: loop {
249            let mut delay = Duration::ZERO;
250            if retries > IMMEDIATE_RETRIES {
251                delay = backoff.next().unwrap();
252                debug!(
253                    "Delaying retry {} of peer {:?} subscription, in {} seconds",
254                    retries,
255                    peer.clone(),
256                    delay.as_secs_f32(),
257                );
258                sleep(delay).await;
259            } else if retries > 0 {
260                tokio::task::yield_now().await;
261            }
262            retries += 1;
263
264            // Hold off (re)connecting while commit-lagging: streamed blocks would be
265            // rejected by the observer service and later re-fetched via commit sync.
266            if !Self::wait_while_commit_lagging(&context, &commit_vote_monitor, &dag_state).await {
267                return;
268            }
269
270            // Recompute highest rounds from DagState before each connection attempt
271            // so reconnections resume from where we left off rather than re-fetching
272            // already-seen blocks. Clamp to the GC round, since blocks below it would
273            // be skipped anyway.
274            let highest_round_per_authority = {
275                let Some(dag_state) = dag_state.upgrade() else {
276                    return;
277                };
278                let ds = dag_state.read();
279                let gc_round = ds.gc_round();
280                let mut rounds = vec![0 as Round; context.committee.size()];
281                for (authority, _) in context.committee.authorities() {
282                    rounds[authority.value()] = ds
283                        .get_last_block_for_authority(authority)
284                        .round()
285                        .max(gc_round);
286                }
287                rounds
288            };
289
290            // Subscribe to stream blocks from the peer.
291            let request_timeout = MIN_TIMEOUT.max(delay);
292            let mut blocks = match network_client
293                .stream_blocks(peer.clone(), highest_round_per_authority, request_timeout)
294                .await
295            {
296                Ok(blocks) => {
297                    debug!("Subscribed to peer {:?} after {} attempts", peer, retries);
298                    blocks
299                }
300                Err(e) => {
301                    debug!("Failed to subscribe to blocks from peer {:?}: {}", peer, e);
302                    continue 'subscription;
303                }
304            };
305
306            let max_parallel_tasks = context.committee.size();
307            'stream: loop {
308                let _scope = monitored_scope("ObserverSubscriberStreamConsumer");
309
310                let next_item = tokio::select! {
311                    result = tasks.join_next(), if !tasks.is_empty() => {
312                        Self::handle_task_result(result);
313                        continue 'stream;
314                    }
315                    item = blocks.next(), if tasks.len() < max_parallel_tasks => item,
316                };
317
318                match next_item {
319                    Some(item) => {
320                        context
321                            .metrics
322                            .node_metrics
323                            .observer_subscribed_blocks_batch_size
324                            .observe(item.blocks.len() as f64);
325
326                        // During catch-up (commit lagging behind quorum), streamed blocks
327                        // would be rejected by the observer service and re-fetched via
328                        // commit sync, so drop the stream and hold off reconnecting until
329                        // nearly caught up, saving the bandwidth on both ends. The
330                        // disconnection is deliberate, so reconnect without backoff.
331                        let is_commit_lagging = dag_state.upgrade().is_none_or(|dag_state| {
332                            is_commit_lagging(
333                                &context,
334                                dag_state.read().last_commit_index(),
335                                commit_vote_monitor.quorum_commit_index(),
336                            )
337                        });
338                        if is_commit_lagging {
339                            retries = 0;
340                            backoff.reset();
341                            break 'stream;
342                        }
343
344                        if let Some(handler) = &randomness_signature_handler {
345                            for sig in item.auxiliary_data.randomness_signatures {
346                                handler.handle_randomness_signature(sig);
347                            }
348                        }
349
350                        for block in item.blocks {
351                            // Backpressure: wait if we've hit max parallelism.
352                            while tasks.len() >= max_parallel_tasks {
353                                Self::handle_task_result(tasks.join_next().await);
354                            }
355
356                            let observer_service = observer_service.clone();
357                            let peer_cloned = peer.clone();
358                            tasks.spawn(async move {
359                                let _scope =
360                                    monitored_scope("ObserverSubscriberTask::handle_block");
361
362                                let Some(observer_service) = observer_service.upgrade() else {
363                                    return;
364                                };
365                                let result = observer_service
366                                    .handle_block(peer_cloned.clone(), block)
367                                    .await;
368                                if let Err(e) = result {
369                                    match e {
370                                        ConsensusError::BlockRejected {
371                                            block_ref,
372                                            reason,
373                                        } => {
374                                            debug!(
375                                                "Failed to process block from peer for block {:?}: {}",
376                                                block_ref, reason
377                                            );
378                                        }
379                                        _ => {
380                                            info!("Received invalid block from peer: {}", e);
381                                        }
382                                    }
383                                }
384                            });
385                        }
386
387                        // Reset retries when a block is received and also reset the backoff.
388                        retries = 0;
389                        backoff.reset();
390                    }
391                    None => {
392                        debug!("Subscription to blocks from peer {:?} ended", peer);
393                        retries += 1;
394                        break 'stream;
395                    }
396                }
397            }
398
399            while !tasks.is_empty() {
400                Self::handle_task_result(tasks.join_next().await);
401            }
402        }
403    }
404
405    // While the local commit index lags the quorum commit index too much, waits for
406    // commit sync to bring it within `RESUBSCRIBE_LAG_BATCHES` batches before returning;
407    // the band between the suspension and resume thresholds is hysteresis preventing
408    // rapid flapping. While waiting, the quorum commit index stays effectively frozen:
409    // with the stream suspended no new blocks — and hence no new commit votes — arrive
410    // from the peer, and the blocks fetched by commit sync only carry votes for commits
411    // at or below the already-known quorum commit index, so they cannot advance it.
412    // Only once the subscription resumes do the freshly streamed higher-round blocks
413    // carry votes that push the quorum commit index forward; if that reveals the node
414    // is still too far behind, the stream is dropped and suspended again.
415    // Returns false when the node is shutting down.
416    async fn wait_while_commit_lagging(
417        context: &Context,
418        commit_vote_monitor: &CommitVoteMonitor,
419        dag_state: &Weak<parking_lot::RwLock<DagState>>,
420    ) -> bool {
421        let mut suspended = false;
422        loop {
423            let Some(dag_state) = dag_state.upgrade() else {
424                return false;
425            };
426            let local_commit_index = dag_state.read().last_commit_index();
427            drop(dag_state);
428            let quorum_commit_index = commit_vote_monitor.quorum_commit_index();
429
430            let caught_up = if suspended {
431                quorum_commit_index.saturating_sub(local_commit_index)
432                    <= context.parameters.commit_sync_batch_size * RESUBSCRIBE_LAG_BATCHES
433            } else {
434                !is_commit_lagging(context, local_commit_index, quorum_commit_index)
435            };
436            // The gauge writes are level-triggered (re-asserted every iteration) rather
437            // than transition-triggered: a task torn down mid-suspension (peer switch or
438            // stop) never performs the closing transition, so a stale value must be
439            // overwritten by whichever task evaluates the suspension next. Logs stay on
440            // transitions.
441            if caught_up {
442                if suspended {
443                    info!(
444                        "Resuming block stream subscription: local commit index {} caught up with quorum commit index {}",
445                        local_commit_index, quorum_commit_index,
446                    );
447                }
448                context
449                    .metrics
450                    .node_metrics
451                    .observer_subscription_suspended
452                    .set(0);
453                return true;
454            }
455            if !suspended {
456                suspended = true;
457                info!(
458                    "Suspending block stream subscription: local commit index {} lags quorum commit index {}, blocks will arrive via commit sync",
459                    local_commit_index, quorum_commit_index,
460                );
461            }
462            context
463                .metrics
464                .node_metrics
465                .observer_subscription_suspended
466                .set(1);
467            sleep(SUSPENSION_CHECK_INTERVAL).await;
468        }
469    }
470
471    fn handle_task_result(result: Option<Result<(), JoinError>>) {
472        if let Some(Err(error)) = result {
473            if error.is_panic() {
474                std::panic::resume_unwind(error.into_panic());
475            }
476            warn!("Observer block handler task was cancelled: {error}");
477        }
478    }
