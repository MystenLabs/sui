9/// Awaits the task and propagates its panic if it panicked. Task cancellation is ignored.
10pub(crate) async fn join_and_propagate_panic<T>(task: JoinHandle<T>) {
11    if let Err(error) = task.await
12        && error.is_panic()
13    {
14        std::panic::resume_unwind(error.into_panic());
15    }
16}
17
18/// Aborts all tasks in the set and awaits them, propagating the first panic among them.
19/// Task cancellations are ignored.
20pub(crate) async fn shutdown_join_set<T: 'static>(tasks: &mut JoinSet<T>) {
21    tasks.abort_all();
22    // Drain the whole set before propagating a panic, so no task is dropped without
23    // being awaited.
24    let mut first_panic = None;
25    while let Some(result) = tasks.join_next().await {
26        if let Err(error) = result
27            && error.is_panic()
28            && first_panic.is_none()
29        {
30            first_panic = Some(error.into_panic());
31        }
32    }
33    if let Some(panic) = first_panic {
34        std::panic::resume_unwind(panic);
35    }
36}

38/// Reaps the task if it has finished, propagating its panic if it panicked.
39/// Returns true when the task has been reaped and can be dropped.
40pub(crate) fn reap_finished_task(task: &mut JoinHandle<()>) -> bool {
41    if !task.is_finished() {
42        return false;
43    }
44    // A finished task completes immediately when polled.
45    if let Some(Err(error)) = task.now_or_never()
46        && error.is_panic()
47    {
48        std::panic::resume_unwind(error.into_panic());
49    }
50    true
51}

53/// Runs the closure on the blocking pool, like tokio::task::spawn_blocking, but resolves
54/// to ConsensusError::Shutdown instead of a JoinError when the task is cancelled by
55/// runtime shutdown. Panics from the closure are resumed on the calling thread.
56pub(crate) async fn spawn_blocking<F, T>(f: F) -> ConsensusResult<T>
57where
58    F: FnOnce() -> T + Send + 'static,
59    T: Send + 'static,
60{
61    tokio::task::spawn_blocking(f)
62        .await
63        .map_err(spawn_blocking_join_error)
64}
65
66/// Converts a JoinError from a spawn_blocking task into a ConsensusError.
67/// Panics from the blocking task are resumed on the calling thread. Otherwise the task
68/// was cancelled, which only happens when the runtime is shutting down.
69fn spawn_blocking_join_error(e: tokio::task::JoinError) -> ConsensusError {
70    if e.is_panic() {
71        std::panic::resume_unwind(e.into_panic());
72    }
73    ConsensusError::Shutdown
74}

76#[cfg(test)]
77mod test {
78    use super::*;
79
80    /// Reproduces the production scenario: spawn_blocking racing with runtime shutdown
81    /// gets its task cancelled instead of run, so its JoinHandle resolves to
82    /// JoinError::Cancelled.
83    #[tokio::test]
84    async fn test_spawn_blocking_join_error_on_runtime_shutdown() {
85        let runtime = tokio::runtime::Builder::new_multi_thread()
86            .worker_threads(1)
87            .build()
88            .unwrap();
89        let handle = runtime.handle().clone();
90        runtime.shutdown_background();
91
92        let cancelled = handle.spawn_blocking(|| ());
93        let join_error = cancelled.await.unwrap_err();
94        assert!(join_error.is_cancelled());
95        assert!(matches!(
96            spawn_blocking_join_error(join_error),
97            ConsensusError::Shutdown
98        ));
99    }
100
101    #[tokio::test]
102    #[should_panic(expected = "boom")]
103    async fn test_spawn_blocking_resumes_panic() {
104        let _ = spawn_blocking(|| panic!("boom")).await;
105    }
