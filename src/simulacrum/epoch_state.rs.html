50pub struct EpochState {
51    epoch_start_state: EpochStartSystemState,
52    committee: Committee,
53    protocol_config: ProtocolConfig,
54    execution_metrics: Arc<ExecutionMetrics>,
55    bytecode_verifier_metrics: Arc<BytecodeVerifierMetrics>,
56    executor: Arc<dyn Executor + Send + Sync>,
57    /// Keeps arbitrary simulated packages out of committed execution's VM cache.
58    simulation_executor: Arc<dyn Executor + Send + Sync>,
59    chain_identifier: ChainIdentifier,
60    unsettled_object_withdrawals: Arc<UnsettledObjectWithdrawals>,
61    /// A counter that advances each time we advance the clock in order to ensure that each update
62    /// txn has a unique digest. This is reset on epoch changes
63    next_consensus_round: u64,
64}

66impl EpochState {
67    pub fn new(system_state: SuiSystemState, chain_identifier: ChainIdentifier) -> Self {
68        let protocol_config =
69            ProtocolConfig::get_for_version(system_state.protocol_version().into(), Chain::Unknown);
70        Self::new_with_protocol_config(system_state, protocol_config, chain_identifier)
71    }
72
73    pub fn new_with_protocol_config(
74        system_state: SuiSystemState,
75        protocol_config: ProtocolConfig,
76        chain_identifier: ChainIdentifier,
77    ) -> Self {
78        let epoch_start_state = system_state.into_epoch_start_state();
79        let committee = epoch_start_state.get_sui_committee();
80        let registry = prometheus::Registry::new();
81        let execution_metrics = Arc::new(ExecutionMetrics::new(&registry));
82        let bytecode_verifier_metrics = Arc::new(BytecodeVerifierMetrics::new(&registry));
83        let executor = sui_execution::executor(&protocol_config, true).unwrap();
84        let simulation_executor = sui_execution::executor(&protocol_config, true).unwrap();
85        let unsettled_object_withdrawals = Arc::new(UnsettledObjectWithdrawals::new(Arc::new(
86            ObjectFundsCheckerMetrics::new(&registry),
87        )));
88
89        Self {
90            epoch_start_state,
91            committee,
92            protocol_config,
93            execution_metrics,
94            bytecode_verifier_metrics,
95            executor,
96            simulation_executor,
97            chain_identifier,
98            unsettled_object_withdrawals,
99            next_consensus_round: 0,
100        }
101    }

112        let round = self.next_consensus_round;

137    pub fn execute_transaction(
138        &self,
139        store: &dyn SimulatorStore,
140        deny_config: &TransactionDenyConfig,
141        verifier_signing_config: &VerifierSigningConfig,
142        transaction: &VerifiedTransaction,
143    ) -> Result<(
144        InnerTemporaryStore,
145        SuiGasStatus,
146        TransactionEffects,
147        Result<(), sui_types::error::ExecutionError>,
148    )> {
149        let tx_digest = *transaction.digest();
150        let tx_data = &transaction.data().intent_message().value;
151        let input_object_kinds = tx_data.input_objects()?;
152        let receiving_object_refs = tx_data.receiving_objects();
153
154        sui_transaction_checks::deny::check_transaction_for_signing(
155            tx_data,
156            transaction.tx_signatures(),
157            &input_object_kinds,
158            &receiving_object_refs,
159            deny_config,
160            &store,
161        )?;
162
163        let (input_objects, receiving_objects) = store.read_objects_for_synchronous_execution(
164            &tx_digest,
165            &input_object_kinds,
166            &receiving_object_refs,
167        )?;
168
169        // Run the transaction input checks that would run when submitting the txn to a validator
170        // for signing
171        let (gas_status, checked_input_objects) = sui_transaction_checks::check_transaction_input(
172            &self.protocol_config,
173            self.epoch_start_state.reference_gas_price(),
174            transaction.data().transaction_data(),
175            input_objects,
176            &receiving_objects,
177            &self.bytecode_verifier_metrics,
178            verifier_signing_config,
179        )?;
180
181        let transaction_data = transaction.data().transaction_data();
182        let (kind, signer, gas_data) = transaction_data.execution_parts();
183        let system_object_versions =
184            sui_types::base_types::SystemObjectVersions::from_latest_in_store(
185                store.backing_store(),
186            );
187        let (inner_temp_store, gas_status, effects, _timings, result) = self
188            .executor
189            .execute_transaction_to_effects_and_execution_error(
190                store.backing_store(),
191                &self.protocol_config,
192                self.execution_metrics.clone(),
193                false, // enable_expensive_checks
194                // TODO: Integrate with early execution error
195                ExecutionOrEarlyError::ok(None),
196                &self.epoch_start_state.epoch(),
197                self.epoch_start_state.epoch_start_timestamp_ms(),
198                checked_input_objects,
199                system_object_versions,
200                self.unsettled_object_withdrawals.as_ref() as &dyn UnsettledObjectFundsRead,
201                gas_data,
202                gas_status,
203                kind,
204                None, // compat_args
205                signer,
206                tx_digest,
207                &mut None,
208            );
209        if self
210            .protocol_config
211            .check_object_funds_withdraw_in_execution()
212            && effects.status().is_ok()
213            && let Some(accumulator_version) =
214                system_object_versions.get(&SUI_ACCUMULATOR_ROOT_OBJECT_ID)
215        {
216            self.unsettled_object_withdrawals
217                .record_object_funds_withdraws(
218                    transaction.transaction_data(),
219                    &effects,
220                    &inner_temp_store.accumulator_running_max_withdraws,
221                    accumulator_version.version,
222                    self.chain_identifier,
223                );
224        }
225        Ok((inner_temp_store, gas_status, effects, result))
226    }

228    pub(crate) fn commit_accumulator_versions(
229        &self,
230        committed_accumulator_versions: Vec<SequenceNumber>,
231    ) {
232        self.unsettled_object_withdrawals
233            .commit_accumulator_versions(committed_accumulator_versions);
234    }

236    pub(crate) fn simulate_transaction<S: SimulatorStore + Send + Sync>(
237        &self,
238        store: &S,
239        transaction_deny_config: &TransactionDenyConfig,
240        verifier_signing_config: &VerifierSigningConfig,
241        transaction: TransactionData,
242        checks: TransactionChecks,
243        allow_mock_gas_coin: bool,
244    ) -> SuiResult<SimulateTransactionResult> {
245        let input_loader = SimulatorInputLoader(store);
246        let account_funds_read = SimulatorAccountFundsRead(store);
247        let coin_reservation_resolver = BorrowedCoinReservationResolver::new(store);
248        let certificate_deny_set = HashSet::new();
249
250        simulate_transaction(
251            transaction,
252            checks,
253            allow_mock_gas_coin,
254            Some(self.reference_gas_price()),
255            TxValidityCheckContext {
256                config: &self.protocol_config,
257                epoch: self.epoch(),
258                chain_identifier: self.chain_identifier,
259                reference_gas_price: self.reference_gas_price(),
260                committee_size: self.committee.num_members() as u32,
261            },
262            self.epoch(),
263            self.epoch_start_state.epoch_start_timestamp_ms(),
264            self.chain_identifier,
265            transaction_deny_config,
266            &certificate_deny_set,
267            &input_loader,
268            store,
269            store,
270            self.simulation_executor.as_ref(),
271            &coin_reservation_resolver,
272            &account_funds_read,
273            verifier_signing_config,
274            &self.bytecode_verifier_metrics,
275            &self.execution_metrics,
276        )
277    }

280impl<S: SimulatorStore + Send + Sync> SimulatorAccountFundsRead<'_, S> {
281    fn account_amount(
282        &self,
283        account_id: &AccumulatorObjId,
284        version: Option<SequenceNumber>,
285    ) -> u128 {
286        AccumulatorValue::load_by_id::<U128>(self.0, version, *account_id)
287            .expect("simulator accumulator reads must succeed")
288            .map(|value| value.value)
289            .unwrap_or(0)
290    }
291}

293impl<S: SimulatorStore + Send + Sync> AccountFundsRead for SimulatorAccountFundsRead<'_, S> {
294    fn get_latest_account_amount(&self, account_id: &AccumulatorObjId) -> u128 {
295        self.account_amount(account_id, None)
296    }
297
298    fn get_consistent_latest_account_amount_and_version(
299        &self,
300        account_id: &AccumulatorObjId,
301    ) -> (u128, SequenceNumber) {
302        let root_version = SimulatorStore::get_object(self.0, &SUI_ACCUMULATOR_ROOT_OBJECT_ID)
303            .expect("simulator accumulator root must exist")
304            .version();
305        (
306            self.get_account_amount_at_version(account_id, root_version),
307            root_version,
308        )
309    }
310
311    fn get_account_amount_at_version(
312        &self,
313        account_id: &AccumulatorObjId,
314        version: SequenceNumber,
315    ) -> u128 {
316        self.account_amount(account_id, Some(version))
317    }
318}
