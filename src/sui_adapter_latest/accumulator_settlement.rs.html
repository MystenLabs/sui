4use std::collections::{BTreeMap, BTreeSet};

6use move_core_types::account_address::AccountAddress;
7use sui_types::{
8    SUI_ACCUMULATOR_ROOT_OBJECT_ID, SUI_FRAMEWORK_ADDRESS, SUI_FRAMEWORK_PACKAGE_ID,
9    accumulator_root::{
10        ACCUMULATOR_ROOT_SETTLE_U128_FUNC, ACCUMULATOR_ROOT_SETTLEMENT_PROLOGUE_FUNC,
11        ACCUMULATOR_ROOT_SETTLEMENT_SETTLE_EVENTS_FUNC, ACCUMULATOR_SETTLEMENT_MODULE,
12    },
13    transaction::{
14        Argument, CallArg, Command, ObjectArg, ProgrammableTransaction, SharedObjectMutability,
15    },

18pub(crate) struct ExecutionBatch {
19    pub transaction: ProgrammableTransaction,
20    pub command_indices: Vec<usize>,
21}
22
23pub(crate) fn execution_batches(transaction: ProgrammableTransaction) -> Vec<ExecutionBatch> {
24    let groups = settlement_groups(&transaction);
25    let Some(mut groups) = groups.filter(|groups| groups.len() > 1) else {
26        return vec![ExecutionBatch {
27            command_indices: (0..transaction.commands.len()).collect(),
28            transaction,
29        }];
30    };
31
32    // The prologue records conservation totals for the entire transaction, exactly once.
33    groups.first_entry().unwrap().get_mut().insert(0, 0);
34    let ProgrammableTransaction { inputs, commands } = transaction;
35    let mut commands: Vec<_> = commands.into_iter().map(Some).collect();
36    groups
37        .into_values()
38        .map(|command_indices| {
39            let mut batch_inputs = vec![];
40            let mut input_indices = BTreeMap::new();
41            let batch_commands = command_indices
42                .iter()
43                .map(|&i| {
44                    let Command::MoveCall(mut call) = commands[i].take().unwrap() else {
45                        unreachable!("settlement commands were checked before partitioning")
46                    };
47                    for argument in &mut call.arguments {
48                        let Argument::Input(i) = argument else {
49                            unreachable!("settlement arguments were checked before partitioning")
50                        };
51                        *i = *input_indices.entry(*i).or_insert_with(|| {
52                            let new_index = u16::try_from(batch_inputs.len())
53                                .expect("a batch has no more inputs than the original transaction");
54                            batch_inputs.push(inputs[usize::from(*i)].clone());
55                            new_index
56                        });
57                    }
58                    Command::MoveCall(call)
59                })
60                .collect();
61            ExecutionBatch {
62                transaction: ProgrammableTransaction {
63                    inputs: batch_inputs,
64                    commands: batch_commands,
65                },
66                command_indices,
67            }
68        })
69        .collect()
70}

72fn settlement_groups(
73    transaction: &ProgrammableTransaction,
74) -> Option<BTreeMap<BTreeSet<AccountAddress>, Vec<usize>>> {
75    let mut root = None;
76    for (i, input) in transaction.inputs.iter().enumerate() {
77        match input {
78            CallArg::Pure(_) => (),
79            CallArg::Object(ObjectArg::SharedObject {
80                id,
81                mutability: SharedObjectMutability::NonExclusiveWrite,
82                ..
83            }) if *id == SUI_ACCUMULATOR_ROOT_OBJECT_ID && root.is_none() => {
84                root = Some(Argument::Input(u16::try_from(i).ok()?));
85            }
86            _ => return None,
87        }
88    }
89    let root = root?;
90    let mut groups: BTreeMap<BTreeSet<AccountAddress>, Vec<usize>> = BTreeMap::new();
91    for (i, command) in transaction.commands.iter().enumerate() {
92        let Command::MoveCall(call) = command else {
93            return None;
94        };
95        if call.package != SUI_FRAMEWORK_PACKAGE_ID
96            || call.module != ACCUMULATOR_SETTLEMENT_MODULE.as_str()
97            || call.arguments.first() != Some(&root)
98        {
99            return None;
100        }
101        let (function, arity, arguments) = if i == 0 {
102            (ACCUMULATOR_ROOT_SETTLEMENT_PROLOGUE_FUNC, 0, 6)
103        } else if call.function == ACCUMULATOR_ROOT_SETTLE_U128_FUNC.as_str() {
104            (ACCUMULATOR_ROOT_SETTLE_U128_FUNC, 1, 4)
105        } else {
106            (ACCUMULATOR_ROOT_SETTLEMENT_SETTLE_EVENTS_FUNC, 0, 5)
107        };
108        if call.function != function.as_str()
109            || call.type_arguments.len() != arity
110            || call.arguments.len() != arguments
111            || !call.arguments.iter().skip(1).all(|argument| {
112                matches!(argument, Argument::Input(i)
113                    if matches!(transaction.inputs.get(usize::from(*i)), Some(CallArg::Pure(_))))
114            })
115        {
116            return None;
117        }
118        if i != 0 {
119            // settle_events has no type arguments; its EventStreamHead is a framework type.
120            let mut addresses = BTreeSet::from([SUI_FRAMEWORK_ADDRESS]);
121            for ty in &call.type_arguments {
122                addresses.extend(ty.to_type_tag().ok()?.all_addresses());
123            }
124            groups.entry(addresses).or_default().push(i);
125        }
126    }
127    Some(groups)
128}
