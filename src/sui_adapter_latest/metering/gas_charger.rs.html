27// The gas state and the payment of a transaction. The engine settles each phase here, and
28// `charge` applies the final charge to the payment.
29#[derive(Debug)]
30pub(crate) struct GasCharger {
31    tx_digest: TransactionDigest,
32    payment: PaymentMetadata,
33    state: GasState,
34}
35
36// Internal representation of how a transaction's gas is being paid.
37// `Unmetered` for no payment (dev inspect and system transactions).
38// `Gasless` for metered-but-free transactions (gas is metered but not charged).
39// `Unsmashed` until `smash_gas` combines the signed payment methods.
40// `Smash` when one or more user-provided payment methods have been combined into a single
41// source.
42#[derive(Debug)]
43enum PaymentMetadata {
44    Unmetered,
45    Gasless,
46    // Never empty.
47    Unsmashed(Vec<PaymentMethod>),
48    // Contains the list of payments (coins and address balances) and additional metadata
49    Smash(SmashMetadata),
50}
51
52// State produced by smashing the gas payment sources into one. The charge goes to the smash
53// target unless the PTB moved the gas coin.
54#[derive(Debug)]
55struct SmashMetadata {
56    // Where the PTB moved the gas coin, if it did.
57    moved_to: Option<PaymentLocation>,
58    // The total balance of all smashed payment methods.
59    total_smashed: u64,
60    // The primary payment method, which receives `total_smashed`.
61    smash_target: PaymentMethod,
62    // The original payment methods to be smashed into the `smash_target`. It does not include
63    // the `smash_target` itself. Keyed by location to guarantee uniqueness.
64    smashed_payments: IndexMap<PaymentLocation, PaymentMethod>,
65}
66
67// A single source of SUI used to pay for gas: either a coin object or a withdrawal
68// reservation from an address balance.
69#[derive(Debug, Clone, Copy)]
70enum PaymentMethod {
71    Coin(ObjectRef),
72    AddressBalance(SuiAddress, /* withdrawal reservation */ u64),
73}
74
75// Identifies where a gas payment lives, independent of its value (`ObjectRef` or reservation).
76// Used often as a key, e.g. during smashing and during gas final charging.
77#[derive(Debug, PartialEq, Eq, Clone, Copy, Hash)]
78pub(crate) enum PaymentLocation {
79    Coin(ObjectID),
80    AddressBalance(SuiAddress),
81}

83// A resolved gas payment: the location that will receive the final charge or refund,
84// paired with the total SUI available after smashing. Produced by
85// `GasMeter::payment` and consumed by PTB execution to set up the
86// runtime gas coin.
87#[derive(Debug, Clone, Copy)]
88pub(crate) struct GasPayment {
89    // The location of the gas payment (coin or address balance), which also serves as the
90    // target for smashed gas payments.
91    pub location: PaymentLocation,
92    // The total amount available for gas payment after smashing
93    pub amount: u64,
94}

96impl GasCharger {
97    // Budget and price follow the rule signing uses. The store is untouched until
98    // `smash_gas`.
99    pub fn new(
100        tx_digest: TransactionDigest,
101        gas_data: &GasData,
102        transaction_kind: &TransactionKind,
103        reference_gas_price: u64,
104        protocol_config: &ProtocolConfig,
105    ) -> Self {
106        let metering = match execution_gas_params(
107            gas_data,
108            transaction_kind,
109            reference_gas_price,
110            protocol_config,
111        ) {
112            None => Metering::Unmetered,
113            Some((budget, price)) => Metering::Metered {
114                budget,
115                price,
116                reference_gas_price,
117            },
118        };
119        let payment = match metering {
120            Metering::Unmetered => PaymentMetadata::Unmetered,
121            Metering::Metered { .. } if is_gasless_transaction(gas_data, transaction_kind) => {
122                PaymentMetadata::Gasless
123            }
124            Metering::Metered { .. } => {
125                PaymentMetadata::Unsmashed(PaymentMethod::from_gas_data(gas_data))
126            }
127        };
128        Self {
129            tx_digest,
130            payment,
131            state: GasState::new(metering, protocol_config),
132        }
133    }

155    // The smash target and the total smashed. `None` for unmetered and gasless.
156    fn gas_payment(&self) -> Option<GasPayment> {
157        match &self.payment {
158            PaymentMetadata::Unmetered | PaymentMetadata::Gasless => None,
159            PaymentMetadata::Unsmashed(_) => unreachable!("gas is smashed before execution"),
160            PaymentMetadata::Smash(metadata) => Some(GasPayment {
161                location: metadata.smash_target.location(),
162                amount: metadata.total_smashed,
163            }),
164        }
165    }

172    // Payment setup.
173
174    // Smash the signed payments into the first one. Input checks validate the gas coins, so
175    // a bad coin panics.
176    pub fn smash_gas(
177        &mut self,
178        temporary_store: &mut TemporaryStore<'_>,
179    ) -> Result<(), ExecutionError> {
180        let PaymentMetadata::Unsmashed(methods) = &self.payment else {
181            debug_assert!(
182                !matches!(self.payment, PaymentMetadata::Smash(_)),
183                "gas is smashed once"
184            );
185            return Ok(());
186        };
187        let mut metadata = SmashMetadata::new(methods)?;
188        metadata.smash_gas(&self.tx_digest, temporary_store);
189        self.payment = PaymentMetadata::Smash(metadata);
190        Ok(())
191    }
192
193    // Smash again after the writes were dropped. The charge goes back to the smash target.
194    pub fn resmash(&mut self, temporary_store: &mut TemporaryStore<'_>) {
195        match &mut self.payment {
196            PaymentMetadata::Smash(metadata) => {
197                metadata.reset_charge_location();
198                metadata.smash_gas(&self.tx_digest, temporary_store);
199            }
200            PaymentMetadata::Unmetered | PaymentMetadata::Gasless => {}
201            // A failed smash leaves through the BumpOnly exit before any reset.
202            PaymentMetadata::Unsmashed(_) => debug_assert!(false, "gas is smashed before a reset"),
203        }
204    }

223    // An execution error wins over running out of gas.
224    pub fn settle_execution<T, E: ExecutionErrorTrait>(
225        &mut self,
226        gas_meter: GasMeter,
227        result: Result<T, E>,
228    ) -> Result<T, E> {
229        let GasMeter {
230            execution_meter,
231            charge_location,
232            ..
233        } = gas_meter;
234        if let (Some(location), PaymentMetadata::Smash(metadata)) =
235            (charge_location, &mut self.payment)
236        {
237            metadata.move_charge_location(location);
238        }
239        let is_move_abort = matches!(
240            result.as_ref().err().map(|e| e.kind()),
241            Some(sui_types::execution_status::ExecutionErrorKind::MoveAbort(
242                ..
243            ))
244        );
245        let charge = self.state.settle_execution(execution_meter, is_move_abort);
246        match result {
247            Ok(v) => charge.map(|_| v).map_err(Into::into),
248            Err(e) => Err(e),
249        }
250    }

264    // Payment.
265
266    // Apply the final charge to the payment.
267    pub fn charge<T, E: ExecutionErrorTrait>(
268        self,
269        temporary_store: &mut TemporaryStore<'_>,
270        execution_result: &Result<T, E>,
271    ) -> ChargedGas {
272        let cost_summary = match &self.payment {
273            PaymentMetadata::Unmetered => {
274                // Park unmetered (system-tx) storage rebate into 0x5 so SUI is not dropped.
275                let unmetered_storage_rebate = self.state.unmetered_storage_rebate();
276                temporary_store.conserve_unmetered_storage_rebate(unmetered_storage_rebate);
277                GasCostSummary::default()
278            }
279            PaymentMetadata::Unsmashed(_) => {
280                unreachable!("gas is smashed before it is charged")
281            }
282            PaymentMetadata::Gasless => {
283                if execution_result.is_err() {
284                    GasCostSummary::default()
285                } else {
286                    let cost_summary = self.state.summary();
287                    let storage_cost = cost_summary.storage_cost;
288                    assert!(
289                        storage_cost == 0,
290                        "Gasless transaction must not incur storage cost, got {storage_cost}"
291                    );
292                    let sender_rebate = cost_summary.storage_rebate;
293                    GasCostSummary {
294                        computation_cost: sender_rebate,
295                        storage_cost: 0,
296                        storage_rebate: sender_rebate,
297                        non_refundable_storage_fee: cost_summary.non_refundable_storage_fee,
298                    }
299                }
300            }
301            PaymentMetadata::Smash(metadata) => {
302                if let PaymentLocation::Coin(_) = metadata.charge_location() {
303                    trace!(target: "replay_gas_info", "Gas smashing has occurred for this transaction");
304                }
305                let cost_summary = self.state.summary();
306                metadata.apply(temporary_store, cost_summary)
307            }
308        };
309        let (used_coins, charge_location) = match &self.payment {
310            PaymentMetadata::Smash(metadata) => (
311                metadata.used_coins().copied().collect(),
312                Some(metadata.charge_location()),
313            ),
314            _ => (vec![], None),
315        };
316        ChargedGas {
317            cost_summary,
318            gas_report: self.state.gas_usage_report(),
319            gas_coin: self
320                .gas_payment()
321                .and_then(|payment| match payment.location {
322                    PaymentLocation::Coin(coin_id) => Some(coin_id),
323                    PaymentLocation::AddressBalance(_) => None,
324                }),
325            used_coins,
326            charge_location,
327        }
328    }
329}

417        protocol_config: &ProtocolConfig,

420        // times two for the global pause and per-address settings
421        // this "overcharges" slightly since it does not check the global pause for each owner
422        // but rather each coin type.
423        let bytes_read_per_owner = CONFIG_SETTING_DYNAMIC_FIELD_SIZE_FOR_GAS;
424        // associate the cost with dynamic field access so that it will increase if/when this
425        // cost increases
426        let cost_per_byte =
427            protocol_config.dynamic_field_borrow_child_object_type_cost_per_byte() as usize;
428        let owner_cost = bytes_read_per_owner
429            .saturating_mul(cost_per_byte)
430            .saturating_mul(num_non_gas_coin_owners as usize);
431        self.execution_meter.charge_read_bytes(owner_cost)
432    }

435impl SmashMetadata {
436    // The first payment is the smash target. Reservations on one address are summed. A
437    // duplicate gas coin or an overflowing sum is an invariant violation.
438    fn new(methods: &[PaymentMethod]) -> Result<Self, ExecutionError> {
439        let invalid = || {
440            ExecutionError::invariant_violation(
441                "invalid gas payment: duplicate gas coin or reservation overflow",
442            )
443        };
444        let mut unique_methods: IndexMap<PaymentLocation, PaymentMethod> = IndexMap::new();
445        for &method in methods {
446            match (unique_methods.entry(method.location()), method) {
447                (indexmap::map::Entry::Vacant(entry), method) => {
448                    entry.insert(method);
449                }
450                (
451                    indexmap::map::Entry::Occupied(mut occupied),
452                    PaymentMethod::AddressBalance(other, additional),
453                ) => {
454                    let PaymentMethod::AddressBalance(addr, amount) = occupied.get_mut() else {
455                        unreachable!("Payment method does not match location")
456                    };
457                    assert_eq!(*addr, other, "Payment method does not match location");
458                    *amount = amount.checked_add(additional).ok_or_else(invalid)?;
459                }
460                (indexmap::map::Entry::Occupied(_), _) => return Err(invalid()),
461            }
462        }
463        let (_, smash_target) = unique_methods.shift_remove_index(0).ok_or_else(invalid)?;
464        Ok(Self {
465            moved_to: None,
466            // set by smash_gas
467            total_smashed: 0,
468            smash_target,
469            smashed_payments: unique_methods,
470        })
471    }
472
473    // The charge location.
474
475    fn charge_location(&self) -> PaymentLocation {
476        self.moved_to.unwrap_or(self.smash_target.location())
477    }
478
479    fn move_charge_location(&mut self, location: PaymentLocation) {
480        self.moved_to = Some(location);
481    }
482
483    fn reset_charge_location(&mut self) {
484        self.moved_to = None;
485    }
486
487    // Coin and balance operations.
488
489    // Iterates over all payment methods: the smash target followed by the smashed payments.
490    fn payment_methods(&self) -> impl Iterator<Item = &'_ PaymentMethod> {
491        std::iter::once(&self.smash_target).chain(self.smashed_payments.values())
492    }
493
494    fn smash_gas(
495        &mut self,
496        tx_digest: &TransactionDigest,
497        temporary_store: &mut TemporaryStore<'_>,
498    ) {
499        // sum the value of all gas coins
500        let total_smashed = self
501            .payment_methods()
502            .map(|payment| match payment {
503                PaymentMethod::AddressBalance(_, reservation) => Ok(*reservation),
504                PaymentMethod::Coin(obj_ref) => {
505                    let obj_data = temporary_store
506                        .objects()
507                        .get(&obj_ref.0)
508                        .map(|obj| &obj.data);
509                    let Some(Data::Move(move_obj)) = obj_data else {
510                        return Err(ExecutionError::invariant_violation(
511                            "Provided non-gas coin object as input for gas!",
512                        ));
513                    };
514                    if !move_obj.type_().is_gas_coin() {
515                        return Err(ExecutionError::invariant_violation(
516                            "Provided non-gas coin object as input for gas!",
517                        ));
518                    }
519                    Ok(move_obj.get_coin_value_unsafe())
520                }
521            })
522            .collect::<Result<Vec<u64>, ExecutionError>>()
523            // transaction and certificate input checks must have insured that all gas coins
524            // are valid
525            .unwrap_or_else(|_| {
526                panic!(
527                    "Unable to process gas payments for transaction {}",
528                    tx_digest
529                )
530            })
531            .iter()
532            .sum();
533        // If it is 0, then we are smashing for the first time (at the beginning of execution).
534        // If it is non-zero, then we are re-smashing after a reset (due to some sort of
535        // failure in charging for gas), and the total should not change.
536        debug_assert!(
537            self.total_smashed == 0 || self.total_smashed == total_smashed,
538            "Gas smashing should not change after a reset"
539        );
540        self.total_smashed = total_smashed;
541
542        let smash_location = self.smash_target.location();
543        // delete all gas objects except the smash target
544        for payment_method in self.smashed_payments.values() {
545            let location = payment_method.location();
546            assert_ne!(location, smash_location, "Payment methods must be unique");
547            match payment_method {
548                PaymentMethod::AddressBalance(sui_address, reservation) => {
549                    assert_reachable!("smashed payment is address-balance reservation");
550                    let balance_type =
551                        sui_types::balance::Balance::type_tag(sui_types::gas_coin::GAS::type_tag());
552                    let event = AccumulatorEvent::from_balance_change(
553                        *sui_address,
554                        balance_type,
555                        i64::try_from(*reservation).unwrap().checked_neg().unwrap(),
556                    )
557                    .expect("Failed to create accumulator event for gas smashing");
558                    temporary_store.add_accumulator_event(event);
559                }
560                PaymentMethod::Coin((id, _, _)) => {
561                    assert_reachable!("smashed payment is coin object");
562                    temporary_store.delete_input_object(id);
563                }
564            }
565        }
566        match &self.smash_target {
567            PaymentMethod::AddressBalance(sui_address, reservation) => {
568                assert_reachable!("smash target is address-balance reservation");
569                // The reservation here is only a maximal withdrawal from this address balance
570                // We do not need to withdraw here unless necessary, which will be done during
571                // gas charging
572                let deposit = total_smashed
573                    .checked_sub(*reservation)
574                    .expect("the total smashed includes the target's reservation");
575                if deposit != 0 {
576                    let balance_type =
577                        sui_types::balance::Balance::type_tag(sui_types::gas_coin::GAS::type_tag());
578                    let event = AccumulatorEvent::from_balance_change(
579                        *sui_address,
580                        balance_type,
581                        i64::try_from(deposit).unwrap(),
582                    )
583                    .expect("Failed to create accumulator event for gas smashing");
584                    temporary_store.add_accumulator_event(event);
585                }
586            }
587            PaymentMethod::Coin((gas_coin_id, _, _)) => {
588                let mut primary_gas_object = temporary_store
589                    .objects()
590                    .get(gas_coin_id)
591                    // unwrap should be safe because we checked that this exists in `self.objects()` above
592                    .unwrap_or_else(|| {
593                        panic!(
594                            "Invariant violation: gas coin not found in store in txn {}",
595                            tx_digest
596                        )
597                    })
598                    .clone();
599                primary_gas_object
600                    .data
601                    .try_as_move_mut()
602                    // unwrap should be safe because we checked that the primary gas object was a coin object above.
603                    .unwrap_or_else(|| {
604                        panic!(
605                            "Invariant violation: invalid coin object in txn {}",
606                            tx_digest
607                        )
608                    })
609                    .set_coin_value_unsafe(total_smashed);
610                temporary_store.mutate_input_object(primary_gas_object);
611            }
612        }
613    }
614
615    // Apply the net charge or refund to the gas coin, or to the address balance through an
616    // accumulator event.
617    fn apply(
618        &self,
619        temporary_store: &mut TemporaryStore<'_>,
620        cost_summary: GasCostSummary,
621    ) -> GasCostSummary {
622        let net_change = cost_summary.net_gas_usage();
623        match self.charge_location() {
624            PaymentLocation::AddressBalance(payer_address) => {
625                if net_change != 0 {
626                    let balance_type =
627                        sui_types::balance::Balance::type_tag(sui_types::gas_coin::GAS::type_tag());
628                    let event = AccumulatorEvent::from_balance_change(
629                        payer_address,
630                        balance_type,
631                        net_change
632                            .checked_neg()
633                            .expect("net gas usage is never i64::MIN"),
634                    )
635                    .expect("Failed to create accumulator event for gas charging");
636                    temporary_store.add_accumulator_event(event);
637                }
638            }
639            PaymentLocation::Coin(gas_object_id) => {
640                let mut gas_object = temporary_store
641                    .read_object(&gas_object_id)
642                    .expect("gas coin is an input object and present after smashing")
643                    .clone();
644                deduct_gas(&mut gas_object, net_change);
645                trace!(net_change, gas_obj_id =? gas_object.id(), gas_obj_ver =? gas_object.version(), "Updated gas object");
646                temporary_store.mutate_new_or_input_object(gas_object);
647            }
648        }
649        cost_summary
650    }
651
652    fn used_coins(&self) -> impl Iterator<Item = &'_ ObjectRef> {
653        self.payment_methods().filter_map(|method| match method {
654            PaymentMethod::Coin(obj_ref) => Some(obj_ref),
655            PaymentMethod::AddressBalance(_, _) => None,
656        })
657    }
658}

660impl PaymentMethod {
661    // The payments as signed: each gas object is a coin, or an address balance reservation
662    // when its digest encodes one. With no gas objects the budget is reserved on the owner.
663    fn from_gas_data(gas_data: &GasData) -> Vec<Self> {
664        if gas_data.payment.is_empty() {
665            return vec![PaymentMethod::AddressBalance(
666                gas_data.owner,
667                gas_data.budget,
668            )];
669        }
670        gas_data
671            .payment
672            .iter()
673            .map(|entry| match ParsedDigest::try_from(entry.2) {
674                Ok(parsed) => {
675                    PaymentMethod::AddressBalance(gas_data.owner, parsed.reservation_amount())
676                }
677                Err(_) => PaymentMethod::Coin(*entry),
678            })
679            .collect()
680    }
681
682    fn location(&self) -> PaymentLocation {
683        match self {
684            PaymentMethod::Coin(obj_ref) => PaymentLocation::Coin(obj_ref.0),
685            PaymentMethod::AddressBalance(addr, _) => PaymentLocation::AddressBalance(*addr),
686        }
687    }
