21// MIST amounts, and prices in MIST per gas unit.
22type Mist = u64;
23// Gas units, as the transaction is charged.
24type GasUnits = u64;
25// Internal gas units, as the Move gas status counts them.
26type InternalUnits = u64;
27
28// How a transaction is metered
29#[derive(Debug, Clone, Copy)]
30pub(crate) enum Metering {
31    Unmetered,
32    Metered {
33        budget: Mist,
34        price: Mist,
35        reference_gas_price: Mist,
36    },
37}
38
39impl Metering {
40    fn is_unmetered(&self) -> bool {
41        matches!(self, Metering::Unmetered)
42    }
43
44    fn budget(&self) -> Mist {
45        match self {
46            Metering::Unmetered => 0,
47            Metering::Metered { budget, .. } => *budget,
48        }
49    }
50
51    fn price(&self) -> Mist {
52        match self {
53            Metering::Unmetered => 0,
54            Metering::Metered { price, .. } => *price,
55        }
56    }
57
58    fn reference_gas_price(&self) -> Mist {
59        match self {
60            Metering::Unmetered => 0,
61            Metering::Metered {
62                reference_gas_price,
63                ..
64            } => *reference_gas_price,
65        }
66    }
67}

69// Holds protocol config values and configuration/cost data for gas charging
70#[derive(Debug)]
71struct GasConfig {
72    gas_model_version: u64,
73    cost_table: CostTable,
74    // Internal gas units per byte of input object read.
75    read_cost_per_byte: InternalUnits,
76    // Internal gas units per KiB of package read. Packages use `read_cost_per_byte` when
77    // unset.
78    package_read_per_kb_cost: Option<InternalUnits>,
79    // Internal gas units per byte of published module.
80    publish_cost_per_byte: InternalUnits,
81    // Cap, in gas units, on the execution meter budget.
82    max_computation_bucket: GasUnits,
83    // Minimum computation charge in gas units: the base transaction cost.
84    min_computation_units: GasUnits,
85    storage_gas_price: Mist,
86    storage_per_byte_cost: GasUnits,
87    rebate_rate: u64,
88    // RGP multiplier capping the effective gas price of aborted transactions.
89    abort_price_factor: Option<u64>,
90}

99            publish_cost_per_byte: config.package_publish_cost_per_byte(),
100            max_computation_bucket: config.max_gas_computation_bucket(),
101            min_computation_units: config.base_tx_cost_fixed(),

107        }
108    }

111// Gas state of a transaction, in MIST. The phases are pre-execution, execution (input
112// objects and the PTB) and storage. Each phase settles once against the budget in MIST.
113// A computation phase that does not fit marks the state out of gas. Storage that does not
114// fit leaves the state unchanged.
115#[derive(Debug)]
116pub(crate) struct GasState {
117    metering: Metering,
118    config: GasConfig,
119    // MIST owed before the VM runs.
120    pre_execution: Mist,
121    computation: Computation,
122    storage: StorageCharge,
123    // A computation check failed, or storage settled at `RebatesOnly`.
124    out_of_gas: bool,
125}
126
127// The units the execution meter used, input objects included, and the price fixed when
128// execution settles.
129#[derive(Debug, Default)]
130struct Computation {
131    units: InternalUnits,
132    settled_price: Option<Mist>,
133}
134
135impl Computation {
136    fn mist(&self, price: Mist, min_units: GasUnits) -> Mist {
137        let units = half_digits_rounding(to_gas_units(self.units)).max(min_units);
138        units.saturating_mul(price)
139    }
140}

142// What storage charges for. A successful transaction settles AllWrites. After an error it
143// settles InputsOnly, or RebatesOnly when the inputs do not fit.
144#[derive(Debug, Clone, Copy, PartialEq, Eq)]
145pub(crate) enum StorageTier {
146    // Every object the transaction wrote or deleted.
147    AllWrites,
148    // The mutable inputs, with their versions bumped, and the smashed gas coins.
149    InputsOnly,
150    // Only the rebates of the smashed gas coins. The whole budget goes to computation.
151    RebatesOnly,
152}

154impl GasState {
155    pub fn new(metering: Metering, config: &ProtocolConfig) -> Self {
156        Self {
157            metering,
158            config: GasConfig::new(config),
159            pre_execution: 0,
160            computation: Computation::default(),
161            storage: StorageCharge::default(),
162            out_of_gas: false,
163        }
164    }
165
166    pub fn is_unmetered(&self) -> bool {
167        self.metering.is_unmetered()
168    }
169
170    pub fn budget(&self) -> Mist {
171        self.metering.budget()
172    }
173
174    pub fn price(&self) -> Mist {
175        self.metering.price()
176    }
177
178    pub fn reference_gas_price(&self) -> Mist {
179        self.metering.reference_gas_price()
180    }
181
182    // MIST charged so far: pre-execution plus computation. Computation counts only once
183    // execution settles.
184    fn charged(&self) -> Mist {
185        let computation = self.computation.settled_price.map_or(0, |price| {
186            self.computation
187                .mist(price, self.config.min_computation_units)
188        });
189        self.pre_execution.saturating_add(computation)
190    }
191
192    // First phase, just the budget
193    fn pre_execution_limit(&self) -> Mist {
194        self.budget()
195    }
196
197    // budget - pre_execution
198    fn execution_limit(&self) -> Mist {
199        self.budget().saturating_sub(self.pre_execution)
200    }
201
202    // budget - computation - pre_execution
203    fn storage_limit(&self) -> Mist {
204        self.budget().saturating_sub(self.charged())
205    }
206
207    // `charge` must stay below `limit`, else out of gas.
208    fn check_below_limit(&mut self, charge: Mist, limit: Mist) -> Result<(), ExecutionError> {
209        if !self.is_unmetered() && charge >= limit {
210            self.out_of_gas = true;
211            return Err(ExecutionErrorKind::InsufficientGas.into());
212        }
213        Ok(())
214    }
215
216    // Pre execution phase.
217
218    pub fn settle_pre_execution(&mut self, mist: Mist) -> Result<(), ExecutionError> {
219        self.pre_execution = mist;
220        self.check_below_limit(self.pre_execution, self.pre_execution_limit())
221    }
222
223    // Execution phase.
224
225    // Open the execution phase. The meter budget is the execution limit, capped by the
226    // computation bucket.
227    pub fn execution_meter(&mut self) -> Result<ExecutionMeter, ExecutionError> {
228        assert_invariant!(
229            self.computation.settled_price.is_none(),
230            "the execution phase opens before it settles"
231        );
232        let status = match self.metering {
233            Metering::Metered { price, .. } => {
234                let cap = self.config.max_computation_bucket.saturating_mul(price);
235                GasStatus::new(
236                    self.config.cost_table.clone(),
237                    self.execution_limit().min(cap),
238                    price,
239                    self.config.gas_model_version,
240                )
241            }
242            Metering::Unmetered => GasStatus::new_unmetered(),
243        };
244        Ok(ExecutionMeter::new(
245            status,
246            InputRates {
247                cost_per_byte: self.config.read_cost_per_byte,
248                package_cost_per_kb: self.config.package_read_per_kb_cost,
249            },
250            self.config.publish_cost_per_byte,
251        ))
252    }
253
254    // Close the execution phase at the signed price, or at the capped price on abort.
255    pub fn settle_execution(
256        &mut self,
257        meter: ExecutionMeter,
258        aborted: bool,
259    ) -> Result<(), ExecutionError> {
260        assert_invariant!(
261            self.computation.settled_price.is_none(),
262            "the execution phase settles once"
263        );
264        let price = match (self.metering, self.config.abort_price_factor) {
265            (Metering::Unmetered, _) => 0,
266            (
267                Metering::Metered {
268                    price,
269                    reference_gas_price,
270                    ..
271                },
272                Some(factor),
273            ) if aborted => price.min(factor.saturating_mul(reference_gas_price)),
274            (Metering::Metered { price, .. }, _) => price,
275        };
276        self.computation.units = meter.used_internal_units();
277        self.computation.settled_price = Some(price);
278        let computation = self
279            .computation
280            .mist(price, self.config.min_computation_units);
281        self.check_below_limit(computation, self.execution_limit())
282    }
283
284    // Storage phase.
285
286    pub fn storage_rates(&self) -> StorageRates {
287        if self.is_unmetered() {
288            StorageRates::unmetered()
289        } else {
290            StorageRates {
291                price: self.config.storage_gas_price,
292                per_byte_cost: self.config.storage_per_byte_cost,
293                metered: true,
294            }
295        }
296    }
297
298    // AllWrites and InputsOnly must fit net of the sender's rebate, and leave the state
299    // unchanged when they do not. RebatesOnly always settles.
300    pub fn settle_storage(
301        &mut self,
302        charge: StorageCharge,
303        tier: StorageTier,
304    ) -> Result<(), ExecutionError> {
305        match tier {
306            StorageTier::RebatesOnly => self.out_of_gas = true,
307            StorageTier::AllWrites | StorageTier::InputsOnly => {
308                let net_cost = charge
309                    .cost
310                    .saturating_sub(sender_rebate(charge.rebate, self.config.rebate_rate));
311                if !self.is_unmetered() && net_cost > self.storage_limit() {
312                    return Err(ExecutionErrorKind::InsufficientGas.into());
313                }
314            }
315        }
316        self.storage = charge;
317        Ok(())
318    }
319
320    // Reporting.
321
322    // Computation is the whole budget once out of gas. Storage settled only if it fit, so
323    // the net charge never exceeds the budget.
324    pub fn summary(&self) -> GasCostSummary {
325        let storage_rebate = self.storage.rebate;
326        let sender_rebate = sender_rebate(storage_rebate, self.config.rebate_rate);
327        let non_refundable_storage_fee = storage_rebate
328            .checked_sub(sender_rebate)
329            .expect("sender rebate must not exceed storage rebate");
330        let storage_cost = self.storage.cost;
331        let computation_cost = if self.out_of_gas {
332            self.budget()
333        } else {
334            self.charged()
335        };
336        debug_assert!(
337            self.out_of_gas
338                || self.is_unmetered()
339                || computation_cost.saturating_add(storage_cost.saturating_sub(sender_rebate))
340                    <= self.budget(),
341            "storage must fit in what computation left"
342        );
343        GasCostSummary {
344            computation_cost,
345            storage_cost,
346            storage_rebate: sender_rebate,
347            non_refundable_storage_fee,
348        }
349    }

351    pub fn gas_usage_report(&self) -> GasUsageReport {
352        GasUsageReport {
353            cost_summary: self.summary(),
354            gas_used: to_gas_units(self.computation.units),
355            gas_budget: self.budget(),
356            gas_price: self.price(),
357            reference_gas_price: self.reference_gas_price(),
358            storage_gas_price: if self.is_unmetered() {
359                0
360            } else {
361                self.config.storage_gas_price
362            },
363            rebate_rate: self.config.rebate_rate,
364            per_object_storage: self.storage.per_object.clone(),
365        }
366    }

370    }

373// Rounds down, as the Move meter does.
374fn to_gas_units(internal_units: InternalUnits) -> GasUnits {
375    let units: Gas = InternalGas::new(internal_units).to_unit_round_down();
376    u64::from(units)
377}
378
379// Round up to half the digits: 1234 becomes 1240, 123456 becomes 123500.
380fn half_digits_rounding(n: u64) -> u64 {
381    if n < 10 {
382        return n;
383    }
384    let digits = n.ilog10();
385    let drop = digits / 2;
386    let base = 10u64.pow(drop);
387    n.div_ceil(base).saturating_mul(base)
388}

390// Rates of the input objects, in internal gas units.
391#[derive(Debug, Clone, Copy)]
392struct InputRates {
393    cost_per_byte: InternalUnits,
394    package_cost_per_kb: Option<InternalUnits>,
395}
396
397impl InputRates {
398    fn units(&self, size: usize, is_package: bool) -> InternalUnits {
399        match (is_package, self.package_cost_per_kb) {
400            (true, Some(cost_per_kb)) => package_read_internal_gas(size, cost_per_kb),
401            _ => (size as u64).saturating_mul(self.cost_per_byte),
402        }
403    }
404}

406// The meter of the execution phase: the Move gas status, with byte charges for input objects
407// and the PTB.
408#[derive(Debug)]
409pub(crate) struct ExecutionMeter {
410    status: GasStatus,
411    initial_internal_units: InternalUnits,
412    input_rates: InputRates,
413    publish_cost_per_byte: InternalUnits,
414}
415
416impl ExecutionMeter {
417    fn new(
418        status: GasStatus,
419        input_rates: InputRates,
420        publish_cost_per_byte: InternalUnits,
421    ) -> Self {
422        // A fresh meter has its whole budget left.
423        let initial_internal_units = u64::from(status.gas_left);
424        Self {
425            status,
426            initial_internal_units,
427            input_rates,
428            publish_cost_per_byte,
429        }
430    }
431
432    // A meter that charges nothing, for genesis.
433    pub fn unmetered(config: &ProtocolConfig) -> Self {
434        Self::new(
435            GasStatus::new_unmetered(),
436            InputRates {
437                cost_per_byte: config.obj_access_cost_read_per_byte(),
438                package_cost_per_kb: config.obj_access_cost_read_per_package_kb_as_option(),
439            },
440            config.package_publish_cost_per_byte(),
441        )
442    }
443
444    pub fn status(&self) -> &GasStatus {
445        &self.status
446    }
447
448    pub fn status_mut(&mut self) -> &mut GasStatus {
449        &mut self.status
450    }
451
452    fn used_internal_units(&self) -> InternalUnits {
453        self.initial_internal_units
454            .saturating_sub(u64::from(self.status.gas_left))
455    }
456
457    pub fn charge_input_object(
458        &mut self,
459        size: usize,
460        is_package: bool,
461    ) -> Result<(), ExecutionError> {
462        let units = self.input_rates.units(size, is_package);
463        self.status
464            .deduct_gas(InternalGas::new(units))
465            .map_err(out_of_gas)
466    }
467
468    pub fn charge_publish_package(&mut self, size: usize) -> Result<(), ExecutionError> {
469        self.charge_bytes(size, self.publish_cost_per_byte)
470    }
471
472    pub fn charge_read_bytes(&mut self, size: usize) -> Result<(), ExecutionError> {
473        self.charge_bytes(size, self.input_rates.cost_per_byte)
474    }
475
476    fn charge_bytes(
477        &mut self,
478        size: usize,
479        cost_per_byte: InternalUnits,
480    ) -> Result<(), ExecutionError> {
481        self.status
482            .charge_bytes(size, cost_per_byte)
483            .map_err(out_of_gas)
484    }
485}
486
487fn out_of_gas(e: PartialVMError) -> ExecutionError {
488    debug_assert_eq!(e.major_status(), StatusCode::OUT_OF_GAS);
489    ExecutionErrorKind::InsufficientGas.into()
490}

510// Storage cost and rebate of a write set, with the per-object breakdown. The store builds
511// it and the state settles it. `unmetered_rebate` is the rebate of an unmetered
512// transaction, which the charger moves to the system state object.
513#[derive(Debug, Default)]
514pub(crate) struct StorageCharge {
515    cost: Mist,
516    rebate: Mist,
517    unmetered_rebate: Mist,
518    per_object: Vec<(ObjectID, PerObjectStorage)>,
519}
520
521impl StorageCharge {
522    // Track a written object, or a deleted one with `new_size` zero. Returns the new storage
523    // rebate, or `None` on overflow.
524    pub fn track_mutation(
525        &mut self,
526        rates: &StorageRates,
527        object_id: ObjectID,
528        new_size: usize,
529        storage_rebate: Mist,
530    ) -> Option<Mist> {
531        if !rates.metered {
532            self.unmetered_rebate = self.unmetered_rebate.checked_add(storage_rebate)?;
533            return Some(0);
534        }
535        let new_size = new_size as u64;
536        let storage_cost = new_size
537            .checked_mul(rates.per_byte_cost)?
538            .checked_mul(rates.price)?;
539        self.cost = self.cost.checked_add(storage_cost)?;
540        self.rebate = self.rebate.checked_add(storage_rebate)?;
541        self.per_object.push((
542            object_id,
543            PerObjectStorage {
544                storage_cost,
545                storage_rebate,
546                new_size,
547            },
548        ));
549        Some(storage_cost)
550    }
