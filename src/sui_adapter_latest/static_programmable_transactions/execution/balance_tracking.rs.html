16/// What a user PTB read and wrote, from which its `BalanceTracking` is built.
17pub(crate) struct ObjectData<'a> {
18    /// The sender, the sponsor, and each declared withdrawal's funder and allowance ID, whether or
19    /// not the withdrawal is redeemed.
20    pub transaction_addresses: Vec<SuiAddress>,
21    /// Object inputs, including the gas coin, with their owners at the start of the transaction.
22    pub inputs: Vec<(ObjectID, Owner)>,
23    /// Objects loaded during execution: mutable inputs, dynamic field children, and received
24    /// objects.
25    pub loaded_runtime_objects: BTreeSet<ObjectID>,
26    /// UIDs found in the objects read from storage.
27    pub contained_uids: BTreeSet<ObjectID>,
28    /// Includes IDs created and then deleted in the transaction.
29    pub generated_ids: BTreeSet<ObjectID>,
30    pub results: &'a ExecutionResultsV2,
31}
32
33/// Fails if an object is written or deleted without being read or created, since its UID was then
34/// missed when walking the objects read.
35pub(crate) fn finish(
36    before: BalanceTotals,
37    after: BalanceTotals,
38    objects: ObjectData<'_>,
39) -> Result<BalanceTracking, ExecutionError> {
40    let ObjectData {
41        transaction_addresses,
42        inputs,
43        loaded_runtime_objects,
44        contained_uids,
45        generated_ids,
46        results,
47    } = objects;
48    let mut seen: BTreeSet<ObjectID> = inputs.iter().map(|(id, _)| *id).collect();
49    seen.extend(
50        contained_uids
51            .iter()
52            .chain(&loaded_runtime_objects)
53            .chain(&generated_ids),
54    );
55    let unseen = results
56        .written_objects
57        .iter()
58        .filter(|(_, object)| !object.is_package())
59        .map(|(id, _)| id)
60        .chain(&results.deleted_object_ids)
61        .find(|id| !seen.contains(id));
62    if let Some(id) = unseen {
63        return Err(make_invariant_violation!(
64            "Object {id} was written or deleted without being read or created"
65        ));
66    }
67    let mut addresses: BTreeSet<SuiAddress> = transaction_addresses.into_iter().collect();
68    let mut packages = BTreeSet::new();
69    for (id, owner) in &inputs {
70        addresses.insert((*id).into());
71        add_owner_addresses(&mut addresses, owner);
72    }
73    for (id, object) in &results.written_objects {
74        if object.is_package() {
75            packages.insert(SuiAddress::from(*id));
76        }
77        addresses.insert((*id).into());
78        add_owner_addresses(&mut addresses, &object.owner);
79    }
80    let ids = contained_uids
81        .iter()
82        .chain(&generated_ids)
83        .chain(&loaded_runtime_objects)
84        .chain(&results.deleted_object_ids);
85    addresses.extend(ids.map(|id| SuiAddress::from(*id)));
86    let mut tracking = BalanceTracking {
87        before,
88        after,
89        addresses,
90    };
91    add_accumulator_events(&mut tracking, &results.accumulator_events)?;
92    // Freezing a package or a system object blocks nothing.
93    tracking
94        .addresses
95        .retain(|a| !packages.contains(a) && !ObjectID::from(*a).is_system_object());
96    Ok(tracking)
97}

99/// Adds each raw `Balance<T>` Split to `before` and each raw Merge to `after`, and records every
100/// event's address as involved.
101///
102/// The events must not be merged yet, so a withdraw-and-redeposit counts on both sides.
103pub(crate) fn add_accumulator_events<'a>(
104    tracking: &mut BalanceTracking,
105    events: impl IntoIterator<Item = &'a AccumulatorEvent>,
106) -> Result<(), ExecutionError> {
107    for event in events {
108        let address = event.write.address.address;
109        tracking.addresses.insert(address);
110        let Some(type_) = Balance::maybe_get_balance_type_param(&event.write.address.ty) else {
111            continue;
112        };
113        let AccumulatorValue::Integer(amount) = event.write.value else {
114            continue;
115        };
116        let totals = match event.write.operation {
117            AccumulatorOperation::Split => &mut tracking.before,
118            AccumulatorOperation::Merge => &mut tracking.after,
119        };
120        add_to_totals(totals, address, &Rc::new(type_), u128::from(amount))
121            .map_err(ExecutionError::invariant_violation)?;
122    }
123    Ok(())
124}

126/// Adds the addresses that own an object: a parent object's ID, or every member of a party.
127fn add_owner_addresses(addresses: &mut BTreeSet<SuiAddress>, owner: &Owner) {
128    match owner {
129        Owner::AddressOwner(address)
130        | Owner::ObjectOwner(address)
131        | Owner::ConsensusAddressOwner { owner: address, .. } => {
132            addresses.insert(*address);
133        }
134        Owner::Party { permissions, .. } => addresses.extend(permissions.members()),
135        Owner::Shared { .. } | Owner::Immutable => (),
136    }
137}
