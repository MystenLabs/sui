4use crate::static_programmable_transactions::{
5    linkage::resolved_linkage::{ExecutableLinkage, ResolvedLinkage},
6    loading::ast::PackagePayload,
7};
8use move_binary_format::file_format::Visibility;
9use sui_types::{
10    Identifier, TypeTag,
11    base_types::ObjectID,
12    transaction::{Argument, CallArg},
13};
14
15/// Names are defining-ID based, but type instantiations have not yet been validated by the VM.
16#[derive(Debug)]
17pub struct Transaction {
18    pub inputs: Vec<CallArg>,
19    pub commands: Vec<Command>,
20    pub unified_linkage: ExecutableLinkage,
21}
22
23#[derive(Debug)]
24pub struct MoveCall {
25    pub package: ObjectID,
26    pub module: Identifier,
27    pub function: Identifier,
28    pub visibility: Visibility,
29    pub type_arguments: Vec<TypeTag>,
30    pub arguments: Vec<Argument>,
31}
32
33#[derive(Debug)]
34pub enum Command {
35    MoveCall(Box<MoveCall>),
36    TransferObjects(Vec<Argument>, Argument),
37    SplitCoins(Argument, Vec<Argument>),
38    MergeCoins(Argument, Vec<Argument>),
39    MakeMoveVec(Option<TypeTag>, Vec<Argument>),
40    Publish(PackagePayload, Vec<ObjectID>, ResolvedLinkage),
41    Upgrade(
42        PackagePayload,
43        Vec<ObjectID>,
44        ObjectID,
45        Argument,
46        ResolvedLinkage,
47    ),
48}
