16/// Tracks the references live at any point in the transaction, and the total returned
17struct Context<'pc> {
18    protocol_config: &'pc ProtocolConfig,
19    live: u64,
20    total_returned: u64,
21}
22
23/// Charges for the references live at each command, and checks three limits:
24/// - The number of references live at any point
25/// - The number of references returned by any command
26/// - The number of references returned over the whole transaction
27pub fn meter<E: ExecutionErrorTrait>(
28    meter: &mut TranslationMeter,
29    protocol_config: &ProtocolConfig,
30    transaction: &T::Transaction,
31) -> Result<(), E> {
32    let mut context = Context {
33        protocol_config,
34        live: 0,
35        total_returned: 0,
36    };
37    for (idx, c) in transaction.commands.iter().enumerate() {
38        command::<E>(&mut context, meter, c).map_err(|e| e.with_command_index(idx))?;
39    }
40    Ok(())
41}
42
43fn command<E: ExecutionErrorTrait>(
44    context: &mut Context,
45    meter: &mut TranslationMeter,
46    sp!(_, c): &T::Command,
47) -> Result<(), E> {
48    // The references created by the arguments are held by the command until the return values
49    // are fully processed
50    let held = arguments::<E>(context, c.command.arguments())?;
51
52    let returned = c.result_type.iter().filter(|ty| ty.is_reference()).count() as u64;
53    context.returned_n::<E>(returned)?;
54    // The references held for the arguments are consumed once the command has its return values
55    context.free_n::<E>(held)?;
56
57    // Unused results are dropped at the end of the command
58    let dropped = {
59        assert_invariant!(
60            c.drop_values.len() == c.result_type.len(),
61            "command drop_values length does not match result_type length"
62        );
63        c.drop_values
64            .iter()
65            .zip_debug_eq(&c.result_type)
66            .filter(|(drop, ty)| **drop && ty.is_reference())
67            .count() as u64
68    };
69    context.free_n::<E>(dropped)?;
70    meter.charge_num_live_references(context.live)
71}

73/// Returns the number of references for the arguments, excluding the TxContext.
74/// For each argument, appropriately tracks the creation/freeing of references
75fn arguments<'a, E: ExecutionErrorTrait>(
76    context: &mut Context,
77    args: impl IntoIterator<Item = &'a T::Argument>,
78) -> Result<u64, E> {
79    args.into_iter().try_fold(0u64, |held, arg| {
80        let is_ref = argument::<E>(context, arg)?;
81        let incr = if is_ref { 1 } else { 0 };
82        Ok(held.saturating_add(incr))
83    })
84}

86/// Returns true iff the argument is a reference type, excluding the TxContext.
87/// Counts the creation/freeing of references for the argument and locations/usage
88fn argument<E: ExecutionErrorTrait>(
89    context: &mut Context,
90    sp!(_, (arg, ty)): &T::Argument,
91) -> Result</* is ref */ bool, E> {
92    if ty.is_tx_context() != TxContextKind::None {
93        // TxContext is excluded from reference safety
94        return Ok(false);
95    }
96    Ok(match arg {
97        T::Argument__::Borrow(_, _location) => {
98            context.create::<E>()?;
99            true
100        }
101
102        T::Argument__::Freeze(u) => {
103            let usage_is_ref = usage(context, u, ty)?;
104            assert_invariant!(usage_is_ref, "freeze argument must be a reference type");
105            context.free::<E>()?;
106            context.create::<E>()?;
107            true
108        }
109        T::Argument__::Read(u) => {
110            let usage_is_ref = usage::<E>(context, u, ty)?;
111            assert_invariant!(usage_is_ref, "read argument must be a reference type");
112            context.free::<E>()?;
113            false
114        }
115        T::Argument__::Use(u) => usage::<E>(context, u, ty)?,
116    })
117}

119/// Returns true iff the argument is a reference type
120/// Counts the creation of a new reference for `Copy`
121fn usage<E: ExecutionErrorTrait>(
122    context: &mut Context,
123    u: &T::Usage,
124    ty: &T::Type,
125) -> Result<bool, E> {
126    if !ty.is_reference() {
127        return Ok(false);
128    }
129    match u {
130        T::Usage::Move(_) => Ok(true),
131        T::Usage::Copy { .. } => {
132            context.create::<E>()?;
133            Ok(true)
134        }
135    }
136}

138impl Context<'_> {
139    fn create<E: ExecutionErrorTrait>(&mut self) -> Result<(), E> {
140        self.create_n::<E>(1)
141    }
142
143    fn create_n<E: ExecutionErrorTrait>(&mut self, n: u64) -> Result<(), E> {
144        self.live = self.live.saturating_add(n);
145        let max_live = self.protocol_config.max_ptb_live_references();
146        if self.live > max_live {
147            return Err(E::new_with_source(
148                // TODO introduce an ExecutionErrorKind for limits
149                ExecutionErrorKind::InsufficientGas,
150                format!(
151                    "Command has {} live references, exceeding the maximum of {max_live}",
152                    self.live
153                ),
154            ));
155        }
156        Ok(())
157    }
158
159    /// Creates the `n` references returned by a command, checking both the per-command limit and
160    /// the transaction wide total. The total bounds the length of any chain of returned
161    /// references, which other analyses are sensitive to beyond the number live at once.
162    fn returned_n<E: ExecutionErrorTrait>(&mut self, n: u64) -> Result<(), E> {
163        let max_per_command = self.protocol_config.max_ptb_returned_references();
164        if n > max_per_command {
165            return Err(E::new_with_source(
166                // TODO introduce an ExecutionErrorKind for limits
167                ExecutionErrorKind::InsufficientGas,
168                format!(
169                    "Command returns {n} references, exceeding the maximum of {max_per_command}"
170                ),
171            ));
172        }
173        self.total_returned = self.total_returned.saturating_add(n);
174        let max_total = self.protocol_config.max_ptb_total_returned_references();
175        if self.total_returned > max_total {
176            return Err(E::new_with_source(
177                // TODO introduce an ExecutionErrorKind for limits
178                ExecutionErrorKind::InsufficientGas,
179                format!(
180                    "Transaction returns {} references, exceeding the maximum of {max_total}",
181                    self.total_returned
182                ),
183            ));
184        }
185        self.create_n::<E>(n)
186    }
187
188    fn free<E: ExecutionErrorTrait>(&mut self) -> Result<(), E> {
189        self.free_n::<E>(1)
190    }
191
192    fn free_n<E: ExecutionErrorTrait>(&mut self, n: u64) -> Result<(), E> {
193        let Some(rem) = self.live.checked_sub(n) else {
194            invariant_violation!("freeing {n} reference(s) when {} are live", self.live)
195        };
196        self.live = rem;
197        Ok(())
198    }
199}
