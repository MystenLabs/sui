21/// A packed set of small indices. Index `n` is stored in word `n / 64` at bit `n % 64`. Used both
22/// for node-id ancestor sets and, in `Memory::parent_commands`, for command-id sets.
23#[derive(Debug, Clone, Default)]
24struct BitSet {
25    words: Vec<u64>,
26}
27
28#[derive(Debug)]
29enum NodeKind {
30    Root,
31    Delta { command: u16 },
32}
33
34#[derive(Debug)]
35struct Node {
36    kind: NodeKind,
37    /// Contains the node itself and all of its transitive ancestors
38    ancestors: BitSet,
39    children: Vec<NodeID>,
40}

42#[derive(Debug)]
43struct Memory {
44    nodes: Vec<Node>,
45    roots: BTreeMap<RootLocation, NodeID>,
46    /// Commands that created delta children of each parent. Used to detect when a parent
47    /// has children from more than one command.
48    parent_commands: IndexMap<NodeID, BitSet>,
49    /// Parents with delta children from multiple commands. Only these parents can
50    /// cause cross-command overlap.
51    conflict_parents: IndexSet<NodeID>,
52}

54#[derive(Debug)]
55enum Value {
56    NonRef,
57    Ref { is_mut: bool, node: NodeID },
58}

60#[derive(Debug)]
61struct Location {
62    /// The location represented by this root. Its graph node is created on first borrow,
63    /// so values that are never borrowed need no node.
64    root: RootLocation,
65    value: Option<Value>,

86impl BitSet {
87    /// Creates an empty set with capacity for `bits` indices.
88    fn with_bits(bits: usize) -> Self {
89        Self {
90            words: vec![0; bits.div_ceil(64).max(1)],
91        }
92    }
93
94    /// Sets the bit and returns whether it was already set. Errors if it is out of bounds.
95    fn set(&mut self, bit: usize) -> anyhow::Result<bool> {
96        let word = self
97            .words
98            .get_mut(bit / 64)
99            .ok_or_else(|| anyhow::anyhow!("BitSet index {bit} is out of bounds"))?;
100        let mask = 1 << (bit % 64);
101        let was_set = *word & mask != 0;
102        *word |= mask;
103        Ok(was_set)
104    }
105
106    /// Sets the bit, growing the allocation to fit.
107    fn set_grow(&mut self, bit: usize) {
108        let word_index = bit / 64;
109        if word_index >= self.words.len() {
110            self.words.resize(word_index.saturating_add(1), 0);
111        }
112        if let Some(word) = self.words.get_mut(word_index) {
113            *word |= 1 << (bit % 64);
114        }
115    }
116
117    /// Out-of-range bits are treated as unset.
118    fn contains(&self, bit: usize) -> bool {
119        self.words
120            .get(bit / 64)
121            .is_some_and(|word| word & (1 << (bit % 64)) != 0)
122    }
123
124    fn count(&self) -> u32 {
125        self.words.iter().map(|word| word.count_ones()).sum()
126    }
127
128    /// Adds all bits from `other`, growing the allocation if needed.
129    fn union(&mut self, other: &Self) {
130        if self.words.len() < other.words.len() {
131            self.words.resize(other.words.len(), 0);
132        }
133        // Extra words in `self` are unchanged, so zip may stop at the end of `other`.
134        #[allow(clippy::disallowed_methods)]
135        let words = self.words.iter_mut().zip(&other.words);
136        for (word, other_word) in words {
137            *word |= other_word;
138        }
139    }
140
141    /// Tests shared indices until `f` returns true or an error.
142    fn try_any_common(
143        &self,
144        other: &Self,
145        mut f: impl FnMut(usize) -> anyhow::Result<bool>,
146    ) -> anyhow::Result<bool> {
147        // Missing words are zero, so the intersection ends with the shorter set.
148        #[allow(clippy::disallowed_methods)]
149        let words = self.words.iter().zip(&other.words);
150        for (word_index, (left, right)) in words.enumerate() {
151            let mut common = left & right;
152            while common != 0 {
153                if f(Self::bit_index(word_index, common)?)? {
154                    return Ok(true);
155                }
156                common = Self::clear_lowest_bit(common)?;
157            }
158        }
159        Ok(false)
160    }
161
162    /// Returns the absolute index of the lowest set bit of `word`, which lives in `word_index`.
163    fn bit_index(word_index: usize, word: u64) -> anyhow::Result<usize> {
164        debug_assert!(word != 0);
165        word_index
166            .checked_mul(64)
167            .and_then(|base| base.checked_add(word.trailing_zeros() as usize))
168            .ok_or_else(|| anyhow::anyhow!("BitSet index overflow for word {word_index}"))
169    }
170
171    /// Removes the lowest set bit of `word`, which must be non-zero.
172    fn clear_lowest_bit(word: u64) -> anyhow::Result<u64> {
173        let sub = word
174            .checked_sub(1)
175            .ok_or_else(|| anyhow::anyhow!("clear_lowest_bit called on 0"))?;
176        Ok(word & sub)
177    }
178}

180impl Memory {
181    fn new() -> Self {
182        Self {
183            nodes: vec![],
184            roots: BTreeMap::new(),
185            parent_commands: IndexMap::new(),
186            conflict_parents: IndexSet::new(),
187        }
188    }
189
190    fn node(&self, id: NodeID) -> anyhow::Result<&Node> {
191        self.nodes
192            .get(id)
193            .ok_or_else(|| anyhow::anyhow!("Node index {id} is out of bounds"))
194    }
195
196    fn node_mut(&mut self, id: NodeID) -> anyhow::Result<&mut Node> {
197        self.nodes
198            .get_mut(id)
199            .ok_or_else(|| anyhow::anyhow!("Node index {id} is out of bounds"))
200    }
201
202    /// Creates a node whose ancestor set is itself plus every ancestor of its parents, and records
203    /// reverse parent-to-child edges for identifying extensions from distinct commands.
204    fn new_node(&mut self, kind: NodeKind, parents: &[NodeID]) -> anyhow::Result<NodeID> {
205        let id = self.nodes.len();
206        let bits = id
207            .checked_add(1)
208            .ok_or_else(|| anyhow::anyhow!("Node ID overflow"))?;
209        let mut ancestors = BitSet::with_bits(bits);
210        anyhow::ensure!(
211            !ancestors.set(id)?,
212            "Node {id} is already set in its fresh ancestor set"
213        );
214        for &parent in parents {
215            ancestors.union(&self.node(parent)?.ancestors);
216        }
217        for &parent in parents {
218            self.node_mut(parent)?.children.push(id);
219        }
220        if let NodeKind::Delta { command } = &kind {
221            let command = *command as usize;
222            for &parent in parents {
223                let is_conflict = {
224                    let commands = self.parent_commands.entry(parent).or_default();
225                    commands.set_grow(command);
226                    commands.count() >= 2
227                };
228                if is_conflict {
229                    self.conflict_parents.insert(parent);
230                }
231            }
232        }
233        self.nodes.push(Node {
234            kind,
235            ancestors,
236            children: vec![],
237        });
238        Ok(id)
239    }
240
241    /// Returns the unique node for a PTB location, creating it on first use.
242    fn root(&mut self, root: RootLocation) -> anyhow::Result<NodeID> {
243        if let Some(id) = self.roots.get(&root) {
244            return Ok(*id);
245        }
246        let id = self.new_node(NodeKind::Root, &[])?;
247        self.roots.insert(root, id);
248        Ok(id)
249    }
250
251    /// Returns the node for a PTB location if one has been created
252    fn get_root(&self, root: RootLocation) -> Option<NodeID> {
253        self.roots.get(&root).copied()
254    }
255
256    /// Creates a call-return delta reference, whose ancestors are derived from parent reference
257    /// arguments passed to the command.
258    /// For immutable references, the parents are all reference arguments.
259    /// For mutable references, the parents are the mutable reference arguments.
260    fn call_return(&mut self, command: u16, parents: &[NodeID]) -> anyhow::Result<NodeID> {
261        self.new_node(NodeKind::Delta { command }, parents)
262    }
263
264    /// Returns whether `node` can derive from `ancestor`.
265    fn is_ancestor(&self, node: NodeID, ancestor: NodeID) -> anyhow::Result<bool> {
266        Ok(self.node(node)?.ancestors.contains(ancestor))
267    }
268
269    /// Checks whether `common` has delta children from different commands in the two ancestor sets.
270    fn common_has_cross_command(
271        &self,
272        common: NodeID,
273        left_ancestors: &BitSet,
274        right_ancestors: &BitSet,
275    ) -> anyhow::Result<bool> {
276        let mut left_cmd: Option<u16> = None;
277        let mut left_multi = false;
278        let mut right_cmd: Option<u16> = None;
279        let mut right_multi = false;
280        for &child in &self.node(common)?.children {
281            let NodeKind::Delta { command } = &self.node(child)?.kind else {
282                continue;
283            };
284            let command = *command;
285            if left_ancestors.contains(child) {
286                match left_cmd {
287                    None => left_cmd = Some(command),
288                    Some(c) if c != command => left_multi = true,
289                    _ => {}
290                }
291            }
292            if right_ancestors.contains(child) {
293                match right_cmd {
294                    None => right_cmd = Some(command),
295                    Some(c) if c != command => right_multi = true,
296                    _ => {}
297                }
298            }
299        }
300        // Both sides must contain a delta child. If either side contains children from multiple
301        // commands, at least one pair must have different commands.
302        Ok(match (left_cmd, right_cmd) {
303            (Some(l), Some(r)) => left_multi || right_multi || l != r,
304            _ => false,
305        })
306    }
307
308    /// Checks for paths from a shared ancestor through delta children from different commands.
309    /// Such paths may overlap; only ancestors in `conflict_parents` need checking.
310    fn has_cross_command_extensions(&self, left: NodeID, right: NodeID) -> anyhow::Result<bool> {
311        if self.conflict_parents.is_empty() {
312            return Ok(false);
313        }
314        let left_ancestors = &self.node(left)?.ancestors;
315        let right_ancestors = &self.node(right)?.ancestors;
316        let min_width = left_ancestors.words.len().min(right_ancestors.words.len());
317        // Heuristic: estimate scan cost by comparing the number of conflict parents with the number
318        // of words in the shorter ancestor bitset.
319        if self.conflict_parents.len() <= min_width {
320            for &parent in &self.conflict_parents {
321                if left_ancestors.contains(parent)
322                    && right_ancestors.contains(parent)
323                    && self.common_has_cross_command(parent, left_ancestors, right_ancestors)?
324                {
325                    return Ok(true);
326                }
327            }
328            Ok(false)
329        } else {
330            left_ancestors.try_any_common(right_ancestors, |common| {
331                let is_conflict_parent = self
332                    .parent_commands
333                    .get(&common)
334                    .is_some_and(|commands| commands.count() >= 2);
335                Ok(is_conflict_parent
336                    && self.common_has_cross_command(common, left_ancestors, right_ancestors)?)
337            })
338        }
339    }
340
341    /// Returns whether `left` may extend `right` through ancestry or cross-command extensions.
342    fn extends(&self, left: NodeID, right: NodeID) -> anyhow::Result<bool> {
343        Ok((left != right && self.is_ancestor(left, right)?)
344            || self.has_cross_command_extensions(left, right)?)
345    }
346
347    /// Returns whether neither node derives from the other and neither has cross-command extensions.
348    fn is_disjoint(&self, left: NodeID, right: NodeID) -> anyhow::Result<bool> {
349        Ok(left != right
350            && !self.is_ancestor(left, right)?
351            && !self.is_ancestor(right, left)?
352            && !self.has_cross_command_extensions(left, right)?)
353    }
354}

356impl Value {
357    fn copy(&self) -> Value {
358        match self {
359            Value::NonRef => Value::NonRef,
360            Value::Ref { is_mut, node } => Value::Ref {
361                is_mut: *is_mut,
362                node: *node,
363            },
364        }
365    }
366
367    fn freeze(&mut self) -> anyhow::Result<Value> {
368        match self.copy() {
369            Value::NonRef => anyhow::bail!("Cannot freeze a non-reference value"),
370            Value::Ref { is_mut, node } => {
371                anyhow::ensure!(is_mut, "Cannot freeze an immutable reference");
372                Ok(Value::Ref {
373                    is_mut: false,
374                    node,
375                })
376            }
377        }
378    }
379}

381impl Location {
382    fn non_ref(root: RootLocation) -> Self {
383        Self {
384            root,
385            value: Some(Value::NonRef),
386        }
387    }
388
389    fn copy_value(&self) -> anyhow::Result<Value> {
390        self.value
391            .as_ref()
392            .map(Value::copy)
393            .ok_or_else(|| anyhow::anyhow!("Use of invalid memory location"))
394    }
395
396    fn move_value(&mut self) -> anyhow::Result<Value> {
397        self.value
398            .take()
399            .ok_or_else(|| anyhow::anyhow!("Use of invalid memory location"))
400    }
401
402    fn use_(&mut self, usage: &T::Usage) -> anyhow::Result<Value> {
403        match usage {
404            T::Usage::Move(_) => self.move_value(),
405            T::Usage::Copy { .. } => self.copy_value(),
406        }
407    }
408
409    fn assert_borrowable(&self) -> anyhow::Result<()> {
410        match self.value.as_ref() {
411            None => anyhow::bail!("Borrow of invalid memory location"),
412            Some(Value::Ref { .. }) => anyhow::bail!("Cannot borrow a reference"),
413            Some(Value::NonRef) => Ok(()),
414        }
415    }

418impl Context {
419    fn new<Mode: ExecutionMode>(_env: &Env<Mode>, txn: &T::Transaction) -> anyhow::Result<Self> {
420        let T::Transaction {
421            gas_payment,
422            bytes: _,
423            objects,
424            withdrawals,
425            pure,
426            receiving,
427            withdrawal_compatibility_conversions: _,
428            original_command_len: _,
429            commands: _,
430            unified_linkage: _,
431        } = txn;
432        let memory = Memory::new();
433        let tx_context = Location::non_ref(RootLocation::Known(T::Location::TxContext));
434        let mut gas = Location::non_ref(RootLocation::Known(T::Location::GasCoin));
435        if gas_payment.is_none() {
436            gas.move_value()
437                .map_err(|_| anyhow::anyhow!("gas coin should be initialized"))?;
438        }
439        let object_inputs = (0..objects.len())
440            .map(|i| {
441                Ok(Location::non_ref(RootLocation::Known(
442                    T::Location::ObjectInput(checked_as!(i, u16)?),
443                )))
444            })
445            .collect::<anyhow::Result<_>>()?;
446        let withdrawal_inputs = (0..withdrawals.len())
447            .map(|i| {
448                Ok(Location::non_ref(RootLocation::Known(
449                    T::Location::WithdrawalInput(checked_as!(i, u16)?),
450                )))
451            })
452            .collect::<anyhow::Result<_>>()?;
453        let pure_inputs = (0..pure.len())
454            .map(|i| {
455                Ok(Location::non_ref(RootLocation::Known(
456                    T::Location::PureInput(checked_as!(i, u16)?),
457                )))
458            })
459            .collect::<anyhow::Result<_>>()?;
460        let receiving_inputs = (0..receiving.len())
461            .map(|i| {
462                Ok(Location::non_ref(RootLocation::Known(
463                    T::Location::ReceivingInput(checked_as!(i, u16)?),
464                )))
465            })
466            .collect::<anyhow::Result<_>>()?;
467        Ok(Self {
468            memory,
469            tx_context,
470            gas,
471            object_inputs,
472            withdrawal_inputs,
473            pure_inputs,
474            receiving_inputs,
475            results: vec![],
476            result_ref_rows: vec![],

478        })
479    }

481    fn current_command(&self) -> anyhow::Result<u16> {
482        Ok(checked_as!(self.results.len(), u16)?)
483    }

485    fn add_result_values(
486        &mut self,
487        results: impl IntoIterator<Item = Option<Value>>,
488    ) -> anyhow::Result<()> {
489        let command = self.current_command()?;
490        let row = results
491            .into_iter()
492            .enumerate()
493            .map(|(i, v)| {
494                Ok(Location {
495                    root: RootLocation::Known(T::Location::Result(command, checked_as!(i, u16)?)),
496                    value: v,
497                })
498            })
499            .collect::<anyhow::Result<Vec<_>>>()?;
500        if row
501            .iter()
502            .any(|loc| matches!(loc.value, Some(Value::Ref { .. })))
503        {
504            self.result_ref_rows.push(self.results.len());
505        }
506        self.results.push(row);
507        Ok(())
508    }

566    /// Returns true iff any live reference borrows `location`.
567    fn location_is_borrowed(&self, location: &Location) -> anyhow::Result<bool> {
568        match self.memory.get_root(location.root) {
569            Some(node) => self.any_extends(node, /* ignore alias */ false),
570            None => Ok(false),
571        }
572    }

574    /// Whether the location is borrowed by a reference argument already seen this command, i.e. its
575    /// root node is an ancestor of one of `arg_reference_nodes`. A location never borrowed has no
576    /// node and cannot be such an ancestor.
577    fn borrowed_by_arg_reference(&self, location: T::Location) -> anyhow::Result<bool> {
578        let Some(loc_node) = self.memory.get_root(RootLocation::Known(location)) else {
579            return Ok(false);
580        };
581        for &node in &self.arg_reference_nodes {
582            if self.memory.is_ancestor(node, loc_node)? {
583                return Ok(true);
584            }
585        }
586        Ok(false)
587    }
588
589    fn check_usage(&self, usage: &T::Usage, location: &Location) -> anyhow::Result<()> {
590        let is_borrowed = self.location_is_borrowed(location)?
591            || self.borrowed_by_arg_reference(usage.location())?;
592        match usage {
593            T::Usage::Move(_) => {
594                anyhow::ensure!(!is_borrowed, "Cannot move a value that is borrowed");
595            }
596            T::Usage::Copy {
597                borrowed: borrowed_flag,
598                ..
599            } => {
600                let Some(borrowed_flag) = borrowed_flag.get().copied() else {
601                    anyhow::bail!("Borrowed flag not set for copy usage");
602                };
603                // Drop-safety refinement can release references earlier than the original borrow
604                // check. A true borrowed flag may therefore be stale, but a location that is still
605                // borrowed must have the flag set.
606                if is_borrowed {
607                    anyhow::ensure!(
608                        borrowed_flag,
609                        "Copy of borrowed location {:?} in command {} is not flagged as borrowed",
610                        location.root,
611                        self.current_command()?
612                    );
613                }
614            }
615        }
616        Ok(())
617    }

619    fn argument(&mut self, sp!(_, (arg, _)): &T::Argument) -> anyhow::Result<Value> {
620        let location = self.location(arg.location())?;
621        match arg {
622            T::Argument__::Use(usage)
623            | T::Argument__::Freeze(usage)
624            | T::Argument__::Read(usage) => self.check_usage(usage, location)?,
625            T::Argument__::Borrow(_, _) => (),
626        };
627        let value = match arg {
628            T::Argument__::Use(usage) => self.location_mut(arg.location())?.use_(usage)?,
629            T::Argument__::Freeze(usage) => {
630                self.location_mut(arg.location())?.use_(usage)?.freeze()?
631            }
632            T::Argument__::Borrow(is_mut, _) => self.borrow_location(arg.location(), *is_mut)?,
633            T::Argument__::Read(usage) => {
634                self.location_mut(arg.location())?.use_(usage)?;
635                Value::NonRef
636            }
637        };
638        if let Value::Ref { node, .. } = &value {
639            self.arg_reference_nodes.push(*node);
640        }
641        Ok(value)
642    }
643
644    /// Borrows a location, creating its graph node on first use.
645    fn borrow_location(&mut self, loc: T::Location, is_mut: bool) -> anyhow::Result<Value> {
646        let root = {
647            let location = self.location(loc)?;
648            location.assert_borrowable()?;
649            location.root
650        };
651        let node = self.memory.root(root)?;
652        Ok(Value::Ref { is_mut, node })
653    }
654
655    fn arguments(&mut self, args: &[T::Argument]) -> anyhow::Result<Vec<Value>> {
656        args.iter()
657            .map(|arg| self.argument(arg))
658            .collect::<anyhow::Result<Vec<_>>>()
659    }
660
661    fn all_references(&self) -> impl Iterator<Item = NodeID> + '_ {
662        // Only results can hold references. `tx_context`, `gas`, and every input are created
663        // non-reference and never reassigned, so they are not scanned. Among results, only rows
664        // produced by reference-returning commands can hold references.
665        self.result_ref_rows
666            .iter()
667            .filter_map(move |&i| self.results.get(i))
668            .flatten()
669            .filter_map(|v| match v.value.as_ref() {
670                Some(Value::Ref { node, .. }) => Some(*node),
671                Some(Value::NonRef) | None => None,
672            })
673    }
674
675    /// Returns whether any live reference may extend `node`. Excludes aliases when requested.
676    fn any_extends(&self, node: NodeID, ignore_aliases: bool) -> anyhow::Result<bool> {
677        for other in self.all_references() {
678            if (!ignore_aliases && other == node) || self.memory.extends(other, node)? {
679                return Ok(true);
680            }
681        }
682        Ok(false)
683    }
684}

686/// Verifies memory safety using an alternative implementation of `verify::memory_safety`.
687///
688/// It represents memory locations and call returns as graph nodes. Root nodes represent known PTB
689/// locations (or in rare cases unknown values from calls without reference arguments). Each node
690/// tracks its transitive ancestors in a bitset. A call-return delta records its command identity,
691/// analogous to the regex verifier's `.*` extension, while retaining enough identity to use the
692/// guarantee that mutable references returned by one call do not overlap with other references
693/// returned from that call. Extensions from distinct commands are conservatively treated as
694/// potentially overlapping. PTBs have no control flow, which makes this representation sufficient
695/// as an invariant check for the regex implementation.
696/// Checks the following
697/// - Values are not used after being moved
698/// - Reference safety is upheld (no dangling references)
699pub fn verify<Mode: ExecutionMode>(
700    env: &Env<Mode>,
701    txn: &T::Transaction,
702) -> Result<(), Mode::Error> {
703    Ok(verify_(env, txn).map_err(|e| make_invariant_violation!("{}. Transaction {:?}", e, txn))?)
704}
705
706pub(crate) fn verify_<Mode: ExecutionMode>(
707    env: &Env<Mode>,
708    txn: &T::Transaction,
709) -> anyhow::Result<()> {
710    let mut context = Context::new(env, txn)?;
711    let T::Transaction {
712        gas_payment: _,
713        bytes: _,
714        objects: _,
715        withdrawals: _,
716        pure: _,
717        receiving: _,
718        withdrawal_compatibility_conversions: _,
719        original_command_len: _,
720        commands,
721        unified_linkage: _,
722    } = txn;
723    for c in commands {
724        command(&mut context, c)?;
725    }
726    Ok(())
727}
728
729fn command(context: &mut Context, c: &T::Command) -> anyhow::Result<()> {
730    debug_assert!(context.arg_reference_nodes.is_empty());
731    let results = command_(context, c)?;
732    // drop unused result values by marking them as `None`
733    assert_invariant!(
734        results.len() == c.value.drop_values.len(),
735        "result length mismatch. expected {}, got {}",
736        c.value.drop_values.len(),
737        results.len()
738    );
739    context.add_result_values(
740        results
741            .into_iter()
742            .zip_debug_eq(c.value.drop_values.iter().copied())
743            .map(|(v, drop)| if drop { None } else { Some(v) }),
744    )?;
745    context.arg_reference_nodes.clear();
746    Ok(())
747}
748
749fn command_(context: &mut Context, sp!(_, c): &T::Command) -> anyhow::Result<Vec<Value>> {
750    let result_tys = &c.result_type;
751    let results = match &c.command {
752        T::Command__::MoveCall(move_call) => {
753            let T::MoveCall {
754                function,
755                arguments,
756            } = &**move_call;
757            let arg_values = context.arguments(arguments)?;
758            call(context, &function.signature, arg_values)?
759        }
760        T::Command__::TransferObjects(objs, recipient) => {
761            context.arguments(objs)?;
762            context.argument(recipient)?;
763            non_ref_results(result_tys)?
764        }
765        T::Command__::SplitCoins(_, coin, amounts) => {
766            context.arguments(amounts)?;
767            let coin_value = context.argument(coin)?;
768            write_ref(context, coin_value)?;
769            non_ref_results(result_tys)?
770        }
771        T::Command__::MergeCoins(_, target, coins) => {
772            context.arguments(coins)?;
773            let target_value = context.argument(target)?;
774            write_ref(context, target_value)?;
775            non_ref_results(result_tys)?
776        }
777        T::Command__::MakeMoveVec(_, arguments) => {
778            context.arguments(arguments)?;
779            non_ref_results(result_tys)?
780        }
781        T::Command__::Publish(_, _, _) => non_ref_results(result_tys)?,
782        T::Command__::Upgrade(_, _, _, ticket, _) => {
783            context.argument(ticket)?;
784            non_ref_results(result_tys)?
785        }
786    };
787    assert_invariant!(
788        result_tys.len() == results.len(),
789        "result length mismatch. Expected {}, got {}",
790        result_tys.len(),
791        results.len()
792    );
793    Ok(results)
794}
795
796fn write_ref(context: &Context, value: Value) -> anyhow::Result<()> {
797    match value {
798        Value::NonRef => {
799            anyhow::bail!("Cannot write to a non-reference value");
800        }
801
802        Value::Ref { is_mut: false, .. } => {
803            anyhow::bail!("Cannot write to an immutable reference");
804        }
805        Value::Ref { is_mut: true, node } => {
806            anyhow::ensure!(
807                !context.any_extends(node, /* ignore alias */ true)?,
808                "Cannot write to a mutable reference that has extensions"
809            );
810            Ok(())
811        }
812    }
813}
814
815fn call(
816    context: &mut Context,
817    signature: &T::LoadedFunctionInstantiation,
818    arguments: Vec<Value>,
819) -> anyhow::Result<Vec<Value>> {
820    let return_ = &signature.return_;
821    let mut all_nodes = Vec::new();
822    let mut imm_nodes = Vec::new();
823    let mut mut_nodes = Vec::new();
824    for arg in arguments {
825        match arg {
826            Value::NonRef => (),
827            Value::Ref { is_mut: true, node } => {
828                anyhow::ensure!(
829                    !context.any_extends(node, /* ignore alias */ true)?,
830                    "Cannot transfer a mutable ref with extensions"
831                );
832                for other in &mut_nodes {
833                    anyhow::ensure!(
834                        context.memory.is_disjoint(*other, node)?,
835                        "Double mutable borrow"
836                    );
837                }
838                all_nodes.push(node);
839                mut_nodes.push(node);
840            }
841            Value::Ref {
842                is_mut: false,
843                node,
844            } => {
845                all_nodes.push(node);
846                imm_nodes.push(node);
847            }
848        }
849    }
850    // All mutable references must be disjoint from all immutable references
851    for immutable in &imm_nodes {
852        for mutable in &mut_nodes {
853            anyhow::ensure!(
854                context.memory.is_disjoint(*immutable, *mutable)?,
855                "Mutable and immutable borrows cannot overlap"
856            );
857        }
858    }
859    // Returned references cannot borrow from TxContext, so exclude arguments derived from it
860    // when choosing parents for return nodes.
861    if let Some(tx_context_root) = context
862        .memory
863        .get_root(RootLocation::Known(T::Location::TxContext))
864    {
865        let mut tx_context_nodes = BTreeSet::new();
866        // `mut_nodes` is a subset of `all_nodes`, so all candidates are covered.
867        for node in &all_nodes {
868            if context.memory.is_ancestor(*node, tx_context_root)? {
869                tx_context_nodes.insert(*node);
870            }
871        }
872        mut_nodes.retain(|node| !tx_context_nodes.contains(node));
873        all_nodes.retain(|node| !tx_context_nodes.contains(node));
874    }
875    let command = context.current_command()?;
876    // Reference returns derive from an unknown root when no reference arguments feed them. Calls
877    // that return no references need no such node, so avoid creating one.
878    let has_reference_return = return_
879        .iter()
880        .any(|ty| matches!(ty, T::Type::Reference(_, _)));
881    let mut_nodes = if mut_nodes.is_empty() && has_reference_return {
882        vec![context.memory.root(RootLocation::Unknown { command })?]
883    } else {
884        mut_nodes
885    };
886    let all_nodes = if all_nodes.is_empty() && has_reference_return {
887        vec![context.memory.root(RootLocation::Unknown { command })?]
888    } else {
889        all_nodes
890    };
891    return_
892        .iter()
893        .enumerate()
894        .map(|(i, ty)| {
895            let _result = checked_as!(i, u16)?;
896            Ok(match ty {
897                T::Type::Reference(/* is mut */ true, _) => Value::Ref {
898                    is_mut: true,
899                    node: context.memory.call_return(command, &mut_nodes)?,
900                },
901                T::Type::Reference(/* is mut */ false, _) => Value::Ref {
902                    is_mut: false,
903                    node: context.memory.call_return(command, &all_nodes)?,
904                },
905                _ => Value::NonRef,
906            })
907        })
908        .collect::<anyhow::Result<Vec<_>>>()
909}
910
911fn non_ref_results(results: &[T::Type]) -> anyhow::Result<Vec<Value>> {
912    results
913        .iter()
914        .map(|t| {
915            anyhow::ensure!(
916                !matches!(t, T::Type::Reference(_, _)),
917                "attempted to create a non-reference result from a reference type",
918            );
919            Ok(Value::NonRef)
920        })
921        .collect()
922}
