21/// A packed set of node IDs. Node `n` is stored in word `n / 64` at bit `n % 64`.

75impl BitSet {
76    /// Allocates enough zeroed 64-bit words for `bits` flags. Zero means no flags are set.
77    fn with_bits(bits: usize) -> Self {
78        Self {
79            words: vec![0; bits.div_ceil(64).max(1)],
80        }
81    }
82
83    /// Sets the flag by OR-ing its one-bit mask into the word that contains it.
84    /// Returns true iff flag was already set.
85    fn set(&mut self, bit: usize) -> anyhow::Result<bool> {
86        let word = self
87            .words
88            .get_mut(bit / 64)
89            .ok_or_else(|| anyhow::anyhow!("BitSet index {bit} is out of bounds"))?;
90        let mask = 1 << (bit % 64);
91        let was_set = *word & mask != 0;
92        *word |= mask;
93        Ok(was_set)
94    }
95
96    /// Tests the flag by AND-ing its one-bit mask with the containing word.
97    fn contains(&self, bit: usize) -> bool {
98        self.words
99            .get(bit / 64)
100            .is_some_and(|word| word & (1 << (bit % 64)) != 0)
101    }
102
103    /// Computes set union 64 flags at a time with a wordwise OR, potentially growing to hold
104    /// `other`'s flags.
105    fn union(&mut self, other: &Self) {
106        if self.words.len() < other.words.len() {
107            self.words.resize(other.words.len(), 0);
108        }
109        // After the resize `self` is at least as long as `other`, and any words past `other`'s
110        // end are unaffected by the union, so truncating to `other`'s length is correct.
111        #[allow(clippy::disallowed_methods)]
112        let words = self.words.iter_mut().zip(&other.words);
113        for (word, other_word) in words {
114            *word |= other_word;
115        }
116    }
117
118    /// Visits the intersection of two sets. `trailing_zeros` finds the lowest set bit and
119    /// `clear_lowest_bit` clears it, so only set bits are visited.
120    fn try_for_each_common(
121        &self,
122        other: &Self,
123        mut f: impl FnMut(usize) -> anyhow::Result<()>,
124    ) -> anyhow::Result<()> {
125        // Ancestor sets are sized by node ID, so the two sets can have different word counts. The
126        // missing trailing words are all zeros, so nothing past the shorter set can be common. As
127        // such, truncating to the shorter length is correct.
128        #[allow(clippy::disallowed_methods)]
129        let words = self.words.iter().zip(&other.words);
130        for (word_index, (left, right)) in words.enumerate() {
131            let mut common = left & right;
132            while common != 0 {
133                f(Self::bit_index(word_index, common)?)?;
134                common = Self::clear_lowest_bit(common)?;
135            }
136        }
137        Ok(())
138    }
139
140    /// Visits each set bit using `trailing_zeros` and clears it before the next iteration.
141    fn try_for_each_set(
142        &self,
143        mut f: impl FnMut(usize) -> anyhow::Result<()>,
144    ) -> anyhow::Result<()> {
145        for (word_index, word) in self.words.iter().enumerate() {
146            let mut bits = *word;
147            while bits != 0 {
148                f(Self::bit_index(word_index, bits)?)?;
149                bits = Self::clear_lowest_bit(bits)?;
150            }
151        }
152        Ok(())
153    }
154
155    /// Returns the absolute index of the lowest set bit of `word`, which lives in `word_index`.
156    fn bit_index(word_index: usize, word: u64) -> anyhow::Result<usize> {
157        debug_assert!(word != 0);
158        word_index
159            .checked_mul(64)
160            .and_then(|base| base.checked_add(word.trailing_zeros() as usize))
161            .ok_or_else(|| anyhow::anyhow!("BitSet index overflow for word {word_index}"))
162    }
163
164    /// Removes the lowest set bit of `word`, which must be non-zero.
165    /// Subtracting one clears the lowest set bit and sets all bits below it, so the `&` preserves
166    /// the remaining bits.
167    fn clear_lowest_bit(word: u64) -> anyhow::Result<u64> {
168        let sub = word
169            .checked_sub(1)
170            .ok_or_else(|| anyhow::anyhow!("clear_lowest_bit called on 0"))?;
171        Ok(word & sub)
172    }
173}

175impl Memory {
176    fn node(&self, id: NodeID) -> anyhow::Result<&Node> {
177        self.nodes
178            .get(id)
179            .ok_or_else(|| anyhow::anyhow!("Node index {id} is out of bounds"))
180    }
181
182    fn node_mut(&mut self, id: NodeID) -> anyhow::Result<&mut Node> {
183        self.nodes
184            .get_mut(id)
185            .ok_or_else(|| anyhow::anyhow!("Node index {id} is out of bounds"))
186    }
187
188    /// Creates a node whose ancestor set is itself plus every ancestor of its parents, and records
189    /// reverse parent-to-child edges for identifying extensions from distinct commands.
190    fn new_node(&mut self, kind: NodeKind, parents: &[NodeID]) -> anyhow::Result<NodeID> {
191        let id = self.nodes.len();
192        let bits = id
193            .checked_add(1)
194            .ok_or_else(|| anyhow::anyhow!("Node ID overflow"))?;
195        let mut ancestors = BitSet::with_bits(bits);
196        anyhow::ensure!(
197            !ancestors.set(id)?,
198            "Node {id} is already set in its fresh ancestor set"
199        );
200        for &parent in parents {
201            ancestors.union(&self.node(parent)?.ancestors);
202        }
203        for &parent in parents {
204            self.node_mut(parent)?.children.push(id);
205        }
206        self.nodes.push(Node {
207            kind,
208            ancestors,
209            children: vec![],
210        });
211        Ok(id)
212    }
213
214    /// Returns the unique node for a PTB location, creating it on first use.
215    fn root(&mut self, root: RootLocation) -> anyhow::Result<NodeID> {
216        if let Some(id) = self.roots.get(&root) {
217            return Ok(*id);
218        }
219        let id = self.new_node(NodeKind::Root(root), &[])?;
220        self.roots.insert(root, id);
221        Ok(id)
222    }
223
224    /// Creates a call-return delta reference, whose ancestors are derived from parent reference
225    /// arguments passed to the command.
226    /// For immutable references, the parents are all reference arguments.
227    /// For mutable references, the parents are the mutable reference arguments.
228    fn call_return(&mut self, command: u16, parents: &[NodeID]) -> anyhow::Result<NodeID> {
229        self.new_node(NodeKind::Delta { command }, parents)
230    }
231
232    /// Returns whether `node` can derive from `ancestor`.
233    fn is_ancestor(&self, node: NodeID, ancestor: NodeID) -> anyhow::Result<bool> {
234        Ok(self.node(node)?.ancestors.contains(ancestor))
235    }
236
237    /// Returns whether paths from a shared ancestor pass through delta children from different
238    /// commands. Such extensions are conservatively treated as potentially overlapping.
239    fn has_cross_command_extensions(&self, left: NodeID, right: NodeID) -> anyhow::Result<bool> {
240        let left_ancestors = &self.node(left)?.ancestors;
241        let right_ancestors = &self.node(right)?.ancestors;
242        let mut has_extensions = false;
243        left_ancestors.try_for_each_common(right_ancestors, |common| {
244            if has_extensions {
245                return Ok(());
246            }
247            let mut left_commands = IndexSet::new();
248            let mut right_commands = IndexSet::new();
249            for &child in &self.node(common)?.children {
250                let NodeKind::Delta { command } = &self.node(child)?.kind else {
251                    continue;
252                };
253                if left_ancestors.contains(child) {
254                    left_commands.insert(*command);
255                }
256                if right_ancestors.contains(child) {
257                    right_commands.insert(*command);
258                }
259            }
260            has_extensions = left_commands
261                .iter()
262                .any(|left| right_commands.iter().any(|right| left != right));
263            Ok(())
264        })?;
265        Ok(has_extensions)
266    }
267
268    /// Returns whether `left` may extend `right` through ancestry or cross-command extensions.
269    fn extends(&self, left: NodeID, right: NodeID) -> anyhow::Result<bool> {
270        Ok((left != right && self.is_ancestor(left, right)?)
271            || self.has_cross_command_extensions(left, right)?)
272    }
273
274    /// Returns whether neither node derives from the other and neither has cross-command extensions.
275    fn is_disjoint(&self, left: NodeID, right: NodeID) -> anyhow::Result<bool> {
276        Ok(left != right
277            && !self.is_ancestor(left, right)?
278            && !self.is_ancestor(right, left)?
279            && !self.has_cross_command_extensions(left, right)?)
280    }
281
282    /// Returns the known PTB locations among `node`'s ancestors, excluding unknown roots.
283    fn known_roots(&self, node: NodeID) -> anyhow::Result<IndexSet<T::Location>> {
284        let mut roots = IndexSet::new();
285        self.node(node)?.ancestors.try_for_each_set(|ancestor| {
286            if let NodeKind::Root(RootLocation::Known(location)) = &self.node(ancestor)?.kind {
287                roots.insert(*location);
288            }
289            Ok(())
290        })?;
291        Ok(roots)
292    }
293}

295impl Value {
296    fn copy(&self) -> Value {
297        match self {
298            Value::NonRef => Value::NonRef,
299            Value::Ref { is_mut, node } => Value::Ref {
300                is_mut: *is_mut,
301                node: *node,
302            },
303        }
304    }
305
306    fn freeze(&mut self) -> anyhow::Result<Value> {
307        match self.copy() {
308            Value::NonRef => anyhow::bail!("Cannot freeze a non-reference value"),
309            Value::Ref { is_mut, node } => {
310                anyhow::ensure!(is_mut, "Cannot freeze an immutable reference");
311                Ok(Value::Ref {
312                    is_mut: false,
313                    node,
314                })
315            }
316        }
317    }
318}

320impl Location {
321    fn non_ref(root: NodeID) -> Self {
322        Self {
323            root,
324            value: Some(Value::NonRef),
325        }
326    }
327
328    fn copy_value(&self) -> anyhow::Result<Value> {
329        self.value
330            .as_ref()
331            .map(Value::copy)
332            .ok_or_else(|| anyhow::anyhow!("Use of invalid memory location"))
333    }
334
335    fn move_value(&mut self) -> anyhow::Result<Value> {
336        self.value
337            .take()
338            .ok_or_else(|| anyhow::anyhow!("Use of invalid memory location"))
339    }
340
341    fn use_(&mut self, usage: &T::Usage) -> anyhow::Result<Value> {
342        match usage {
343            T::Usage::Move(_) => self.move_value(),
344            T::Usage::Copy { .. } => self.copy_value(),
345        }
346    }
347
348    fn borrow(&self, is_mut: bool) -> anyhow::Result<Value> {
349        match self.value.as_ref() {
350            None => anyhow::bail!("Borrow of invalid memory location"),
351            Some(Value::Ref { .. }) => anyhow::bail!("Cannot borrow a reference"),
352            Some(Value::NonRef) => Ok(Value::Ref {
353                is_mut,
354                node: self.root,
355            }),
356        }
357    }
358}

360impl Context {
361    fn new<Mode: ExecutionMode>(_env: &Env<Mode>, txn: &T::Transaction) -> anyhow::Result<Self> {
362        let T::Transaction {
363            gas_payment,
364            bytes: _,
365            objects,
366            withdrawals,
367            pure,
368            receiving,
369            withdrawal_compatibility_conversions: _,
370            original_command_len: _,
371            commands: _,
372            unified_linkage: _,
373        } = txn;
374        let mut memory = Memory::default();
375        let tx_context =
376            Location::non_ref(memory.root(RootLocation::Known(T::Location::TxContext))?);
377        let mut gas = Location::non_ref(memory.root(RootLocation::Known(T::Location::GasCoin))?);
378        if gas_payment.is_none() {
379            gas.move_value()
380                .map_err(|_| anyhow::anyhow!("gas coin should be initialized"))?;
381        }
382        let object_inputs = (0..objects.len())
383            .map(|i| {
384                Ok(Location::non_ref(memory.root(RootLocation::Known(
385                    T::Location::ObjectInput(checked_as!(i, u16)?),
386                ))?))
387            })
388            .collect::<anyhow::Result<_>>()?;
389        let withdrawal_inputs = (0..withdrawals.len())
390            .map(|i| {
391                Ok(Location::non_ref(memory.root(RootLocation::Known(
392                    T::Location::WithdrawalInput(checked_as!(i, u16)?),
393                ))?))
394            })
395            .collect::<anyhow::Result<_>>()?;
396        let pure_inputs = (0..pure.len())
397            .map(|i| {
398                Ok(Location::non_ref(memory.root(RootLocation::Known(
399                    T::Location::PureInput(checked_as!(i, u16)?),
400                ))?))
401            })
402            .collect::<anyhow::Result<_>>()?;
403        let receiving_inputs = (0..receiving.len())
404            .map(|i| {
405                Ok(Location::non_ref(memory.root(RootLocation::Known(
406                    T::Location::ReceivingInput(checked_as!(i, u16)?),
407                ))?))
408            })
409            .collect::<anyhow::Result<_>>()?;
410        Ok(Self {
411            memory,
412            allow_references_in_ptbs: _env.protocol_config.allow_references_in_ptbs(),
413            tx_context,
414            gas,
415            object_inputs,
416            withdrawal_inputs,
417            pure_inputs,
418            receiving_inputs,
419            results: vec![],
420            arg_roots: IndexSet::new(),
421        })
422    }

429        &mut self,
430        results: impl IntoIterator<Item = Option<Value>>,

433        self.results.push(

443                        value: v,

445                })

449    }

452        Ok(match loc {
453            T::Location::TxContext => &self.tx_context,
454            T::Location::GasCoin => &self.gas,
455            T::Location::ObjectInput(i) => self

459            T::Location::WithdrawalInput(i) => self

463            T::Location::PureInput(i) => self

467            T::Location::ReceivingInput(i) => self

507    fn check_usage(&self, usage: &T::Usage, location: &Location) -> anyhow::Result<()> {
508        // by marking "ignore alias" as `false`, we will also check for `Alias` paths, i.e. paths
509        // that point to the location itself without any extensions.
510        let is_borrowed = self.any_extends(location.root, /* ignore alias */ false)?
511            || self.arg_roots.contains(&usage.location());
512        match usage {
513            T::Usage::Move(_) => {
514                anyhow::ensure!(!is_borrowed, "Cannot move a value that is borrowed");
515            }
516            T::Usage::Copy {
517                borrowed: borrowed_flag,
518                ..
519            } => {
520                let Some(borrowed_flag) = borrowed_flag.get().copied() else {
521                    anyhow::bail!("Borrowed flag not set for copy usage");
522                };
523                // `verify::memory_safety` sets `borrowed` before `drop_safety` rewrites last-use
524                // copies of references into moves. Those moves release references earlier than
525                // an originally annotated copies did, which might mean that a reference that
526                // caused the `borrowed_flag` to be set might have been "optimized" to being
527                // released early.
528                // As such, we can just check that if the location `is_borrowed` then the
529                // flag must be consistent, i.e.
530                // is_borrowed ==> borrowed_flag
531                if is_borrowed {
532                    anyhow::ensure!(
533                        borrowed_flag,
534                        "Copy of borrowed location {:?} in command {} is not flagged as borrowed",
535                        location.root,
536                        self.current_command()?
537                    );
538                }
539            }
540        }
541        Ok(())
542    }
543
544    fn argument(&mut self, sp!(_, (arg, _)): &T::Argument) -> anyhow::Result<Value> {
545        let location = self.location(arg.location())?;
546        match arg {
547            T::Argument__::Use(usage)
548            | T::Argument__::Freeze(usage)
549            | T::Argument__::Read(usage) => self.check_usage(usage, location)?,
550            T::Argument__::Borrow(_, _) => (),
551        };
552        let location = self.location_mut(arg.location())?;
553        let value = match arg {
554            T::Argument__::Use(usage) => location.use_(usage)?,
555            T::Argument__::Freeze(usage) => location.use_(usage)?.freeze()?,
556            T::Argument__::Borrow(is_mut, _) => location.borrow(*is_mut)?,
557            T::Argument__::Read(usage) => {
558                location.use_(usage)?;
559                Value::NonRef
560            }
561        };
562        if let Value::Ref { node, .. } = &value {
563            self.arg_roots.extend(self.memory.known_roots(*node)?);
564        }
565        Ok(value)
566    }
567
568    fn arguments(&mut self, args: &[T::Argument]) -> anyhow::Result<Vec<Value>> {
569        args.iter()
570            .map(|arg| self.argument(arg))
571            .collect::<anyhow::Result<Vec<_>>>()
572    }

574    fn all_references(&self) -> impl Iterator<Item = NodeID> + '_ {
575        let Self {
576            tx_context,
577            gas,
578            object_inputs,
579            withdrawal_inputs,
580            pure_inputs,
581            receiving_inputs,
582            results,
583            arg_roots: _,
584            ..
585        } = self;
586        std::iter::once(tx_context)
587            .chain(std::iter::once(gas))
588            .chain(object_inputs)
589            .chain(withdrawal_inputs)
590            .chain(pure_inputs)
591            .chain(receiving_inputs)
592            .chain(results.iter().flatten())
593            .filter_map(|v| match v.value.as_ref() {
594                Some(Value::Ref { node, .. }) => Some(*node),
595                Some(Value::NonRef) | None => None,
596            })
597    }
598
599    /// Returns whether any live reference may extend `node`. Excludes aliases when requested.
600    fn any_extends(&self, node: NodeID, ignore_aliases: bool) -> anyhow::Result<bool> {
601        for other in self.all_references() {
602            if (!ignore_aliases && other == node) || self.memory.extends(other, node)? {
603                return Ok(true);
604            }
605        }
606        Ok(false)
607    }

610/// Verifies memory safety using an alternative implementation of `verify::memory_safety`.
611///
612/// It represents memory locations and call returns as graph nodes. Root nodes represent known PTB
613/// locations (or in rare cases unknown values from calls without reference arguments). Each node
614/// tracks its transitive ancestors in a bitset. A call-return delta records its command identity,
615/// analogous to the regex verifier's `.*` extension, while retaining enough identity to use the
616/// guarantee that mutable references returned by one call do not overlap with other references
617/// returned from that call. Extensions from distinct commands are conservatively treated as
618/// potentially overlapping. PTBs have no control flow, which makes this representation sufficient
619/// as an invariant check for the regex implementation.
620/// Checks the following
621/// - Values are not used after being moved
622/// - Reference safety is upheld (no dangling references)
623pub fn verify<Mode: ExecutionMode>(
624    env: &Env<Mode>,
625    txn: &T::Transaction,
626) -> Result<(), Mode::Error> {
627    if !env.protocol_config.memory_safety_invariant_check_v2() {
628        return legacy::verify(env, txn);
629    }
630    Ok(verify_(env, txn).map_err(|e| make_invariant_violation!("{}. Transaction {:?}", e, txn))?)
631}

633pub(crate) fn verify_<Mode: ExecutionMode>(
634    env: &Env<Mode>,
635    txn: &T::Transaction,
636) -> anyhow::Result<()> {
637    let mut context = Context::new(env, txn)?;
638    let T::Transaction {
639        gas_payment: _,
640        bytes: _,
641        objects: _,
642        withdrawals: _,
643        pure: _,
644        receiving: _,
645        withdrawal_compatibility_conversions: _,
646        original_command_len: _,
647        commands,
648        unified_linkage: _,
649    } = txn;
650    for c in commands {
651        command(&mut context, c)?;
652    }
653    Ok(())
654}
655
656fn command(context: &mut Context, c: &T::Command) -> anyhow::Result<()> {
657    // process the command
658    debug_assert!(context.arg_roots.is_empty());
659    let results = command_(context, c)?;
660    // drop unused result values by marking them as `None`
661    assert_invariant!(
662        results.len() == c.value.drop_values.len(),
663        "result length mismatch. expected {}, got {}",
664        c.value.drop_values.len(),
665        results.len()
666    );
667    context.add_result_values(
668        results
669            .into_iter()
670            .zip_debug_eq(c.value.drop_values.iter().copied())
671            .map(|(v, drop)| if drop { None } else { Some(v) }),
672    )?;
673    context.arg_roots.clear();
674    Ok(())
675}
676
677fn command_(context: &mut Context, sp!(_, c): &T::Command) -> anyhow::Result<Vec<Value>> {
678    let result_tys = &c.result_type;
679    let results = match &c.command {
680        T::Command__::MoveCall(move_call) => {
681            let T::MoveCall {
682                function,
683                arguments,
684            } = &**move_call;
685            let arg_values = context.arguments(arguments)?;
686            call(context, &function.signature, arg_values)?
687        }
688        T::Command__::TransferObjects(objs, recipient) => {
689            context.arguments(objs)?;
690            context.argument(recipient)?;
691            non_ref_results(result_tys)?
692        }
693        T::Command__::SplitCoins(_, coin, amounts) => {
694            context.arguments(amounts)?;
695            let coin_value = context.argument(coin)?;
696            write_ref(context, coin_value)?;
697            non_ref_results(result_tys)?
698        }
699        T::Command__::MergeCoins(_, target, coins) => {
700            context.arguments(coins)?;
701            let target_value = context.argument(target)?;
702            write_ref(context, target_value)?;
703            non_ref_results(result_tys)?
704        }
705        T::Command__::MakeMoveVec(_, arguments) => {
706            context.arguments(arguments)?;
707            non_ref_results(result_tys)?
708        }
709        T::Command__::Publish(_, _, _) => non_ref_results(result_tys)?,
710        T::Command__::Upgrade(_, _, _, ticket, _) => {
711            context.argument(ticket)?;
712            non_ref_results(result_tys)?
713        }
714    };
715    assert_invariant!(
716        result_tys.len() == results.len(),
717        "result length mismatch. Expected {}, got {}",
718        result_tys.len(),
719        results.len()
720    );
721    Ok(results)
722}

724fn write_ref(context: &Context, value: Value) -> anyhow::Result<()> {
725    match value {
726        Value::NonRef => {
727            anyhow::bail!("Cannot write to a non-reference value");
728        }
729
730        Value::Ref { is_mut: false, .. } => {
731            anyhow::bail!("Cannot write to an immutable reference");
732        }
733        Value::Ref { is_mut: true, node } => {
734            anyhow::ensure!(
735                !context.any_extends(node, /* ignore alias */ true)?,
736                "Cannot write to a mutable reference that has extensions"
737            );
738            Ok(())
739        }
740    }
741}

743fn call(
744    context: &mut Context,
745    signature: &T::LoadedFunctionInstantiation,
746    arguments: Vec<Value>,
747) -> anyhow::Result<Vec<Value>> {
748    let return_ = &signature.return_;
749    let mut all_nodes = Vec::new();
750    let mut imm_nodes = Vec::new();
751    let mut mut_nodes = Vec::new();
752    for arg in arguments {
753        match arg {
754            Value::NonRef => (),
755            Value::Ref { is_mut: true, node } => {
756                anyhow::ensure!(
757                    !context.any_extends(node, /* ignore alias */ true)?,
758                    "Cannot transfer a mutable ref with extensions"
759                );
760                for other in &mut_nodes {
761                    anyhow::ensure!(
762                        context.memory.is_disjoint(*other, node)?,
763                        "Double mutable borrow"
764                    );
765                }
766                all_nodes.push(node);
767                mut_nodes.push(node);
768            }
769            Value::Ref {
770                is_mut: false,
771                node,
772            } => {
773                all_nodes.push(node);
774                imm_nodes.push(node);
775            }
776        }
777    }
778    // All mutable references must be disjoint from all immutable references
779    for immutable in &imm_nodes {
780        for mutable in &mut_nodes {
781            anyhow::ensure!(
782                context.memory.is_disjoint(*immutable, *mutable)?,
783                "Mutable and immutable borrows cannot overlap"
784            );
785        }
786    }
787    if context.allow_references_in_ptbs {
788        // `mut_nodes` is a subset of `all_nodes`, so all candidates are covered.
789        let mut tx_context_nodes = BTreeSet::new();
790        for node in &all_nodes {
791            if context
792                .memory
793                .known_roots(*node)?
794                .contains(&T::Location::TxContext)
795            {
796                tx_context_nodes.insert(*node);
797            }
798        }
799        mut_nodes.retain(|node| !tx_context_nodes.contains(node));
800        all_nodes.retain(|node| !tx_context_nodes.contains(node));
801    }
802    let command = context.current_command()?;
803    let mut_nodes = if mut_nodes.is_empty() {
804        vec![context.memory.root(RootLocation::Unknown { command })?]
805    } else {
806        mut_nodes
807    };
808    let all_nodes = if all_nodes.is_empty() {
809        vec![context.memory.root(RootLocation::Unknown { command })?]
810    } else {
811        all_nodes
812    };
813    return_
814        .iter()
815        .enumerate()
816        .map(|(i, ty)| {
817            let _result = checked_as!(i, u16)?;
818            Ok(match ty {
819                T::Type::Reference(/* is mut */ true, _) => Value::Ref {
820                    is_mut: true,
821                    node: context.memory.call_return(command, &mut_nodes)?,
822                },
823                T::Type::Reference(/* is mut */ false, _) => Value::Ref {
824                    is_mut: false,
825                    node: context.memory.call_return(command, &all_nodes)?,
826                },
827                _ => Value::NonRef,
828            })
829        })
830        .collect::<anyhow::Result<Vec<_>>>()
831}
832
833fn non_ref_results(results: &[T::Type]) -> anyhow::Result<Vec<Value>> {
834    results
835        .iter()
836        .map(|t| {
837            anyhow::ensure!(
838                !matches!(t, T::Type::Reference(_, _)),
839                "attempted to create a non-reference result from a reference type",
840            );
841            Ok(Value::NonRef)
842        })
843        .collect()
844}

857    /// A dot-star like extension, but with a unique identifier. Deltas can be compared between
858    /// different Deltas of the same command, otherwise they behave like .* in the regex based
859    /// implementation. This means that it represents an arbitrary field extension of the reference
860    /// in question. However, due to invariants within reference safety, for mutable references these
861    /// extensions cannot with other references from the same command.
862    #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
863    struct Delta {
864        command: u16,
865        result: u16,
866    }
867
868    #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
869    enum RootLocation {
870        /// The result of a command, specifically a `MoveCall`, without any input references.
871        /// These calls will always abort, but we still must track them.
872        Unknown {
873            command: u16,
874        },
875        Known(T::Location),
876    }
877
878    /// A path points to an abstract memory location, rooted in an input or command result. Any
879    /// extension is the result from a reference being returned from a command.
880    #[derive(Debug, Clone, PartialEq, Eq, Hash)]
881    struct Path {
882        root: RootLocation,
883        extensions: Vec<Delta>,
884    }
885
886    #[derive(Debug)]
887    struct PathSet(IndexSet<Path>);
888
889    #[derive(Debug)]
890    enum Value {
891        NonRef,
892        Ref { is_mut: bool, paths: Rc<PathSet> },
893    }
894
895    #[derive(Debug)]
896    struct Location {
897        // A singleton set pointing to the location itself
898        self_path: Rc<PathSet>,
899        value: Option<Value>,
900    }

917    enum PathComparison {
918        /// `self` is a strict prefix of `other`
919        Prefix,
920        /// `self` and `other` are the same path
921        Aliases,
922        /// `self` extends `other`
923        Extends,
924        /// `self` and `other` point to distinct regions of memory. They might however be rooted
925        /// in the same parent region, i.e. the same `root` location or a prefix of the same
926        /// `extensions`
927        Disjoint,
928    }
929
930    impl Path {
931        fn initial(location: T::Location) -> Self {
932            Self {
933                root: RootLocation::Known(location),
934                extensions: vec![],
935            }
936        }
937
938        /// See `PathComparison` for the meaning of the return value
939        fn compare(&self, other: &Self) -> PathComparison {
940            if self.root != other.root {
941                return PathComparison::Disjoint;
942            };
943            let mut self_extensions = self.extensions.iter();
944            let mut other_extensions = other.extensions.iter();
945            loop {
946                match (self_extensions.next(), other_extensions.next()) {
947                    (Some(self_ext), Some(other_ext)) => {
948                        if self_ext.command != other_ext.command {
949                            // Cannot compare `Delta` from distinct commands, as such we do assume
950                            // the possibility that `self` extends `other`
951                            return PathComparison::Extends;
952                        }
953                        if self_ext.result != other_ext.result {
954                            // If the command is the same, but the result is different, we know that
955                            // they must be disjoint. Or they are immutable references, in which case
956                            // we do not care.
957                            return PathComparison::Disjoint;
958                        }
959                    }
960                    (None, Some(_)) => return PathComparison::Prefix,
961                    (Some(_), None) => return PathComparison::Extends,
962                    (None, None) => return PathComparison::Aliases,
963                }
964            }
965        }
966
967        /// Create a new `Path` that extends the current path with the given `extension`.
968        fn extend(&self, extension: Delta) -> Self {
969            let mut new_extensions = self.extensions.clone();
970            new_extensions.push(extension);
971            Self {
972                root: self.root,
973                extensions: new_extensions,
974            }
975        }
976    }
977
978    impl PathSet {
979        /// Should be used only for `call` for creating initial path sets.
980        fn empty() -> Self {
981            Self(IndexSet::new())
982        }
983
984        fn initial(location: T::Location) -> Self {
985            Self(IndexSet::from([Path::initial(location)]))
986        }
987
988        fn unknown_root(command: u16) -> Self {
989            Self(IndexSet::from([Path {
990                root: RootLocation::Unknown { command },
991                extensions: vec![],
992            }]))
993        }
994
995        fn is_empty(&self) -> bool {
996            self.0.is_empty()
997        }
998
999        /// Returns true if any path in `self` `Extends` with any path in `other`.
1000        /// Excludes `Aliases` if `ignore_aliases` is true.
1001        fn extends(&self, other: &Self, ignore_aliases: bool) -> bool {
1002            self.0.iter().any(|self_path| {
1003                other
1004                    .0
1005                    .iter()
1006                    .any(|other_path| match self_path.compare(other_path) {
1007                        PathComparison::Prefix | PathComparison::Disjoint => false,
1008                        PathComparison::Aliases => !ignore_aliases,
1009                        PathComparison::Extends => true,
1010                    })
1011            })
1012        }
1013
1014        /// Returns true if all paths in `self` are `Disjoint` with all paths in `other`.
1015        fn is_disjoint(&self, other: &Self) -> bool {
1016            self.0.iter().all(|self_path| {
1017                other
1018                    .0
1019                    .iter()
1020                    .all(|other_path| match self_path.compare(other_path) {
1021                        PathComparison::Disjoint => true,
1022                        PathComparison::Prefix
1023                        | PathComparison::Aliases
1024                        | PathComparison::Extends => false,
1025                    })
1026            })
1027        }
1028
1029        /// Insert all paths from `other` into `self`.
1030        fn union(&mut self, other: &PathSet) {
1031            // We might be able to optimize this slightly by not including paths that are extensions
1032            // of existing paths
1033            self.0.extend(other.0.iter().cloned());
1034        }
1035
1036        /// Create a new `PathSet` where all paths in `self` are extended with the given `extension`.
1037        fn extend(&self, extension: Delta) -> Self {
1038            let mut new_paths = IndexSet::with_capacity(self.0.len());
1039            for path in &self.0 {
1040                new_paths.insert(path.extend(extension));
1041            }
1042            Self(new_paths)
1043        }
1044    }

1203        fn add_result_values(
1204            &mut self,
1205            results: impl IntoIterator<Item = Option<Value>>,
1206        ) -> anyhow::Result<()> {
1207            let command = self.current_command()?;
1208            let allow_references_in_ptbs = self.allow_references_in_ptbs;
1209            self.results.push(
1210                results
1211                    .into_iter()
1212                    .enumerate()
1213                    .map(|(i, v)| {
1214                        // Post-condition of the source strip in `call`: for path sets, `TxContext`
1215                        // can never be returned. Gated because flag-off dev-inspect legitimately
1216                        // produces ctx-rooted results.
1217                        if allow_references_in_ptbs && let Some(Value::Ref { paths, .. }) = &v {
1218                            anyhow::ensure!(
1219                                paths
1220                                    .0
1221                                    .iter()
1222                                    .all(|p| p.root != RootLocation::Known(T::Location::TxContext)),
1223                                "TxContext can never be returned"
1224                            );
1225                        }
1226                        Ok(Location {
1227                            self_path: Rc::new(PathSet::initial(T::Location::Result(
1228                                command,
1229                                checked_as!(i, u16)?,
1230                            ))),
1231                            value: v,
1232                        })
1233                    })
1234                    .collect::<anyhow::Result<_>>()?,
1235            );
1236            Ok(())
1237        }

1295        fn check_usage(&self, usage: &T::Usage, location: &Location) -> anyhow::Result<()> {
1296            // by marking "ignore alias" as `false`, we will also check for `Alias` paths, i.e. paths
1297            // that point to the location itself without any extensions.
1298            let is_borrowed = self.any_extends(&location.self_path, /* ignore alias */ false)
1299                || self.arg_roots.contains(&usage.location());
1300            match usage {
1301                T::Usage::Move(_) => {
1302                    anyhow::ensure!(!is_borrowed, "Cannot move a value that is borrowed");
1303                }
1304                T::Usage::Copy { borrowed, .. } => {
1305                    let Some(borrowed) = borrowed.get().copied() else {
1306                        anyhow::bail!("Borrowed flag not set for copy usage");
1307                    };
1308                    anyhow::ensure!(
1309                        borrowed == is_borrowed,
1310                        "Borrowed flag mismatch for copy usage: expected {borrowed}, got {is_borrowed} \
1311                    location {:?} for in command {}",
1312                        location.self_path,
1313                        self.current_command()?
1314                    );
1315                }
1316            }
1317            Ok(())
1318        }
1319
1320        fn argument(&mut self, sp!(_, (arg, _)): &T::Argument) -> anyhow::Result<Value> {
1321            let location = self.location(arg.location())?;
1322            match arg {
1323                T::Argument__::Use(usage)
1324                | T::Argument__::Freeze(usage)
1325                | T::Argument__::Read(usage) => self.check_usage(usage, location)?,
1326                T::Argument__::Borrow(_, _) => (),
1327            };
1328            let location = self.location_mut(arg.location())?;
1329            let value = match arg {
1330                T::Argument__::Use(usage) => location.use_(usage)?,
1331                T::Argument__::Freeze(usage) => location.use_(usage)?.freeze()?,
1332                T::Argument__::Borrow(is_mut, _) => location.borrow(*is_mut)?,
1333                T::Argument__::Read(usage) => {
1334                    location.use_(usage)?;
1335                    Value::NonRef
1336                }
1337            };
1338            if let Value::Ref { paths, .. } = &value {
1339                for p in &paths.0 {
1340                    match p.root {
1341                        RootLocation::Unknown { .. } => (),
1342                        RootLocation::Known(location) => {
1343                            self.arg_roots.insert(location);
1344                        }
1345                    }
1346                }
1347            }
1348            Ok(value)
1349        }
1350
1351        fn arguments(&mut self, args: &[T::Argument]) -> anyhow::Result<Vec<Value>> {
1352            args.iter()
1353                .map(|arg| self.argument(arg))
1354                .collect::<anyhow::Result<Vec<_>>>()
1355        }

1357        fn all_references(&self) -> impl Iterator<Item = Rc<PathSet>> {
1358            let Self {
1359                allow_references_in_ptbs: _,
1360                tx_context,
1361                gas,
1362                object_inputs,
1363                withdrawal_inputs,
1364                pure_inputs,
1365                receiving_inputs,
1366                results,
1367                arg_roots: _,
1368            } = self;
1369            std::iter::once(tx_context)
1370                .chain(std::iter::once(gas))
1371                .chain(object_inputs)
1372                .chain(withdrawal_inputs)
1373                .chain(pure_inputs)
1374                .chain(receiving_inputs)
1375                .chain(results.iter().flatten())
1376                .filter_map(|v| -> Option<Rc<PathSet>> {
1377                    match v.value.as_ref() {
1378                        Some(Value::Ref { paths, .. }) => Some(paths.clone()),
1379                        Some(Value::NonRef) | None => None,
1380                    }
1381                })
1382        }
1383
1384        /// Returns true if any of the references in a given `T::Location` extends a path in `paths`.
1385        /// Excludes `Aliases` if `ignore_aliases` is true.
1386        fn any_extends(&self, paths: &PathSet, ignore_aliases: bool) -> bool {
1387            self.all_references()
1388                .any(|other| other.extends(paths, ignore_aliases))
1389        }
1390    }

1392    /// Verifies memory safety of a transaction. This is a re-implementation of `verify::memory_safety`
1393    /// using an alternative approach given the newness of the Regex based borrow graph in that
1394    /// implementation.
1395    /// This is a set based approach were each reference is represent as a set of paths. A path
1396    /// is has a root (basically a `T::Location` plus some edge case massaging) and a list of extensions
1397    /// resulting from Move function calls. Each one of those Move calls gets a `Delta` extension for
1398    /// each return value. The `Delta` is like the ".*" in the regex based implementation but where it
1399    /// carries a sense of identity. This identity allows for invariants from the return values of the
1400    /// Move call to be leveraged. For example, mutable references returned from a call cannot overlap.
1401    /// If we just used ".*", we would not be able to express this invariant without some sense of
1402    /// identity for the reference itself (which is what is going on in the Regex based implementation).
1403    /// This implementation stems from research work for the Move borrow checker, but would normally
1404    /// not be expressive enough in the presence of control flow. Luckily, PTBs do not have control flow
1405    /// so we can use this approach as a safety net for the Regex based implementation until that
1406    /// code is sufficiently. tested and hardened.
1407    /// Strip TxContext input arguments so that they do not flow as inputs
1408    /// Checks the following
1409    /// - Values are not used after being moved
1410    /// - Reference safety is upheld (no dangling references)
1411    pub fn verify<Mode: ExecutionMode>(
1412        env: &Env<Mode>,
1413        txn: &T::Transaction,
1414    ) -> Result<(), Mode::Error> {
1415        Ok(verify_(env, txn)
1416            .map_err(|e| make_invariant_violation!("{}. Transaction {:?}", e, txn))?)
1417    }
1418
1419    fn verify_<Mode: ExecutionMode>(env: &Env<Mode>, txn: &T::Transaction) -> anyhow::Result<()> {
1420        if env
1421            .protocol_config
1422            .max_ptb_live_references_as_option()
1423            .is_some()
1424        {
1425            return Ok(());
1426        }
1427        let mut context = Context::new(env, txn)?;
1428        let T::Transaction {
1429            gas_payment: _,
1430            bytes: _,
1431            objects: _,
1432            withdrawals: _,
1433            pure: _,
1434            receiving: _,
1435            withdrawal_compatibility_conversions: _,
1436            original_command_len: _,
1437            commands,
1438            unified_linkage: _,
1439        } = txn;
1440        for c in commands {
1441            command(&mut context, c)?;
1442        }
1443        Ok(())
1444    }

1446    fn command(context: &mut Context, c: &T::Command) -> anyhow::Result<()> {
1447        // process the command
1448        debug_assert!(context.arg_roots.is_empty());
1449        let results = command_(context, c)?;
1450        // drop unused result values by marking them as `None`
1451        assert_invariant!(
1452            results.len() == c.value.drop_values.len(),
1453            "result length mismatch. expected {}, got {}",
1454            c.value.drop_values.len(),
1455            results.len()
1456        );
1457        context.add_result_values(
1458            results
1459                .into_iter()
1460                .zip_debug_eq(c.value.drop_values.iter().copied())
1461                .map(|(v, drop)| if drop { None } else { Some(v) }),
1462        )?;
1463        context.arg_roots.clear();
1464        Ok(())
1465    }
1466
1467    fn command_(context: &mut Context, sp!(_, c): &T::Command) -> anyhow::Result<Vec<Value>> {
1468        let result_tys = &c.result_type;
1469        let results = match &c.command {
1470            T::Command__::MoveCall(move_call) => {
1471                let T::MoveCall {
1472                    function,
1473                    arguments,
1474                } = &**move_call;
1475                let arg_values = context.arguments(arguments)?;
1476                call(context, &function.signature, arg_values)?
1477            }
1478            T::Command__::TransferObjects(objs, recipient) => {
1479                context.arguments(objs)?;
1480                context.argument(recipient)?;
1481                non_ref_results(result_tys)?
1482            }
1483            T::Command__::SplitCoins(_, coin, amounts) => {
1484                context.arguments(amounts)?;
1485                let coin_value = context.argument(coin)?;
1486                write_ref(context, coin_value)?;
1487                non_ref_results(result_tys)?
1488            }
1489            T::Command__::MergeCoins(_, target, coins) => {
1490                context.arguments(coins)?;
1491                let target_value = context.argument(target)?;
1492                write_ref(context, target_value)?;
1493                non_ref_results(result_tys)?
1494            }
1495            T::Command__::MakeMoveVec(_, arguments) => {
1496                context.arguments(arguments)?;
1497                non_ref_results(result_tys)?
1498            }
1499            T::Command__::Publish(_, _, _) => non_ref_results(result_tys)?,
1500            T::Command__::Upgrade(_, _, _, ticket, _) => {
1501                context.argument(ticket)?;
1502                non_ref_results(result_tys)?
1503            }
1504        };
1505        assert_invariant!(
1506            result_tys.len() == results.len(),
1507            "result length mismatch. Expected {}, got {}",
1508            result_tys.len(),
1509            results.len()
1510        );
1511        Ok(results)
1512    }

1514    fn write_ref(context: &Context, value: Value) -> anyhow::Result<()> {
1515        match value {
1516            Value::NonRef => {
1517                anyhow::bail!("Cannot write to a non-reference value");
1518            }
1519
1520            Value::Ref { is_mut: false, .. } => {
1521                anyhow::bail!("Cannot write to an immutable reference");
1522            }
1523            Value::Ref {
1524                is_mut: true,
1525                paths,
1526            } => {
1527                anyhow::ensure!(
1528                    !context.any_extends(&paths, /* ignore alias */ true),
1529                    "Cannot write to a mutable reference that has extensions"
1530                );
1531                Ok(())
1532            }
1533        }
1534    }

1536    fn call(
1537        context: &mut Context,
1538        signature: &T::LoadedFunctionInstantiation,
1539        arguments: Vec<Value>,
1540    ) -> anyhow::Result<Vec<Value>> {
1541        let return_ = &signature.return_;
1542        let mut all_paths: PathSet = PathSet::empty();
1543        let mut imm_paths: PathSet = PathSet::empty();
1544        let mut mut_paths: PathSet = PathSet::empty();
1545        for arg in arguments {
1546            match arg {
1547                Value::NonRef => (),
1548                Value::Ref {
1549                    is_mut: true,
1550                    paths,
1551                } => {
1552                    // Allow alias conflicts with references not passed as arguments
1553                    anyhow::ensure!(
1554                        !context.any_extends(&paths, /* ignore alias */ true),
1555                        "Cannot transfer a mutable ref with extensions"
1556                    );
1557                    // All mutable argument references must be disjoint from all other references
1558                    anyhow::ensure!(mut_paths.is_disjoint(&paths), "Double mutable borrow");
1559                    all_paths.union(&paths);
1560                    mut_paths.union(&paths);
1561                }
1562                Value::Ref {
1563                    is_mut: false,
1564                    paths,
1565                } => {
1566                    all_paths.union(&paths);
1567                    imm_paths.union(&paths);
1568                }
1569            }
1570        }
1571        // All mutable references must be disjoint from all immutable references
1572        anyhow::ensure!(
1573            imm_paths.is_disjoint(&mut_paths),
1574            "Mutable and immutable borrows cannot overlap"
1575        );
1576        // With references enabled in PTBs, TxContext still cannot be the
1577        // root of a returned reference. We ensure this by removing them from
1578        // any possible input.
1579        if context.allow_references_in_ptbs {
1580            let is_ctx = |p: &Path| p.root == RootLocation::Known(T::Location::TxContext);
1581            mut_paths.0.retain(|p| !is_ctx(p));
1582            all_paths.0.retain(|p| !is_ctx(p));
1583        }
1584        let command = context.current_command()?;
1585        let mut_paths = if mut_paths.is_empty() {
1586            PathSet::unknown_root(command)
1587        } else {
1588            mut_paths
1589        };
1590        let all_paths = if all_paths.is_empty() {
1591            PathSet::unknown_root(command)
1592        } else {
1593            all_paths
1594        };
1595        return_
1596            .iter()
1597            .enumerate()
1598            .map(|(i, ty)| {
1599                let delta = Delta {
1600                    command,
1601                    result: checked_as!(i, u16)?,
1602                };
1603                match ty {
1604                    T::Type::Reference(/* is mut */ true, _) => {
1605                        Value::ref_(true, mut_paths.extend(delta))
1606                    }
1607                    T::Type::Reference(/* is mut */ false, _) => {
1608                        Value::ref_(false, all_paths.extend(delta))
1609                    }
1610                    _ => Ok(Value::NonRef),
1611                }
1612            })
1613            .collect::<anyhow::Result<Vec<_>>>()
1614    }
1615
1616    fn non_ref_results(results: &[T::Type]) -> anyhow::Result<Vec<Value>> {
1617        results
1618            .iter()
1619            .map(|t| {
1620                anyhow::ensure!(
1621                    !matches!(t, T::Type::Reference(_, _)),
1622                    "attempted to create a non-reference result from a reference type",
1623                );
1624                Ok(Value::NonRef)
1625            })
1626            .collect()
1627    }
