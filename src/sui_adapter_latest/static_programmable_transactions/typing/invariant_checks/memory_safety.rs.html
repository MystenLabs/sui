14/// A dot-star like extension, but with a unique identifier. Deltas can be compared between
15/// different Deltas of the same command, otherwise they behave like .* in the regex based
16/// implementation. This means that it represents an arbitrary field extension of the reference
17/// in question. However, due to invariants within reference safety, for mutable references these
18/// extensions cannot with other references from the same command.
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
20struct Delta {
21    command: u16,
22    result: u16,
23}
24
25#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
26enum RootLocation {
27    /// The result of a command, specifically a `MoveCall`, without any input references.
28    /// These calls will always abort, but we still must track them.
29    Unknown {
30        command: u16,
31    },
32    Known(T::Location),
33}
34
35/// A path points to an abstract memory location, rooted in an input or command result. Any
36/// extension is the result from a reference being returned from a command.
37#[derive(Debug, Clone, PartialEq, Eq, Hash)]
38struct Path {
39    root: RootLocation,
40    extensions: Vec<Delta>,
41}
42
43#[derive(Debug)]
44struct PathSet(IndexSet<Path>);
45
46#[derive(Debug)]
47enum Value {
48    NonRef,
49    Ref { is_mut: bool, paths: Rc<PathSet> },
50}
51
52#[derive(Debug)]
53struct Location {
54    // A singleton set pointing to the location itself
55    self_path: Rc<PathSet>,
56    value: Option<Value>,
57}
58
59#[derive(Debug)]
60struct Context {
61    allow_references_in_ptbs: bool,
62    tx_context: Location,
63    gas: Location,
64    object_inputs: Vec<Location>,
65    withdrawal_inputs: Vec<Location>,
66    pure_inputs: Vec<Location>,
67    receiving_inputs: Vec<Location>,
68    results: Vec<Vec<Location>>,
69    // Temporary set of locations borrowed by arguments seen thus far for the current command.
70    // Used exclusively for checking the validity copy/move.
71    arg_roots: IndexSet<T::Location>,
72}
73
74enum PathComparison {
75    /// `self` is a strict prefix of `other`
76    Prefix,
77    /// `self` and `other` are the same path
78    Aliases,
79    /// `self` extends `other`
80    Extends,
81    /// `self` and `other` point to distinct regions of memory. They might however be rooted
82    /// in the same parent region, i.e. the same `root` location or a prefix of the same
83    /// `extensions`
84    Disjoint,
85}
86
87impl Path {
88    fn initial(location: T::Location) -> Self {
89        Self {
90            root: RootLocation::Known(location),
91            extensions: vec![],
92        }
93    }
94
95    /// See `PathComparison` for the meaning of the return value
96    fn compare(&self, other: &Self) -> PathComparison {
97        if self.root != other.root {
98            return PathComparison::Disjoint;
99        };
100        let mut self_extensions = self.extensions.iter();
101        let mut other_extensions = other.extensions.iter();
102        loop {
103            match (self_extensions.next(), other_extensions.next()) {
104                (Some(self_ext), Some(other_ext)) => {
105                    if self_ext.command != other_ext.command {
106                        // Cannot compare `Delta` from distinct commands, as such we do assume
107                        // the possibility that `self` extends `other`
108                        return PathComparison::Extends;
109                    }
110                    if self_ext.result != other_ext.result {
111                        // If the command is the same, but the result is different, we know that
112                        // they must be disjoint. Or they are immutable references, in which case
113                        // we do not care.
114                        return PathComparison::Disjoint;
115                    }
116                }
117                (None, Some(_)) => return PathComparison::Prefix,
118                (Some(_), None) => return PathComparison::Extends,
119                (None, None) => return PathComparison::Aliases,
120            }
121        }
122    }
123
124    /// Create a new `Path` that extends the current path with the given `extension`.
125    fn extend(&self, extension: Delta) -> Self {
126        let mut new_extensions = self.extensions.clone();
127        new_extensions.push(extension);
128        Self {
129            root: self.root,
130            extensions: new_extensions,
131        }
132    }
133}

135impl PathSet {
136    /// Should be used only for `call` for creating initial path sets.
137    fn empty() -> Self {
138        Self(IndexSet::new())
139    }
140
141    fn initial(location: T::Location) -> Self {
142        Self(IndexSet::from([Path::initial(location)]))
143    }
144
145    fn unknown_root(command: u16) -> Self {
146        Self(IndexSet::from([Path {
147            root: RootLocation::Unknown { command },
148            extensions: vec![],
149        }]))
150    }
151
152    fn is_empty(&self) -> bool {
153        self.0.is_empty()
154    }
155
156    /// Returns true if any path in `self` `Extends` with any path in `other`.
157    /// Excludes `Aliases` if `ignore_aliases` is true.
158    fn extends(&self, other: &Self, ignore_aliases: bool) -> bool {
159        self.0.iter().any(|self_path| {
160            other
161                .0
162                .iter()
163                .any(|other_path| match self_path.compare(other_path) {
164                    PathComparison::Prefix | PathComparison::Disjoint => false,
165                    PathComparison::Aliases => !ignore_aliases,
166                    PathComparison::Extends => true,
167                })
168        })
169    }
170
171    /// Returns true if all paths in `self` are `Disjoint` with all paths in `other`.
172    fn is_disjoint(&self, other: &Self) -> bool {
173        self.0.iter().all(|self_path| {
174            other
175                .0
176                .iter()
177                .all(|other_path| match self_path.compare(other_path) {
178                    PathComparison::Disjoint => true,
179                    PathComparison::Prefix | PathComparison::Aliases | PathComparison::Extends => {
180                        false
181                    }
182                })
183        })
184    }
185
186    /// Insert all paths from `other` into `self`.
187    fn union(&mut self, other: &PathSet) {
188        // We might be able to optimize this slightly by not including paths that are extensions
189        // of existing paths
190        self.0.extend(other.0.iter().cloned());
191    }
192
193    /// Create a new `PathSet` where all paths in `self` are extended with the given `extension`.
194    fn extend(&self, extension: Delta) -> Self {
195        let mut new_paths = IndexSet::with_capacity(self.0.len());
196        for path in &self.0 {
197            new_paths.insert(path.extend(extension));
198        }
199        Self(new_paths)
200    }
201}

203impl Value {
204    /// Create a new reference value
205    fn ref_(is_mut: bool, paths: PathSet) -> anyhow::Result<Value> {
206        anyhow::ensure!(
207            !paths.is_empty(),
208            "Cannot create a reference with an empty path set"
209        );
210        Ok(Value::Ref {
211            is_mut,
212            paths: Rc::new(paths),
213        })
214    }
215
216    fn copy(&self) -> Value {
217        match self {
218            Value::NonRef => Value::NonRef,
219            Value::Ref { is_mut, paths } => Value::Ref {
220                is_mut: *is_mut,
221                paths: paths.clone(),
222            },
223        }
224    }
225
226    fn freeze(&mut self) -> anyhow::Result<Value> {
227        let copied = self.copy();
228        match copied {
229            Value::NonRef => {
230                anyhow::bail!("Cannot freeze a non-reference value")
231            }
232            Value::Ref { is_mut, paths } => {
233                anyhow::ensure!(is_mut, "Cannot freeze an immutable reference");
234                Ok(Value::Ref {
235                    is_mut: false,
236                    paths,
237                })
238            }
239        }
240    }
241}

243impl Location {
244    fn non_ref(location: T::Location) -> Self {
245        Self {
246            self_path: Rc::new(PathSet::initial(location)),
247            value: Some(Value::NonRef),
248        }
249    }
250
251    fn copy_value(&self) -> anyhow::Result<Value> {
252        let Some(value) = self.value.as_ref() else {
253            anyhow::bail!("Use of invalid memory location")
254        };
255        Ok(value.copy())
256    }
257
258    fn move_value(&mut self) -> anyhow::Result<Value> {
259        let Some(value) = self.value.take() else {
260            anyhow::bail!("Use of invalid memory location")
261        };
262        Ok(value)
263    }
264
265    fn use_(&mut self, usage: &T::Usage) -> anyhow::Result<Value> {
266        match usage {
267            T::Usage::Move(_) => self.move_value(),
268            T::Usage::Copy { .. } => self.copy_value(),
269        }
270    }
271
272    fn borrow(&mut self, is_mut: bool) -> anyhow::Result<Value> {
273        let Some(value) = self.value.as_ref() else {
274            anyhow::bail!("Borrow of invalid memory location")
275        };
276        match value {
277            Value::Ref { .. } => {
278                anyhow::bail!("Cannot borrow a reference")
279            }
280            Value::NonRef => {
281                anyhow::ensure!(
282                    !self.self_path.is_empty(),
283                    "Cannot have an empty location to borrow from"
284                );
285                // a new reference that borrows from this location
286                Ok(Value::Ref {
287                    is_mut,
288                    paths: self.self_path.clone(),
289                })
290            }
291        }
292    }
293}

295impl Context {
296    fn new<Mode: ExecutionMode>(env: &Env<Mode>, txn: &T::Transaction) -> anyhow::Result<Self> {
297        let T::Transaction {
298            gas_payment,
299            bytes: _,
300            objects,
301            withdrawals,
302            pure,
303            receiving,
304            withdrawal_compatibility_conversions: _,
305            original_command_len: _,
306            commands: _,
307            unified_linkage: _,
308        } = txn;
309        let tx_context = Location::non_ref(T::Location::TxContext);
310        let mut gas = Location::non_ref(T::Location::GasCoin);
311        if gas_payment.is_none() {
312            gas.move_value()
313                .map_err(|_| anyhow::anyhow!("gas coin should be initialized"))?;
314        }
315        let object_inputs = (0..objects.len())
316            .map(|i| {
317                Ok(Location::non_ref(T::Location::ObjectInput(checked_as!(
318                    i, u16
319                )?)))
320            })
321            .collect::<Result<_, ExecutionError>>()?;
322        let withdrawal_inputs = (0..withdrawals.len())
323            .map(|i| {
324                Ok(Location::non_ref(T::Location::WithdrawalInput(
325                    checked_as!(i, u16)?,
326                )))
327            })
328            .collect::<Result<_, ExecutionError>>()?;
329        let pure_inputs = (0..pure.len())
330            .map(|i| {
331                Ok(Location::non_ref(T::Location::PureInput(checked_as!(
332                    i, u16
333                )?)))
334            })
335            .collect::<Result<_, ExecutionError>>()?;
336        let receiving_inputs = (0..receiving.len())
337            .map(|i| {
338                Ok(Location::non_ref(T::Location::ReceivingInput(checked_as!(
339                    i, u16
340                )?)))
341            })
342            .collect::<Result<_, ExecutionError>>()?;
343        Ok(Self {
344            allow_references_in_ptbs: env.protocol_config.allow_references_in_ptbs(),
345            tx_context,
346            gas,
347            object_inputs,
348            withdrawal_inputs,
349            pure_inputs,
350            receiving_inputs,
351            results: vec![],
352            arg_roots: IndexSet::new(),
353        })
354    }

356    fn current_command(&self) -> anyhow::Result<u16> {
357        Ok(checked_as!(self.results.len(), u16)?)
358    }
359
360    fn add_result_values(
361        &mut self,
362        results: impl IntoIterator<Item = Option<Value>>,
363    ) -> anyhow::Result<()> {
364        let command = self.current_command()?;
365        let allow_references_in_ptbs = self.allow_references_in_ptbs;
366        self.results.push(
367            results
368                .into_iter()
369                .enumerate()
370                .map(|(i, v)| {
371                    // Post-condition of the source strip in `call`: for path sets, `TxContext`
372                    // can never be returned. Gated because flag-off dev-inspect legitimately
373                    // produces ctx-rooted results.
374                    if allow_references_in_ptbs && let Some(Value::Ref { paths, .. }) = &v {
375                        anyhow::ensure!(
376                            paths
377                                .0
378                                .iter()
379                                .all(|p| p.root != RootLocation::Known(T::Location::TxContext)),
380                            "TxContext can never be returned"
381                        );
382                    }
383                    Ok(Location {
384                        self_path: Rc::new(PathSet::initial(T::Location::Result(
385                            command,
386                            checked_as!(i, u16)?,
387                        ))),
388                        value: v,
389                    })
390                })
391                .collect::<anyhow::Result<_>>()?,
392        );
393        Ok(())
394    }

396    fn location(&self, loc: T::Location) -> anyhow::Result<&Location> {
397        Ok(match loc {
398            T::Location::TxContext => &self.tx_context,
399            T::Location::GasCoin => &self.gas,
400            T::Location::ObjectInput(i) => self
401                .object_inputs
402                .get(i as usize)
403                .ok_or_else(|| anyhow::anyhow!("Object input index out of bounds {i}"))?,
404            T::Location::WithdrawalInput(i) => self
405                .withdrawal_inputs
406                .get(i as usize)
407                .ok_or_else(|| anyhow::anyhow!("Withdrawal input index out of bounds {i}"))?,
408            T::Location::PureInput(i) => self
409                .pure_inputs
410                .get(i as usize)
411                .ok_or_else(|| anyhow::anyhow!("Pure input index out of bounds {i}"))?,
412            T::Location::ReceivingInput(i) => self
413                .receiving_inputs
414                .get(i as usize)
415                .ok_or_else(|| anyhow::anyhow!("Receiving input index out of bounds {i}"))?,
416            T::Location::Result(i, j) => self
417                .results
418                .get(i as usize)
419                .and_then(|r| r.get(j as usize))
420                .ok_or_else(|| anyhow::anyhow!("Result index out of bounds ({i},{j})"))?,
421        })
422    }

436            T::Location::PureInput(i) => self

442                .get_mut(i as usize)
443                .ok_or_else(|| anyhow::anyhow!("Receiving input index out of bounds {i}"))?,
444            T::Location::Result(i, j) => self

446                .get_mut(i as usize)

452    fn check_usage(&self, usage: &T::Usage, location: &Location) -> anyhow::Result<()> {
453        // by marking "ignore alias" as `false`, we will also check for `Alias` paths, i.e. paths
454        // that point to the location itself without any extensions.
455        let is_borrowed = self.any_extends(&location.self_path, /* ignore alias */ false)
456            || self.arg_roots.contains(&usage.location());
457        match usage {
458            T::Usage::Move(_) => {
459                anyhow::ensure!(!is_borrowed, "Cannot move a value that is borrowed");
460            }
461            T::Usage::Copy { borrowed, .. } => {
462                let Some(borrowed) = borrowed.get().copied() else {
463                    anyhow::bail!("Borrowed flag not set for copy usage");
464                };
465                anyhow::ensure!(
466                    borrowed == is_borrowed,
467                    "Borrowed flag mismatch for copy usage: expected {borrowed}, got {is_borrowed} \
468                    location {:?} for in command {}",
469                    location.self_path,
470                    self.current_command()?
471                );
472            }
473        }
474        Ok(())
475    }

477    fn argument(&mut self, sp!(_, (arg, _)): &T::Argument) -> anyhow::Result<Value> {
478        let location = self.location(arg.location())?;
479        match arg {
480            T::Argument__::Use(usage)
481            | T::Argument__::Freeze(usage)
482            | T::Argument__::Read(usage) => self.check_usage(usage, location)?,
483            T::Argument__::Borrow(_, _) => (),
484        };
485        let location = self.location_mut(arg.location())?;
486        let value = match arg {
487            T::Argument__::Use(usage) => location.use_(usage)?,
488            T::Argument__::Freeze(usage) => location.use_(usage)?.freeze()?,
489            T::Argument__::Borrow(is_mut, _) => location.borrow(*is_mut)?,
490            T::Argument__::Read(usage) => {
491                location.use_(usage)?;
492                Value::NonRef
493            }
494        };
495        if let Value::Ref { paths, .. } = &value {
496            for p in &paths.0 {
497                match p.root {
498                    RootLocation::Unknown { .. } => (),
499                    RootLocation::Known(location) => {
500                        self.arg_roots.insert(location);
501                    }
502                }
503            }
504        }
505        Ok(value)
506    }

508    fn arguments(&mut self, args: &[T::Argument]) -> anyhow::Result<Vec<Value>> {
509        args.iter()
510            .map(|arg| self.argument(arg))
511            .collect::<anyhow::Result<Vec<_>>>()
512    }

514    fn all_references(&self) -> impl Iterator<Item = Rc<PathSet>> {
515        let Self {
516            allow_references_in_ptbs: _,
517            tx_context,
518            gas,
519            object_inputs,
520            withdrawal_inputs,
521            pure_inputs,
522            receiving_inputs,
523            results,
524            arg_roots: _,
525        } = self;
526        std::iter::once(tx_context)
527            .chain(std::iter::once(gas))
528            .chain(object_inputs)
529            .chain(withdrawal_inputs)
530            .chain(pure_inputs)
531            .chain(receiving_inputs)
532            .chain(results.iter().flatten())
533            .filter_map(|v| -> Option<Rc<PathSet>> {
534                match v.value.as_ref() {
535                    Some(Value::Ref { paths, .. }) => Some(paths.clone()),
536                    Some(Value::NonRef) | None => None,
537                }
538            })
539    }

541    /// Returns true if any of the references in a given `T::Location` extends a path in `paths`.
542    /// Excludes `Aliases` if `ignore_aliases` is true.
543    fn any_extends(&self, paths: &PathSet, ignore_aliases: bool) -> bool {
544        self.all_references()
545            .any(|other| other.extends(paths, ignore_aliases))
546    }

549/// Verifies memory safety of a transaction. This is a re-implementation of `verify::memory_safety`
550/// using an alternative approach given the newness of the Regex based borrow graph in that
551/// implementation.
552/// This is a set based approach were each reference is represent as a set of paths. A path
553/// is has a root (basically a `T::Location` plus some edge case massaging) and a list of extensions
554/// resulting from Move function calls. Each one of those Move calls gets a `Delta` extension for
555/// each return value. The `Delta` is like the ".*" in the regex based implementation but where it
556/// carries a sense of identity. This identity allows for invariants from the return values of the
557/// Move call to be leveraged. For example, mutable references returned from a call cannot overlap.
558/// If we just used ".*", we would not be able to express this invariant without some sense of
559/// identity for the reference itself (which is what is going on in the Regex based implementation).
560/// This implementation stems from research work for the Move borrow checker, but would normally
561/// not be expressive enough in the presence of control flow. Luckily, PTBs do not have control flow
562/// so we can use this approach as a safety net for the Regex based implementation until that
563/// code is sufficiently. tested and hardened.
564/// Strip TxContext input arguments so that they do not flow as inputs
565/// Checks the following
566/// - Values are not used after being moved
567/// - Reference safety is upheld (no dangling references)
568pub fn verify<Mode: ExecutionMode>(
569    env: &Env<Mode>,
570    txn: &T::Transaction,
571) -> Result<(), Mode::Error> {
572    Ok(verify_(env, txn).map_err(|e| make_invariant_violation!("{}. Transaction {:?}", e, txn))?)
573}
574
575fn verify_<Mode: ExecutionMode>(env: &Env<Mode>, txn: &T::Transaction) -> anyhow::Result<()> {
576    if env
577        .protocol_config
578        .max_ptb_live_references_as_option()
579        .is_some()
580    {
581        return Ok(());
582    }
583    let mut context = Context::new(env, txn)?;
584    let T::Transaction {
585        gas_payment: _,
586        bytes: _,
587        objects: _,
588        withdrawals: _,
589        pure: _,
590        receiving: _,
591        withdrawal_compatibility_conversions: _,
592        original_command_len: _,
593        commands,
594        unified_linkage: _,
595    } = txn;
596    for c in commands {
597        command(&mut context, c)?;
598    }
599    Ok(())
600}
601
602fn command(context: &mut Context, c: &T::Command) -> anyhow::Result<()> {
603    // process the command
604    debug_assert!(context.arg_roots.is_empty());
605    let results = command_(context, c)?;
606    // drop unused result values by marking them as `None`
607    assert_invariant!(
608        results.len() == c.value.drop_values.len(),
609        "result length mismatch. expected {}, got {}",
610        c.value.drop_values.len(),
611        results.len()
612    );
613    context.add_result_values(
614        results
615            .into_iter()
616            .zip_debug_eq(c.value.drop_values.iter().copied())
617            .map(|(v, drop)| if drop { None } else { Some(v) }),
618    )?;
619    context.arg_roots.clear();
620    Ok(())
621}

623fn command_(context: &mut Context, sp!(_, c): &T::Command) -> anyhow::Result<Vec<Value>> {
624    let result_tys = &c.result_type;
625    let results = match &c.command {
626        T::Command__::MoveCall(move_call) => {
627            let T::MoveCall {
628                function,
629                arguments,
630            } = &**move_call;
631            let arg_values = context.arguments(arguments)?;
632            call(context, &function.signature, arg_values)?
633        }
634        T::Command__::TransferObjects(objs, recipient) => {
635            context.arguments(objs)?;
636            context.argument(recipient)?;
637            non_ref_results(result_tys)?
638        }
639        T::Command__::SplitCoins(_, coin, amounts) => {
640            context.arguments(amounts)?;
641            let coin_value = context.argument(coin)?;
642            write_ref(context, coin_value)?;
643            non_ref_results(result_tys)?
644        }
645        T::Command__::MergeCoins(_, target, coins) => {
646            context.arguments(coins)?;
647            let target_value = context.argument(target)?;
648            write_ref(context, target_value)?;
649            non_ref_results(result_tys)?
650        }
651        T::Command__::MakeMoveVec(_, arguments) => {
652            context.arguments(arguments)?;
653            non_ref_results(result_tys)?
654        }
655        T::Command__::Publish(_, _, _) => non_ref_results(result_tys)?,
656        T::Command__::Upgrade(_, _, _, ticket, _) => {
657            context.argument(ticket)?;
658            non_ref_results(result_tys)?
659        }
660    };
661    assert_invariant!(
662        result_tys.len() == results.len(),
663        "result length mismatch. Expected {}, got {}",
664        result_tys.len(),
665        results.len()
666    );
667    Ok(results)
668}

670fn write_ref(context: &Context, value: Value) -> anyhow::Result<()> {
671    match value {
672        Value::NonRef => {
673            anyhow::bail!("Cannot write to a non-reference value");
674        }
675
676        Value::Ref { is_mut: false, .. } => {
677            anyhow::bail!("Cannot write to an immutable reference");
678        }
679        Value::Ref {
680            is_mut: true,
681            paths,
682        } => {
683            anyhow::ensure!(
684                !context.any_extends(&paths, /* ignore alias */ true),
685                "Cannot write to a mutable reference that has extensions"
686            );
687            Ok(())
688        }
689    }
690}
691
692fn call(
693    context: &mut Context,
694    signature: &T::LoadedFunctionInstantiation,
695    arguments: Vec<Value>,
696) -> anyhow::Result<Vec<Value>> {
697    let return_ = &signature.return_;
698    let mut all_paths: PathSet = PathSet::empty();
699    let mut imm_paths: PathSet = PathSet::empty();
700    let mut mut_paths: PathSet = PathSet::empty();
701    for arg in arguments {
702        match arg {
703            Value::NonRef => (),
704            Value::Ref {
705                is_mut: true,
706                paths,
707            } => {
708                // Allow alias conflicts with references not passed as arguments
709                anyhow::ensure!(
710                    !context.any_extends(&paths, /* ignore alias */ true),
711                    "Cannot transfer a mutable ref with extensions"
712                );
713                // All mutable argument references must be disjoint from all other references
714                anyhow::ensure!(mut_paths.is_disjoint(&paths), "Double mutable borrow");
715                all_paths.union(&paths);
716                mut_paths.union(&paths);
717            }
718            Value::Ref {
719                is_mut: false,
720                paths,
721            } => {
722                all_paths.union(&paths);
723                imm_paths.union(&paths);
724            }
725        }
726    }
727    // All mutable references must be disjoint from all immutable references
728    anyhow::ensure!(
729        imm_paths.is_disjoint(&mut_paths),
730        "Mutable and immutable borrows cannot overlap"
731    );
732    // With references enabled in PTBs, TxContext still cannot be the
733    // root of a returned reference. We ensure this by removing them from
734    // any possible input.
735    if context.allow_references_in_ptbs {
736        let is_ctx = |p: &Path| p.root == RootLocation::Known(T::Location::TxContext);
737        mut_paths.0.retain(|p| !is_ctx(p));
738        all_paths.0.retain(|p| !is_ctx(p));
739    }
740    let command = context.current_command()?;
741    let mut_paths = if mut_paths.is_empty() {
742        PathSet::unknown_root(command)
743    } else {
744        mut_paths
745    };
746    let all_paths = if all_paths.is_empty() {
747        PathSet::unknown_root(command)
748    } else {
749        all_paths
750    };
751    return_
752        .iter()
753        .enumerate()
754        .map(|(i, ty)| {
755            let delta = Delta {
756                command,
757                result: checked_as!(i, u16)?,
758            };
759            match ty {
760                T::Type::Reference(/* is mut */ true, _) => {
761                    Value::ref_(true, mut_paths.extend(delta))
762                }
763                T::Type::Reference(/* is mut */ false, _) => {
764                    Value::ref_(false, all_paths.extend(delta))
765                }
766                _ => Ok(Value::NonRef),
767            }
768        })
769        .collect::<anyhow::Result<Vec<_>>>()
770}
771
772fn non_ref_results(results: &[T::Type]) -> anyhow::Result<Vec<Value>> {
773    results
774        .iter()
775        .map(|t| {
776            anyhow::ensure!(
777                !matches!(t, T::Type::Reference(_, _)),
778                "attempted to create a non-reference result from a reference type",
779            );
780            Ok(Value::NonRef)
781        })
782        .collect()
783}

789impl Path {
790    #[cfg(debug_assertions)]
791    #[allow(unused)]
792    fn print(&self) {
793        print!("{:?}", self.root);
794        for ext in &self.extensions {
795            let Delta { command, result } = ext;
796            print!(".d{}_{}", command, result);
797        }
798        println!(",");
799    }
800}

802impl PathSet {
803    #[cfg(debug_assertions)]
804    #[allow(unused)]
805    fn print(&self) {
806        println!("{{");
807        for path in &self.0 {
808            path.print();
809        }
810        println!("}}");
811    }
812}
813
814impl Value {
815    #[cfg(debug_assertions)]
816    #[allow(unused)]
817    fn print(&self) {
818        match self {
819            Value::NonRef => print!("NonRef"),
820            Value::Ref { is_mut, paths } => {
821                if *is_mut {
822                    print!("mut ");
823                } else {
824                    print!("imm ");
825                }
826                paths.print();
827            }
828        }
829    }
830}
