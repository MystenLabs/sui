4//! Post-execution system-invariant checks for transaction execution.
5//!
6//! All of the defense-in-depth checks that verify the finalized execution results satisfy system
7//! invariants live here, behind the [`InvariantChecker`] API: that the transaction neither mints
8//! nor burns SUI, that balance-accumulator withdrawals stay authorized, and that every modified
9//! object traces back to an authenticated owner. [`TemporaryStore`] owns one `InvariantChecker`,
10//! forwards execution observations needed by invariant checks to it, and defers to it for
11//! the checks. Keeping it in its own module keeps the invariant accounting -- which reaches across
12//! gas, accumulator events, settlement SUI, per-object storage rebates, and object ownership -- out
13//! of the main store code.
14//!
15//! Note these are invariant *assertions* (a failure means a bug, and aborts or panics), distinct
16//! from transaction-validation guards like `TemporaryStore::check_accumulator_amounts_representable`
17//! that can legitimately reject a well-formed transaction.
18//!

42/// Holds invariant-check-only bookkeeping accumulated during execution and exposes the
43/// post-execution system-invariant checks (SUI conservation, balance-accumulator authorization,
44/// object ownership) as its API.
45///
46/// Immutable check inputs and execution results are read from the owning [`TemporaryStore`]
47/// passed to each check method.
48#[derive(Default)]
49pub(crate) struct InvariantChecker {
50    /// A map from wrapped object to its container, used to authenticate wrapped mutations.
51    wrapped_object_containers: BTreeMap<ObjectID, ObjectID>,
52    /// All object IDs generated by the object runtime, including created-and-then-deleted objects.
53    generated_runtime_ids: BTreeSet<ObjectID>,
54    /// Index ranges into `execution_results.accumulator_events` for events emitted from PTB
55    /// (Move) execution. Each `record_ptb_event_range` call appends a contiguous range
56    /// bracketing the merge. Any index outside these ranges was emitted by the runtime outside
57    /// of PTB execution (currently only `gas_charger`'s `add_accumulator_event`). Consumed by
58    /// `check_address_balance_changes` to gate non-PTB-emitted events behind input reservations.
59    /// Cleared on `drop_writes` since the underlying events are also cleared. A `Vec` is
60    /// sufficient because real transactions run the PTB at most a handful of times.
61    ptb_emitted_accumulator_event_ranges: Vec<Range<usize>>,
62}

64impl InvariantChecker {
65    pub(crate) fn save_wrapped_object_containers(
66        &mut self,
67        wrapped_object_containers: BTreeMap<ObjectID, ObjectID>,
68    ) {
69        #[cfg(debug_assertions)]
70        {
71            for (id, container1) in &wrapped_object_containers {
72                if let Some(container2) = self.wrapped_object_containers.get(id) {
73                    assert_eq!(container1, container2);
74                }
75            }
76            for (id, container1) in &self.wrapped_object_containers {
77                if let Some(container2) = wrapped_object_containers.get(id) {
78                    assert_eq!(container1, container2);
79                }
80            }
81        }
82        // Merge the two maps because we may be calling the execution engine more than once
83        // (e.g. in advance epoch transaction, where we may be publishing a new system package).
84        self.wrapped_object_containers
85            .extend(wrapped_object_containers);
86    }

88    pub(crate) fn save_generated_object_ids(&mut self, generated_ids: BTreeSet<ObjectID>) {
89        #[cfg(debug_assertions)]
90        {
91            for id in &self.generated_runtime_ids {
92                assert!(!generated_ids.contains(id))
93            }
94            for id in &generated_ids {
95                assert!(!self.generated_runtime_ids.contains(id));
96            }
97        }
98        self.generated_runtime_ids.extend(generated_ids);
99    }

101    /// Record that the accumulator events in `[start, end)` (indices into
102    /// `execution_results.accumulator_events`) were emitted by PTB (Move) execution. `start`/`end`
103    /// are the lengths of the accumulator-event vec before/after a `merge_results`. The
104    /// address-balance change invariant uses this set to distinguish trusted PTB-emitted events
105    /// from runtime-emitted ones.
106    pub(crate) fn record_ptb_event_range(&mut self, start: usize, end: usize) {
107        debug_assert!(
108            start <= end,
109            "merge_results should not shrink accumulator_events"
110        );
111        let (start, end) = (start.min(end), start.max(end));
112        let range = start..end;
113        match self.ptb_emitted_accumulator_event_ranges.last_mut() {
114            // Coalesce with the previous PTB range if no runtime events were added in between.
115            Some(last) if last.end == range.start => last.end = range.end,
116            _ => self.ptb_emitted_accumulator_event_ranges.push(range),
117        }
118    }

120    /// Check that this transaction neither creates nor destroys SUI. This should hold for all txes
121    /// except the epoch change tx, which mints staking rewards equal to the gas fees burned in the
122    /// previous epoch.  Specifically, this checks two key invariants about storage
123    /// fees and storage rebate:
124    ///
125    /// 1. all SUI in storage rebate fields of input objects should flow either to the transaction
126    ///    storage rebate, or the transaction non-refundable storage rebate
127    /// 2. all SUI charged for storage should flow into the storage rebate field of some output
128    ///    object
129    ///
130    /// This function is intended to be called *after* we have charged for
131    /// gas + applied the storage rebate to the gas object, but *before* we
132    /// have updated object versions.
133    pub(crate) fn check_sui_conserved(
134        &self,
135        store: &TemporaryStore<'_>,
136        simple_conservation_checks: bool,
137        gas_summary: &GasCostSummary,
138    ) -> Result<(), ExecutionError> {
139        if !simple_conservation_checks {
140            return Ok(());
141        }
142        // total amount of SUI in storage rebate of input objects
143        let mut total_input_rebate = 0;
144        // total amount of SUI in storage rebate of output objects
145        let mut total_output_rebate = 0;
146        for (_id, input, output) in get_modified_objects(store) {
147            if let Some(input) = input {
148                total_input_rebate += input.storage_rebate;
149            }
150            if let Some(object) = output {
151                total_output_rebate += object.storage_rebate;
152            }
153        }
154
155        if gas_summary.storage_cost == 0 {
156            // this condition is usually true when the transaction went OOG and no
157            // gas is left for storage charges.
158            // The storage cost has to be there at least for the gas coin which
159            // will not be deleted even when going to 0.
160            // However if the storage cost is 0 and if there is any object touched
161            // or deleted the value in input must be equal to the output plus rebate and
162            // non refundable.
163            // Rebate and non refundable will be positive when there are object deleted
164            // (gas smashing being the primary and possibly only example).
165            // A more typical condition is for all storage charges in summary to be 0 and
166            // then input and output must be the same value
167            if total_input_rebate
168                != total_output_rebate
169                    + gas_summary.storage_rebate
170                    + gas_summary.non_refundable_storage_fee
171            {
172                return Err(ExecutionError::invariant_violation(format!(
173                    "SUI conservation failed -- no storage charges in gas summary \
174                        and total storage input rebate {} not equal  \
175                        to total storage output rebate {}",
176                    total_input_rebate, total_output_rebate,
177                )));
178            }
179        } else {
180            // all SUI in storage rebate fields of input objects should flow either to
181            // the transaction storage rebate, or the non-refundable storage rebate pool
182            if total_input_rebate
183                != gas_summary.storage_rebate + gas_summary.non_refundable_storage_fee
184            {
185                return Err(ExecutionError::invariant_violation(format!(
186                    "SUI conservation failed -- {} SUI in storage rebate field of input objects, \
187                        {} SUI in tx storage rebate or tx non-refundable storage rebate",
188                    total_input_rebate, gas_summary.non_refundable_storage_fee,
189                )));
190            }
191
192            // all SUI charged for storage should flow into the storage rebate field
193            // of some output object
194            if gas_summary.storage_cost != total_output_rebate {
195                return Err(ExecutionError::invariant_violation(format!(
196                    "SUI conservation failed -- {} SUI charged for storage, \
197                        {} SUI in storage rebate field of output objects",
198                    gas_summary.storage_cost, total_output_rebate
199                )));
200            }
201        }
202        Ok(())
203    }

205    /// Defense-in-depth invariant on funds-accumulator events. Per `(address, type)`:
206    /// - If the pair is in `input_reservations`, net withdrawal <= budget.
207    /// - Else if the PTB emitted a Split at this key, we assume there must be an object withdrawal.
208    ///   As such, any net change is acceptable.
209    /// - Else if the PTB emitted only Merges at this key, we can assume there might not be an
210    ///   object withdrawal. In any case, the net balance at the end of the transaction should be
211    ///   non-negative, since there could be additional withdrawals from gas, but they should
212    ///   not exceed the deposits.
213    /// - Else, any event is unauthorized.
214    ///
215    /// Currently the only funds-accumulator type is `Balance<T>`, so the check is scoped to
216    /// those events. As more accumulator shapes are added the filter and the integer
217    /// arithmetic in this method will need to grow with them.
218    ///
219    /// PTB-emitted events are identified via `ptb_emitted_accumulator_event_ranges`, populated
220    /// at `record_execution_results` time. They are trusted because Move enforces `&mut UID`
221    /// and the native checks the actual balance.
222    pub(crate) fn check_address_balance_changes(
223        &self,
224        store: &TemporaryStore<'_>,
225    ) -> Result<(), ExecutionError> {
226        use sui_types::balance::Balance;
227
228        let input_reservations = &store.post_execution_check_inputs.input_reservations;
229        let mut actual_changes: BTreeMap<(SuiAddress, TypeTag), i128> = BTreeMap::new();
230        let mut has_ptb_withdrawals: BTreeSet<(SuiAddress, TypeTag)> = BTreeSet::new();
231        let mut has_ptb_deposits: BTreeSet<(SuiAddress, TypeTag)> = BTreeSet::new();
232        for (idx, event) in store
233            .execution_results
234            .accumulator_events
235            .iter()
236            .enumerate()
237        {
238            // Filter on the value shape first: only `Integer` carries the funds-flow we care
239            // about. Other shapes (e.g. `EventDigest` for event-stream heads) belong to
240            // non-Balance accumulators and are out of scope here. If we ever see an `Integer`
241            // value at a non-`Balance<T>` type, the accounting invariants below don't apply
242            // -- debug_fatal so that case is surfaced instead of silently accepted.
243            let amount = match event.write.value {
244                AccumulatorValue::Integer(amount) => amount as i128,
245                AccumulatorValue::IntegerTuple(_, _) | AccumulatorValue::EventDigest(_) => {
246                    assert_invariant!(
247                        !sui_types::balance::Balance::is_balance_type(&event.write.address.ty),
248                        "Non-integer accumulator changes should not be balances"
249                    );
250                    continue;
251                }
252            };
253            if !Balance::is_balance_type(&event.write.address.ty) {
254                debug_fatal!(
255                    "Integer accumulator value at non-Balance type: {:?}",
256                    event.write.address.ty
257                );
258                continue;
259            }
260            let is_ptb_emitted = self
261                .ptb_emitted_accumulator_event_ranges
262                .iter()
263                .any(|range| range.contains(&idx));
264            let key = (event.write.address.address, event.write.address.ty.clone());
265            let change = match event.write.operation {
266                AccumulatorOperation::Split => {
267                    if is_ptb_emitted {
268                        has_ptb_withdrawals.insert(key.clone());
269                    }
270                    -amount
271                }
272                AccumulatorOperation::Merge => {
273                    if is_ptb_emitted {
274                        has_ptb_deposits.insert(key.clone());
275                    }
276                    amount
277                }
278            };
279            *actual_changes.entry(key.clone()).or_insert(0) += change;
280        }
281
282        for (key, actual) in actual_changes {
283            let (address, type_tag) = &key;
284            if let Some(budget) = input_reservations.get(&key).copied() {
285                let net_withdrawn = -actual.min(0) as u128;
286                assert_invariant!(
287                    net_withdrawn <= budget as u128,
288                    "Balance accumulator withdrawal exceeds reservation budget at address \
289                    {address} for type {type_tag}: net Split {net_withdrawn}, budget {budget}"
290                );
291            } else if has_ptb_withdrawals.contains(&key) {
292                // Move authorized the PTB Split against the on-chain balance, so any
293                // resulting net (including a net withdrawal beyond any PTB Merges here)
294                // is trusted.
295            } else if has_ptb_deposits.contains(&key) {
296                // PTB only deposited at this key. As such, the final net change must be
297                // non-negative, since there was no authorization for any withdrawal.
298                // We cannot compare this value to the sum of the PTB deposits due to intricacies
299                // with gas charging and storage rebate.
300                assert_invariant!(
301                    actual >= 0,
302                    "PTB-emitted Balance accumulator deposits do not cover the runtime \
303                    withdrawal at address {address} for type {type_tag}: net change {actual}"
304                );
305            } else {
306                invariant_violation!(
307                    "Unauthorized runtime Balance accumulator event at address {address} for \
308                    type {type_tag}: net change {actual} (no input reservation, no PTB-emitted \
309                    events)"
310                );
311            }
312        }
313
314        Ok(())
315    }

317    /// Check that this transaction neither creates nor destroys SUI.
318    /// This more expensive check will check a third invariant on top of the 2 performed
319    /// by `check_sui_conserved` above:
320    ///
321    /// * all SUI in input objects (including coins etc in the Move part of an object) should flow
322    ///   either to an output object, or be burned as part of computation fees or non-refundable
323    ///   storage rebate
324    ///
325    /// This function is intended to be called *after* we have charged for gas + applied the
326    /// storage rebate to the gas object, but *before* we have updated object versions. The
327    /// advance epoch transaction would mint `epoch_fees` amount of SUI, and burn `epoch_rebates`
328    /// amount of SUI. We need these information for this check.
329    pub(crate) fn check_sui_conserved_expensive(
330        &self,
331        store: &TemporaryStore<'_>,
332        gas_summary: &GasCostSummary,
333        layout_resolver: &mut impl LayoutResolver,
334    ) -> Result<(), ExecutionError> {
335        let advance_epoch_gas_summary = store.post_execution_check_inputs.advance_epoch_gas_summary;
336        // Accumulate in u128. The per-object SUI totals are bounded by the real supply, but the
337        // accumulator-event terms below are not: an object-sourced withdrawal/deposit (backing
338        // verified only at settlement) can contribute up to u64::MAX on each side, and a transaction
339        // can stack several across distinct keys, so a u64 running total could overflow. These
340        // amounts net out, so a u128 sum stays exact and conservation is decided correctly.
341        // total amount of SUI in input objects, including both coins and storage rebates
342        let mut total_input_sui: u128 = 0;
343        // total amount of SUI in output objects, including both coins and storage rebates
344        let mut total_output_sui: u128 = 0;
345
346        // settlement input/output sui is used by the settlement transactions to account for
347        // Sui that has been gathered from the accumulator writes of transactions which it is
348        // settling.
349        total_input_sui += store.execution_results.settlement_input_sui as u128;
350        total_output_sui += store.execution_results.settlement_output_sui as u128;
351
352        for (id, input, output) in get_modified_objects(store) {
353            if let Some(input) = input {
354                total_input_sui +=
355                    get_input_sui(store, &id, input.version, layout_resolver)? as u128;
356            }
357            if let Some(object) = output {
358                total_output_sui += object.get_total_sui(layout_resolver).map_err(|e| {
359                    make_invariant_violation!(
360                        "Failed looking up output SUI in SUI conservation checking for \
361                         mutated type {:?}: {e:#?}",
362                        object.struct_tag(),
363                    )
364                })? as u128;
365            }
366        }
367
368        for event in &store.execution_results.accumulator_events {
369            let (input, output) = event.total_sui_in_event();
370            total_input_sui += input as u128;
371            total_output_sui += output as u128;
372        }
373
374        // note: storage_cost flows into the storage_rebate field of the output objects, which is
375        // why it is not accounted for here.
376        // similarly, all of the storage_rebate *except* the storage_fund_rebate_inflow
377        // gets credited to the gas coin both computation costs and storage rebate inflow are
378        total_output_sui +=
379            gas_summary.computation_cost as u128 + gas_summary.non_refundable_storage_fee as u128;
380        if let Some((epoch_fees, epoch_rebates)) = advance_epoch_gas_summary {
381            total_input_sui += epoch_fees as u128;
382            total_output_sui += epoch_rebates as u128;
383        }
384        if total_input_sui != total_output_sui {
385            return Err(ExecutionError::invariant_violation(format!(
386                "SUI conservation failed: input={}, output={}, \
387                    this transaction either mints or burns SUI",
388                total_input_sui, total_output_sui,
389            )));
390        }
391        Ok(())
392    }
393}

395type ModifiedObjectInfo<'a> = (
396    ObjectID,
397    // old object metadata, including version, digest, owner, and storage rebate.
398    Option<DynamicallyLoadedObjectMetadata>,
399    Option<&'a Object>,
400);
401
402/// Return the list of all modified objects, for each object, returns
403/// - Object ID,
404/// - Input: If the object existed prior to this transaction, include their version and storage_rebate,
405/// - Output: If a new version of the object is written, include the new object.
406fn get_modified_objects<'a>(store: &'a TemporaryStore<'_>) -> Vec<ModifiedObjectInfo<'a>> {
407    store
408        .execution_results
409        .modified_objects
410        .iter()
411        .map(|id| {
412            let metadata = store.get_object_modified_at(id);
413            let output = store.execution_results.written_objects.get(id);
414            (*id, metadata, output)
415        })
416        .chain(
417            store
418                .execution_results
419                .written_objects
420                .iter()
421                .filter_map(|(id, object)| {
422                    if store.execution_results.modified_objects.contains(id) {
423                        None
424                    } else {
425                        Some((*id, None, Some(object)))
426                    }
427                }),
428        )
429        .collect()
430}
431
432fn get_input_sui(
433    store: &TemporaryStore<'_>,
434    id: &ObjectID,
435    expected_version: SequenceNumber,
436    layout_resolver: &mut impl LayoutResolver,
437) -> Result<u64, ExecutionError> {
438    if let Some(obj) = store.input_objects.get(id) {
439        // the assumption here is that if it is in the input objects must be the right one
440        if obj.version() != expected_version {
441            invariant_violation!(
442                "Version mismatching when resolving input object to check conservation--\
443                 expected {}, got {}",
444                expected_version,
445                obj.version(),
446            );
447        }
448        obj.get_total_sui(layout_resolver).map_err(|e| {
449            make_invariant_violation!(
450                "Failed looking up input SUI in SUI conservation checking for input with \
451                     type {:?}: {e:#?}",
452                obj.struct_tag(),
453            )
454        })
455    } else {
456        // not in input objects, must be a dynamic field
457        let Some(obj) = store.store.get_object_by_key(id, expected_version) else {
458            invariant_violation!(
459                "Failed looking up dynamic field {id} in SUI conservation checking"
460            );
461        };
462        obj.get_total_sui(layout_resolver).map_err(|e| {
463            make_invariant_violation!(
464                "Failed looking up input SUI in SUI conservation checking for type \
465                     {:?}: {e:#?}",
466                obj.struct_tag(),
467            )
468        })
469    }
470}

472impl InvariantChecker {
473    /// Run the SUI-conservation and balance-accumulator invariant checks against the
474    /// (already-finalized, gas-charged) `store`. Read-only: the caller (the execution engine's
475    /// `run_conservation_checks`) owns any recovery that mutates state.
476    ///
477    /// Returns `Ok(())` when the checks are not applicable: the genesis transaction mints the SUI
478    /// supply, and dev-inspect mode is allowed to violate conservation. Immutable check inputs are
479    /// read from the store.
480    pub(crate) fn check_conservation_invariants<Mode: ExecutionMode>(
481        &self,
482        store: &TemporaryStore<'_>,
483        move_vm: &Arc<MoveRuntime>,
484        enable_expensive_checks: bool,
485        cost_summary: &GasCostSummary,
486    ) -> Result<(), ExecutionError> {
487        if store.post_execution_check_inputs.is_genesis || Mode::skip_conservation_checks() {
488            return Ok(());
489        }
490        let simple_conservation_checks = store.protocol_config().simple_conservation_checks();
491        self.check_sui_conserved(store, simple_conservation_checks, cost_summary)
492            .and_then(|()| {
493                if enable_expensive_checks {
494                    let mut layout_resolver =
495                        TypeLayoutResolver::new(move_vm, store.protocol_config(), Box::new(store));
496                    self.check_sui_conserved_expensive(store, cost_summary, &mut layout_resolver)
497                } else {
498                    Ok(())
499                }
500            })
501            .and_then(|()| self.check_address_balance_changes(store))
502    }
503
504    /// If the transaction executed successfully, every package written by the transaction must
505    /// correspond to exactly one `Publish`/`Upgrade` command and must contain the modules that
506    /// command supplied and record the dependencies it declared as the package's dependencies.
507    ///
508    /// Compared as multisets rather than pairwise: written packages are keyed by id, so they
509    /// cannot be correlated back to their originating command by position.
510    pub(crate) fn check_published_packages(
511        &self,
512        store: &TemporaryStore<'_>,
513    ) -> Result<(), ExecutionError> {
514        if !store.protocol_config().harden_linkage_consistency() {
515            return Ok(());
516        }
517
518        let Some(declared) = &store.post_execution_check_inputs.declared_packages else {
519            return Ok(());
520        };
521
522        let mut written: Vec<(usize, BTreeSet<ObjectID>)> = store
523            .execution_results
524            .written_objects
525            .values()
526            .filter_map(|object| object.data.try_as_package())
527            .map(|package| {
528                (
529                    package.serialized_module_map().len(),
530                    package
531                        .linkage_table()
532                        .values()
533                        .map(|upgrade_info| upgrade_info.upgraded_id)
534                        .collect(),
535                )
536            })
537            .collect();
538        let mut declared = declared.clone();
539        written.sort();
540        declared.sort();
541
542        if written != declared {
543            return Err(ExecutionError::invariant_violation(format!(
544                "packages written by this transaction do not correspond to its publish and upgrade \
545                 commands: commands declared {declared:?}, packages recorded {written:?}"
546            )));
547        }
548        Ok(())
549    }

551    // check that every object read is owned directly or indirectly by sender, sponsor,
552    // or a shared object input
553    pub(crate) fn check_ownership_invariants(
554        &self,
555        store: &TemporaryStore<'_>,
556        sender: &SuiAddress,
557        sponsor: &Option<SuiAddress>,
558        gas_charger: &GasCharger,
559        is_epoch_change: bool,
560    ) -> SuiResult<()> {
561        // The funds-accumulator reservation budget is shared with the conservation checks; see
562        // `Self::check_address_balance_changes`.
563        let input_reservations = &store.post_execution_check_inputs.input_reservations;
564        let gas_objs: HashSet<&ObjectID> = gas_charger.used_coins().map(|g| &g.0).collect();
565        let gas_owner = sponsor.as_ref().unwrap_or(sender);
566
567        // mark input objects as authenticated
568        let objects_authenticated_for_mutation: HashSet<SuiAddress> = store
569            .input_objects
570            .iter()
571            .filter_map(|(id, obj)| {
572                match &obj.owner {
573                    Owner::AddressOwner(a) => {
574                        if gas_objs.contains(id) {
575                            // gas object must be owned by sender or sponsor
576                            assert!(
577                                a == gas_owner,
578                                "Gas object must be owned by sender or sponsor"
579                            );
580                        } else {
581                            assert!(sender == a, "Input object must be owned by sender");
582                        }
583                        Some(id)
584                    }
585                    Owner::Shared { .. } | Owner::ConsensusAddressOwner { .. } => Some(id),
586                    Owner::Immutable => {
587                        // object is authenticated, but it cannot own other objects,
588                        // so we should not add it to `authenticated_objs`
589                        // However, we would definitely want to add immutable objects
590                        // to the set of authenticated roots if we were doing runtime
591                        // checks inside the VM instead of after-the-fact in the temporary
592                        // store. Here, we choose not to add them because this will catch a
593                        // bug where we mutate or delete an object that belongs to an immutable
594                        // object (though it will show up somewhat opaquely as an authentication
595                        // failure), whereas adding the immutable object to the roots will prevent
596                        // us from catching this.
597                        None
598                    }
599                    Owner::Party { permissions, .. } => {
600                        let sender_permissions = permissions.permissions_for(sender);
601                        let sponsor_permissions = sponsor
602                            .as_ref()
603                            .map(|s| permissions.permissions_for(s))
604                            .unwrap_or(ObjectPermissions::NONE);
605                        (sender_permissions | sponsor_permissions)
606                            .can_use_mutably()
607                            .then_some(id)
608                    }
609                    Owner::ObjectOwner(_parent) => {
610                        unreachable!(
611                            "Input objects must be address owned, shared, consensus, or immutable"
612                        )
613                    }
614                }
615            })
616            .filter(|id| {
617                // remove any non-mutable inputs. This will remove deleted or readonly shared
618                // objects.
619                store.mutable_input_refs.contains_key(id)
620                    || store.non_exclusive_input_original_versions.contains_key(id)
621            })
622            .copied()
623            // Add any object IDs generated in the object runtime during execution to the
624            // authenticated set (i.e., new (non-package) objects, and possibly ephemeral UIDs).
625            .chain(self.generated_runtime_ids.iter().copied())
626            .map(SuiAddress::from)
627            .collect();
628
629        // Add sender and sponsor (if present) to authenticated set
630        let mut authenticated_for_mutation = {
631            assert!(
632                !objects_authenticated_for_mutation.contains(sender),
633                "Sender cannot be an object"
634            );
635            assert!(
636                sponsor
637                    .is_none_or(|sponsor| !objects_authenticated_for_mutation.contains(&sponsor)),
638                "Sponsor cannot be an object"
639            );
640            let mut s = objects_authenticated_for_mutation.clone();
641            s.insert(*sender);
642            if let Some(sponsor) = sponsor {
643                s.insert(*sponsor);
644            }
645            s
646        };
647
648        // check all modified objects are authenticated
649        let mut objects_to_authenticate = store
650            .execution_results
651            .modified_objects
652            .iter()
653            .copied()
654            .collect::<Vec<_>>();
655
656        while let Some(to_authenticate) = objects_to_authenticate.pop() {
657            if authenticated_for_mutation.contains(&to_authenticate.into()) {
658                // object has already been authenticated
659                continue;
660            }
661
662            let parent = if let Some(container_id) =
663                self.wrapped_object_containers.get(&to_authenticate)
664            {
665                // It's a wrapped object, so check that the container is authenticated
666                *container_id
667            } else {
668                // It's non-wrapped, so check the owner -- we can load the object from the
669                // store.
670                let Some(old_obj) = store.store.get_object(&to_authenticate) else {
671                    panic!(
672                        "Failed to load object {to_authenticate:?}.\n \
673                         If it cannot be loaded, we would expect it to be in the wrapped object map: {:#?}",
674                        &self.wrapped_object_containers
675                    )
676                };
677
678                match &old_obj.owner {
679                    // We mutated a dynamic field, we can continue to trace this back to verify
680                    // proper ownership.
681                    Owner::ObjectOwner(parent) => ObjectID::from(*parent),
682                    // We mutated an address owned or sequenced address owned object -- one of two cases apply:
683                    // 1) the object is owned by an object or address in the authenticated set,
684                    // 2) the object is owned by some other address, in which case we should
685                    //    continue to trace this back.
686                    Owner::AddressOwner(parent)
687                    | Owner::ConsensusAddressOwner { owner: parent, .. } => {
688                        // For Receiving<_> objects, the address owner is actually an object.
689                        // If it was actually an address, we should have caught it as an input and
690                        // it would already have been in authenticated_for_mutation
691                        ObjectID::from(*parent)
692                    }
693                    // We mutated a shared object -- we checked if this object was in the
694                    // authenticated set at the top of this loop and it wasn't so this is a failure.
695                    owner @ Owner::Shared { .. } | owner @ Owner::Party { .. } => {
696                        panic!(
697                            "Unauthenticated root at {to_authenticate:?} with owner {owner:?}\n\
698                             Potentially covering objects in: {authenticated_for_mutation:#?}"
699                        );
700                    }
701
702                    Owner::Immutable => {
703                        assert!(
704                            is_epoch_change,
705                            "Immutable objects cannot be written, except for \
706                             Sui Framework/Move stdlib upgrades at epoch change boundaries"
707                        );
708                        // Note: this assumes that the only immutable objects an epoch change
709                        // tx can update are system packages,
710                        // but in principle we could allow others.
711                        assert!(
712                            is_system_package(to_authenticate),
713                            "Only system packages can be upgraded"
714                        );
715                        continue;
716                    }
717                }
718            };
719
720            // we now assume the object is authenticated and check the parent
721            authenticated_for_mutation.insert(to_authenticate.into());
722            objects_to_authenticate.push(parent);
723        }
724
725        // Check that all funds accumulator splits are authorized
726        let sui_balance_type =
727            sui_types::balance::Balance::type_tag(sui_types::gas_coin::GAS::type_tag());
728        let gas_payment_address_balance =
729            gas_charger
730                .gas_payment_location()
731                .and_then(|location| match location {
732                    PaymentLocation::Coin(_) => None,
733                    PaymentLocation::AddressBalance(address) => Some(address),
734                });
735        // A Split at a non-signer key requires every allowance id declared for it as a loaded,
736        // matching input.
737        let is_allowance_backed = |key: &(SuiAddress, TypeTag)| {
738            let Some(allowance_ids) = store.post_execution_check_inputs.allowance_ids.get(key)
739            else {
740                return false;
741            };
742            allowance_ids.iter().all(|id| {
743                let resolved = store.input_objects.get(id).map(parse_allowance_object);
744                matches!(resolved, Some(Ok(a)) if a.funder == key.0 && a.funds_type == key.1)
745            })
746        };
747        let mut funds_net_changes: BTreeMap<(SuiAddress, TypeTag), i128> = BTreeMap::new();
748        for event in store.execution_results.accumulator_events.iter() {
749            let amount = match event.write.value {
750                AccumulatorValue::Integer(a) => a as i128,
751                AccumulatorValue::IntegerTuple(_, _) | AccumulatorValue::EventDigest(_) => {
752                    assert!(
753                        !sui_types::balance::Balance::is_balance_type(&event.write.address.ty),
754                        "Non-integer accumulator changes should not be balances"
755                    );
756                    continue;
757                }
758            };
759            let signed = match event.write.operation {
760                AccumulatorOperation::Split => -amount,
761                AccumulatorOperation::Merge => amount,
762            };
763            let address = event.write.address.address;
764            let type_tag = &event.write.address.ty;
765            let key = (address, type_tag.clone());
766            *funds_net_changes.entry(key.clone()).or_insert(0) += signed;
767            // Authorized if it is:
768            // - A merge/deposit (anyone can deposit)
769            // - A withdrawal
770            //   - with a corresponding input reservation (the signer's or allowance-backed)
771            //   - from an object authenticated for mutation
772            //   - for the gas payment (potentially from a GasCoin send_funds transfer)
773            let authorized = match event.write.operation {
774                AccumulatorOperation::Merge => true,
775                AccumulatorOperation::Split => {
776                    let is_authorized_input_reservation = input_reservations.contains_key(&key)
777                        && (address == *sender
778                            || address == *gas_owner
779                            || is_allowance_backed(&key));
780                    is_authorized_input_reservation
781                        || objects_authenticated_for_mutation.contains(&address)
782                        || (*type_tag == sui_balance_type
783                            && gas_payment_address_balance
784                                .is_some_and(|gas_addr| gas_addr == address))
785                }
786            };
787            assert!(
788                authorized,
789                "Unauthenticated funds-accumulator Split at address {address} for type \
790                 {type_tag}: no signer or allowance-backed input reservation, address is not an \
791                 authenticated object, and it is not the final gas payment address balance"
792            );
793        }
794
795        // For all net negative changes (net withdrawals), the net changes _must_ be less than the
796        // reservation amount, or it must be from an object. This excludes the final gas payment
797        // address since the case where that withdrawal is allowed should be a net positive (or
798        // zero) since it occurs only in the case where the gas coin is transferred via send_funds
799        for (key, change) in funds_net_changes {
800            // skip if deposit or for an object
801            if change >= 0 || objects_authenticated_for_mutation.contains(&key.0) {
802                continue;
803            }
804            let reservation = input_reservations.get(&key).copied().unwrap_or(0) as u128;
805            let withdrawn = change.unsigned_abs();
806            assert!(
807                withdrawn <= reservation,
808                "Net withdrawal of {withdrawn} for {key:?} exceeds input reservation of \
809                 {reservation}"
810            );
811        }
812
813        Ok(())
814    }
815}
