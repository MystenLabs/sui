117impl InputObjectMetadata {
118    pub fn id(&self) -> ObjectID {
119        match self {
120            InputObjectMetadata::Receiving { id, .. } => *id,
121            InputObjectMetadata::InputObject { id, .. } => *id,
122        }
123    }
124
125    pub fn version(&self) -> SequenceNumber {
126        match self {
127            InputObjectMetadata::Receiving { version, .. } => *version,
128            InputObjectMetadata::InputObject { version, .. } => *version,
129        }
130    }
131}

133impl InputValue {
134    pub fn new_object(object_metadata: InputObjectMetadata, value: ObjectValue) -> Self {
135        let mut inner = ResultValue::new(Value::Object(value));
136        if let InputObjectMetadata::InputObject {
137            id,
138            owner: Owner::Shared { .. },
139            ..
140        } = &object_metadata
141        {
142            inner.shared_object_ids.insert(*id);
143        }
144        InputValue {
145            object_metadata: Some(object_metadata),
146            inner,
147        }
148    }
149
150    pub fn new_raw(ty: RawValueType, value: Vec<u8>) -> Self {
151        InputValue {
152            object_metadata: None,
153            inner: ResultValue::new(Value::Raw(ty, value)),
154        }
155    }
156
157    pub fn new_receiving_object(id: ObjectID, version: SequenceNumber) -> Self {
158        InputValue {
159            object_metadata: Some(InputObjectMetadata::Receiving { id, version }),
160            inner: ResultValue::new(Value::Receiving(id, version, None)),
161        }
162    }
163
164    pub fn withdrawal(withdrawal_ty: RawValueType, owner: SuiAddress, limit: U256) -> Self {
165        let value = Value::Raw(
166            withdrawal_ty,
167            bcs::to_bytes(&Withdrawal::new(owner, limit)).unwrap(),
168        );
169        InputValue {
170            object_metadata: None,
171            inner: ResultValue {
172                last_usage_kind: None,
173                value: Some(value),
174                shared_object_ids: BTreeSet::new(),
175            },
176        }
177    }
178}

190impl Value {
191    pub fn is_copyable(&self) -> bool {
192        match self {
193            Value::Object(_) => false,
194            Value::Raw(RawValueType::Any, _) => true,
195            Value::Raw(RawValueType::Loaded { abilities, .. }, _) => abilities.has_copy(),
196            Value::Receiving(_, _, _) => false,
197        }
198    }
199
200    pub fn write_bcs_bytes(
201        &self,
202        buf: &mut Vec<u8>,
203        bound: Option<SizeBound>,
204    ) -> Result<(), ExecutionError> {
205        match self {
206            Value::Object(obj_value) => obj_value.write_bcs_bytes(buf, bound)?,
207            Value::Raw(_, bytes) => buf.extend(bytes),
208            Value::Receiving(id, version, _) => {
209                buf.extend(Receiving::new(*id, *version).to_bcs_bytes())
210            }
211        }
212        if let Some(bound) = bound {
213            ensure_serialized_size(buf.len() as u64, bound)?;
214        }
215
216        Ok(())
217    }
218
219    pub fn was_used_in_non_entry_move_call(&self) -> bool {
220        match self {
221            Value::Object(obj) => obj.used_in_non_entry_move_call,
222            // Any is only used for Pure inputs, and if it was used by &mut it would have switched
223            // to Loaded
224            Value::Raw(RawValueType::Any, _) => false,
225            Value::Raw(
226                RawValueType::Loaded {
227                    used_in_non_entry_move_call,
228                    ..
229                },
230                _,
231            ) => *used_in_non_entry_move_call,
232            // Only thing you can do with a `Receiving<T>` is consume it, so once it's used it
233            // can't be used again.
234            Value::Receiving(_, _, _) => false,
235        }
236    }
237}

239impl ObjectValue {
240    /// # Safety
241    /// We must have the Type is the coin type, but we are unable to check it at this spot
242    pub unsafe fn coin(type_: Type, coin: Coin) -> Self {
243        Self {
244            type_,
245            has_public_transfer: true,
246            used_in_non_entry_move_call: false,
247            contents: ObjectContents::Coin(coin),
248        }
249    }
250
251    pub fn ensure_public_transfer_eligible(&self) -> Result<(), ExecutionError> {
252        if !self.has_public_transfer {
253            return Err(ExecutionErrorKind::InvalidTransferObject.into());
254        }
255        Ok(())
256    }
257
258    pub fn write_bcs_bytes(
259        &self,
260        buf: &mut Vec<u8>,
261        bound: Option<SizeBound>,
262    ) -> Result<(), ExecutionError> {
263        match &self.contents {
264            ObjectContents::Raw(bytes) => buf.extend(bytes),
265            ObjectContents::Coin(coin) => buf.extend(coin.to_bcs_bytes()),
266        }
267        if let Some(bound) = bound {
268            ensure_serialized_size(buf.len() as u64, bound)?;
269        }
270        Ok(())
271    }
272}

274pub fn ensure_serialized_size(size: u64, bound: SizeBound) -> Result<(), ExecutionError> {
275    let bound_size = match bound {
276        SizeBound::Object(bound_size)
277        | SizeBound::VectorElem(bound_size)
278        | SizeBound::Raw(bound_size) => bound_size,
279    };
280    if size > bound_size {
281        let e = match bound {
282            SizeBound::Object(_) => ExecutionErrorKind::MoveObjectTooBig {
283                object_size: size,
284                max_object_size: bound_size,
285            },
286            SizeBound::VectorElem(_) => ExecutionErrorKind::MoveVectorElemTooBig {
287                value_size: size,
288                max_scaled_size: bound_size,
289            },
290            SizeBound::Raw(_) => ExecutionErrorKind::MoveRawValueTooBig {
291                value_size: size,
292                max_scaled_size: bound_size,
293            },
294        };
295        let msg = "Serialized bytes of value too large".to_owned();
296        return Err(ExecutionError::new_with_source(e, msg));
297    }
298    Ok(())
299}

301pub trait TryFromValue: Sized {
302    fn try_from_value(value: Value) -> Result<Self, CommandArgumentError>;
303}
304
305impl TryFromValue for Value {
306    fn try_from_value(value: Value) -> Result<Self, CommandArgumentError> {
307        Ok(value)
308    }
309}
310
311impl TryFromValue for ObjectValue {
312    fn try_from_value(value: Value) -> Result<Self, CommandArgumentError> {
313        match value {
314            Value::Object(o) => Ok(o),
315            Value::Raw(RawValueType::Any, _) => Err(CommandArgumentError::TypeMismatch),
316            Value::Raw(RawValueType::Loaded { .. }, _) => Err(CommandArgumentError::TypeMismatch),
317            Value::Receiving(_, _, _) => Err(CommandArgumentError::TypeMismatch),
318        }
319    }
320}

322impl TryFromValue for SuiAddress {
323    fn try_from_value(value: Value) -> Result<Self, CommandArgumentError> {
324        try_from_value_prim(&value, Type::Address)
325    }
326}

328impl TryFromValue for u64 {
329    fn try_from_value(value: Value) -> Result<Self, CommandArgumentError> {
330        try_from_value_prim(&value, Type::U64)
331    }
332}
333
334fn try_from_value_prim<'a, T: Deserialize<'a>>(
335    value: &'a Value,
336    expected_ty: Type,
337) -> Result<T, CommandArgumentError> {
338    match value {
339        Value::Object(_) => Err(CommandArgumentError::TypeMismatch),
340        Value::Receiving(_, _, _) => Err(CommandArgumentError::TypeMismatch),
341        Value::Raw(RawValueType::Any, bytes) => {
342            bcs::from_bytes(bytes).map_err(|_| CommandArgumentError::InvalidBCSBytes)
343        }
344        Value::Raw(RawValueType::Loaded { ty, .. }, bytes) => {
345            if ty != &expected_ty {
346                return Err(CommandArgumentError::TypeMismatch);
347            }
348            bcs::from_bytes(bytes).map_err(|_| CommandArgumentError::InvalidBCSBytes)
349        }
350    }
