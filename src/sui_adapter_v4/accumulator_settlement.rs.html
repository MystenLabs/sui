58struct Context<'a> {
59    state: &'a dyn ExecutionState,
60    protocol_config: &'a ProtocolConfig,
61    tx_context: &'a TxContext,
62    inputs: Vec<CallArg>,
63    root_argument: Argument,
64    root: Object,
65    children: BTreeMap<ObjectID, ChildObject>,
66    generated_ids: BTreeSet<ObjectID>,
67    input_sui: u64,
68    output_sui: u64,
69}
70
71impl<'a> Context<'a> {
72    fn new(
73        protocol_config: &'a ProtocolConfig,
74        state: &'a dyn ExecutionState,
75        tx_context: &'a TxContext,
76        inputs: Vec<CallArg>,
77    ) -> Result<Self, ExecutionError> {
78        assert_invariant!(
79            tx_context.sender() == SuiAddress::ZERO,
80            "Settlement sender must be the system address"
81        );
82        let mut root_argument = None;
83        for (index, input) in inputs.iter().enumerate() {
84            match input {
85                CallArg::Pure(_) => (),
86                CallArg::Object(ObjectArg::SharedObject {
87                    id,
88                    mutability:
89                        SharedObjectMutability::NonExclusiveWrite | SharedObjectMutability::Mutable,
90                    ..
91                }) => {
92                    assert_invariant!(
93                        *id == SUI_ACCUMULATOR_ROOT_OBJECT_ID,
94                        "Settlement shared input must be the accumulator root"
95                    );
96                    assert_invariant!(
97                        root_argument.is_none(),
98                        "Settlement must have only one accumulator root input"
99                    );
100                    root_argument = Some(Argument::Input(checked_as!(index, u16)?));
101                }
102                _ => invariant_violation!("Unexpected settlement input: {input:?}"),
103            }
104        }
105        let Some(root_argument) = root_argument else {
106            invariant_violation!("Settlement must have an accumulator root input");
107        };
108        let Some(root) = state.read_object(&SUI_ACCUMULATOR_ROOT_OBJECT_ID).cloned() else {
109            invariant_violation!("Settlement accumulator root is missing");
110        };
111        Ok(Self {
112            state,
113            protocol_config,
114            tx_context,
115            inputs,
116            root_argument,
117            root,
118            children: BTreeMap::new(),
119            generated_ids: BTreeSet::new(),
120            input_sui: 0,
121            output_sui: 0,
122        })
123    }
124
125    fn read_dynamic_field<K: Serialize + DeserializeOwned, V: DeserializeOwned>(
126        &mut self,
127        key: &K,
128        key_type: &TypeTag,
129        value_type: &TypeTag,
130    ) -> Result<(ObjectID, Option<V>), ExecutionError> {
131        let bytes = bcs::to_bytes(key)
132            .map_err(|e| make_invariant_violation!("Cannot serialize settlement key: {e}"))?;
133        let id = derive_dynamic_field_id(SUI_ACCUMULATOR_ROOT_OBJECT_ID, key_type, &bytes)
134            .map_err(|e| make_invariant_violation!("Cannot derive settlement field ID: {e}"))?;
135        let child = match self.children.entry(id) {
136            Entry::Occupied(entry) => entry.into_mut(),
137            Entry::Vacant(entry) => {
138                let original = self
139                    .state
140                    .read_child_object(&SUI_ACCUMULATOR_ROOT_OBJECT_ID, &id, self.root.version())
141                    .map_err(|e| {
142                        make_invariant_violation!("Cannot read settlement field {id}: {e}")
143                    })?;
144                entry.insert(ChildObject {
145                    current: original.clone(),
146                    original,
147                })
148            }
149        };
150        let Some(object) = &child.current else {
151            return Ok((id, None));
152        };
153        assert_invariant!(
154            object.id() == id,
155            "Invalid settlement field ID: expected {id}, got {}",
156            object.id()
157        );
158        assert_invariant!(
159            object.owner == Owner::ObjectOwner(SUI_ACCUMULATOR_ROOT_OBJECT_ID.into()),
160            "Invalid settlement field owner for {id}: {:?}",
161            object.owner
162        );
163        let Some(object) = object.data.try_as_move() else {
164            invariant_violation!("Settlement field {id} must be a Move object");
165        };
166        assert_invariant!(
167            *object.type_()
168                == DynamicFieldInfo::dynamic_field_type(key_type.clone(), value_type.clone())
169                    .into(),
170            "Settlement field {id} has an unexpected type"
171        );
172        let field: Field<K, V> = bcs::from_bytes(object.contents()).map_err(|e| {
173            make_invariant_violation!("Cannot deserialize settlement field {id}: {e}")
174        })?;
175        Ok((id, Some(field.value)))
176    }
177
178    fn write_dynamic_field<K: Serialize, V: Serialize>(
179        &mut self,
180        id: ObjectID,
181        key: K,
182        key_type: TypeTag,
183        value: V,
184        value_type: TypeTag,
185    ) -> Result<(), ExecutionError> {
186        let child = self
187            .children
188            .get_mut(&id)
189            .ok_or_else(|| make_invariant_violation!("Settlement field {id} was not loaded"))?;
190        if child.current.is_none() {
191            self.generated_ids.insert(id);
192        }
193        let contents = bcs::to_bytes(&Field {
194            id: UID::new(id),
195            name: key,
196            value,
197        })
198        .map_err(|e| make_invariant_violation!("Cannot serialize settlement field {id}: {e}"))?;
199        // Field has key but not store, regardless of its type arguments.
200        let object = unsafe {
201            MoveObject::new_from_execution(
202                DynamicFieldInfo::dynamic_field_type(key_type, value_type).into(),
203                /* has_public_transfer */ false,
204                child
205                    .original
206                    .as_ref()
207                    .map_or(SequenceNumber::MIN, |object| object.version()),
208                contents,
209                self.protocol_config,
210                /* system_mutation */ true,
211            )?
212        };
213        child.current = Some(Object::new_move(
214            object,
215            Owner::ObjectOwner(SUI_ACCUMULATOR_ROOT_OBJECT_ID.into()),
216            self.tx_context.digest(),
217        ));
218        Ok(())
219    }
220
221    fn finish(self) -> (ExecutionResultsV2, BTreeSet<ObjectID>) {
222        let Self {
223            root,
224            children,
225            generated_ids,
226            input_sui: settlement_input_sui,
227            output_sui: settlement_output_sui,
228            state: _,
229            protocol_config: _,
230            tx_context: _,
231            inputs: _,
232            root_argument: _,
233        } = self;
234        let root_id = root.id();
235        let mut written_objects = BTreeMap::new();
236        let mut modified_objects = BTreeSet::new();
237        let mut created_object_ids = BTreeSet::new();
238        let mut deleted_object_ids = BTreeSet::new();
239        written_objects.insert(root_id, root);
240        modified_objects.insert(root_id);
241        for (id, ChildObject { original, current }) in children {
242            match (original, current) {
243                (None, None) => (),
244                (None, Some(object)) => {
245                    created_object_ids.insert(id);
246                    written_objects.insert(id, object);
247                }
248                (Some(_), None) => {
249                    modified_objects.insert(id);
250                    deleted_object_ids.insert(id);
251                }
252                (Some(original), Some(object)) => {
253                    // Match Move's child-object change detection:
254                    // unchanged values produce no write.
255                    // Transaction metadata and storage rebates are finalized later.
256                    if original.data != object.data || original.owner != object.owner {
257                        modified_objects.insert(id);
258                        written_objects.insert(id, object);
259                    }
260                }
261            }
262        }
263        let execution_results = ExecutionResultsV2 {
264            settlement_input_sui,
265            settlement_output_sui,
266            written_objects,
267            modified_objects,
268            created_object_ids,
269            deleted_object_ids,
270            user_events: vec![],
271            accumulator_events: vec![],
272        };
273        (execution_results, generated_ids)
274    }
275}

277/// Executes accumulator settlement calls directly in Rust.
278/// Type arguments come from accumulator events and already contain defining IDs. Settlement
279/// only uses those types as phantom dynamic-field keys, so it needs neither code nor layouts.
280pub(crate) fn execute<E: ExecutionErrorTrait>(
281    protocol_config: &ProtocolConfig,
282    state: &mut dyn ExecutionState,
283    tx_context: &TxContext,
284    transaction: ProgrammableTransaction,
285) -> Result<(), E> {
286    let ProgrammableTransaction { inputs, commands } = transaction;
287    let Some(last_command_index) = commands.len().checked_sub(1) else {
288        invariant_violation!("Settlement must contain a prologue");
289    };
290    // Execute the commands
291    let (execution_results, generated_ids) = {
292        let mut context = Context::new(protocol_config, state, tx_context, inputs)?;
293        let command_result = commands
294            .into_iter()
295            .enumerate()
296            .try_for_each(|(index, cmd)| {
297                command(&mut context, index, last_command_index, cmd)
298                    .map_err(|e| e.with_command_index(index))
299            });
300        let loaded = context
301            .children
302            .iter()
303            .filter_map(|(id, child)| {
304                let object = child.original.as_ref()?;
305                Some((
306                    *id,
307                    DynamicallyLoadedObjectMetadata {
308                        version: object.version(),
309                        digest: object.digest(),
310                        owner: object.owner.clone(),
311                        storage_rebate: object.storage_rebate,
312                        previous_transaction: object.previous_transaction,
313                    },
314                ))
315            })
316            .collect();
317        let final_result = context.finish();
318        // Settlement must succeed. If an invariant or storage error is returned, retain
319        // the reads for replay and leave the writes unapplied.
320        state.save_loaded_runtime_objects(loaded);
321        command_result?;
322        final_result
323    };
324    state.record_execution_results(ExecutionResults::V2(execution_results))?;
325    state.record_generated_object_ids(generated_ids);
326    Ok(())
327}

329fn command(
330    context: &mut Context<'_>,
331    index: usize,
332    last_command_index: usize,
333    command: Command,
334) -> Result<(), ExecutionError> {
335    let Command::MoveCall(call) = command else {
336        invariant_violation!("Settlement commands must be Move calls");
337    };
338    let ProgrammableMoveCall {
339        package,
340        module,
341        function,
342        type_arguments,
343        arguments,
344    } = *call;
345    assert_invariant!(
346        package == SUI_FRAMEWORK_PACKAGE_ID,
347        "Unexpected settlement package: {package}"
348    );
349    match (module.as_str(), function.as_str()) {
350        (ACCUMULATOR_SETTLEMENT_MODULE_NAME, SETTLEMENT_PROLOGUE) => {
351            assert_invariant!(index == 0, "Settlement prologue must be the first command");
352            let [] = type_arguments.as_slice() else {
353                invariant_violation!("Settlement prologue expects no type arguments");
354            };
355            let [root, epoch, height, offset, input, output] = arguments.as_slice() else {
356                invariant_violation!("Settlement prologue expects six arguments");
357            };
358            assert_invariant!(
359                *root == context.root_argument,
360                "Settlement call has an incorrect root argument"
361            );
362            // These values distinguish transaction digests. Move also ignores their values,
363            // but execution still validates their encoding as u64 arguments.
364            let _: u64 = pure(&context.inputs, epoch)?;
365            let _: u64 = pure(&context.inputs, height)?;
366            let _: u64 = pure(&context.inputs, offset)?;
367            context.input_sui = pure::<u64>(&context.inputs, input)?;
368            context.output_sui = pure::<u64>(&context.inputs, output)?;
369            Ok(())
370        }
371        (ACCUMULATOR_SETTLEMENT_MODULE_NAME, SETTLE_U128) => {
372            assert_invariant!(index > 0, "Balance settlement must follow the prologue");
373            let [ty] = type_arguments.as_slice() else {
374                invariant_violation!("Balance settlement expects one type argument");
375            };
376            let [root, owner, merge, split] = arguments.as_slice() else {
377                invariant_violation!("Balance settlement expects four arguments");
378            };
379            let ty = ty
380                .to_type_tag()
381                .map_err(|e| make_invariant_violation!("Invalid settlement type: {e}"))?;
382            assert_invariant!(
383                *root == context.root_argument,
384                "Settlement call has an incorrect root argument"
385            );
386            let owner: SuiAddress = pure(&context.inputs, owner)?;
387            let merge: u128 = pure(&context.inputs, merge)?;
388            let split: u128 = pure(&context.inputs, split)?;
389            settle_u128(context, ty, owner, merge, split)
390        }
391        (ACCUMULATOR_SETTLEMENT_MODULE_NAME, SETTLE_EVENTS) => {
392            assert_invariant!(index > 0, "Event settlement must follow the prologue");
393            let [] = type_arguments.as_slice() else {
394                invariant_violation!("Event settlement expects no type arguments");
395            };
396            let [root, stream, new_root, count, checkpoint] = arguments.as_slice() else {
397                invariant_violation!("Event settlement expects five arguments");
398            };
399            assert_invariant!(
400                *root == context.root_argument,
401                "Settlement call has an incorrect root argument"
402            );
403            let stream: SuiAddress = pure(&context.inputs, stream)?;
404            let new_root: U256 = pure(&context.inputs, new_root)?;
405            let count: u64 = pure(&context.inputs, count)?;
406            let checkpoint: u64 = pure(&context.inputs, checkpoint)?;
407            settle_events(context, stream, new_root, count, checkpoint)
408        }
409        (ACCUMULATOR_METADATA_MODULE_NAME, RECORD_ACCUMULATOR_OBJECT_CHANGES) => {
410            assert_invariant!(index > 0, "Object-count update must follow the prologue");
411            assert_invariant!(
412                index == last_command_index,
413                "Object-count update must be the last command"
414            );
415            let [] = type_arguments.as_slice() else {
416                invariant_violation!("Object-count update expects no type arguments");
417            };
418            let [root, created, destroyed] = arguments.as_slice() else {
419                invariant_violation!("Object-count update expects three arguments");
420            };
421            assert_invariant!(
422                *root == context.root_argument,
423                "Settlement call has an incorrect root argument"
424            );
425            let created: u64 = pure(&context.inputs, created)?;
426            let destroyed: u64 = pure(&context.inputs, destroyed)?;
427            record_object_changes(context, created, destroyed)
428        }
429        _ => invariant_violation!("Unexpected settlement call: {module}::{function}"),
430    }
431}

433fn pure<T: DeserializeOwned>(inputs: &[CallArg], argument: &Argument) -> Result<T, ExecutionError> {
434    let Argument::Input(index) = argument else {
435        invariant_violation!("Settlement arguments must refer to inputs: {argument:?}");
436    };
437    let Some(input) = inputs.get(usize::from(*index)) else {
438        invariant_violation!("Settlement input index out of bounds: {index}")
439    };
440    let CallArg::Pure(bytes) = input else {
441        invariant_violation!("Settlement expected a pure argument: {argument:?}");
442    };
443    bcs::from_bytes(bytes)
444        .map_err(|e| make_invariant_violation!("Invalid settlement argument: {e}"))
445}

447fn settle_u128(
448    context: &mut Context<'_>,
449    ty: TypeTag,
450    owner: SuiAddress,
451    merge: u128,
452    split: u128,
453) -> Result<(), ExecutionError> {
454    let key = AccumulatorKey { owner };
455    let key_type = AccumulatorKey::get_type_tag(&[ty]);
456    let (id, previous) =
457        context.read_dynamic_field::<_, U128>(&key, &key_type, &U128::get_type_tag())?;
458    assert_invariant!(
459        previous.is_some() || split == 0,
460        "Cannot withdraw from a missing accumulator"
461    );
462    let previous = previous.map_or(0, |v| v.value);
463    let value = match (merge, split) {
464        (0, 0) => invariant_violation!("Settlement must merge or split a nonzero amount"),
465        (merge, 0) => previous.checked_add(merge),
466        (0, split) => previous.checked_sub(split),
467        (_, _) => invariant_violation!("Settlement cannot both merge and split"),
468    };
469    let Some(value) = value else {
470        invariant_violation!("Settlement balance overflow or underflow");
471    };
472    if value == 0 {
473        context
474            .children
475            .get_mut(&id)
476            .ok_or_else(|| make_invariant_violation!("Settlement field was not loaded"))?
477            .current = None;
478        Ok(())
479    } else {
480        context.write_dynamic_field(id, key, key_type, U128 { value }, U128::get_type_tag())
481    }
482}

484fn settle_events(
485    context: &mut Context<'_>,
486    stream: SuiAddress,
487    new_root: U256,
488    count: u64,
489    checkpoint: u64,
490) -> Result<(), ExecutionError> {
491    let key = AccumulatorKey { owner: stream };
492    let value_type = EventStreamHead::get_type_tag();
493    let key_type = AccumulatorKey::get_type_tag(std::slice::from_ref(&value_type));
494    let (id, previous) =
495        context.read_dynamic_field::<_, EventStreamHead>(&key, &key_type, &value_type)?;
496    let mut head = previous.unwrap_or_default();
497    add_to_mmr(new_root, &mut head.mmr);
498    head.num_events = head
499        .num_events
500        .checked_add(count)
501        .ok_or_else(|| make_invariant_violation!("Settlement event count overflow"))?;
502    head.checkpoint_seq = checkpoint;
503    context.write_dynamic_field(id, key, key_type, head, value_type)
504}
505
506fn add_to_mmr(new_root: U256, mmr: &mut Vec<U256>) {
507    let mut current = new_root;
508    for peak in mmr.iter_mut() {
509        if *peak == U256::zero() {
510            *peak = current;
511            return;
512        }
513        current = hash_two_to_one_u256(*peak, current);
514        *peak = U256::zero();
515    }
516    mmr.push(current);
517}
518
519fn hash_two_to_one_u256(left: U256, right: U256) -> U256 {
520    let mut hash = Blake2b256::default();
521    hash.update(left.to_le_bytes());
522    hash.update(right.to_le_bytes());
523    U256::from_le_bytes(&hash.finalize().digest)
524}

526fn record_object_changes(
527    context: &mut Context<'_>,
528    created: u64,
529    destroyed: u64,
530) -> Result<(), ExecutionError> {
531    let key = AccumulatorObjectCountKey::default();
532    let key_type = AccumulatorObjectCountKey::get_type_tag();
533    let (id, previous) = context.read_dynamic_field::<_, u64>(&key, &key_type, &TypeTag::U64)?;
534    let count = previous
535        .unwrap_or(0)
536        .checked_add(created)
537        .and_then(|v| v.checked_sub(destroyed))
538        .ok_or_else(|| {
539            make_invariant_violation!("Settlement object count overflow or underflow")
540        })?;
541    context.write_dynamic_field(id, key, key_type, count, TypeTag::U64)
542}

544/// These framework layouts contain no SUI. Their type arguments are phantom, so conservation
545/// checks do not need to resolve user packages just to inspect settlement outputs.
546pub(crate) fn has_no_sui(object: &Object) -> bool {
547    static EVENT_STREAM_FIELD_TYPE: LazyLock<StructTag> = LazyLock::new(|| {
548        let event = EventStreamHead::get_type_tag();
549        DynamicFieldInfo::dynamic_field_type(
550            AccumulatorKey::get_type_tag(std::slice::from_ref(&event)),
551            event,
552        )
553    });
554    static OBJECT_COUNT_FIELD_TYPE: LazyLock<StructTag> = LazyLock::new(|| {
555        DynamicFieldInfo::dynamic_field_type(
556            AccumulatorObjectCountKey::get_type_tag(),
557            TypeTag::U64,
558        )
559    });
560
561    let Some(move_object) = object.data.try_as_move() else {
562        return false;
563    };
564    if move_object.type_().is_balance_accumulator_field() {
565        return !move_object.type_().is_sui_balance_accumulator_field();
566    }
567    let Some(tag) = move_object.type_().other() else {
568        return false;
569    };
570    if tag.address == SUI_FRAMEWORK_ADDRESS
571        && tag.module.as_ident_str() == ACCUMULATOR_ROOT_MODULE
572        && tag.name.as_str() == ACCUMULATOR_ROOT_STRUCT_NAME
573        && tag.type_params.is_empty()
574    {
575        return true;
576    }
577    tag == &*EVENT_STREAM_FIELD_TYPE || tag == &*OBJECT_COUNT_FIELD_TYPE
578}
