4//! Tunable RocksDB options, resolved per column family.
5//!
6//! This module turns a serde-friendly [`RocksDbConfig`] into concrete
7//! [`rocksdb::Options`] — one set of database-wide options plus one
8//! set per column family. The intent is that operators tune
9//! performance knobs (compression, write buffers, write-stall
10//! thresholds, compaction parallelism) through configuration, while
11//! correctness-bearing settings (merge operators and compaction
12//! filters) stay in the schema's Rust code and are layered on top of
13//! the resolved options by the schema itself.
14//!
15//! # Resolution model
16//!
17//! Every field is an [`Option`], so three states compose cleanly:
18//!
19//! - `None` — leave the RocksDB native default in place (the setter
20//!   is simply not called).
21//! - a code-supplied default — the downstream crate that owns the
22//!   schema provides a fully populated [`RocksDbConfig`] as its
23//!   baseline (see `sui_rpc_store::default_rocksdb_config`).
24//! - a configuration override — values parsed from TOML are layered
25//!   over the code defaults via [`RocksDbConfig::merge_over`].
26//!
27//! Within a single resolved config, a per-column-family entry in
28//! [`RocksDbConfig::column_family`] is layered over
29//! [`RocksDbConfig::default_cf`] at resolve time, so a CF override
30//! only has to name the fields it wants to change.
31//!
32//! # What is *not* configurable here
33//!
34//! Merge operators and compaction filters are correctness-bearing and
35//! deliberately absent from [`CfTuning`]. The schema attaches them in
36//! code, on top of the [`rocksdb::Options`] this module produces.
37

45/// Compression algorithm for a column family.
46///
47/// Maps onto [`rocksdb::DBCompressionType`]. The variants here are the
48/// ones the crate's RocksDB build enables.
49#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
50#[serde(rename_all = "kebab-case")]
51pub enum Compression {
52    None,
53    Snappy,
54    Lz4,
55    Zstd,
56    Zlib,
57}
58
59impl From<Compression> for rocksdb::DBCompressionType {
60    fn from(c: Compression) -> Self {
61        match c {
62            Compression::None => Self::None,
63            Compression::Snappy => Self::Snappy,
64            Compression::Lz4 => Self::Lz4,
65            Compression::Zstd => Self::Zstd,
66            Compression::Zlib => Self::Zlib,
67        }
68    }
69}

71/// Write-stall thresholds for a column family.
72///
73/// RocksDB slows or stops writes when L0 file counts or the estimated
74/// pending-compaction byte debt cross these thresholds. The values
75/// are exposed individually (rather than behind a single "disable"
76/// flag) so each can be tuned per CF, and so the chosen policy is
77/// explicit rather than hidden behind a boolean.
78///
79/// The pending-compaction limits use a sentinel of `0` to mean
80/// "disabled". The L0 file-count triggers have no disable sentinel;
81/// the idiom is to set them to a value high enough never to fire.
82#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
83#[serde(rename_all = "kebab-case", default, deny_unknown_fields)]
84pub struct WriteStallConfig {
85    /// Soft limit on estimated pending-compaction bytes before writes
86    /// slow down, in MiB. `0` disables the soft pending-bytes stall.
87    #[serde(skip_serializing_if = "Option::is_none")]
88    pub soft_pending_compaction_bytes_limit_mb: Option<u64>,
89
90    /// Hard limit on estimated pending-compaction bytes before writes
91    /// stop, in MiB. `0` disables the hard pending-bytes stall.
92    #[serde(skip_serializing_if = "Option::is_none")]
93    pub hard_pending_compaction_bytes_limit_mb: Option<u64>,
94
95    /// Number of L0 files that triggers a compaction. Forms the floor
96    /// of the L0 trigger chain; must not exceed the slowdown trigger.
97    #[serde(skip_serializing_if = "Option::is_none")]
98    pub level0_file_num_compaction_trigger: Option<i32>,
99
100    /// Number of L0 files at which writes are slowed. Must be greater
101    /// than or equal to the compaction trigger and less than or equal
102    /// to the stop trigger.
103    #[serde(skip_serializing_if = "Option::is_none")]
104    pub level0_slowdown_writes_trigger: Option<i32>,
105
106    /// Number of L0 files at which writes are stopped. Top of the L0
107    /// trigger chain.
108    #[serde(skip_serializing_if = "Option::is_none")]
109    pub level0_stop_writes_trigger: Option<i32>,
110}
111
112impl WriteStallConfig {
113    /// Whether every threshold is unset. Used to skip serializing an
114    /// otherwise-empty `[...write-stall]` table.
115    fn is_empty(&self) -> bool {
116        *self == Self::default()
117    }
118
119    /// Field-wise overlay: every field set on `self` wins; otherwise
120    /// fall back to `base`.
121    pub fn merge_over(&self, base: &Self) -> Self {
122        Self {
123            soft_pending_compaction_bytes_limit_mb: self
124                .soft_pending_compaction_bytes_limit_mb
125                .or(base.soft_pending_compaction_bytes_limit_mb),
126            hard_pending_compaction_bytes_limit_mb: self
127                .hard_pending_compaction_bytes_limit_mb
128                .or(base.hard_pending_compaction_bytes_limit_mb),
129            level0_file_num_compaction_trigger: self
130                .level0_file_num_compaction_trigger
131                .or(base.level0_file_num_compaction_trigger),
132            level0_slowdown_writes_trigger: self
133                .level0_slowdown_writes_trigger
134                .or(base.level0_slowdown_writes_trigger),
135            level0_stop_writes_trigger: self
136                .level0_stop_writes_trigger
137                .or(base.level0_stop_writes_trigger),
138        }
139    }
140
141    /// Validate the relationships RocksDB requires between the set
142    /// fields. Only constraints between fields that are both present
143    /// are checked; absent fields fall back to RocksDB defaults that
144    /// are internally consistent.
145    ///
146    /// `cf` names the column family for error messages.
147    fn validate(&self, cf: &str) -> Result<(), OpenError> {
148        let c = self.level0_file_num_compaction_trigger;
149        let sd = self.level0_slowdown_writes_trigger;
150        let st = self.level0_stop_writes_trigger;
151
152        for (name, value) in [
153            ("level0-file-num-compaction-trigger", c),
154            ("level0-slowdown-writes-trigger", sd),
155            ("level0-stop-writes-trigger", st),
156        ] {
157            if let Some(v) = value
158                && v < 0
159            {
160                return Err(OpenError::msg(format!(
161                    "rocksdb config for column family `{cf}`: {name} must be non-negative, got {v}"
162                )));
163            }
164        }
165
166        if let (Some(c), Some(sd)) = (c, sd)
167            && c > sd
168        {
169            return Err(OpenError::msg(format!(
170                "rocksdb config for column family `{cf}`: level0-file-num-compaction-trigger \
171                 ({c}) must be <= level0-slowdown-writes-trigger ({sd})"
172            )));
173        }
174        if let (Some(sd), Some(st)) = (sd, st)
175            && sd > st
176        {
177            return Err(OpenError::msg(format!(
178                "rocksdb config for column family `{cf}`: level0-slowdown-writes-trigger ({sd}) \
179                 must be <= level0-stop-writes-trigger ({st})"
180            )));
181        }
182        if let (Some(c), Some(st)) = (c, st)
183            && c > st
184        {
185            return Err(OpenError::msg(format!(
186                "rocksdb config for column family `{cf}`: level0-file-num-compaction-trigger ({c}) \
187                 must be <= level0-stop-writes-trigger ({st})"
188            )));
189        }
190
191        if let (Some(soft), Some(hard)) = (
192            self.soft_pending_compaction_bytes_limit_mb,
193            self.hard_pending_compaction_bytes_limit_mb,
194        ) && soft != 0
195            && hard != 0
196            && soft > hard
197        {
198            return Err(OpenError::msg(format!(
199                "rocksdb config for column family `{cf}`: soft-pending-compaction-bytes-limit-mb \
200                 ({soft}) must be <= hard-pending-compaction-bytes-limit-mb ({hard})"
201            )));
202        }
203
204        Ok(())
205    }
206
207    /// Apply the set fields to `opts`.
208    fn apply(&self, opts: &mut rocksdb::Options) {
209        if let Some(mb) = self.soft_pending_compaction_bytes_limit_mb {
210            opts.set_soft_pending_compaction_bytes_limit(mib_usize(mb));
211        }
212        if let Some(mb) = self.hard_pending_compaction_bytes_limit_mb {
213            opts.set_hard_pending_compaction_bytes_limit(mib_usize(mb));
214        }
215        if let Some(n) = self.level0_file_num_compaction_trigger {
216            opts.set_level_zero_file_num_compaction_trigger(n);
217        }
218        if let Some(n) = self.level0_slowdown_writes_trigger {
219            opts.set_level_zero_slowdown_writes_trigger(n);
220        }
221        if let Some(n) = self.level0_stop_writes_trigger {
222            opts.set_level_zero_stop_writes_trigger(n);
223        }
224    }
225}

227/// Sliding-window deletion-compaction trigger for a column family.
228///
229/// Wraps RocksDB's `CompactOnDeletionCollectorFactory`: while writing
230/// an SST, RocksDB marks the file for compaction whenever any
231/// `window_size` consecutive entries contain at least
232/// `deletion_trigger` deletes, or when the whole file's deletion ratio
233/// reaches `deletion_ratio`. Setting `window_size = 0` disables the
234/// sliding window and `deletion_ratio = 0.0` disables the ratio check.
235#[derive(Clone, Copy, Debug, Default, PartialEq, Serialize, Deserialize)]
236#[serde(rename_all = "kebab-case", default, deny_unknown_fields)]
237pub struct CompactOnDeletionConfig {
238    /// Sliding window size, in entries. `0` disables the sliding window.
239    #[serde(skip_serializing_if = "Option::is_none")]
240    pub window_size: Option<usize>,
241
242    /// Deletion count within `window_size` entries that marks the SST
243    /// for compaction.
244    #[serde(skip_serializing_if = "Option::is_none")]
245    pub deletion_trigger: Option<usize>,
246
247    /// Whole-SST deletion fraction in `(0.0, 1.0]` that marks the SST for
248    /// compaction. `0.0` disables the whole-file ratio check.
249    #[serde(skip_serializing_if = "Option::is_none")]
250    pub deletion_ratio: Option<f64>,
251}
252
253impl CompactOnDeletionConfig {
254    fn is_empty(&self) -> bool {
255        *self == Self::default()
256    }
257
258    /// Field-wise overlay: every field set on `self` wins; otherwise
259    /// fall back to `base`. Setting `window_size` to `Some(0)` without
260    /// an explicit `deletion_trigger` also clears an inherited trigger
261    /// to `Some(0)` so disabling the sliding window only requires
262    /// zeroing `window_size`.
263    pub fn merge_over(&self, base: &Self) -> Self {
264        let window_size = self.window_size.or(base.window_size);
265        let deletion_trigger = match (self.window_size, self.deletion_trigger) {
266            (Some(0), None) => Some(0),
267            (_, trigger) => trigger.or(base.deletion_trigger),
268        };
269        Self {
270            window_size,
271            deletion_trigger,
272            deletion_ratio: self.deletion_ratio.or(base.deletion_ratio),
273        }
274    }
275
276    fn validate(&self, cf: &str) -> Result<(), OpenError> {
277        if let Some(ratio) = self.deletion_ratio
278            && (!ratio.is_finite() || !(0.0..=1.0).contains(&ratio))
279        {
280            return Err(OpenError::msg(format!(
281                "rocksdb config for column family `{cf}`: compact-on-deletion deletion-ratio \
282                 must be in 0.0..=1.0, got {ratio}"
283            )));
284        }
285
286        let window = self.window_size.unwrap_or(0);
287        let trigger = self.deletion_trigger.unwrap_or(0);
288        if window > 0 && (trigger == 0 || trigger > window) {
289            return Err(OpenError::msg(format!(
290                "rocksdb config for column family `{cf}`: compact-on-deletion deletion-trigger \
291                 ({trigger}) must be in 1..=window-size ({window})"
292            )));
293        }
294        if window == 0 && trigger > 0 {
295            return Err(OpenError::msg(format!(
296                "rocksdb config for column family `{cf}`: compact-on-deletion deletion-trigger \
297                 ({trigger}) requires a non-zero window-size"
298            )));
299        }
300
301        Ok(())
302    }
303
304    fn apply(&self, opts: &mut rocksdb::Options) {
305        let window = self.window_size.unwrap_or(0);
306        let trigger = self.deletion_trigger.unwrap_or(0);
307        let ratio = self.deletion_ratio.unwrap_or(0.0);
308        if window > 0 || ratio > 0.0 {
309            opts.add_compact_on_deletion_collector_factory(window, trigger, ratio);
310        }
311    }
312}

314/// Per-column-family performance tuning.
315///
316/// Correctness-bearing settings (merge operators, compaction filters)
317/// are intentionally not here; the schema adds those in code.
318#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
319#[serde(rename_all = "kebab-case", default, deny_unknown_fields)]
320pub struct CfTuning {
321    /// Per-memtable size, in MiB.
322    #[serde(skip_serializing_if = "Option::is_none")]
323    pub write_buffer_size_mb: Option<usize>,
324
325    /// Maximum number of memtables before writes are throttled by
326    /// flush back-pressure.
327    #[serde(skip_serializing_if = "Option::is_none")]
328    pub max_write_buffer_number: Option<i32>,
329
330    /// Compression for all but the bottommost level.
331    #[serde(skip_serializing_if = "Option::is_none")]
332    pub compression: Option<Compression>,
333
334    /// Compression for the bottommost level (where most data settles).
335    #[serde(skip_serializing_if = "Option::is_none")]
336    pub bottommost_compression: Option<Compression>,
337
338    /// Block size, in KiB.
339    #[serde(skip_serializing_if = "Option::is_none")]
340    pub block_size_kb: Option<usize>,
341
342    /// Bits per key for a full (non-block-based) bloom filter. Set on
343    /// point-lookup-heavy CFs; leave unset on range-scanned CFs.
344    #[serde(skip_serializing_if = "Option::is_none")]
345    pub bloom_filter_bits: Option<f64>,
346
347    /// Memtable prefix-bloom size as a fraction of the memtable.
348    #[serde(skip_serializing_if = "Option::is_none")]
349    pub memtable_prefix_bloom_ratio: Option<f64>,
350
351    /// Target SST file size at level base, in MiB.
352    #[serde(skip_serializing_if = "Option::is_none")]
353    pub target_file_size_mb: Option<u64>,
354
355    /// SST age, in seconds, after which RocksDB may include a file in
356    /// periodic compaction. Zero disables periodic compaction.
357    #[serde(skip_serializing_if = "Option::is_none")]
358    pub periodic_compaction_seconds: Option<u64>,
359
360    /// Write-stall thresholds.
361    #[serde(default, skip_serializing_if = "WriteStallConfig::is_empty")]
362    pub write_stall: WriteStallConfig,
363
364    /// Deletion-triggered compaction collector.
365    #[serde(default, skip_serializing_if = "CompactOnDeletionConfig::is_empty")]
366    pub compact_on_deletion: CompactOnDeletionConfig,
367}

369impl CfTuning {
370    /// Field-wise overlay: every field set on `self` wins; otherwise
371    /// fall back to `base`. The nested [`WriteStallConfig`] is merged
372    /// recursively.
373    pub fn merge_over(&self, base: &Self) -> Self {
374        Self {
375            write_buffer_size_mb: self.write_buffer_size_mb.or(base.write_buffer_size_mb),
376            max_write_buffer_number: self
377                .max_write_buffer_number
378                .or(base.max_write_buffer_number),
379            compression: self.compression.or(base.compression),
380            bottommost_compression: self.bottommost_compression.or(base.bottommost_compression),
381            block_size_kb: self.block_size_kb.or(base.block_size_kb),
382            bloom_filter_bits: self.bloom_filter_bits.or(base.bloom_filter_bits),
383            memtable_prefix_bloom_ratio: self
384                .memtable_prefix_bloom_ratio
385                .or(base.memtable_prefix_bloom_ratio),
386            target_file_size_mb: self.target_file_size_mb.or(base.target_file_size_mb),
387            periodic_compaction_seconds: self
388                .periodic_compaction_seconds
389                .or(base.periodic_compaction_seconds),
390            write_stall: self.write_stall.merge_over(&base.write_stall),
391            compact_on_deletion: self
392                .compact_on_deletion
393                .merge_over(&base.compact_on_deletion),
394        }
395    }
396
397    fn validate(&self, cf: &str) -> Result<(), OpenError> {
398        self.write_stall.validate(cf)?;
399        self.compact_on_deletion.validate(cf)?;
400        Ok(())
401    }
402
403    /// Build a fresh [`rocksdb::Options`] with the set fields applied.
404    /// `block_cache`, when present, is shared into this CF's
405    /// block-based table factory.
406    fn to_options(&self, block_cache: Option<&rocksdb::Cache>) -> rocksdb::Options {
407        let mut opts = rocksdb::Options::default();
408
409        if let Some(mb) = self.write_buffer_size_mb {
410            opts.set_write_buffer_size(mib_usize(mb as u64));
411        }
412        if let Some(n) = self.max_write_buffer_number {
413            opts.set_max_write_buffer_number(n);
414        }
415        if let Some(c) = self.compression {
416            opts.set_compression_type(c.into());
417        }
418        if let Some(c) = self.bottommost_compression {
419            opts.set_bottommost_compression_type(c.into());
420        }
421
422        // The block-based table factory bundles block size, the shared
423        // block cache, and the bloom filter, so it is constructed once
424        // and only when at least one of them is configured.
425        if self.block_size_kb.is_some() || block_cache.is_some() || self.bloom_filter_bits.is_some()
426        {
427            let mut bbt = rocksdb::BlockBasedOptions::default();
428            if let Some(kb) = self.block_size_kb {
429                bbt.set_block_size(kb << 10);
430            }
431            if let Some(cache) = block_cache {
432                bbt.set_block_cache(cache);
433            }
434            if let Some(bits) = self.bloom_filter_bits {
435                bbt.set_bloom_filter(bits, false);
436            }
437            opts.set_block_based_table_factory(&bbt);
438        }
439
440        if let Some(ratio) = self.memtable_prefix_bloom_ratio {
441            opts.set_memtable_prefix_bloom_ratio(ratio);
442        }
443        if let Some(mb) = self.target_file_size_mb {
444            opts.set_target_file_size_base(mb << 20);
445        }
446        if let Some(seconds) = self.periodic_compaction_seconds {
447            opts.set_periodic_compaction_seconds(seconds);
448        }
449
450        self.write_stall.apply(&mut opts);
451        self.compact_on_deletion.apply(&mut opts);
452
453        opts
454    }
455}

518/// Tunable RocksDB configuration for a [`Db`](crate::Db).
519///
520/// Combines database-wide options, a default per-CF profile applied to
521/// every column family, and per-CF overrides keyed by column-family
522/// name. See the [module docs](self) for the resolution model.
523#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
524#[serde(rename_all = "kebab-case", default, deny_unknown_fields)]
525pub struct RocksDbConfig {
526    /// Database-wide options.
527    pub db: DbWideConfig,
528
529    /// Profile applied to every column family as the baseline.
530    pub default_cf: CfTuning,
531
532    /// Per-column-family overrides, layered over [`Self::default_cf`].
533    /// Keyed by column-family name.
534    pub column_family: BTreeMap<String, CfTuning>,
535}
536
537impl RocksDbConfig {
538    /// Field-wise overlay producing the effective configuration:
539    /// values set on `self` (typically parsed from TOML) win;
540    /// otherwise fall back to `base` (typically the code defaults).
541    /// The `column_family` maps are unioned by key, with each shared
542    /// key's [`CfTuning`] merged recursively.
543    pub fn merge_over(&self, base: &Self) -> Self {
544        let mut column_family = base.column_family.clone();
545        for (name, cf) in &self.column_family {
546            let merged = match base.column_family.get(name) {
547                Some(b) => cf.merge_over(b),
548                None => cf.clone(),
549            };
550            column_family.insert(name.clone(), merged);
551        }
552        Self {
553            db: self.db.merge_over(&base.db),
554            default_cf: self.default_cf.merge_over(&base.default_cf),
555            column_family,
556        }
557    }
558
559    /// Validate the write-stall threshold relationships for the
560    /// default profile and for every per-CF override (resolved
561    /// against the default profile). Returns the first violation.
562    pub fn validate(&self) -> Result<(), OpenError> {
563        self.default_cf.validate("default-cf")?;
564        for (name, cf) in &self.column_family {
565            cf.merge_over(&self.default_cf).validate(name)?;
566        }
567        Ok(())
568    }
569}

571/// Resolves per-column-family [`rocksdb::Options`] from a validated
572/// [`RocksDbConfig`].
573///
574/// Built once by [`Db::open`](crate::Db::open) and handed to
575/// [`Schema::open`](crate::Schema::open). Owns the single shared block
576/// cache so every CF that opts into a block cache shares one instance.
577pub struct CfOptionsResolver {
578    config: RocksDbConfig,
579    block_cache: Option<rocksdb::Cache>,
580}
581
582impl CfOptionsResolver {
583    /// Validate `config` and build the resolver, allocating the shared
584    /// block cache if one is configured.
585    pub fn new(config: RocksDbConfig) -> Result<Self, OpenError> {
586        config.validate()?;
587        let block_cache = config
588            .db
589            .block_cache_size_mb
590            .map(|mb| rocksdb::Cache::new_lru_cache(mib_usize(mb as u64)));
591        Ok(Self {
592            config,
593            block_cache,
594        })
595    }
596
597    /// Database-wide options for `open_cf_descriptors`. Always sets
598    /// `create_if_missing` and `create_missing_column_families`, and
599    /// enables RocksDB statistics so [`Db::compaction_bytes`](crate::Db::compaction_bytes)
600    /// can read the compaction byte tickers; the remaining knobs are
601    /// applied only when configured.
602    pub(crate) fn db_options(&self) -> rocksdb::Options {
603        let mut opts = rocksdb::Options::default();
604        opts.create_if_missing(true);
605        opts.create_missing_column_families(true);
606        opts.enable_statistics();
607        // rust-rocksdb's `StatsLevel` discriminants are one above the C
608        // API's, so this selects RocksDB's `kExceptTimers`: tickers and
609        // size histograms, with timer histograms off. RocksDB still times
610        // every merge-operator call whenever statistics are enabled.
611        opts.set_statistics_level(rocksdb::statistics::StatsLevel::ExceptHistogramOrTimers);
612
613        let db = &self.config.db;
614        if let Some(n) = db.parallelism {
615            opts.increase_parallelism(n);
616        }
617        if let Some(n) = db.max_background_jobs {
618            opts.set_max_background_jobs(n);
619        }
620        if let Some(n) = db.max_open_files {
621            opts.set_max_open_files(n);
622        }
623        if let Some(mb) = db.db_write_buffer_size_mb {
624            opts.set_db_write_buffer_size(mib_usize(mb as u64));
625        }
626        if let Some(mb) = db.max_total_wal_size_mb {
627            opts.set_max_total_wal_size(mb << 20);
628        }
629        if let Some(v) = db.enable_pipelined_write {
630            opts.set_enable_pipelined_write(v);
631        }
632        if let Some(bits) = db.table_cache_num_shard_bits {
633            opts.set_table_cache_num_shard_bits(bits);
634        }
635        opts
636    }
637
638    /// Resolved [`rocksdb::Options`] for the column family named `cf`:
639    /// the default profile with the per-CF override (if any) layered
640    /// on top, plus the shared block cache. Does **not** attach merge
641    /// operators or compaction filters — the schema adds those.
642    pub fn options(&self, cf: &str) -> rocksdb::Options {
643        let tuning = match self.config.column_family.get(cf) {
644            Some(o) => o.merge_over(&self.config.default_cf),
645            None => self.config.default_cf.clone(),
646        };
647        tuning.to_options(self.block_cache.as_ref())
648    }
649
650    /// Names of the column families with an explicit per-CF override.
651    /// Used by [`Db::open_cfs`](crate::Db::open_cfs) to reject
652    /// overrides that name a column family the schema does not declare.
653    pub(crate) fn configured_cf_names(&self) -> impl Iterator<Item = &str> {
654        self.config.column_family.keys().map(String::as_str)
655    }
656}
657
658/// Convert a MiB count to bytes as a `usize`, saturating rather than
659/// overflowing on absurd inputs.
660fn mib_usize(mb: u64) -> usize {
661    mb.saturating_mul(1024 * 1024)
662        .try_into()
663        .unwrap_or(usize::MAX)
664}

908    fn compact_on_deletion_validation_and_merge() {
909        let base = CfTuning {
910            compact_on_deletion: CompactOnDeletionConfig {
911                window_size: Some(10_000),
912                deletion_trigger: Some(8_000),
913                deletion_ratio: Some(0.0),
914            },
915            ..Default::default()
916        };
917        // Zeroing window_size alone clears the inherited deletion_trigger.
918        let disabled = CfTuning {
919            compact_on_deletion: CompactOnDeletionConfig {
920                window_size: Some(0),
921                ..Default::default()
922            },
923            ..Default::default()
924        };
925        let merged = disabled.merge_over(&base);
926        assert_eq!(merged.compact_on_deletion.window_size, Some(0));
927        assert_eq!(merged.compact_on_deletion.deletion_trigger, Some(0));
928        assert_eq!(merged.compact_on_deletion.deletion_ratio, Some(0.0));
929        assert!(merged.validate("object_by_owner").is_ok());
930
931        let zero_window_with_trigger = CfTuning {
932            compact_on_deletion: CompactOnDeletionConfig {
933                window_size: Some(0),
934                deletion_trigger: Some(500),
935                ..Default::default()
936            },
937            ..Default::default()
938        };
939        assert!(
940            zero_window_with_trigger
941                .validate("object_by_owner")
942                .is_err()
943        );
944
945        let bad_trigger = CfTuning {
946            compact_on_deletion: CompactOnDeletionConfig {
947                window_size: Some(1_000),
