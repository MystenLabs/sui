41/// Version assignments for a single transaction
42#[derive(Debug, Clone, PartialEq, Eq)]
43pub struct AssignedVersions {
44    pub shared_object_versions: Vec<(ConsensusObjectSequenceKey, SequenceNumber)>,
45    /// Versions of system objects, keyed by object ID, that this transaction may read during
46    /// execution but that are not part of its declared shared inputs. Each version is assigned
47    /// deterministically during consensus sequencing, so that every validator reads the same
48    /// version of the object.
49    ///
50    /// Today this holds the accumulator root version (as of the beginning of the consensus
51    /// commit this transaction belongs to) and the forwarding address registry version (as of the
52    /// point in the commit at which this transaction is sequenced). The accumulator root qualifies
53    /// because it is written at the end of every commit, so there is always a well-defined prior
54    /// version to read from.
55    pub system_object_versions: SystemObjectVersions,
56}
57
58impl AssignedVersions {
59    pub fn new(
60        shared_object_versions: Vec<(ConsensusObjectSequenceKey, SequenceNumber)>,
61        system_object_versions: SystemObjectVersions,
62    ) -> Self {
63        debug_assert!(
64            shared_object_versions
65                .iter()
66                .filter(|((id, _), version)| {
67                    id.is_implicitly_read_system_object() && !version.is_cancelled()
68                })
69                .all(|((id, _), version)| {
70                    system_object_versions
71                        .get(id)
72                        .is_none_or(|system_version| system_version.version == *version)
73                }),
74            "declared versions {shared_object_versions:?} disagree with {system_object_versions:?}"
75        );
76        Self {
77            shared_object_versions,
78            system_object_versions,
79        }
80    }
81
82    pub fn empty() -> Self {
83        Self::new(vec![], SystemObjectVersions::empty())
84    }
85
86    /// Construct with only the accumulator root as the system object read during execution.
87    /// Production callers build the full `system_object_versions` map directly.
88    #[cfg(test)]
89    pub fn new_for_testing(
90        shared_object_versions: Vec<(ConsensusObjectSequenceKey, SequenceNumber)>,
91        accumulator_version: Option<SequenceNumber>,
92    ) -> Self {
93        Self::new(
94            shared_object_versions,
95            SystemObjectVersions::new(
96                accumulator_version.map(|v| ConsensusObjectVersion {
97                    initial_shared_version: sui_types::object::OBJECT_START_VERSION,
98                    version: v,
99                }),
100                None,
101            ),
102        )
103    }
104
105    /// The accumulator root version this transaction reads, if any.
106    pub fn accumulator_version(&self) -> Option<SequenceNumber> {
107        self.system_object_versions
108            .get(&SUI_ACCUMULATOR_ROOT_OBJECT_ID)
109            .map(|v| v.version)
110    }
111
112    pub fn iter(&self) -> impl Iterator<Item = &(ConsensusObjectSequenceKey, SequenceNumber)> {
113        self.shared_object_versions.iter()
114    }
115
116    pub fn as_slice(&self) -> &[(ConsensusObjectSequenceKey, SequenceNumber)] {
117        &self.shared_object_versions
118    }
119}

321    pub fn assign_versions_from_consensus<'a, T>(
322        epoch_store: &AuthorityPerEpochStore,
323        cache_reader: &dyn ObjectCacheRead,
324        assignables: impl Iterator<Item = &'a Schedulable<T>> + Clone,
325        cancelled_txns: &BTreeMap<TransactionDigest, CancelConsensusCertificateReason>,
326    ) -> SuiResult<ConsensusSharedObjVerAssignment>
327    where
328        T: AsTx + 'a,
329    {
330        let mut shared_input_next_versions = get_or_init_versions(
331            assignables
332                .clone()
333                .flat_map(|a| a.shared_input_objects(epoch_store)),
334            epoch_store,
335            cache_reader,
336        )?;
337        let mut assigned_versions = Vec::new();
338        for assignable in assignables {
339            assert!(
340                !matches!(assignable, Schedulable::AccumulatorSettlement(_, _))
341                    || epoch_store.accumulators_enabled(),
342                "AccumulatorSettlement should not be scheduled when accumulators are disabled"
343            );
344
345            let cert_assigned_versions = Self::assign_versions_for_certificate(
346                epoch_store,
347                assignable,
348                &mut shared_input_next_versions,
349                cancelled_txns,
350            );
351            assigned_versions.push((assignable.key(), cert_assigned_versions));
352        }
353
354        Ok(ConsensusSharedObjVerAssignment {
355            shared_input_next_versions,
356            assigned_versions: AssignedTxAndVersions::new(assigned_versions),
357        })
358    }

360    pub fn assign_versions_from_effects(
361        certs_and_effects: &[(
362            &VerifiedExecutableTransaction,
363            &TransactionEffects,
364            // Accumulator version
365            Option<SequenceNumber>,
366        )],
367        epoch_store: &AuthorityPerEpochStore,
368        cache_reader: &dyn ObjectCacheRead,
369    ) -> AssignedTxAndVersions {
370        // We don't care about the results since we can use effects to assign versions.
371        // But we must call it to make sure whenever a consensus object is touched the first time
372        // during an epoch, either through consensus or through checkpoint executor,
373        // its next version must be initialized. This is because we initialize the next version
374        // of a consensus object in an epoch by reading the current version from the object store.
375        // This must be done before we mutate it the first time, otherwise we would be initializing
376        // it with the wrong version.
377        let _ = get_or_init_versions(
378            certs_and_effects.iter().flat_map(|(cert, _, _)| {
379                cert.transaction_data().shared_input_objects().into_iter()
380            }),
381            epoch_store,
382            cache_reader,
383        );
384        let mut assigned_versions = Vec::new();
385        for (cert, effects, accumulator_version) in certs_and_effects {
386            let initial_version_map: BTreeMap<_, _> = cert
387                .transaction_data()
388                .shared_input_objects()
389                .into_iter()
390                .map(|input| input.into_id_and_version())
391                .collect();
392            // Implicitly read system objects retrieve their accessed versions from this map.
393            let accessed_versions: BTreeMap<ObjectID, SequenceNumber> = effects
394                .accessed_consensus_objects()
395                .into_iter()
396                .map(|iso| iso.id_and_version())
397                .collect();
398            let cert_assigned_versions: Vec<_> = accessed_versions
399                .iter()
400                .filter_map(|(id, version)| {
401                    let v = initial_version_map
402                        .get(id)
403                        .map(|initial_version| ((*id, *initial_version), *version));
404                    if v.is_none() {
405                        debug_assert!(
406                            IMPLICITLY_READ_SYSTEM_OBJECTS.contains(id),
407                            "accessed consensus object is neither a declared input nor a known implicitly read system object: \
408                             accessed={accessed_versions:?} declared={initial_version_map:?}"
409                        );
410                    }
411                    v
412                })
413                .collect();
414            if let (Some(effects_version), Some(sequenced_version)) = (
415                accessed_versions.get(&SUI_ACCUMULATOR_ROOT_OBJECT_ID),
416                accumulator_version,
417            ) && !effects_version.is_cancelled()
418                && effects_version != sequenced_version
419            {
420                debug_fatal!(
421                    "accumulator root version from effects {:?} disagrees \
422                        with the reconstructed accumulator version {:?} for tx {:?}",
423                    effects_version,
424                    sequenced_version,
425                    cert.digest()
426                );
427            }
428            let mut system_object_versions: BTreeMap<_, _> =
429                implicitly_read_system_objects(epoch_store)
430                    .into_iter()
431                    .filter_map(|(id, initial_shared_version)| {
432                        let version = *accessed_versions.get(&id)?;
433                        // A cancelled sentinel only appears when the object was also a declared
434                        // input of a cancelled transaction; it is not a version to pin.
435                        (!version.is_cancelled()).then_some((
436                            id,
437                            ConsensusObjectVersion {
438                                initial_shared_version,
439                                version,
440                            },
441                        ))
442                    })
443                    .collect();
444            // Note that for accumulator version, we cannot rely on the one from effects yet, since it won't
445            // be produced until implicitly read system objects are fully shipped. But the old object funds withdraw
446            // still need it. Hence we always use the one provided from the caller (i.e. checkpoint executor).
447            // TODO: Once check_object_funds_withdraw_in_execution is enabled everywhere and the
448            // post-execution ObjectFundsChecker is removed, drop this override (and the
449            // accumulator version threaded from the checkpoint executor) so the accumulator root
450            // is reconstructed from effects exactly like the forwarding address registry.
451            if let Some(version) = *accumulator_version {
452                let initial_shared_version = epoch_store
453                    .epoch_start_config()
454                    .accumulator_root_obj_initial_shared_version()
455                    .expect(
456                        "initial shared version must be known for an implicitly read system object",
457                    );
458                system_object_versions.insert(
459                    SUI_ACCUMULATOR_ROOT_OBJECT_ID,
460                    ConsensusObjectVersion {
461                        initial_shared_version,
462                        version,
463                    },
464                );
465            }
466            let system_object_versions = SystemObjectVersions::from_map(system_object_versions);
467            let tx_key = cert.key();
468            trace!(
469                ?tx_key,
470                ?cert_assigned_versions,
471                ?system_object_versions,
472                "assigned consensus object versions from effects"
473            );
474            assigned_versions.push((
475                tx_key,
476                AssignedVersions::new(cert_assigned_versions, system_object_versions),
477            ));
478        }
479        AssignedTxAndVersions::new(assigned_versions)
480    }

482    pub fn assign_versions_for_certificate(
483        epoch_store: &AuthorityPerEpochStore,
484        assignable: &Schedulable<impl AsTx>,
485        shared_input_next_versions: &mut HashMap<ConsensusObjectSequenceKey, SequenceNumber>,
486        cancelled_txns: &BTreeMap<TransactionDigest, CancelConsensusCertificateReason>,
487    ) -> AssignedVersions {
488        let shared_input_objects: Vec<_> = assignable.shared_input_objects(epoch_store).collect();
489
490        let system_object_versions = SystemObjectVersions::from_map(
491            implicitly_read_system_objects(epoch_store)
492                .into_iter()
493                .map(|(id, initial_shared_version)| {
494                    let version = *shared_input_next_versions
495                        .get(&(id, initial_shared_version))
496                        .expect(
497                            "implicitly read system objects must be in shared_input_next_versions",
498                        );
499                    (
500                        id,
501                        ConsensusObjectVersion {
502                            initial_shared_version,
503                            version,
504                        },
505                    )
506                })
507                .collect(),
508        );
509
510        if shared_input_objects.is_empty() {
511            // No shared object used by this transaction. No need to assign versions.
512            return AssignedVersions::new(vec![], system_object_versions);
513        }
514
515        let tx_key = assignable.key();
516
517        // Check if the transaction is cancelled due to congestion.
518        let cancellation_info = tx_key
519            .as_digest()
520            .and_then(|tx_digest| cancelled_txns.get(tx_digest));
521        let congested_objects_info: Option<HashSet<_>> =
522            if let Some(CancelConsensusCertificateReason::CongestionOnObjects(congested_objects)) =
523                &cancellation_info
524            {
525                Some(congested_objects.iter().cloned().collect())
526            } else {
527                None
528            };
529        let txn_cancelled = cancellation_info.is_some();
530
531        let mut input_object_keys = assignable.non_shared_input_object_keys();
532        let mut assigned_versions = Vec::with_capacity(shared_input_objects.len());
533        let mut is_exclusively_accessed_input = Vec::with_capacity(shared_input_objects.len());
534        // Record receiving object versions towards the shared version computation.
535        let receiving_object_keys = assignable.receiving_object_keys();
536        input_object_keys.extend(receiving_object_keys);
537
538        if txn_cancelled {
539            // For cancelled transaction due to congestion, assign special versions to all shared objects.
540            // Note that new lamport version does not depend on any shared objects.
541            for SharedInputObject {
542                id,
543                initial_shared_version,
544                ..
545            } in shared_input_objects.iter()
546            {
547                let assigned_version = match cancellation_info {
548                    Some(CancelConsensusCertificateReason::CongestionOnObjects(_)) => {
549                        if congested_objects_info
550                            .as_ref()
551                            .is_some_and(|info| info.contains(id))
552                        {
553                            SequenceNumber::CONGESTED
554                        } else {
555                            SequenceNumber::CANCELLED_READ
556                        }
557                    }
558                    Some(CancelConsensusCertificateReason::DkgFailed) => {
559                        if id == &SUI_RANDOMNESS_STATE_OBJECT_ID {
560                            SequenceNumber::RANDOMNESS_UNAVAILABLE
561                        } else {
562                            SequenceNumber::CANCELLED_READ
563                        }
564                    }
565                    None => unreachable!("cancelled transaction should have cancellation info"),
566                };
567                assigned_versions.push(((*id, *initial_shared_version), assigned_version));
568                is_exclusively_accessed_input.push(false);
569            }
570        } else {
571            for (
572                SharedInputObject {
573                    id,
574                    initial_shared_version,
575                    mutability,
576                },
577                assigned_version,
578            ) in shared_input_objects.iter().map(|obj| {
579                (
580                    obj,
581                    *shared_input_next_versions
582                        .get(&obj.id_and_version())
583                        .unwrap(),
584                )
585            }) {
586                assigned_versions.push(((*id, *initial_shared_version), assigned_version));
587                input_object_keys.push(ObjectKey(*id, assigned_version));
588                is_exclusively_accessed_input.push(mutability.is_exclusive());
589            }
590        }
591
592        let next_version =
593            SequenceNumber::lamport_increment(input_object_keys.iter().map(|obj| obj.1));
594        assert!(
595            next_version.is_valid(),
596            "Assigned version must be valid. Got {:?}",
597            next_version
598        );
599
600        if !txn_cancelled {
601            // Update the next version for the shared objects.
602            assigned_versions
603                .iter()
604                .zip_debug_eq(is_exclusively_accessed_input)
605                .filter_map(|((id, _), mutable)| {
606                    if mutable {
607                        Some((*id, next_version))
608                    } else {
609                        None
610                    }
611                })
612                .for_each(|(id, version)| {
613                    assert!(
614                        version.is_valid(),
615                        "Assigned version must be a valid version."
616                    );
617                    shared_input_next_versions
618                        .insert(id, version)
619                        .expect("Object must exist in shared_input_next_versions.");
620                });
621        }
622
623        trace!(
624            ?tx_key,
625            ?assigned_versions,
626            ?next_version,
627            ?txn_cancelled,
628            "locking shared objects"
629        );
630
631        AssignedVersions::new(assigned_versions, system_object_versions)
632    }

635fn implicitly_read_system_objects(
636    epoch_store: &AuthorityPerEpochStore,
637) -> Vec<ConsensusObjectSequenceKey> {
638    let epoch_start_config = epoch_store.epoch_start_config();
639    let mut objects = Vec::new();
640    if epoch_store.accumulators_enabled() {
641        objects.push((
642            SUI_ACCUMULATOR_ROOT_OBJECT_ID,
643            epoch_start_config
644                .accumulator_root_obj_initial_shared_version()
645                .expect("accumulator root obj initial shared version should be set"),
646        ));
647    }
648    if epoch_store.protocol_config().enable_forwarding_addresses()
649        && let Some(initial_shared_version) =
650            epoch_start_config.forwarding_address_registry_obj_initial_shared_version()

660fn get_or_init_versions<'a>(
661    shared_input_objects: impl Iterator<Item = SharedInputObject> + 'a,
662    epoch_store: &AuthorityPerEpochStore,
663    cache_reader: &dyn ObjectCacheRead,
664) -> SuiResult<HashMap<ConsensusObjectSequenceKey, SequenceNumber>> {
665    let mut shared_input_objects: Vec<_> = shared_input_objects
666        .map(|so| so.into_id_and_version())
667        .collect();

669    shared_input_objects.extend(implicitly_read_system_objects(epoch_store));
670
671    shared_input_objects.sort();
672    shared_input_objects.dedup();
673
674    epoch_store.get_or_init_next_object_versions(&shared_input_objects, cache_reader)
675}

1414    /// Generate a transaction that uses shared objects as specified in the parameters.
1415    /// Also uses a gas object with specified version.
1416    /// The version of the gas object is used to manipulate the lamport version of this transaction.
1417    fn generate_shared_objs_tx_with_gas_version(
1418        shared_objects: &[(ObjectID, SequenceNumber, bool)],
1419        gas_object_version: u64,
1420    ) -> VerifiedExecutableTransaction {
1421        let mut tx_builder = TestTransactionBuilder::new(
1422            SuiAddress::ZERO,
1423            (
1424                ObjectID::random(),
1425                SequenceNumber::from_u64(gas_object_version),
1426                ObjectDigest::random(),
1427            ),
1428            0,
1429        );
1430        let tx_data = {
1431            let builder = tx_builder.ptb_builder_mut();
1432            for (shared_object_id, shared_object_init_version, shared_object_mutable) in
1433                shared_objects
1434            {
1435                builder
1436                    .obj(ObjectArg::SharedObject {
1437                        id: *shared_object_id,
1438                        initial_shared_version: *shared_object_init_version,
1439                        mutability: if *shared_object_mutable {
1440                            SharedObjectMutability::Mutable
1441                        } else {
1442                            SharedObjectMutability::Immutable
1443                        },
1444                    })
1445                    .unwrap();
1446            }
1447            tx_builder.build()
1448        };
1449        let tx = SenderSignedData::new(tx_data, vec![]);
1450        VerifiedExecutableTransaction::new_unchecked(ExecutableTransaction::new_from_data_and_sig(
1451            tx,
1452            CertificateProof::new_system(0),
1453        ))
1454    }
