4//! Publishes the checkpoints a validator builds as soon as it builds them, before the network
5//! certifies them, and the ones it executes without having built them first.

25/// Checkpoints that can wait to be loaded and published. Neither the checkpoint builder nor the
26/// executor waits for the publisher: a checkpoint that does not fit is dropped, which
27/// subscribers see as a gap.
28const MAX_PENDING_CHECKPOINTS: usize = 128;
29
30/// Hands the summary of every checkpoint this validator builds or executes to a background task,
31/// which loads the checkpoint's contents and full data and broadcasts it to subscribers. A
32/// checkpoint reaches the task twice, once when built and once when executed after
33/// certification; it is published the first time. Clones share the task, which stops once every
34/// clone is dropped.
35#[derive(Clone)]
36pub struct LocallyBuiltCheckpointPublisher {
37    sender: mpsc::Sender<CheckpointSummary>,
38}
39
40impl LocallyBuiltCheckpointPublisher {
41    /// Published checkpoints have strictly increasing sequence numbers. Checkpoints up to
42    /// `highest_executed_checkpoint`, the highest checkpoint this node had executed when it
43    /// started, are never published: after a restart the builder can rebuild checkpoints this
44    /// node already executed, and their objects may already be pruned. Checkpoints rebuilt above
45    /// it are published, so a client that connects right after a restart can receive checkpoints
46    /// that were already streamed before it.
47    pub fn spawn(
48        state: Arc<AuthorityState>,
49        highest_executed_checkpoint: Option<CheckpointSequenceNumber>,
50        subscribers: broadcast::Sender<Arc<LocallyBuiltCheckpoint>>,
51    ) -> Self {
52        let (sender, receiver) = mpsc::channel(MAX_PENDING_CHECKPOINTS);
53        spawn_monitored_task!(publish_checkpoints(
54            receiver,
55            state,
56            highest_executed_checkpoint,
57            subscribers,
58        ));
59        Self { sender }
60    }
61
62    /// Wraps `output` so that every checkpoint it handles is also published.
63    pub fn wrap_output(&self, output: Box<dyn CheckpointOutput>) -> Box<dyn CheckpointOutput> {
64        Box::new(PublishLocallyBuiltCheckpoint {
65            output,
66            publisher: self.clone(),
67        })
68    }
69
70    pub(crate) fn publish(&self, summary: &CheckpointSummary) {
71        match self.sender.try_send(summary.clone()) {
72            Ok(()) => {}
73            Err(mpsc::error::TrySendError::Full(_)) => warn!(
74                checkpoint = summary.sequence_number,
75                "locally built checkpoint publisher is backed up; dropping checkpoint"
76            ),
77            Err(mpsc::error::TrySendError::Closed(_)) => warn!(
78                checkpoint = summary.sequence_number,
79                "locally built checkpoint publisher stopped; dropping checkpoint"
80            ),
81        }
82    }
83}
84
85struct PublishLocallyBuiltCheckpoint {
86    output: Box<dyn CheckpointOutput>,
87    publisher: LocallyBuiltCheckpointPublisher,
88}
89
90#[async_trait]
91impl CheckpointOutput for PublishLocallyBuiltCheckpoint {
92    async fn checkpoint_created(
93        &self,
94        summary: &CheckpointSummary,
95        contents: &CheckpointContents,
96        epoch_store: &Arc<AuthorityPerEpochStore>,
97        checkpoint_store: &Arc<CheckpointStore>,
98    ) -> SuiResult {
99        // The wrapped output runs first, so a validator submits its checkpoint signature before
100        // spending anything on publishing.
101        self.output
102            .checkpoint_created(summary, contents, epoch_store, checkpoint_store)
103            .await?;
104        self.publisher.publish(summary);
105        Ok(())
106    }
107}
108
109async fn publish_checkpoints(
110    mut receiver: mpsc::Receiver<CheckpointSummary>,
111    state: Arc<AuthorityState>,
112    mut last_published: Option<CheckpointSequenceNumber>,
113    subscribers: broadcast::Sender<Arc<LocallyBuiltCheckpoint>>,
114) {
115    while let Some(summary) = receiver.recv().await {
116        let sequence_number = summary.sequence_number;
117        if last_published.is_some_and(|last| sequence_number <= last) {
118            continue;
119        }
120        // Set before loading, so a checkpoint that fails to load is skipped rather than retried.
121        last_published = Some(sequence_number);
122
123        match load_locally_built_checkpoint(&state, summary).await {
124            Ok(checkpoint) => {
125                // A send error only means that the subscription service has stopped.
126                let _ = subscribers.send(Arc::new(checkpoint));
127            }
128            Err(error) => warn!(
129                checkpoint = sequence_number,
130                ?error,
131                "failed to load locally built checkpoint; subscribers will see a gap"
132            ),
133        }
134    }
135}

137/// Waits until every transaction in the checkpoint has executed, then reads its contents,
138/// transactions, effects, events, and objects. The builder includes the epoch's advance-epoch
139/// transaction without executing it; the checkpoint executor executes it after the network
140/// certifies the epoch's last checkpoint. Every other transaction has executed by the time its
141/// checkpoint is built, so this only waits on the last checkpoint of an epoch, which is therefore
142/// published after certification.
143async fn load_locally_built_checkpoint(
144    state: &AuthorityState,
145    summary: CheckpointSummary,
146) -> SuiResult<LocallyBuiltCheckpoint> {
147    let contents: CheckpointContents = state.get_checkpoint_contents(summary.content_digest)?;
148    let tx_digests: Vec<_> = contents.iter().map(|digests| digests.transaction).collect();
149
150    let cache = state.get_transaction_cache_reader();
151    let effects = cache
152        .notify_read_executed_effects_may_fail(
153            "LocallyBuiltCheckpointPublisher::load_checkpoint",
154            &tx_digests,
155        )
156        .await?;
157    let transactions = cache
158        .multi_get_transaction_blocks(&tx_digests)
159        .into_iter()
160        .zip_debug_eq(&tx_digests)
161        .map(|(transaction, digest)| {
162            transaction.ok_or_else(|| SuiErrorKind::TransactionNotFound { digest: *digest }.into())
163        })
164        .collect::<SuiResult<Vec<_>>>()?;
165
166    let (transactions, object_set) = load_executed_transactions(
167        transactions.iter().map(|transaction| transaction.data()),
168        &effects,
169        state.get_object_store().as_ref(),
170        cache.as_ref(),
171    )?;
172
173    Ok(LocallyBuiltCheckpoint {
174        summary,
175        contents,
176        transactions,
177        object_set,
178    })
