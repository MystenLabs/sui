4//! Causal-order bookkeeping for execution scheduling.
5//!
6//! Every unit the ExecutionScheduler enqueues is assigned a *causal index* - its
7//! position in enqueue order. Both sources (consensus handler, checkpoint executor)
8//! enqueue in causal order, so anything a transaction can wait for during execution
9//! is produced by a unit with a lower index. The execution driver admits
10//! transactions by index (see `execution_driver.rs`) and retires them - on completing
11//! execution, or on dropping a transaction it does not need to run (stale, or already
12//! executed). Every index retires in the driver.
13//!
14//! Enqueues are deduplicated by accumulator root version: all transactions of a
15//! version are enqueued together, so a unit at or below the enqueue watermark was
16//! already enqueued in full by the other source. [`CausalAdmission::dedup_and_assign`]
17//! filters, indexes, and bumps the watermark for a whole batch under one lock,
18//! serializing version admission across the two sources; combined with every
19//! committed transaction executing before its epoch closes, this guarantees every
20//! assigned index reaches the driver. This design depends on accumulator versions
21//! being present, which holds because accumulators cannot be disabled once enabled.
22//! Units without a version (pre-accumulator epochs replayed from checkpoints, where
23//! nothing blocks, and the end-of-epoch transaction) bypass deduplication.
24//!
25//! A unit that already-indexed units may wait on must never be re-assigned a new,
26//! higher index (its blocked waiters would pin the concurrency limit while the
27//! causal-next lane can never reach it). Hence the funds-withdraw retry keeps its
28//! original index, and settlement transactions - which materialize long after their
29//! waiters enqueue - are index-less: never blocked by admission, though they occupy
30//! a concurrency slot when one is free.

44/// Shared state between the ExecutionScheduler (index assignment and enqueue
45/// deduplication) and the execution driver (admission and retirement). See the module
46/// comment and `execution_driver.rs`.
47pub struct CausalAdmission {
48    inner: Mutex<AdmissionInner>,
49    /// Wakes the driver loop when the watermark or in-flight count changes.
50    notify: Notify,
51    /// Max transactions admitted for execution concurrently via the capacity branch (K).
52    concurrency_limit: usize,
53    metrics: Option<CausalAdmissionMetrics>,
54}

56pub struct CausalAdmissionMetrics {
57    watermark: IntGauge,
58    next_index: IntGauge,
59    in_flight: IntGauge,
60    done_above: IntGauge,
61    next_admitted: IntGauge,
62}

64impl CausalAdmissionMetrics {

77                registry,
78            )
79            .unwrap(),
80            in_flight: register_int_gauge_with_registry!(
81                "causal_admission_in_flight",

83                registry,
84            )
85            .unwrap(),
86            done_above: register_int_gauge_with_registry!(
87                "causal_admission_done_above",
88                "Indices done above the watermark, waiting for the gap below them to fill",
89                registry,
90            )
91            .unwrap(),
92            next_admitted: register_int_gauge_with_registry!(
93                "causal_admission_next_admitted",
94                "1 while a causal-next admission over the concurrency limit is outstanding",
95                registry,

97            .unwrap(),
98        }
99    }

102struct AdmissionInner {
103    /// The next index to assign. Indices start at 1; watermark 0 means "nothing done".
104    next_index: u64,
105    /// The highest accumulator root version whose group has been fully enqueued.
106    /// Groups at or below it are duplicates.
107    enqueue_watermark: Option<SequenceNumber>,
108    /// The watermark C: every index <= C is done (executed or retired).
109    watermark: u64,
110    /// Done indices > C, awaiting the gap below them to fill.
111    done_above: BTreeSet<u64>,
112    /// Admitted (submitted for execution) transactions not yet finished. Includes
113    /// transactions blocked inside execution.
114    in_flight: usize,
115    /// Whether a causal-next admission (over the concurrency limit) is outstanding.
116    /// At most one at a time, bounding in-flight at concurrency_limit + 1.
117    next_admitted: bool,
118}
119
120impl CausalAdmission {
121    /// Default sizing: execution parallelism K at half the CPUs, leaving the rest for
122    /// consensus, networking and checkpointing. Test configurations use a fixed limit
123    /// so the host's CPU count does not influence simulation behavior. The
124    /// `execution-concurrency-limit` fail point overrides it, letting selected tests
125    /// explore admission interleavings, including limits small enough to force
126    /// transactions through the causal-next lane.
127    pub fn new_with_default_sizing(registry: &Registry) -> Arc<Self> {
128        #[allow(unused_mut)]
129        let mut concurrency_limit = if mysten_common::in_test_configuration() {
130            TEST_CONCURRENCY_LIMIT
131        } else {
132            std::cmp::max(1, num_cpus::get() / 2)
133        };
134        sui_macros::fail_point_arg!("execution-concurrency-limit", |limit: usize| {
135            concurrency_limit = limit;
136        });
137        tracing::info!("execution concurrency limit: {concurrency_limit}");
138        Self::build(
139            concurrency_limit,
140            Some(CausalAdmissionMetrics::new(registry)),
141        )
142    }
143
144    pub fn new(concurrency_limit: usize) -> Arc<Self> {
145        Self::build(concurrency_limit, None)
146    }
147
148    fn build(concurrency_limit: usize, metrics: Option<CausalAdmissionMetrics>) -> Arc<Self> {
149        assert!(concurrency_limit > 0);
150        let admission = Arc::new(Self {
151            inner: Mutex::new(AdmissionInner {
152                next_index: 1,
153                enqueue_watermark: None,
154                watermark: 0,
155                done_above: BTreeSet::new(),
156                in_flight: 0,
157                next_admitted: false,
158            }),
159            notify: Notify::new(),
160            concurrency_limit,
161            metrics,
162        });
163        admission.publish(&admission.inner.lock());
164        admission
165    }
166
167    fn publish(&self, inner: &AdmissionInner) {
168        if let Some(metrics) = &self.metrics {
169            metrics.watermark.set(inner.watermark as i64);
170            metrics.next_index.set(inner.next_index as i64);
171            metrics.in_flight.set(inner.in_flight as i64);
172            metrics.done_above.set(inner.done_above.len() as i64);
173            metrics.next_admitted.set(inner.next_admitted as i64);
174        }
175    }
176
177    /// Deduplicates an enqueue batch and assigns causal indices to the survivors, in
178    /// batch order, atomically: units whose assigned accumulator root version is at
179    /// or below the enqueue watermark were already enqueued in full by the other
180    /// source and are dropped; the watermark then rises to the batch's highest
181    /// version. Units without an accumulator version are never deduplicated.
182    ///
183    /// Processing the whole batch under one lock serializes version admission across
184    /// the consensus and checkpoint paths.
185    pub fn dedup_and_assign<T>(&self, certs: Vec<(T, ExecutionEnv)>) -> Vec<(T, ExecutionEnv)> {
186        // Internal re-submissions of already-indexed certificates must not pass
187        // through here again - they keep their original index.
188        debug_assert!(certs.iter().all(|(_, env)| env.causal_index.is_none()));
189        let mut inner = self.inner.lock();
190        let watermark = inner.enqueue_watermark;
191        let mut max_version = watermark;
192        let certs = certs
193            .into_iter()
194            .filter_map(|(item, mut env)| {
195                if let Some(version) = env.assigned_versions.accumulator_version() {
196                    if watermark.is_some_and(|w| version <= w) {
197                        assert_reachable!("rejected duplicate enqueue of a version group");
198                        return None;
199                    }
200                    max_version = max_version.max(Some(version));
201                }
202                env.causal_index = Some(inner.next_index);
203                inner.next_index += 1;
204                Some((item, env))
205            })
206            .collect();
207        inner.enqueue_watermark = max_version;
208        self.publish(&inner);
209        certs
210    }
211
212    /// Attempts to admit the transaction with `index` for execution. On success,
213    /// returns a slot that must be held for the duration of execution; dropping it
214    /// releases the concurrency slot and retires the index.
215    ///
216    /// Admission policy: under the concurrency limit, anything goes. At the limit, the
217    /// next transaction in causal order is still admitted, one at a time - it can
218    /// never block (everything below it is done), which is what guarantees progress no
219    /// matter how many admitted transactions are parked. In-flight is thereby bounded
220    /// at concurrency_limit + 1.
221    pub fn try_admit(self: &Arc<Self>, index: u64) -> Option<InFlightSlot> {
222        let mut inner = self.inner.lock();
223        debug_assert!(index > inner.watermark, "admitting an already-done index");
224        let is_next = if inner.in_flight < self.concurrency_limit {
225            assert_reachable!("admitted under the concurrency limit");
226            false
227        } else if index == inner.watermark + 1 && !inner.next_admitted {
228            inner.next_admitted = true;
229            true
230        } else {
231            return None;
232        };
233        assert_sometimes!(is_next, "admitted via the causal-next lane over the limit");
234        inner.in_flight += 1;
235        self.publish(&inner);
236        Some(InFlightSlot {
237            admission: self.clone(),
238            is_next,
239            retire_on_drop: Some(index),
240        })
241    }
242
243    /// Takes a concurrency slot if one is free, without any causal-index conditions.
244    /// For settlement transactions: they must never be blocked by admission
245    /// (transactions may be parked waiting on the very versions they write), but they
246    /// respect the concurrency limit when there is room. The returned slot releases on
247    /// drop and retires nothing.
248    pub fn try_take_slot(self: &Arc<Self>) -> Option<InFlightSlot> {
249        let mut inner = self.inner.lock();
250        if inner.in_flight >= self.concurrency_limit {
251            return None;
252        }
253        inner.in_flight += 1;
254        self.publish(&inner);
255        Some(InFlightSlot {
256            admission: self.clone(),
257            is_next: false,
258            retire_on_drop: None,
259        })
260    }
261
262    /// Waits until the watermark or in-flight count may have changed. A single permit
263    /// is buffered, so a change occurring before this call is not missed.
264    pub async fn changed(&self) {
265        self.notify.notified().await;
266    }
267
268    #[cfg(test)]
269    pub fn mark_done(&self, index: u64) {
270        let mut inner = self.inner.lock();
271        Self::mark_done_locked(&mut inner, index);
272        self.publish(&inner);
273        drop(inner);
274        self.notify.notify_one();
275    }
276
277    fn mark_done_locked(inner: &mut AdmissionInner, index: u64) {
278        if index == inner.watermark + 1 {
279            inner.watermark = index;
280            loop {
281                let next = inner.watermark + 1;
282                if !inner.done_above.remove(&next) {
283                    break;
284                }
285                inner.watermark = next;
286            }
287        } else {
288            debug_assert!(index > inner.watermark, "index done twice");
289            inner.done_above.insert(index);
290        }
291    }
292
293    /// Whether every assigned index is retired and nothing is in flight.
294    pub fn is_quiescent(&self) -> bool {
295        Self::is_quiescent_locked(&self.inner.lock())
296    }
297
298    fn is_quiescent_locked(inner: &AdmissionInner) -> bool {
299        inner.in_flight == 0
300            && !inner.next_admitted
301            && inner.done_above.is_empty()
302            && inner.watermark + 1 == inner.next_index
303    }
304
305    /// Checks quiescence at an epoch boundary: every unit enqueued in an epoch executes
306    /// or is dropped before the epoch closes, and the next epoch has not enqueued yet.
307    /// A violation means an index leaked, which silently disables the causal-next
308    /// lane for the rest of the process since the watermark can never pass it. State
309    /// is deliberately not reset: indices are process-global, and a straggling slot
310    /// retiring into reset bookkeeping would corrupt the watermark.
311    ///
312    /// Must run after `epoch_terminated`: the driver drops a slot only after its
313    /// transaction's effects are visible, so an earlier check can see slots that are
314    /// about to retire.
315    pub fn check_quiescent_at_epoch_boundary(&self) {
316        let inner = self.inner.lock();
317        if !Self::is_quiescent_locked(&inner) {
318            debug_fatal!(
319                "causal admission not quiescent at epoch boundary: watermark={} next_index={} \
320                 in_flight={} next_admitted={} done_above={:?}",
321                inner.watermark,
322                inner.next_index,
323                inner.in_flight,
324                inner.next_admitted,
325                inner.done_above
326            );
327        }
328    }
329
330    #[cfg(test)]
331    pub fn watermark_for_testing(&self) -> u64 {
332        self.inner.lock().watermark
333    }
334}

336/// A held execution-concurrency slot. Dropping it releases the slot and retires the
337/// admitted index - one lock and one driver wakeup for the whole completion.
338pub struct InFlightSlot {
339    admission: Arc<CausalAdmission>,
340    is_next: bool,
341    retire_on_drop: Option<u64>,
342}
343
344impl InFlightSlot {
345    /// Keeps the admitted causal index alive past this execution attempt, for a
346    /// transaction that will be re-submitted under the same index (RetryLater).
347    pub fn skip_retire(&mut self) {
348        self.retire_on_drop = None;
349    }
350}
351
352impl Drop for InFlightSlot {
353    fn drop(&mut self) {
354        let mut inner = self.admission.inner.lock();
355        inner.in_flight -= 1;
356        if self.is_next {
357            inner.next_admitted = false;
358        }
359        if let Some(index) = self.retire_on_drop {
360            CausalAdmission::mark_done_locked(&mut inner, index);
361        }
362        self.admission.publish(&inner);
363        drop(inner);
364        self.admission.notify.notify_one();
365    }
366}

380    /// Enqueues `n` version-less units and returns their assigned indices.
381    fn assign(admission: &CausalAdmission, n: usize) -> Vec<u64> {
382        admission
383            .dedup_and_assign((0..n).map(|_| ((), env(None))).collect())
384            .into_iter()
385            .map(|(_, env)| env.causal_index.unwrap())
386            .collect()
387    }

389    #[test]
390    fn watermark_advances_over_done_indices() {
391        let admission = CausalAdmission::new(2);
392        assert_eq!(assign(&admission, 3), vec![1, 2, 3]);
393        assert_eq!(admission.watermark_for_testing(), 0);
394
395        // Out-of-order completion parks above the watermark until the gap fills.
396        admission.mark_done(2);
397        assert_eq!(admission.watermark_for_testing(), 0);
398        admission.mark_done(1);
399        assert_eq!(admission.watermark_for_testing(), 2);
400        admission.mark_done(3);
401        assert_eq!(admission.watermark_for_testing(), 3);
402    }

404    #[test]
405    fn duplicate_version_groups_are_rejected() {
406        let admission = CausalAdmission::new(2);
407
408        // A batch with versions 5 and 6 is admitted in full.
409        let batch = vec![((), env(Some(5))), ((), env(Some(5))), ((), env(Some(6)))];
410        assert_eq!(admission.dedup_and_assign(batch).len(), 3);
411        // Versions at or below the watermark are duplicates from the other source.
412        let batch = vec![((), env(Some(5))), ((), env(Some(6)))];
413        assert!(admission.dedup_and_assign(batch).is_empty());
414        // A higher version is new work, and version-less units always pass.
415        let batch = vec![((), env(Some(7))), ((), env(None))];
416        let admitted = admission.dedup_and_assign(batch);
417        assert_eq!(
418            admitted
419                .into_iter()
420                .map(|(_, env)| env.causal_index.unwrap())
421                .collect::<Vec<_>>(),
422            vec![4, 5]
423        );
424    }

426    #[test]
427    fn next_consecutive_is_admitted_at_the_limit() {
428        let admission = CausalAdmission::new(1);
429        assign(&admission, 2);
430
431        // Fill the single concurrency slot with index 2.
432        let mut slot2 = admission.try_admit(2).unwrap();
433        // The capacity branch is closed, but index 1 == watermark+1 is still admitted.
434        let mut slot1 = admission.try_admit(1).unwrap();
435        // Only one causal-next admission may be outstanding at a time.
436        assert!(admission.try_admit(1).is_none());
437        // A RetryLater release keeps the index alive, and it can be admitted again.
438        slot1.skip_retire();
439        drop(slot1);
440        let slot1b = admission.try_admit(1).unwrap();
441
442        // Completing index 1 retires it via the slot drop.
443        drop(slot1b);
444        assert_eq!(admission.watermark_for_testing(), 1);
445        // Index 2 is now watermark+1; re-admitting it bypasses the (full) limit.
446        slot2.skip_retire();
447        drop(slot2);
448        let slot2b = admission.try_admit(2).unwrap();
449        drop(slot2b);
450        assert_eq!(admission.watermark_for_testing(), 2);
451    }

453    #[test]
454    fn concurrency_limit_bounds_admissions() {
455        let admission = CausalAdmission::new(2);
456        assign(&admission, 5);
457
458        // Admit indices 2 and 3 through the capacity branch, filling the limit.
459        let _s2 = admission.try_admit(2).unwrap();
460        let _s3 = admission.try_admit(3).unwrap();
461        // Index 4 is neither under the limit nor the causal-next transaction.
462        assert!(admission.try_admit(4).is_none());
463
464        // Releasing a slot reopens the capacity branch, regardless of how far the
465        // index runs ahead of the watermark.
466        drop(_s2);
467        assert!(admission.try_admit(5).is_some());
468    }

481    #[test]
482    fn quiescence_requires_every_index_retired_and_nothing_in_flight() {
483        let admission = CausalAdmission::new(2);
484        assert!(admission.is_quiescent());
485        assign(&admission, 3);
486        assert!(!admission.is_quiescent());
487
488        let slot1 = admission.try_admit(1).unwrap();
489        admission.mark_done(2);
490        admission.mark_done(3);
491        // Indices 2 and 3 are done above the watermark; 1 is still in flight.
492        assert!(!admission.is_quiescent());
493        drop(slot1);
494        assert!(admission.is_quiescent());
495
496        // A settlement slot holds no index but still counts as in flight.
497        let settle = admission.try_take_slot().unwrap();
498        assert!(!admission.is_quiescent());
499        drop(settle);
500        assert!(admission.is_quiescent());
501
502        // A retry that kept its index alive (skip_retire) leaves it unretired.
503        assign(&admission, 1);
504        let mut slot4 = admission.try_admit(4).unwrap();
505        slot4.skip_retire();
506        drop(slot4);
507        assert!(!admission.is_quiescent());
508        admission.mark_done(4);
509        assert!(admission.is_quiescent());
510    }
