35pub(crate) struct EagerFundsWithdrawScheduler {
36    funds_read: Arc<dyn AccountFundsRead>,
37    inner_state: Arc<Mutex<InnerState>>,
38}
39
40struct InnerState {
41    /// For each address funds account that we have seen withdraws through `schedule_withdraws`,
42    /// we track the current state of that account, and only remove it from the map after
43    /// we have settled all withdraws for that account.
44    tracked_accounts: BTreeMap<AccumulatorObjId, AccountState>,
45    /// Tracks all the address funds accounts that have a withdraw transaction tracked,
46    /// mapping from the accumulator version that the withdraw transaction reads from.
47    /// If a withdraw transaction needs to withdraw from account O at version V,
48    /// we must process and settle that withdraw transaction whenever we settle all transactions
49    /// scheduled for version V.
50    pending_settlements: BTreeMap<SequenceNumber, BTreeSet<AccumulatorObjId>>,
51    /// The current version of the accumulator object known to the scheduler.
52    accumulator_version: SequenceNumber,
53}
54
55impl EagerFundsWithdrawScheduler {
56    pub fn new(
57        funds_read: Arc<dyn AccountFundsRead>,
58        starting_accumulator_version: SequenceNumber,
59    ) -> Arc<Self> {
60        Arc::new(Self {
61            funds_read,
62            inner_state: Arc::new(Mutex::new(InnerState {
63                tracked_accounts: BTreeMap::new(),
64                pending_settlements: BTreeMap::new(),
65                accumulator_version: starting_accumulator_version,
66            })),
67        })
68    }
69
70    /// Reserves `reservations` against each account's tracked state, starting to track
71    /// accounts from `init_balances`, and records the version for settlement.
72    fn reserve_withdraws(
73        inner_state: &mut InnerState,
74        reservations: WithdrawReservations,
75        init_balances: &BTreeMap<AccumulatorObjId, (u128, SequenceNumber)>,
76        cur_accumulator_version: SequenceNumber,
77        all_accounts: BTreeSet<AccumulatorObjId>,
78    ) -> Vec<(Arc<PendingWithdraw>, oneshot::Receiver<ScheduleStatus>)> {
79        let mut pending = Vec::with_capacity(reservations.withdraws.len());
80        for withdraw in reservations.withdraws {
81            let (sender, receiver) = oneshot::channel();
82            let accounts = withdraw.reservations.keys().cloned().collect::<Vec<_>>();
83            let pending_withdraw =
84                PendingWithdraw::new(reservations.accumulator_version, withdraw, sender);
85            for account_id in accounts {
86                let entry = inner_state
87                    .tracked_accounts
88                    .entry(account_id)
89                    .or_insert_with(|| {
90                        let (balance, version) = init_balances.get(&account_id).cloned().unwrap();
91                        AccountState::new(account_id, balance, version)
92                    });
93                entry.try_reserve_new_withdraw(pending_withdraw.clone(), cur_accumulator_version);
94            }
95            pending.push((pending_withdraw, receiver));
96        }
97        let old = inner_state
98            .pending_settlements
99            .insert(reservations.accumulator_version, all_accounts);
100        assert!(old.is_none());
101        pending
102    }
103}

105#[async_trait::async_trait]
106impl FundsWithdrawSchedulerTrait for EagerFundsWithdrawScheduler {
107    fn schedule_withdraws(
108        &self,
109        mut reservations: WithdrawReservations,
110    ) -> BTreeMap<TransactionDigest, ScheduleResult> {
111        let mut inner_state = self.inner_state.lock();
112        let cur_accumulator_version = inner_state.accumulator_version;
113        if reservations.accumulator_version < cur_accumulator_version {
114            // This accumulator version is already settled.
115            // There is no need to schedule the withdraws.
116            return reservations.notify_skip_schedule();
117        }
118        let mut all_accounts = reservations.all_accounts();
119        let mut init_balances = BTreeMap::new();
120        let mut settled_in_storage = false;
121        for account_id in &all_accounts {
122            if inner_state.tracked_accounts.contains_key(account_id) {
123                continue;
124            }
125            // TODO: We can warm up the cache prior to holding the lock.
126            let (balance, version) = self
127                .funds_read
128                .get_consistent_latest_account_amount_and_version(account_id);
129            if version > reservations.accumulator_version {
130                settled_in_storage = true;
131                break;
132            }
133            init_balances.insert(*account_id, (balance, version));
134        }
135
136        if settled_in_storage {
137            let skip_results = reservations.notify_skip_schedule();
138            // Storage has settled this batch, but this scheduler's `settle_funds` for it is
139            // still to come, so tracked accounts must hold the batch's reservations until then.
140            // Untracked accounts stay untracked; a later read of them includes this batch.
141            let tracked = &inner_state.tracked_accounts;
142            reservations.withdraws.retain_mut(|withdraw| {
143                withdraw
144                    .reservations
145                    .retain(|account_id, _| tracked.contains_key(account_id));
146                !withdraw.reservations.is_empty()
147            });
148            all_accounts.retain(|account_id| tracked.contains_key(account_id));
149            if !all_accounts.is_empty() {
150                assert_reachable!("storage-settled batch keeps tracked-account reservations");
151                Self::reserve_withdraws(
152                    &mut inner_state,
153                    reservations,
154                    &BTreeMap::new(),
155                    cur_accumulator_version,
156                    all_accounts,
157                );
158            }
159            return skip_results;
160        }
161
162        Self::reserve_withdraws(
163            &mut inner_state,
164            reservations,
165            &init_balances,
166            cur_accumulator_version,
167            all_accounts,
168        )
169        .into_iter()
170        .map(|(pending_withdraw, mut receiver)| {
171            let result = if pending_withdraw.is_scheduled() {
172                ScheduleResult::ScheduleResult(receiver.try_recv().unwrap())
173            } else {
174                ScheduleResult::Pending(receiver)
175            };
176            (pending_withdraw.tx_digest(), result)
177        })
178        .collect()
179    }
180
181    fn settle_funds(&self, settlement: FundsSettlement) {
182        let next_accumulator_version = settlement.next_accumulator_version;
183        let mut inner_state = self.inner_state.lock();
184        if next_accumulator_version <= inner_state.accumulator_version {
185            // This accumulator version is already settled.
186            // There is no need to settle the funds.
187            debug!(
188                next_accumulator_version =? next_accumulator_version.value(),
189                "Skipping settlement since it is already settled",
190            );
191            return;
192        }
193        assert_eq!(
194            next_accumulator_version,
195            inner_state.accumulator_version.next()
196        );
197        let cleanup_version = inner_state.accumulator_version;
198        inner_state.accumulator_version = next_accumulator_version;
199        debug!(
200            "Bumping accumulator version from {:?} to {:?}",
201            cleanup_version.value(),
202            next_accumulator_version.value(),
203        );
204
205        // A settlement bumps the accumulator version from `cleanup_version` to `next_accumulator_version`.
206        // We must process the following types of accounts:
207        // 1. Accounts that had withdraws scheduled at version `cleanup_version`. These withdraws
208        // are now settled and we know their exact funds changes.
209        // 2. Accounts that had withdraws scheduled at version `next_accumulator_version`. These withdraws
210        // if not yet scheduled, can now be deterministically scheduled, since
211        // we have the final state before them.
212        // 3. Accounts that have funds changes through settle_funds, this can include
213        // both withdraws and deposits.
214        let mut affected_accounts = inner_state
215            .pending_settlements
216            .remove(&cleanup_version)
217            .unwrap_or_default();
218        // Since we just bumped the accumulator version, any withdraws scheduled that depend
219        // on this version can now be deterministically scheduled.
220        if let Some(current_version_accounts) = inner_state
221            .pending_settlements
222            .get(&next_accumulator_version)
223        {
224            affected_accounts.extend(current_version_accounts);
225        }
226        affected_accounts.extend(settlement.funds_changes.keys().cloned());
227
228        for object_id in affected_accounts {
229            let Some(account_state) = inner_state.tracked_accounts.get_mut(&object_id) else {
230                continue;
231            };
232            account_state.settle_funds(
233                settlement
234                    .funds_changes
235                    .get(&object_id)
236                    .cloned()
237                    .unwrap_or_default(),
238                next_accumulator_version,
239            );
240
241            if account_state.is_empty() {
242                debug!(
243                    account_id = ?object_id,
244                    "Removing account state since it is empty",
245                );
246                inner_state.tracked_accounts.remove(&object_id);
247            }
248        }
249        debug!("Settled accumulator version {:?}", next_accumulator_version);
250    }
251
252    fn close_epoch(&self) {
253        debug!("Closing epoch in EagerFundsWithdrawScheduler");
254    }
255
256    fn num_tracked_accounts(&self) -> usize {
257        self.inner_state.lock().tracked_accounts.len()
258    }
259
260    #[cfg(test)]
261    fn get_current_accumulator_version(&self) -> SequenceNumber {
262        self.inner_state.lock().accumulator_version
263    }
264}
