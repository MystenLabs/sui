15// 1. There may be times when the entire network is unstable, and in that case
16//    we may not want to punish validators when they have errors.
17// 2. Some reports are more critical than others. For example, a health check
18//    report is more critical than a submit report in terms of failures status.

20/// Maximum adjusted latency from completely unreachable (reliability = 0.0) or very slow validators.
21const MAX_LATENCY: Duration = Duration::from_secs(10);
22
23/// Complete client-observed statistics for validator interactions.
24///
25/// This struct maintains client-side metrics for all validators in the network,
26/// including latency measurements, reliability measurements, and failure tracking
27/// as observed from the client's perspective. It uses exponential moving averages (EMA)
28/// to smooth out transient spikes while still responding to sustained changes.
29#[derive(Debug, Clone)]
30pub struct ClientObservedStats {
31    /// Per-validator statistics mapping authority names to their client-observed metrics
32    pub validator_stats: HashMap<AuthorityName, ValidatorClientStats>,
33    /// Configuration parameters for scoring and exclusion policies
34    pub config: ValidatorClientMonitorConfig,
35}
36
37/// Client-observed stats for a single validator.
38///
39/// Tracks reliability, latency, and failure patterns for a specific validator
40/// as observed from the client's perspective. Uses exponential moving averages
41/// to smooth measurements while maintaining responsiveness to changes.
42#[derive(Debug, Clone)]
43pub struct ValidatorClientStats {
44    /// Moving window of success rate (0.0 to 1.0). Reliability is class-agnostic:
45    /// holds delay transactions but do not fail them.
46    pub reliability: MovingWindow<f64>,
47    /// Moving window of latencies per operation type and transaction class.
48    /// Class-neutral observations are recorded under both classes.
49    pub average_latencies: BTreeMap<(OperationType, TransactionClass), MovingWindow<Duration>>,
50    /// Size of the moving window for latency measurements
51    pub latency_moving_window_size: usize,
52}
53
54impl ValidatorClientStats {
55    pub fn new(
56        init_reliability: f64,
57        reliability_moving_window_size: usize,
58        latency_moving_window_size: usize,
59    ) -> Self {
60        Self {
61            reliability: MovingWindow::new(init_reliability, reliability_moving_window_size),
62            average_latencies: BTreeMap::new(),
63            latency_moving_window_size,
64        }
65    }
66
67    pub fn update_average_latency(
68        &mut self,
69        operation: OperationType,
70        tx_class: Option<TransactionClass>,
71        new_latency: Duration,
72    ) {
73        for class in [TransactionClass::Restricted, TransactionClass::Unrestricted] {
74            if tx_class.is_some_and(|tx_class| tx_class != class) {
75                continue;
76            }
77            match self.average_latencies.entry((operation, class)) {
78                Entry::Occupied(mut entry) => {
79                    entry.get_mut().add_value(new_latency);
80                }
81                Entry::Vacant(entry) => {
82                    entry.insert(MovingWindow::new(
83                        new_latency,
84                        self.latency_moving_window_size,
85                    ));
86                }
87            }
88        }
89    }
90}
91
92impl ClientObservedStats {
93    pub fn new(config: ValidatorClientMonitorConfig) -> Self {
94        Self {
95            validator_stats: HashMap::new(),
96            config,
97        }
98    }
99
100    /// Record client-observed interaction result with a validator.
101    ///
102    /// Updates reliability scores and latency measurements.
103    pub fn record_interaction_result(&mut self, feedback: OperationFeedback) {
104        let validator_stats = self
105            .validator_stats
106            .entry(feedback.authority_name)
107            .or_insert_with(|| {
108                ValidatorClientStats::new(
109                    1.0,
110                    self.config.reliability_moving_window_size,
111                    self.config.latency_moving_window_size,
112                )
113            });
114
115        match feedback.result {
116            Ok(latency) => {
117                validator_stats.reliability.add_value(1.0);
118                validator_stats.update_average_latency(
119                    feedback.operation,
120                    feedback.tx_class,
121                    latency,
122                );
123            }
124            Err(()) => {
125                validator_stats.reliability.add_value(0.0);
126            }
127        }
128    }
129
130    /// Get validator latencies for all validators in the committee.
131    ///
132    /// Returns a map of all tracked validators to their stats. For a validator
133    pub fn get_all_validator_stats(
134        &self,
135        committee: &Committee,
136        tx_class: TransactionClass,
137    ) -> HashMap<AuthorityName, Duration> {
138        committee
139            .names()
140            .map(|validator| {
141                let latency = self.calculate_client_latency(validator, tx_class);
142                (*validator, latency)
143            })
144            .collect()
145    }
146
147    /// Calculate adjusted latency for a single validator.
148    ///
149    /// Returns the average latency for Consensus operations with reliability penalty applied.
150    /// Lower values are better.
151    ///
152    /// Returns latency in seconds, with reliability penalty applied as a multiplier.
153    fn calculate_client_latency(
154        &self,
155        validator: &AuthorityName,
156        tx_class: TransactionClass,
157    ) -> Duration {
158        let Some(stats) = self.validator_stats.get(validator) else {
159            return MAX_LATENCY;
160        };
161
162        let operation = OperationType::SharedObjectFinality;
163        // While a validator has no observations in the requested class yet, fall back
164        // to the other class rather than treating it as unmeasured.
165        let other_class = match tx_class {
166            TransactionClass::Restricted => TransactionClass::Unrestricted,
167            TransactionClass::Unrestricted => TransactionClass::Restricted,
168        };
169        let Some(latency) = stats
170            .average_latencies
171            .get(&(operation, tx_class))
172            .or_else(|| stats.average_latencies.get(&(operation, other_class)))
173        else {
174            // No latency measurements yet, but health checks may have already told us the
175            // validator is unreachable. Rank known-unhealthy below merely-unmeasured, so that
176            // cold-start selection avoids validators we know are down.
177            let reliability_penalty = MAX_LATENCY
178                .mul_f64((1.0 - stats.reliability.get()) * self.config.reliability_weight);
179            return MAX_LATENCY + reliability_penalty;
180        };
181
182        // Get the latency for the relevant operation
183        let base_latency = latency.get();
184        let reliability = stats.reliability.get();
185        let reliability_weight = self.config.reliability_weight;
186        let penalty = MAX_LATENCY.mul_f64((1.0 - reliability) * reliability_weight);
187        (base_latency + penalty).min(MAX_LATENCY)
188    }
189
190    /// Retain only the specified validators, removing any others.
191    ///
192    /// Called periodically during health checks to clean up statistics for validators
193    /// that are no longer in the active set. This prevents memory leaks and
194    /// ensures scores are only calculated for current validators.
195    pub fn retain_validators(&mut self, current_validators: &[AuthorityName]) {
196        let cur_len = self.validator_stats.len();
197        let validator_set: HashSet<_> = current_validators.iter().collect();
198        self.validator_stats
199            .retain(|validator, _| validator_set.contains(validator));
200        let removed_count = cur_len - self.validator_stats.len();
201        if removed_count > 0 {
202            debug!("Removed {} stale validator data", removed_count);
203        }
204    }
205}
