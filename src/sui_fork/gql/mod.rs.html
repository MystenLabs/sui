24/// Signed transaction envelope paired with its execution effects and the checkpoint it was
25/// finalized in. The checkpoint is used by [`crate::store::ForkStore`] as a pre-fork guard, because
26/// remote results whose `checkpoint > forked_at_checkpoint` must not leak into a fork that has
27/// already diverged from the upstream chain.
28#[derive(Clone, Debug)]
29pub(crate) struct TransactionInfo {
30    pub(crate) transaction: VerifiedTransaction,
31    pub(crate) effects: TransactionEffects,
32    pub(crate) checkpoint: CheckpointSequenceNumber,
33}
34
35/// `TransactionRead` trait is used to retrieve transaction data for a given digest.
36pub(crate) trait TransactionRead {
37    /// Given a transaction digest, return the signed transaction, its effects, and the checkpoint
38    /// it was finalized in. Returns `None` if the transaction is not found.
39    fn transaction_data_and_effects(
40        &self,
41        tx_digest: &str,
42    ) -> Result<Option<TransactionInfo>, Error>;
43}

45/// Query for an object, specifying an `ObjectID` and the rule to retrieve it.
46#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
47pub(crate) struct ObjectKey {
48    pub(crate) object_id: ObjectID,
49    pub(crate) version_query: VersionQuery,
50}
51
52/// Version rule for an object query.
53#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
54pub(crate) enum VersionQuery {
55    /// Request the highest version of the object at or below the given version.
56    RootVersion(u64),
57
58    /// Request the object as of the given checkpoint. Useful when the version is unknown.
59    AtCheckpoint(u64),
60
61    /// Request an exact version, but only if it existed by the given checkpoint.
62    VersionAtCheckpoint { version: u64, checkpoint: u64 },
63}
64
65/// The `ObjectRead` trait is used to retrieve objects by their keys, with different query options.
66pub(crate) trait ObjectRead {
67    /// Retrieve objects by their keys, with different query options.
68    ///
69    /// If the object is not found, the element in the vector is `None`. Otherwise each tuple
70    /// contains:
71    /// - `Object`: The object data
72    /// - `u64`: The actual version of the object
73    fn get_objects(&self, keys: &[ObjectKey]) -> Result<Vec<Option<(Object, u64)>>, Error>;
74}
75
76/// Checkpoint read data.
77pub(crate) trait CheckpointRead {
78    /// Return the verified checkpoint summary together with its decoded contents. If `sequence` is
79    /// `None`, return the latest checkpoint.
80    fn get_checkpoint(
81        &self,
82        sequence: Option<CheckpointSequenceNumber>,
83    ) -> Result<Option<(VerifiedCheckpoint, CheckpointContents)>, Error>;
84}
