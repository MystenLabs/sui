49/// Signed transaction envelope paired with its execution effects and the checkpoint it was
50/// finalized in. The checkpoint is used by [`crate::store::ForkStore`] as a pre-fork guard, because
51/// remote results whose `checkpoint > forked_at_checkpoint` must not leak into a fork that has
52/// already diverged from the upstream chain.
53#[derive(Clone, Debug)]
54pub(crate) struct TransactionInfo {
55    pub(crate) transaction: VerifiedTransaction,
56    pub(crate) effects: TransactionEffects,
57    pub(crate) checkpoint: CheckpointSequenceNumber,
58}
59
60/// `TransactionRead` trait is used to retrieve transaction data for a given digest.
61pub(crate) trait TransactionRead {
62    /// Given a transaction digest, return the signed transaction, its effects, and the checkpoint
63    /// it was finalized in. Returns `None` if the transaction is not found.
64    fn transaction_data_and_effects(
65        &self,
66        tx_digest: &str,
67    ) -> Result<Option<TransactionInfo>, Error>;
68}

70/// Query for an object, specifying an `ObjectID` and the rule to retrieve it.
71#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
72pub(crate) struct ObjectKey {
73    pub(crate) object_id: ObjectID,
74    pub(crate) version_query: VersionQuery,
75}
76
77/// Version rule for an object query.
78#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
79pub(crate) enum VersionQuery {
80    /// Request the highest version of the object at or below the given version.
81    RootVersion(u64),
82
83    /// Request the object as of the given checkpoint. Useful when the version is unknown.
84    AtCheckpoint(u64),
85
86    /// Request an exact version, but only if it existed by the given checkpoint.
87    VersionAtCheckpoint { version: u64, checkpoint: u64 },
88}
89
90/// The `ObjectRead` trait is used to retrieve objects by their keys, with different query options.
91pub(crate) trait ObjectRead {
92    /// Retrieve objects by their keys, with different query options.
93    ///
94    /// If the object is not found, the element in the vector is `None`. Otherwise each tuple
95    /// contains:
96    /// - `Object`: The object data
97    /// - `u64`: The actual version of the object
98    fn get_objects(&self, keys: &[ObjectKey]) -> Result<Vec<Option<(Object, u64)>>, Error>;
99}

101/// Checkpoint read data.
102pub(crate) trait CheckpointRead {
103    /// Return the verified checkpoint summary together with its decoded contents. If `sequence` is
104    /// `None`, return the latest checkpoint.
105    fn get_checkpoint(
106        &self,
107        sequence: Option<CheckpointSequenceNumber>,
108    ) -> Result<Option<(VerifiedCheckpoint, CheckpointContents)>, Error>;
109}
