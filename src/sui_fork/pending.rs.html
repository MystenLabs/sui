4//! In-memory staging for the checkpoint currently being sealed.
5//!
6//! Simulacrum hands the store a checkpoint summary, then its transactions,
7//! effects, events, and object diffs piecemeal through the `SimulatorStore`
8//! insert methods, and finally the checkpoint contents. This buffer holds
9//! those pieces until the contents arrive, so the caller can persist the
10//! whole checkpoint into the rpc-store as one atomic batch.
11//!
12//! Object diffs are the exception to "reads never consult this buffer":
13//! execution resolves the next transaction's inputs (and the settlement and
14//! barrier system transactions sealed with the checkpoint) through the store,
15//! so the staged diffs double as a read overlay over the rpc-store until the
16//! seal commits them. Overlay versions always outrank persisted ones — local
17//! execution Lamport-bumps past every live version, and remote fetches are
18//! pinned at or below the fork checkpoint — so "overlay first, else store" is
19//! exact. Checkpoint, transaction, effects, and event entries stay write-only:
20//! sealing completes synchronously inside the checkpoint publication path, so
21//! by the time an execution returns, its rows are already in the rpc-store.
22//!
23//! Nothing here is persisted: staged entries that have not been sealed are
24//! lost on process restart, which is the crash-safety contract — a checkpoint
25//! either commits whole at seal time or leaves no trace.

56/// One fully staged transaction, cloned out of the buffer at seal time.
57pub(crate) struct StagedTransaction {
58    pub(crate) digest: TransactionDigest,
59    pub(crate) transaction: VerifiedTransaction,
60    pub(crate) effects: TransactionEffects,
61    pub(crate) events: TransactionEvents,
62}

64/// One execution result's object writes and removals, staged until seal.
65#[derive(Clone)]
66pub(crate) struct ObjectDiff {
67    pub(crate) written: BTreeMap<ObjectID, Object>,
68    pub(crate) removed: Vec<ObjectRemoval>,
69}

71/// Staged object diffs plus the merged views the read overlay answers from.
72///
73/// The views mirror the rows `LocalStore::stage_local_object_diff` will write
74/// at seal: `current` is what the checkpoint-pinned version row would say per
75/// object, `versions` is what the raw object rows would say per version. They
76/// are maintained on record rather than derived on read so overlay lookups
77/// stay O(log n) on the execution hot path.
78#[derive(Default)]
79struct PendingObjects {
80    /// Diffs in arrival order, replayed verbatim into the seal batch.
81    diffs: Vec<ObjectDiff>,
82    current: BTreeMap<ObjectID, (SequenceNumber, Status)>,
83    versions: BTreeMap<ObjectID, BTreeMap<SequenceNumber, Status>>,
84}

86/// Staging buffer for the in-flight checkpoint and its transactions.
87///
88/// `RwLock` provides the interior mutability needed behind the shared
89/// `Arc<ForkStoreInner>` that every cloned `ForkStore` holds.
90#[derive(Default)]
91pub(crate) struct PendingCheckpointBuffer {
92    checkpoint: RwLock<Option<VerifiedCheckpoint>>,
93    transactions: RwLock<BTreeMap<TransactionDigest, PendingTransaction>>,
94    objects: RwLock<PendingObjects>,
95}

102    pub(crate) fn record_checkpoint(&self, checkpoint: VerifiedCheckpoint) -> anyhow::Result<()> {
103        let mut pending = self
104            .checkpoint
105            .write()
106            .ok()
107            .context("pending checkpoint lock poisoned")?;
108        *pending = Some(checkpoint);
109        Ok(())
110    }

112    pub(crate) fn record_transaction(
113        &self,
114        transaction: VerifiedTransaction,
115    ) -> anyhow::Result<()> {
116        let digest = *transaction.digest();
117        let mut pending = self
118            .transactions
119            .write()
120            .ok()
121            .context("pending transaction lock poisoned")?;
122        pending.entry(digest).or_default().transaction = Some(transaction);
123        Ok(())

151    /// Stage one execution result's object writes and removals.
152    ///
153    /// The merged views mirror the row-level semantics of
154    /// `LocalStore::stage_local_object_diff`: removals apply before writes so
155    /// an object removed and rewritten in the same result ends up live, and an
156    /// object created and terminally deleted in the same result keeps only its
157    /// historical version row.
158    pub(crate) fn record_object_diff(
159        &self,
160        written: BTreeMap<ObjectID, Object>,
161        removed: Vec<ObjectRemoval>,
162    ) -> anyhow::Result<()> {
163        let mut pending = self
164            .objects
165            .write()
166            .ok()
167            .context("pending objects lock poisoned")?;
168
169        let terminal_deleted: BTreeSet<ObjectID> = removed
170            .iter()
171            .filter_map(|removal| {
172                (removal.kind == TombstoneKind::Deleted).then_some(removal.object_id)
173            })
174            .collect();
175
176        for removal in &removed {
177            pending
178                .versions
179                .entry(removal.object_id)
180                .or_default()
181                .insert(removal.version, Status::Tombstone(removal.kind));
182            pending.current.insert(
183                removal.object_id,
184                (removal.version, Status::Tombstone(removal.kind)),
185            );
186        }
187        for object in written.values() {
188            pending
189                .versions
190                .entry(object.id())
191                .or_default()
192                .insert(object.version(), Status::Live(object.clone()));
193            if !terminal_deleted.contains(&object.id()) {
194                pending.current.insert(
195                    object.id(),
196                    (object.version(), Status::Live(object.clone())),
197                );
198            }
199        }
200
201        pending.diffs.push(ObjectDiff { written, removed });
202        Ok(())
203    }

205    /// The staged current state for an object, if the in-flight checkpoint
206    /// touched it.
207    pub(crate) fn latest_object(
208        &self,

219    /// The staged status at one exact version, if the in-flight checkpoint
220    /// wrote it.
221    pub(crate) fn object_at_version(
222        &self,
223        id: ObjectID,
224        version: SequenceNumber,
225    ) -> anyhow::Result<Option<Status>> {
226        let pending = self
227            .objects
228            .read()
229            .ok()
230            .context("pending objects lock poisoned")?;
231        Ok(pending
232            .versions
233            .get(&id)
234            .and_then(|versions| versions.get(&version))
235            .cloned())
236    }

238    /// The highest staged version at or below `upper_bound`, if any.
239    ///
240    /// Staged versions always outrank persisted ones for the same object, so a
241    /// hit here is the bounded read's answer; only a miss falls through to the
242    /// store.
243    pub(crate) fn object_at_or_before(
244        &self,
245        id: ObjectID,
246        upper_bound: SequenceNumber,
247    ) -> anyhow::Result<Option<(SequenceNumber, Status)>> {
248        let pending = self
249            .objects
250            .read()
251            .ok()
252            .context("pending objects lock poisoned")?;
253        Ok(pending.versions.get(&id).and_then(|versions| {
254            versions
255                .range(..=upper_bound)
256                .next_back()
257                .map(|(version, status)| (*version, status.clone()))
258        }))
259    }

271    /// Return the staged checkpoint matching `contents`, validating that the
272    /// contents digest is the one the checkpoint committed to.
273    pub(crate) fn checkpoint_for_contents(
274        &self,
275        contents: &CheckpointContents,
276    ) -> anyhow::Result<VerifiedCheckpoint> {
277        let pending = self
278            .checkpoint
279            .read()
280            .ok()
281            .context("pending checkpoint lock poisoned")?;
282        let Some(checkpoint) = pending.as_ref() else {
283            bail!(
284                "checkpoint contents {} inserted without a pending checkpoint",
285                contents.digest(),
286            );
287        };
288        if checkpoint.data().content_digest != *contents.digest() {
289            bail!(
290                "pending checkpoint {} references contents {}, but inserted contents are {}",
291                checkpoint.data().sequence_number,
292                checkpoint.data().content_digest,
293                contents.digest(),
294            );
295        }
296        Ok(checkpoint.clone())
297    }

299    /// Clone out every staged transaction referenced by `contents`, in
300    /// checkpoint order. Entries stay staged until [`Self::clear_sealed`]
301    /// confirms they were durably saved.
302    pub(crate) fn staged_transactions_for(
303        &self,
304        checkpoint: &VerifiedCheckpoint,
305        contents: &CheckpointContents,
306    ) -> anyhow::Result<Vec<StagedTransaction>> {
307        let pending = self
308            .transactions
309            .read()
310            .ok()
311            .context("pending transaction lock poisoned")?;
312        let mut staged = Vec::new();
313        for execution in contents.iter() {
314            let digest = execution.transaction;
315            let Some(entry) = pending.get(&digest) else {
316                bail!(
317                    "checkpoint {} references transaction {digest}, but no pending transaction was recorded",
318                    checkpoint.data().sequence_number,
319                );
320            };
321            let transaction = entry.transaction.clone().with_context(|| {
322                format!(
323                    "checkpoint {} references transaction {digest}, but transaction data is missing",
324                    checkpoint.data().sequence_number,
325                )
326            })?;
327            let effects = entry.effects.clone().with_context(|| {
328                format!(
329                    "checkpoint {} references transaction {digest}, but transaction effects are missing",
330                    checkpoint.data().sequence_number,
331                )
332            })?;
333            let events = entry.events.clone().unwrap_or_default();
334            staged.push(StagedTransaction {
335                digest,
336                transaction,
337                effects,
338                events,
339            });
340        }
341        Ok(staged)
342    }

344    /// Drop the staged entries for a checkpoint whose rows were durably saved.
345    pub(crate) fn clear_sealed(
346        &self,
347        checkpoint: &VerifiedCheckpoint,
348        digests: impl IntoIterator<Item = TransactionDigest>,
349    ) -> anyhow::Result<()> {
350        // Sealing is serialized, so every staged diff belongs to the
351        // checkpoint that just committed.
352        let mut objects = self
353            .objects
354            .write()
355            .ok()
356            .context("pending objects lock poisoned")?;
357        *objects = PendingObjects::default();
358        drop(objects);
359
360        let mut pending = self
361            .transactions
362            .write()
363            .ok()
364            .context("pending transaction lock poisoned")?;
365        for digest in digests {
366            pending.remove(&digest);
367        }
368        drop(pending);
369
370        let mut pending_checkpoint = self
371            .checkpoint
372            .write()
373            .ok()
374            .context("pending checkpoint lock poisoned")?;
375        if pending_checkpoint
376            .as_ref()
377            .is_some_and(|pending| pending.digest() == checkpoint.digest())
378        {
379            *pending_checkpoint = None;
380        }
381        Ok(())
382    }
