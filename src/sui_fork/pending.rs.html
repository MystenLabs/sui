4//! In-memory staging for the checkpoint currently being sealed.
5//!
6//! Simulacrum hands the store a checkpoint summary, then its transactions, effects, events, and
7//! object diffs piecemeal through the `SimulatorStore` insert methods, and finally the checkpoint
8//! contents. This buffer holds those pieces until the contents arrive, so the caller can persist
9//! the whole checkpoint into the DB as one atomic batch.
10//!
11//! Object diffs are the only staged entries that reads consult, because execution is the only
12//! reader that runs before seal. Each transaction in the in-flight checkpoint, including the
13//! settlement and barrier system transactions sealed with it, resolves its inputs through the
14//! store, so the staged diffs serve as a read overlay over the rpc-store until the seal commits
15//! them. A read returns the overlay entry when one exists and falls through to the store otherwise.
16//! No version comparison is needed, because an overlay entry is always newer than any persisted
17//! row. Local execution Lamport-bumps past every live version, while remote fetches stay pinned at
18//! or below the fork checkpoint. Checkpoint, transaction, effects, and event entries stay
19//! write-only because their only readers run after execution returns, and sealing completes
20//! synchronously inside checkpoint publication, so those rows are already in the rpc-store by then.
21//!
22//! Nothing here is persisted. Staged entries that have not been sealed are lost on process restart,
23//! which is the crash-safety contract, because a checkpoint either commits whole at seal time or
24//! leaves no trace.

55/// One fully staged transaction, cloned out of the buffer at seal time.

59    pub(crate) effects: TransactionEffects,

63/// One execution result's object writes and removals, staged until seal.
64#[derive(Clone)]
65pub(crate) struct ObjectDiff {
66    pub(crate) written: BTreeMap<ObjectID, Object>,
67    pub(crate) removed: Vec<ObjectRemoval>,
68}

70/// Staged object diffs plus the merged views the read overlay serves.
71///
72/// The views mirror the rows `LocalStore::stage_local_object_diff` will write at seal, and they are
73/// maintained on record rather than derived on read so overlay lookups stay O(log n) on the
74/// execution hot path.
75#[derive(Default)]
76struct PendingObjects {
77    /// Diffs in arrival order, replayed verbatim into the seal batch.
78    diffs: Vec<ObjectDiff>,
79
80    /// What the checkpoint-pinned version row would say per object once sealed.
81    current: BTreeMap<ObjectID, (SequenceNumber, Status)>,
82
83    /// What the raw object rows would say per version once sealed.
84    versions: BTreeMap<ObjectID, BTreeMap<SequenceNumber, Status>>,
85}

87/// Staging buffer for the in-flight checkpoint and its transactions.
88///
89/// `RwLock` provides the interior mutability needed behind the shared `Arc<ForkStoreInner>` that
90/// every cloned `ForkStore` holds.
91#[derive(Default)]
92pub(crate) struct PendingCheckpointBuffer {
93    checkpoint: RwLock<Option<VerifiedCheckpoint>>,
94    transactions: RwLock<BTreeMap<TransactionDigest, PendingTransaction>>,
95    objects: RwLock<PendingObjects>,
96}

103    pub(crate) fn record_checkpoint(&self, checkpoint: VerifiedCheckpoint) -> anyhow::Result<()> {
104        let mut pending = self
105            .checkpoint
106            .write()
107            .ok()
108            .context("pending checkpoint lock poisoned")?;
109        *pending = Some(checkpoint);
110        Ok(())
111    }

113    pub(crate) fn record_transaction(
114        &self,
115        transaction: VerifiedTransaction,
116    ) -> anyhow::Result<()> {
117        let digest = *transaction.digest();
118        let mut pending = self
119            .transactions
120            .write()
121            .ok()
122            .context("pending transaction lock poisoned")?;
123        pending.entry(digest).or_default().transaction = Some(transaction);
124        Ok(())

152    /// Stage one execution result's object writes and removals.
153    ///
154    /// The merged views mirror the row-level semantics of `LocalStore::stage_local_object_diff`.
155    /// Removals apply before writes so an object removed and rewritten in the same result ends up
156    /// live, and an object created and terminally deleted in the same result keeps only its
157    /// historical version row.
158    pub(crate) fn record_object_diff(
159        &self,
160        written: BTreeMap<ObjectID, Object>,
161        removed: Vec<ObjectRemoval>,
162    ) -> anyhow::Result<()> {
163        let mut pending = self
164            .objects
165            .write()
166            .ok()
167            .context("pending objects lock poisoned")?;
168
169        let terminal_deleted: BTreeSet<ObjectID> = removed
170            .iter()
171            .filter_map(|removal| {
172                (removal.kind == TombstoneKind::Deleted).then_some(removal.object_id)
173            })
174            .collect();
175
176        for removal in &removed {
177            pending
178                .versions
179                .entry(removal.object_id)
180                .or_default()
181                .insert(removal.version, Status::Tombstone(removal.kind));
182            pending.current.insert(
183                removal.object_id,
184                (removal.version, Status::Tombstone(removal.kind)),
185            );
186        }
187        for object in written.values() {
188            pending
189                .versions
190                .entry(object.id())
191                .or_default()
192                .insert(object.version(), Status::Live(object.clone()));
193            if !terminal_deleted.contains(&object.id()) {
194                pending.current.insert(
195                    object.id(),
196                    (object.version(), Status::Live(object.clone())),
197                );
198            }
199        }
200
201        pending.diffs.push(ObjectDiff { written, removed });
202        Ok(())
203    }

205    /// Return the staged current state for an object, if the in-flight checkpoint touched it.
206    pub(crate) fn latest_object(
207        &self,
208        id: ObjectID,

210        let pending = self
211            .objects

213            .ok()

218    /// Return the staged status at one exact version, if the in-flight checkpoint wrote it.
219    pub(crate) fn object_at_version(
220        &self,
221        id: ObjectID,
222        version: SequenceNumber,
223    ) -> anyhow::Result<Option<Status>> {
224        let pending = self
225            .objects
226            .read()
227            .ok()
228            .context("pending objects lock poisoned")?;
229        Ok(pending
230            .versions
231            .get(&id)
232            .and_then(|versions| versions.get(&version))
233            .cloned())
234    }

236    /// Return the highest staged version at or below `upper_bound`, if any.
237    ///
238    /// Staged versions always outrank persisted ones for the same object, so callers take a hit
239    /// here as the bounded read's result and fall through to the store only on a miss.
240    pub(crate) fn object_at_or_before(
241        &self,
242        id: ObjectID,
243        upper_bound: SequenceNumber,
244    ) -> anyhow::Result<Option<(SequenceNumber, Status)>> {
245        let pending = self
246            .objects
247            .read()
248            .ok()
249            .context("pending objects lock poisoned")?;
250        Ok(pending.versions.get(&id).and_then(|versions| {
251            versions
252                .range(..=upper_bound)
253                .next_back()
254                .map(|(version, status)| (*version, status.clone()))
255        }))
256    }

258    /// Clone every staged object diff, in arrival order, for the seal.

268    /// Return the staged checkpoint matching `contents`, validating that the contents digest is the
269    /// one the checkpoint committed to.
270    pub(crate) fn checkpoint_for_contents(
271        &self,
272        contents: &CheckpointContents,
273    ) -> anyhow::Result<VerifiedCheckpoint> {
274        let pending = self
275            .checkpoint
276            .read()
277            .ok()
278            .context("pending checkpoint lock poisoned")?;
279        let Some(checkpoint) = pending.as_ref() else {
280            bail!(
281                "checkpoint contents {} inserted without a pending checkpoint",
282                contents.digest(),
283            );
284        };
285        if checkpoint.data().content_digest != *contents.digest() {
286            bail!(
287                "pending checkpoint {} references contents {}, but inserted contents are {}",
288                checkpoint.data().sequence_number,
289                checkpoint.data().content_digest,
290                contents.digest(),
291            );
292        }
293        Ok(checkpoint.clone())
294    }

296    /// Clone every staged transaction referenced by `contents`, in checkpoint order.
297    ///
298    /// Entries stay staged until [`Self::clear_sealed`] confirms they were durably saved.
299    pub(crate) fn staged_transactions_for(
300        &self,
301        checkpoint: &VerifiedCheckpoint,
302        contents: &CheckpointContents,
303    ) -> anyhow::Result<Vec<StagedTransaction>> {
304        let pending = self
305            .transactions
306            .read()
307            .ok()
308            .context("pending transaction lock poisoned")?;
309        let mut staged = Vec::new();
310        for execution in contents.iter() {
311            let digest = execution.transaction;
312            let Some(entry) = pending.get(&digest) else {
313                bail!(
314                    "checkpoint {} references transaction {digest}, but no pending transaction was recorded",
315                    checkpoint.data().sequence_number,
316                );
317            };
318            let transaction = entry.transaction.clone().with_context(|| {
319                format!(
320                    "checkpoint {} references transaction {digest}, but transaction data is missing",
321                    checkpoint.data().sequence_number,
322                )
323            })?;
324            let effects = entry.effects.clone().with_context(|| {
325                format!(
326                    "checkpoint {} references transaction {digest}, but transaction effects are missing",
327                    checkpoint.data().sequence_number,
328                )
329            })?;
330            let events = entry.events.clone().unwrap_or_default();
331            staged.push(StagedTransaction {
332                digest,
333                transaction,
334                effects,
335                events,
336            });
337        }
338        Ok(staged)
339    }

341    /// Drop every staged entry for a checkpoint whose rows were durably saved.
342    pub(crate) fn clear_sealed(
343        &self,
344        checkpoint: &VerifiedCheckpoint,
345        digests: impl IntoIterator<Item = TransactionDigest>,
346    ) -> anyhow::Result<()> {
347        // Sealing is serialized, so every staged diff belongs to the
348        // checkpoint that just committed.
349        let mut objects = self
350            .objects
351            .write()
352            .ok()
353            .context("pending objects lock poisoned")?;
354        *objects = PendingObjects::default();
355        drop(objects);
356
357        let mut pending = self
358            .transactions
359            .write()
360            .ok()
361            .context("pending transaction lock poisoned")?;
362        for digest in digests {
363            pending.remove(&digest);
364        }
365        drop(pending);
366
367        let mut pending_checkpoint = self
368            .checkpoint
369            .write()
370            .ok()
371            .context("pending checkpoint lock poisoned")?;
372        if pending_checkpoint
373            .as_ref()
374            .is_some_and(|pending| pending.digest() == checkpoint.digest())
375        {
376            *pending_checkpoint = None;
377        }
378        Ok(())
379    }
