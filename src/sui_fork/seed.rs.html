4//! Fork manifest and seed resolution for seed-bounded index reads.
5//!
6//! Seeding happens once, at fork creation, in two steps. Resolution enumerates the requested
7//! addresses and objects against GraphQL pinned at the fork checkpoint and records the resulting
8//! object references in an immutable manifest stored on disk. The fork fetches that data from the
9//! live netork RPC and hands them to `sui-rpc-store`'s `Restore` pipelines, which is where the
10//! fork's whole pre-fork derived-index surface comes from.
11

37/// Objects fetched per GraphQL query when querying for seed objects. The load commits as one batch
38/// regardless, and this only bounds the size of an individual query.
39const OBJECTS_PER_QUERY: usize = 50;
40
41/// CLI seed input before it has been resolved against the upstream chain.
42#[derive(Clone, Debug, Default)]
43pub struct SeedInput {
44    /// Addresses whose owned objects should be recorded in the seed manifest.
45    pub addresses: BTreeSet<SuiAddress>,
46    /// Object IDs to fetch and seed.
47    pub object_ids: BTreeSet<ObjectID>,
48}
49
50/// Object reference recorded in the manifest and hydrated by the seed load.
51#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
52pub(crate) struct SeedEntry {
53    pub(crate) object_ref: ObjectRef,
54}
55
56/// Durable manifest for fork metadata and optional pre-fork seed metadata.
57#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
58pub(crate) struct SeedManifest {
59    pub(crate) network: String,
60    pub(crate) checkpoint: CheckpointSequenceNumber,
61    /// Addresses that were fully enumerated to produce this manifest. Nothing reads the list, and
62    /// it exists as a record for whoever inspects the fork directory of which addresses the seeding
63    /// used.
64    #[serde(default)]
65    pub(crate) addresses: Vec<SuiAddress>,
66    pub(crate) entries: Vec<SeedEntry>,
67}
68
69impl SeedManifest {
70    fn empty(network: String, checkpoint: CheckpointSequenceNumber) -> Self {
71        Self {
72            network,
73            checkpoint,
74            addresses: Vec::new(),
75            entries: Vec::new(),
76        }
77    }
78}
79
80impl SeedInput {
81    /// Return true when no addresses or objects were requested for seeding.
82    pub fn is_empty(&self) -> bool {
83        self.addresses.is_empty() && self.object_ids.is_empty()
84    }
85}

87impl From<AddressOwnedObject> for SeedEntry {
88    fn from(object: AddressOwnedObject) -> Self {
89        Self {
90            object_ref: object.object_ref,
91        }
92    }
93}

95/// Reject seed inputs that would overwrite or reinterpret an existing manifest.
96pub(crate) fn ensure_seed_policy(local: &MetadataStore, input: &SeedInput) -> Result<(), Error> {
97    if local.seed_manifest_exists() && !input.is_empty() {
98        bail!(
99            "A seed manifest already exists at {}. To fork the same checkpoint with different seeds, use a different --data-dir.",
100            local.seed_manifest_path().display(),
101        );
102    }
103    Ok(())
104}

106/// Ensure an existing fork manifest matches the requested network and checkpoint.
107pub(crate) fn ensure_seed_manifest_matches(
108    manifest: &SeedManifest,
109    network: &str,
110    checkpoint: Option<CheckpointSequenceNumber>,
111) -> Result<(), Error> {
112    if manifest.network != network {
113        bail!(
114            "Seed manifest network {} does not match requested network {}. Use a different --data-dir.",
115            manifest.network,
116            network,
117        );
118    }
119
120    if let Some(checkpoint) = checkpoint
121        && manifest.checkpoint != checkpoint
122    {
123        bail!(
124            "Seed manifest checkpoint {} does not match requested checkpoint {}. Use a different --data-dir.",
125            manifest.checkpoint,
126            checkpoint,
127        );
128    }
129
130    Ok(())
131}

133/// Load or create the seed manifest for the current fork directory.
134pub(crate) async fn prepare_seed_manifest(
135    store: &ForkStore,
136    network: String,
137    input: &SeedInput,
138) -> Result<SeedManifest, Error> {
139    if store.metadata().seed_manifest_exists() {
140        if !input.is_empty() {
141            bail!(
142                "A seed manifest already exists at {}. To fork the same checkpoint with different seeds, use a different --data-dir.",
143                store.metadata().seed_manifest_path().display(),
144            );
145        }
146        let manifest = store.metadata().read_seed_manifest()?;
147        ensure_seed_manifest_matches(&manifest, &network, Some(store.forked_at_checkpoint()))?;
148        return Ok(manifest);
149    }
150
151    let manifest = if input.is_empty() {
152        SeedManifest::empty(network, store.forked_at_checkpoint())
153    } else {
154        resolve_seeds(input, network, store).await?
155    };
156    store.metadata().write_seed_manifest(&manifest)?;
157    Ok(manifest)
158}
159
160/// Load every manifest entry into the rpc-store with its full derived-index surface, once, before
161/// the fork executes anything.
162///
163/// The manifest holds object references rather than objects, so each entry is fetched by id and
164/// version and handed to the restore pipelines, which rebuild the owned-object index and the rest
165/// of the derived surface. The load runs at most once per fork directory, because `Balance`
166/// accumulates through a merge operator and a second pass would double-count every seeded coin, and
167/// it commits its own completion marker atomically with the rows to make that unrepeatable rather
168/// than merely unlikely.
169pub(crate) fn load_seed_objects(store: &ForkStore, manifest: &SeedManifest) -> Result<(), Error> {
170    if store.local_store().seed_load_complete()? {
171        return Ok(());
172    }
173
174    let object_refs: Vec<_> = manifest
175        .entries
176        .iter()
177        .map(|entry| entry.object_ref)
178        .collect();
179
180    let mut objects = Vec::with_capacity(object_refs.len());
181    for chunk in object_refs.chunks(OBJECTS_PER_QUERY) {
182        objects.extend(store.fetch_seed_objects(chunk)?);
183    }
184
185    store.local_store().restore_seed_objects(&objects)
186}

188/// Resolve the owned objects and balances belonging to `address`.
189async fn resolve_address_seed(
190    remote: &RemoteSource,
191    address: SuiAddress,
192) -> Result<Vec<SeedEntry>, Error> {
193    let mut entries: Vec<SeedEntry> = remote
194        .address_owned_objects_at_fork(address)
195        .await?
196        .into_iter()
197        .map(SeedEntry::from)
198        .collect();
199    entries.extend(resolve_address_balance_seed(remote, address).await?);
200    Ok(entries)
201}
202
203/// Resolve the accumulator balance fields belonging to `address`.
204///
205/// An address balance lives in a dynamic field under the accumulator root rather than in an object
206/// the address owns, so the owned-object enumeration above never surfaces it, and seeding an
207/// address without this step would establish its coins while silently leaving its balance at zero.
208/// Local execution does maintain address balances, so a withdrawal would then apply a delta to a
209/// baseline that was never seeded.
210///
211/// The field's id is derivable from `(address, coin type)`, so the only thing that has to come from
212/// the remote is which coin types to derive for, which is the one part nothing local can know. Each
213/// derived id is then resolved like any other object reference and seeded as an ordinary object, so
214/// the stock `Balance` restore pipeline picks it up through its accumulator-root arm without this
215/// crate writing a balance row itself.
216async fn resolve_address_balance_seed(
217    remote: &RemoteSource,
218    address: SuiAddress,
219) -> Result<Vec<SeedEntry>, Error> {
220    let checkpoint = remote.forked_at_checkpoint();
221    let coin_types = remote
222        .address_balance_coin_types_at_fork(address)
223        .await
224        .with_context(|| format!("failed to resolve address balances for {address}"))?;
225    if coin_types.is_empty() {
226        return Ok(Vec::new());
227    }
228
229    let mut field_ids = Vec::with_capacity(coin_types.len());
230    for coin_type in &coin_types {
231        match accumulator_field_id(address, coin_type) {
232            Ok(field_id) => field_ids.push(field_id),
233            // A coin type the accumulator cannot key on is not a failure of the
234            // seed: it just has no field to fetch.
235            Err(e) => warn!(%address, %coin_type, "skipping address balance seed: {e:#}"),
236        }
237    }
238
239    let refs = remote
240        .object_refs_at_fork(&field_ids)
241        .await
242        .with_context(|| format!("failed to resolve accumulator fields for {address}"))?;
243
244    let mut entries = Vec::new();
245    for (field_id, object_ref) in field_ids.iter().zip_eq(refs) {
246        match object_ref {
247            Some(object_ref) => entries.push(SeedEntry { object_ref }),
248            // The balances connection reported a coin type, but no field exists
249            // at the fork checkpoint. Nothing to seed, and nothing wrong.
250            None => warn!(
251                %address,
252                %field_id,
253                checkpoint,
254                "address balance field not found at fork checkpoint",
255            ),
256        }
257    }
258    Ok(entries)
259}
260
261/// Derive the object id of the accumulator field holding `address`'s balance of `coin_type`.
262///
263/// `coin_type` arrives as the inner type (`0x2::sui::SUI`), while the accumulator keys its fields
264/// on the wrapped `0x2::balance::Balance<T>`, so the id is derived from the wrapped form.
265fn accumulator_field_id(address: SuiAddress, coin_type: &str) -> Result<ObjectID, Error> {
266    let inner: TypeTag = coin_type
267        .parse()
268        .with_context(|| format!("invalid coin type {coin_type}"))?;
269    Ok(*AccumulatorValue::get_field_id(address, &Balance::type_tag(inner))?.inner())
270}

272/// Resolve the requested object ids against the remote source at the fork checkpoint.
273async fn resolve_object_seeds(
274    remote: &RemoteSource,
275    object_ids: &[ObjectID],
276) -> Result<Vec<SeedEntry>, Error> {
277    if object_ids.is_empty() {
278        return Ok(Vec::new());
279    }
280
281    let checkpoint = remote.forked_at_checkpoint();
282    let objects = remote.object_seed_metadata_at_fork(object_ids).await?;
283    let mut entries = Vec::new();
284
285    for (object_id, object) in object_ids.iter().zip_eq(objects) {
286        match object {
287            ObjectSeedMetadata::Missing => {
288                warn!(%object_id, checkpoint, "object seed not found at fork checkpoint");
289            }
290            ObjectSeedMetadata::NonAddressOwned => {
291                warn!(
292                    %object_id,
293                    checkpoint,
294                    "object seed is not owned by an address and will not be added to the seed manifest",
295                );
296            }
297            ObjectSeedMetadata::AddressOwned(object) => entries.push(SeedEntry::from(object)),
298        }
299    }
300
301    Ok(entries)
302}

304/// Resolve the requested addresses and object ids against the remote source at the fork checkpoint.
305async fn resolve_seeds(
306    input: &SeedInput,
307    network: String,
308    store: &ForkStore,
309) -> Result<SeedManifest, Error> {
310    let checkpoint = store.forked_at_checkpoint();
311    let mut entries = BTreeMap::new();
312
313    // Address seeds are ignored when the fork checkpoint is older
314    // than the remote's ownership-enumeration window. The skipped addresses must NOT be
315    // recorded in the manifest. The address list claims a complete scan, and
316    // claiming one that never ran is worse than recording nothing.
317    let addresses: Vec<SuiAddress> = if input.addresses.is_empty() {
318        Vec::new()
319    } else {
320        let lowest_available = store.remote().lowest_available_checkpoint_objects()?;
321        if checkpoint < lowest_available {
322            let message = format!(
323                "ignoring --address seeds: checkpoint {checkpoint} is older than the remote's \
324                 object-ownership window (available from checkpoint {lowest_available}, roughly \
325                 the last hour). If the object IDs are known, seed them directly with --object \
326                 instead."
327            );
328            warn!(
329                addresses = ?input.addresses,
330                checkpoint,
331                lowest_available,
332                "{message}",
333            );
334            // The CLI may run without a tracing subscriber, and a silently
335            // dropped seed address would read as a bug, so the warning also
336            // goes to stderr directly.
337            eprintln!("warning: {message}");
338
339            Vec::new()
340        } else {
341            input.addresses.iter().copied().collect()
342        }
343    };
344    for address in addresses.iter().copied() {
345        let address_entries = resolve_address_seed(store.remote(), address).await?;
346        if address_entries.is_empty() {
347            warn!(%address, checkpoint, "address seed resolved no owned objects");
348        }
349        for entry in address_entries {
350            entries.insert(entry.object_ref.0, entry);
351        }
352    }
353
354    let remaining_object_ids: Vec<_> = input
355        .object_ids
356        .iter()
357        .copied()
358        .filter(|object_id| !entries.contains_key(object_id))
359        .collect();
360    for entry in resolve_object_seeds(store.remote(), &remaining_object_ids).await? {
361        entries.insert(entry.object_ref.0, entry);
362    }
363
364    Ok(SeedManifest {
365        network,
366        checkpoint,
367        addresses,
368        entries: entries.into_values().collect(),
369    })
370}
