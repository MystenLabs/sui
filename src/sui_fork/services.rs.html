4//! Initialization and ownership of the fork's underlying services.
5//!
6//! `ServiceManager` owns the durable RPC-store state under the fork data
7//! directory. Opening it validates or creates fork metadata, opens the local
8//! `sui-rpc-store` RocksDB instance, records the fork chain identifier, and
9//! refreshes RPC-store pruning metadata before any readers are handed out.
10//!
11//! After the startup path creates Simulacrum, this module can start the embedded
12//! `sui-rpc-store` indexer. The indexer consumes checkpoints produced by
13//! Simulacrum, saves them into the local RPC store, and runs the checkpoint
14//! broadcast pipeline used by RPC subscriptions.
15//!
16//! The module deliberately stays below orchestration concerns. `startup`
17//! chooses the remote checkpoint, initializes remote readers, seeds Simulacrum
18//! state, and builds the RPC server; this module keeps the opened store and
19//! indexer service alive for the rest of the process.

84/// The pipelines the embedded indexer owns, by name — the same set
85/// [`ServiceManager::pipeline_layer`] enables, which the test at the bottom of
86/// this file pins.
87///
88/// These are the only pipelines whose watermarks anything writes, so they are
89/// also the only ones that can bound the fork's indexed tip. The column
90/// families `LocalStore` writes are committed in the same batch as their data
91/// and never lag, so including them would peg the bound at `None` forever.
92/// Two callers need the names rather than the layer: the reader's pipeline set
93/// in [`LocalStore::new`], and the watermarks the seed load writes.
94pub(crate) const FORK_INDEXER_PIPELINES: &[&str] = &[
95    ObjectByOwner::NAME,
96    ObjectByType::NAME,
97    Balance::NAME,
98    PackageVersions::NAME,
99    TransactionBitmap::NAME,
100    EventBitmap::NAME,
101    Epochs::NAME,
102];

104/// Opened fork services backed by `sui-rpc-store`.
105///
106/// The manager owns the local RPC store and, once started, keeps the embedded
107/// `sui-rpc-store` indexer alive for local Simulacrum checkpoints.
108pub(crate) struct ServiceManager {
109    db: Db,
110    schema: Arc<RpcStoreSchema>,
111    metadata: Metadata,
112    indexer_pipelines: Vec<&'static str>,
113    /// Handle to the running indexer. Holding it keeps the indexer alive
114    /// (dropping the `Service` stops the background tasks), and
115    /// [`Self::indexer_stopped`] joins it to observe failures. Behind an async
116    /// mutex because `Service::join` needs exclusive access while the manager
117    /// is shared behind the `Context`.
118    indexer_service: Option<tokio::sync::Mutex<Service>>,
119}

121/// What fork a data directory belongs to, written once and thereafter compared
122/// on every open. Equality is the whole check, so every field is part of the
123/// fork's identity.
124#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
125struct Metadata {
126    format_version: u32,
127    network: String,
128    forked_at_checkpoint: CheckpointSequenceNumber,
129    chain_identifier: [u8; 32],
130}
131
132impl ServiceManager {
133    /// Open the fork's durable state under `root`, creating it if this is a new
134    /// fork directory.
135    ///
136    /// Everything a reader depends on is established here, before any reader
137    /// exists: metadata is written or matched against what is already on disk,
138    /// the rpc-store is opened, its pruning watermarks are refreshed, and the
139    /// chain identifier is recorded. Reusing a directory that describes a
140    /// different network or checkpoint fails rather than reinterpreting it.
141    pub(crate) fn open(
142        root: &Path,
143        network: String,
144        forked_at_checkpoint: CheckpointSequenceNumber,
145        chain_identifier: ChainIdentifier,
146    ) -> anyhow::Result<Self> {
147        fs::create_dir_all(root)
148            .with_context(|| format!("failed to create fork data directory {}", root.display()))?;
149
150        let metadata = Metadata {
151            format_version: FORK_METADATA_FORMAT_VERSION,
152            network,
153            forked_at_checkpoint,
154            chain_identifier: *chain_identifier.as_bytes(),
155        };
156        Self::load_or_write_metadata(root, &metadata)?;
157
158        let db_path = Self::rpc_store_path(root);
159        let (db, schema) = Db::open::<RpcStoreSchema>(
160            &db_path,
161            DbOptions {
162                rocksdb: default_rocksdb_config(),
163                ..DbOptions::default()
164            },
165        )
166        .with_context(|| format!("failed to open rpc store at {}", db_path.display()))?;
167        let schema = Arc::new(schema);
168
169        Self::seed_chain_identifier(&db, metadata.chain_identifier)?;
170
171        Ok(Self {
172            db,
173            schema,
174            metadata,
175            indexer_pipelines: Vec::new(),
176            indexer_service: None,
177        })
178    }

180    /// The checkpoint an existing fork directory was forked at, or `None` if
181    /// `root` holds no fork yet.
182    ///
183    /// Reads the metadata sidecar without opening the store, so startup can
184    /// decide whether it is resuming a fork or creating one before committing
185    /// to either. A directory that belongs to a different network, or to a
186    /// different checkpoint than the one requested, is an error.
187    pub(crate) fn existing_forked_checkpoint(
188        root: &Path,
189        network: &str,
190        requested_checkpoint: Option<CheckpointSequenceNumber>,
191    ) -> anyhow::Result<Option<CheckpointSequenceNumber>> {
192        let path = Self::metadata_path(root);
193        if !path.exists() {
194            return Ok(None);
195        }
196        let stored: Metadata = serde_json::from_slice(
197            &fs::read(&path)
198                .with_context(|| format!("failed to read fork metadata {}", path.display()))?,
199        )
200        .with_context(|| format!("failed to parse fork metadata {}", path.display()))?;
201        ensure!(
202            stored.network == network,
203            "fork metadata network {} does not match requested network {}. Use a different --data-dir.",
204            stored.network,
205            network,
206        );
207        if let Some(checkpoint) = requested_checkpoint {
208            ensure!(
209                stored.forked_at_checkpoint == checkpoint,
210                "fork metadata checkpoint {} does not match requested checkpoint {}. Use a different --data-dir.",
211                stored.forked_at_checkpoint,
212                checkpoint,
213            );
214        }
215        Ok(Some(stored.forked_at_checkpoint))
216    }

218    /// Start the embedded rpc-store indexer over locally produced checkpoints.
219    ///
220    /// Ingestion reads checkpoints back out of `simulacrum`, starting at the
221    /// checkpoint after the fork point — everything at or below it is pre-fork
222    /// state the seed load already placed. Registers the pipelines in
223    /// [`Self::pipeline_layer`] plus the broadcast pipeline that feeds RPC
224    /// subscriptions, and errors if an indexer is already running.
225    pub(crate) async fn start_indexer(
226        &mut self,
227        simulacrum: Arc<RwLock<ForkedSimulacrum>>,
228        checkpoint_sender: broadcast::Sender<Arc<Checkpoint>>,
229        registry: &Registry,
230    ) -> anyhow::Result<()> {
231        ensure!(
232            self.indexer_service.is_none(),
233            "fork rpc-store indexer is already running",
234        );
235
236        let first_checkpoint = self
237            .metadata
238            .forked_at_checkpoint
239            .checked_add(1)
240            .context("forked_at_checkpoint cannot be u64::MAX")?;
241        let ingestion_metrics = IngestionMetrics::new(Some(METRICS_PREFIX), registry);
242        let ingestion_client = IngestionClient::from_trait(
243            Arc::new(SimulacrumIngestion::new(
244                simulacrum,
245                self.chain_identifier(),
246            )),
247            ingestion_metrics,
248        );
249        let store = Store::new(self.db.clone(), self.schema.clone());
250        let mut indexer = Indexer::from_store(
251            store,
252            IndexerArgs {
253                first_checkpoint: Some(first_checkpoint),
254                ..IndexerArgs::default()
255            },
256            ingestion_client,
257            None,
258            sui_rpc_store::ConsistencyConfig::default(),
259            None,
260            IngestionConfig::default(),
261            registry,
262        )
263        .await
264        .context("failed to construct fork rpc-store indexer")?;
265
266        let committer_config = CommitterConfig::default();
267        indexer
268            .add_pipelines(Self::pipeline_layer(), committer_config.clone())
269            .await
270            .context("failed to register fork rpc-store pipelines")?;
271        indexer
272            .add_checkpoint_broadcast(checkpoint_sender, committer_config)
273            .await
274            .context("failed to register fork checkpoint broadcast pipeline")?;
275
276        self.indexer_pipelines = indexer.pipelines().collect();
277        self.indexer_service = Some(tokio::sync::Mutex::new(
278            indexer
279                .run()
280                .await
281                .context("failed to start fork rpc-store indexer")?,
282        ));
283        Ok(())
284    }

286    /// The pipelines the embedded indexer owns.
287    ///
288    /// Only those whose column families the fork does *not* write itself. The
289    /// fork is in the same position as a fullnode running this indexer beside a
290    /// perpetual store: it already holds the raw chain data and serves it
291    /// directly, so re-deriving it here would write every one of those rows
292    /// twice per checkpoint. It has to be the indexer that stands down rather
293    /// than `LocalStore`, because the indexer's own ingestion source is the
294    /// fork's store — [`crate::ingestion::SimulacrumIngestion`] reads each
295    /// checkpoint back out of it — so those rows must already be there before
296    /// ingestion can run at all.
297    ///
298    /// Two of the enabled pipelines are load-bearing in ways that are easy to
299    /// miss. `package_versions` is the *only* writer for a package published
300    /// after the fork point: `stage_local_object_diff` never stages a
301    /// package-version row, so disabling it would silently lose them. `balance`
302    /// accumulates through a merge operator, so it must stay disjoint — the
303    /// fork writes pre-fork balances during the seed load, the indexer writes
304    /// post-fork ones, and any overlap doubles rather than being idempotent.
305    fn pipeline_layer() -> PipelineLayer {
306        PipelineLayer {
307            object_by_owner: Some(CommitterLayer::default()),
308            object_by_type: Some(CommitterLayer::default()),
309            balance: Some(CommitterLayer::default()),
310            package_versions: Some(CommitterLayer::default()),
311            transaction_bitmap: Some(CommitterLayer::default()),
312            event_bitmap: Some(CommitterLayer::default()),
313            epochs: Some(CommitterLayer::default()),
314            // Left off deliberately: `checkpoint_summary`,
315            // `checkpoint_contents`, `checkpoint_seq_by_digest`,
316            // `transactions`, `effects`, `events`, `tx_seq_by_digest`,
317            // `tx_metadata_by_seq`, `objects`, and
318            // `object_version_by_checkpoint` are all written synchronously by
319            // `LocalStore`.
320            ..PipelineLayer::default()
321        }
322    }

341    /// Resolves when the embedded indexer stops: `Ok(())` if all its tasks
342    /// completed (unexpected while the fork is serving), or the task error if
343    /// any pipeline failed or panicked. Pends forever if the indexer was never
344    /// started.
345    pub(crate) async fn indexer_stopped(&self) -> anyhow::Result<()> {
346        let Some(service) = &self.indexer_service else {
347            return std::future::pending().await;
348        };
349        service.lock().await.join().await
350    }

352    /// Wait until the indexer has committed `checkpoint` on every pipeline it
353    /// owns, so a caller that just produced a checkpoint can rely on the
354    /// derived indexes reflecting it.
355    ///
356    /// Polls rather than subscribing, and gives up after
357    /// [`INDEXED_CHECKPOINT_TIMEOUT`] so a stalled pipeline surfaces as an
358    /// error instead of hanging the caller forever.
359    pub(crate) async fn wait_for_indexed_checkpoint(
360        &self,
361        checkpoint: CheckpointSequenceNumber,
362    ) -> anyhow::Result<()> {
363        let deadline = Instant::now() + INDEXED_CHECKPOINT_TIMEOUT;
364        loop {
365            if self
366                .highest_indexed_checkpoint()?
367                .is_some_and(|indexed| indexed >= checkpoint)
368            {
369                return Ok(());
370            }
371
372            ensure!(
373                Instant::now() < deadline,
374                "timed out waiting for rpc-store to index checkpoint {checkpoint}",
375            );
376            tokio::time::sleep(INDEXED_CHECKPOINT_POLL_INTERVAL).await;
377        }
378    }

380    /// The highest checkpoint *every* indexer pipeline has committed — the
381    /// lowest of their watermarks, since a checkpoint is only fully indexed
382    /// once the slowest pipeline has it.
383    ///
384    /// `None` while the answer is not yet meaningful: no indexer running, or a
385    /// pipeline that has not written a watermark at all.
386    fn highest_indexed_checkpoint(&self) -> anyhow::Result<Option<CheckpointSequenceNumber>> {
387        if self.indexer_pipelines.is_empty() {
388            return Ok(None);
389        }
390
391        let framework = FrameworkSchema::new(self.db.clone());
392        let mut indexed: Option<CheckpointSequenceNumber> = None;
393        for pipeline in &self.indexer_pipelines {
394            let Some(watermark) = framework
395                .watermarks
396                .get(&PipelineTaskKey::new(*pipeline))
397                .with_context(|| format!("failed to read {pipeline} watermark"))?
398            else {
399                return Ok(None);
400            };
401            indexed = Some(
402                indexed.map_or(watermark.checkpoint_hi_inclusive, |checkpoint| {
403                    checkpoint.min(watermark.checkpoint_hi_inclusive)
404                }),
405            );
406        }
407        Ok(indexed)
408    }

425    /// Establish that `root` belongs to the fork described by `expected`,
426    /// writing the metadata if the directory is new and rejecting it if what is
427    /// already there disagrees.
428    fn load_or_write_metadata(root: &Path, expected: &Metadata) -> anyhow::Result<()> {
429        let path = Self::metadata_path(root);
430        if path.exists() {
431            let stored: Metadata = serde_json::from_slice(
432                &fs::read(&path)
433                    .with_context(|| format!("failed to read fork metadata {}", path.display()))?,
434            )
435            .with_context(|| format!("failed to parse fork metadata {}", path.display()))?;
436            ensure!(
437                stored == *expected,
438                "fork metadata at {} does not match requested fork. Use a different --data-dir.",
439                path.display(),
440            );
441            return Ok(());
442        }
443
444        // Temp-file + rename, like the other metadata sidecars: a crash
445        // mid-write must not leave a truncated fork_metadata.json, which
446        // open() would reject as a mismatched fork on every later launch.
447        crate::metadata::write_json_exclusive(&path, expected, "fork metadata")
448    }

450    /// Record the forked-from chain identifier in the rpc-store, under the
451    /// fork's own pipeline key.
452    ///
453    /// The store pins each pipeline to the chain it first ingested and refuses
454    /// checkpoints from another one, so every identifier already present must
455    /// equal `expected` — one that disagrees means this directory holds a
456    /// different chain's data, whatever the metadata sidecar says.
457    fn seed_chain_identifier(db: &Db, expected: [u8; 32]) -> anyhow::Result<()> {
458        let expected = ChainId(expected);
459        let framework = FrameworkSchema::new(db.clone());
460        for entry in framework
461            .chain_ids
462            .iter(..)
463            .context("failed to iterate rpc store chain identifiers")?
464        {
465            let (_, chain_id) = entry.context("failed to read rpc store chain identifier")?;
466            ensure!(
467                chain_id == expected,
468                "rpc store chain identifier does not match fork metadata. Use a different --data-dir.",
469            );
470        }
471
472        let mut batch = db.batch();
473        batch
474            .put(
475                &framework.chain_ids,
476                &PipelineTaskKey::new(FORK_CHAIN_ID_PIPELINE),
477                &expected,
478            )
479            .context("failed to stage rpc store chain identifier")?;
480        batch
481            .commit()
482            .context("failed to commit rpc store chain identifier")
483    }

492    /// The indexer must own exactly the column families `LocalStore` does not
493    /// write, and no others.
494    ///
495    /// Both directions bite silently. Enabling one the fork already writes
496    /// duplicates every row of it per checkpoint — and for `balance`, which
497    /// accumulates through a merge operator, duplication would double the
498    /// value rather than being idempotent. Disabling one the fork does *not*
499    /// write loses those rows outright: `package_versions` is the only writer
500    /// for a package published after the fork point, since
501    /// `stage_local_object_diff` never stages a package-version row.
502    ///
503    /// Neither failure shows up as a test error elsewhere, so the split is
504    /// pinned here.
505    #[test]
506    fn indexer_owns_only_the_pipelines_the_fork_does_not_write() {
507        let layer = ServiceManager::pipeline_layer();
508
509        for (name, enabled) in [
510            ("object_by_owner", layer.object_by_owner.is_some()),
511            ("object_by_type", layer.object_by_type.is_some()),
512            ("balance", layer.balance.is_some()),
513            ("package_versions", layer.package_versions.is_some()),
514            ("transaction_bitmap", layer.transaction_bitmap.is_some()),
515            ("event_bitmap", layer.event_bitmap.is_some()),
516            ("epochs", layer.epochs.is_some()),
517        ] {
518            assert!(enabled, "{name} has no other writer and must stay enabled");
519            assert!(
520                FORK_INDEXER_PIPELINES.contains(&name),
521                "{name} is registered but missing from FORK_INDEXER_PIPELINES, so nothing \
522                 seeds its watermark and the reported indexed tip stays None",
523            );
524        }
525
526        assert_eq!(
527            FORK_INDEXER_PIPELINES.len(),
528            7,
529            "FORK_INDEXER_PIPELINES lists a pipeline the layer does not register; its \
530             watermark would never advance and would pin the reported tip",
531        );
532
533        for (name, enabled) in [
534            ("checkpoint_summary", layer.checkpoint_summary.is_some()),
535            ("checkpoint_contents", layer.checkpoint_contents.is_some()),
536            (
537                "checkpoint_seq_by_digest",
538                layer.checkpoint_seq_by_digest.is_some(),
539            ),
540            ("transactions", layer.transactions.is_some()),
541            ("effects", layer.effects.is_some()),
542            ("events", layer.events.is_some()),
543            ("tx_seq_by_digest", layer.tx_seq_by_digest.is_some()),
544            ("tx_metadata_by_seq", layer.tx_metadata_by_seq.is_some()),
545            ("objects", layer.objects.is_some()),
546            (
547                "object_version_by_checkpoint",
548                layer.object_version_by_checkpoint.is_some(),
549            ),
550        ] {
551            assert!(
552                !enabled,
553                "{name} is written by LocalStore; enabling it here double-writes every row"
554            );
555        }
556    }
