4//! Initialization and ownership of the fork's underlying services.
5//!
6//! `ServiceManager` owns the durable RPC-store state under the fork data directory. Opening it
7//! validates or creates fork metadata, opens the local `sui-rpc-store` RocksDB instance, and
8//! records the fork chain identifier before any readers are handed out.
9//!
10//! After the startup path creates Simulacrum, this module can start the embedded `sui-rpc-store`
11//! indexer. The indexer consumes checkpoints produced by Simulacrum, saves them into the local RPC
12//! store, and runs the checkpoint broadcast pipeline used by RPC subscriptions.
13//!
14//! The module deliberately stays below orchestration concerns. `startup` chooses the remote
15//! checkpoint, initializes remote readers, seeds Simulacrum state, and builds the RPC server, while
16//! this module keeps the opened store and indexer service alive for the rest of the process.
17

81/// The pipelines the embedded indexer owns, by name, matching the set
82/// [`ServiceManager::pipeline_layer`] enables, which the test at the bottom of this file pins.
83///
84/// These are the only pipelines whose watermarks anything writes, so they are also the only ones
85/// that can bound the fork's indexed tip. The column families `LocalStore` writes are committed in
86/// the same batch as their data and never lag, so including them would peg the bound at `None`
87/// forever. Two callers need the names rather than the layer, the reader's pipeline set in
88/// [`LocalStore::new`] and the watermarks the seed load writes.
89pub(crate) const FORK_INDEXER_PIPELINES: &[&str] = &[
90    ObjectByOwner::NAME,
91    ObjectByType::NAME,
92    Balance::NAME,
93    PackageVersions::NAME,
94    TransactionBitmap::NAME,
95    EventBitmap::NAME,
96    Epochs::NAME,
97];

99/// Opened fork services backed by `sui-rpc-store`.
100///
101/// The manager owns the local RPC store and, once started, keeps the embedded `sui-rpc-store`
102/// indexer alive for local Simulacrum checkpoints.
103pub(crate) struct ServiceManager {
104    db: Db,
105    schema: Arc<RpcStoreSchema>,
106    metadata: Metadata,
107    indexer_pipelines: Vec<&'static str>,
108    /// Handle to the running indexer. Holding it keeps the indexer alive (dropping the `Service`
109    /// stops the background tasks), and [`Self::indexer_stopped`] joins it to observe failures.
110    /// Behind an async mutex because `Service::join` needs exclusive access while the manager is
111    /// shared behind the `Context`.
112    indexer_service: Option<tokio::sync::Mutex<Service>>,
113}

115/// Various fork metadata, written to a JSON file in the fork data directory.
116#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
117struct Metadata {
118    format_version: u32,
119    network: String,
120    forked_at_checkpoint: CheckpointSequenceNumber,
121    chain_identifier: [u8; 32],
122}
123
124impl ServiceManager {
125    /// Open the fork's durable state under `root`, creating it if this is a new fork directory.
126    ///
127    /// Everything a reader depends on is established here, before any reader exists. Metadata is
128    /// written or matched against what is already on disk, the rpc-store is opened, and the chain
129    /// identifier is recorded. Reusing a directory that describes a different network or checkpoint
130    /// fails rather than reinterpreting it.
131    pub(crate) fn open(
132        root: &Path,
133        network: String,
134        forked_at_checkpoint: CheckpointSequenceNumber,
135        chain_identifier: ChainIdentifier,
136    ) -> anyhow::Result<Self> {
137        fs::create_dir_all(root)
138            .with_context(|| format!("failed to create fork data directory {}", root.display()))?;
139
140        let metadata = Metadata {
141            format_version: FORK_METADATA_FORMAT_VERSION,
142            network,
143            forked_at_checkpoint,
144            chain_identifier: *chain_identifier.as_bytes(),
145        };
146        Self::load_or_write_metadata(root, &metadata)?;
147
148        let db_path = Self::rpc_store_path(root);
149        let (db, schema) = Db::open::<RpcStoreSchema>(
150            &db_path,
151            DbOptions {
152                rocksdb: default_rocksdb_config(),
153                ..DbOptions::default()
154            },
155        )
156        .with_context(|| format!("failed to open rpc store at {}", db_path.display()))?;
157        let schema = Arc::new(schema);
158
159        Self::seed_chain_identifier(&db, metadata.chain_identifier)?;
160
161        Ok(Self {
162            db,
163            schema,
164            metadata,
165            indexer_pipelines: Vec::new(),
166            indexer_service: None,
167        })
168    }

170    /// Return the checkpoint an existing fork directory was forked at, or `None` if `root` holds no
171    /// fork yet.
172    ///
173    /// The metadata sidecar is read without opening the store, so startup can decide whether it is
174    /// resuming a fork or creating one before committing to either. A directory that belongs to a
175    /// different network, or to a different checkpoint than the one requested, is an error.
176    pub(crate) fn existing_forked_checkpoint(
177        root: &Path,
178        network: &str,
179        requested_checkpoint: Option<CheckpointSequenceNumber>,
180    ) -> anyhow::Result<Option<CheckpointSequenceNumber>> {
181        let path = Self::metadata_path(root);
182        if !path.exists() {
183            return Ok(None);
184        }
185        let stored: Metadata = serde_json::from_slice(
186            &fs::read(&path)
187                .with_context(|| format!("failed to read fork metadata {}", path.display()))?,
188        )
189        .with_context(|| format!("failed to parse fork metadata {}", path.display()))?;
190        ensure!(
191            stored.network == network,
192            "fork metadata network {} does not match requested network {}. Use a different --data-dir.",
193            stored.network,
194            network,
195        );
196        if let Some(checkpoint) = requested_checkpoint {
197            ensure!(
198                stored.forked_at_checkpoint == checkpoint,
199                "fork metadata checkpoint {} does not match requested checkpoint {}. Use a different --data-dir.",
200                stored.forked_at_checkpoint,
201                checkpoint,
202            );
203        }
204        Ok(Some(stored.forked_at_checkpoint))
205    }

207    /// Start the embedded rpc-store indexer over locally produced checkpoints.
208    ///
209    /// Ingestion reads checkpoints back out of `simulacrum`, starting at the checkpoint after the
210    /// fork point, because everything at or below it is pre-fork state the seed load already
211    /// placed. Registers the pipelines in [`Self::pipeline_layer`] plus the broadcast pipeline that
212    /// feeds RPC subscriptions, and errors if an indexer is already running.
213    pub(crate) async fn start_indexer(
214        &mut self,
215        simulacrum: Arc<RwLock<ForkedSimulacrum>>,
216        checkpoint_sender: broadcast::Sender<Arc<Checkpoint>>,
217        registry: &Registry,
218    ) -> anyhow::Result<()> {
219        ensure!(
220            self.indexer_service.is_none(),
221            "fork rpc-store indexer is already running",
222        );
223
224        let first_checkpoint = self
225            .metadata
226            .forked_at_checkpoint
227            .checked_add(1)
228            .context("forked_at_checkpoint cannot be u64::MAX")?;
229        let ingestion_metrics = IngestionMetrics::new(Some(METRICS_PREFIX), registry);
230        let ingestion_client = IngestionClient::from_trait(
231            Arc::new(SimulacrumIngestion::new(
232                simulacrum,
233                self.chain_identifier(),
234            )),
235            ingestion_metrics,
236        );
237        let store = Store::new(self.db.clone(), self.schema.clone());
238        let mut indexer = Indexer::from_store(
239            store,
240            IndexerArgs {
241                first_checkpoint: Some(first_checkpoint),
242                ..IndexerArgs::default()
243            },
244            ingestion_client,
245            None,
246            sui_rpc_store::ConsistencyConfig::default(),
247            None,
248            IngestionConfig::default(),
249            registry,
250        )
251        .await
252        .context("failed to construct fork rpc-store indexer")?;
253
254        let committer_config = CommitterConfig::default();
255        indexer
256            .add_pipelines(Self::pipeline_layer(), committer_config.clone())
257            .await
258            .context("failed to register fork rpc-store pipelines")?;
259        indexer
260            .add_checkpoint_broadcast(checkpoint_sender, committer_config)
261            .await
262            .context("failed to register fork checkpoint broadcast pipeline")?;
263
264        self.indexer_pipelines = indexer.pipelines().collect();
265        self.indexer_service = Some(tokio::sync::Mutex::new(
266            indexer
267                .run()
268                .await
269                .context("failed to start fork rpc-store indexer")?,
270        ));
271        Ok(())
272    }

274    /// The pipelines the embedded indexer owns.
275    ///
276    /// Only those whose column families the fork does not write itself. The fork is in the same
277    /// position as a fullnode running this indexer beside a perpetual store, in that it already
278    /// holds the raw chain data and serves it directly, so re-deriving it here would write every
279    /// one of those rows twice per checkpoint. It has to be the indexer that stands down rather
280    /// than `LocalStore`, because the indexer's own ingestion source is the fork's store
281    /// ([`crate::ingestion::SimulacrumIngestion`] reads each checkpoint back out of it), so those
282    /// rows must already be there before ingestion can run at all.
283    ///
284    /// Two of the enabled pipelines are load-bearing in ways that are easy to miss.
285    /// `package_versions` is the only writer for a package published after the fork point, since
286    /// `stage_local_object_diff` never stages a package-version row, so disabling it would silently
287    /// lose those rows. `balance` accumulates through a merge operator, so its writers must stay
288    /// disjoint. The fork writes pre-fork balances during the seed load and the indexer writes
289    /// post-fork ones, and any overlap doubles the value rather than being idempotent.
290    fn pipeline_layer() -> PipelineLayer {
291        PipelineLayer {
292            object_by_owner: Some(CommitterLayer::default()),
293            object_by_type: Some(CommitterLayer::default()),
294            balance: Some(CommitterLayer::default()),
295            package_versions: Some(CommitterLayer::default()),
296            transaction_bitmap: Some(CommitterLayer::default()),
297            event_bitmap: Some(CommitterLayer::default()),
298            epochs: Some(CommitterLayer::default()),
299            // Left off deliberately: `checkpoint_summary`,
300            // `checkpoint_contents`, `checkpoint_seq_by_digest`,
301            // `transactions`, `effects`, `events`, `tx_seq_by_digest`,
302            // `tx_metadata_by_seq`, `objects`, and
303            // `object_version_by_checkpoint` are all written synchronously by
304            // `LocalStore`.
305            ..PipelineLayer::default()
306        }
307    }

326    /// Resolve when the embedded indexer stops, with `Ok(())` if all its tasks completed
327    /// (unexpected while the fork is serving), or the task error if any pipeline failed or
328    /// panicked. Pends forever if the indexer was never started.
329    pub(crate) async fn indexer_stopped(&self) -> anyhow::Result<()> {
330        let Some(service) = &self.indexer_service else {
331            return std::future::pending().await;
332        };
333        service.lock().await.join().await
334    }

336    /// Wait until the indexer has committed `checkpoint` on every pipeline it owns, so a caller
337    /// that just produced a checkpoint can rely on the derived indexes reflecting it.
338    ///
339    /// Polls rather than subscribing, and gives up after [`INDEXED_CHECKPOINT_TIMEOUT`] so a
340    /// stalled pipeline surfaces as an error instead of hanging the caller forever.
341    pub(crate) async fn wait_for_indexed_checkpoint(
342        &self,
343        checkpoint: CheckpointSequenceNumber,
344    ) -> anyhow::Result<()> {
345        let deadline = Instant::now() + INDEXED_CHECKPOINT_TIMEOUT;
346        loop {
347            if self
348                .highest_indexed_checkpoint()?
349                .is_some_and(|indexed| indexed >= checkpoint)
350            {
351                return Ok(());
352            }
353
354            ensure!(
355                Instant::now() < deadline,
356                "timed out waiting for rpc-store to index checkpoint {checkpoint}",
357            );
358            tokio::time::sleep(INDEXED_CHECKPOINT_POLL_INTERVAL).await;
359        }
360    }

362    /// Return the highest checkpoint every indexer pipeline has committed, which is the lowest of
363    /// their watermarks, since a checkpoint is only fully indexed once the slowest pipeline has it.
364    ///
365    /// Returns `None` when the value is not yet meaningful, either because no indexer is running or
366    /// because a pipeline has not written a watermark at all.
367    fn highest_indexed_checkpoint(&self) -> anyhow::Result<Option<CheckpointSequenceNumber>> {
368        if self.indexer_pipelines.is_empty() {
369            return Ok(None);
370        }
371
372        let framework = FrameworkSchema::new(self.db.clone());
373        let mut indexed: Option<CheckpointSequenceNumber> = None;
374        for pipeline in &self.indexer_pipelines {
375            let Some(watermark) = framework
376                .watermarks
377                .get(&PipelineTaskKey::new(*pipeline))
378                .with_context(|| format!("failed to read {pipeline} watermark"))?
379            else {
380                return Ok(None);
381            };
382            indexed = Some(
383                indexed.map_or(watermark.checkpoint_hi_inclusive, |checkpoint| {
384                    checkpoint.min(watermark.checkpoint_hi_inclusive)
385                }),
386            );
387        }
388        Ok(indexed)
389    }

406    /// Establish that `root` belongs to the fork described by `expected`, writing the metadata if
407    /// the directory is new and rejecting it if what is already there disagrees.
408    fn load_or_write_metadata(root: &Path, expected: &Metadata) -> anyhow::Result<()> {
409        let path = Self::metadata_path(root);
410        if path.exists() {
411            let stored: Metadata = serde_json::from_slice(
412                &fs::read(&path)
413                    .with_context(|| format!("failed to read fork metadata {}", path.display()))?,
414            )
415            .with_context(|| format!("failed to parse fork metadata {}", path.display()))?;
416            ensure!(
417                stored == *expected,
418                "fork metadata at {} does not match requested fork. Use a different --data-dir.",
419                path.display(),
420            );
421            return Ok(());
422        }
423
424        // Temp-file + rename, like the other metadata sidecars: a crash
425        // mid-write must not leave a truncated fork_metadata.json, which
426        // open() would reject as a mismatched fork on every later launch.
427        crate::metadata::write_json_exclusive(&path, expected, "fork metadata")
428    }

430    /// Record the live network's chain identifier in the rpc-store, under the fork's own pipeline
431    /// key.
432    ///
433    /// The store pins each pipeline to the chain it first ingested and refuses checkpoints from
434    /// another one, so every identifier already present must equal `expected`. One that disagrees
435    /// means this directory holds a different chain's data, whatever the metadata sidecar says.
436    fn seed_chain_identifier(db: &Db, expected: [u8; 32]) -> anyhow::Result<()> {
437        let expected = ChainId(expected);
438        let framework = FrameworkSchema::new(db.clone());
439        for entry in framework
440            .chain_ids
441            .iter(..)
442            .context("failed to iterate rpc store chain identifiers")?
443        {
444            let (_, chain_id) = entry.context("failed to read rpc store chain identifier")?;
445            ensure!(
446                chain_id == expected,
447                "rpc store chain identifier does not match fork metadata. Use a different --data-dir.",
448            );
449        }
450
451        let mut batch = db.batch();
452        batch
453            .put(
454                &framework.chain_ids,
455                &PipelineTaskKey::new(FORK_CHAIN_ID_PIPELINE),
456                &expected,
457            )
458            .context("failed to stage rpc store chain identifier")?;
459        batch
460            .commit()
461            .context("failed to commit rpc store chain identifier")
462    }

471    /// The indexer must own exactly the column families `LocalStore` does not write, and no others.
472    ///
473    /// Both directions bite silently. Enabling one the fork already writes duplicates every row of
474    /// it per checkpoint, and for `balance`, which accumulates through a merge operator,
475    /// duplication would double the value rather than being idempotent. Disabling one the fork does
476    /// not write loses those rows outright. `package_versions` is the only writer for a package
477    /// published after the fork point, since `stage_local_object_diff` never stages a
478    /// package-version row.
479    ///
480    /// Neither failure shows up as a test error elsewhere, so the split is pinned here.
481    #[test]
482    fn indexer_owns_only_the_pipelines_the_fork_does_not_write() {
483        let layer = ServiceManager::pipeline_layer();
484
485        for (name, enabled) in [
486            ("object_by_owner", layer.object_by_owner.is_some()),
487            ("object_by_type", layer.object_by_type.is_some()),
488            ("balance", layer.balance.is_some()),
489            ("package_versions", layer.package_versions.is_some()),
490            ("transaction_bitmap", layer.transaction_bitmap.is_some()),
491            ("event_bitmap", layer.event_bitmap.is_some()),
492            ("epochs", layer.epochs.is_some()),
493        ] {
494            assert!(enabled, "{name} has no other writer and must stay enabled");
495            assert!(
496                FORK_INDEXER_PIPELINES.contains(&name),
497                "{name} is registered but missing from FORK_INDEXER_PIPELINES, so nothing \
498                 seeds its watermark and the reported indexed tip stays None",
499            );
500        }
501
502        assert_eq!(
503            FORK_INDEXER_PIPELINES.len(),
504            7,
505            "FORK_INDEXER_PIPELINES lists a pipeline the layer does not register; its \
506             watermark would never advance and would pin the reported tip",
507        );
508
509        for (name, enabled) in [
510            ("checkpoint_summary", layer.checkpoint_summary.is_some()),
511            ("checkpoint_contents", layer.checkpoint_contents.is_some()),
512            (
513                "checkpoint_seq_by_digest",
514                layer.checkpoint_seq_by_digest.is_some(),
515            ),
516            ("transactions", layer.transactions.is_some()),
517            ("effects", layer.effects.is_some()),
518            ("events", layer.events.is_some()),
519            ("tx_seq_by_digest", layer.tx_seq_by_digest.is_some()),
520            ("tx_metadata_by_seq", layer.tx_metadata_by_seq.is_some()),
521            ("objects", layer.objects.is_some()),
522            (
523                "object_version_by_checkpoint",
524                layer.object_version_by_checkpoint.is_some(),
525            ),
526        ] {
527            assert!(
528                !enabled,
529                "{name} is written by LocalStore; enabling it here double-writes every row"
530            );
531        }
532    }
