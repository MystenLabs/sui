49/// Checkpoint selected for startup, plus whether existing local fork state was found.
50#[derive(Clone, Copy, Debug, Eq, PartialEq)]
51pub(crate) struct ResolvedStartCheckpoint {
52    pub(crate) checkpoint: Option<CheckpointSequenceNumber>,
53    pub(crate) resuming: bool,
54}

56/// Resolve the fork checkpoint from local state when possible.
57///
58/// Explicit data directories can be inspected directly before the remote GraphQL client asks for
59/// latest. Default roots cannot be inspected without a checkpoint because their path contains the
60/// checkpoint.
61pub(crate) fn resolve_start_checkpoint_from_local(
62    node: &Node,
63    requested_checkpoint: Option<CheckpointSequenceNumber>,
64    data_dir: Option<&Path>,
65) -> Result<ResolvedStartCheckpoint> {
66    let local = match (data_dir, requested_checkpoint) {
67        (Some(data_dir), _) => Some(MetadataStore::new_with_root(data_dir.to_path_buf())),
68        (None, Some(checkpoint)) => Some(MetadataStore::new(node, checkpoint, None)?),
69        (None, None) => None,
70    };
71
72    let Some(local) = local else {
73        return Ok(ResolvedStartCheckpoint {
74            checkpoint: requested_checkpoint,
75            resuming: false,
76        });
77    };
78
79    if local.seed_manifest_exists() {
80        let manifest = local.read_seed_manifest()?;
81        ensure_seed_manifest_matches(&manifest, &node.network_name(), requested_checkpoint)?;
82        return Ok(ResolvedStartCheckpoint {
83            checkpoint: Some(manifest.checkpoint),
84            resuming: true,
85        });
86    }
87
88    if let Some(checkpoint) = ServiceManager::existing_forked_checkpoint(
89        local.root(),
90        &node.network_name(),
91        requested_checkpoint,
92    )? {
93        return Ok(ResolvedStartCheckpoint {
94            checkpoint: Some(checkpoint),
95            resuming: true,
96        });
97    }
98
99    Ok(ResolvedStartCheckpoint {
100        checkpoint: requested_checkpoint,
101        resuming: false,
102    })
103}

105/// Initialize a forked network by fetching the fork checkpoint from the remote endpoint when
106/// needed, applying seed metadata, and starting a local Simulacrum instance from the highest
107/// checkpoint already persisted locally. Also builds the checkpoint subscription broker, whose
108/// returned handle must be passed to [`run`] so the gRPC server exposes the streaming RPC.
109///
110/// `data_dir` is the root folder where the fork state is persisted. If `None`, a default path is
111/// used. See the `[MetadataStore]` docs for details.
112pub async fn initialize(
113    node: Node,
114    forked_at_checkpoint: CheckpointSequenceNumber,
115    version: &str,
116    data_dir: Option<PathBuf>,
117    seed_input: SeedInput,
118) -> Result<(Context, SubscriptionServiceHandle)> {
119    // 1. Prepare metadata and GraphQL, then open the RPC store before constructing ForkStore.
120    let gql = GraphQLClient::new(node.clone(), version)?;
121    let chain_identifier = gql.chain();
122    let local = MetadataStore::new(&node, forked_at_checkpoint, data_dir)?;
123    let network_name = node.network_name();
124    crate::seed::ensure_seed_policy(&local, &seed_input)?;
125
126    // 2. Fetch the startup checkpoint, open the RPC store using its chain identity,
127    //    then construct the ForkStore with the RPC store already attached.
128    let (checkpoint, checkpoint_contents) = gql
129        .get_checkpoint(Some(forked_at_checkpoint))?
130        .ok_or_else(|| anyhow!("checkpoint {} not found", forked_at_checkpoint))?;
131    let rpc_chain_identifier = fork_chain_identifier(chain_identifier, &checkpoint);
132    let services = ServiceManager::open(
133        local.root(),
134        network_name.clone(),
135        forked_at_checkpoint,
136        rpc_chain_identifier,
137    )?;
138    let store = ForkStore::from_parts(forked_at_checkpoint, gql, local, services.local_store());
139    store.save_checkpoint(&checkpoint, &checkpoint_contents)?;
140    let seed_manifest =
141        crate::seed::prepare_seed_manifest(&store, network_name, &seed_input).await?;
142
143    // Seeding is eager and one-shot: the enumerations behind the manifest are
144    // pinned at the fork checkpoint and cannot be re-run once the fork has
145    // diverged, so the whole seed set is loaded here, before anything executes,
146    // and never again. On a resumed fork the load is a no-op.
147    load_seed_objects(&store, &seed_manifest)?;
148
149    // Resume support: the Simulacrum must build its next checkpoint on the
150    // fork's own local tip. On a fresh fork that tip is the fork-point
151    // checkpoint persisted just above; on a resumed fork that has advanced,
152    // it is the highest locally sealed checkpoint — re-seeding from the fork
153    // point would rebuild an already-persisted sequence number with different
154    // contents and fail the seal.
155    let base_checkpoint = resume_base_checkpoint(&store)?;
156
157    // 3. Read system state — fetches object 0x5 + dynamic fields from remote via ObjectStore.
158    let system_state = sui_types::sui_system_state::get_sui_system_state(&store).map_err(|e| {
159        anyhow!(
160            "failed to read system state at checkpoint {}: {}",
161            forked_at_checkpoint,
162            e
163        )
164    })?;
165
166    let protocol_version = system_state.protocol_version();
167
168    // 4. Build NetworkConfig with local test validators.
169    let mut rng = OsRng;
170    let config = ConfigBuilder::new_with_temp_dir()
171        .rng(&mut rng)
172        .with_chain_start_timestamp_ms(base_checkpoint.timestamp_ms)
173        .deterministic_committee_size(NonZeroUsize::MIN)
174        .with_protocol_version(ProtocolVersion::new(protocol_version))
175        .with_chain_override(node.chain())
176        .build();
177
178    // 5. Override validators in system state with local keys from config.
179    let system_state = override_validators(system_state, &config)?;
180
181    // 6. Build KeyStore.
182    let keystore = KeyStore::from_network_config(&config);
183
184    // 7. Create Simulacrum from custom state.
185    let simulacrum = Simulacrum::new_from_custom_state(
186        keystore,
187        base_checkpoint,
188        system_state,
189        rpc_chain_identifier,
190        &config,
191        store,
192        rng,
193    );
194
195    // 8. Build the checkpoint subscription broker. The sender is owned by
196    //    `Context` (producers push here on `advance_checkpoint`); the handle
197    //    is wired into `RpcService` in `run` so subscribers can register.
198    let registry = Registry::new();
199    let (checkpoint_sender, subscription_handle) =
200        SubscriptionService::build(&registry, None, None, None, None);
201
202    let context = Context::new(simulacrum, services, checkpoint_sender, &registry).await?;
203
204    Ok((context, subscription_handle))
205}

207fn fork_chain_identifier(chain: Chain, checkpoint: &VerifiedCheckpoint) -> ChainIdentifier {
208    match chain {
209        Chain::Mainnet => get_mainnet_chain_identifier(),
210        Chain::Testnet => get_testnet_chain_identifier(),
211        Chain::Unknown => ChainIdentifier::from(*checkpoint.digest()),
212    }
213}

215/// Return the checkpoint the Simulacrum should build on, which is the highest locally persisted
216/// checkpoint. On a fresh fork this is the fork-point checkpoint, persisted before this is called.
217/// On a resumed fork it is the local tip, so the next sealed checkpoint gets the correct sequence
218/// number and previous-digest chain instead of colliding with an already-persisted one.
219pub(crate) fn resume_base_checkpoint(store: &ForkStore) -> Result<VerifiedCheckpoint> {
220    store
221        .get_highest_verified_checkpoint()?
222        .ok_or_else(|| anyhow!("no local checkpoint available to resume from"))
223}

225/// Run the forked network. Spawns the `sui-rpc-api` `RpcService` bound to `rpc_addr`, backed by the
226/// `ForkStore`'s RPC trait impls, then blocks on Ctrl+C.
227pub async fn run(
228    context: Context,
229    subscription_handle: SubscriptionServiceHandle,
230    rpc_addr: SocketAddr,
231    version: &'static str,
232) -> Result<()> {
233    let store = {
234        let sim = context.simulacrum().read().await;
235        sim.store().clone()
236    };
237    let reader: Arc<dyn RpcStateReader> = Arc::new(store);
238
239    // Serve through `sui-rpc-api`'s `RpcService` directly (the fork does not
240    // depend on `sui-rpc-node`). The `ForkStore` itself is the
241    // `RpcStateReader`, with the fork admin service and executor attached.
242    let mut service = RpcService::new(reader);
243    service.with_server_version(ServerVersion::new("sui-fork", version));
244    service.with_subscription_service(subscription_handle);
245    let context = Arc::new(context);
246    service.with_executor(Arc::new(ForkedTransactionExecutor::new(context.clone())));
247    service.with_custom_service(ForkingServiceServer::new(ForkingServiceImpl::new(
248        context.clone(),
249    )));
250    service.with_file_descriptor_set(crate::proto::FILE_DESCRIPTOR_SET);
251
252    info!("starting sui-rpc-api server on {rpc_addr}");
253    let server_handle = tokio::spawn(async move { service.start_service(rpc_addr).await });
254
255    info!("forked network running, waiting for shutdown signal (Ctrl+C)");
256    tokio::select! {
257        res = tokio::signal::ctrl_c() => {
258            res?;
259            info!("shutdown signal received, stopping forked network");
260        }
261        join = server_handle => {
262            if let Err(e) = join {
263                return Err(anyhow!("rpc server task panicked: {e}"));
264            }
265            return Err(anyhow!("rpc server task exited unexpectedly"));
266        }
267        stopped = context.indexer_stopped() => {
268            // Without this watchdog an indexer failure would only surface as
269            // a 30s publication timeout on the next executed transaction.
270            return match stopped {
271                Ok(()) => Err(anyhow!("embedded rpc-store indexer stopped unexpectedly")),
272                Err(e) => Err(e.context("embedded rpc-store indexer failed")),
273            };
274        }
275    }
276    Ok(())
277}

279/// Replace the validator set in the system state with local validators from the NetworkConfig
280/// genesis, so the simulacrum can sign checkpoints with locally available keys.
281fn override_validators(
282    system_state: SuiSystemState,
283    config: &NetworkConfig,
284) -> Result<SuiSystemState> {
285    let genesis_validators = match config.genesis.sui_system_object() {
286        SuiSystemState::V1(inner) => inner.validators,
287        SuiSystemState::V2(inner) => inner.validators,
288        #[cfg(msim)]
289        _ => anyhow::bail!("unsupported genesis system state variant"),
290    };
291
292    match system_state {
293        SuiSystemState::V1(mut inner) => {
294            inner.validators = genesis_validators;
295            Ok(SuiSystemState::V1(inner))
296        }
297        SuiSystemState::V2(mut inner) => {
298            inner.validators = genesis_validators;
299            Ok(SuiSystemState::V2(inner))
300        }
301        #[cfg(msim)]
302        _ => anyhow::bail!("unsupported system state variant"),
303    }
304}

314    fn write_manifest(root: &Path, network: &str, checkpoint: CheckpointSequenceNumber) {
315        MetadataStore::new_with_root(root.to_path_buf())
316            .write_seed_manifest(&SeedManifest {
317                network: network.to_owned(),
318                checkpoint,
319                addresses: Vec::new(),
320                entries: Vec::new(),
321            })
322            .expect("seed manifest should write");
323    }

325    #[test]
326    fn resolve_start_checkpoint_uses_manifest_when_checkpoint_is_omitted() {
327        let temp = tempfile::tempdir().expect("tempdir");
328        write_manifest(temp.path(), "mainnet", 42);
329
330        let resolved = resolve_start_checkpoint_from_local(&Node::Mainnet, None, Some(temp.path()))
331            .expect("checkpoint resolution should not fail");
332
333        assert_eq!(
334            resolved,
335            ResolvedStartCheckpoint {
336                checkpoint: Some(42),
337                resuming: true,
338            }
339        );
340    }
