4//! Locate the `sui` and `sui-fork` binaries by building them with cargo.
5//!
6//! Cargo exports `CARGO_BIN_EXE_<name>` only to integration tests of the package that owns the
7//! binary, and a workspace test run never links the `sui-fork` binary at all because that package
8//! has no integration tests. Guessing `target/<profile>/<name>` instead would run whatever was
9//! built there last, which silently tests stale code. Building on demand costs one cargo
10//! invocation per test process, which is a fingerprint check once the binaries are current, and
11//! cargo's build directory lock serialises the test processes that nextest runs in parallel.
12
13use std::ffi::OsStr;
14use std::ffi::OsString;
15use std::path::Path;
16use std::path::PathBuf;
17use std::process::Command;
18use std::process::Stdio;
19use std::sync::OnceLock;
20
21use anyhow::Context;
22use anyhow::Result;
23use anyhow::anyhow;
24use anyhow::ensure;
25use serde_json::Value;
26
27/// Paths of the freshly built `sui` and `sui-fork` executables.
28#[derive(Clone, Debug)]
29pub struct Binaries {
30    pub sui: PathBuf,
31    pub sui_fork: PathBuf,
32}
33
34impl Binaries {
35    /// Build both binaries in the running test's profile and cache the result for this process.
36    pub fn build() -> Result<&'static Self> {
37        static BINARIES: OnceLock<Result<Binaries, String>> = OnceLock::new();
38        BINARIES
39            .get_or_init(|| build_binaries().map_err(|error| format!("{error:#}")))
40            .as_ref()
41            .map_err(|error| anyhow!("{error}"))
42    }
43
44    /// Return `path` with the directories holding `sui` and `sui-fork` prepended.
45    pub fn path_with(&self, path: &OsStr) -> Result<OsString> {
46        let mut dirs = vec![
47            self.sui
48                .parent()
49                .context("sui has no parent dir")?
50                .to_path_buf(),
51            self.sui_fork
52                .parent()
53                .context("sui-fork has no parent dir")?
54                .to_path_buf(),
55        ];
56        dirs.dedup();
57        dirs.extend(std::env::split_paths(path));
58        std::env::join_paths(dirs).context("PATH entry contains a separator")
59    }
60}
61
62/// Run `cargo build` for both binaries and read their paths from the JSON artifact messages.
63///
64/// Reading `executable` from the messages rather than composing `target/<profile>/<name>` keeps
65/// the harness correct under a custom `CARGO_TARGET_DIR` or profile, because cargo reports where
66/// it actually put the file. Fresh artifacts are reported too, so a no-op build still resolves
67/// both paths.
68fn build_binaries() -> Result<Binaries> {
69    let cargo = std::env::var_os("CARGO").unwrap_or_else(|| env!("CARGO").into());
70    let workspace = Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
71    let output = Command::new(cargo)
72        .current_dir(&workspace)
73        .args(["build", "--message-format=json-render-diagnostics"])
74        .args(["--package", "sui", "--bin", "sui"])
75        .args(["--package", "sui-fork", "--bin", "sui-fork"])
76        .args(profile_flags()?)
77        .stdin(Stdio::null())
78        .stderr(Stdio::inherit())
79        .output()
80        .context("failed to run cargo build for the sui and sui-fork binaries")?;
81    ensure!(
82        output.status.success(),
83        "cargo build of the sui and sui-fork binaries failed with {}",
84        output.status
85    );
86
87    let mut sui = None;
88    let mut sui_fork = None;
89    for line in String::from_utf8_lossy(&output.stdout).lines() {
90        let Ok(message) = serde_json::from_str::<Value>(line) else {
91            continue;
92        };
93        if message["reason"] != "compiler-artifact" {
94            continue;
95        }
96        let Some(executable) = message["executable"].as_str() else {
97            continue;
98        };
99        let is_bin = message["target"]["kind"]
100            .as_array()
101            .is_some_and(|kinds| kinds.iter().any(|kind| kind.as_str() == Some("bin")));
102        if !is_bin {
103            continue;
104        }
105        match message["target"]["name"].as_str() {
106            Some("sui") => sui = Some(PathBuf::from(executable)),
107            Some("sui-fork") => sui_fork = Some(PathBuf::from(executable)),
108            _ => {}
109        }
110    }
111
112    Ok(Binaries {
113        sui: sui.context("cargo build did not report the sui executable")?,
114        sui_fork: sui_fork.context("cargo build did not report the sui-fork executable")?,
115    })
116}

118/// Derive cargo profile flags from where the test executable lives.
119///
120/// Test executables sit at `<target>/<profile dir>/deps/<name>`. The profile directory is `debug`
121/// for the dev and test profiles, `release` for release and bench, and the profile's own name
122/// otherwise, so building with the matching flag puts the binaries next to the test and at the
123/// same optimisation level.
124fn profile_flags() -> Result<Vec<String>> {
125    let exe = std::env::current_exe().context("cannot locate the test executable")?;
126    let profile_dir = exe
127        .parent()
128        .and_then(Path::parent)
129        .and_then(Path::file_name)
130        .and_then(OsStr::to_str)
131        .context("test executable is not under <target>/<profile>/deps")?;
132    Ok(match profile_dir {
133        "debug" => vec![],
134        "release" => vec!["--release".to_owned()],
135        custom => vec!["--profile".to_owned(), custom.to_owned()],
136    })
137}
