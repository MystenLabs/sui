4//! A local Sui network with the GraphQL stack a fork reads from.
5//!
6//! `sui-fork` resolves everything it does not hold locally through GraphQL pinned at the fork
7//! checkpoint, so a source network needs a fullnode, an indexer writing to PostgreSQL, a
8//! consistent store for the owned-object and balance enumerations that seeding relies on, and a
9//! GraphQL server over both. That is what `sui start --with-graphql` assembles, so
10//! [`SourceNetwork`] spawns the `sui` binary under test with that command on ports the harness
11//! allocates. The shell harness creates a localnet-only client config for that RPC URL, and
12//! `localnet_setup` in `tests/shell_lib/lib.sh` copies it and funds a new address through the same
13//! binary.
14//!
15//! Two habits of `--force-regenesis` are sandboxed. It keeps its config directory under the
16//! system temp dir without printing or removing it, so the child runs with `TMPDIR` inside the
17//! network's own temp dir, which goes away with the network. And it rewrites the localnet RPC
18//! URL of whatever `client.yaml` sits in `SUI_CONFIG_DIR`, so the child gets an empty directory
19//! there rather than the developer's `~/.sui/sui_config`.
20

40/// Epoch length of the source network, long enough that no test straddles an epoch change.
41const EPOCH_DURATION_MS: u64 = 60 * 60 * 1_000;
42
43/// Longest wait for `sui start` to exit after SIGINT before its process group is killed. A minute,
44/// because a group kill cannot reach the PostgreSQL that `sui start` runs in a session of its own,
45/// and with a dozen localnets shutting down at once on one machine twenty seconds proved too short.
46const STOP_GRACE: Duration = Duration::from_secs(60);
47
48/// Time between the SIGINTs sent to `sui start` while it is still running.
49const INTERRUPT_INTERVAL: Duration = Duration::from_secs(2);
50
51/// Interval between checks for the child's exit.
52const POLL: Duration = Duration::from_millis(200);
53
54/// Lines of `sui start` output quoted in errors.
55const LOG_TAIL_LINES: usize = 40;
56
57/// A running `sui start --force-regenesis --with-graphql --with-faucet`.
58///
59/// [`Self::stop`] interrupts the child and waits for it, so PostgreSQL and the node databases are
60/// shut down before the directory holding them is removed. Dropping a network that is still
61/// running kills its process group instead, which covers every early return in the harness.
62pub struct SourceNetwork {
63    child: Child,
64    group: u32,
65    rpc_url: String,
66    graphql_url: String,
67    faucet_url: String,
68    log_path: PathBuf,
69    _dir: TempDir,
70}
71
72impl SourceNetwork {
73    /// Spawn `sui start` from the `sui` binary at `sui` and return without waiting for it, since
74    /// the scripts poll the faucet until the network answers.
75    pub fn start(sui: &Path) -> Result<Self> {
76        let dir = tempfile::tempdir().context("failed to create the source network directory")?;
77        let tmp_dir = dir.path().join("tmp");
78        let sui_config_dir = dir.path().join("sui_config");
79        for subdir in [&tmp_dir, &sui_config_dir] {
80            std::fs::create_dir(subdir)
81                .with_context(|| format!("failed to create {}", subdir.display()))?;
82        }
83        let log_path = dir.path().join("sui-start.log");
84        let log = File::create(&log_path).context("failed to create the sui start log")?;
85
86        let rpc_port = free_loopback_port();
87        let graphql_port = free_loopback_port();
88        let consistent_store_port = free_loopback_port();
89        let faucet_port = free_loopback_port();
90
91        // The `--with-*` flags take `=` and an explicit host, because a bare port binds 0.0.0.0.
92        let mut command = Command::new(sui);
93        command
94            .arg("start")
95            .arg("--force-regenesis")
96            .args(["--epoch-duration-ms", &EPOCH_DURATION_MS.to_string()])
97            .args(["--fullnode-rpc-port", &rpc_port.to_string()])
98            .arg(format!("--with-graphql=127.0.0.1:{graphql_port}"))
99            .arg(format!(
100                "--with-consistent-store=127.0.0.1:{consistent_store_port}"
101            ))
102            .arg(format!("--with-faucet=127.0.0.1:{faucet_port}"))
103            .env("TMPDIR", &tmp_dir)
104            .env("SUI_CONFIG_DIR", &sui_config_dir)
105            // The binary logs at `error` by default. The `sui` target adds the lines that name
106            // each service as it comes up, which is what a failed start needs to be diagnosed.
107            .env("RUST_LOG", "error,sui=info")
108            .env("RUST_BACKTRACE", "0")
109            .stdin(Stdio::null())
110            .stdout(
111                log.try_clone()
112                    .context("failed to clone the sui start log handle")?,
113            )
114            .stderr(log);
115        let child = spawn_in_group(&mut command).context("failed to spawn sui start")?;
116        let group = child.id();
117
118        Ok(Self {
119            child,
120            group,
121            rpc_url: format!("http://127.0.0.1:{rpc_port}"),
122            graphql_url: format!("http://127.0.0.1:{graphql_port}/graphql"),
123            faucet_url: format!("http://127.0.0.1:{faucet_port}/v2/gas"),
124            log_path,
125            _dir: dir,
126        })
127    }
128
129    /// Return the fullnode RPC URL, which scripts register as their `localnet` env.
130    pub fn rpc_url(&self) -> &str {
131        &self.rpc_url
132    }
133
134    /// Return the GraphQL endpoint, which is what a fork passes as `--network`.
135    pub fn graphql_url(&self) -> &str {
136        &self.graphql_url
137    }
138
139    /// Return the faucet endpoint scripts fund their address from.
140    pub fn faucet_url(&self) -> &str {
141        &self.faucet_url
142    }
143
144    /// Interrupt `sui start` and wait for it to exit, so its services shut down in order and
145    /// PostgreSQL is stopped by the process that started it.
146    ///
147    /// Fails if the child had already exited, because a script then ran against a dead network.
148    pub async fn stop(mut self) -> Result<()> {
149        if let Some(status) = self.try_wait()? {
150            release_group(self.group);
151            return Err(anyhow!(
152                "sui start exited with {status} during the test:\n{}",
153                self.log_tail()
154            ));
155        }
156
157        // `sui start` listens for SIGINT only between the validator health checks of its main
158        // loop, and a signal that lands during a check is lost, so the interrupt is repeated until
159        // the child exits. Once it is shutting down, the extra signals are ignored.
160        let deadline = Instant::now() + STOP_GRACE;
161        let mut next_interrupt = Instant::now();
162        let status = loop {
163            if let Some(status) = self.try_wait()? {
164                break Some(status);
165            }
166            let now = Instant::now();
167            if now >= deadline {
168                break None;
169            }
170            if now >= next_interrupt {
171                signal_process(self.child.id(), "INT");
172                next_interrupt = now + INTERRUPT_INTERVAL;
173            }
174            tokio::time::sleep(POLL).await;
175        };
176        // After a clean exit the group holds at most a PostgreSQL the child failed to stop; after
177        // a timeout it still holds the child.
178        release_group(self.group);
179        match status {
180            Some(status) if status.success() => Ok(()),
181            Some(status) => Err(anyhow!(
182                "sui start exited with {status} on SIGINT:\n{}",
183                self.log_tail()
184            )),
185            None => Err(anyhow!(
186                "sui start did not exit within {STOP_GRACE:?} of SIGINT and was killed:\n{}",
187                self.log_tail()
188            )),
189        }
190    }
191
192    fn try_wait(&mut self) -> Result<Option<std::process::ExitStatus>> {
193        self.child
194            .try_wait()
195            .context("failed to wait for sui start")
196    }
197
198    /// The last lines of the child's output, for error messages.
199    fn log_tail(&self) -> String {
200        let log = std::fs::read_to_string(&self.log_path).unwrap_or_default();
201        let lines: Vec<&str> = log.lines().collect();
202        lines[lines.len().saturating_sub(LOG_TAIL_LINES)..].join("\n")
203    }
204}
205
206impl Drop for SourceNetwork {
207    /// Kill the process group of a network that was never stopped, before its directory goes.
208    fn drop(&mut self) {
209        if matches!(self.child.try_wait(), Ok(None)) {
210            release_group(self.group);
211        }
212    }
213}

215/// Reserve a free loopback port the way `sui start` picks its own internal ports.
216pub fn free_loopback_port() -> u16 {
217    get_available_port("127.0.0.1")
