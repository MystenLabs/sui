4//! The minimum batch a collector waits for before it cuts its pending rows into a batch (see
5//! [`CommitterConfig::min_batch_rows`] and [`CommitterConfig::max_batch_delay_ms`]).
6
7use std::future;
8use std::time::Duration;
9
10use tokio::time::Instant;
11use tokio::time::sleep_until;
12
13use crate::pipeline::CommitterConfig;
14
15/// Tracks whether a collector's pending rows have reached the minimum batch: at least
16/// `min_batch_rows` rows are pending, or rows have been pending for `max_batch_delay`.
17pub(super) struct BatchFloor {
18    min_rows: usize,
19    max_delay: Duration,
20
21    /// When rows started to be pending, or `None` while nothing is pending. Cutting a batch that
22    /// leaves rows behind does not move it, so no row waits longer than `max_delay`.
23    pending_since: Option<Instant>,
24}
25
26impl BatchFloor {
27    pub(super) fn new(config: &CommitterConfig) -> Self {
28        Self {
29            min_rows: config.min_batch_rows,
30            max_delay: config.max_batch_delay(),
31            pending_since: None,
32        }
33    }
34
35    /// Record whether anything is pending, after the collector's pending set changed.
36    pub(super) fn set_pending(&mut self, anything_pending: bool) {
37        if !anything_pending {
38            self.pending_since = None;
39        } else if self.pending_since.is_none() {
40            self.pending_since = Some(Instant::now());
41        }
42    }
43
44    /// Whether the pending rows, `pending_rows` of them, may be cut into a batch now.
45    pub(super) fn is_reached(&self, pending_rows: usize) -> bool {
46        let Some(since) = self.pending_since else {
47            return false;
48        };
49
50        pending_rows >= self.min_rows || Instant::now() >= since + self.max_delay
51    }
52
53    /// Resolves once rows have been pending for `max_delay`. Never resolves while nothing is
54    /// pending.
55    pub(super) async fn delay_elapsed(&self) {
56        match self.pending_since {
57            Some(since) => sleep_until(since + self.max_delay).await,
58            None => future::pending().await,
59        }
60    }
61}
