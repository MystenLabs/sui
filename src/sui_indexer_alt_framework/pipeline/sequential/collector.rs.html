22/// A fully-assembled batch handed off from the collector to the committer task.
23///
24/// Batches are produced strictly in checkpoint order by the collector and consumed in the
25/// same order by the committer, so watermarks advance monotonically even with pipelining.
26pub(super) struct BatchedRows<H: Handler> {
27    pub batch: H::Batch,
28    pub watermark: CommitterWatermark,
29    pub batch_rows: usize,
30}
31
32/// Collector task — drains `rx` into a reorder buffer, assembles batches in checkpoint order,
33/// and hands each ready batch off to the committer via `committer_tx`.
34///
35/// Data arrives out of order, grouped by checkpoint, on `rx`. The collector orders them, and
36/// whenever the next expected checkpoint has arrived and the committer has room for another batch
37/// (`committer_tx` has a free slot), it cuts a batch from the contiguous run of checkpoints
38/// starting there. While the committer is busy, checkpoints keep accumulating, so batches grow
39/// under back-pressure, up to `max_batch_checkpoints` or until the reorder buffer is full (see
40/// below). If `config.committer.min_batch_rows` is set, a batch is only cut once that many rows
41/// are pending (counting rows buffered for checkpoints that arrived ahead of their predecessors),
42/// once rows have been pending for `config.committer.max_batch_delay()`, or once the reorder
43/// buffer is full. When the processor closes its channel, every checkpoint that can still be
44/// written in order is written out.
45///
46/// Backpressure: when `committer_tx` is full (bounded by `pipeline_depth`), the collector stops
47/// cutting batches, and checkpoints accumulate in the reorder buffer. Once `max_pending_rows` are
48/// pending and the next checkpoint has arrived, the buffer is full: the collector stops receiving
49/// from `rx` until it cuts a batch, so the processor→collector channel fills, the processor
50/// blocks, and the ingestion service slows down to this pipeline's pace. A full buffer makes a
51/// batch due regardless of `min_batch_rows`, so a `min_batch_rows` above `max_pending_rows` cannot
52/// leave the collector with receive stopped and no batch to cut.
53///
54/// While the next checkpoint is missing, the collector keeps receiving past `max_pending_rows`,
55/// because the missing checkpoint may still be in `rx` and nothing can be written until it
56/// arrives. Once it arrives, receiving stops again until batches bring the buffer below
57/// `max_pending_rows` or use up the contiguous run of checkpoints.
58pub(super) fn collector<H: Handler>(
59    handler: Arc<H>,
60    config: SequentialConfig,
61    mut next_checkpoint: u64,
62    mut rx: mpsc::Receiver<IndexedCheckpoint<H>>,
63    metrics: Arc<IndexerMetrics>,
64    max_pending_rows: usize,
65    max_batch_checkpoints: usize,
66    committer_tx: mpsc::Sender<BatchedRows<H>>,
67) -> Service {
68    Service::new().spawn_aborting(async move {
69        // The task keeps track of the highest (inclusive) checkpoint it has added to a batch
70        // through `next_checkpoint`. By extension it also knows the next checkpoint to expect.
71
72        // Data for checkpoints that haven't been written yet.
73        let mut pending: BTreeMap<u64, IndexedCheckpoint<H>> = BTreeMap::new();
74        let mut pending_rows: usize = 0;
75        let mut floor = BatchFloor::new(&config.committer);
76
77        info!(pipeline = H::NAME, "Starting collector");
78
79        loop {
80            let ready = has_ready_checkpoint(next_checkpoint, &pending);
81            // Only the next checkpoint counts, not stale duplicates of written ones: while it is
82            // missing, it may still be in `rx`, so receiving must continue.
83            let buffer_full =
84                pending.contains_key(&next_checkpoint) && pending_rows >= max_pending_rows;
85            let batch_due = buffer_full || (ready && floor.is_reached(pending_rows));
86
87            tokio::select! {
88                biased;
89
90                permit = committer_tx.reserve(), if batch_due => {
91                    let Ok(permit) = permit else {
92                        info!(
93                            pipeline = H::NAME,
94                            "Committer task closed; stopping collector"
95                        );
96                        return Ok(());
97                    };
98
99                    if let Some(batched) = gather_batch(
100                        &*handler,
101                        &mut next_checkpoint,
102                        &mut pending,
103                        &mut pending_rows,
104                        max_batch_checkpoints,
105                        &metrics,
106                    ) {
107                        permit.send(batched);
108                    }
109
110                    floor.set_pending(!pending.is_empty());
111                }
112
113                received = rx.recv(), if !buffer_full => {
114                    // The processor has stopped, so write out everything that can be written.
115                    let Some(mut indexed) = received else {
116                        break;
117                    };
118
119                    // Eagerly drain the channel to avoid scheduler ping-pong, until the buffer
120                    // reaches `max_pending_rows`. Whether receiving continues past that is
121                    // decided by `buffer_full` on the next iteration: only while the next
122                    // checkpoint is missing, because it may be among the checkpoints still in
123                    // `rx`.
124                    let mut recv_rows = 0usize;
125                    loop {
126                        recv_rows += indexed.len();
127                        pending_rows += indexed.len();
128                        if let Some(replaced) = pending.insert(indexed.checkpoint(), indexed) {
129                            pending_rows -= replaced.len();
130                        }
131
132                        if pending_rows >= max_pending_rows {
133                            break;
134                        }
135
136                        match rx.try_recv() {
137                            Ok(next) => indexed = next,
138                            Err(_) => break,
139                        }
140                    }
141
142                    // Although there isn't an explicit collector in the sequential pipeline,
143                    // keeping this metric consistent with the concurrent pipeline is useful
144                    // to monitor the backpressure from committer to processor.
145                    metrics
146                        .total_collector_rows_received
147                        .with_label_values(&[H::NAME])
148                        .inc_by(recv_rows as u64);
149
150                    floor.set_pending(true);
151
152                    if pending.len() > WARN_PENDING_WATERMARKS {
153                        warn!(
154                            pipeline = H::NAME,
155                            pending = pending.len(),
156                            "Pipeline has a large number of pending watermarks",
157                        );
158                    }
159                }
160
161                () = floor.delay_elapsed(), if ready && !batch_due => {}
162            }
163        }
164
165        while has_ready_checkpoint(next_checkpoint, &pending) {
166            let Some(batched) = gather_batch(
167                &*handler,
168                &mut next_checkpoint,
169                &mut pending,
170                &mut pending_rows,
171                max_batch_checkpoints,
172                &metrics,
173            ) else {
174                continue;
175            };
176
177            if committer_tx.send(batched).await.is_err() {
178                info!(
179                    pipeline = H::NAME,
180                    "Committer task closed; stopping collector"
181                );
182                return Ok(());
183            }
184        }
185
186        info!(
187            pipeline = H::NAME,
188            "Processor closed channel and no more data to commit"
189        );
190        Ok(())
191    })
192}

194/// Cut the next batch from the contiguous run of checkpoints in `pending` that starts at
195/// `next_checkpoint`, taking at most `max_batch_checkpoints` of them, and discarding stale
196/// duplicates of checkpoints that were already written. Returns `None` if no checkpoint was taken.
197fn gather_batch<H: Handler>(
198    handler: &H,
199    next_checkpoint: &mut u64,
200    pending: &mut BTreeMap<u64, IndexedCheckpoint<H>>,
201    pending_rows: &mut usize,
202    max_batch_checkpoints: usize,
203    metrics: &IndexerMetrics,
204) -> Option<BatchedRows<H>> {
205    let guard = metrics
206        .collector_gather_latency
207        .with_label_values(&[H::NAME])
208        .start_timer();
209
210    let mut batch = H::Batch::default();
211    let mut batch_rows = 0;
212    let mut batch_checkpoints = 0;
213    let mut watermark: Option<CommitterWatermark> = None;
214
215    // Push data into the next batch as long as it's from contiguous checkpoints and we haven't
216    // gathered information from too many checkpoints already.
217    //
218    // We don't worry about overall size because the handler may have optimized writes by
219    // combining rows, but we will limit the number of checkpoints we try and batch together as a
220    // way to impose some limit on the size of the batch (and therefore the length of the write
221    // transaction).
222    // docs::#batch  (see https://docs.sui.io/develop/accessing-data/custom-indexer/custom-indexers)
223    while batch_checkpoints < max_batch_checkpoints {
224        let Some(entry) = pending.first_entry() else {
225            break;
226        };
227
228        match (*next_checkpoint).cmp(entry.key()) {
229            // Next pending checkpoint is from the future.
230            Ordering::Less => break,
231
232            // This is the next checkpoint -- include it.
233            Ordering::Equal => {
234                let indexed = entry.remove();
235                batch_rows += indexed.len();
236                batch_checkpoints += 1;
237                handler.batch(&mut batch, indexed.values.into_iter());
238                watermark = Some(indexed.watermark);
239                *next_checkpoint += 1;
240            }
241
242            // Next pending checkpoint is in the past, ignore it to avoid double writes.
243            Ordering::Greater => {
244                metrics
245                    .total_watermarks_out_of_order
246                    .with_label_values(&[H::NAME])
247                    .inc();
248
249                let indexed = entry.remove();
250                *pending_rows -= indexed.len();
251            }
252        }
253    }
254    // docs::/#batch
255
256    let elapsed = guard.stop_and_record();
257    debug!(
258        pipeline = H::NAME,
259        elapsed_ms = elapsed * 1000.0,
260        rows = batch_rows,
261        pending = *pending_rows,
262        "Gathered batch",
263    );
264
265    let watermark = watermark?;
266    *pending_rows -= batch_rows;
267
268    metrics
269        .collector_batch_size
270        .with_label_values(&[H::NAME])
271        .observe(batch_rows as f64);
272
273    metrics
274        .watermark_epoch
275        .with_label_values(&[H::NAME])
276        .set(watermark.epoch_hi_inclusive as i64);
277
278    metrics
279        .watermark_checkpoint
280        .with_label_values(&[H::NAME])
281        .set(watermark.checkpoint_hi_inclusive as i64);
282
283    metrics
284        .watermark_transaction
285        .with_label_values(&[H::NAME])
286        .set(watermark.tx_hi as i64);
287
288    metrics
289        .watermark_timestamp_ms
290        .with_label_values(&[H::NAME])
291        .set(watermark.timestamp_ms_hi_inclusive as i64);
292
293    Some(BatchedRows {
294        batch,
295        watermark,
296        batch_rows,
297    })
298}

300// Whether the first entry in `pending` is ready to be consumed by the collector — either to be
301// included in the next batch (if it matches `next_checkpoint`) or discarded as a stale duplicate
302// (if it predates `next_checkpoint`).
303fn has_ready_checkpoint<T>(next_checkpoint: u64, pending: &BTreeMap<u64, T>) -> bool {
304    pending
305        .first_key_value()
306        .is_some_and(|(&first, _)| first <= next_checkpoint)
307}
