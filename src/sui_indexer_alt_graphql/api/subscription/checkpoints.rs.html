4//! Checkpoints as a scan-then-live feed: every checkpoint is delivered as its own edge, in
5//! sequence-number order, unfiltered. The cursor is the checkpoint's sequence number. Backfill is
6//! one unfiltered `ListCheckpoints` scan (a sequential read), and both backfilled and live nodes
7//! carry their fully-processed checkpoint in memory, resolving fields from it without further reads.
8
9use std::future::Future;
10use std::ops::RangeInclusive;
11use std::sync::Arc;

18use sui_indexer_alt_reader::alpha_ledger_grpc_reader::StreamPage;
19use sui_rpc::proto::sui::rpc::v2;

23use crate::api::types::checkpoint::Checkpoint;

25use crate::error::RpcError;

35impl Subscribable for Checkpoint {
36    type Item = Self;
37    type Cursor = CCheckpoint;
38    /// Checkpoints are delivered unfiltered.
39    type Filter = ();
40    type ScanItem = v2::Checkpoint;
41
42    fn subscription_type() -> &'static str {
43        "checkpoints"
44    }
45
46    fn scan<'a>(
47        reader: &'a AlphaLedgerGrpcReader,
48        cp_bounds: RangeInclusive<u64>,
49        page: &'a Page<CCheckpoint>,
50        _filter: &'a (),
51    ) -> impl Future<Output = Result<StreamPage<v2::Checkpoint>, RpcError>> + Send + 'a {
52        list_checkpoints_page(reader, cp_bounds, page)
53    }
54
55    /// Fully process the scanned checkpoint and carry it on the node, resolving against it in memory.
56    /// The scan supplied the data cheaply as one sequential read; processing it here is the same work
57    /// the live path does off the broadcast, so a backfilled node resolves identically to a live one.
58    fn build_node(
59        caches: &Arc<StreamedCaches>,
60        resolver_limits: &sui_package_resolver::Limits,
61        payload: &v2::Checkpoint,
62    ) -> Result<Self, RpcError> {
63        let processed = Arc::new(process_checkpoint(payload.clone())?);
64        let scope = Scope::for_streamed_checkpoint(
65            caches.clone(),
66            resolver_limits.clone(),
67            processed.clone(),
68        );
69        Ok(Checkpoint {
70            sequence_number: processed.summary.sequence_number,
71            scope,
72            streamed_data: Some(processed),
73        })
74    }
75
76    /// A live checkpoint is delivered whole, as a single edge carrying its processed data.
77    fn matching_edges(
78        checkpoint: &Arc<ProcessedCheckpoint>,
79        caches: &Arc<StreamedCaches>,
80        resolver_limits: &sui_package_resolver::Limits,
81        _filter: &(),
82    ) -> Result<Vec<Edge<String, Self, EmptyFields>>, RpcError> {
83        let scope = Scope::for_streamed_checkpoint(
84            caches.clone(),
85            resolver_limits.clone(),
86            checkpoint.clone(),
87        );
88        let sequence_number = checkpoint.summary.sequence_number;
89        let cursor = CheckpointToken::cursor(sequence_number).encode_cursor();
90        let node = Checkpoint {
91            sequence_number,
92            scope,
93            streamed_data: Some(checkpoint.clone()),
94        };
95        Ok(vec![Edge::new(cursor, node)])
96    }
97}

99/// Scan one page of checkpoints over `cp_bounds`, resuming after `page`'s cursor. Unfiltered, so the
100/// server does one sequential scan rather than an inverted-index lookup.
101async fn list_checkpoints_page(
102    reader: &AlphaLedgerGrpcReader,
103    cp_bounds: RangeInclusive<u64>,
104    page: &Page<CCheckpoint>,
105) -> Result<StreamPage<v2::Checkpoint>, RpcError> {
106    let after = page.after().map(|c| CursorToken::from(&c.token()).encode());
107    let before = page
108        .before()
109        .map(|c| CursorToken::from(&c.token()).encode());
110
111    let mut options = v2::QueryOptions::default();
112    options.limit = Some(page.limit() as u32);
113    options.after = after;
114    options.before = before;
115    options.ordering = Some(if page.is_from_front() {
116        v2::Ordering::Ascending as i32
117    } else {
118        v2::Ordering::Descending as i32
119    });
120
121    let mut request = v2::ListCheckpointsRequest::default();
122    request.read_mask = Some(checkpoint_field_mask());
123    request.start_checkpoint = Some(*cp_bounds.start());
124    request.end_checkpoint = Some(cp_bounds.end().saturating_add(1));
125    request.filter = None;
126    request.options = Some(options);
127
128    Ok(reader
129        .list_checkpoints(request)
130        .await
131        .context("Failed to list checkpoints")?)
132}
