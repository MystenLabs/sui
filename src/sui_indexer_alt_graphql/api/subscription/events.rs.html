4//! Events as a scan-then-live feed: each matching event is delivered as its own edge, ordered by
5//! checkpoint, then by transaction position within the checkpoint, then by position within the
6//! transaction. The cursor is a three-part position (checkpoint, transaction, event index), and
7//! matching tests each event against the filter.
8

36impl Subscribable for Event {
37    type Item = Self;
38    type Cursor = CEvent;
39    type Filter = EventFilter;
40    type ScanItem = v2::Event;
41    type Output = EventSubscriptionEdge;
42
43    fn subscription_type() -> &'static str {
44        "events"
45    }
46
47    fn output(cursor: String, node: Option<Event>) -> EventSubscriptionEdge {
48        EventSubscriptionEdge { cursor, node }
49    }
50
51    fn scan<'a>(
52        reader: &'a LedgerGrpcReader,
53        cp_bounds: RangeInclusive<u64>,
54        page: &'a Page<CEvent>,
55        filter: &'a EventFilter,
56    ) -> impl Future<Output = Result<StreamPage<v2::Event>, RpcError>> + Send + 'a {
57        Event::scan_grpc(reader, cp_bounds, page, page.limit(), filter)
58    }
59
60    fn build_node(
61        caches: &Arc<StreamedCaches>,
62        resolver_limits: &sui_package_resolver::Limits,
63        payload: &v2::Event,
64    ) -> Result<Self, RpcError> {
65        let checkpoint = payload
66            .checkpoint
67            .context("ListEvents item missing checkpoint")?;
68        let scope = Scope::for_backfill(caches.clone(), resolver_limits.clone(), checkpoint);
69        event_from_stream_item(scope, payload)
70    }
71
72    fn matching_edges(
73        checkpoint: &Arc<ProcessedCheckpoint>,
74        caches: &Arc<StreamedCaches>,
75        resolver_limits: &sui_package_resolver::Limits,
76        filter: &EventFilter,
77    ) -> Result<Vec<Edge<String, Self, EmptyFields>>, RpcError> {
78        let timestamp_ms = Some(checkpoint.summary.timestamp_ms);
79        let scope = Scope::for_streamed_checkpoint(
80            caches.clone(),
81            resolver_limits.clone(),
82            checkpoint.clone(),
83        );
84        let mut edges = Vec::new();
85        for tx in &checkpoint.transactions {
86            let digest = tx
87                .contents
88                .digest()
89                .expect("ExecutedTransaction digest is infallible");
90            let events = tx.contents.events().unwrap_or_default();
91            for (idx, native) in events.into_iter().enumerate() {
92                if !filter.matches(&native) {
93                    continue;
94                }
95                let cursor = EventToken::cursor(
96                    checkpoint.summary.sequence_number,
97                    tx.tx_sequence_number,
98                    idx as u32,
99                )
100                .encode_cursor();
101                edges.push(Edge::new(
102                    cursor,
103                    Event {
104                        scope: scope.with_active_transaction_contents(digest, tx.contents.clone()),
105                        native,
106                        transaction_digest: digest,
107                        sequence_number: idx as u64,
108                        timestamp_ms: OnceCell::from(timestamp_ms),
109                    },
110                ));
111            }
112        }
113        Ok(edges)
114    }
115
116    fn boundary_after(checkpoint: &ProcessedCheckpoint) -> Option<CEvent> {
117        // `network_total_transactions` counts through this checkpoint, so it is the sequence number
118        // of the next checkpoint's first transaction, whose first event is index 0.
119        Some(EventToken::boundary(
120            checkpoint.summary.sequence_number + 1,
121            checkpoint.summary.network_total_transactions,
122            0,
123        ))
124    }
125}

127#[cfg(test)]
128mod tests {
129    use sui_rpc_cursor::CursorKind;
130    use sui_rpc_cursor::CursorToken;
131    use sui_rpc_cursor::Position;
132
133    use super::*;
134
135    /// `boundary_after` marks the start of the next checkpoint: a `Boundary` at that checkpoint's
136    /// first transaction and event index 0, so resuming after it begins at the first event of
137    /// `checkpoint + 1`.
138    #[test]
139    fn boundary_after_marks_start_of_next_checkpoint() {
140        let checkpoint = ProcessedCheckpoint::for_test(10, 42);
141        let cursor = Event::boundary_after(&checkpoint).expect("event feed mints a boundary");
142        let token = CursorToken::from(&cursor);
143        assert_eq!(token.kind, CursorKind::Boundary);
144        assert_eq!(
145            token.position,
146            Position::Events {
147                checkpoint: 11,
148                tx_seq: 42,
149                event_index: 0,
150            }
151        );
152    }
