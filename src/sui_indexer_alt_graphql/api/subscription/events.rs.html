4//! Events as a scan-then-live feed: each matching event is delivered as its own edge, ordered by
5//! checkpoint, then by transaction position within the checkpoint, then by position within the
6//! transaction. The cursor is a three-part position (checkpoint, transaction, event index), and
7//! matching tests each event against the filter.
8
9use std::future::Future;
10use std::ops::RangeInclusive;
11use std::sync::Arc;
12
13use async_graphql::connection::CursorType;
14use async_graphql::connection::Edge;
15use async_graphql::connection::EmptyFields;
16use sui_indexer_alt_reader::alpha_ledger_grpc_reader::AlphaLedgerGrpcReader;
17use sui_indexer_alt_reader::alpha_ledger_grpc_reader::StreamPage;
18use sui_rpc::proto::sui::rpc::v2;
19use tokio::sync::OnceCell;
20
21use crate::api::types::event::CEvent;
22use crate::api::types::event::Event;
23use crate::api::types::event::EventToken;
24use crate::api::types::event::event_from_stream_item;
25use crate::api::types::event::filter::EventFilter;
26use crate::error::RpcError;
27use crate::pagination::Page;
28use crate::scope::Scope;
29use crate::task::streaming::ProcessedCheckpoint;
30use crate::task::streaming::StreamedCaches;
31
32use super::scan_then_live::Subscribable;
33
34impl Subscribable for Event {
35    type Item = Self;
36    type Cursor = CEvent;
37    type Filter = EventFilter;
38    type ScanItem = v2::Event;
39
40    fn subscription_type() -> &'static str {
41        "events"
42    }
43
44    fn scan<'a>(
45        reader: &'a AlphaLedgerGrpcReader,
46        cp_bounds: RangeInclusive<u64>,
47        page: &'a Page<CEvent>,
48        filter: &'a EventFilter,
49    ) -> impl Future<Output = Result<StreamPage<v2::Event>, RpcError>> + Send + 'a {
50        Event::scan_grpc(reader, cp_bounds, page, filter)
51    }
52
53    fn build_node(
54        caches: &Arc<StreamedCaches>,
55        resolver_limits: &sui_package_resolver::Limits,
56        payload: &v2::Event,
57    ) -> Result<Self, RpcError> {
58        let scope = Scope::for_indexed(caches.clone(), resolver_limits.clone());
59        event_from_stream_item(scope, payload)
60    }
61
62    fn matching_edges(
63        checkpoint: &Arc<ProcessedCheckpoint>,
64        caches: &Arc<StreamedCaches>,
65        resolver_limits: &sui_package_resolver::Limits,
66        filter: &EventFilter,
67    ) -> Result<Vec<Edge<String, Self, EmptyFields>>, RpcError> {
68        let timestamp_ms = Some(checkpoint.summary.timestamp_ms);
69        let scope = Scope::for_streamed_checkpoint(
70            caches.clone(),
71            resolver_limits.clone(),
72            checkpoint.clone(),
73        );
74        let mut edges = Vec::new();
75        for tx in &checkpoint.transactions {
76            let digest = tx
77                .contents
78                .digest()
79                .expect("ExecutedTransaction digest is infallible");
80            let events = tx.contents.events().unwrap_or_default();
81            for (idx, native) in events.into_iter().enumerate() {
82                if !filter.matches(&native) {
83                    continue;
84                }
85                let cursor = EventToken::cursor(
86                    checkpoint.summary.sequence_number,
87                    tx.tx_sequence_number,
88                    idx as u32,
89                )
90                .encode_cursor();
91                edges.push(Edge::new(
92                    cursor,
93                    Event {
94                        scope: scope.with_active_transaction_contents(digest, tx.contents.clone()),
95                        native,
96                        transaction_digest: digest,
97                        sequence_number: idx as u64,
98                        timestamp_ms: OnceCell::from(timestamp_ms),
99                    },
100                ));
101            }
102        }
103        Ok(edges)
104    }
105}
