4//! Events as a scan-then-live feed: each matching event is delivered as its own edge, ordered by
5//! checkpoint, then by transaction position within the checkpoint, then by position within the
6//! transaction. The cursor places the event by those three indices, and matching tests each event
7//! against the filter.
8

10use std::ops::RangeInclusive;
11use std::sync::Arc;
12

14use async_graphql::connection::Edge;
15use async_graphql::connection::EmptyFields;

17use sui_rpc::proto::sui::rpc::v2;

37impl Subscribable for Event {
38    type Item = Self;
39    type Cursor = CEvent;
40    type Filter = EventFilter;
41    type ScanItem = v2::Event;
42    type Output = EventSubscriptionEdge;
43
44    fn subscription_type() -> &'static str {
45        "events"
46    }
47
48    fn output(cursor: String, node: Option<Event>) -> EventSubscriptionEdge {
49        EventSubscriptionEdge { cursor, node }
50    }
51
52    fn scan<'a>(
53        reader: &'a LedgerGrpcReader,
54        cp_bounds: RangeInclusive<u64>,
55        page: &'a Page<CEvent>,
56        filter: &'a EventFilter,
57    ) -> impl Future<Output = Result<LedgerPage<v2::Event, CEvent>, RpcError>> + Send + 'a {
58        Event::scan_grpc(reader, cp_bounds, page, page.limit(), filter)
59    }
60
61    fn resume_from(cursor: &CEvent) -> Option<u64> {
62        cursor.last_checkpoint()
63    }
64
65    fn made_progress(end: &CEvent, after: &CEvent, reached_tip: bool) -> bool {
66        ledger_scan_moved(end, after, reached_tip)
67    }
68
69    fn build_node(
70        caches: &Arc<StreamedCaches>,
71        resolver_limits: &sui_package_resolver::Limits,
72        checkpoint: u64,
73        payload: &v2::Event,
74    ) -> Result<Self, RpcError> {
75        let scope = Scope::for_backfill(caches.clone(), resolver_limits.clone(), checkpoint);
76        event_from_stream_item(scope, payload)
77    }
78
79    fn matching_edges(
80        checkpoint: &Arc<ProcessedCheckpoint>,
81        caches: &Arc<StreamedCaches>,
82        resolver_limits: &sui_package_resolver::Limits,
83        filter: &EventFilter,
84    ) -> Result<Vec<Edge<String, Self, EmptyFields>>, RpcError> {
85        let timestamp_ms = Some(checkpoint.summary.timestamp_ms);
86        let scope = Scope::for_streamed_checkpoint(
87            caches.clone(),
88            resolver_limits.clone(),
89            checkpoint.clone(),
90        );
91        let mut edges = Vec::new();
92        // `transactions` holds every transaction in the checkpoint, in order.
93        for (transaction_index, tx) in checkpoint.transactions.iter().enumerate() {
94            let digest = tx
95                .contents
96                .digest()
97                .expect("ExecutedTransaction digest is infallible");
98            let events = tx.contents.events().unwrap_or_default();
99            for (idx, native) in events.into_iter().enumerate() {
100                if !filter.matches(&native) {
101                    continue;
102                }
103                let cursor = EventToken::at(EventKey {
104                    checkpoint: checkpoint.summary.sequence_number,
105                    transaction_index: transaction_index as u64,
106                    event_index: idx as u32,
107                })
108                .encode_cursor();
109                edges.push(Edge::new(
110                    cursor,
111                    Event {
112                        scope: scope.with_active_transaction_contents(digest, tx.contents.clone()),
113                        native,
114                        transaction_digest: digest,
115                        sequence_number: idx as u64,
116                        timestamp_ms: OnceCell::from(timestamp_ms),
117                    },
118                ));
119            }
120        }
121        Ok(edges)
122    }
123
124    fn boundary_after(checkpoint: &ProcessedCheckpoint) -> Option<CEvent> {
125        Some(EventToken::checkpoint_start(
126            checkpoint.summary.sequence_number + 1,
127        ))
128    }
129}
