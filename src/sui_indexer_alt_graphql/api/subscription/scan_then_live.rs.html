4//! Scan-then-live subscription driver: stream the items matching a filter, in checkpoint order,
5//! resuming from a cursor or checkpoint, in two phases:
6//!
7//! 1. Backfill ([`backfill`]): scan indexed data forward from the resume point to catch up, using
8//!    the same pagination path.
9//! 2. Live ([`live`]): read matching items from the checkpoint broadcast.
10//!
11//! Once the scan comes within `handoff_threshold` of the tip, the driver subscribes to the broadcast,
12//! so reading is handed off to the live phase at `handoff + 1` with no gap or duplicate. The coverage
13//! claim the ledger reports on each page lets the handoff advance even through stretches that match
14//! nothing.
15

56/// How long to wait before re-requesting when the scan has drained the indexer's current tip but not
57/// yet reached the handoff. The wait doubles while the ledger makes no progress, up to
58/// `BACKFILL_POLL_MAX_INTERVAL`.
59const BACKFILL_POLL_INTERVAL: Duration = Duration::from_millis(100);
60const BACKFILL_POLL_MAX_INTERVAL: Duration = Duration::from_secs(2);
61
62/// How long the backfill waits for the ledger to make progress before asking the client to
63/// reconnect, so a stalled ledger does not hold subscription slots indefinitely.
64const BACKFILL_POLL_DEADLINE: Duration = Duration::from_secs(180);

66/// A feed that can be streamed with the scan-then-live driver. One implementation per subscribable
67/// GraphQL type supplies its cursor, filter, matching, and payload shape, and the free functions
68/// here drive the shared machinery over it.
69pub(super) trait Subscribable {
70    /// The GraphQL node delivered in each edge.
71    type Item: OutputType + Send + 'static;
72    /// The opaque cursor minted for resumption.
73    type Cursor: CursorType + Clone + Send + Sync + 'static;
74    /// The filter selecting matching items.
75    type Filter: Clone + Send + Sync + 'static;
76    /// The raw scan payload one page yields before conversion to edges.
77    type ScanItem: Send + 'static;
78
79    /// The GraphQL payload delivered for each item: an edge carrying the cursor and the node.
80    type Output: OutputType + Send + 'static;
81
82    /// Scan one page of matches over `cp_bounds`, starting after `page`'s cursor.
83    fn scan<'a>(
84        reader: &'a LedgerGrpcReader,
85        cp_bounds: RangeInclusive<u64>,
86        page: &'a Page<Self::Cursor>,
87        filter: &'a Self::Filter,
88    ) -> impl Future<Output = Result<LedgerPage<Self::ScanItem, Self::Cursor>, RpcError>> + Send + 'a;
89
90    /// The last checkpoint a resume cursor shows the client has seen any of, if it shows one.
91    fn resume_from(cursor: &Self::Cursor) -> Option<u64>;
92
93    /// Whether a scan page that ended at `end` moved past `after`, the cursor it started from.
94    /// `reached_tip` is whether the page ended with `LEDGER_TIP`: at the ledger's indexed tip, or,
95    /// for an `after` ahead of that tip, where `after` resumes. Only such a page can end where it
96    /// started.
97    fn made_progress(end: &Self::Cursor, after: &Self::Cursor, reached_tip: bool) -> bool;
98
99    /// Build the GraphQL node from one scanned item's payload, which sits in `checkpoint`,
100    /// constructing whatever scope the feed resolves against. The driver mints the cursor from the
101    /// item's position and assembles the edge.
102    fn build_node(
103        caches: &Arc<StreamedCaches>,
104        resolver_limits: &sui_package_resolver::Limits,
105        checkpoint: u64,
106        payload: &Self::ScanItem,
107    ) -> Result<Self::Item, RpcError>;
108
109    /// The filter-matching edges in one live checkpoint, in delivery order.
110    fn matching_edges(
111        checkpoint: &Arc<ProcessedCheckpoint>,
112        caches: &Arc<StreamedCaches>,
113        resolver_limits: &sui_package_resolver::Limits,
114        filter: &Self::Filter,
115    ) -> Result<Vec<Edge<String, Self::Item, EmptyFields>>, RpcError>;
116
117    /// Assemble one delivered payload from its cursor and node: a match passes the node, a progress
118    /// payload passes `None`. A feed that does not opt into progress is only ever called with `Some`.
119    fn output(cursor: String, node: Option<Self::Item>) -> Self::Output;
120
121    /// The cursor marking the end of `checkpoint`, minted for a live progress payload: at the start
122    /// of the next checkpoint, so a scan starting after it covers everything from there on.
123    /// Returning `None` suppresses live progress for the feed; backfill progress is gated
124    /// separately, by the per-subscription opt-in.
125    fn boundary_after(_checkpoint: &ProcessedCheckpoint) -> Option<Self::Cursor> {
126        None
127    }
128
129    /// The feed's name, used to label its lifecycle metrics.
130    fn subscription_type() -> &'static str;
131}

133/// One backfill scan output: either a matching edge at the checkpoint its payload names, or a
134/// coverage marker at a page's coverage claim, at or below which every match has been scanned
135/// (advances the handoff even on a match-less page), carrying the page's end cursor, the resume
136/// point past everything scanned so far.
137enum Scanned<I: OutputType> {
138    Match {
139        checkpoint: u64,
140        edge: Edge<String, I, EmptyFields>,
141    },
142    Covered {
143        checkpoint: u64,
144        cursor: String,
145    },
146}
147
148/// Where the backfill scan hands off to the live broadcast. The scan delivers every match at or
149/// below the pinned handoff; live delivers everything above it.
150struct Handoff {
151    /// Pin once the scan frontier is within this many checkpoints of the streamed tip.
152    threshold: u64,
153    /// The last checkpoint the client has seen any of, when known up front (from `afterCheckpoint`
154    /// or the `after` cursor).
155    resume_from: Option<u64>,
156    /// The client resumed from a cursor. Its checkpoint is unknown up front (a legacy cursor), or
157    /// comes from the client, so `resume_from` is raised to the first scan output, which the ledger
158    /// only returns once it is past the cursor.
159    resume_unknown: bool,
160    pinned: Option<u64>,
161}
162
163/// What the backfill does with one scan output, as decided by [`Handoff::step`].
164enum ScanStep<I: OutputType> {
165    /// Deliver this match.
166    Deliver(Edge<String, I, EmptyFields>),
167    /// Keep scanning.
168    Continue,
169    /// The scan has covered the handoff; live takes over.
170    Done,
171}

173/// Where a subscription resumes, from its `after` cursor and `afterCheckpoint`.
174#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
175struct ResumePoint {
176    /// The last checkpoint the client has seen any of, when known up front: the later of
177    /// `afterCheckpoint` and the cursor's.
178    checkpoint: Option<u64>,
179    /// Whether the client resumed from a cursor.
180    from_cursor: bool,
181}
182
183impl<I: OutputType> Scanned<I> {
184    /// The checkpoint this output sits at (the match's, or the page's coverage claim).
185    fn checkpoint(&self) -> u64 {
186        match self {
187            Scanned::Match { checkpoint, .. } | Scanned::Covered { checkpoint, .. } => *checkpoint,
188        }
189    }
190}

192/// Decides when the driver emits a progress payload and with what cursor. Every progress cursor
193/// sits at or past the last one delivered and goes out only after all matches up to it have been
194/// delivered, so resuming after it neither skips nor repeats.
195///
196/// Backfill and live produce progress differently. Backfill reuses each scan page's own end cursor
197/// at the page boundary, so it needs no pacing. Live has no scan cursor to reuse: it mints one, and
198/// only after a run of checkpoints that matched nothing, so a sparse filter does not emit a payload
199/// on every checkpoint.
200struct ProgressTracker {
201    /// Whether this subscription opted into progress payloads. When false, nothing is emitted on
202    /// either phase and every payload is a match.
203    enabled: bool,
204    /// Emit live progress after this many consecutive live checkpoints with no match (0 disables
205    /// live progress; backfill is unaffected).
206    unmatched_threshold: u64,
207    /// Consecutive live checkpoints with no match since the last match or progress payload.
208    consecutive_unmatched: u64,
209    /// Cursor of the last payload the backfill delivered, match or progress. Backfill emits progress
210    /// at a page end only when the page's end cursor moved past this, so a client is never handed a
211    /// cursor it already holds.
212    last_delivered_cursor: Option<String>,
213    /// Highest checkpoint a delivered progress cursor covers. A backfill page can scan past the
214    /// handoff, so live must not deliver at or below this, or the resume point would move backward.
215    covered_checkpoint: Option<u64>,
216}
217
218impl ProgressTracker {
219    fn new(enabled: bool, unmatched_threshold: u64) -> Self {
220        Self {
221            enabled,
222            unmatched_threshold,
223            consecutive_unmatched: 0,
224            last_delivered_cursor: None,
225            covered_checkpoint: None,
226        }
227    }
228
229    /// Remember the last cursor handed to the client, so a later page end can tell whether it
230    /// advanced past it.
231    fn record_delivered(&mut self, cursor: &str) {
232        self.last_delivered_cursor = Some(cursor.to_owned());
233    }
234
235    /// The progress cursor to emit for a scan page that covered through `checkpoint` and ended at
236    /// `end_cursor`, or `None` when it did not advance past the last cursor delivered (a full page
237    /// ends on its last match, so it emits nothing). Records the cursor when it does advance.
238    fn progress_if_advanced(&mut self, checkpoint: u64, end_cursor: String) -> Option<String> {
239        if !self.enabled {
240            return None;
241        }
242        if self.last_delivered_cursor.as_ref() == Some(&end_cursor) {
243            return None;
244        }
245        self.covered_checkpoint = Some(checkpoint);
246        self.last_delivered_cursor = Some(end_cursor.clone());
247        Some(end_cursor)
248    }
249
250    /// Whether a delivered progress cursor already covers live `checkpoint`.
251    fn covers(&self, checkpoint: u64) -> bool {
252        self.covered_checkpoint.is_some_and(|c| checkpoint <= c)
253    }
254
255    /// Reset the unmatched run after a live checkpoint that matched.
256    fn reset_unmatched_run(&mut self) {
257        self.consecutive_unmatched = 0;
258    }
259
260    /// The progress cursor to emit for a checkpoint that matched nothing, minted by `cursor` only
261    /// once the unmatched run reaches the threshold; `None` otherwise.
262    fn progress_after_unmatched(
263        &mut self,
264        checkpoint: u64,
265        cursor: impl FnOnce() -> String,
266    ) -> Option<String> {
267        if !self.enabled || self.unmatched_threshold == 0 {
268            return None;
269        }
270        self.consecutive_unmatched += 1;
271        if self.consecutive_unmatched < self.unmatched_threshold {
272            return None;
273        }
274        self.consecutive_unmatched = 0;
275        self.covered_checkpoint = Some(checkpoint);
276        Some(cursor())
277    }
278}

280impl Handoff {
281    fn new(threshold: u64, resume_from: Option<u64>) -> Self {
282        Self {
283            threshold,
284            resume_from,
285            resume_unknown: false,
286            pinned: None,
287        }
288    }
289
290    /// Raise the resume point to the first scan output, for an `after` cursor whose checkpoint is
291    /// unknown up front or comes from the client.
292    fn resume_from_first_output(mut self) -> Self {
293        self.resume_unknown = true;
294        self
295    }
296
297    /// Decide what to do with one scan output, given this replica's streamed `tip`.
298    ///
299    /// Pins first if the scan frontier is within the threshold of `tip`. `resubscribe` must
300    /// subscribe the live receiver and then return the streamed tip, so live sees every checkpoint
301    /// above the pin. A match past the handoff belongs to live, so the scan stops there.
302    fn step<I: OutputType>(
303        &mut self,
304        scanned: Scanned<I>,
305        tip: u64,
306        resubscribe: impl FnOnce() -> u64,
307    ) -> ScanStep<I> {
308        if std::mem::take(&mut self.resume_unknown) {
309            // The first output is past the cursor. A match sits in the cursor's checkpoint or a
310            // later one, with no match in between. A coverage claim is at least the checkpoint
311            // before the cursor's, so the one after the claim is at least the cursor's. Either way
312            // the scan stays responsible for the rest of the cursor's checkpoint.
313            let floor = match &scanned {
314                Scanned::Match { checkpoint, .. } => *checkpoint,
315                Scanned::Covered { checkpoint, .. } => checkpoint.saturating_add(1),
316            };
317            self.resume_from = Some(self.resume_from.map_or(floor, |from| from.max(floor)));
318        }
319
320        if self.pinned.is_none() && tip.saturating_sub(scanned.checkpoint()) <= self.threshold {
321            self.pin(resubscribe());
322        }
323
324        match scanned {
325            Scanned::Match { checkpoint, edge } => {
326                if self.pinned.is_none_or(|h| checkpoint <= h) {
327                    ScanStep::Deliver(edge)
328                } else {
329                    ScanStep::Done
330                }
331            }
332            Scanned::Covered { checkpoint, .. } => {
333                if self.pinned.is_some_and(|h| checkpoint >= h) {
334                    ScanStep::Done
335                } else {
336                    ScanStep::Continue
337                }
338            }
339        }
340    }
341
342    /// Pin the handoff at the streamed `tip` (read after resubscribing), floored at the resume
343    /// point, which can be ahead of this replica's tip behind a load balancer. Flooring here keeps
344    /// the scan responsible for the matches after an `after` cursor in its own checkpoint.
345    fn pin(&mut self, tip: u64) {
346        self.pinned = Some(match self.resume_from {
347            Some(from) => tip.max(from),
348            None => tip,
349        });
350    }
351
352    /// The last checkpoint the scan was responsible for; live delivers from the next one.
353    fn pinned(&self) -> Option<u64> {
354        self.pinned
355    }
356}

358impl ResumePoint {
359    /// `after` is the `after` cursor's [`Subscribable::resume_from`], if the client passed a cursor.
360    fn new(after: Option<Option<u64>>, after_checkpoint: Option<u64>) -> Self {
361        Self {
362            // `None` orders before any checkpoint, so this is the later of the two when both are set.
363            checkpoint: after.flatten().max(after_checkpoint),
364            from_cursor: after.is_some(),
365        }
366    }
367
368    /// Whether the client asked to resume at all; with neither `after` nor `afterCheckpoint`, the
369    /// subscription is live-only.
370    fn is_set(&self) -> bool {
371        self.from_cursor || self.checkpoint.is_some()
372    }
373
374    /// Whether this resume point's checkpoint sits more than `max_ahead` checkpoints past the
375    /// streamed `tip`. A resume point that names no checkpoint is never too far ahead.
376    fn is_too_far_ahead(&self, tip: u64, max_ahead: u64) -> bool {
377        self.checkpoint
378            .is_some_and(|checkpoint| checkpoint > tip.saturating_add(max_ahead))
379    }
380
381    /// Wait for every pipeline to index the resume point's checkpoint.
382    async fn wait_for_pipelines(
383        &self,
384        watermarks_rx: &mut watch::Receiver<Arc<Watermarks>>,
385    ) -> Result<(), RpcError> {
386        if let Some(checkpoint) = self.checkpoint {
387            wait_for_pipelines_catching_up_at(checkpoint, watermarks_rx).await?;
388        }
389        Ok(())
390    }
391}

393/// Subscribe to items matching `filter`: backfill from `resume` toward the tip, then follow live.
394/// The handoff is pinned mid-scan, once the scan frontier comes within `handoff_threshold` of the
395/// tip, so even a deep backfill catches up within one connection instead of lagging the receiver.
396#[allow(clippy::type_complexity)]
397pub(super) fn subscribe<S: Subscribable>(
398    reader: LedgerGrpcReader,
399    broadcast: Arc<SubscriptionBroadcast>,
400    caches: Arc<StreamedCaches>,
401    resolver_limits: sui_package_resolver::Limits,
402    mut watermarks_rx: watch::Receiver<Arc<Watermarks>>,
403    filter: S::Filter,
404    after: Option<S::Cursor>,
405    after_checkpoint: Option<u64>,
406    send_progress: bool,
407    subscriber_limit: SubscriberLimit,
408    config: SubscriptionConfig,
409) -> Result<impl Stream<Item = Result<S::Output, RpcError>>, RpcError<Error>> {
410    // Where to resume: the checkpoint the `after` cursor places itself in, and `afterCheckpoint`.
411    let resume = ResumePoint::new(after.as_ref().map(S::resume_from), after_checkpoint);
412
413    // Reject a resume point more than `max_ahead` checkpoints past the streamed tip before
414    // admitting, so bad input never claims a slot or lands in the subscription lifecycle metrics.
415    // There is nothing to backfill ahead of the tip, so such a request would only wait for the
416    // chain to reach it, and a far-future one would hold the connection open indefinitely.
417    let max_ahead = config.max_start_checkpoints_ahead_of_tip;
418    if resume.is_too_far_ahead(broadcast.network_tip(), max_ahead) {
419        return Err(bad_user_input(Error::TooFarAheadOfTip { max: max_ahead }));
420    }
421
422    // Claim a concurrency slot for the subscription's lifetime; at capacity this is `Err` with no
423    // guard created (and so no lifecycle metric).
424    let guard = SubscriptionLifecycleGuard::new(
425        S::subscription_type(),
426        broadcast.metrics(),
427        &subscriber_limit,
428    )
429    .map_err(upcast)?;
430
431    // Size the backfill scan page to the resolve concurrency. Scans are sequential (each needs the
432    // previous page's cursor), so feeding one window of `n` concurrent resolutions takes ceil(n /
433    // page) scans: a page much smaller than the concurrency makes scanning the bottleneck, a much
434    // larger one just holds a bigger page in memory. Matching them is roughly one scan per window.
435    let scan_page_size = config.max_concurrent_resolutions;
436
437    // Pin the handoff once the scan comes within half the live buffer of the tip, leaving room for
438    // checkpoints that arrive during the handoff so the receiver does not lag.
439    let handoff_threshold = config.broadcast_buffer as u64 / 2;
440
441    // Progress payloads are opt-in per subscription; once enabled, this threshold paces how often
442    // live progress fires.
443    let unmatched_threshold = config.unmatched_checkpoints_before_progress;
444
445    Ok(stream! {
446        let mut pending_receiver = None;
447        let mut handoff = Handoff::new(handoff_threshold, resume.checkpoint);
448        if resume.from_cursor {
449            handoff = handoff.resume_from_first_output();
450        }
451        let mut tracker = ProgressTracker::new(send_progress, unmatched_threshold);
452        let mut last_checkpoint: Option<u64> = None;
453
454        // Phase 1: scan toward the tip, pinning the live receiver near it. `after` and/or
455        // `afterCheckpoint` resume the backfill; neither means live-only.
456        if resume.is_set() {
457            // The resume point can be ahead of this server's watermarks (a cursor minted from the
458            // live stream, or by a more caught-up server). Waiting for them to reach it raises the
459            // ledger readers' monotonic-read floor past it, so no replica scans from a view behind
460            // the resume point. `scan_page` then waits for each page's end the same way, keeping
461            // the floor at or above every later resume point.
462            if let Err(e) = resume.wait_for_pipelines(&mut watermarks_rx).await {
463                guard.terminate(SubscriptionTerminationReason::BackfillError);
464                yield Err(e);
465                return;
466            }
467
468            let checkpoint_lo = after_checkpoint.map_or(0, |cp| cp.saturating_add(1));
469            let scan = backfill::<S>(
470                reader,
471                caches.clone(),
472                resolver_limits.clone(),
473                watermarks_rx,
474                filter.clone(),
475                after,
476                checkpoint_lo,
477                scan_page_size,
478            );
479            for await scanned in scan {
480                let scanned = match scanned {
481                    Ok(scanned) => scanned,
482                    Err(e) => {
483                        guard.terminate(SubscriptionTerminationReason::BackfillError);
484                        yield Err(e);
485                        return;
486                    }
487                };
488
489                // A page's end is progress once every match on it has been delivered. A match past
490                // the handoff ends the scan before its coverage marker is reached, so `page_end`
491                // for this page would not run; that is fine, live resumes from the handoff.
492                //
493                // Consecutive match-less pages each send one, uncoalesced: at most one small payload
494                // per ledger request (which the scan already paces, waiting at the tip and never
495                // resending a repeated end cursor), each moving a deep backfill's resume point on.
496                let page_end = match &scanned {
497                    Scanned::Covered { checkpoint, cursor } => Some((*checkpoint, cursor.clone())),
498                    Scanned::Match { .. } => None,
499                };
500
501                let step = handoff.step(scanned, broadcast.network_tip(), || {
502                    pending_receiver = Some(broadcast.broadcaster().resubscribe());
503                    broadcast.network_tip()
504                });
505                if let ScanStep::Deliver(edge) = &step {
506                    tracker.record_delivered(&edge.cursor);
507                }
508                if let Some((checkpoint, end_cursor)) = page_end
509                    && let Some(cursor) = tracker.progress_if_advanced(checkpoint, end_cursor)
510                {
511                    yield Ok(S::output(cursor, None));
512                }
513                match step {
514                    ScanStep::Deliver(edge) => {
515                        yield Ok(S::output(edge.cursor, Some(edge.node)));
516                        guard.record_backfill_delivered();
517                    }
518                    ScanStep::Continue => {}
519                    ScanStep::Done => break,
520                }
521            }
522            last_checkpoint = handoff.pinned();
523        }
524
525        // Phase 2: follow live from `handoff + 1` (a fresh receiver if there was no backfill).
526        let receiver = pending_receiver.unwrap_or_else(|| broadcast.broadcaster().resubscribe());
527        for await delivery in
528            live::<S>(receiver, last_checkpoint, caches, resolver_limits, filter, tracker, guard)
529        {
530            yield delivery;
531        }
532    })
533}

535/// Scan matches from the resume point toward the tip, yielding each match and a per-page coverage
536/// marker. Open-ended: the caller stops it once the handoff is covered.
537fn backfill<S: Subscribable>(
538    reader: LedgerGrpcReader,
539    caches: Arc<StreamedCaches>,
540    resolver_limits: sui_package_resolver::Limits,
541    mut watermarks_rx: watch::Receiver<Arc<Watermarks>>,
542    filter: S::Filter,
543    mut after: Option<S::Cursor>,
544    checkpoint_lo: u64,
545    scan_page_size: usize,
546) -> impl Stream<Item = Result<Scanned<S::Item>, RpcError>> {
547    stream! {
548        let limits = PageLimits {
549            default: scan_page_size as u32,
550            max: scan_page_size as u32,
551        };
552
553        // Resume from whichever is later: the `after` cursor or `[checkpoint_lo, ..]`. The upper
554        // bound is `u64::MAX` because backfill is open-ended, chasing the moving tip; `subscribe`
555        // stops it once the handoff is covered.
556        let cp_bounds = checkpoint_lo..=u64::MAX;
557
558        loop {
559            let (items, next_after) = match scan_page::<S>(
560                &reader,
561                &caches,
562                &resolver_limits,
563                &limits,
564                cp_bounds.clone(),
565                after.as_ref(),
566                &filter,
567                &mut watermarks_rx,
568            )
569            .await
570            {
571                Ok(page) => page,
572                Err(e) => {
573                    yield Err(e);
574                    return;
575                }
576            };
577            for item in items {
578                yield Ok(item);
579            }
580            after = Some(next_after);
581        }
582    }
583}

585/// Whether a transaction or event scan page that ended at `end` moved past `after`.
586///
587/// Only a page that reached the ledger's tip can end where it started: an `ItemLimit` page ends on
588/// an item past `after`, and a `ScanLimit` page at a frontier past it. A page at the tip is
589/// compared by place, because ledger servers of different versions encode the same position in
590/// different bytes. Its place moves whenever it scanned anything, because a page that scans up to
591/// the tip claims coverage through it.
592///
593/// Where either cursor has no place (a legacy `after`, or a page that claimed nothing), the ledger
594/// bytes decide. A page that drains the tip echoes `after`'s bytes, and a page whose `after` is
595/// ahead of the ledger's indexed tip ends where that cursor resumes, claiming nothing, rather than
596/// at the ledger's own tip, so a page never ends behind `after`. A cursor without ledger bytes was
597/// never sent to the ledger, so any page moves past it.
598pub(super) fn ledger_scan_moved<C: ScanCursor>(end: &C, after: &C, reached_tip: bool) -> bool {
599    let end = end.bound();
600    let after = after.bound();
601    if reached_tip && let (Some(end), Some(after)) = (end.point, after.point) {
602        return end > after.gap_after();
603    }
604
605    match (end.ledger, after.ledger) {
606        (Some(end), Some(after)) => end != after,
607        _ => true,
608    }
609}

611/// Follow the live broadcast from `last_checkpoint + 1`, delivering matching items. Drops the
612/// one-checkpoint seam overlap (the resubscribe/tip race), gap-checks, and disconnects on anomalies.
613fn live<S: Subscribable>(
614    mut receiver: CheckpointBroadcaster,
615    mut last_checkpoint: Option<u64>,
616    caches: Arc<StreamedCaches>,
617    resolver_limits: sui_package_resolver::Limits,
618    filter: S::Filter,
619    mut tracker: ProgressTracker,
620    guard: SubscriptionLifecycleGuard,
621) -> impl Stream<Item = Result<S::Output, RpcError>> {
622    stream! {
623        let mut delivered_live = false;
624        loop {
625            match receiver.recv().await {
626                Ok(checkpoint) => {
627                    let seq = checkpoint.summary.sequence_number;
628                    if let Some(last) = last_checkpoint {
629                        // Already covered by the scan (resubscribe/tip overlap): skip.
630                        if seq <= last {
631                            continue;
632                        }
633                        if seq > last + 1 {
634                            warn!(
635                                last_checkpoint = last,
636                                received = seq,
637                                "Unexpected gap between scan and live; disconnecting"
638                            );
639                            guard.terminate(SubscriptionTerminationReason::UnexpectedGap);
640                            yield Err(reconnect_error());
641                            return;
642                        }
643                    }
644                    // The last backfill page can scan past the handoff, and its end went out as
645                    // progress. The scan found nothing to deliver up to there, so a live match here
646                    // means the ledger and the stream disagree. Dropping it keeps the client's
647                    // cursor from moving backwards.
648                    if tracker.covers(seq) {
649                        if let Ok(edges) =
650                            S::matching_edges(&checkpoint, &caches, &resolver_limits, &filter)
651                            && !edges.is_empty()
652                        {
653                            warn!(
654                                checkpoint = seq,
655                                matches = edges.len(),
656                                "Dropping live matches in a checkpoint backfill progress already covered"
657                            );
658                        }
659                        last_checkpoint = Some(seq);
660                        continue;
661                    }
662
663                    // Deliver each matching item as its own payload, ordered within the checkpoint.
664                    // Empty checkpoints yield nothing but may advance the resume point as progress.
665                    let edges = match S::matching_edges(&checkpoint, &caches, &resolver_limits, &filter) {
666                        Ok(edges) => edges,
667                        Err(e) => {
668                            guard.terminate(SubscriptionTerminationReason::Error);
669                            yield Err(e);
670                            return;
671                        }
672                    };
673                    if edges.is_empty() {
674                        // This live checkpoint matched nothing. If the feed sends progress and the
675                        // unmatched run is long enough, emit one progress payload at the boundary
676                        // past this checkpoint, rather than one on every match-less checkpoint.
677                        if let Some(boundary) = S::boundary_after(&checkpoint)
678                            && let Some(cursor) =
679                                tracker.progress_after_unmatched(seq, || boundary.encode_cursor())
680                        {
681                            yield Ok(S::output(cursor, None));
682                        }
683                    } else {
684                        tracker.reset_unmatched_run();
685                    }
686                    for edge in edges {
687                        yield Ok(S::output(edge.cursor, Some(edge.node)));
688                        guard.record_delivered(checkpoint.summary.timestamp_ms);
689                    }
690                    last_checkpoint = Some(seq);
691                    delivered_live = true;
692                }
693                // A lag before the first live checkpoint is catch-up overflow (likely kv-rpc lag).
694                Err(broadcast::error::RecvError::Lagged(missed)) if !delivered_live => {
695                    warn!(missed, "Subscriber fell behind during catch-up; disconnecting");
696                    guard.terminate(SubscriptionTerminationReason::Lagged);
697                    yield Err(reconnect_error());
698                    return;
699                }
700                Err(e) => {
701                    guard.terminate(SubscriptionTerminationReason::from_recv_error(&e));
702                    yield Err(broadcast_error(e));
703                    return;
704                }
705            }
706        }
707    }
708}

710/// Retry policy for the backfill scan: jittered exponential backoff, giving up after 60s (past which
711/// a failure is unlikely to be a transient blip).
712fn scan_backoff() -> ExponentialBackoff {
713    ExponentialBackoff {
714        initial_interval: Duration::from_millis(100),
715        max_interval: Duration::from_secs(5),
716        max_elapsed_time: Some(Duration::from_secs(60)),
717        ..Default::default()
718    }
719}

721/// Paces re-requests while the ledger has nothing past the scan's position: waits double from
722/// `BACKFILL_POLL_INTERVAL` up to `BACKFILL_POLL_MAX_INTERVAL`, and once `BACKFILL_POLL_DEADLINE`
723/// has passed without progress the client is asked to reconnect.
724struct LedgerPoll {
725    interval: Duration,
726    deadline: tokio::time::Instant,
727}
728
729impl LedgerPoll {
730    fn new() -> Self {
731        Self {
732            interval: BACKFILL_POLL_INTERVAL,
733            deadline: tokio::time::Instant::now() + BACKFILL_POLL_DEADLINE,
734        }
735    }
736
737    async fn wait(&mut self) -> Result<(), RpcError> {
738        if tokio::time::Instant::now() >= self.deadline {
739            warn!("Ledger made no progress within the backfill deadline; disconnecting");
740            return Err(reconnect_error());
741        }
742        tokio::time::sleep(self.interval).await;
743        self.interval = (self.interval * 2).min(BACKFILL_POLL_MAX_INTERVAL);
744        Ok(())
745    }
746}

748/// Scan the next page of matches, waiting at the indexer tip if nothing is ready yet. Returns the
749/// matches, a trailing coverage marker, and the cursor to resume from.
750async fn scan_page<S: Subscribable>(
751    reader: &LedgerGrpcReader,
752    caches: &Arc<StreamedCaches>,
753    resolver_limits: &sui_package_resolver::Limits,
754    limits: &PageLimits,
755    cp_bounds: RangeInclusive<u64>,
756    after: Option<&S::Cursor>,
757    filter: &S::Filter,
758    watermarks_rx: &mut watch::Receiver<Arc<Watermarks>>,
759) -> Result<(Vec<Scanned<S::Item>>, S::Cursor), RpcError> {
760    let mut poll = LedgerPoll::new();
761    loop {
762        let result = scan_with_retry::<S>(reader, limits, cp_bounds.clone(), after, filter).await?;
763
764        // No end cursor means nothing past the indexer tip is indexed yet; wait, then re-request. A
765        // range that was scanned but matched nothing still returns a boundary cursor, not `None`.
766        let Some(end_cursor) = result.last_cursor else {
767            poll.wait().await?;
768            continue;
769        };
770
771        // A page that reached the ledger's tip without moving: wait for the ledger to index more
772        // before asking again, rather than re-polling at full speed.
773        let reached_tip = result.end_reason == Some(QueryEndReason::LedgerTip);
774        if after.is_some_and(|after| !S::made_progress(&end_cursor, after, reached_tip)) {
775            poll.wait().await?;
776            continue;
777        }
778
779        // Hold the page until all pipelines have indexed through its end: its last item, its
780        // coverage claim, and, for a page cut short by the scan limit, the checkpoint after the
781        // claim, which its end cursor's ledger bytes sit in. The delivered items' nested queries
782        // resolve against indexed data, and the ledger readers' floor reaches the next page's
783        // resume point.
784        let last_item = result.items.last().map(|item| item.checkpoint);
785        let end = match result.end_reason {
786            Some(QueryEndReason::ScanLimit) => result.covered_checkpoint.map(|c| c + 1),
787            _ => result.covered_checkpoint,
788        };
789        if let Some(checkpoint) = last_item.max(end) {
790            wait_for_pipelines_catching_up_at(checkpoint, watermarks_rx).await?;
791        }
792
793        // A forward page keeps `items` in ascending order, one delivered edge each.
794        let mut items = Vec::with_capacity(result.items.len() + 1);
795        for item in result.items {
796            items.push(Scanned::Match {
797                checkpoint: item.checkpoint,
798                edge: Edge::new(
799                    item.cursor.encode_cursor(),
800                    S::build_node(caches, resolver_limits, item.checkpoint, &item.payload)?,
801                ),
802            });
803        }
804
805        // Coverage marker at the page's coverage claim: advances the handoff even on a match-less
806        // page, and carries the page's end cursor so the backfill can deliver it as progress. A
807        // page without a claim (some ledger servers send none for a page that scanned nothing)
808        // adds no marker; a later page's claim covers it.
809        if let Some(checkpoint) = result.covered_checkpoint {
810            items.push(Scanned::Covered {
811                checkpoint,
812                cursor: end_cursor.encode_cursor(),
813            });
814        }
815
816        return Ok((items, end_cursor));
817    }
818}

820/// Scan one page over `cp_bounds`, retrying transient failures under a bounded budget.
821async fn scan_with_retry<S: Subscribable>(
822    reader: &LedgerGrpcReader,
823    limits: &PageLimits,
824    cp_bounds: RangeInclusive<u64>,
825    after: Option<&S::Cursor>,
826    filter: &S::Filter,
827) -> Result<LedgerPage<S::ScanItem, S::Cursor>, RpcError> {
828    let page = Page::from_params(
829        limits,
830        Some(limits.default as u64),
831        after.cloned(),
832        None,
833        None,
834    )
835    .map_err(anyhow::Error::from)?;
836
837    backoff::future::retry(scan_backoff(), || async {
838        S::scan(reader, cp_bounds.clone(), &page, filter)
839            .await
840            .map_err(|e| {
841                warn!(error = ?e, "scan failed, retrying");
842                backoff::Error::transient(e)
843            })
844    })
845    .await
846}

885    /// [`drive`] with a prepared `handoff`.
886    fn drive_handoff(mut handoff: Handoff, tip: u64, outputs: &[Out]) -> (Vec<u64>, Option<u64>) {
887        let mut delivered = vec![];
888        for out in outputs {
889            // Matches carry their checkpoint as the node, so delivery order is observable.
890            let scanned = match *out {
891                Out::Match(checkpoint) => Scanned::Match {
892                    checkpoint,
893                    edge: Edge::new(String::new(), checkpoint),
894                },
895                Out::End(checkpoint) => Scanned::Covered {
896                    checkpoint,
897                    cursor: String::new(),
898                },
899            };
900            match handoff.step(scanned, tip, || tip) {
901                ScanStep::Deliver(edge) => delivered.push(edge.node),
902                ScanStep::Continue => {}
903                ScanStep::Done => return (delivered, handoff.pinned()),
904            }
905        }
906        (delivered, None)
907    }
