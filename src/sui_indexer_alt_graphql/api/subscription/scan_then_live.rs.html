4//! Scan-then-live subscription driver: stream the items matching a filter, in checkpoint order,
5//! resuming from a cursor or checkpoint, in two phases:
6//!
7//! 1. Backfill ([`backfill`]): scan indexed data forward from the resume point to catch up, using
8//!    the same pagination path.
9//! 2. Live ([`live`]): read matching items from the checkpoint broadcast.
10//!
11//! Once the scan comes within `handoff_threshold` of the tip, the driver subscribes to the broadcast,
12//! so reading is handed off to the live phase at `handoff + 1` with no gap or duplicate. A coverage
13//! marker at each page's end lets the handoff advance even through stretches that match nothing.
14

56/// How long to wait before re-requesting when the scan has drained the indexer's current tip but not
57/// yet reached the handoff.
58const BACKFILL_POLL_INTERVAL: Duration = Duration::from_millis(100);

60/// A feed that can be streamed with the scan-then-live driver. One implementation per subscribable
61/// GraphQL type (transactions, events) supplies its cursor, filter, and matching, and the free
62/// functions here drive the shared machinery over it.
63pub(super) trait Subscribable {
64    /// The GraphQL node delivered in each edge.
65    type Item: OutputType + Send + 'static;
66    /// The opaque cursor minted for resumption.
67    type Cursor: CursorType
68        + Eq
69        + Clone
70        + Send
71        + Sync
72        + TryFrom<CursorToken, Error = anyhow::Error>
73        + 'static;
74    /// The filter selecting matching items.
75    type Filter: Clone + Send + Sync + 'static;
76    /// The raw scan payload one page yields before conversion to edges.
77    type ScanItem: Send + 'static;
78
79    /// Scan one page of matches over `cp_bounds`, starting after `page`'s cursor.
80    fn scan<'a>(
81        reader: &'a AlphaLedgerGrpcReader,
82        cp_bounds: RangeInclusive<u64>,
83        page: &'a Page<Self::Cursor>,
84        filter: &'a Self::Filter,
85    ) -> impl Future<Output = Result<StreamPage<Self::ScanItem>, RpcError>> + Send + 'a;
86
87    /// Build the GraphQL node from one scanned item's payload, constructing whatever scope the feed
88    /// resolves against. The driver mints the cursor from the item's position and assembles the edge.
89    fn build_node(
90        caches: &Arc<StreamedCaches>,
91        resolver_limits: &sui_package_resolver::Limits,
92        payload: &Self::ScanItem,
93    ) -> Result<Self::Item, RpcError>;
94
95    /// The filter-matching edges in one live checkpoint, in delivery order.
96    fn matching_edges(
97        checkpoint: &Arc<ProcessedCheckpoint>,
98        caches: &Arc<StreamedCaches>,
99        resolver_limits: &sui_package_resolver::Limits,
100        filter: &Self::Filter,
101    ) -> Result<Vec<Edge<String, Self::Item, EmptyFields>>, RpcError>;
102
103    /// The feed's name, used to label its lifecycle metrics.
104    fn subscription_type() -> &'static str;
105}

107/// One backfill scan output: either a matching edge, or a coverage marker whose `checkpoint` is the
108/// fully-scanned frontier (advances the handoff even on a match-less page).
109enum Scanned<I: OutputType> {
110    Match {
111        checkpoint: u64,
112        edge: Edge<String, I, EmptyFields>,
113    },
114    Covered {
115        checkpoint: u64,
116    },
117}
118
119impl<I: OutputType> Scanned<I> {
120    /// The checkpoint this output sits at (the match's, or the covered frontier).
121    fn checkpoint(&self) -> u64 {
122        match self {
123            Scanned::Match { checkpoint, .. } | Scanned::Covered { checkpoint } => *checkpoint,
124        }
125    }
126}

128/// Subscribe to items matching `filter`: backfill from `resume` toward the tip, then follow live.
129/// The handoff is pinned mid-scan, once the scan frontier comes within `handoff_threshold` of the
130/// tip, so even a deep backfill catches up within one connection instead of lagging the receiver.
131#[allow(clippy::type_complexity)]
132pub(super) fn subscribe<S: Subscribable>(
133    reader: AlphaLedgerGrpcReader,
134    broadcast: Arc<SubscriptionBroadcast>,
135    caches: Arc<StreamedCaches>,
136    resolver_limits: sui_package_resolver::Limits,
137    watermarks_rx: watch::Receiver<Arc<Watermarks>>,
138    filter: S::Filter,
139    after: Option<S::Cursor>,
140    after_checkpoint: Option<u64>,
141    subscriber_limit: SubscriberLimit,
142    config: SubscriptionConfig,
143) -> Result<impl Stream<Item = Result<Edge<String, S::Item, EmptyFields>, RpcError>>, RpcError<Error>>
144where
145    for<'a> CursorToken: From<&'a S::Cursor>,
146{
147    // The checkpoint to resume after: the later of the `after` cursor and `afterCheckpoint`. A
148    // cursor's checkpoint is read through its `CursorToken` position, so the driver needs no
149    // per-feed accessor.
150    let start_from = match (
151        after
152            .as_ref()
153            .map(|c| CursorToken::from(c).position.checkpoint()),
154        after_checkpoint,
155    ) {
156        (Some(a), Some(b)) => Some(a.max(b)),
157        (a, b) => a.or(b),
158    };
159
160    // Reject a far-ahead resume before admitting, so bad input never claims a slot or lands in the
161    // subscription lifecycle metrics.
162    reject_if_start_too_far_ahead(start_from, &broadcast, &config)?;
163
164    // Claim a concurrency slot for the subscription's lifetime; at capacity this is `Err` with no
165    // guard created (and so no lifecycle metric).
166    let guard = SubscriptionLifecycleGuard::new(
167        S::subscription_type(),
168        broadcast.metrics(),
169        &subscriber_limit,
170    )
171    .map_err(upcast)?;
172
173    // Size the backfill scan page to the resolve concurrency. Scans are sequential (each needs the
174    // previous page's cursor), so feeding one window of `n` concurrent resolutions takes ceil(n /
175    // page) scans: a page much smaller than the concurrency makes scanning the bottleneck, a much
176    // larger one just holds a bigger page in memory. Matching them is roughly one scan per window.
177    let scan_page_size = config.max_concurrent_resolutions;
178
179    // Pin the handoff once the scan comes within half the live buffer of the tip, leaving room for
180    // checkpoints that arrive during the handoff so the receiver does not lag.
181    let handoff_threshold = config.broadcast_buffer as u64 / 2;
182
183    Ok(stream! {
184        let mut pending_receiver = None;
185        let mut handoff: Option<u64> = None;
186        let mut last_checkpoint: Option<u64> = None;
187
188        // Phase 1: scan toward the tip, pinning the live receiver near it. `after` and/or
189        // `afterCheckpoint` resume the backfill; neither means live-only.
190        if after.is_some() || after_checkpoint.is_some() {
191            let checkpoint_lo = after_checkpoint.map_or(0, |cp| cp.saturating_add(1));
192            let scan = backfill::<S>(
193                reader,
194                caches.clone(),
195                resolver_limits.clone(),
196                watermarks_rx,
197                filter.clone(),
198                after,
199                checkpoint_lo,
200                scan_page_size,
201            );
202            for await scanned in scan {
203                let scanned = match scanned {
204                    Ok(scanned) => scanned,
205                    Err(e) => {
206                        guard.terminate(SubscriptionTerminationReason::BackfillError);
207                        yield Err(e);
208                        return;
209                    }
210                };
211
212                // Resubscribe-first and pin once the scan frontier is within threshold of the tip.
213                if pending_receiver.is_none()
214                    && broadcast.network_tip().saturating_sub(scanned.checkpoint()) <= handoff_threshold
215                {
216                    pending_receiver = Some(broadcast.broadcaster().resubscribe());
217                    handoff = Some(broadcast.network_tip());
218                }
219
220                match scanned {
221                    // A match: deliver it, unless it is past the handoff (then it is live's).
222                    Scanned::Match { checkpoint, edge } => {
223                        if handoff.is_some_and(|h| checkpoint > h) {
224                            break;
225                        }
226                        yield Ok(edge);
227                        guard.record_backfill_delivered();
228                    }
229                    // A coverage marker (its `checkpoint` is the fully-scanned frontier): stop once
230                    // it has covered the handoff.
231                    Scanned::Covered { checkpoint } => {
232                        if handoff.is_some_and(|h| checkpoint >= h) {
233                            break;
234                        }
235                    }
236                }
237            }
238            last_checkpoint = handoff;
239        }
240
241        // Phase 2: follow live from `handoff + 1` (a fresh receiver if there was no backfill).
242        let receiver = pending_receiver.unwrap_or_else(|| broadcast.broadcaster().resubscribe());
243        for await edge in
244            live::<S>(receiver, last_checkpoint, caches, resolver_limits, filter, guard)
245        {
246            yield edge;
247        }
248    })
249}

251/// Reject a start point sitting more than `max_ahead` checkpoints past the chain tip. There is
252/// nothing to backfill ahead of the tip, so such a request would only wait for the chain to reach
253/// it, and a far-future one would hold the connection open indefinitely.
254fn reject_if_start_too_far_ahead(
255    start_from: Option<u64>,
256    broadcast: &SubscriptionBroadcast,
257    config: &SubscriptionConfig,
258) -> Result<(), RpcError<Error>> {
259    let max_ahead = config.max_start_checkpoints_ahead_of_tip;
260    if let Some(start) = start_from
261        && start > broadcast.network_tip().saturating_add(max_ahead)
262    {
263        return Err(bad_user_input(Error::TooFarAheadOfTip { max: max_ahead }));
264    }
265    Ok(())
266}

268/// Scan matches from the resume point toward the tip, yielding each match and a per-page coverage
269/// marker. Open-ended: the caller stops it once the handoff is covered.
270fn backfill<S: Subscribable>(
271    reader: AlphaLedgerGrpcReader,
272    caches: Arc<StreamedCaches>,
273    resolver_limits: sui_package_resolver::Limits,
274    mut watermarks_rx: watch::Receiver<Arc<Watermarks>>,
275    filter: S::Filter,
276    mut after: Option<S::Cursor>,
277    checkpoint_lo: u64,
278    scan_page_size: usize,
279) -> impl Stream<Item = Result<Scanned<S::Item>, RpcError>> {
280    stream! {
281        let limits = PageLimits {
282            default: scan_page_size as u32,
283            max: scan_page_size as u32,
284        };
285
286        // Resume from whichever is later: the `after` cursor or `[checkpoint_lo, ..]`. The upper
287        // bound is `u64::MAX` because backfill is open-ended, chasing the moving tip; `subscribe`
288        // stops it once the handoff is covered.
289        let cp_bounds = checkpoint_lo..=u64::MAX;
290
291        loop {
292            let (items, next_after) = match scan_page::<S>(
293                &reader,
294                &caches,
295                &resolver_limits,
296                &limits,
297                cp_bounds.clone(),
298                after.as_ref(),
299                &filter,
300                &mut watermarks_rx,
301            )
302            .await
303            {
304                Ok(page) => page,
305                Err(e) => {
306                    yield Err(e);
307                    return;
308                }
309            };
310            for item in items {
311                yield Ok(item);
312            }
313            after = Some(next_after);
314        }
315    }
316}

318/// The highest checkpoint fully scanned as of `token`: its own checkpoint for a `Boundary`, one less
319/// for an `Item` (whose checkpoint may still hold later matches).
320fn covered_checkpoint(token: &CursorToken) -> u64 {
321    let checkpoint = token.position.checkpoint();
322    match token.kind {
323        CursorKind::Boundary => checkpoint,
324        CursorKind::Item => checkpoint.saturating_sub(1),
325    }
326}

328/// Follow the live broadcast from `last_checkpoint + 1`, delivering matching items. Drops the
329/// one-checkpoint seam overlap (the resubscribe/tip race), gap-checks, and disconnects on anomalies.
330fn live<S: Subscribable>(
331    mut receiver: CheckpointBroadcaster,
332    mut last_checkpoint: Option<u64>,
333    caches: Arc<StreamedCaches>,
334    resolver_limits: sui_package_resolver::Limits,
335    filter: S::Filter,
336    guard: SubscriptionLifecycleGuard,
337) -> impl Stream<Item = Result<Edge<String, S::Item, EmptyFields>, RpcError>> {
338    stream! {
339        let mut delivered_live = false;
340        loop {
341            match receiver.recv().await {
342                Ok(checkpoint) => {
343                    let seq = checkpoint.summary.sequence_number;
344                    if let Some(last) = last_checkpoint {
345                        // Already covered by the scan (resubscribe/tip overlap): skip.
346                        if seq <= last {
347                            continue;
348                        }
349                        if seq > last + 1 {
350                            warn!(
351                                last_checkpoint = last,
352                                received = seq,
353                                "Unexpected gap between scan and live; disconnecting"
354                            );
355                            guard.terminate(SubscriptionTerminationReason::UnexpectedGap);
356                            yield Err(reconnect_error());
357                            return;
358                        }
359                    }
360                    // Deliver each matching item as its own payload, ordered within the checkpoint.
361                    // Empty checkpoints yield nothing.
362                    let edges = match S::matching_edges(&checkpoint, &caches, &resolver_limits, &filter) {
363                        Ok(edges) => edges,
364                        Err(e) => {
365                            guard.terminate(SubscriptionTerminationReason::Error);
366                            yield Err(e);
367                            return;
368                        }
369                    };
370                    for edge in edges {
371                        yield Ok(edge);
372                        guard.record_delivered(checkpoint.summary.timestamp_ms);
373                    }
374                    last_checkpoint = Some(seq);
375                    delivered_live = true;
376                }
377                // A lag before the first live checkpoint is catch-up overflow (likely kv-rpc lag).
378                Err(broadcast::error::RecvError::Lagged(missed)) if !delivered_live => {
379                    warn!(missed, "Subscriber fell behind during catch-up; disconnecting");
380                    guard.terminate(SubscriptionTerminationReason::Lagged);
381                    yield Err(reconnect_error());
382                    return;
383                }
384                Err(e) => {
385                    guard.terminate(SubscriptionTerminationReason::from_recv_error(&e));
386                    yield Err(broadcast_error(e));
387                    return;
388                }
389            }
390        }
391    }
392}

394/// Retry policy for the backfill scan: jittered exponential backoff, giving up after 60s (past which
395/// a failure is unlikely to be a transient blip).
396fn scan_backoff() -> ExponentialBackoff {
397    ExponentialBackoff {
398        initial_interval: Duration::from_millis(100),
399        max_interval: Duration::from_secs(5),
400        max_elapsed_time: Some(Duration::from_secs(60)),
401        ..Default::default()
402    }
403}

405/// Scan the next page of matches, waiting at the indexer tip if nothing is ready yet. Returns the
406/// matches, a trailing coverage marker, and the cursor to resume from.
407async fn scan_page<S: Subscribable>(
408    reader: &AlphaLedgerGrpcReader,
409    caches: &Arc<StreamedCaches>,
410    resolver_limits: &sui_package_resolver::Limits,
411    limits: &PageLimits,
412    cp_bounds: RangeInclusive<u64>,
413    after: Option<&S::Cursor>,
414    filter: &S::Filter,
415    watermarks_rx: &mut watch::Receiver<Arc<Watermarks>>,
416) -> Result<(Vec<Scanned<S::Item>>, S::Cursor), RpcError> {
417    loop {
418        let result = scan_with_retry::<S>(reader, limits, cp_bounds.clone(), after, filter).await?;
419
420        // No end cursor means nothing past the indexer tip is indexed yet; wait, then re-request. A
421        // range that was scanned but matched nothing still returns a boundary cursor, not `None`.
422        let Some(end_cursor) = result.last_cursor() else {
423            tokio::time::sleep(BACKFILL_POLL_INTERVAL).await;
424            continue;
425        };
426        let end_cursor = CursorToken::decode(end_cursor).context("Failed to decode scan cursor")?;
427
428        // Hold the page until all pipelines have indexed through its end.
429        wait_for_pipelines_catching_up_at(end_cursor.position.checkpoint(), watermarks_rx).await?;
430
431        // A forward page keeps `items` in ascending order, one delivered edge each. The decoded
432        // token yields both the checkpoint and the edge cursor, so it is only decoded once.
433        let mut items = Vec::with_capacity(result.items.len() + 1);
434        for item in &result.items {
435            let token =
436                CursorToken::decode(&item.cursor).context("Failed to decode scan item cursor")?;
437            let checkpoint = token.position.checkpoint();
438            let cursor = S::Cursor::try_from(token)
439                .context("Unexpected position in scan item cursor")?
440                .encode_cursor();
441            items.push(Scanned::Match {
442                checkpoint,
443                edge: Edge::new(
444                    cursor,
445                    S::build_node(caches, resolver_limits, &item.payload)?,
446                ),
447            });
448        }
449        // Coverage marker at the fully-scanned end: advances the handoff even on a match-less page.
450        items.push(Scanned::Covered {
451            checkpoint: covered_checkpoint(&end_cursor),
452        });
453
454        let next_after =
455            S::Cursor::try_from(end_cursor).context("Unexpected position in scan end cursor")?;
456        return Ok((items, next_after));
457    }
458}

460/// Scan one page over `cp_bounds`, retrying transient failures under a bounded budget.
461async fn scan_with_retry<S: Subscribable>(
462    reader: &AlphaLedgerGrpcReader,
463    limits: &PageLimits,
464    cp_bounds: RangeInclusive<u64>,
465    after: Option<&S::Cursor>,
466    filter: &S::Filter,
467) -> Result<StreamPage<S::ScanItem>, RpcError> {
468    let page = Page::from_params(
469        limits,
470        Some(limits.default as u64),
471        after.cloned(),
472        None,
473        None,
474    )
475    .map_err(anyhow::Error::from)?;
476
477    backoff::future::retry(scan_backoff(), || async {
478        S::scan(reader, cp_bounds.clone(), &page, filter)
479            .await
480            .map_err(|e| {
481                warn!(error = ?e, "scan failed, retrying");
482                backoff::Error::transient(e)
483            })
484    })
485    .await
