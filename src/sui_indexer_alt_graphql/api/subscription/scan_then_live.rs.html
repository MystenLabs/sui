4//! Scan-then-live subscription driver: stream the items matching a filter, in checkpoint order,
5//! resuming from a cursor or checkpoint, in two phases:
6//!
7//! 1. Backfill ([`backfill`]): scan indexed data forward from the resume point to catch up, using
8//!    the same pagination path.
9//! 2. Live ([`live`]): read matching items from the checkpoint broadcast.
10//!
11//! Once the scan comes within `handoff_threshold` of the tip, the driver subscribes to the broadcast,
12//! so reading is handed off to the live phase at `handoff + 1`. The coverage claim the ledger reports
13//! on each page lets the handoff advance even through stretches that match nothing. Every payload
14//! goes out only past the position the client's latest cursor resumes from, so the two phases leave
15//! no gap and repeat nothing.
16

53/// How long to wait before re-requesting when the scan has drained the indexer's current tip but not
54/// yet reached the handoff. The wait doubles while the ledger makes no progress, up to
55/// `BACKFILL_POLL_MAX_INTERVAL`.
56const BACKFILL_POLL_INTERVAL: Duration = Duration::from_millis(100);
57const BACKFILL_POLL_MAX_INTERVAL: Duration = Duration::from_secs(2);
58
59/// How long the backfill waits for the ledger to make progress before asking the client to
60/// reconnect, so a stalled ledger does not hold subscription slots indefinitely.
61const BACKFILL_POLL_DEADLINE: Duration = Duration::from_secs(180);

63/// The filter-matching items in one live checkpoint, each with its cursor, in delivery order.
64type LiveMatches<S> = Vec<(<S as Subscribable>::Cursor, <S as Subscribable>::Item)>;

66/// A feed that can be streamed with the scan-then-live driver. One implementation per subscribable
67/// GraphQL type supplies its cursor, filter, matching, and payload shape, and the free functions
68/// here drive the shared machinery over it.
69pub(super) trait Subscribable {
70    /// The GraphQL node delivered in each edge.
71    type Item: OutputType + Send + 'static;
72    /// The opaque cursor minted for resumption.
73    type Cursor: CursorType + Clone + Send + Sync + 'static;
74    /// A position in the feed's order: the boundary just before an item, whether or not an item
75    /// sits there.
76    type Position: Ord + Copy + Send + Sync + 'static;
77    /// The filter selecting matching items.
78    type Filter: Clone + Send + Sync + 'static;
79    /// The raw scan payload one page yields before conversion to edges.
80    type ScanItem: Send + 'static;
81
82    /// The GraphQL payload delivered for each item: an edge carrying the cursor and the node.
83    type Output: OutputType + Send + 'static;
84
85    /// Scan one page of matches over `cp_bounds`, starting after `page`'s cursor.
86    fn scan<'a>(
87        reader: &'a LedgerGrpcReader,
88        cp_bounds: RangeInclusive<u64>,
89        page: &'a Page<Self::Cursor>,
90        filter: &'a Self::Filter,
91    ) -> impl Future<Output = Result<LedgerPage<Self::ScanItem, Self::Cursor>, RpcError>> + Send + 'a;
92
93    /// The last checkpoint a resume cursor shows the client has seen any of, if it shows one.
94    fn resume_from(cursor: &Self::Cursor) -> Option<u64>;
95
96    /// Where a scan after `cursor` starts: every item before this position comes at or before the
97    /// cursor.
98    fn resume_position(cursor: &Self::Cursor) -> Self::Position;
99
100    /// The position at the start of `checkpoint`, before its first item.
101    fn checkpoint_start(checkpoint: u64) -> Self::Position;
102
103    /// Build the GraphQL node from one scanned item's payload, which sits in `checkpoint`,
104    /// constructing whatever scope the feed resolves against. The driver assembles the edge from
105    /// the item's cursor and this node.
106    fn build_node(
107        caches: &Arc<StreamedCaches>,
108        resolver_limits: &sui_package_resolver::Limits,
109        checkpoint: u64,
110        payload: &Self::ScanItem,
111    ) -> Result<Self::Item, RpcError>;
112
113    /// The filter-matching items in one live checkpoint, each with its cursor, in delivery order.
114    fn matching_items(
115        checkpoint: &Arc<ProcessedCheckpoint>,
116        caches: &Arc<StreamedCaches>,
117        resolver_limits: &sui_package_resolver::Limits,
118        filter: &Self::Filter,
119    ) -> Result<LiveMatches<Self>, RpcError>;
120
121    /// Assemble one delivered payload from its cursor and node: a match passes the node, a progress
122    /// payload passes `None`. A feed that does not opt into progress is only ever called with `Some`.
123    fn output(cursor: String, node: Option<Self::Item>) -> Self::Output;
124
125    /// The cursor marking the end of `checkpoint`, minted for a live progress payload: at the start
126    /// of the next checkpoint, so a scan starting after it covers everything from there on.
127    /// Returning `None` suppresses live progress for the feed; backfill progress is gated
128    /// separately, by the per-subscription opt-in.
129    fn boundary_after(_checkpoint: &ProcessedCheckpoint) -> Option<Self::Cursor> {
130        None
131    }
132
133    /// The feed's name, used to label its lifecycle metrics.
134    fn subscription_type() -> &'static str;
135}

137/// One backfill scan output: either a match at the checkpoint its payload names, or a coverage
138/// marker at a page's coverage claim, at or below which every match has been scanned (advances the
139/// handoff even on a match-less page), carrying the page's end cursor, the resume point past
140/// everything scanned so far.
141enum Scanned<C, I> {
142    Match { checkpoint: u64, cursor: C, node: I },
143    Covered { checkpoint: u64, cursor: C },
144}
145
146/// Where the backfill scan hands off to the live broadcast. The scan delivers matches at or below
147/// the pinned handoff; live delivers the rest, from the checkpoint after it.
148struct Handoff {
149    /// Pin once the scan frontier is within this many checkpoints of the streamed tip.
150    threshold: u64,
151    pinned: Option<u64>,
152}
153
154/// What the backfill does with one scan output, as decided by [`Handoff::step`].
155enum ScanStep<C, I> {
156    /// Deliver this match.
157    Deliver { cursor: C, node: I },
158    /// Keep scanning.
159    Continue,
160    /// The scan has covered the handoff; live takes over.
161    Done,
162}
163
164impl<C, I> Scanned<C, I> {
165    /// The checkpoint this output sits at (the match's, or the page's coverage claim).
166    fn checkpoint(&self) -> u64 {
167        match self {
168            Scanned::Match { checkpoint, .. } | Scanned::Covered { checkpoint, .. } => *checkpoint,
169        }
170    }
171}

173/// Tracks what the client holds, and decides when the driver emits a progress payload.
174///
175/// The committed position is where the client's latest cursor resumes: every match before it has
176/// been delivered. A payload, match or progress, goes out only when its cursor resumes past the
177/// committed position, which it then becomes. So resuming from the latest cursor neither skips nor
178/// repeats, and the client's resume point never moves backward. Live drops what the backfill
179/// already delivered, or what a progress payload already covered, by position.
180///
181/// Backfill and live produce progress differently. Backfill reuses each scan page's own end cursor
182/// at the page boundary, so it needs no pacing. Live has no scan cursor to reuse: it mints one, and
183/// only after a run of checkpoints that matched nothing, so a sparse filter does not emit a payload
184/// on every checkpoint.
185struct DeliveryTracker<P> {
186    /// Whether this subscription opted into progress payloads. When false, no progress is emitted
187    /// on either phase and every payload is a match.
188    progress_enabled: bool,
189    /// Emit live progress after this many consecutive live checkpoints with no match (0 disables
190    /// live progress; backfill is unaffected).
191    unmatched_threshold: u64,
192    /// Consecutive live checkpoints with no match since the last match or progress payload.
193    consecutive_unmatched: u64,
194    /// Where the client's latest cursor resumes, or `None` before it holds one.
195    committed: Option<P>,
196}
197
198impl<P: Ord + Copy> DeliveryTracker<P> {
199    fn new(progress_enabled: bool, unmatched_threshold: u64, committed: Option<P>) -> Self {
200        Self {
201            progress_enabled,
202            unmatched_threshold,
203            consecutive_unmatched: 0,
204            committed,
205        }
206    }
207
208    /// Commit a match whose cursor resumes at `position`, or return `false` if the client already
209    /// holds it. A committed match ends the run of unmatched live checkpoints.
210    fn commit_match(&mut self, position: P) -> bool {
211        let advanced = self.advance(position);
212        if advanced {
213            self.consecutive_unmatched = 0;
214        }
215        advanced
216    }
217
218    /// Commit the end of a backfill page, whose cursor resumes at `position`, as progress. Returns
219    /// `false` when progress is off, or when the client already holds a cursor at or past it (a full
220    /// page ends on its last match, so it emits nothing).
221    fn commit_page_end(&mut self, position: P) -> bool {
222        self.progress_enabled && self.advance(position)
223    }
224
225    /// Commit progress at `position`, the boundary after a live checkpoint that delivered no match,
226    /// only once the unmatched run reaches the threshold. Returns whether to emit it.
227    fn commit_unmatched(&mut self, position: P) -> bool {
228        if !self.progress_enabled || self.unmatched_threshold == 0 {
229            return false;
230        }
231        self.consecutive_unmatched += 1;
232        if self.consecutive_unmatched < self.unmatched_threshold {
233            return false;
234        }
235        self.consecutive_unmatched = 0;
236        self.advance(position)
237    }
238
239    /// Move the committed position to `position`, if it is past it.
240    fn advance(&mut self, position: P) -> bool {
241        if self
242            .committed
243            .is_some_and(|committed| position <= committed)
244        {
245            return false;
246        }
247        self.committed = Some(position);
248        true
249    }
250}

252impl Handoff {
253    fn new(threshold: u64) -> Self {
254        Self {
255            threshold,
256            pinned: None,
257        }
258    }
259
260    /// Decide what to do with one scan output, given this replica's streamed `tip`.
261    ///
262    /// Pins first if the scan frontier is within the threshold of `tip`. `resubscribe` must
263    /// subscribe the live receiver and then return the streamed tip, so live sees every checkpoint
264    /// above the pin. A match past the handoff belongs to live, so the scan stops there.
265    fn step<C, I>(
266        &mut self,
267        scanned: Scanned<C, I>,
268        tip: u64,
269        resubscribe: impl FnOnce() -> u64,
270    ) -> ScanStep<C, I> {
271        if self.pinned.is_none() && tip.saturating_sub(scanned.checkpoint()) <= self.threshold {
272            self.pinned = Some(resubscribe());
273        }
274
275        match scanned {
276            Scanned::Match {
277                checkpoint,
278                cursor,
279                node,
280            } => {
281                if self.pinned.is_none_or(|h| checkpoint <= h) {
282                    ScanStep::Deliver { cursor, node }
283                } else {
284                    ScanStep::Done
285                }
286            }
287            Scanned::Covered { checkpoint, .. } => {
288                if self.pinned.is_some_and(|h| checkpoint >= h) {
289                    ScanStep::Done
290                } else {
291                    ScanStep::Continue
292                }
293            }
294        }
295    }
296
297    /// The last checkpoint the scan was responsible for; live delivers from the next one.
298    fn pinned(&self) -> Option<u64> {
299        self.pinned
300    }
301}

303/// Whether a subscription resuming after `checkpoint` starts more than `max_ahead` checkpoints past
304/// the streamed `tip`. A resume point that names no checkpoint is never too far ahead.
305fn is_too_far_ahead(checkpoint: Option<u64>, tip: u64, max_ahead: u64) -> bool {
306    checkpoint.is_some_and(|checkpoint| checkpoint > tip.saturating_add(max_ahead))
307}

309/// Subscribe to items matching `filter`: backfill from `resume` toward the tip, then follow live.
310/// The handoff is pinned mid-scan, once the scan frontier comes within `handoff_threshold` of the
311/// tip, so even a deep backfill catches up within one connection instead of lagging the receiver.
312#[allow(clippy::type_complexity)]
313pub(super) fn subscribe<S: Subscribable>(
314    reader: LedgerGrpcReader,
315    broadcast: Arc<SubscriptionBroadcast>,
316    caches: Arc<StreamedCaches>,
317    resolver_limits: sui_package_resolver::Limits,
318    mut watermarks_rx: watch::Receiver<Arc<Watermarks>>,
319    filter: S::Filter,
320    after: Option<S::Cursor>,
321    after_checkpoint: Option<u64>,
322    send_progress: bool,
323    subscriber_limit: SubscriberLimit,
324    config: SubscriptionConfig,
325) -> Result<impl Stream<Item = Result<S::Output, RpcError>>, RpcError<Error>> {
326    // The last checkpoint the client has seen any of, when it has seen one: the later of the
327    // cursor's and `afterCheckpoint`.
328    let resume_checkpoint = after
329        .as_ref()
330        .and_then(S::resume_from)
331        .max(after_checkpoint);
332
333    // Reject a resume point more than `max_ahead` checkpoints past the streamed tip before
334    // admitting, so bad input never claims a slot or lands in the subscription lifecycle metrics.
335    // There is nothing to backfill ahead of the tip, so such a request would only wait for the
336    // chain to reach it, and a far-future one would hold the connection open indefinitely.
337    let max_ahead = config.max_start_checkpoints_ahead_of_tip;
338    if is_too_far_ahead(resume_checkpoint, broadcast.network_tip(), max_ahead) {
339        return Err(bad_user_input(Error::TooFarAheadOfTip { max: max_ahead }));
340    }
341
342    // Claim a concurrency slot for the subscription's lifetime; at capacity this is `Err` with no
343    // guard created (and so no lifecycle metric).
344    let guard = SubscriptionLifecycleGuard::new(
345        S::subscription_type(),
346        broadcast.metrics(),
347        &subscriber_limit,
348    )
349    .map_err(upcast)?;
350
351    // Size the backfill scan page to the resolve concurrency. Scans are sequential (each needs the
352    // previous page's cursor), so feeding one window of `n` concurrent resolutions takes ceil(n /
353    // page) scans: a page much smaller than the concurrency makes scanning the bottleneck, a much
354    // larger one just holds a bigger page in memory. Matching them is roughly one scan per window.
355    let scan_page_size = config.max_concurrent_resolutions;
356
357    // Pin the handoff once the scan comes within half the live buffer of the tip, leaving room for
358    // checkpoints that arrive during the handoff so the receiver does not lag.
359    let handoff_threshold = config.broadcast_buffer as u64 / 2;
360
361    // Progress payloads are opt-in per subscription; once enabled, this threshold paces how often
362    // live progress fires.
363    let unmatched_threshold = config.unmatched_checkpoints_before_progress;
364
365    // The client holds every match before the later of its cursor and `afterCheckpoint`. With
366    // neither, it holds nothing and the subscription is live-only.
367    let committed = after
368        .as_ref()
369        .map(S::resume_position)
370        .max(after_checkpoint.map(|checkpoint| S::checkpoint_start(checkpoint.saturating_add(1))));
371
372    Ok(stream! {
373        let mut pending_receiver = None;
374        let mut handoff = Handoff::new(handoff_threshold);
375        let mut tracker = DeliveryTracker::new(send_progress, unmatched_threshold, committed);
376        let mut last_checkpoint: Option<u64> = None;
377
378        // Phase 1: scan toward the tip, pinning the live receiver near it. `after` and/or
379        // `afterCheckpoint` resume the backfill; neither means live-only.
380        if committed.is_some() {
381            // The resume point can be ahead of this server's watermarks (a cursor minted from the
382            // live stream, or by a more caught-up server). Waiting for them to reach it raises the
383            // ledger readers' monotonic-read floor past it, so no replica scans from a view behind
384            // the resume point. `scan_page` then waits for each page's end the same way, keeping
385            // the floor at or above every later resume point.
386            if let Some(checkpoint) = resume_checkpoint
387                && let Err(e) = wait_for_pipelines_catching_up_at(checkpoint, &mut watermarks_rx).await
388            {
389                guard.terminate(SubscriptionTerminationReason::BackfillError);
390                yield Err(RpcError::from(e));
391                return;
392            }
393
394            let checkpoint_lo = after_checkpoint.map_or(0, |cp| cp.saturating_add(1));
395            let scan = backfill::<S>(
396                reader,
397                caches.clone(),
398                resolver_limits.clone(),
399                watermarks_rx,
400                filter.clone(),
401                after,
402                checkpoint_lo,
403                scan_page_size,
404            );
405            for await scanned in scan {
406                let scanned = match scanned {
407                    Ok(scanned) => scanned,
408                    Err(e) => {
409                        guard.terminate(SubscriptionTerminationReason::BackfillError);
410                        yield Err(e);
411                        return;
412                    }
413                };
414
415                // A page's end is progress once every match on it has been delivered. A match past
416                // the handoff ends the scan before its coverage marker is reached, so `page_end`
417                // for this page would not run; that is fine, live resumes from the handoff.
418                //
419                // Consecutive match-less pages each send one, uncoalesced: at most one small payload
420                // per ledger request (which the scan already paces, waiting at the tip and never
421                // resending an end that did not move), each moving a deep backfill's resume point on.
422                let page_end = match &scanned {
423                    Scanned::Covered { cursor, .. } => Some(cursor.clone()),
424                    Scanned::Match { .. } => None,
425                };
426
427                let step = handoff.step(scanned, broadcast.network_tip(), || {
428                    pending_receiver = Some(broadcast.broadcaster().resubscribe());
429                    broadcast.network_tip()
430                });
431                if let Some(end_cursor) = page_end
432                    && tracker.commit_page_end(S::resume_position(&end_cursor))
433                {
434                    yield Ok(S::output(end_cursor.encode_cursor(), None));
435                }
436                match step {
437                    ScanStep::Deliver { cursor, node } => {
438                        if tracker.commit_match(S::resume_position(&cursor)) {
439                            yield Ok(S::output(cursor.encode_cursor(), Some(node)));
440                            guard.record_backfill_delivered();
441                        }
442                    }
443                    ScanStep::Continue => {}
444                    ScanStep::Done => break,
445                }
446            }
447            last_checkpoint = handoff.pinned();
448        }
449
450        // Phase 2: follow live from `handoff + 1` (a fresh receiver if there was no backfill).
451        let receiver = pending_receiver.unwrap_or_else(|| broadcast.broadcaster().resubscribe());
452        for await delivery in
453            live::<S>(receiver, last_checkpoint, caches, resolver_limits, filter, tracker, guard)
454        {
455            yield delivery;
456        }
457    })
458}

460/// Scan matches from the resume point toward the tip, yielding each match and a per-page coverage
461/// marker. Open-ended: the caller stops it once the handoff is covered.
462fn backfill<S: Subscribable>(
463    reader: LedgerGrpcReader,
464    caches: Arc<StreamedCaches>,
465    resolver_limits: sui_package_resolver::Limits,
466    mut watermarks_rx: watch::Receiver<Arc<Watermarks>>,
467    filter: S::Filter,
468    mut after: Option<S::Cursor>,
469    checkpoint_lo: u64,
470    scan_page_size: usize,
471) -> impl Stream<Item = Result<Scanned<S::Cursor, S::Item>, RpcError>> {
472    stream! {
473        let limits = PageLimits {
474            default: scan_page_size as u32,
475            max: scan_page_size as u32,
476        };
477
478        // Resume from whichever is later: the `after` cursor or `[checkpoint_lo, ..]`. The upper
479        // bound is `u64::MAX` because backfill is open-ended, chasing the moving tip; `subscribe`
480        // stops it once the handoff is covered.
481        let cp_bounds = checkpoint_lo..=u64::MAX;
482
483        loop {
484            let (items, next_after) = match scan_page::<S>(
485                &reader,
486                &caches,
487                &resolver_limits,
488                &limits,
489                cp_bounds.clone(),
490                after.as_ref(),
491                &filter,
492                &mut watermarks_rx,
493            )
494            .await
495            {
496                Ok(page) => page,
497                Err(e) => {
498                    yield Err(e);
499                    return;
500                }
501            };
502            for item in items {
503                yield Ok(item);
504            }
505            after = Some(next_after);
506        }
507    }
508}

510/// Whether a scan page that ended at `end` moved past `after`, the cursor it started from. A page
511/// that scanned nothing new (at the ledger's tip, or past it, where the ledger echoes `after`) ends
512/// where `after` resumes.
513fn moved_past<S: Subscribable>(end: &S::Cursor, after: &S::Cursor) -> bool {
514    S::resume_position(end) > S::resume_position(after)
515}

517/// Follow the live broadcast from `last_checkpoint + 1`, delivering matching items the client does
518/// not already hold. Drops the one-checkpoint seam overlap (the resubscribe/tip race), gap-checks,
519/// and disconnects on anomalies.
520fn live<S: Subscribable>(
521    mut receiver: CheckpointBroadcaster,
522    mut last_checkpoint: Option<u64>,
523    caches: Arc<StreamedCaches>,
524    resolver_limits: sui_package_resolver::Limits,
525    filter: S::Filter,
526    mut tracker: DeliveryTracker<S::Position>,
527    guard: SubscriptionLifecycleGuard,
528) -> impl Stream<Item = Result<S::Output, RpcError>> {
529    stream! {
530        let mut delivered_live = false;
531        loop {
532            match receiver.recv().await {
533                Ok(checkpoint) => {
534                    let seq = checkpoint.summary.sequence_number;
535                    if let Some(last) = last_checkpoint {
536                        // Already covered by the scan (resubscribe/tip overlap): skip.
537                        if seq <= last {
538                            continue;
539                        }
540                        if seq > last + 1 {
541                            warn!(
542                                last_checkpoint = last,
543                                received = seq,
544                                "Unexpected gap between scan and live; disconnecting"
545                            );
546                            guard.terminate(SubscriptionTerminationReason::UnexpectedGap);
547                            yield Err(reconnect_error());
548                            return;
549                        }
550                    }
551
552                    // Deliver each matching item as its own payload, ordered within the checkpoint,
553                    // skipping those the client already holds: the rest of a resume cursor's
554                    // checkpoint up to the cursor, or what backfill progress past the handoff
555                    // covered. Checkpoints that deliver nothing may advance the resume point as
556                    // progress.
557                    let items = match S::matching_items(&checkpoint, &caches, &resolver_limits, &filter) {
558                        Ok(items) => items,
559                        Err(e) => {
560                            guard.terminate(SubscriptionTerminationReason::Error);
561                            yield Err(e);
562                            return;
563                        }
564                    };
565                    let mut delivered_match = false;
566                    for (cursor, node) in items {
567                        if !tracker.commit_match(S::resume_position(&cursor)) {
568                            continue;
569                        }
570                        delivered_match = true;
571                        yield Ok(S::output(cursor.encode_cursor(), Some(node)));
572                        guard.record_delivered(checkpoint.summary.timestamp_ms);
573                    }
574
575                    if !delivered_match
576                        && let Some(boundary) = S::boundary_after(&checkpoint)
577                        && tracker.commit_unmatched(S::resume_position(&boundary))
578                    {
579                        // If the feed sends progress and the unmatched run is long enough, emit one
580                        // progress payload at the boundary past this checkpoint, rather than one on
581                        // every match-less checkpoint.
582                        yield Ok(S::output(boundary.encode_cursor(), None));
583                    }
584                    last_checkpoint = Some(seq);
585                    delivered_live = true;
586                }
587                // A lag before the first live checkpoint is catch-up overflow (likely kv-rpc lag).
588                Err(broadcast::error::RecvError::Lagged(missed)) if !delivered_live => {
589                    warn!(missed, "Subscriber fell behind during catch-up; disconnecting");
590                    guard.terminate(SubscriptionTerminationReason::Lagged);
591                    yield Err(reconnect_error());
592                    return;
593                }
594                Err(e) => {
595                    guard.terminate(SubscriptionTerminationReason::from_recv_error(&e));
596                    yield Err(broadcast_error(e));
597                    return;
598                }
599            }
600        }
601    }
602}

604/// Retry policy for the backfill scan: jittered exponential backoff, giving up after 60s (past which
605/// a failure is unlikely to be a transient blip).
606fn scan_backoff() -> ExponentialBackoff {
607    ExponentialBackoff {
608        initial_interval: Duration::from_millis(100),
609        max_interval: Duration::from_secs(5),
610        max_elapsed_time: Some(Duration::from_secs(60)),
611        ..Default::default()
612    }
613}

615/// Paces re-requests while the ledger has nothing past the scan's position: waits double from
616/// `BACKFILL_POLL_INTERVAL` up to `BACKFILL_POLL_MAX_INTERVAL`, and once `BACKFILL_POLL_DEADLINE`
617/// has passed without progress the client is asked to reconnect.
618struct LedgerPoll {
619    interval: Duration,
620    deadline: tokio::time::Instant,
621}
622
623impl LedgerPoll {
624    fn new() -> Self {
625        Self {
626            interval: BACKFILL_POLL_INTERVAL,
627            deadline: tokio::time::Instant::now() + BACKFILL_POLL_DEADLINE,
628        }
629    }
630
631    async fn wait(&mut self) -> Result<(), RpcError> {
632        if tokio::time::Instant::now() >= self.deadline {
633            warn!("Ledger made no progress within the backfill deadline; disconnecting");
634            return Err(reconnect_error());
635        }
636        tokio::time::sleep(self.interval).await;
637        self.interval = (self.interval * 2).min(BACKFILL_POLL_MAX_INTERVAL);
638        Ok(())
639    }
640}

642/// Scan the next page of matches, waiting at the indexer tip if nothing is ready yet. Returns the
643/// matches, a trailing coverage marker, and the cursor to resume from.
644async fn scan_page<S: Subscribable>(
645    reader: &LedgerGrpcReader,
646    caches: &Arc<StreamedCaches>,
647    resolver_limits: &sui_package_resolver::Limits,
648    limits: &PageLimits,
649    cp_bounds: RangeInclusive<u64>,
650    after: Option<&S::Cursor>,
651    filter: &S::Filter,
652    watermarks_rx: &mut watch::Receiver<Arc<Watermarks>>,
653) -> Result<(Vec<Scanned<S::Cursor, S::Item>>, S::Cursor), RpcError> {
654    let mut poll = LedgerPoll::new();
655    loop {
656        let result = scan_with_retry::<S>(reader, limits, cp_bounds.clone(), after, filter).await?;
657
658        // No end cursor means nothing past the indexer tip is indexed yet; wait, then re-request. A
659        // range that was scanned but matched nothing still returns a boundary cursor, not `None`.
660        let Some(end_cursor) = result.last_cursor else {
661            poll.wait().await?;
662            continue;
663        };
664
665        // A page that did not move past where it started: wait for the ledger to index more before
666        // asking again, rather than re-polling at full speed.
667        if after.is_some_and(|after| !moved_past::<S>(&end_cursor, after)) {
668            poll.wait().await?;
669            continue;
670        }
671
672        // Hold the page until all pipelines have indexed through the checkpoint its end cursor
673        // resumes in, which is at or past its coverage claim. The delivered items' nested queries
674        // resolve against indexed data, and the ledger readers' floor reaches the next page's
675        // resume point.
676        if let Some(checkpoint) = S::resume_from(&end_cursor) {
677            wait_for_pipelines_catching_up_at(checkpoint, watermarks_rx).await?;
678        }
679
680        // A forward page keeps `items` in ascending order, one delivered edge each.
681        let mut items = Vec::with_capacity(result.items.len() + 1);
682        for item in result.items {
683            items.push(Scanned::Match {
684                checkpoint: item.checkpoint,
685                node: S::build_node(caches, resolver_limits, item.checkpoint, &item.payload)?,
686                cursor: item.cursor,
687            });
688        }
689
690        // Coverage marker at the page's coverage claim: advances the handoff even on a match-less
691        // page, and carries the page's end cursor so the backfill can deliver it as progress. A
692        // page without a claim (some ledger servers send none for a page that scanned nothing)
693        // adds no marker; a later page's claim covers it.
694        if let Some(checkpoint) = result.covered_checkpoint {
695            items.push(Scanned::Covered {
696                checkpoint,
697                cursor: end_cursor.clone(),
698            });
699        }
700
701        return Ok((items, end_cursor));
702    }
703}
704
705/// Scan one page over `cp_bounds`, retrying transient failures under a bounded budget.
706async fn scan_with_retry<S: Subscribable>(
707    reader: &LedgerGrpcReader,
708    limits: &PageLimits,
709    cp_bounds: RangeInclusive<u64>,
710    after: Option<&S::Cursor>,
711    filter: &S::Filter,
712) -> Result<LedgerPage<S::ScanItem, S::Cursor>, RpcError> {
713    let page = Page::from_params(
714        limits,
715        Some(limits.default as u64),
716        after.cloned(),
717        None,
718        None,
719    )
720    .map_err(anyhow::Error::from)?;
721
722    backoff::future::retry(scan_backoff(), || async {
723        S::scan(reader, cp_bounds.clone(), &page, filter)
724            .await
725            .map_err(|e| {
726                warn!(error = ?e, "scan failed, retrying");
727                backoff::Error::transient(e)
728            })
729    })
730    .await
731}

759    /// Runs scan outputs through [`Handoff::step`] against a fixed streamed `tip`, as the backfill
760    /// loop in `subscribe` does. Returns the delivered match checkpoints and the checkpoint live
761    /// starts after (`None` if still scanning).
762    fn drive(threshold: u64, tip: u64, outputs: &[Out]) -> (Vec<u64>, Option<u64>) {
763        let mut handoff = Handoff::new(threshold);
764        let mut delivered = vec![];
765        for out in outputs {
766            // Matches carry their checkpoint as the node, so delivery order is observable.
767            let scanned = match *out {
768                Out::Match(checkpoint) => Scanned::Match {
769                    checkpoint,
770                    cursor: (),
771                    node: checkpoint,
772                },
773                Out::End(checkpoint) => Scanned::Covered {
774                    checkpoint,
775                    cursor: (),
776                },
777            };
778            match handoff.step(scanned, tip, || tip) {
779                ScanStep::Deliver { node, .. } => delivered.push(node),
780                ScanStep::Continue => {}
781                ScanStep::Done => return (delivered, handoff.pinned()),
782            }
783        }
784        (delivered, None)
785    }
