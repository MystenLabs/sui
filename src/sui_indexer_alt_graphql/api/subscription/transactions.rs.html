4//! Transactions as a scan-then-live feed: each matching transaction is delivered as its own edge,
5//! ordered by checkpoint then by position within the checkpoint. The cursor places the transaction
6//! by checkpoint and index within it, and matching tests a transaction's contents against the
7//! filter.
8
9use std::future::Future;
10use std::ops::RangeInclusive;
11use std::sync::Arc;
12

17use sui_rpc::proto::sui::rpc::v2;

36impl Subscribable for Transaction {
37    type Item = Self;
38    type Cursor = CTransaction;
39    type Filter = TransactionFilter;
40    type ScanItem = v2::ExecutedTransaction;
41    type Output = TransactionSubscriptionEdge;
42
43    fn subscription_type() -> &'static str {
44        "transactions"
45    }
46
47    fn output(cursor: String, node: Option<Transaction>) -> TransactionSubscriptionEdge {
48        TransactionSubscriptionEdge { cursor, node }
49    }
50
51    fn scan<'a>(
52        reader: &'a LedgerGrpcReader,
53        cp_bounds: RangeInclusive<u64>,
54        page: &'a Page<CTransaction>,
55        filter: &'a TransactionFilter,
56    ) -> impl Future<Output = Result<LedgerPage<v2::ExecutedTransaction, CTransaction>, RpcError>>
57    + Send
58    + 'a {
59        Transaction::scan_grpc(reader, cp_bounds, page, page.limit(), filter)
60    }
61
62    fn resume_from(cursor: &CTransaction) -> Option<u64> {
63        cursor.last_checkpoint()
64    }
65
66    fn made_progress(end: &CTransaction, after: &CTransaction, reached_tip: bool) -> bool {
67        ledger_scan_moved(end, after, reached_tip)
68    }
69
70    fn build_node(
71        caches: &Arc<StreamedCaches>,
72        resolver_limits: &sui_package_resolver::Limits,
73        checkpoint: u64,
74        payload: &v2::ExecutedTransaction,
75    ) -> Result<Self, RpcError> {
76        let scope = Scope::for_backfill(caches.clone(), resolver_limits.clone(), checkpoint);
77        transaction_from_stream_item(scope, payload)
78    }
79
80    fn matching_edges(
81        checkpoint: &Arc<ProcessedCheckpoint>,
82        caches: &Arc<StreamedCaches>,
83        resolver_limits: &sui_package_resolver::Limits,
84        filter: &TransactionFilter,
85    ) -> Result<Vec<Edge<String, Self, EmptyFields>>, RpcError> {
86        let scope = Scope::for_streamed_checkpoint(
87            caches.clone(),
88            resolver_limits.clone(),
89            checkpoint.clone(),
90        );
91        let mut edges = Vec::new();
92        // `transactions` holds every transaction in the checkpoint, in order.
93        for (index, tx) in checkpoint.transactions.iter().enumerate() {
94            if !filter.matches(&tx.contents) {
95                continue;
96            }
97            let transaction = Transaction::with_contents(scope.clone(), tx.contents.clone())?;
98            let cursor = TransactionToken::at(TransactionKey {
99                checkpoint: checkpoint.summary.sequence_number,
100                transaction_index: index as u64,
101            })
102            .encode_cursor();
103            edges.push(Edge::new(cursor, transaction));
104        }
105        Ok(edges)
106    }
107
108    fn boundary_after(checkpoint: &ProcessedCheckpoint) -> Option<CTransaction> {
109        Some(TransactionToken::checkpoint_start(
110            checkpoint.summary.sequence_number + 1,
111        ))
112    }
113}
