20/// Input for withdrawing funds from an accumulator.
21#[derive(SimpleObject, Clone)]
22pub struct BalanceWithdraw {
23    /// How much to withdraw from the accumulator.
24    pub reservation: Option<WithdrawalReservation>,
25
26    /// The type of the funds accumulator to withdraw from (e.g. `0x2::balance::Balance<0x2::sui::SUI>`).
27    #[graphql(name = "type")]
28    pub type_: Option<MoveType>,
29
30    /// The account to withdraw funds from.
31    pub withdraw_from: Option<WithdrawFrom>,
32
33    /// The account whose funds are withdrawn, if `withdrawFrom` is `SENDER_ALLOWANCE`.
34    pub funder: Option<Address>,
35
36    /// The allowance authorizing the withdrawal, if `withdrawFrom` is `SENDER_ALLOWANCE`. Use `asTransactionObject` to see it as this transaction referenced it.
37    pub allowance: Option<Address>,
38}
39
40/// Reservation details for a withdrawal.
41#[derive(Union, Clone)]
42pub enum WithdrawalReservation {
43    MaxAmountU64(WithdrawMaxAmountU64),
44}
45
46#[derive(SimpleObject, Clone)]
47pub struct WithdrawMaxAmountU64 {
48    pub amount: Option<BigInt>,
49}
50
51/// The account to withdraw funds from.
52#[derive(Enum, Copy, Clone, Eq, PartialEq)]
53pub enum WithdrawFrom {
54    /// The funds are withdrawn from the transaction sender's account.
55    Sender,
56    /// The funds are withdrawn from the sponsor's account.
57    Sponsor,
58    /// The funds are withdrawn from the account of an allowance's funder, under an allowance granted to the transaction sender. See `BalanceWithdraw.funder` and `BalanceWithdraw.allowance`.
59    SenderAllowance,
60}
61
62impl BalanceWithdraw {
63    pub fn from_native(withdrawal: NativeFundsWithdrawalArg, scope: Scope) -> Self {
64        let NativeFundsWithdrawalArg {
65            reservation,
66            type_arg,
67            withdraw_from,
68        } = withdrawal;
69
70        let reservation = Some(match reservation {
71            NativeReservation::MaxAmountU64(amount) => {
72                WithdrawalReservation::MaxAmountU64(WithdrawMaxAmountU64 {
73                    amount: Some(amount.into()),
74                })
75            }
76        });
77
78        let (withdraw_from, funder, allowance) = match withdraw_from {
79            NativeWithdrawFrom::Sender => (WithdrawFrom::Sender, None, None),
80            NativeWithdrawFrom::Sponsor => (WithdrawFrom::Sponsor, None, None),
81            NativeWithdrawFrom::SenderAllowance { funder, allowance } => (
82                WithdrawFrom::SenderAllowance,
83                Some(Address::with_address(scope.without_root_bound(), funder)),
84                Some(Address::with_address(
85                    scope.without_root_bound(),
86                    allowance.into(),
87                )),
88            ),
89        };
90
91        let type_ = {
92            let NativeWithdrawalTypeArg::Balance(t) = type_arg;
93            let balance_struct = StructInput {
94                address: AccountAddress::TWO,
95                module: "balance".to_string(),
96                name: "Balance".to_string(),
97                type_params: vec![t.into()],
98            };
99            Some(MoveType::from_input(
100                TypeInput::Struct(Box::new(balance_struct)),
101                scope,
102            ))
103        };
104
105        Self {
106            reservation,
107            type_,
108            withdraw_from: Some(withdraw_from),
109            funder,
110            allowance,
111        }
112    }
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118
119    use async_graphql::ScalarType;
120    use sui_types::base_types::ObjectID;
121    use sui_types::base_types::SuiAddress as NativeSuiAddress;
122    use sui_types::gas_coin::GAS;
123    use sui_types::transaction::FundsWithdrawalArg;
124    use sui_types::transaction::Reservation as NativeReservation;
125    use sui_types::transaction::WithdrawFrom as NativeWithdrawFrom;
126    use sui_types::transaction::WithdrawalTypeArg as NativeWithdrawalTypeArg;
127
128    #[test]
129    fn test_from_native() {
130        let withdrawal = FundsWithdrawalArg {
131            reservation: NativeReservation::MaxAmountU64(42),
132            type_arg: NativeWithdrawalTypeArg::Balance(GAS::type_tag()),
133            withdraw_from: NativeWithdrawFrom::Sender,
134        };
135        let expected_type_tag = withdrawal.type_arg.to_type_tag();
136
137        let withdraw = BalanceWithdraw::from_native(withdrawal, Scope::for_tests());
138
139        let Some(WithdrawalReservation::MaxAmountU64(reservation)) = withdraw.reservation else {
140            panic!("expected MaxAmountU64 reservation");
141        };
142        assert_eq!(
143            reservation.amount.as_ref().map(|a| {
144                let async_graphql::Value::String(s) = a.to_value() else {
145                    panic!("expected string value");
146                };
147                s
148            }),
149            Some("42".to_string())
150        );
151        assert!(matches!(withdraw.withdraw_from, Some(WithdrawFrom::Sender)));
152        assert!(withdraw.funder.is_none());
153        assert!(withdraw.allowance.is_none());
154
155        let type_tag = withdraw
156            .type_
157            .as_ref()
158            .and_then(|t| t.to_type_tag())
159            .unwrap();
160
161        assert_eq!(type_tag, expected_type_tag);
162    }
163
164    #[test]
165    fn test_from_native_allowance() {
166        let funder = NativeSuiAddress::random_for_testing_only();
167        let allowance = ObjectID::random();
168        let withdrawal =
169            FundsWithdrawalArg::balance_from_allowance(42, GAS::type_tag(), funder, allowance);
170
171        let withdraw = BalanceWithdraw::from_native(withdrawal, Scope::for_tests());
172
173        assert!(matches!(
174            withdraw.withdraw_from,
175            Some(WithdrawFrom::SenderAllowance)
176        ));
177        assert_eq!(withdraw.funder.map(|a| a.address), Some(funder));
178        assert_eq!(
179            withdraw.allowance.map(|a| a.address),
180            Some(allowance.into())
181        );
182    }
