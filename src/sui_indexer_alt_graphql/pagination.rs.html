26/// Configuration for page size limits, specifying a max multi-get size, as well as a default and
27/// max page size for each paginated fields. Page limits can be customized for specific fields,
28/// otherwise falling back to a blanket default.
29pub(crate) struct PaginationConfig {
30    /// Maximum number of keys that can be fetched in a single multi-get.
31    max_multi_get_size: u32,
32
33    /// Fallback page limit configuration.
34    fallback: PageLimits,
35
36    /// Type and field name-specific overrides for page limits.
37    overrides: BTreeMap<(&'static str, &'static str), PageLimits>,
38}
39
40/// The configuration for a single paginated field.
41pub(crate) struct PageLimits {
42    pub(crate) default: u32,
43    pub(crate) max: u32,
44}
45
46#[derive(Debug, Eq, PartialEq)]
47pub(crate) struct Page<C> {
48    /// The exclusive lower bound of the page (no bound means start from the beginning of the
49    /// data-set).
50    after: Option<C>,
51
52    /// The exclusive upper bound of the page (no bound means continue to the end of the data-set).
53    before: Option<C>,
54
55    /// Maximum number of entries in the page.
56    limit: u64,
57
58    /// In case there are more than `limit` entries in the range described by `(after, before)`,
59    /// this field states whether the entries up to limit are taken from the `Front` or `Back` of
60    /// that range.
61    end: End,
62}
63
64/// Whether the page is extracted from the beginning or the end of the range bounded by the cursors.
65#[derive(PartialEq, Eq, Debug, Clone, Copy)]
66pub(crate) enum End {
67    Front,
68    Back,
69}

71/// Connection for fields served by a gRPC stream. Unlike the stock `Connection`, pages may be
72/// partially filled, with `PageInfo` cursors decoupled from the edge list.
73pub(crate) struct StreamConnection<N: OutputType> {
74    pub edges: Vec<Edge<String, N, EmptyFields>>,
75    pub page_info: PageInfo,
76}

78#[derive(thiserror::Error, Debug, Clone)]
79pub(crate) enum Error {
80    #[error("Cannot provide both 'first' and 'last' parameters for connection")]
81    FirstAndLast,
82
83    #[error("Page size of {limit} exceeds max of {max} for connection")]
84    TooLarge { limit: u64, max: u32 },
85}

89        max_multi_get_size: u32,
90        fallback: PageLimits,

92    ) -> Self {
93        Self {

95            fallback,
96            overrides,
97        }
98    }

106    pub(crate) fn limits<'l, 't: 'l>(&'l self, type_: &'t str, name: &'t str) -> &'l PageLimits {

108    }

118impl<C> Page<C> {
119    /// Convert connection parameters into a page. Entries for the page are drawn from the range
120    /// `(after, before)` (Both bounds are optional). The number of entries in the page is
121    /// controlled by `first` and `last`.
122    ///
123    /// - Setting both is in an error.
124    /// - Setting `first` indicates that the entries are taken from the front of the range.
125    /// - Setting `last` indicates that the entries are taken from the end of the range.
126    /// - Setting neither defaults the limit to the default page size in `limits`, taken from the
127    ///   front of the range.
128    ///
129    /// It is an error to set a limit on page size that is greater than the `limit`'s max page
130    /// size.
131    pub(crate) fn from_params(
132        limits: &PageLimits,
133        first: Option<u64>,
134        after: Option<C>,
135        last: Option<u64>,
136        before: Option<C>,
137    ) -> Result<Self, Error> {
138        let page = match (first, after, last, before) {
139            (Some(_), _, Some(_), _) => return Err(Error::FirstAndLast),
140
141            (limit, after, None, before) => Page {
142                after,
143                before,
144                limit: limit.unwrap_or(limits.default as u64),
145                end: End::Front,
146            },
147
148            (None, after, Some(limit), before) => Page {
149                after,
150                before,
151                limit,
152                end: End::Back,
153            },
154        };
155
156        if page.limit > limits.max as u64 {
157            return Err(Error::TooLarge {
158                limit: page.limit,
159                max: limits.max,
160            });
161        }
162
163        Ok(page)
164    }
165
166    pub(crate) fn after(&self) -> Option<&C> {
167        self.after.as_ref()
168    }
169
170    pub(crate) fn before(&self) -> Option<&C> {
171        self.before.as_ref()
172    }
173
174    pub(crate) fn limit(&self) -> usize {
175        self.limit as usize
176    }
177
178    /// Returns the limit + 2 for has_previous_page and has_next_page pagination calculations
179    pub(crate) fn limit_with_overhead(&self) -> usize {
180        self.limit() + 2
181    }
182
183    pub(crate) fn is_from_front(&self) -> bool {
184        matches!(self.end, End::Front)
185    }
186
187    /// Direction for sorting SQL queries.
188    pub(crate) fn order_by_direction(&self) -> Query<'_> {
189        if self.is_from_front() {
190            query!("ASC")
191        } else {
192            query!("DESC")
193        }
194    }
195}

197impl Page<JsonCursor<usize>> {
198    /// Translate the cursors of this Page into a range of indices, in [0, total).
199    ///
200    /// Returns `None` if the cursors are inconsistent, otherwise returns Some(lo..hi) where `lo`
201    /// and `hi` are the inclusive lower and exclusive upper bounds of the range of indices
202    /// corresponding to the cursors for this page.
203    pub(crate) fn range(&self, total: usize) -> Option<Range<usize>> {
204        let mut lo = self.after().map_or(0, |a| a.saturating_add(1)).min(total);
205        let mut hi = self.before().map_or(total, |b| **b).min(total);
206
207        if hi <= lo {
208            return None;
209        }
210
211        if (hi - lo) > self.limit() {
212            if self.is_from_front() {
213                hi = lo + self.limit();
214            } else {
215                lo = hi - self.limit();
216            }
217        }
218
219        Some(lo..hi)
220    }
221
222    /// Treat the cursors of this Page as indices into a range [0, total).
223    ///
224    /// Returns a connection where the cursors correspond to a sub-range of indices and the
225    /// nodes are selected by calling `node` with each index in that sub-range.
226    pub(crate) fn paginate_indices<N: OutputType, E>(
227        &self,
228        total: usize,
229        node: impl Fn(usize) -> Result<N, E>,
230    ) -> Result<Connection<String, N>, E> {
231        let mut conn = Connection::new(false, false);
232        let Some(range) = self.range(total) else {
233            return Ok(conn);
234        };
235
236        conn.has_previous_page = 0 < range.start;
237        conn.has_next_page = range.end < total;
238        for i in range {
239            conn.edges
240                .push(Edge::new(JsonCursor::new(i).encode_cursor(), node(i)?));
241        }
242
243        Ok(conn)
244    }
245}

247impl<C: CursorType + Eq + PartialEq + Clone> Page<C> {
248    /// Process the results of a paginated query.
249    ///
250    /// `results` is expected to be the result of a query modified to fetch the page indicated by
251    /// `self`. This function determines whether those results are consistent with the cursors for
252    /// this page, and whether there are more results in the set being paginated, before or after
253    /// the current page.
254    ///
255    /// `results` should contain the elements of the page, in order, as well as at least one record
256    /// either side of the page, if there is one.
257    ///
258    /// `cursor` is a function that extracts the cursor from an element of `results`, and `node` is
259    /// a function that extracts the node. Returns a GraphQL `Connection` populated with edges
260    /// derived from `results`.
261    pub(crate) fn paginate_results<T, N: OutputType, E>(
262        &self,
263        results: Vec<T>,
264        cursor: impl Fn(&T) -> C,
265        node: impl Fn(T) -> Result<N, E>,
266    ) -> Result<Connection<String, N>, E> {
267        let edges: Vec<_> = results.into_iter().map(|r| (cursor(&r), r)).collect();
268        let first = edges.first().map(|(c, _)| c.clone());
269        let last = edges.last().map(|(c, _)| c.clone());
270
271        let (prev, next, prefix, suffix) =
272            match (self.after(), first, last, self.before(), self.end) {
273                // Results came back empty, despite supposedly including the `after` and `before`
274                // cursors, so the bounds must have been invalid, no matter which end the page was
275                // drawn from.
276                (_, None, _, _, _) | (_, _, None, _, _) => {
277                    return Ok(Connection::new(false, false));
278                }
279
280                // Page drawn from the front, and the cursor for the first element does not match
281                // `after`. This absence implies the bound was invalid, so we return an empty
282                // result.
283                (Some(a), Some(f), _, _, End::Front) if f != *a => {
284                    return Ok(Connection::new(false, false));
285                }
286
287                // Similar to above case, but for back of results.
288                (_, _, Some(l), Some(b), End::Back) if l != *b => {
289                    return Ok(Connection::new(false, false));
290                }
291
292                // From here onwards, we know that the results are non-empty. In the forward
293                // pagination scenario, the presence of a previous page is determined by whether a
294                // cursor supplied on the end the page is being drawn from is found in the first
295                // position. The presence of a next page is determined by whether we have more
296                // results than the provided limit, and/ or if the end cursor element appears in
297                // the result set.
298                (after, Some(f), Some(l), before, End::Front) => {
299                    let has_previous_page = after.is_some_and(|a| *a == f);
300                    let prefix = has_previous_page as usize;
301
302                    // If results end with the before cursor, we will at least need to trim one
303                    // element from the suffix and we trim more off the end if there is more after
304                    // applying the limit.
305                    let mut suffix = before.is_some_and(|b| *b == l) as usize;
306                    suffix += edges.len().saturating_sub(self.limit() + prefix + suffix);
307                    let has_next_page = suffix > 0;
308
309                    (has_previous_page, has_next_page, prefix, suffix)
310                }
311
312                // Symmetric to the previous case, but drawing from the back.
313                (after, Some(f), Some(l), before, End::Back) => {
314                    let has_next_page = before.is_some_and(|b| *b == l);
315                    let suffix = has_next_page as usize;
316
317                    let mut prefix = after.is_some_and(|a| *a == f) as usize;
318                    prefix += edges.len().saturating_sub(self.limit() + prefix + suffix);
319                    let has_previous_page = prefix > 0;
320
321                    (has_previous_page, has_next_page, prefix, suffix)
322                }
323            };
324
325        // If there are no elements left after trimming, then forget whether there's a previous or
326        // next page, because there will be no start or end cursor for this page to anchor on.
327        if edges.len() == prefix + suffix {
328            return Ok(Connection::new(false, false));
329        }
330
331        // We finally made it -- trim the prefix and suffix edges from the result and send it!
332        let mut edges = edges.into_iter();
333        if prefix > 0 {
334            edges.nth(prefix - 1);
335        }
336        if suffix > 0 {
337            edges.nth_back(suffix - 1);
338        }
339
340        let mut conn = Connection::new(prev, next);
341        for (c, n) in edges {
342            conn.edges.push(Edge::new(c.encode_cursor(), node(n)?));
343        }
344
345        Ok(conn)
346    }
347}

349impl<C> Page<C>
350where
351    C: CursorType + TryFrom<CursorToken, Error = anyhow::Error>,
352{
353    /// Translate a `StreamPage` into a `StreamConnection`. The server is expected to have already
354    /// applied page bounds and limit.
355    ///
356    /// `node` converts one stream payload into the connection's node type; cursors are minted
357    /// here, from the server's per-item tokens. Edges are returned in ascending order (the stream
358    /// emits descending order when paginating from the back).
359    pub(crate) fn paginate_stream_results<T, N: OutputType>(
360        &self,
361        result: StreamPage<T>,
362        node: impl Fn(&T) -> Result<N, RpcError>,
363    ) -> Result<StreamConnection<N>, RpcError> {
364        let more = result.has_more();
365        let start = result.first_cursor().cloned();
366        let end = result.last_cursor().cloned();
367        let mut items = result.items;
368
369        let (has_previous_page, has_next_page, start, end) = if self.is_from_front() {
370            (self.after().is_some(), more, start, end)
371        } else {
372            items.reverse();
373            (more, self.before().is_some(), end, start)
374        };
375
376        let mut edges = Vec::with_capacity(items.len());
377        for item in items {
378            edges.push(Edge::new(
379                Self::encode_stream_cursor(&item.cursor)?,
380                node(&item.payload)?,
381            ));
382        }
383
384        Ok(StreamConnection {
385            edges,
386            page_info: PageInfo {
387                has_previous_page,
388                has_next_page,
389                start_cursor: start.map(|b| Self::encode_stream_cursor(&b)).transpose()?,
390                end_cursor: end.map(|b| Self::encode_stream_cursor(&b)).transpose()?,
391            },
392        })
393    }
394
395    /// Re-encode a server-minted cursor (raw encoded `CursorToken` bytes from the gRPC stream) as a
396    /// GraphQL cursor string in this page's cursor format.
397    fn encode_stream_cursor(bytes: &[u8]) -> Result<String, RpcError> {
398        let token = CursorToken::decode(bytes).context("Failed to decode stream cursor")?;
399        let cursor = C::try_from(token).context("Unexpected position in stream cursor")?;
400        Ok(cursor.encode_cursor())
401    }
402}

405impl<N: OutputType> StreamConnection<N> {
406    /// Information to aid in pagination.
407    async fn page_info(&self) -> &PageInfo {
408        &self.page_info
409    }

411    /// A list of edges.

414    }

416    /// A list of nodes.

418        self.edges.iter().map(|e| &e.node).collect()
419    }

444impl<N: OutputType> From<Connection<String, N>> for StreamConnection<N> {
445    /// Convert a stock async-graphql `Connection` (as produced by `Page::paginate_results`) into
446    /// the custom shape. Cursors are derived from edges, matching stock semantics.
447    fn from(conn: Connection<String, N>) -> Self {
448        let start_cursor = conn.edges.first().map(|e| e.cursor.clone());
449        let end_cursor = conn.edges.last().map(|e| e.cursor.clone());
450        Self {
451            edges: conn.edges,
452            page_info: PageInfo {
453                has_previous_page: conn.has_previous_page,
454                has_next_page: conn.has_next_page,
455                start_cursor,
456                end_cursor,
457            },
458        }
459    }
460}
