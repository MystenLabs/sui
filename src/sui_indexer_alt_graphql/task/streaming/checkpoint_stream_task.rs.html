4//! Checkpoint stream task.
5//!
6//! Maintains the connection to the fullnode's `SubscribeCheckpoints` gRPC
7//! stream and broadcasts every checkpoint, in order, to GraphQL subscribers.
8//!
9//! ```text
10//!   loop forever:
11//!     connect (retry on transient errors, fail on auth/config)
12//!     for each incoming checkpoint:
13//!       if seq > last_broadcast + 1:
14//!         recover_gap(last_broadcast + 1 ..= seq - 1)
15//!       broadcast(checkpoint)
16//!     on stream end: reconnect
17//! ```
18//!
19//! # Reconnect
20//!
21//! gRPC connections drop for many reasons: rolling deploys, network blips,
22//! server restarts. When the stream errors or closes, the task reconnects
23//! with bounded exponential backoff. Transient errors retry indefinitely;
24//! permanent gRPC errors (auth, config) terminate the task.
25//!
26//! # Gap recovery
27//!
28//! `SubscribeCheckpoints` resumes at the current tip on reconnect, not where
29//! we left off. So after a drop, the first new message typically has a
30//! sequence number well past the last we broadcast — checkpoints produced
31//! during the drop are now invisible on the live stream.
32//!
33//! Detection happens on every incoming message: if `seq > last_broadcast + 1`,
34//! the task fetches the missing checkpoints from kv-rpc (a `LedgerService`
35//! endpoint serving historical checkpoints) and broadcasts them in order
36//! before broadcasting the live message.
37//!
38//! ```text
39//!   stream:    ... 10, 11   ╳ drop ╳   16, 17, 18 ...
40//!                                       ↑ first after reconnect
41//!
42//!   on receive 16 (last_broadcast = 11):
43//!     fetch 12..=15 from kv-rpc, broadcast 12..=15
44//!     then broadcast 16
45//! ```
46//!
47//! Recovery is chunked, with each chunk waiting for the kv-rpc indexer to
48//! catch up. See `gap_recovery` for chunk size and retry semantics.
49//!
50//! # Subscriber view
51//!
52//! From a subscriber's perspective, the broadcast is strictly contiguous,
53//! in-order, and contains no duplicates — there's no visible boundary
54//! between recovered and live messages. Long disconnects converge naturally:
55//! if recovery itself takes long enough that the validator advances past
56//! the recovery target, the next live message is itself a gap, and detection
57//! fires again.

124/// A broadcast receiver used by subscription resolvers to receive processed checkpoints.
125/// Stored in the GraphQL context; each subscriber calls `resubscribe()` to get its own
126/// receiver. Using a Receiver (not Sender) ensures that when the stream task drops its
127/// Sender, all subscribers receive `RecvError::Closed`.
128pub(crate) type CheckpointBroadcaster = broadcast::Receiver<Arc<ProcessedCheckpoint>>;

145        ProtoCheckpoint::path_builder()
146            .transactions()
147            .events()
148            .bcs()
149            .value(),
150        ProtoCheckpoint::path_builder()
151            .transactions()
152            .signatures()
153            .bcs()
154            .value(),
155        ProtoCheckpoint::path_builder()
156            .transactions()
157            .balance_changes()
158            .finish(),
159        // Checkpoint-level objects (deduplicated across transactions).
160        // Per-transaction `transactions.objects` is not populated in streamed checkpoints.

163            .objects()
164            .bcs()
165            .value(),

169/// Handle on the checkpoint broadcast that subscription resolvers consume from. Registered in
170/// the GraphQL context as a nominal type so its members do not clash with other context entries
171/// that may be added later.
172#[cfg(any(feature = "staging", test))]
173pub(crate) struct SubscriptionBroadcast {
174    /// Receiver created with the channel and never `.recv()`'d. Subscribers call `resubscribe()`
175    /// on it to get their own receivers; we also read `broadcaster.len()` to count how many
176    /// checkpoints the live stream has broadcast since channel creation, which combined with
177    /// `first_live_checkpoint` gives the current network tip without a separate atomic.
178    broadcaster: CheckpointBroadcaster,
179    /// Sequence number of the first checkpoint the live upstream stream broadcast through this
180    /// channel. Distinct from any `resume_from` arg passed by subscribers (those refer to the
181    /// kv-rpc resume fallback, not the live broadcast).
182    first_live_checkpoint: u64,
183    /// Per-subscriber metrics, shared with every stream this broadcast spawns.
184    metrics: Arc<SubscriptionMetrics>,
185}
186
187#[cfg(any(feature = "staging", test))]
188impl SubscriptionBroadcast {
189    pub(crate) fn new(
190        broadcaster: CheckpointBroadcaster,
191        first_live_checkpoint: u64,
192        metrics: Arc<SubscriptionMetrics>,
193    ) -> Self {
194        Self {
195            broadcaster,
196            first_live_checkpoint,
197            metrics,
198        }
199    }
200
201    /// Per-subscriber metrics shared by every stream this broadcast spawns.
202    #[cfg(feature = "staging")]
203    pub(crate) fn metrics(&self) -> &SubscriptionMetrics {
204        &self.metrics
205    }
206
207    /// Direct access to the broadcast receiver template. Subscribers should call
208    /// `.resubscribe()` to get their own receiver.
209    #[cfg(feature = "staging")]
210    pub(crate) fn broadcaster(&self) -> &CheckpointBroadcaster {
211        &self.broadcaster
212    }
213
214    /// Sequence number of the most recently broadcast checkpoint. Derived from the tokio-managed
215    /// `broadcaster.len()` plus the immutable `first_live_checkpoint`, so there is no separate
216    /// shared counter that could race with the broadcast `send`.
217    pub(crate) fn network_tip(&self) -> u64 {
218        self.first_live_checkpoint
219            .saturating_add(self.broadcaster.len() as u64)
220            .saturating_sub(1)
221    }
222}
223
224/// Convert a broadcast `RecvError` into an `RpcError` to be yielded to the subscriber.
225#[cfg(any(feature = "staging", test))]
226pub(crate) fn broadcast_error(e: broadcast::error::RecvError) -> RpcError {
227    match e {
228        broadcast::error::RecvError::Lagged(missed_count) => {
229            warn!(missed_count, "Subscription lagged, disconnecting");
230            anyhow::anyhow!(
231                "Subscription too slow: missed {missed_count} checkpoints. \
232                 Please reconnect and resume from your last seen checkpoint."
233            )
234            .into()
235        }
236        broadcast::error::RecvError::Closed => {
237            warn!("Checkpoint broadcast channel closed");
238            anyhow::anyhow!("Checkpoint stream has been shut down. Please reconnect.").into()
239        }
240    }
241}

243/// General error for a subscription interrupted by a gap or catch-up overflow. The specific reason
244/// is logged at the call site; clients only see that they should reconnect and resume.
245#[cfg(any(feature = "staging", test))]
246pub(crate) fn reconnect_error() -> RpcError {
247    anyhow::anyhow!(
248        "Subscription interrupted. Please reconnect and resume from your last seen checkpoint."
249    )
250    .into()
251}

253/// Background service that connects to a fullnode's gRPC SubscribeCheckpoints endpoint,
254/// processes incoming checkpoints, and broadcasts them to subscription resolvers.
255pub(crate) struct CheckpointStreamTask {
256    uri: Uri,
257    sender: broadcast::Sender<Arc<ProcessedCheckpoint>>,
258    streaming_packages: Arc<StreamingPackageStore>,
259    streaming_transactions: Arc<StreamedTransactionStore>,
260    // Populated only under the staging feature (from the checkpoint's `execution_objects`).
261    #[cfg_attr(not(feature = "staging"), allow(dead_code))]
262    streaming_objects: Arc<StreamedObjectStore>,
263    readiness: Arc<SubscriptionReadiness>,
264    /// kv-rpc reader used to fill upstream gaps. Required: streaming subscriptions need a
265    /// fallback source to recover from disconnects. lib.rs ensures this is configured when
266    /// streaming is enabled.
267    ledger_grpc_reader: LedgerGrpcReader,
268    /// Watermarks receiver used by gap recovery to wait for kv-rpc to be at or past a target
269    /// before fetching.
270    watermarks_rx: watch::Receiver<Arc<Watermarks>>,
271    gap_recovery_chunk_size: usize,
272    metrics: Arc<SubscriptionMetrics>,
273}
274
275impl CheckpointStreamTask {
276    /// Returns the task and the anchor receiver created with the broadcast channel. The
277    /// receiver must outlive `run()` (which consumes the task), so the caller holds onto it
278    /// and pairs it with the first streamed checkpoint (from `SubscriptionReadiness`) to
279    /// build a `SubscriptionBroadcast` once readiness fires.
280    pub(crate) fn new(
281        uri: Uri,
282        config: &SubscriptionConfig,
283        streaming_packages: Arc<StreamingPackageStore>,
284        streaming_transactions: Arc<StreamedTransactionStore>,
285        streaming_objects: Arc<StreamedObjectStore>,
286        readiness: Arc<SubscriptionReadiness>,
287        ledger_grpc_reader: LedgerGrpcReader,
288        watermarks_rx: watch::Receiver<Arc<Watermarks>>,
289        metrics: Arc<SubscriptionMetrics>,
290    ) -> (Self, CheckpointBroadcaster) {
291        let (sender, broadcaster) = broadcast::channel(config.broadcast_buffer);
292        let task = Self {
293            uri,
294            sender,
295            streaming_packages,
296            streaming_transactions,
297            streaming_objects,
298            readiness,
299            ledger_grpc_reader,
300            watermarks_rx,
301            gap_recovery_chunk_size: config.gap_recovery_chunk_size,
302            metrics,
303        };
304        (task, broadcaster)
305    }
306
307    /// Connect to the fullnode's gRPC SubscribeCheckpoints endpoint.
308    async fn connect(&self) -> anyhow::Result<Streaming<SubscribeCheckpointsResponse>> {
309        let endpoint = Endpoint::from(self.uri.clone()).connect_timeout(CONNECTION_TIMEOUT);
310
311        let mut client = SubscriptionServiceClient::connect(endpoint)
312            .await
313            .context("Failed to connect to checkpoint stream")?
314            .max_decoding_message_size(MAX_GRPC_MESSAGE_SIZE_BYTES);
315
316        let mut request = SubscribeCheckpointsRequest::default();
317        request.read_mask = Some(checkpoint_field_mask());
318
319        let stream = client
320            .subscribe_checkpoints(request)
321            .await
322            .context("Failed to subscribe to checkpoint stream")?
323            .into_inner();
324
325        Ok(stream)
326    }
327
328    /// Drive the upstream checkpoint stream. Reconnects on any failure with bounded backoff,
329    /// detects gaps on each incoming checkpoint, and fills them via kv-rpc before broadcasting
330    /// the live message. Permanent connect errors (auth / config) terminate the task; transient
331    /// errors retry indefinitely.
332    pub(crate) fn run(self) -> Service {
333        Service::new().spawn_aborting(async move {
334            let mut last_broadcast: Option<u64> = None;
335            let mut first_live_recorded = false;
336
337            loop {
338                info!("Connecting to checkpoint stream at {}...", self.uri);
339                let stream = backoff::future::retry(reconnect_backoff(), || async {
340                    self.connect().await.map_err(|e| {
341                        self.metrics.record_connect_failure(&e);
342                        classify_connect_error(e)
343                    })
344                })
345                .await
346                .inspect_err(|_| {
347                    self.metrics.record_termination("connect_error");
348                })?;
349                info!("Connected to checkpoint stream at {}", self.uri);
350
351                self.consume_stream(stream, &mut last_broadcast, &mut first_live_recorded)
352                    .await
353                    .inspect_err(|_| {
354                        self.metrics.record_termination("stream_error");
355                    })?;
356                self.metrics.upstream_disconnections.inc();
357                warn!("Checkpoint stream ended, reconnecting");
358            }
359        })
360    }
361
362    /// Consume one connected stream until it ends or errors. On stream-level errors, logs
363    /// and returns `Ok(())` so the caller reconnects. Per-message errors (malformed proto,
364    /// gap recovery failure) propagate as `Err` and terminate the task.
365    ///
366    /// Generic over the stream type so tests can feed synthetic messages without standing
367    /// up a real gRPC server. Tonic's `Streaming<T>` already implements `Stream`, so the
368    /// production call site is unchanged.
369    async fn consume_stream<S>(
370        &self,
371        mut stream: S,
372        last_broadcast: &mut Option<u64>,
373        first_live_recorded: &mut bool,
374    ) -> anyhow::Result<()>
375    where
376        S: futures::Stream<Item = Result<SubscribeCheckpointsResponse, tonic::Status>> + Unpin,
377    {
378        while let Some(result) = stream.next().await {
379            let response = match result {
380                Ok(r) => r,
381                Err(e) => {
382                    warn!("Stream error: {e:#}");
383                    return Ok(());
384                }
385            };
386
387            let Some(checkpoint) = response.checkpoint else {
388                continue;
389            };
390            let seq = checkpoint
391                .sequence_number
392                .context("Checkpoint without sequence_number")?;
393
394            if !*first_live_recorded {
395                self.readiness.record_first_live_checkpoint(seq);
396                *first_live_recorded = true;
397            }
398
399            // Gap detection: synchronously fill any hole between last_broadcast and this
400            // message before broadcasting the live cp.
401            if let Some(last) = *last_broadcast
402                && seq > last + 1
403            {
404                info!(from = last + 1, to = seq - 1, "Recovering gap");
405                self.metrics.upstream_gap_recoveries.inc();
406                recover_gap(
407                    &self.ledger_grpc_reader,
408                    &self.watermarks_rx,
409                    &self.sender,
410                    &self.metrics,
411                    last + 1,
412                    seq - 1,
413                    self.gap_recovery_chunk_size,
414                )
415                .await?;
416            }
417
418            self.index_and_broadcast(checkpoint, seq)?;
419            *last_broadcast = Some(seq);
420        }
421
422        Ok(())
423    }
424
425    /// Index packages from the checkpoint into the streaming store, signal eviction,
426    /// then process and broadcast. Indexing exposes the checkpoint's packages to
427    /// subscribers immediately, ahead of `kv_packages`. Recovered checkpoints from
428    /// `recover_gap` skip this step because the gate already waits for both
429    /// `ledger_grpc` and `kv_packages` to catch up.
430    fn index_and_broadcast(&self, checkpoint: ProtoCheckpoint, seq: u64) -> anyhow::Result<()> {
431        // Index this checkpoint's packages and transactions into the streaming caches so subscribers
432        // resolve them (e.g. `Object.previousTransaction`) while the checkpoint is ahead of the
433        // persistent index; the eviction task drains each cache once the index catches up.
434        self.streaming_packages
435            .index_packages(seq, &extract_packages(&checkpoint));
436        let processed = process_checkpoint(checkpoint).inspect_err(|_| {
437            self.metrics.upstream_malformed_checkpoints.inc();
438        })?;
439        self.streaming_transactions
440            .index_transactions(seq, &processed.transactions);
441        // `execution_objects` only exists under the staging feature (it's the streamed object source).
442        #[cfg(feature = "staging")]
443        self.streaming_objects
444            .index_objects(seq, &processed.execution_objects);
445
446        self.metrics.record_processed_checkpoint(
447            "live",
448            processed.summary.sequence_number,
449            processed.summary.timestamp_ms,
450        );
451
452        // Ignore send errors: no active subscribers is a normal state.
453        let _ = self.sender.send(Arc::new(processed));
454        Ok(())
455    }
456}

458/// Backoff between reconnect attempts: 1s growing to 5s max, no overall deadline.
459/// Tight cap so we recover quickly from rolling deploys; infinite retry so the streaming
460/// server stays alive through extended outages.
461fn reconnect_backoff() -> ExponentialBackoff {
462    ExponentialBackoff {
463        initial_interval: Duration::from_secs(1),
464        max_interval: Duration::from_secs(5),
465        max_elapsed_time: None,
466        ..Default::default()
467    }
468}

470/// Classify connect errors as transient or permanent. gRPC codes that signal config problems
471/// (auth, missing endpoint, malformed request) terminate the task; everything else retries.
472fn classify_connect_error(e: anyhow::Error) -> backoff::Error<anyhow::Error> {
473    use tonic::Code;
474    let permanent = e.downcast_ref::<tonic::Status>().is_some_and(|s| {
475        matches!(
476            s.code(),
477            Code::Unauthenticated
478                | Code::PermissionDenied
479                | Code::InvalidArgument
480                | Code::Unimplemented
481        )
482    });
483
484    if permanent {
485        warn!("Permanent connect failure (config / auth issue): {e:#}");
486        backoff::Error::permanent(e)
487    } else {
488        warn!("Connect failed, will retry: {e:#}");
489        backoff::Error::transient(e)
490    }
491}

493pub(crate) fn process_checkpoint(
494    checkpoint: ProtoCheckpoint,
495) -> anyhow::Result<ProcessedCheckpoint> {
496    let sequence_number = checkpoint
497        .sequence_number
498        .context("Checkpoint without sequence_number")?;
499
500    let summary: CheckpointSummary = checkpoint
501        .summary
502        .as_ref()
503        .and_then(|s| s.bcs.as_ref())
504        .context("Missing summary.bcs")?
505        .deserialize()
506        .context("Failed to deserialize checkpoint summary")?;
507
508    let contents: CheckpointContents = checkpoint
509        .contents
510        .as_ref()
511        .and_then(|c| c.bcs.as_ref())
512        .context("Missing contents.bcs")?
513        .deserialize()
514        .context("Failed to deserialize checkpoint contents")?;
515
516    let signature: AuthorityStrongQuorumSignInfo = {
517        let sdk_sig = ValidatorAggregatedSignature::try_from(
518            checkpoint.signature.as_ref().context("Missing signature")?,
519        )
520        .context("Failed to parse checkpoint signature")?;
521        AuthorityStrongQuorumSignInfo::from(sdk_sig)
522    };
523
524    let timestamp_ms = summary.timestamp_ms;
525    let cp_sequence_number = sequence_number;
526    let tx_lo = summary.network_total_transactions - checkpoint.transactions.len() as u64;
527    #[cfg(feature = "staging")]
528    let checkpoint_objects = deserialize_checkpoint_objects(&checkpoint)?;
529
530    // Seed the checkpoint-wide execution-objects map with every existing object version
531    // carried by the proto. Tombstones for deleted/wrapped outputs are added below from each
532    // transaction's effects because the proto doesn't carry deleted-object payloads.
533    #[cfg(feature = "staging")]
534    let mut execution_objects: BTreeMap<(ObjectID, SequenceNumber), Option<NativeObject>> =
535        checkpoint_objects
536            .iter()
537            .map(|(k, v)| (*k, Some(v.clone())))
538            .collect();
539
540    let mut transactions = Vec::with_capacity(checkpoint.transactions.len());
541    for (i, proto) in checkpoint.transactions.iter().enumerate() {
542        #[cfg(feature = "staging")]
543        add_tombstones(&mut execution_objects, proto)?;
544        transactions.push(process_transaction(
545            proto,
546            timestamp_ms,
547            cp_sequence_number,
548            tx_lo + i as u64,
549        )?);
550    }
551
552    Ok(ProcessedCheckpoint {
553        summary,
554        contents,
555        signature,
556        transactions,
557        #[cfg(feature = "staging")]
558        execution_objects: Arc::new(execution_objects),
559    })
560}

562fn process_transaction(
563    proto: &ProtoExecutedTransaction,
564    timestamp_ms: u64,
565    cp_sequence_number: u64,
566    tx_sequence_number: u64,
567) -> anyhow::Result<ProcessedTransaction> {
568    let contents = hydrate_executed_transaction(proto, timestamp_ms, cp_sequence_number)?;
569    Ok(ProcessedTransaction {
570        tx_sequence_number,
571        contents: Arc::new(contents),
572    })
573}

575/// Deserialize a proto `ExecutedTransaction` (from a streamed checkpoint or the scanning API) into
576/// native transaction contents, tagging it with the checkpoint `timestamp_ms` and `cp_sequence_number`
577/// the proto belongs to (the proto carries these on `timestamp`/`checkpoint` for scanned txs, and the
578/// caller supplies them from the checkpoint summary for streamed txs).
579fn hydrate_executed_transaction(
580    proto: &ProtoExecutedTransaction,
581    timestamp_ms: u64,
582    cp_sequence_number: u64,
583) -> anyhow::Result<NativeTransactionContents> {
584    let transaction_data: TransactionData = proto
585        .transaction
586        .as_ref()
587        .and_then(|t| t.bcs.as_ref())
588        .context("Missing transaction.bcs")?
589        .deserialize()
590        .context("Failed to deserialize transaction data")?;
591
592    let effects: TransactionEffects = proto
593        .effects
594        .as_ref()
595        .and_then(|e| e.bcs.as_ref())
596        .context("Missing effects.bcs")?
597        .deserialize()
598        .context("Failed to deserialize transaction effects")?;
599
600    let events: Vec<Arc<_>> = proto
601        .events
602        .as_ref()
603        .and_then(|e| e.bcs.as_ref())
604        .map(|bcs| {
605            bcs.deserialize()
606                .context("Failed to deserialize transaction events")
607        })
608        .transpose()?
609        .map(|e: TransactionEvents| e.data.into_iter().map(Arc::new).collect())
610        .unwrap_or_default();
611
612    let signatures: Vec<GenericSignature> = proto
613        .signatures
614        .iter()
615        .map(|sig| {
616            let bytes = sig
617                .bcs
618                .as_ref()
619                .context("Missing signature bcs")?
620                .value
621                .as_deref()
622                .unwrap_or(&[]);
623            GenericSignature::from_bytes(bytes)
624                .map_err(|e| anyhow::anyhow!("Failed to parse user signature: {e}"))
625        })
626        .collect::<anyhow::Result<_>>()?;
627
628    Ok(NativeTransactionContents::ExecutedTransaction(
629        sui_indexer_alt_reader::kv_loader::ExecutedTransactionData {
630            effects: Box::new(effects),
631            events,
632            transaction_data: Box::new(transaction_data),
633            signatures,
634            balance_changes: proto.balance_changes.clone(),
635            proto_effects: proto.effects.clone(),
636            proto_transaction: proto.transaction.clone(),
637            timestamp_ms: Some(timestamp_ms),
638            cp_sequence_number: Some(cp_sequence_number),
639        },
640    ))
641}

643/// Extract package objects from a streamed checkpoint. Returned packages will be
644/// inserted into the streaming index and queued for eventual eviction.
645fn extract_packages(checkpoint: &ProtoCheckpoint) -> Vec<Arc<Package>> {
646    let Some(object_set) = &checkpoint.objects else {
647        return Vec::new();
648    };
649    let mut packages = Vec::new();
650    for obj in &object_set.objects {
651        let Some(bcs) = &obj.bcs else { continue };
652        let Ok(native_obj) = bcs.deserialize::<NativeObject>() else {
653            continue;
654        };
655        let Some(move_package) = native_obj.data.try_as_package() else {
656            continue;
657        };
658        let Ok(package) = Package::read_from_package(move_package) else {
659            continue;
660        };
661        packages.push(Arc::new(package));
662    }
663    packages
664}

666/// Deserialize all objects from the checkpoint-level ObjectSet.
667#[cfg(feature = "staging")]
668fn deserialize_checkpoint_objects(
669    checkpoint: &ProtoCheckpoint,
670) -> anyhow::Result<BTreeMap<(ObjectID, SequenceNumber), NativeObject>> {
671    let mut map = BTreeMap::new();
672    if let Some(object_set) = &checkpoint.objects {
673        for obj in &object_set.objects {
674            if let Some(bcs) = &obj.bcs {
675                let native_obj: NativeObject = bcs
676                    .deserialize()
677                    .context("Failed to deserialize checkpoint object BCS")?;
678                map.insert((native_obj.id(), native_obj.version()), native_obj);
679            }
680        }
681    }
682    Ok(map)
683}

685/// Add tombstones (`None` entries) to a checkpoint-wide execution-objects map for any
686/// `DoesNotExist` outputs in this transaction's effects. The proto's `objects` field doesn't
687/// carry payloads for deleted/wrapped objects, so tombstones must come from effects; without
688/// them, `execution_output_object_latest` would return the pre-deletion version of an object
689/// that no longer exists at end-of-checkpoint.
690#[cfg(feature = "staging")]
691fn add_tombstones(
692    map: &mut BTreeMap<(ObjectID, SequenceNumber), Option<NativeObject>>,
693    proto_tx: &ProtoExecutedTransaction,
694) -> anyhow::Result<()> {
695    let Some(effects) = &proto_tx.effects else {
696        return Ok(());
697    };
698
699    let lamport_version = SequenceNumber::from_u64(
700        effects
701            .lamport_version
702            .context("Effects should have lamport_version")?,
703    );
704
705    for changed_obj in &effects.changed_objects {
706        if changed_obj.output_state() != OutputObjectState::DoesNotExist {
707            continue;
708        }
709        let object_id: ObjectID = changed_obj
710            .object_id
711            .as_ref()
712            .and_then(|id| id.parse().ok())
713            .context("ChangedObject should have valid object_id")?;
714        map.insert((object_id, lamport_version), None);
715    }
716
717    Ok(())
718}
