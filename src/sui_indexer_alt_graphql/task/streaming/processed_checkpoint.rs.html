4use std::sync::Arc;
5
6use sui_indexer_alt_reader::kv_loader::TransactionContents;
7use sui_types::crypto::AuthorityStrongQuorumSignInfo;
8use sui_types::messages_checkpoint::CheckpointContents;
9use sui_types::messages_checkpoint::CheckpointSummary;
10
11use crate::scope::ExecutionObjectMap;
12
13/// A checkpoint received from gRPC with pre-deserialized data for subscriber consumption.
14pub(crate) struct ProcessedCheckpoint {
15    pub(crate) summary: CheckpointSummary,
16    pub(crate) contents: CheckpointContents,
17    pub(crate) signature: AuthorityStrongQuorumSignInfo,
18    pub(crate) transactions: Vec<ProcessedTransaction>,
19    /// Checkpoint-wide execution objects (inputs and outputs across all transactions in the
20    /// checkpoint, including tombstones for deleted/wrapped objects). Object visibility in a
21    /// streamed scope is end-of-checkpoint, matching what the indexed Query API exposes.
22    pub(crate) execution_objects: ExecutionObjectMap,
23}
24
25/// A transaction from a streamed checkpoint with pre-deserialized contents.
26pub(crate) struct ProcessedTransaction {
27    pub(crate) tx_sequence_number: u64,
28    /// Wrapped in `Arc` so that subscribers (and resolvers within them) share a single deep
29    /// copy of the per-tx contents instead of each cloning the whole `TransactionContents`.
30    pub(crate) contents: Arc<TransactionContents>,
31}
32
33#[cfg(test)]
34impl ProcessedCheckpoint {
35    /// A checkpoint for tests where only `sequence_number` and `network_total_transactions` carry
36    /// meaning; the remaining fields are empty or default.
37    pub(crate) fn for_test(sequence_number: u64, network_total_transactions: u64) -> Self {
38        use std::collections::BTreeMap;
39
40        use sui_types::crypto::AggregateAuthoritySignature;
41        use sui_types::gas::GasCostSummary;
42
43        let contents = CheckpointContents::new_with_digests_only_for_tests(vec![]);
44        let summary = CheckpointSummary {
45            epoch: 0,
46            sequence_number,
47            network_total_transactions,
48            content_digest: *contents.digest(),
49            previous_digest: None,
50            epoch_rolling_gas_cost_summary: GasCostSummary::default(),
51            timestamp_ms: 0,
52            checkpoint_commitments: vec![],
53            end_of_epoch_data: None,
54            version_specific_data: vec![],
55        };
56        let signature = AuthorityStrongQuorumSignInfo {
57            epoch: 0,
58            signature: AggregateAuthoritySignature::default(),
59            signers_map: Default::default(),
60        };
61        Self {
62            summary,
63            contents,
64            signature,
65            transactions: vec![],
66            execution_objects: Arc::new(BTreeMap::new()),
67        }
68    }
69}
