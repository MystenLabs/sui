4use std::sync::Arc;
5
6use super::EvictableCache;
7use super::StreamedObjectStore;
8use super::StreamedTransactionStore;
9use super::StreamingPackageStore;
10
11/// The in-memory caches a streamed checkpoint reads ahead of the durable index: packages for type
12/// resolution, transaction contents for live `previousTransaction` lookups, and object contents for
13/// live cross-checkpoint object lookups. Threaded as one handle from the stream task down to the
14/// subscription resolvers, so a scope carries every streamed source it can read from in one place.
15pub(crate) struct StreamedCaches {
16    pub(crate) package_store: Arc<StreamingPackageStore>,
17    pub(crate) transaction_store: Arc<StreamedTransactionStore>,
18    pub(crate) object_store: Arc<StreamedObjectStore>,
19}
20
21impl StreamedCaches {
22    pub(crate) fn new(
23        package_store: Arc<StreamingPackageStore>,
24        transaction_store: Arc<StreamedTransactionStore>,
25        object_store: Arc<StreamedObjectStore>,
26    ) -> Self {
27        Self {
28            package_store,
29            transaction_store,
30            object_store,
31        }
32    }
33
34    /// Each cache as an [`EvictableCache`], for registration with the eviction task. Eviction stays
35    /// per-store: each drains against its own watermark pipeline.
36    pub(crate) fn to_evictable(&self) -> Vec<Arc<dyn EvictableCache>> {
37        vec![
38            self.package_store.clone(),
39            self.transaction_store.clone(),
40            self.object_store.clone(),
41        ]
42    }
43}
