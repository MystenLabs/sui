4//! Per-subscriber delivery throttle for subscriptions. See [`Throttle`].
5
6use std::time::Duration;
7
8use async_graphql::Response;
9use async_graphql::Value;
10use futures::Stream;
11use futures::StreamExt;
12
13use crate::extensions::query_limits::QueryDepth;
14
15/// Fixed per-level surcharge added to a payload's cost, approximating the sequential DB round trips a
16/// payload makes (roughly one per level of query depth), which the output-node count alone does not
17/// capture. The weight follows Shopify's rate-limiting model, which prices a fetch at two points plus
18/// the objects it returns; the returned objects are already counted as output nodes, so each level of
19/// depth carries only the fixed two-point part.
20///
21/// Ref: <https://shopify.engineering/rate-limiting-graphql-apis-calculating-query-complexity>
22const DEPTH_NODE_COST: u32 = 2;
23
24/// Paces subscription delivery to a sustained rate of `nodes_per_second` output nodes per second.
25///
26/// A payload's cost is normalized to output-node-equivalents, then held for `cost / rate` seconds
27/// before delivery:
28///
29/// ```text
30/// cost  = output_nodes + query_depth * DEPTH_NODE_COST
31/// delay = cost / nodes_per_second
32/// ```
33///
34/// So a 10-node payload at query depth 5 costs `10 + 5*2 = 20`, and at 40 nodes/second is held
35/// `20 / 40 = 0.5s`. A rate of `0` disables pacing, and payloads are never dropped or reordered.
36#[derive(Clone, Copy)]
37pub(crate) struct Throttle {
38    nodes_per_second: u32,
39}
40
41impl Throttle {
42    pub(crate) fn new(nodes_per_second: u32) -> Self {
43        Self { nodes_per_second }
44    }
45
46    /// Pace `stream`, delivering each payload immediately then pausing before the next for its delay.
47    pub(crate) fn wrap<S>(self, stream: S, query_depth: QueryDepth) -> impl Stream<Item = Response>
48    where
49        S: Stream<Item = Response>,
50    {
51        async_stream::stream! {
52            let mut stream = std::pin::pin!(stream);
53            while let Some(response) = stream.next().await {
54                // Deliver immediately, then pause before pulling the next payload, which also
55                // backpressures its resolution. Depth is constant but only known once validation has
56                // run, so read the slot here.
57                let delay = self.calculate_delay(&response.data, query_depth.get());
58                yield response;
59                if !delay.is_zero() {
60                    tokio::time::sleep(delay).await;
61                }
62            }
63        }
64    }
65
66    /// The delay for one payload: its cost spread over `cost / rate` seconds, or `Duration::ZERO`
67    /// when disabled.
68    pub(crate) fn calculate_delay(&self, payload: &Value, query_depth: u32) -> Duration {
69        if self.nodes_per_second == 0 {
70            return Duration::ZERO;
71        }
72        let cost = payload_cost(payload, query_depth);
73        Duration::from_secs_f64(cost as f64 / self.nodes_per_second as f64)
74    }
75}
76
77/// The delivery cost of a payload in output-node-equivalents: its actual output nodes plus a surcharge
78/// for the subscription's query depth (constant across the subscription's payloads).
79fn payload_cost(value: &Value, query_depth: u32) -> u32 {
80    count_output_nodes(value).saturating_add(query_depth.saturating_mul(DEPTH_NODE_COST))
81}
82
83/// Count the output nodes in a resolved payload: one per object, one per list, and one per scalar
84/// leaf, summed over the tree.
85fn count_output_nodes(value: &Value) -> u32 {
86    match value {
87        Value::Object(fields) => fields
88            .values()
89            .map(count_output_nodes)
90            .fold(1, u32::saturating_add),
91        Value::List(items) => items
92            .iter()
93            .map(count_output_nodes)
94            .fold(1, u32::saturating_add),
95        _ => 1,
96    }
97}

99#[cfg(test)]
100mod tests {
101    use async_graphql::value;
102
103    use super::*;
104
105    #[test]
106    fn counts_scalars_and_objects() {
107        // A single scalar.
108        assert_eq!(count_output_nodes(&value!(1)), 1);
109        // An object (1) with two scalar fields (1 each).
110        assert_eq!(count_output_nodes(&value!({ "a": 1, "b": 2 })), 3);
111        // Nesting: outer object (1) + inner object (1 + two scalars = 3).
112        assert_eq!(count_output_nodes(&value!({ "a": { "b": 1, "c": 2 } })), 4);
113    }
114
115    #[test]
116    fn counts_lists_and_elements() {
117        // Object (1) + list (1) + three scalar elements (1 each).
118        assert_eq!(count_output_nodes(&value!({ "items": [1, 2, 3] })), 5);
119        // Object (1) + list (1) + two element objects (1 + one scalar each = 2).
120        assert_eq!(
121            count_output_nodes(&value!({ "nodes": [{ "x": 1 }, { "x": 2 }] })),
122            6
123        );
124    }
125
126    #[test]
127    fn payload_cost_adds_depth_surcharge() {
128        let payload = value!({ "a": 1, "b": 2 }); // 3 output nodes
129        assert_eq!(payload_cost(&payload, 0), 3);
130        assert_eq!(payload_cost(&payload, 4), 3 + 4 * DEPTH_NODE_COST);
131    }
132
133    #[test]
134    fn zero_rate_disables_pacing() {
135        let payload = value!({ "a": 1, "b": 2 });
136        assert_eq!(
137            Throttle::new(0).calculate_delay(&payload, 100),
138            Duration::ZERO
139        );
140    }
141
142    #[test]
143    fn delay_is_cost_over_rate() {
144        let payload = value!({ "a": 1, "b": 2, "c": 3, "d": 4 }); // 5 output nodes
145
146        // No depth surcharge: cost 5 at 10 nodes/sec = 0.5s.
147        assert_eq!(
148            Throttle::new(10).calculate_delay(&payload, 0),
149            Duration::from_millis(500)
150        );
151
152        // Depth surcharge included: cost 5 + 5 * 2 = 15 at 15 nodes/sec = 1s.
153        assert_eq!(
154            Throttle::new(15).calculate_delay(&payload, 5),
155            Duration::from_secs(1)
156        );
157
158        // A higher rate paces the same payload proportionally faster.
159        assert!(
160            Throttle::new(20).calculate_delay(&payload, 0)
161                < Throttle::new(10).calculate_delay(&payload, 0)
162        );
163    }
164
165    #[tokio::test(start_paused = true)]
166    async fn wrap_delivers_two_payloads_per_second() {
167        // A 5-node payload at 10 nodes/sec is held 5 / 10 = 0.5s. The first is delivered immediately,
168        // then each subsequent one is paced 0.5s behind, so delivery holds at two per second.
169        let payload = value!({ "a": 1, "b": 2, "c": 3, "d": 4 }); // 5 output nodes
170        let responses = vec![
171            Response::new(payload.clone()),
172            Response::new(payload.clone()),
173            Response::new(payload),
174        ];
175        let mut paced = Box::pin(
176            Throttle::new(10).wrap(futures::stream::iter(responses), QueryDepth::default()),
177        );
178
179        let start = tokio::time::Instant::now();
180        paced.next().await.unwrap();
181        assert_eq!(start.elapsed(), Duration::ZERO);
182        paced.next().await.unwrap();
183        assert_eq!(start.elapsed(), Duration::from_millis(500));
184        paced.next().await.unwrap();
185        assert_eq!(start.elapsed(), Duration::from_secs(1));
186        assert!(paced.next().await.is_none());
187    }
188
189    #[tokio::test(start_paused = true)]
190    async fn wrap_includes_depth_surcharge() {
191        // 4-node payloads at query depth 3 each cost 4 + 3 * 2 = 10, so at 20 nodes/sec the gap
192        // between deliveries is 10 / 20 = 0.5s.
193        let query_depth = QueryDepth::new_for_test(3);
194        let payload = value!({ "a": 1, "b": 2, "c": 3 }); // 4 output nodes
195        let responses = vec![Response::new(payload.clone()), Response::new(payload)];
196        let mut paced =
197            Box::pin(Throttle::new(20).wrap(futures::stream::iter(responses), query_depth));
198
199        let start = tokio::time::Instant::now();
200        paced.next().await.unwrap();
201        assert_eq!(start.elapsed(), Duration::ZERO);
202        paced.next().await.unwrap();
203        assert_eq!(start.elapsed(), Duration::from_millis(500));
204    }
205
206    #[tokio::test(start_paused = true)]
207    async fn wrap_does_not_pace_when_disabled() {
208        // A rate of 0 disables pacing, so both payloads arrive immediately with no gap.
209        let payload = value!({ "a": 1, "b": 2 });
210        let responses = vec![Response::new(payload.clone()), Response::new(payload)];
211        let mut paced = Box::pin(
212            Throttle::new(0).wrap(futures::stream::iter(responses), QueryDepth::default()),
213        );
214
215        let start = tokio::time::Instant::now();
216        paced.next().await.unwrap();
217        paced.next().await.unwrap();
218        assert_eq!(start.elapsed(), Duration::ZERO);
219    }
