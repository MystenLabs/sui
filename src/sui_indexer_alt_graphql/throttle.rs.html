4//! Per-subscriber delivery throttle for subscriptions. See [`Throttle`].
5
6use std::time::Duration;
7
8use async_graphql::Response;
9use async_graphql::Value;
10use futures::Stream;
11use futures::StreamExt;
12use prometheus::Histogram;
13
14use crate::extensions::query_limits::QueryDepth;
15
16/// Fixed per-level surcharge added to a payload's cost, approximating the sequential DB round trips a
17/// payload makes (roughly one per level of query depth), which the output-node count alone does not
18/// capture. The weight follows Shopify's rate-limiting model, which prices a fetch at two points plus
19/// the objects it returns; the returned objects are already counted as output nodes, so each level of
20/// depth carries only the fixed two-point part.
21///
22/// Ref: <https://shopify.engineering/rate-limiting-graphql-apis-calculating-query-complexity>
23const DEPTH_NODE_COST: u32 = 2;
24
25/// Paces subscription delivery to a sustained rate of `nodes_per_second` output nodes per second.
26///
27/// A payload's cost is normalized to output-node-equivalents, then held for `cost / rate` seconds
28/// before delivery:
29///
30/// ```text
31/// cost  = output_nodes + query_depth * DEPTH_NODE_COST
32/// delay = cost / nodes_per_second
33/// ```
34///
35/// So a 10-node payload at query depth 5 costs `10 + 5*2 = 20`, and at 40 nodes/second is held
36/// `20 / 40 = 0.5s`. A rate of `0` disables pacing, and payloads are never dropped or reordered.
37#[derive(Clone)]
38pub(crate) struct Throttle {
39    nodes_per_second: u32,
40    /// Observes each payload's pacing delay in seconds (including zero when the budget is not
41    /// binding), for the `throttle_delay` metric.
42    delay_metric: Histogram,
43}
44
45impl Throttle {
46    pub(crate) fn new(nodes_per_second: u32, delay_metric: Histogram) -> Self {
47        Self {
48            nodes_per_second,
49            delay_metric,
50        }
51    }
52
53    /// Pace `stream`, delivering each payload immediately then pausing before the next for its delay.
54    pub(crate) fn wrap<S>(self, stream: S, query_depth: QueryDepth) -> impl Stream<Item = Response>
55    where
56        S: Stream<Item = Response>,
57    {
58        async_stream::stream! {
59            let mut stream = std::pin::pin!(stream);
60            while let Some(response) = stream.next().await {
61                // Deliver immediately, then pause before pulling the next payload, which also
62                // backpressures its resolution. Depth is constant but only known once validation has
63                // run, so read the slot here.
64                let delay = self.calculate_delay(&response.data, query_depth.get());
65                self.delay_metric.observe(delay.as_secs_f64());
66                yield response;
67                if !delay.is_zero() {
68                    tokio::time::sleep(delay).await;
69                }
70            }
71        }
72    }
73
74    /// The delay for one payload: its cost spread over `cost / rate` seconds, or `Duration::ZERO`
75    /// when disabled.
76    pub(crate) fn calculate_delay(&self, payload: &Value, query_depth: u32) -> Duration {
77        if self.nodes_per_second == 0 {
78            return Duration::ZERO;
79        }
80        let cost = payload_cost(payload, query_depth);
81        Duration::from_secs_f64(cost as f64 / self.nodes_per_second as f64)
82    }
83}
84
85/// The delivery cost of a payload in output-node-equivalents: its actual output nodes plus a surcharge
86/// for the subscription's query depth (constant across the subscription's payloads).
87fn payload_cost(value: &Value, query_depth: u32) -> u32 {
88    count_output_nodes(value).saturating_add(query_depth.saturating_mul(DEPTH_NODE_COST))
89}
90
91/// Count the output nodes in a resolved payload: one per object, one per list, and one per scalar
92/// leaf, summed over the tree.
93fn count_output_nodes(value: &Value) -> u32 {
94    match value {
95        Value::Object(fields) => fields
96            .values()
97            .map(count_output_nodes)
98            .fold(1, u32::saturating_add),
99        Value::List(items) => items
100            .iter()
101            .map(count_output_nodes)
102            .fold(1, u32::saturating_add),
103        _ => 1,
104    }
105}

107#[cfg(test)]
108mod tests {
109    use async_graphql::value;
110
111    use super::*;
112
113    fn throttle(nodes_per_second: u32) -> Throttle {
114        Throttle::new(nodes_per_second, test_histogram())
115    }
116
117    fn test_histogram() -> Histogram {
118        Histogram::with_opts(prometheus::HistogramOpts::new("test", "test")).unwrap()
119    }
120
121    #[test]
122    fn counts_scalars_and_objects() {
123        // A single scalar.
124        assert_eq!(count_output_nodes(&value!(1)), 1);
125        // An object (1) with two scalar fields (1 each).
126        assert_eq!(count_output_nodes(&value!({ "a": 1, "b": 2 })), 3);
127        // Nesting: outer object (1) + inner object (1 + two scalars = 3).
128        assert_eq!(count_output_nodes(&value!({ "a": { "b": 1, "c": 2 } })), 4);
129    }
130
131    #[test]
132    fn counts_lists_and_elements() {
133        // Object (1) + list (1) + three scalar elements (1 each).
134        assert_eq!(count_output_nodes(&value!({ "items": [1, 2, 3] })), 5);
135        // Object (1) + list (1) + two element objects (1 + one scalar each = 2).
136        assert_eq!(
137            count_output_nodes(&value!({ "nodes": [{ "x": 1 }, { "x": 2 }] })),
138            6
139        );
140    }
141
142    #[test]
143    fn payload_cost_adds_depth_surcharge() {
144        let payload = value!({ "a": 1, "b": 2 }); // 3 output nodes
145        assert_eq!(payload_cost(&payload, 0), 3);
146        assert_eq!(payload_cost(&payload, 4), 3 + 4 * DEPTH_NODE_COST);
147    }
148
149    #[test]
150    fn zero_rate_disables_pacing() {
151        let payload = value!({ "a": 1, "b": 2 });
152        assert_eq!(throttle(0).calculate_delay(&payload, 100), Duration::ZERO);
153    }
154
155    #[test]
156    fn delay_is_cost_over_rate() {
157        let payload = value!({ "a": 1, "b": 2, "c": 3, "d": 4 }); // 5 output nodes
158
159        // No depth surcharge: cost 5 at 10 nodes/sec = 0.5s.
160        assert_eq!(
161            throttle(10).calculate_delay(&payload, 0),
162            Duration::from_millis(500)
163        );
164
165        // Depth surcharge included: cost 5 + 5 * 2 = 15 at 15 nodes/sec = 1s.
166        assert_eq!(
167            throttle(15).calculate_delay(&payload, 5),
168            Duration::from_secs(1)
169        );
170
171        // A higher rate paces the same payload proportionally faster.
172        assert!(
173            throttle(20).calculate_delay(&payload, 0) < throttle(10).calculate_delay(&payload, 0)
174        );
175    }
176
177    #[tokio::test(start_paused = true)]
178    async fn wrap_delivers_two_payloads_per_second() {
179        // A 5-node payload at 10 nodes/sec is held 5 / 10 = 0.5s. The first is delivered immediately,
180        // then each subsequent one is paced 0.5s behind, so delivery holds at two per second.
181        let payload = value!({ "a": 1, "b": 2, "c": 3, "d": 4 }); // 5 output nodes
182        let responses = vec![
183            Response::new(payload.clone()),
184            Response::new(payload.clone()),
185            Response::new(payload),
186        ];
187        let mut paced =
188            Box::pin(throttle(10).wrap(futures::stream::iter(responses), QueryDepth::default()));
189
190        let start = tokio::time::Instant::now();
191        paced.next().await.unwrap();
192        assert_eq!(start.elapsed(), Duration::ZERO);
193        paced.next().await.unwrap();
194        assert_eq!(start.elapsed(), Duration::from_millis(500));
195        paced.next().await.unwrap();
196        assert_eq!(start.elapsed(), Duration::from_secs(1));
197        assert!(paced.next().await.is_none());
198    }
199
200    #[tokio::test(start_paused = true)]
201    async fn wrap_includes_depth_surcharge() {
202        // 4-node payloads at query depth 3 each cost 4 + 3 * 2 = 10, so at 20 nodes/sec the gap
203        // between deliveries is 10 / 20 = 0.5s.
204        let query_depth = QueryDepth::new_for_test(3);
205        let payload = value!({ "a": 1, "b": 2, "c": 3 }); // 4 output nodes
206        let responses = vec![Response::new(payload.clone()), Response::new(payload)];
207        let mut paced = Box::pin(throttle(20).wrap(futures::stream::iter(responses), query_depth));
208
209        let start = tokio::time::Instant::now();
210        paced.next().await.unwrap();
211        assert_eq!(start.elapsed(), Duration::ZERO);
212        paced.next().await.unwrap();
213        assert_eq!(start.elapsed(), Duration::from_millis(500));
214    }
215
216    #[tokio::test(start_paused = true)]
217    async fn wrap_does_not_pace_when_disabled() {
218        // A rate of 0 disables pacing, so both payloads arrive immediately with no gap.
219        let payload = value!({ "a": 1, "b": 2 });
220        let responses = vec![Response::new(payload.clone()), Response::new(payload)];
221        let mut paced =
222            Box::pin(throttle(0).wrap(futures::stream::iter(responses), QueryDepth::default()));
223
224        let start = tokio::time::Instant::now();
225        paced.next().await.unwrap();
226        paced.next().await.unwrap();
227        assert_eq!(start.elapsed(), Duration::ZERO);
228    }

230    #[tokio::test(start_paused = true)]
231    async fn wrap_observes_one_delay_sample_per_payload() {
232        let metric = test_histogram();
233        let payload = value!({ "a": 1, "b": 2 });
234        let responses = vec![Response::new(payload.clone()), Response::new(payload)];
235        let mut paced = Box::pin(
236            Throttle::new(10, metric.clone())
237                .wrap(futures::stream::iter(responses), QueryDepth::default()),
238        );
239
240        while paced.next().await.is_some() {}
241
242        assert_eq!(metric.get_sample_count(), 2);
243    }
