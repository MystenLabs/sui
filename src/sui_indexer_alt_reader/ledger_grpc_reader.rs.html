132/// Implemented by `LedgerGrpcReader` for each key type whose `Loader::load` needs to stay under
133/// the ledger service's batch-size limit — `DataLoader`'s own `max_batch_size` is only a dispatch
134/// trigger, not a hard cap on how many keys reach a single `Loader::load` call. `load_chunk`
135/// supplies the raw, single-chunk fetch and `chunk_size` supplies the limit; `load_chunked`'s
136/// default body splits `keys` into chunks, dispatches `load_chunk` concurrently per chunk, and
137/// merges the results.
138///
139/// `pub`, not `pub(crate)`: it's referenced through `LedgerGrpcReader`'s blanket `Loader<K>` impl
140/// below (`type Value = <Self as ChunkedLoader<K>>::Value`), and that impl is part of
141/// `LedgerGrpcReader`'s public interface since both the type and `Loader` are public.
142#[async_trait::async_trait]
143pub trait ChunkedLoader<K>
144where
145    K: Send + Sync + Hash + Eq + Clone + 'static,
146{
147    type Value: Send + Sync + Clone + 'static;
148    type Error: Send + Sync + Clone + 'static;
149
150    fn chunk_size(&self) -> usize;
151
152    async fn load_chunk(&self, keys: &[K]) -> Result<HashMap<K, Self::Value>, Self::Error>;
153
154    async fn load_chunked(&self, keys: &[K]) -> Result<HashMap<K, Self::Value>, Self::Error>
155    where
156        Self: Sync,
157    {
158        let limit = self.chunk_size();
159
160        let mut results = HashMap::new();
161        for batch in try_join_all(keys.chunks(limit).map(|chunk| self.load_chunk(chunk))).await? {
162            results.extend(batch);
163        }
164        Ok(results)
165    }
166}
167
168/// Covers every key type `LedgerGrpcReader` implements [`ChunkedLoader`] for. Coherent because
169/// `Self` (`LedgerGrpcReader`) is a concrete local type — only `K` is generic — unlike a bare
170/// `impl<K, T: ChunkedLoader<K>> Loader<K> for T`, which the orphan rules reject since neither
171/// `Loader` (foreign) nor `T` (an uncovered generic) is local.
172#[async_trait::async_trait]
173impl<K> Loader<K> for LedgerGrpcReader
174where
175    K: Send + Sync + Hash + Eq + Clone + 'static,
176    Self: ChunkedLoader<K>,
177{
178    type Value = <Self as ChunkedLoader<K>>::Value;
179    type Error = <Self as ChunkedLoader<K>>::Error;
180
181    async fn load(&self, keys: &[K]) -> Result<HashMap<K, Self::Value>, Self::Error> {
182        self.load_chunked(keys).await
183    }
184}

223    /// With a `min_checkpoint` floor, every request asks the ledger service for a view at least as
224    /// new as the floor (see [`crate::monotonic_read`]).
225    pub async fn new(
226        uri: Uri,
227        args: LedgerGrpcArgs,
228        prefix: Option<&str>,
229        registry: &Registry,
230        max_batch_get_transactions: usize,
231        max_batch_get_objects: usize,
232        min_checkpoint: Option<CheckpointFloor>,
233    ) -> anyhow::Result<Self> {
234        let timeout = args.statement_timeout();
235        let mut client = Client::new(uri)?
236            .with_num_connections(args.ledger_grpc_num_connections)
237            .with_max_decoding_message_size(args.ledger_grpc_max_decoding_message_size)
238            .request_layer(
239                ServiceBuilder::new()
240                    .layer(GrpcMetricsLayer::new(
241                        prefix.unwrap_or("ledger_grpc"),
242                        registry,
243                    ))
244                    .option_layer(min_checkpoint.map(MonotonicReadLayer::new)),
245            );
246
247        if let Some(timeout) = timeout {
248            client = client.with_response_headers_timeout(timeout);
249        }
250
251        Ok(Self {
252            client,
253            timeout,
254            max_batch_get_transactions,
255            max_batch_get_objects,
256        })
257    }

271    pub async fn checkpoint_watermark(&self) -> anyhow::Result<CheckpointSummary> {
272        use grpc::GetCheckpointRequest;
273        use prost_types::FieldMask;
274        use sui_rpc::field::FieldMaskUtil;
275
276        let request =
277            GetCheckpointRequest::default().with_read_mask(FieldMask::from_paths(["summary.bcs"]));
278
279        let response = self.get_checkpoint(request).await?;
280
281        let checkpoint = response.checkpoint.context("No checkpoint returned")?;
282
283        checkpoint
284            .summary
285            .as_ref()
286            .and_then(|s| s.bcs.as_ref())
287            .context("Missing summary.bcs")?
288            .deserialize()
289            .context("Failed to deserialize checkpoint summary")
290    }

292    /// Resolve a checkpoint digest to its sequence number via the ledger service. Returns `None`
293    /// if no checkpoint with that digest is known.
294    pub async fn checkpoint_seq_by_digest(
295        &self,
296        digest: sui_types::digests::CheckpointDigest,
297    ) -> anyhow::Result<Option<u64>> {
298        use grpc::GetCheckpointRequest;
299        use prost_types::FieldMask;
300        use sui_rpc::field::FieldMaskUtil;
301
302        let sdk_digest = sui_sdk_types::Digest::new(digest.inner().to_owned());
303        let request = GetCheckpointRequest::by_digest(&sdk_digest)
304            .with_read_mask(FieldMask::from_paths(["sequence_number"]));
305
306        match self.get_checkpoint(request).await {
307            Ok(response) => {
308                let checkpoint = response.checkpoint.context("No checkpoint returned")?;
309                Ok(checkpoint.sequence_number)
310            }
311            Err(status) if status.code() == tonic::Code::NotFound => Ok(None),
312            Err(e) => Err(anyhow::anyhow!(e)),
313        }
314    }

414    /// The system packages that existed as of the end of checkpoint `at_checkpoint`, as
415    /// `(package ID, version)` pairs ordered by package ID, each at its latest version as of that
416    /// checkpoint. Returns `None` if the ledger service no longer serves `at_checkpoint` because
417    /// it lies below its earliest available checkpoint (`BELOW_EARLIEST_AVAILABLE`). Any other
418    /// failure, including `at_checkpoint` lying above the service's latest checkpoint, is an error.
419    pub async fn list_system_packages(
420        &self,
421        at_checkpoint: u64,
422    ) -> anyhow::Result<Option<Vec<(ObjectID, SequenceNumber)>>> {
423        let request = grpc::ListSystemPackagesRequest::new(at_checkpoint);
424        let response = match self
425            .client
426            .clone()
427            .package_client()
428            .list_system_packages(self.request(request))
429            .await
430        {
431            Ok(response) => response.into_inner(),
432            Err(status) if has_error_reason(&status, grpc::ErrorReason::BelowEarliestAvailable) => {
433                return Ok(None);
434            }
435            Err(status) => {
436                return Err(anyhow::anyhow!(status).context("ListSystemPackages failed"));
437            }
438        };
439
440        let packages = response
441            .system_packages
442            .into_iter()
443            .map(|package| {
444                let id = package
445                    .package_id
446                    .as_deref()
447                    .context("System package missing package_id")?
448                    .parse::<ObjectID>()
449                    .context("Failed to parse system package ID")?;
450                let version = package
451                    .version
452                    .with_context(|| format!("System package {id} missing version"))?;
453                Ok((id, SequenceNumber::from_u64(version)))
454            })
455            .collect::<anyhow::Result<_>>()?;
456
457        Ok(Some(packages))
458    }

470impl TryFrom<&grpc::ExecutedTransaction> for CheckpointedTransaction {
471    type Error = anyhow::Error;
472
473    fn try_from(executed: &grpc::ExecutedTransaction) -> anyhow::Result<Self> {
474        let full_tx: sui_types::full_checkpoint_content::ExecutedTransaction = executed
475            .try_into()
476            .context("Failed to convert ExecutedTransaction from proto")?;
477
478        let timestamp_ms = executed
479            .timestamp
480            .map(proto_to_timestamp_ms)
481            .transpose()
482            .with_context(|| format!("Failed to parse timestamp {:?}", executed.timestamp))?;
483
484        Ok(Self {
485            effects: Box::new(full_tx.effects),
486            events: full_tx.events.map(|events| events.data),
487            transaction_data: Box::new(full_tx.transaction),
488            signatures: full_tx.signatures,
489            timestamp_ms,
490            cp_sequence_number: executed.checkpoint,
491            balance_changes: executed.balance_changes.clone(),
492        })
493    }
494}

506impl<T> StreamPage<T> {
507    /// Whether further data may exist in the direction of pagination.
508    ///
509    /// `false` iff one of:
510    /// - `reason ∈ {LedgerTip, CheckpointBound}` — authoritative range terminals.
511    /// - `reason = CursorBound` AND no cursor was emitted - the server did not do any scanning and
512    ///   short-circuited. Typically implies that the cursors for the request fell outside the
513    ///   available range.
514    pub fn has_more(&self) -> bool {
515        use grpc::QueryEndReason as R;
516        match self.end_reason {
517            None => true,
518            Some(R::Unknown | R::ItemLimit | R::ScanLimit) => true,
519            Some(R::LedgerTip | R::CheckpointBound) => false,
520            Some(R::CursorBound) => self.last_cursor().is_some(),
521            // `QueryEndReason` is non exhaustive — conservatively `true` if a
522            // future variant slips past `apply()`'s `unwrap_or(Unknown)`.
523            Some(_) => true,
524        }
525    }
526
527    /// The page's resume cursor: a standalone-watermark cursor emitted after the last item if one
528    /// exists, otherwise the last item's own cursor.
529    pub fn last_cursor(&self) -> Option<&Bytes> {
530        self.last_watermark
531            .as_ref()
532            .map(|boundary| &boundary.cursor)
533            .or_else(|| self.items.last().map(|item| &item.cursor))
534    }
535
536    /// The standalone watermark that came before every item, if one did.
537    pub fn first_watermark(&self) -> Option<&PageBoundary> {
538        self.first_watermark.as_ref()
539    }
540
541    /// The standalone watermark that came after the last item, if one did.
542    pub fn last_watermark(&self) -> Option<&PageBoundary> {
543        self.last_watermark.as_ref()
544    }
545
546    /// The server's latest checkpoint coverage claim on this page (`Watermark.checkpoint`), or
547    /// `None` if no frame carried one. In the scan's direction, every match within the request's
548    /// interval at checkpoints up to and including this one (ascending) or down to and including it
549    /// (descending) has been emitted. Claims never regress within a stream but may repeat, so a
550    /// frame without a claim leaves the latest one in place.
551    pub fn covered_checkpoint(&self) -> Option<u64> {
552        self.covered_checkpoint
553    }
554
555    /// Construct a page directly for cross-crate tests, bypassing the drain loop. The watermark
556    /// fields are private (their invariant is maintained by [`Self::apply`]); this is the only
557    /// sanctioned way to set them from outside the crate.
558    #[cfg(feature = "testing")]
559    pub fn for_test(
560        items: Vec<PageItem<T>>,
561        first_watermark: Option<PageBoundary>,
562        last_watermark: Option<PageBoundary>,
563        end_reason: Option<grpc::QueryEndReason>,
564    ) -> Self {
565        Self {
566            items,
567            first_watermark,
568            last_watermark,
569            covered_checkpoint: None,
570            end_reason,
571        }
572    }
573
574    /// Fold one frame into the page.
575    ///
576    /// Every frame with a cursor must carry `Watermark.cursor_position`.
577    ///
578    /// Returns `true` when the frame is `QueryEnd`.
579    fn apply(&mut self, frame: FrameKind<T>) -> anyhow::Result<bool> {
580        let FrameKind::Frame {
581            payload,
582            cursor,
583            position,
584            covered_checkpoint,
585            end_reason,
586        } = frame
587        else {
588            warn!("ignoring unrecognized frame");
589            return Ok(false);
590        };
591        match payload {
592            Some(payload) => {
593                let cursor = cursor.expect("TryFrom validated item cursor");
594                let position = position.context(MISSING_CURSOR_POSITION)?;
595                self.last_watermark = None;
596                self.items.push(PageItem {
597                    payload,
598                    cursor,
599                    position,
600                });
601            }
602            None => {
603                if let Some(cursor) = cursor {
604                    let position = position.context(MISSING_CURSOR_POSITION)?;
605                    let boundary = PageBoundary { cursor, position };
606                    if self.items.is_empty() && self.first_watermark.is_none() {
607                        self.first_watermark = Some(boundary.clone());
608                    }
609                    self.last_watermark = Some(boundary);
610                }
611            }
612        }
613        if covered_checkpoint.is_some() {
614            self.covered_checkpoint = covered_checkpoint;
615        }
616        if let Some(reason) = end_reason {
617            // `QueryEnd::reason()` folds an absent or unknown reason into
618            // `Unknown`, so `None` here remains unambiguous shorthand for
619            // "no End frame received" (i.e. the deadline cut the stream short).
620            self.end_reason = Some(reason);
621            return Ok(true);
622        }
623        Ok(false)
624    }
625}

663/// Classify a raw list-stream response into a [`FrameKind`], given its payload field. Per-API
664/// implementations only select which response field is the payload.
665fn classify_frame<T>(
666    payload: Option<T>,
667    watermark: Option<grpc::Watermark>,
668    end: Option<grpc::QueryEnd>,
669) -> anyhow::Result<FrameKind<T>> {
670    let (cursor, position, covered_checkpoint) = watermark.map_or((None, None, None), |w| {
671        (w.cursor, w.cursor_position, w.checkpoint)
672    });
673    let end_reason = end.map(|e| e.reason());
674
675    if payload.is_none() && cursor.is_none() && end_reason.is_none() {
676        return Ok(FrameKind::Unknown);
677    }
678    if payload.is_some() && cursor.is_none() {
679        bail!("Item frame missing watermark.cursor");
680    }
681
682    Ok(FrameKind::Frame {
683        payload,
684        cursor,
685        position,
686        covered_checkpoint,
687        end_reason,
688    })
689}

691/// Whether `status` carries a `google.rpc.ErrorInfo` detail with `reason` in domain
692/// `ErrorReason::DOMAIN`. `false` when the status details do not decode as a `google.rpc.Status`.
693fn has_error_reason(status: &tonic::Status, reason: grpc::ErrorReason) -> bool {
694    let Ok(rpc_status) = RpcStatus::decode(status.details()) else {
695        return false;
696    };
697
698    rpc_status
699        .details
700        .iter()
701        .filter_map(|detail| detail.to_msg::<ErrorInfo>().ok())
702        .any(|info| info.reason == reason.as_str_name() && info.domain == grpc::ErrorReason::DOMAIN)
703}
704
705async fn drain_list_stream<R, T, S>(
706    rpc_name: &'static str,
707    stream: S,
708) -> anyhow::Result<StreamPage<T>>
709where
710    R: TryInto<FrameKind<T>, Error = anyhow::Error>,
711    S: Stream<Item = Result<R, tonic::Status>>,
712{
713    futures::pin_mut!(stream);
714    let mut page = StreamPage::default();
715    while let Some(result) = stream.next().await {
716        match result {
717            Ok(response) => {
718                let frame = response
719                    .try_into()
720                    .with_context(|| format!("{rpc_name}: malformed frame"))?;
721                // Process and break on receiving `QueryEnd`.
722                if page
723                    .apply(frame)
724                    .with_context(|| format!("{rpc_name}: unusable frame"))?
725                {
726                    break;
727                }
728            }
729            // `DeadlineExceeded`: server-side `grpc-timeout` header fired. `Cancelled`: client-side
730            // channel timeout fired (or upstream cancel). In either case, preserve partial work if
731            // any progress was made.
732            Err(status)
733                if matches!(
734                    status.code(),
735                    tonic::Code::DeadlineExceeded | tonic::Code::Cancelled
736                ) =>
737            {
738                break;
739            }
740            // Consider other errors as the request failed, safest to discard partial work.
741            Err(status) => {
742                bail!(
743                    "{rpc_name}: stream error {:?}: {}",
744                    status.code(),
745                    status.message()
746                );
747            }
748        }
749    }
750
751    // Exited via `break` or via `None`. If `has_more()` promises further data, the resume cursor
752    // comes from the latest watermark received — on the fused last item, a ScanLimit end frame, or
753    // a prior beacon; a bare end frame is only sent when no progress claim exists.
754    ensure!(
755        !page.has_more() || page.last_cursor().is_some(),
756        "{rpc_name}: server reported more results but did not provide resume cursor — cannot continue",
757    );
758
759    Ok(page)
760}
