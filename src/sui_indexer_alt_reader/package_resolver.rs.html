27/// Package store that reads packages from the ledger gRPC service (kv-rpc or fullnode) by their
28/// storage ID.
29pub struct GrpcPackageStore(Arc<DataLoader<LedgerGrpcReader>>);
30
31#[derive(Copy, Clone, Hash, Eq, PartialEq, Debug)]
32struct PackageKey(AccountAddress);
33
34impl GrpcPackageStore {
35    pub fn new(reader: &LedgerGrpcReader) -> Self {
36        Self(Arc::new(reader.as_data_loader()))
37    }
38}
39
40#[async_trait::async_trait]
41impl PackageStore for GrpcPackageStore {
42    async fn fetch(&self, id: AccountAddress) -> Result<Arc<Package>> {
43        let Self(loader) = self;
44        let Some(package) = loader.load_one(PackageKey(id)).await? else {
45            return Err(Error::PackageNotFound(id));
46        };
47
48        Ok(package)
49    }
50}
51
52#[async_trait::async_trait]
53impl ChunkedLoader<PackageKey> for LedgerGrpcReader {
54    type Value = Arc<Package>;
55    type Error = Error;
56
57    fn chunk_size(&self) -> usize {
58        self.max_batch_get_objects()
59    }
60
61    async fn load_chunk(&self, keys: &[PackageKey]) -> Result<HashMap<PackageKey, Arc<Package>>> {
62        let mut id_to_package = HashMap::new();
63        if keys.is_empty() {
64            return Ok(id_to_package);
65        }
66
67        let requests = keys
68            .iter()
69            .map(|PackageKey(id)| proto::GetObjectRequest::new(&ObjectID::from(*id).into()))
70            .collect();
71
72        let mut request = proto::BatchGetObjectsRequest::default();
73        request.requests = requests;
74        request.read_mask = Some(FieldMask::from_paths(["bcs"]));
75
76        let batch_response = self.batch_get_objects(request).await.map_err(store_err)?;
77
78        for obj_result in batch_response.objects {
79            let object = match obj_result.result {
80                Some(proto::get_object_result::Result::Object(object)) => object,
81
82                Some(proto::get_object_result::Result::Error(status))
83                    if status.code == tonic::Code::NotFound as i32 =>
84                {
85                    continue;
86                }
87
88                Some(proto::get_object_result::Result::Error(status)) => {
89                    return Err(store_err(format!(
90                        "{:?}: {}",
91                        tonic::Code::from(status.code),
92                        status.message
93                    )));
94                }
95
96                _ => {
97                    return Err(store_err("Empty or unrecognized result in batch response"));
98                }
99            };
100
101            let object: Object = object
102                .bcs
103                .as_ref()
104                .ok_or_else(|| store_err("Missing bcs in object"))?
105                .deserialize()
106                .map_err(|e| store_err(format!("Failed to deserialize object: {e}")))?;
107            // Any object ID can be requested, e.g. from a user-supplied type. A non-package object
108            // is a miss for its own key, not an error for every key in the shared batch.
109            let Some(move_package) = object.data.try_as_package() else {
110                continue;
111            };
112
113            let package = Package::read_from_package(move_package)?;
114            id_to_package.insert(PackageKey(*move_package.id()), Arc::new(package));
115        }
116
117        Ok(id_to_package)
118    }
119}
120
121fn store_err(e: impl ToString) -> Error {
122    Error::Store {
123        store: STORE,
124        error: e.to_string(),
125    }
126}

128#[cfg(test)]
129mod tests {
130    use sui_rpc::proto::google::rpc::Status as StatusProto;
131    use sui_rpc::proto::sui::rpc::v2::GetObjectResult;
132
133    use super::*;
134    use crate::ledger_grpc_reader::test_support::mock_reader;
135
136    fn error_result(code: tonic::Code, message: &str) -> GetObjectResult {
137        GetObjectResult::new_error(StatusProto {
138            code: code as i32,
139            message: message.to_owned(),
140            ..Default::default()
141        })
142    }
143
144    #[tokio::test]
145    async fn per_object_not_found_is_package_not_found() {
146        let (reader, mock, _server) = mock_reader().await;
147        mock.set_object_results(vec![error_result(
148            tonic::Code::NotFound,
149            "object not found",
150        )]);
151
152        let store = GrpcPackageStore::new(&reader);
153        let err = store.fetch(AccountAddress::TWO).await.unwrap_err();
154        assert!(matches!(err, Error::PackageNotFound(_)), "{err:?}");
155    }
156
157    #[tokio::test]
158    async fn per_object_backend_error_is_store_error() {
159        let (reader, mock, _server) = mock_reader().await;
160        mock.set_object_results(vec![error_result(tonic::Code::Internal, "boom")]);
161
162        let store = GrpcPackageStore::new(&reader);
163        let err = store.fetch(AccountAddress::TWO).await.unwrap_err();
164        assert!(
165            matches!(
166                &err,
167                Error::Store { store, error }
168                    if *store == STORE && error.contains("Internal") && error.contains("boom")
169            ),
170            "{err:?}"
171        );
172    }
173
174    #[tokio::test]
175    async fn per_object_empty_result_is_store_error() {
176        let (reader, mock, _server) = mock_reader().await;
177        mock.set_object_results(vec![GetObjectResult::default()]);
178
179        let store = GrpcPackageStore::new(&reader);
180        let err = store.fetch(AccountAddress::TWO).await.unwrap_err();
181        assert!(matches!(&err, Error::Store { .. }), "{err:?}");
182    }
183
184    #[tokio::test]
185    async fn non_package_object_is_package_not_found() {
186        let (reader, mock, _server) = mock_reader().await;
187        let object = Object::immutable_with_id_for_testing(ObjectID::from(AccountAddress::TWO));
188        let mut proto_object = proto::Object::default();
189        proto_object.bcs = Some(proto::Bcs::serialize(&object).unwrap());
190        mock.set_object_results(vec![GetObjectResult::new_object(proto_object)]);
191
192        let store = GrpcPackageStore::new(&reader);
193        let err = store.fetch(AccountAddress::TWO).await.unwrap_err();
194        assert!(matches!(err, Error::PackageNotFound(_)), "{err:?}");
195    }
