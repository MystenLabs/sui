4//! DNF bitmap index queries over ordered bucket streams.
5//!
6//! Callers build a `BitmapQuery` as an OR of terms. Each term is an AND of
7//! signed dimension-key literals. Evaluation yields matching bitmap members as
8//! they are produced. Back-pressure from downstream consumers, e.g. a
9//! `.take(page_size)`, propagates back to the backend-provided bucket streams
10//! and avoids materializing matches we won't use.
11//!
12//! Queries are intentionally restricted to anchored DNF: every term must contain
13//! at least one positive literal. Positive literals give the evaluator concrete
14//! bitmap streams to scan and intersect; negative literals only shrink those
15//! candidate streams. Negative-only terms such as `NOT sender = A` are
16//! supported by anchoring them upstream on a universe include — a stored
17//! existence marker in event-space (`EventExtant`), a scan-time-synthesized
18//! dense leaf in tx-space (`TxUniverse`, see [`dense_universe_buckets`]) — so
19//! the evaluator itself stays a set of ordered stream merge-joins with no
20//! complement-specific code path. The full-range scan such a term implies is
21//! inherent to negation and is bounded by the per-request bucket budget.
22//!
23//! Backends provide one ordered `(bucket_id, RoaringBitmap)` stream or iterator
24//! per dimension key. The merge-join machinery here is storage-agnostic:
25//! BigTable, RocksDB, or any other backend can reuse it as long as its bucket
26//! source is sparse, ordered by the requested scan direction, and stores bitmap
27//! positions relative to that bucket.

54/// Terminal raised by a single leaf/backend bucket stream. Positionless by
55/// construction — one leaf cannot know the merged multi-term floor. Leaf stops
56/// never reach a List driver: the DNF evaluators consume them and re-raise a
57/// [`ScanStop`] carrying the merged frontier.
58#[derive(Debug, thiserror::Error)]
59pub enum LeafStop {
60    /// This leaf's share of the bucket-scan budget is exhausted.
61    #[error("bitmap scan limit reached")]
62    BudgetExhausted,
63    /// The request's cancellation token fired.
64    #[error("bitmap scan cancelled")]
65    Cancelled,
66    /// A backend/storage fault.
67    #[error(transparent)]
68    Fault(anyhow::Error),
69}

71/// Terminal signal of a merged bitmap eval stream, raised on the error channel
72/// so the `try_stream!` pipeline short-circuits (a clean end-of-stream means
73/// "scanned the whole range"). The List handlers map each variant to a wire
74/// outcome with one exhaustive match.
75///
76/// Mental model: leaves raise positionless [`LeafStop`]s; the evaluator turns
77/// them into a `ScanStop` that ALWAYS carries the resume frontier on
78/// `ScanLimit`; in-band `Watermarked::Watermark`s are only mid-scan progress
79/// beacons, never the resume channel.
80#[derive(Debug, thiserror::Error)]
81pub enum ScanStop {
82    /// Budget exhausted: a graceful early stop. The handler ends the stream
83    /// with `QUERY_END_REASON_SCAN_LIMIT`; the frontier is the resume cursor.
84    #[error("bitmap scan limit reached")]
85    ScanLimit {
86        /// Merged floor position every term provably scanned to when the budget
87        /// died — the exact value the stopping round's progress beacon would
88        /// have carried. It lives in the scanned index's member-id domain
89        /// (tx-seq for transaction/checkpoint indexes; encoded event-seq — see
90        /// `event_seq` — for the event index). Consumers apply their existing
91        /// domain conversion and resume arithmetic.
92        scan_frontier: u64,
93    },
94    /// Cancelled → gRPC `Cancelled` status.
95    #[error("bitmap scan cancelled")]
96    Cancelled,
97    /// Backend/storage fault → gRPC `Internal`, error carried unchanged.
98    #[error(transparent)]
99    Fault(anyhow::Error),
100}
101
102impl From<anyhow::Error> for LeafStop {
103    fn from(err: anyhow::Error) -> Self {
104        LeafStop::Fault(err)
105    }
106}

108impl From<anyhow::Error> for ScanStop {
109    fn from(err: anyhow::Error) -> Self {
110        ScanStop::Fault(err)
111    }

114/// Reduce the leaf stops raised in one evaluator round to the driver-facing
115/// terminal. Precedence: Fault > BudgetExhausted > Cancelled (a real fault must
116/// surface as Internal; a budget stop with its frontier beats a bare Cancel —
117/// the resume point costs nothing to deliver). `frontier` is the merged floor
118/// the caller computed for this round; it is bound into `ScanLimit` only.
119/// Panics on empty input — evaluators only collapse when a leaf actually erred.
120pub(crate) fn collapse(stops: Vec<LeafStop>, scan_frontier: u64) -> ScanStop {
121    assert!(!stops.is_empty(), "collapse on empty Vec");
122    let mut cancelled = false;
123    let mut budget = false;
124    let mut faults: Vec<anyhow::Error> = Vec::new();
125    for s in stops {
126        match s {
127            LeafStop::Cancelled => cancelled = true,
128            LeafStop::BudgetExhausted => budget = true,
129            LeafStop::Fault(err) => faults.push(err),
130        }
131    }
132    match faults.len() {
133        0 => {
134            if budget {
135                ScanStop::ScanLimit { scan_frontier }
136            } else {
137                debug_assert!(cancelled, "collapse saw only non-erroring leaves");
138                ScanStop::Cancelled
139            }
140        }
141        1 => ScanStop::Fault(faults.pop().expect("len == 1")),
142        n => {
143            let combined = faults
144                .iter()
145                .enumerate()
146                .map(|(i, err)| format!("  [{i}] {err}"))
147                .collect::<Vec<_>>()
148                .join("\n");
149            ScanStop::Fault(anyhow::anyhow!(
150                "{n} concurrent bitmap scan faults:\n{combined}"
151            ))
152        }
153    }
154}

156/// Item or progress watermark flowing through a bitmap eval pipeline.
157/// `Watermark(p)` means every Item with position strictly before `p`
158/// in scan direction has been emitted upstream. Downstream stages must
159/// preserve watermark/item ordering — that's what makes the watermark a
160/// safe resume cursor on timeout.
161#[derive(Clone, Copy, Debug, Eq, PartialEq)]
162pub enum Watermarked<T, P = u64> {
163    Item(T),
164    Watermark(P),
165}
166
167impl<T, P> Watermarked<T, P> {
168    pub fn map_item<U>(self, f: impl FnOnce(T) -> U) -> Watermarked<U, P> {
169        match self {
170            Watermarked::Item(t) => Watermarked::Item(f(t)),
171            Watermarked::Watermark(p) => Watermarked::Watermark(p),
172        }
173    }
174
175    pub fn map_watermark<Q>(self, f: impl FnOnce(P) -> Q) -> Watermarked<T, Q> {
176        match self {
177            Watermarked::Item(t) => Watermarked::Item(t),
178            Watermarked::Watermark(p) => Watermarked::Watermark(f(p)),
179        }
180    }
181}

200/// Physical gap-crossing policy. Logical frontier jumps are unconditional;
201/// this only bounds how many charged dead rows a lagging leaf drains from its
202/// open scan per lag episode before seeking past the gap.
203#[derive(Clone, Copy, Debug)]
204pub struct SkipPolicy {
205    /// `None` never seeks and drains dead rows without bound.
206    pub drain_probe_rows: Option<std::num::NonZeroU32>,
207}
208
209impl SkipPolicy {
210    /// Drain dead rows without seeking; logical frontier jumps remain enabled.
211    pub const DRAIN_ONLY: Self = Self {
212        drain_probe_rows: None,
213    };
214
215    /// What a gap catch-up step that reopens a leaf scan costs, in bucket
216    /// rows: the dead rows this policy drains before it judges a reopen
217    /// cheaper. 0 when it never reopens.
218    pub(crate) fn reopen_cost_rows(self) -> u64 {
219        self.drain_probe_rows
220            .map_or(0, |probe| u64::from(probe.get()))
221    }
222}
223
224impl ScanDirection {
225    pub fn is_ascending(self) -> bool {
226        matches!(self, Self::Ascending)
227    }
228}

230/// How often an evaluator emits the progress watermarks it computes.
231///
232/// A watermark is emitted once `min_buckets` of scan work have been done since
233/// the last emission; until then it is held, replacing any older held one.
234/// The scan's first output is never held. Items and stops pass through
235/// unchanged.
236///
237/// Scan work is counted in bucket rows: the rows charged to the budget
238/// ([`BitmapScanMetrics::buckets_evaluated`]), plus, for every gap catch-up
239/// step that reopened at least one leaf scan, the evaluation's
240/// [`SkipPolicy::drain_probe_rows`]. A reopen is a new backend read, and the
241/// skip policy reopens exactly when it judges that cheaper than draining that
242/// many dead rows, so it is priced at them. Leaves reopened in the same step
243/// count once, because the stream evaluator reopens them concurrently.
244///
245/// An emission is an emitted watermark or an item with a match inside the
246/// scan range; such an item also discards the held watermark. Items are whole
247/// buckets, so when the range starts or ends mid-bucket (e.g. a page resuming
248/// from a cursor) an edge item can hold only matches outside it. That item
249/// sends the client nothing and is not an emission.
250///
251/// The held watermark is discarded at every stop: a [`ScanStop::ScanLimit`]
252/// carries a frontier further along, and after a [`ScanStop::Fault`] or
253/// [`ScanStop::Cancelled`] the consumer is left with the last emitted
254/// watermark. At a clean end it is emitted only with `emit_held_at_end`.
255#[derive(Clone, Copy, Debug, Eq, PartialEq)]
256pub struct WatermarkPacing {
257    /// Scan work, in bucket rows, between emissions. With 0 nothing is held.
258    pub min_buckets: u64,
259    /// Emit the held watermark at a clean end instead of discarding it.
260    pub emit_held_at_end: bool,
261}

263/// Bucket-scan accounting for one evaluation, passed once to `on_metrics`
264/// when the evaluator's stream or iterator is dropped. Nothing is reported
265/// when the stream evaluator rejects a misconfigured budget up front.
266#[derive(Clone, Copy, Debug, Eq, PartialEq)]
267pub struct BitmapScanMetrics {
268    /// Bucket rows charged against the budget, including dead rows drained
269    /// during gap catch-up.
270    pub buckets_evaluated: u64,
271    /// Charged dead bucket rows discarded while catching up lagging leaves.
272    pub buckets_discarded: u64,
273    /// Physical leaf scans abandoned and reopened at a later bucket.
274    pub leaf_seeks: u64,
275}

277/// Bucket budget shared by all leaf scans of one evaluation; each leaf charges
278/// a bucket as it pulls it (`stream::budgeted_bucket_stream`,
279/// `iter::BudgetedBucketIter`).
280#[derive(Clone)]
281pub(crate) struct BitmapScanBudget {
282    initial: u64,
283    remaining: Arc<AtomicU64>,
284    discarded: Arc<AtomicU64>,
285    seeks: Arc<AtomicU64>,
286    reopen_steps: Arc<AtomicU64>,
287}
288
289impl BitmapScanBudget {
290    pub(crate) fn new(initial: u64) -> Self {
291        Self {
292            initial,
293            remaining: Arc::new(AtomicU64::new(initial)),
294            discarded: Arc::new(AtomicU64::new(0)),
295            seeks: Arc::new(AtomicU64::new(0)),
296            reopen_steps: Arc::new(AtomicU64::new(0)),
297        }
298    }
299
300    /// Charge one bucket. Returns false on underflow.
301    fn try_take(&self) -> bool {
302        self.remaining
303            .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |b| {
304                if b == 0 { None } else { Some(b - 1) }
305            })
306            .is_ok()
307    }
308
309    /// Charge a leaf's first bucket; always succeeds. Without this
310    /// reservation a sparse term could drain the shared pool before a sibling
311    /// leaf reaches its first bucket, and that leaf could never report a
312    /// position (a cursorless `SCAN_LIMIT`). The pool is still decremented
313    /// when not empty, so `buckets_evaluated` undercounts only first buckets
314    /// taken after it ran out.
315    fn take_first(&self) {
316        let _ = self
317            .remaining
318            .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |b| {
319                Some(b.saturating_sub(1))
320            });
321    }
322
323    /// Bucket rows charged so far ([`BitmapScanMetrics::buckets_evaluated`]);
324    /// [`WatermarkPacer`] counts these.
325    fn buckets_evaluated(&self) -> u64 {
326        self.initial
327            .saturating_sub(self.remaining.load(Ordering::SeqCst))
328    }
329
330    fn note_discarded(&self) {
331        self.discarded.fetch_add(1, Ordering::Relaxed);
332    }
333
334    fn note_seek(&self) {
335        self.seeks.fetch_add(1, Ordering::Relaxed);
336    }
337
338    /// Count one gap catch-up step that reopened at least one leaf scan.
339    fn note_reopen_step(&self) {
340        self.reopen_steps.fetch_add(1, Ordering::Relaxed);
341    }
342
343    fn discarded(&self) -> u64 {
344        self.discarded.load(Ordering::Relaxed)
345    }
346
347    fn seeks(&self) -> u64 {
348        self.seeks.load(Ordering::Relaxed)
349    }
350
351    /// Gap catch-up steps that reopened a leaf scan so far;
352    /// [`WatermarkPacer`] counts these.
353    fn reopen_steps(&self) -> u64 {
354        self.reopen_steps.load(Ordering::Relaxed)
355    }
356}

359/// Evaluators keep it inside the stream or iterator they return.

366    fn drop(&mut self) {
367        if let Some(cb) = self.callback.take() {

370                buckets_discarded: self.budget.discarded(),
371                leaf_seeks: self.budget.seeks(),
372            });
373        }
374    }

377/// Applies a [`WatermarkPacing`] to one evaluation's bucket-level output,
378/// reading the scan work done from the evaluation's shared budget.
379///
380/// The stream evaluator outputs a round before that round's gap catch-up and
381/// the iterator evaluator after it, so when catch-up drains rows or reopens
382/// leaves the two can pace the same scan differently.
383pub(crate) struct WatermarkPacer {
384    pacing: WatermarkPacing,
385    budget: BitmapScanBudget,
386    /// [`SkipPolicy::reopen_cost_rows`] of the evaluation's skip policy.
387    reopen_cost_rows: u64,
388    /// Scan range; the member-level output drops matches outside it.
389    range: Range<u64>,
390    bucket_size: u64,
391    /// Newest watermark not yet emitted; always newer than every emitted one.
392    held: Option<u64>,
393    /// Scan work done at the last emission; `None` before the first.
394    work_at_last_emission: Option<u64>,
395}
396
397impl WatermarkPacer {
398    pub(crate) fn new(
399        pacing: WatermarkPacing,
400        policy: SkipPolicy,
401        budget: BitmapScanBudget,
402        range: Range<u64>,
403        bucket_size: u64,
404    ) -> Self {
405        Self {
406            pacing,
407            budget,
408            reopen_cost_rows: policy.reopen_cost_rows(),
409            range,
410            bucket_size,
411            held: None,
412            work_at_last_emission: None,
413        }
414    }
415
416    /// The output to emit for one unpaced output: the output itself, or
417    /// nothing while a watermark is held.
418    pub(crate) fn pace(&mut self, output: WatermarkedBucket) -> Option<WatermarkedBucket> {
419        match output {
420            Ok(Watermarked::Item((bucket, bitmap))) => {
421                if self.has_member_in_range(bucket, &bitmap) {
422                    self.record_emission(self.scan_work());
423                }
424                Some(Ok(Watermarked::Item((bucket, bitmap))))
425            }
426            Ok(Watermarked::Watermark(position)) => {
427                let work = self.scan_work();
428                let due = self
429                    .work_at_last_emission
430                    .is_none_or(|last| work.saturating_sub(last) >= self.pacing.min_buckets);
431                if due {
432                    self.record_emission(work);
433                    Some(Ok(Watermarked::Watermark(position)))
434                } else {
435                    self.held = Some(position);
436                    None
437                }
438            }
439            Err(stop) => {
440                self.held = None;
441                Some(Err(stop))
442            }
443        }
444    }
445
446    /// Scan work done so far, in bucket rows (see [`WatermarkPacing`]).
447    fn scan_work(&self) -> u64 {
448        self.budget.buckets_evaluated().saturating_add(
449            self.reopen_cost_rows
450                .saturating_mul(self.budget.reopen_steps()),
451        )
452    }
453
454    fn record_emission(&mut self, work: u64) {
455        self.held = None;
456        self.work_at_last_emission = Some(work);
457    }
458
459    /// Whether `bitmap` (offsets within `bucket`) has a member inside the scan
460    /// range.
461    fn has_member_in_range(&self, bucket: u64, bitmap: &RoaringBitmap) -> bool {
462        let bucket_start = bucket.saturating_mul(self.bucket_size);
463        let low = self
464            .range
465            .start
466            .saturating_sub(bucket_start)
467            .min(self.bucket_size);
468        let high = self
469            .range
470            .end
471            .saturating_sub(bucket_start)
472            .min(self.bucket_size);
473        low < high && bitmap.range(low as u32..high as u32).next().is_some()
474    }
475
476    /// The watermark to emit when the unpaced output ends without a stop.
477    pub(crate) fn finish(&mut self) -> Option<u64> {
478        let held = self.held.take();
479        held.filter(|_| self.pacing.emit_held_at_end)
480    }
481}

483/// Synthesized bucket sequence for the dense tx-seq universe: one full bitmap
484/// per bucket touched by `range`, in scan-direction order. Backends return this
485/// for the query-only `IndexDimension::TxUniverse` key instead of reading
486/// storage — the tx-seq namespace is dense, so the universe is computable.
487///
488/// Bitmaps carry all `[0, bucket_size)` relative bits even in edge buckets;
489/// trimming against the requested range happens at the flatten step, same as
490/// for stored buckets.
491pub fn dense_universe_buckets(
492    range: Range<u64>,
493    bucket_size: u64,
494    direction: ScanDirection,
495) -> DenseUniverseBuckets {
496    let bits = u32::try_from(bucket_size).expect("bucket size fits in u32");
497    let buckets = if range.is_empty() {
498        0..0
499    } else {
500        (range.start / bucket_size)..(range.end - 1) / bucket_size + 1
501    };
502    DenseUniverseBuckets {
503        buckets,
504        direction,
505        bits,
506    }
507}
508
509/// Synthesized dense-universe bucket iterator.
510pub struct DenseUniverseBuckets {
511    buckets: Range<u64>,
512    direction: ScanDirection,
513    bits: u32,
514}
515
516impl DenseUniverseBuckets {
517    /// Reposition to the first bucket at or past `bucket` in scan direction.
518    pub fn seek_bucket(&mut self, bucket: u64) {
519        match self.direction {
520            ScanDirection::Ascending => self.buckets.start = self.buckets.start.max(bucket),
521            ScanDirection::Descending => {
522                self.buckets.end = self.buckets.end.min(bucket.saturating_add(1))
523            }
524        }
525    }
526}
527
528impl Iterator for DenseUniverseBuckets {
529    type Item = (u64, RoaringBitmap);
530
531    fn next(&mut self) -> Option<Self::Item> {
532        let bucket = match self.direction {
533            ScanDirection::Ascending => self.buckets.next(),
534            ScanDirection::Descending => self.buckets.next_back(),
535        }?;
536        let mut bitmap = RoaringBitmap::new();
537        bitmap.insert_range(0..self.bits);
538        Some((bucket, bitmap))
539    }
540}

707/// Deduplicated leaf list + per-term references over those leaves, ready for
708/// the evaluator. `keys[i]` is the dimension-key bytes for leaf `i`; each
709/// [`TermSpec`] references leaves by that index.

711    pub(crate) keys: Vec<Vec<u8>>,
712    pub(crate) terms: Vec<TermSpec>,

715/// Deduplicate literals across the whole query and translate each term's
716/// includes/excludes into indices over the shared leaf list.
717///
718/// Dedup is keyed on encoded dimension-key bytes, so the same key appearing
719/// in multiple terms (e.g. `(sender=A AND module=X) OR (sender=A AND
720/// type=Y)`) maps to a single backend scan instead of two duplicate scans.
721pub(crate) fn build_term_specs(terms: Vec<BitmapTerm>) -> DedupedQuery {
722    let mut key_to_idx: HashMap<Vec<u8>, usize> = HashMap::new();
723    let mut keys: Vec<Vec<u8>> = Vec::new();
724    let mut specs: Vec<TermSpec> = Vec::with_capacity(terms.len());
725    for term in terms {
726        let mut include_idx = Vec::with_capacity(term.literals.len());
727        let mut exclude_idx = Vec::with_capacity(term.literals.len());
728        for literal in term.literals {
729            let (push_target, key) = match literal {
730                BitmapLiteral::Include(k) => (&mut include_idx, k.into_inner()),
731                BitmapLiteral::Exclude(k) => (&mut exclude_idx, k.into_inner()),
732            };
733            let idx = match key_to_idx.get(&key) {
734                Some(&i) => i,
735                None => {
736                    let i = keys.len();
737                    key_to_idx.insert(key.clone(), i);
738                    keys.push(key);
739                    i
740                }
741            };
742            push_target.push(idx);
743        }
744        specs.push(TermSpec {
745            includes: include_idx,
746            excludes: exclude_idx,
747            unsatisfiable: false,
748        });
749    }
750    DedupedQuery { keys, terms: specs }
751}

753/// The less-advanced of two frontier positions in scan direction: the min
754/// ascending, the max descending. Used to keep a merged frontier bounded by
755/// the slowest contributor.
756fn bound_in_direction(a: u64, b: u64, direction: ScanDirection) -> u64 {
757    match direction {
758        ScanDirection::Ascending => a.min(b),
759        ScanDirection::Descending => a.max(b),
760    }

762/// The more-advanced of two positions in scan direction: max ascending, min
763/// descending.
764pub(crate) fn advance_in_direction(a: u64, b: u64, direction: ScanDirection) -> u64 {
765    match direction {
766        ScanDirection::Ascending => a.max(b),
767        ScanDirection::Descending => a.min(b),
768    }

771/// Whether `a` is strictly before `b` in scan direction.
772pub(crate) fn strictly_before(a: u64, b: u64, direction: ScanDirection) -> bool {
773    match direction {

779/// Whether emitting `next` as a watermark advances the frontier past the
780/// previously emitted one. Ascending frontiers strictly increase,
781/// descending strictly decrease; the first watermark always advances.
782fn frontier_advanced(prev: Option<u64>, next: u64, direction: ScanDirection) -> bool {
783    match prev {
784        None => true,
785        Some(prev) => match direction {
786            ScanDirection::Ascending => next > prev,
787            ScanDirection::Descending => next < prev,
788        },
789    }

792/// Clamped member-id edges of `bucket` in scan direction: `(pre, post)` where
793/// `pre` is the leading edge (everything before it is already covered) and
794/// `post` is the trailing edge (everything up to and including the bucket is
795/// covered). Ascending: `(low, high)`; descending: `(high, low)`. Both clamped
796/// to the request range so cursors stay in-bounds when they round-trip into a
797/// follow-up request with a different range.
798pub(crate) fn bucket_edges(
799    bucket: u64,
800    bucket_size: u64,
801    range: &Range<u64>,
802    direction: ScanDirection,
803) -> (u64, u64) {
804    let start = bucket.saturating_mul(bucket_size);
805    let end = start.saturating_add(bucket_size);
806    match direction {
807        ScanDirection::Ascending => (start.max(range.start), end.min(range.end)),
808        ScanDirection::Descending => (end.min(range.end), start.max(range.start)),
809    }
810}

812/// Pull the snapshotted bitmap for leaf `i` to give to one referencing term
813/// this round. Returns `None` if the leaf isn't on the floor bucket (so the
814/// term short-circuits). When `remaining_refs[i] > 1`, the bitmap is cloned so
815/// other referencing terms still see it; the last reference takes by value to
816/// avoid an unnecessary copy in the common single-reference case.
817pub(crate) fn take_snapshot_bitmap(
818    snapshot: &mut [Option<RoaringBitmap>],
819    remaining_refs: &mut [usize],
820    on_floor: &[bool],
821    i: usize,
822) -> Option<RoaringBitmap> {
823    if !on_floor[i] {
824        return None;
825    }
826    if remaining_refs[i] > 1 {
827        remaining_refs[i] -= 1;
828        snapshot[i].clone()
829    } else {
830        remaining_refs[i] = remaining_refs[i].saturating_sub(1);
831        snapshot[i].take()
832    }
833}

835/// Per-round refcount: how many satisfiable term-side slots reference each
836/// on-floor leaf. Drives [`take_snapshot_bitmap`]'s take-vs-clone decision so
837/// the last referencing slot reclaims the bitmap by value.
838pub(crate) fn count_on_floor_refs(terms: &[TermSpec], on_floor: &[bool]) -> Vec<usize> {
839    let mut refs = vec![0usize; on_floor.len()];
840    for term in terms {
841        if term.unsatisfiable {
842            continue;
843        }
844        for &i in term.includes.iter().chain(term.excludes.iter()) {
845            if on_floor[i] {
846                refs[i] += 1;
847            }
848        }
849    }
850    refs
851}

853/// Recompute leaf liveness from current term state. A leaf becomes
854/// `unreferenced` when no satisfiable term still points at it, or when its
855/// head is at EOF (the bucket stream is permanently exhausted; any further
856/// peek would be wasted work, and any include-referencing term will be marked
857/// `unsatisfiable` separately).
858///
859/// `unreferenced` is monotonic — entries only transition false → true — so
860/// this is safe to invoke each round; previously-retired leaves stay retired.
861pub(crate) fn recompute_unreferenced(
862    terms: &[TermSpec],
863    class: &[Option<LeafHead>],
864    unreferenced: &mut [bool],
865) {
866    let leaf_count = unreferenced.len();
867    let mut referenced = vec![false; leaf_count];
868    for term in terms {
869        if term.unsatisfiable {
870            continue;
871        }
872        for &i in term.includes.iter().chain(term.excludes.iter()) {
873            if !unreferenced[i] && !matches!(class[i], Some(LeafHead::Eof)) {
874                referenced[i] = true;
875            }
876        }
877    }
878    for i in 0..leaf_count {
879        if !referenced[i] {
880            unreferenced[i] = true;
881        }
882    }
883}

885/// Evaluate one DNF term at a single bucket from the per-leaf bitmaps present
886/// there: intersect the includes (any absent include ⇒ empty term), then
887/// subtract the union of the present excludes (`a AND NOT b`). Returns the
888/// term's matches at the bucket, or `None` if empty. Bitmaps are taken by value
889/// so the caller hands over the consumed leaf rows without cloning.
890pub(crate) fn eval_term_at_bucket(
891    includes: Vec<Option<RoaringBitmap>>,
892    excludes: Vec<Option<RoaringBitmap>>,
893) -> Option<RoaringBitmap> {
894    let mut acc: Option<RoaringBitmap> = None;
895    for include in includes {
896        // A missing include means the intersection is empty at this bucket.
897        let bitmap = include?;
898        acc = Some(match acc {
899            None => bitmap,
900            Some(a) => a & bitmap,
901        });
902    }
903    // Anchored terms always carry at least one include, so `acc` is `Some`.
904    let mut acc = acc?;
905    for exclude in excludes.into_iter().flatten() {
906        acc -= exclude;
907    }
908    (!acc.is_empty()).then_some(acc)
909}

911/// One DNF term, as index spans into the flat leaf vector. Shared by the stream
912/// and iterator drivers.
913pub(crate) struct TermSpec {
914    pub(crate) includes: Vec<usize>,
915    pub(crate) excludes: Vec<usize>,
916    /// Set once any include leaf hits EOF: the term's intersection is
917    /// permanently empty (it can never match again). Latched.
918    pub(crate) unsatisfiable: bool,
919}

921/// A leaf's head this round, from a non-consuming peek.
922pub(crate) enum LeafHead {
923    Bucket(u64),
924    Eof,
925    Error,
926}
927/// Compute the furthest bucket each active leaf can skip to without changing
928/// query results.
929pub(crate) fn leaf_skip_targets(
930    terms: &[TermSpec],
931    class: &[Option<LeafHead>],
932    unreferenced: &[bool],
933    direction: ScanDirection,
934) -> Vec<Option<u64>> {
935    // Go through each conjunction in the top-level query disjunction.
936    let term_not_before: Vec<Option<u64>> = terms
937        .iter()
938        .map(|term| {
939            if term.unsatisfiable {
940                return None;
941            }
942            // For each conjunction, find its furthest-ahead include leaf.
943            // As far as this conjunction is concerned, every leaf it references
944            // can skip to that bucket. A shared leaf may still be constrained by
945            // another conjunction. Excludes don't set this bound. No exclude row
946            // just means nothing to subtract.
947            term.includes
948                .iter()
949                .filter_map(|&i| match class.get(i).and_then(Option::as_ref) {
950                    Some(LeafHead::Bucket(bucket)) => Some(*bucket),
951                    Some(LeafHead::Error | LeafHead::Eof) | None => None,
952                })
953                .reduce(|a, b| advance_in_direction(a, b, direction))
954        })
955        .collect();
956
957    // Now with term-level knowledge, iterate through the deduped leaves. Find the nearest
958    // safe bound that we can skip ahead to among all terms that reference a leaf.
959    class
960        .iter()
961        .enumerate()
962        .map(|(i, head)| {
963            if unreferenced.get(i).copied().unwrap_or(true) {
964                return None;
965            }
966            let Some(LeafHead::Bucket(head)) = head else {
967                return None;
968            };
969
970            let mut target = None;
971            for (term_index, term) in terms.iter().enumerate() {
972                if term.unsatisfiable
973                    || !term
974                        .includes
975                        .iter()
976                        .chain(&term.excludes)
977                        .any(|&leaf| leaf == i)
978                {
979                    continue;
980                }
981
982                let not_before = term_not_before[term_index]?;
983                target = Some(match target {
984                    None => not_before,
985                    Some(current) => bound_in_direction(current, not_before, direction),
986                });
987            }
988            target.filter(|&target| strictly_before(*head, target, direction))
989        })
990        .collect()
991}

1307    /// `0..bucket_count`; `b` also holds member 1 in `matching_buckets`, the
1308    /// only buckets [`parallel_leaves_query`] matches. Neither leaf lags, so
1309    /// each bucket charges exactly two rows.
1310    pub(crate) fn parallel_leaves_source(
1311        bucket_count: u64,
1312        matching_buckets: &[u64],
1313    ) -> TestBucketSource {
1314        const MATCHED_MEMBERS: &[u32] = &[1];
1315        let matches: Vec<(u64, &[u32])> = matching_buckets
1316            .iter()
1317            .map(|&bucket| (bucket, MATCHED_MEMBERS))
1318            .collect();
1319        parallel_leaves_source_with_matches(bucket_count, &matches)
1320    }

1322    /// [`parallel_leaves_source`] where [`parallel_leaves_query`] matches
1323    /// exactly the listed members (relative to the bucket start) of each
1324    /// listed bucket: both leaves hold them in addition to their own member.
1325    pub(crate) fn parallel_leaves_source_with_matches(
1326        bucket_count: u64,
1327        matches: &[(u64, &[u32])],
1328    ) -> TestBucketSource {
1329        let leaf_rows = |own_member: u32| -> Vec<(u64, Vec<u32>)> {
1330            (0..bucket_count)
1331                .map(|bucket| {
1332                    let mut members = vec![own_member];
1333                    for (_, matched) in matches.iter().filter(|(b, _)| *b == bucket) {
1334                        members.extend_from_slice(matched);
1335                    }
1336                    (bucket, members)
1337                })
1338                .collect()
1339        };
1340        TestBucketSource {
1341            buckets: Arc::new(BTreeMap::from([
1342                (test_key(b"a"), leaf_rows(1)),
1343                (test_key(b"b"), leaf_rows(2)),
1344            ])),
1345        }
1346    }

1348    /// A [`TestBucketSource`] that counts the scan work its scans do: the

1350    /// [`BitmapScanMetrics::buckets_evaluated`]) and the leaf scans reopened
1351    /// ([`BitmapScanMetrics::leaf_seeks`]). A stream scan opened again for a

1357        reopens: Arc<AtomicU64>,

1361    impl ScanWorkCountingSource {
1362        pub(crate) fn new(inner: TestBucketSource) -> Self {
1363            Self {
1364                inner,
1365                rows_yielded: Arc::new(AtomicU64::new(0)),
1366                reopens: Arc::new(AtomicU64::new(0)),
1367                opened_keys: Arc::new(Mutex::new(HashSet::new())),
1368            }
1369        }

1372            self.rows_yielded.load(Ordering::SeqCst)

1377        }

1380    impl BitmapBucketSource for ScanWorkCountingSource {
1381        fn scan_bucket_stream(
1382            &self,
1383            dimension_key: Vec<u8>,
1384            range: Range<u64>,
1385            direction: ScanDirection,
1386        ) -> BucketStream {
1387            let first_open = self
1388                .opened_keys
1389                .lock()
1390                .unwrap()
1391                .insert(dimension_key.clone());
1392            if !first_open {
1393                self.reopens.fetch_add(1, Ordering::SeqCst);
1394            }
1395            let rows_yielded = self.rows_yielded.clone();
1396            self.inner
1397                .scan_bucket_stream(dimension_key, range, direction)
1398                .inspect(move |row| {
1399                    if row.is_ok() {
1400                        rows_yielded.fetch_add(1, Ordering::SeqCst);
1401                    }
1402                })
1403                .boxed()
1404        }
1405    }
