4//! Request-scoped BigTable facade.
5//!
6//! Downstream read APIs on this wrapper acquire a permit from a single
7//! `Arc<Semaphore>` before dispatching to the underlying
8//! `sui_kvstore::BigTableClient`.
9//! For streaming methods, the permit is bundled into the returned stream's
10//! state and released only when the stream is dropped. Pipeline helpers drain
11//! those streams inside one chunk future so live permit-holding streams do not
12//! cross stage boundaries. For non-streaming methods, the permit is held for
13//! the duration of the call.
14//!
15//! Bitmap scans are intentionally not gated by this downstream semaphore:
16//! their fanout is bounded by `max_bitmap_filter_literals`, and filtered list
17//! requests must still be able to run bitmap discovery with a low downstream
18//! request budget.
19//!
20//! The wrapper is constructed per request (`BigTableClient::new(client, capacity, ...)`)
21//! and threaded down through the pipeline.
22//!
23//! Each successful acquire wait is labeled with `stage` so the wait histogram
24//! can be sliced by call-site type. Per-request peak permits and successful
25//! acquisition count are recorded once on `BigTableClientContext` drop.
26

75/// Metrics for the request-scoped `BigTableClient` limiter and the reads it
76/// gates. Cardinality is bounded: per method, 5 stages for `permit_wait_ms`,
77/// `permit_hold_ms`, and `reads_started`, 5 stages × 3 outcomes for
78/// `reads_finished`, and one series each for the request-end `permits_peak`
79/// and `ops_total` histograms.
80#[derive(Clone)]
81pub(crate) struct Metrics {
82    /// Time spent waiting for one limiter permit, recorded once per successful
83    /// `acquire_owned()` call.
84    pub permit_wait_ms: HistogramVec,
85    /// Time one limiter permit is held from acquisition through release.
86    pub permit_hold_ms: HistogramVec,
87    /// Peak in-use permit count observed during a single request, recorded
88    /// once when the request's `BigTableClient` is dropped.
89    pub permits_peak: HistogramVec,
90    /// Total limiter acquisitions issued by a single request, recorded once
91    /// when the request's `BigTableClient` is dropped.
92    pub ops_total: HistogramVec,
93    /// Gated reads started, counted when the read's permit is acquired, just
94    /// before it is sent.
95    pub reads_started: IntCounterVec,
96    /// Gated reads that ended, by [`ReadOutcome`], counted when the read's
97    /// permit is released.
98    pub reads_finished: IntCounterVec,
99}

171/// Per-request shared state. Cloned along with `BigTableClient` (each clone is
172/// just an `Arc` bump). When the last clone is dropped, `Drop` records the
173/// per-request peak and ops-total histograms.
174pub(crate) struct BigTableClientContext {
175    metrics: Arc<Metrics>,
176    method: &'static str,
177    active_permits: AtomicUsize,
178    peak_used: AtomicUsize,
179    ops_count: AtomicUsize,
180}
181
182impl Drop for BigTableClientContext {
183    fn drop(&mut self) {
184        let peak = *self.peak_used.get_mut() as f64;
185        let ops = *self.ops_count.get_mut() as f64;
186        self.metrics
187            .permits_peak
188            .with_label_values(&[self.method])
189            .observe(peak);
190        self.metrics
191            .ops_total
192            .with_label_values(&[self.method])
193            .observe(ops);
194    }
195}

379    /// Open a streaming read under a `stage` permit. The permit stays with the
380    /// returned stream until it is dropped, and records how the read ended.
381    async fn open_gated_stream<T, S>(
382        &self,
383        stage: &'static str,
384        open: impl Future<Output = anyhow::Result<S>>,
385    ) -> Result<BoxStream<'static, Result<T, anyhow::Error>>, RpcError>
386    where
387        T: Send + 'static,
388        S: Stream<Item = anyhow::Result<T>> + Send + 'static,
389    {
390        let mut permit = self.acquire(stage).await?;
391        match open.await {
392            Ok(inner) => Ok(gate_stream(permit, inner.boxed())),
393            Err(error) => {
394                permit.finish(ReadOutcome::Failed);
395                Err(RpcError::from(error))
396            }
397        }
398    }
399
400    /// Run a read that returns its whole result under a `stage` permit.
401    async fn call_gated<T>(
402        &self,
403        stage: &'static str,
404        call: impl Future<Output = anyhow::Result<T>>,
405    ) -> Result<T, RpcError> {
406        let mut permit = self.acquire(stage).await?;
407        let result = call.await;
408        permit.finish(if result.is_ok() {
409            ReadOutcome::Completed
410        } else {
411            ReadOutcome::Failed
412        });
413        result.map_err(RpcError::from)
414    }

430    /// Build a resolver closure for tx-bitmap watermarks (identity
431    /// decode). Hand directly to [`crate::pipeline::resolve_scan_watermarks`].
432    pub(crate) fn tx_wm_resolver(
433        &self,
434        direction: ScanDirection,
435    ) -> impl Fn(u64) -> WmResolverFut + Send + 'static {
436        let client = self.clone();
437        move |position| {
438            let client = client.clone();
439            Box::pin(async move {
440                let lookup_tx_seq = if direction.is_ascending() {
441                    match position.checked_sub(1) {
442                        Some(tx_seq) => tx_seq,
443                        None => return Ok(None),
444                    }
445                } else {
446                    position
447                };
448                client
449                    .tx_seq_checkpoint(lookup_tx_seq)
450                    .await
451                    .map(Some)
452                    .map_err(|e| ScanStop::Fault(anyhow::Error::new(e)))
453            })
454        }
455    }

457    /// Build a resolver closure for event-bitmap watermarks. Hand directly to
458    /// [`crate::pipeline::resolve_scan_watermarks`].
459    pub(crate) fn event_wm_resolver(
460        &self,
461        direction: ScanDirection,
462    ) -> impl Fn(IntraTxCoordinate) -> WmResolverFut + Send + 'static {
463        let client = self.clone();
464        move |position| {
465            let client = client.clone();
466            Box::pin(async move {
467                let lookup_tx_seq = if direction.is_ascending() {
468                    if position.index > 0 {
469                        position.tx_seq
470                    } else {
471                        match position.tx_seq.checked_sub(1) {
472                            Some(tx_seq) => tx_seq,
473                            None => return Ok(None),
474                        }
475                    }
476                } else {
477                    position.tx_seq
478                };
479                client
480                    .tx_seq_checkpoint(lookup_tx_seq)
481                    .await
482                    .map(Some)
483                    .map_err(|e| ScanStop::Fault(anyhow::Error::new(e)))
484            })
485        }
486    }

494/// How a gated read ended, recorded once when its permit is released.
495#[derive(Clone, Copy, Debug, Eq, PartialEq)]
496enum ReadOutcome {
497    /// The response was read to its end.
498    Completed,
499    /// BigTable or decoding returned an error.
500    Failed,
501    /// Dropped before completing or failing. If the read was already sent,
502    /// dropping it resets its HTTP/2 stream.
503    Cancelled,
504}
505
506impl ReadOutcome {
507    fn label(self) -> &'static str {
508        match self {
509            Self::Completed => "completed",
510            Self::Failed => "failed",
511            Self::Cancelled => "cancelled",
512        }
513    }
514}

516struct LimitedPermit {
517    _permit: OwnedSemaphorePermit,
518    context: Arc<BigTableClientContext>,
519    stage: &'static str,
520    acquired_at: Instant,
521    /// `Cancelled` until the read records how it ended.
522    outcome: ReadOutcome,
523}
524
525impl LimitedPermit {
526    fn finish(&mut self, outcome: ReadOutcome) {
527        self.outcome = outcome;
528    }
529}
530
531impl Drop for LimitedPermit {
532    fn drop(&mut self) {
533        let hold_ms = self.acquired_at.elapsed().as_secs_f64() * 1000.0;
534        let metrics = &self.context.metrics;
535        let method = self.context.method;
536        metrics
537            .permit_hold_ms
538            .with_label_values(&[method, self.stage])
539            .observe(hold_ms);
540        metrics
541            .reads_finished
542            .with_label_values(&[method, self.stage, self.outcome.label()])
543            .inc();
544        record_release(&self.context);
545    }
546}
547
548async fn acquire_limited(
549    limiter: Arc<Semaphore>,
550    context: Arc<BigTableClientContext>,
551    stage: &'static str,
552) -> Result<LimitedPermit, tokio::sync::AcquireError> {
553    let start = Instant::now();
554    let permit = limiter.acquire_owned().await?;
555    let acquired_at = Instant::now();
556    record_acquired(&context, stage, acquired_at.duration_since(start));
557    Ok(LimitedPermit {
558        _permit: permit,
559        context,
560        stage,
561        acquired_at,
562        outcome: ReadOutcome::Cancelled,
563    })
564}
565
566fn record_acquired(context: &BigTableClientContext, stage: &'static str, wait: Duration) {
567    let wait_ms = wait.as_secs_f64() * 1000.0;
568    context
569        .metrics
570        .permit_wait_ms
571        .with_label_values(&[context.method, stage])
572        .observe(wait_ms);
573    context
574        .metrics
575        .reads_started
576        .with_label_values(&[context.method, stage])
577        .inc();
578    context.ops_count.fetch_add(1, Ordering::Relaxed);
579    let active = context.active_permits.fetch_add(1, Ordering::AcqRel) + 1;
580    context.peak_used.fetch_max(active, Ordering::Relaxed);
581}
582
583fn record_release(context: &BigTableClientContext) {
584    let previous = context
585        .active_permits
586        .fetch_update(Ordering::AcqRel, Ordering::Acquire, |active| {
587            active.checked_sub(1)
588        })
589        .unwrap_or(0);
590    debug_assert!(previous > 0, "limited client permit release underflow");
591}

593/// Wrap a stream so that `permit` is held until the stream is dropped (either
594/// by full drain, early termination by the consumer, or cancellation
595/// propagating through `Drop`). Items and errors pass through unchanged; the
596/// permit records the read as completed when the inner stream ends, failed
597/// when it yields an error, and cancelled if dropped before either.
598fn gate_stream<T>(
599    mut permit: LimitedPermit,
600    inner: BoxStream<'static, Result<T, anyhow::Error>>,
601) -> BoxStream<'static, Result<T, anyhow::Error>>
602where
603    T: Send + 'static,
604{
605    async_stream::try_stream! {
606        futures::pin_mut!(inner);
607        while let Some(row) = inner.next().await {
608            if row.is_err() {
609                permit.finish(ReadOutcome::Failed);
610            }
611            yield row?;
612        }
613        permit.finish(ReadOutcome::Completed);
614    }
615    .boxed()
616}

656    /// `gate_stream` should hold the permit until the wrapped stream is
657    /// dropped. With a capacity-1 semaphore, the wrapped stream's existence
658    /// blocks any other acquire, even between yields. Reading the inner
659    /// stream to its end records the read as completed.
660    #[tokio::test(start_paused = true)]
661    async fn gate_stream_holds_permit_for_full_drain() {
662        let limiter = Arc::new(Semaphore::new(1));
663        let (permit, registry) = test_permit(&limiter).await;
664        assert_eq!(limiter.available_permits(), 0);
665
666        let inner: BoxStream<'static, Result<u64, anyhow::Error>> =
667            stream::iter(vec![Ok(1u64), Ok(2), Ok(3)]).boxed();
668        let mut wrapped = gate_stream(permit, inner);
669
670        for expected in 1u64..=3 {
671            let item = wrapped.try_next().await.unwrap().unwrap();
672            assert_eq!(item, expected);
673            assert_eq!(
674                limiter.available_permits(),
675                0,
676                "permit should still be held mid-drain"
677            );
678        }
679        assert!(wrapped.try_next().await.unwrap().is_none());
680        // Dropping the stream releases the permit.
681        drop(wrapped);
682        assert_eq!(limiter.available_permits(), 1);
683        assert_eq!(reads_finished(&registry, "completed"), 1);
684        assert_eq!(reads_finished(&registry, "cancelled"), 0);
685    }

687    /// Dropping the wrapped stream early — before the inner stream is
688    /// drained — should release the permit immediately and record the read
689    /// as cancelled.
690    #[tokio::test(start_paused = true)]
691    async fn gate_stream_releases_permit_on_early_drop() {
692        let limiter = Arc::new(Semaphore::new(1));
693        let (permit, registry) = test_permit(&limiter).await;
694        assert_eq!(limiter.available_permits(), 0);
695
696        // Wrap an infinite stream so we can drop in the middle.
697        let counter = Arc::new(AtomicUsize::new(0));
698        let counter_clone = counter.clone();
699        let inner: BoxStream<'static, Result<u64, anyhow::Error>> =
700            stream::repeat_with(move || {
701                let n = counter_clone.fetch_add(1, Ordering::SeqCst);
702                Ok(n as u64)
703            })
704            .boxed();
705        let mut wrapped = gate_stream(permit, inner);
706
707        let _ = wrapped.try_next().await.unwrap().unwrap();
708        let _ = wrapped.try_next().await.unwrap().unwrap();
709        assert_eq!(limiter.available_permits(), 0);
710
711        drop(wrapped);
712        // The drop is synchronous; the permit is released immediately.
713        // (`tokio::sync::Semaphore` doesn't need yield-to-runtime to observe
714        // the release.)
715        tokio::time::sleep(Duration::from_millis(0)).await;
716        assert_eq!(limiter.available_permits(), 1);
717        assert_eq!(reads_finished(&registry, "cancelled"), 1);
718        assert_eq!(reads_finished(&registry, "completed"), 0);
719    }

722    /// still release on the resulting stream-end, and the read is recorded
723    /// as failed rather than cancelled.
724    #[tokio::test(start_paused = true)]
725    async fn gate_stream_propagates_inner_error() {

728
729        let inner: BoxStream<'static, Result<u64, anyhow::Error>> =
