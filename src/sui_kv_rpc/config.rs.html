53/// Per-endpoint tunables for one ledger-history list API. Every field is
54/// optional and falls back to a built-in default; see
55/// [`ResolvedLedgerHistoryMethodConfig`].
56#[derive(Clone, Debug, Default, Deserialize, Serialize, schemars::JsonSchema)]
57#[serde(rename_all = "kebab-case")]
58pub struct LedgerHistoryMethodConfig {
59    /// Wall-clock budget covering computation and response delivery. Defaults
60    /// to `30000`
61    #[serde(skip_serializing_if = "Option::is_none")]
62    pub timeout_ms: Option<u64>,
63
64    /// Page size used when a request omits `limit_items`.
65    #[serde(skip_serializing_if = "Option::is_none")]
66    pub default_limit_items: Option<u32>,
67
68    /// Upper bound a request's `limit_items` is clamped to.
69    #[serde(skip_serializing_if = "Option::is_none")]
70    pub max_limit_items: Option<u32>,
71
72    /// Bounds admitted render tasks and retained rendered responses for this
73    /// endpoint's render-ahead stage. Defaults to `4`; `1` serializes rendering
74    /// with emission.
75    #[serde(skip_serializing_if = "Option::is_none")]
76    pub render_ahead: Option<usize>,
77}

79/// A [`LedgerHistoryMethodConfig`] with all defaults applied.
80#[derive(Clone, Copy, Debug)]
81pub struct ResolvedLedgerHistoryMethodConfig {
82    pub timeout: Duration,
83    pub default_limit_items: u32,
84    pub max_limit_items: u32,
85    pub render_ahead: usize,
86}

88impl LedgerHistoryMethodConfig {
89    fn resolve(
90        this: Option<&LedgerHistoryMethodConfig>,
91        defaults: LedgerHistoryMethodDefaults,
92    ) -> ResolvedLedgerHistoryMethodConfig {
93        ResolvedLedgerHistoryMethodConfig {
94            timeout: Duration::from_millis(
95                this.and_then(|c| c.timeout_ms)
96                    .unwrap_or(DEFAULT_LEDGER_HISTORY_METHOD_TIMEOUT_MS),
97            ),
98            default_limit_items: this
99                .and_then(|c| c.default_limit_items)
100                .unwrap_or(defaults.default_limit_items),
101            max_limit_items: this
102                .and_then(|c| c.max_limit_items)
103                .unwrap_or(defaults.max_limit_items),
104            render_ahead: this
105                .and_then(|c| c.render_ahead)
106                .unwrap_or(defaults.render_ahead),
107        }
108    }
109}

111/// One read pipeline stage, identified by the BigTable table it scans. The same
112/// stage is shared by every list method that reads that table, so its cost is a
113/// property of the table (e.g. object rows are large, `tx_seq_digest` rows tiny)
114/// rather than the calling method.
115#[derive(Clone, Copy, Debug)]
116pub enum PipelineStage {
117    TxSeqDigest,
118    Transactions,
119    Objects,
120    Checkpoints,
121}
122
123/// Per-stage tunables. Every field is optional and falls back to a built-in
124/// default; see [`ResolvedStageConfig`].
125#[derive(Clone, Debug, Default, Deserialize, Serialize, schemars::JsonSchema)]
126#[serde(rename_all = "kebab-case")]
127pub struct StageConfig {
128    /// Items per multi-get batch issued to this stage's table. Defaults to `100`.
129    #[serde(skip_serializing_if = "Option::is_none")]
130    pub chunk_size: Option<usize>,
131
132    /// Max in-flight chunk fetches for this stage. Independent of
133    /// `request_bigtable_concurrency`, which still caps total concurrent reads
134    /// across all stages within a request. Defaults are table-specific; see
135    /// [`StagesConfig::stage`].
136    #[serde(skip_serializing_if = "Option::is_none")]
137    pub concurrency: Option<usize>,
138}
139
140/// A [`StageConfig`] with all defaults applied.
141#[derive(Clone, Copy, Debug)]
142pub struct ResolvedStageConfig {
143    pub chunk_size: usize,
144    pub concurrency: usize,
145}
146
147impl StageConfig {
148    fn resolve(this: Option<&StageConfig>, default_concurrency: usize) -> ResolvedStageConfig {
149        ResolvedStageConfig {
150            chunk_size: this
151                .and_then(|c| c.chunk_size)
152                .unwrap_or(DEFAULT_STAGE_CHUNK_SIZE),
153            concurrency: this
154                .and_then(|c| c.concurrency)
155                .unwrap_or(default_concurrency),
156        }
157    }
158}

160/// Per-table-stage pipeline tunables, shared across the list methods that read
161/// each table.
162#[derive(Clone, Debug, Default, Deserialize, Serialize, schemars::JsonSchema)]
163#[serde(rename_all = "kebab-case")]
164pub struct StagesConfig {
165    /// Reads of the `tx_seq_digest` table (sequence number -> digest/checkpoint).
166    /// Defaults to `10`.
167    #[serde(skip_serializing_if = "Option::is_none")]
168    pub tx_seq_digest: Option<StageConfig>,
169
170    /// Reads of the `transactions` table (digest -> transaction body). Defaults
171    /// to `25`.
172    #[serde(skip_serializing_if = "Option::is_none")]
173    pub transactions: Option<StageConfig>,
174
175    /// Multi-get reads of the `objects` table. Defaults to `50`.
176    #[serde(skip_serializing_if = "Option::is_none")]
177    pub objects: Option<StageConfig>,
178
179    /// Reads of the `checkpoints` table. Defaults to `10`.
180    #[serde(skip_serializing_if = "Option::is_none")]
181    pub checkpoints: Option<StageConfig>,
182}
183
184impl StagesConfig {
185    fn get(&self, stage: PipelineStage) -> Option<&StageConfig> {
186        match stage {
187            PipelineStage::TxSeqDigest => self.tx_seq_digest.as_ref(),
188            PipelineStage::Transactions => self.transactions.as_ref(),
189            PipelineStage::Objects => self.objects.as_ref(),
190            PipelineStage::Checkpoints => self.checkpoints.as_ref(),
191        }
192    }
193
194    /// Resolved tunables for one pipeline stage. Unset stages use chunk size
195    /// `100` and table-specific concurrency: tx-sequence-to-digest `10`,
196    /// transactions `25`, objects `50`, and checkpoints `10`.
197    pub fn stage(&self, stage: PipelineStage) -> ResolvedStageConfig {
198        let default_concurrency = match stage {
199            PipelineStage::TxSeqDigest => DEFAULT_TX_SEQ_DIGEST_STAGE_CONCURRENCY,
200            PipelineStage::Transactions => DEFAULT_TRANSACTIONS_STAGE_CONCURRENCY,
201            PipelineStage::Objects => DEFAULT_OBJECTS_STAGE_CONCURRENCY,
202            PipelineStage::Checkpoints => DEFAULT_CHECKPOINTS_STAGE_CONCURRENCY,
203        };
204        StageConfig::resolve(self.get(stage), default_concurrency)
205    }
206}

208/// Tunables for the ledger-history list APIs. Per-endpoint knobs live in
209/// the three [`LedgerHistoryMethodConfig`] fields; the remaining knobs are
210/// global across all three. Every field is optional and falls back to a built-in
211/// default.
212#[derive(Clone, Debug, Default, Deserialize, Serialize, schemars::JsonSchema)]
213#[serde(rename_all = "kebab-case")]
214pub struct LedgerHistoryConfig {
215    /// Per-endpoint tunables for `list_transactions`.
216    #[serde(skip_serializing_if = "Option::is_none")]
217    pub list_transactions: Option<LedgerHistoryMethodConfig>,
218
219    /// Per-endpoint tunables for `list_events`.
220    #[serde(skip_serializing_if = "Option::is_none")]
221    pub list_events: Option<LedgerHistoryMethodConfig>,
222
223    /// Per-endpoint tunables for `list_checkpoints`.
224    #[serde(skip_serializing_if = "Option::is_none")]
225    pub list_checkpoints: Option<LedgerHistoryMethodConfig>,
226
227    /// Per-request evaluated-bucket budget for filtered tx-bitmap scans, shared
228    /// across all DNF dimensions of one query. Caps how many fetched buckets the
229    /// eval evaluates, NOT how many bucket reads BigTable receives — at
230    /// exhaustion each leaf stream may have fetched one additional bucket that is
231    /// discarded rather than evaluated, so observed reads can exceed this by up
232    /// to `max_bitmap_filter_literals`. Exhausting it ends the query with
233    /// `SCAN_LIMIT` and a resume cursor.
234    ///
235    /// Defaults to `4000` if not specified.
236    #[serde(skip_serializing_if = "Option::is_none")]
237    pub bitmap_bucket_budget_tx: Option<u64>,
238
239    /// Per-request evaluated-bucket budget for filtered event-bitmap scans. Event
240    /// buckets cover far fewer source-domain positions than tx buckets, so this
241    /// is tuned separately even though both default to the same number. Same
242    /// fetched-vs-evaluated semantics as `bitmap_bucket_budget_tx`.
243    ///
244    /// Defaults to `4000` if not specified.
245    #[serde(skip_serializing_if = "Option::is_none")]
246    pub bitmap_bucket_budget_event: Option<u64>,
247
248    /// Dead bucket rows drained from an open leaf scan per lag episode before
249    /// abandoning the stream and reopening it past the gap. A fresh reopen is
250    /// disabled by `0`; provably dead gaps still advance the resume cursor.
251    ///
252    /// Defaults to `50` if not specified.
253    #[serde(skip_serializing_if = "Option::is_none")]
254    pub bitmap_drain_probe_rows: Option<u32>,
255
256    /// Maximum total filter literals (bitmap dimensions) accepted in one filtered
257    /// request, across all DNF terms. Each literal becomes one bitmap leaf, so
258    /// this bounds a single filter's scan fanout. Must not exceed either bitmap
259    /// budget (see [`LedgerHistoryConfig::validate`]).
260    ///
261    /// Defaults to `10` if not specified.
262    #[serde(skip_serializing_if = "Option::is_none")]
263    pub max_bitmap_filter_literals: Option<usize>,
264}
265
266impl LedgerHistoryConfig {
267    pub fn list_transactions(&self) -> ResolvedLedgerHistoryMethodConfig {
268        LedgerHistoryMethodConfig::resolve(
269            self.list_transactions.as_ref(),
270            LIST_TRANSACTIONS_DEFAULTS,
271        )
272    }
273
274    pub fn list_events(&self) -> ResolvedLedgerHistoryMethodConfig {
275        LedgerHistoryMethodConfig::resolve(self.list_events.as_ref(), LIST_EVENTS_DEFAULTS)
276    }
277
278    pub fn list_checkpoints(&self) -> ResolvedLedgerHistoryMethodConfig {
279        LedgerHistoryMethodConfig::resolve(
280            self.list_checkpoints.as_ref(),
281            LIST_CHECKPOINTS_DEFAULTS,
282        )
283    }
284
285    pub fn bitmap_bucket_budget_tx(&self) -> u64 {
286        self.bitmap_bucket_budget_tx
287            .unwrap_or(DEFAULT_BITMAP_BUCKET_BUDGET_TX)
288    }
289
290    pub fn bitmap_bucket_budget_event(&self) -> u64 {
291        self.bitmap_bucket_budget_event
292            .unwrap_or(DEFAULT_BITMAP_BUCKET_BUDGET_EVENT)
293    }
294    pub fn bitmap_skip_policy(&self) -> SkipPolicy {
295        SkipPolicy {
296            drain_probe_rows: NonZeroU32::new(
297                self.bitmap_drain_probe_rows
298                    .unwrap_or(DEFAULT_BITMAP_DRAIN_PROBE_ROWS),
299            ),
300        }
301    }
302
303    pub fn max_bitmap_filter_literals(&self) -> usize {
304        self.max_bitmap_filter_literals
305            .unwrap_or(DEFAULT_MAX_BITMAP_FILTER_LITERALS)
306    }
307
308    /// Reject configurations that cannot make forward progress. Each filter
309    /// literal becomes one bitmap leaf that must fetch at least one bucket to
310    /// emit its first watermark; if a per-request budget is below the literal
311    /// cap a `SCAN_LIMIT` can fire before any merged watermark reaches the wire,
312    /// leaving the client a cursorless `QueryEnd` it cannot resume from. Mirrors
313    /// the fullnode side's `LedgerHistoryConfig::validate`.
314    pub fn validate(&self) -> anyhow::Result<()> {
315        for (name, endpoint) in [
316            ("list_transactions", self.list_transactions()),
317            ("list_events", self.list_events()),
318            ("list_checkpoints", self.list_checkpoints()),
319        ] {
320            anyhow::ensure!(
321                endpoint.render_ahead > 0,
322                "ledger_history.{name}.render_ahead must be greater than zero",
323            );
324        }
325        anyhow::ensure!(
326            self.max_bitmap_filter_literals() > 0,
327            "ledger_history.max_bitmap_filter_literals must be greater than zero",
328        );
329        anyhow::ensure!(
330            self.bitmap_bucket_budget_tx() >= self.max_bitmap_filter_literals() as u64,
331            "ledger_history.bitmap_bucket_budget_tx ({}) must be >= \
332             max_bitmap_filter_literals ({}) so every leaf stream gets at least \
333             one bucket before SCAN_LIMIT",
334            self.bitmap_bucket_budget_tx(),
335            self.max_bitmap_filter_literals(),
336        );
337        anyhow::ensure!(
338            self.bitmap_bucket_budget_event() >= self.max_bitmap_filter_literals() as u64,
339            "ledger_history.bitmap_bucket_budget_event ({}) must be >= \
340             max_bitmap_filter_literals ({}) so every leaf stream gets at least \
341             one bucket before SCAN_LIMIT",
342            self.bitmap_bucket_budget_event(),
343            self.max_bitmap_filter_literals(),
344        );
345        Ok(())
346    }
347}

460        let path = path.as_ref();

462            .with_context(|| format!("failed to read config file {}", path.display()))?;

511    /// TLS identity, when both cert and key are set and non-empty.

550    /// Reject configurations that cannot make forward progress. Validates the
551    /// shared read-pipeline knobs (concurrency and per-stage chunk/fan-out) and
552    /// delegates to [`LedgerHistoryConfig::validate`] for the list-API knobs.
553    pub fn validate(&self) -> anyhow::Result<()> {
554        if let Some(plaintext) = self.plaintext_address() {
555            anyhow::ensure!(
556                plaintext != self.address(),
557                "plaintext_address must differ from address",
558            );
559        }
560        anyhow::ensure!(
561            self.request_bigtable_concurrency() > 0,
562            "request_bigtable_concurrency must be greater than zero",
563        );
564        let stages = self.stages();
565        for stage in [
566            PipelineStage::TxSeqDigest,
567            PipelineStage::Transactions,
568            PipelineStage::Objects,
569            PipelineStage::Checkpoints,
570        ] {
571            let resolved = stages.stage(stage);
572            anyhow::ensure!(
573                resolved.chunk_size > 0,
574                "stages.{stage:?} chunk_size must be greater than zero",
575            );
576            anyhow::ensure!(
577                resolved.concurrency > 0,
578                "stages.{stage:?} concurrency must be greater than zero",
579            );
580        }
581        self.ledger_history().validate()
582    }
