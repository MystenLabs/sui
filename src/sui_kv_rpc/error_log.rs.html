4//! Logs every response that ends in a server-side gRPC error, with its method, code and message.
5//!
6//! The status is read where the client reads it: from the response headers of a trailers-only
7//! response (how a handler's `Err` is sent before any message), and from the trailers at the end of
8//! the body (how a streaming handler reports an error after its first message). Codes the client
9//! caused or chose (`Cancelled`, `NotFound`, `InvalidArgument`, ...) are not logged. Errors
10//! returned after a transport failure while reading the inbound request body (e.g. an HTTP/2
11//! stream reset before tonic finishes decoding the request) are logged at `debug` instead of
12//! `warn` without consuming [`LOG_BUDGET`].
13//!
14//! Lines are rate-limited per process to [`MAX_LINES_PER_SECOND`]. A line dropped by the limit is
15//! counted in the `suppressed` field of the next line that is logged, so counting lines plus
16//! `suppressed` still gives the number of failed requests.
17

19use std::sync::LazyLock;

35/// Error lines logged per second by one process before further failures are only counted.
36const MAX_LINES_PER_SECOND: u64 = 20;
37
38/// Longest message logged; longer messages are cut at a character boundary.
39const MAX_MESSAGE_BYTES: usize = 1024;
40
41const GRPC_STATUS: HeaderName = HeaderName::from_static("grpc-status");
42
43static LOG_BUDGET: LogBudget = LogBudget::new(MAX_LINES_PER_SECOND);
44static PROCESS_START: LazyLock<Instant> = LazyLock::new(Instant::now);
45
46/// Builds one [`ServerErrorLog`] per request, for `sui_http`'s `CallbackLayer`.
47///
48/// Like `RpcMetricsMakeCallbackHandler`, this layer must sit outside any layer that introduces its
49/// own request-body errors (body-size limits, request decompression, body-read timeouts), so a
50/// request-body error unambiguously indicates an inbound transport failure.
51#[derive(Clone, Copy, Default)]
52pub(crate) struct MakeServerErrorLog;
53
54impl MakeCallbackHandler for MakeServerErrorLog {
55    type RequestHandler = ServerErrorLogRequest;
56    type ResponseHandler = ServerErrorLog;
57
58    fn make_handler(&self, request: &request::Parts) -> (ServerErrorLogRequest, ServerErrorLog) {
59        let request_body_failed = Arc::new(AtomicBool::new(false));
60        let req_handler = ServerErrorLogRequest {
61            request_body_failed: request_body_failed.clone(),
62        };
63        let resp_handler = ServerErrorLog {
64            uri: request.uri.clone(),
65            start: Instant::now(),
66            messages_sent: 0,
67            request_body_failed,
68        };
69        (req_handler, resp_handler)
70    }
71}

73/// Records whether polling the inbound request body failed at the transport layer.
74pub(crate) struct ServerErrorLogRequest {
75    request_body_failed: Arc<AtomicBool>,
76}
77
78impl RequestHandler for ServerErrorLogRequest {
79    fn on_body_error<E>(&mut self, _error: &E)
80    where
81        E: std::fmt::Display + 'static,
82    {
83        self.request_body_failed.store(true, Ordering::Release);
84    }
85}

87/// Watches one response and logs it if it ends in a server-side error.
88pub(crate) struct ServerErrorLog {
89    /// The request URI; its path is the gRPC method.
90    uri: http::Uri,
91    start: Instant,
92    /// Response body data frames sent before the response ended.
93    messages_sent: u64,
94    /// Whether polling the inbound request body yielded a transport error.
95    request_body_failed: Arc<AtomicBool>,
96}
97
98impl ServerErrorLog {
99    fn log(&self, ended_in: &'static str, code: Code, message: &str) {
100        let Some(suppressed) = LOG_BUDGET.take(PROCESS_START.elapsed().as_secs()) else {
101            return;
102        };
103        tracing::warn!(
104            method = self.uri.path(),
105            ?code,
106            ended_in,
107            messages_sent = self.messages_sent,
108            elapsed_ms = self.start.elapsed().as_millis() as u64,
109            suppressed,
110            error = &message[..message.floor_char_boundary(MAX_MESSAGE_BYTES)],
111            "request failed with a server error",
112        );
113    }
114
115    fn log_status(&self, ended_in: &'static str, status: &Status) {
116        if self.request_body_failed.load(Ordering::Acquire) {
117            let message = status.message();
118            tracing::debug!(
119                method = self.uri.path(),
120                code = ?status.code(),
121                ended_in,
122                messages_sent = self.messages_sent,
123                elapsed_ms = self.start.elapsed().as_millis() as u64,
124                error = &message[..message.floor_char_boundary(MAX_MESSAGE_BYTES)],
125                "request failed while reading request body",
126            );
127            return;
128        }
129        self.log(ended_in, status.code(), status.message());
130    }
131}
132
133impl ResponseHandler for ServerErrorLog {
134    fn on_response(&mut self, response: &response::Parts) {
135        if let Some(status) = server_error(&response.headers) {
136            self.log_status("headers", &status);
137        }
138    }
139
140    fn on_service_error<E>(&mut self, error: &E)
141    where
142        E: std::fmt::Display + 'static,
143    {
144        self.log("service error", Code::Unknown, &error.to_string());
145    }
146
147    fn on_body_chunk<B>(&mut self, _chunk: &B)
148    where
149        B: bytes::Buf,
150    {
151        self.messages_sent += 1;
152    }
153
154    fn on_end_of_stream(&mut self, trailers: Option<&HeaderMap>) {
155        if let Some(status) = trailers.and_then(server_error) {
156            self.log_status("trailers", &status);
157        }
158    }
159
160    fn on_body_error<E>(&mut self, error: &E)
161    where
162        E: std::fmt::Display + 'static,
163    {
164        self.log("body error", Code::Unknown, &error.to_string());
165    }
166}

168/// The status in `headers` (response headers or trailers), if it is a server-side error.
169fn server_error(headers: &HeaderMap) -> Option<Status> {
170    let code = Code::from_bytes(headers.get(GRPC_STATUS)?.as_bytes());
171    if !is_server_error(code) {
172        return None;
173    }
174    Status::from_header_map(headers)
175}
176
177/// Whether `code` reports a failure of the server rather than of the request or the client.
178fn is_server_error(code: Code) -> bool {
179    matches!(
180        code,
181        Code::Unknown
182            | Code::DeadlineExceeded
183            | Code::ResourceExhausted
184            | Code::Aborted
185            | Code::Internal
186            | Code::Unavailable
187            | Code::DataLoss
188    )
189}

191/// A per-second allowance of log lines, shared by every request in the process.
192struct LogBudget {
193    max_per_second: u64,
194    /// The second (since process start) that `taken` counts lines for.
195    second: AtomicU64,
196    taken: AtomicU64,
197    /// Lines refused since a line was last allowed.
198    suppressed: AtomicU64,
199}
200
201impl LogBudget {
202    const fn new(max_per_second: u64) -> Self {
203        Self {
204            max_per_second,
205            second: AtomicU64::new(0),
206            taken: AtomicU64::new(0),
207            suppressed: AtomicU64::new(0),
208        }
209    }
210
211    /// Takes one line from the allowance for `second`. Returns the number of lines refused since
212    /// the last allowed one, or `None` if this line is refused too.
213    fn take(&self, second: u64) -> Option<u64> {
214        let current = self.second.load(Ordering::Relaxed);
215        if second > current
216            && self
217                .second
218                .compare_exchange(current, second, Ordering::Relaxed, Ordering::Relaxed)
219                .is_ok()
220        {
221            self.taken.store(0, Ordering::Relaxed);
222        }
223        if self.taken.fetch_add(1, Ordering::Relaxed) < self.max_per_second {
224            Some(self.suppressed.swap(0, Ordering::Relaxed))
225        } else {
226            self.suppressed.fetch_add(1, Ordering::Relaxed);
227            None
228        }
229    }
230}

236    fn headers(status: &str, message: Option<&str>) -> HeaderMap {
237        let mut headers = HeaderMap::new();
238        headers.insert(GRPC_STATUS, status.parse().unwrap());
239        if let Some(message) = message {
240            headers.insert("grpc-message", message.parse().unwrap());
241        }
242        headers
243    }

245    #[test]
246    fn server_errors_are_reported_with_their_decoded_message() {
247        let status = server_error(&headers("13", Some("BigTable%20read%20failed"))).unwrap();
248        assert_eq!(status.code(), Code::Internal);
249        assert_eq!(status.message(), "BigTable read failed");
250
251        let status = server_error(&headers("14", None)).unwrap();
252        assert_eq!(status.code(), Code::Unavailable);
253    }
254
255    #[test]
256    fn success_and_client_caused_codes_are_not_reported() {
257        // Ok, Cancelled, InvalidArgument, NotFound, FailedPrecondition, OutOfRange.
258        for code in ["0", "1", "3", "5", "9", "11"] {
259            assert!(
260                server_error(&headers(code, Some("x"))).is_none(),
261                "code {code}"
262            );
263        }
264        assert!(server_error(&HeaderMap::new()).is_none());
265    }

267    #[test]
268    fn budget_refuses_lines_past_the_limit_and_reports_them_on_the_next_second() {
269        let budget = LogBudget::new(2);
270        assert_eq!(budget.take(5), Some(0));
271        assert_eq!(budget.take(5), Some(0));
272        assert_eq!(budget.take(5), None);
273        assert_eq!(budget.take(5), None);
274        // A new second refills the allowance; the first line carries the refused count.
275        assert_eq!(budget.take(6), Some(2));
276        assert_eq!(budget.take(6), Some(0));
277        assert_eq!(budget.take(6), None);
278    }

280    #[derive(Clone, Default)]
281    struct LevelCapture {
282        warns: Arc<AtomicU64>,
283        debugs: Arc<AtomicU64>,
284    }
285
286    impl<S: tracing::Subscriber> tracing_subscriber::Layer<S> for LevelCapture {
287        fn on_event(
288            &self,
289            event: &tracing::Event<'_>,
290            _ctx: tracing_subscriber::layer::Context<'_, S>,
291        ) {
292            if event.metadata().target() != module_path!().trim_end_matches("::tests") {
293                return;
294            }
295            match *event.metadata().level() {
296                tracing::Level::WARN => {
297                    self.warns.fetch_add(1, Ordering::Relaxed);
298                }
299                tracing::Level::DEBUG => {
300                    self.debugs.fetch_add(1, Ordering::Relaxed);
301                }
302                _ => {}
303            }
304        }
305    }

307    #[tokio::test]
308    async fn inbound_request_body_error_is_downgraded_to_debug_not_warn() {
309        use sui_http::middleware::callback::CallbackLayer;
310        use tower::{Layer, Service, ServiceExt};
311        use tracing_subscriber::layer::SubscriberExt;
312
313        struct FailingUnaryHandler;
314
315        impl tonic::server::UnaryService<()> for FailingUnaryHandler {
316            type Response = ();
317            type Future = std::future::Ready<Result<tonic::Response<()>, Status>>;
318
319            fn call(&mut self, _req: tonic::Request<()>) -> Self::Future {
320                std::future::ready(Err(Status::internal("BigTable read failed")))
321            }
322        }
323
324        let capture = LevelCapture::default();
325        let subscriber = tracing_subscriber::registry().with(capture.clone());
326        let _guard = tracing::subscriber::set_default(subscriber);
327
328        let inner = tower::service_fn(|req: http::Request<_>| async move {
329            let mut grpc = tonic::server::Grpc::new(tonic_prost::ProstCodec::<(), ()>::default());
330            Ok::<_, std::convert::Infallible>(grpc.unary(FailingUnaryHandler, req).await)
331        });
332        let mut svc = CallbackLayer::new(MakeServerErrorLog).layer(inner);
333
334        let make_req = |body: axum::body::Body| {
335            http::Request::builder()
336                .uri("/sui.rpc.v2.LedgerService/GetCheckpoint")
337                .header(
338                    http::header::CONTENT_TYPE,
339                    tonic::metadata::GRPC_CONTENT_TYPE,
340                )
341                .body(body)
342                .unwrap()
343        };
344
345        // Inbound request-body failure must log at debug, not warn.
346        let failing_body = axum::body::Body::from_stream(futures::stream::once(async {
347            Err::<bytes::Bytes, _>(Status::internal(
348                "h2 protocol error: error reading a body from connection",
349            ))
350        }));
351        let resp = svc
352            .ready()
353            .await
354            .unwrap()
355            .call(make_req(failing_body))
356            .await
357            .unwrap();
358        assert_eq!(resp.headers().get(GRPC_STATUS).unwrap(), "13");
359        assert_eq!(capture.warns.load(Ordering::Relaxed), 0);
360        assert_eq!(capture.debugs.load(Ordering::Relaxed), 1);
361
362        // A genuine handler Internal error after a clean request body still logs at warn.
363        let resp = svc
364            .ready()
365            .await
366            .unwrap()
367            .call(make_req(axum::body::Body::from(bytes::Bytes::from_static(
368                &[0, 0, 0, 0, 0],
369            ))))
370            .await
371            .unwrap();
372        assert_eq!(resp.headers().get(GRPC_STATUS).unwrap(), "13");
373        assert_eq!(capture.warns.load(Ordering::Relaxed), 1);
374        assert_eq!(capture.debugs.load(Ordering::Relaxed), 1);
375    }
