4//! Sets the checkpoint a replica serves from the `WatermarkService` streams of kvstore indexers,
5//! without reading Bigtable's watermark rows.
6//!
7//! Every kvstore indexer writes the same Bigtable instance, and a durable watermark that any of
8//! them reports for a pipeline means that pipeline's rows up to it are in the cluster that indexer
9//! writes to. So each pipeline counts with the highest watermark any indexer has reported
10//! ([`sui_kvstore::streamed_watermarks`] merges them), and the replica serves up to the lowest of
11//! those across the pipelines it serves, the same bound it takes over Bigtable's watermark rows.
12//! Merged watermarks only move forward, so neither does the served checkpoint.
13//!
14//! Bigtable replicates rows between clusters only later, so this bound holds only if the replica
15//! reads from the cluster the indexers write to. Polling Bigtable's watermark rows needs no such
16//! condition, because it reads them through the replica's own app profile.
17//!
18//! Bigtable's watermark rows also carry a `reader_lo`, and a pipeline whose watermark is below it
19//! counts as missing. Streamed watermarks carry none, so this source cannot detect that case. The
20//! kvstore does not prune, so the case does not arise; `GetServiceInfo` reports 0 as the lowest
21//! available checkpoint for the same reason.
22//!
23//! An indexer's first response holds every pipeline it runs at the higher of its stored watermark
24//! row, read when the indexer started, and its durable progress since. So a replica normally learns
25//! its served checkpoint from the first response of one indexer, even during a chain halt. Until
26//! every served pipeline has a watermark, requests that depend on the served checkpoint are
27//! answered `Unavailable`, and [`crate::Health`] reports the replica not ready.
28

45/// Minimum time between log lines naming the served pipelines that no indexer has streamed a
46/// watermark for.
47const MISSING_PIPELINES_LOG_INTERVAL: Duration = Duration::from_secs(30);
48
49/// Where a replica learns how far the data in Bigtable goes.
50#[derive(Clone, Debug, Default, PartialEq, Eq)]
51pub enum WatermarkSource {
52    /// Poll the pipelines' watermark rows in Bigtable.
53    #[default]
54    Bigtable,
55
56    /// Follow the `WatermarkService` streams of the kvstore indexers at these gRPC URLs. Bigtable's
57    /// watermark rows are never read, even while every stream is down. The replica must read from
58    /// the Bigtable cluster the indexers write to.
59    IndexerStreams(Vec<String>),
60}

62/// Follow the indexers at `indexer_urls` and publish the service info their streams support to
63/// `server`'s cache. Following stops when the returned service shuts down.
64pub(crate) fn follow(server: &KvRpcServer, indexer_urls: &[String]) -> anyhow::Result<Service> {
65    anyhow::ensure!(
66        !indexer_urls.is_empty(),
67        "following indexer watermark streams needs at least one indexer URL"
68    );
69    let (streamed, follower) = streamed_watermarks::follow(indexer_urls)
70        .context("Failed to follow the indexers in indexer-urls")?;
71    info!(
72        ?indexer_urls,
73        "Following kvstore indexer watermark streams; Bigtable watermark rows are not read"
74    );
75
76    let publisher = Service::new().spawn_aborting(publish_service_info(server.clone(), streamed));
77    Ok(follower.merge(publisher))
78}

80/// Publish the service info that `streamed` supports to `server`'s cache whenever the served
81/// checkpoint changes, until the sender behind `streamed` is dropped. The cache is never cleared,
82/// so while every stream is down the replica keeps serving the last checkpoint they supported.
83async fn publish_service_info(
84    server: KvRpcServer,
85    mut streamed: watch::Receiver<StreamedWatermarks>,
86) -> anyhow::Result<()> {
87    let pipelines = &server.service_info_watermark_pipelines;
88    let mut served_checkpoint: Option<u64> = None;
89    let mut missing_logged_at: Option<Instant> = None;
90
91    loop {
92        let (lowest, missing) = {
93            let state = streamed.borrow_and_update();
94            let lowest = lowest_watermark(&state.watermarks, pipelines).copied();
95            // Until every indexer has answered or failed once, a missing pipeline can still arrive
96            // in an indexer's first response.
97            let missing: Vec<&str> = if lowest.is_none() && state.unsettled_instances == 0 {
98                pipelines
99                    .iter()
100                    .copied()
101                    .filter(|pipeline| !state.watermarks.contains_key(*pipeline))
102                    .collect()
103            } else {
104                vec![]
105            };
106            (lowest, missing)
107        };
108
109        match lowest {
110            // Most updates advance a pipeline that does not bound the served checkpoint. Publishing
111            // them would wake every monotonic read waiting on the cache without changing its
112            // answer. A checkpoint's epoch and timestamp are fixed, so the checkpoint alone tells
113            // whether the answer changed.
114            Some(watermark) if served_checkpoint != Some(watermark.checkpoint_hi_inclusive) => {
115                if served_checkpoint.is_none() {
116                    info!(
117                        checkpoint = watermark.checkpoint_hi_inclusive,
118                        "Every served pipeline has a streamed watermark; the served checkpoint is \
119                         available",
120                    );
121                }
122                served_checkpoint = Some(watermark.checkpoint_hi_inclusive);
123                let info = v2::service_info(
124                    server.chain_id,
125                    server.server_version.as_ref(),
126                    watermark.epoch_hi_inclusive,
127                    watermark.checkpoint_hi_inclusive,
128                    watermark.timestamp_ms_hi_inclusive,
129                );
130                server.metrics.observe_served_service_info(&info);
131                server.cache.send_replace(Some(info));
132            }
133            Some(_) => {}
134            None if !missing.is_empty()
135                && missing_logged_at
136                    .is_none_or(|logged| logged.elapsed() >= MISSING_PIPELINES_LOG_INTERVAL) =>
137            {
138                warn!(
139                    ?missing,
140                    "No indexer has streamed a watermark for these served pipelines; requests \
141                     that depend on the served checkpoint are answered Unavailable until one does",
142                );
143                missing_logged_at = Some(Instant::now());
144            }
145            None => {}
146        }
147
148        if streamed.changed().await.is_err() {
149            return Ok(());
150        }
151    }
152}

154/// The watermark with the lowest checkpoint among those of `pipelines`, or `None` while any of
155/// `pipelines` has no watermark in `watermarks`.
156fn lowest_watermark<'w>(
157    watermarks: &'w HashMap<String, CommitterWatermark>,
158    pipelines: &[&str],
159) -> Option<&'w CommitterWatermark> {
160    let mut lowest: Option<&CommitterWatermark> = None;
161    for &pipeline in pipelines {
162        let watermark = watermarks.get(pipeline)?;
163        if lowest
164            .is_none_or(|lowest| watermark.checkpoint_hi_inclusive < lowest.checkpoint_hi_inclusive)
165        {
166            lowest = Some(watermark);
167        }
168    }
169    lowest
170}

200    /// Covers subscribing to an indexer that restarted, which waits out the follower's 2 s
201    /// resubscribe interval.

204    /// Long enough for a 10 ms Bigtable polling loop, if one ran, to read the watermark rows many
205    /// times, and for a streamed update to reach the replica.
206    const QUIET_PERIOD: Duration = Duration::from_millis(300);

208    fn watermark(checkpoint: u64) -> CommitterWatermark {
209        CommitterWatermark {
210            epoch_hi_inclusive: checkpoint / 10,
211            checkpoint_hi_inclusive: checkpoint,
212            tx_hi: 0,
213            timestamp_ms_hi_inclusive: checkpoint * 1_000,
214        }
215    }

217    /// A watermark for every pipeline a replica serves, at the checkpoint `checkpoint` picks for
218    /// it.
219    fn served_watermarks(checkpoint: impl Fn(&str) -> u64) -> HashMap<String, CommitterWatermark> {
220        service_info_watermark_pipelines()
221            .iter()
222            .map(|&pipeline| (pipeline.to_string(), watermark(checkpoint(pipeline))))
223            .collect()
224    }

234    /// A kvstore indexer's `WatermarkService`, streaming the durable progress in `progress`.
235    struct Indexer {
236        url: String,
237        progress: watch::Sender<HashMap<String, CommitterWatermark>>,
238        server: Service,
239    }
240
241    impl Indexer {
242        async fn start(address: SocketAddr, progress: HashMap<String, CommitterWatermark>) -> Self {
243            let progress = watch::Sender::new(progress);
244            let rpc = RpcService::new(address);
245            let (watermarks, relay) =
246                WatermarkService::new(&HashMap::new(), progress.subscribe(), rpc.shutdown_token());
247            let server = rpc
248                .add_service(WatermarkServiceServer::new(watermarks))
249                .run(std::future::ready(()))
250                .await
251                .unwrap()
252                .attach(relay);
253            Self {
254                url: format!("http://{address}"),
255                progress,
256                server,
257            }
258        }
259
260        fn record(&self, progress: HashMap<String, CommitterWatermark>) {
261            self.progress
262                .send_modify(|current| current.extend(progress));
263        }
264    }

318    async fn wait_for_served_checkpoint(
319        server: &KvRpcServer,
320        checkpoint: u64,
321    ) -> GetServiceInfoResponse {
322        let mut cache = server.cache.subscribe();
323        let info = timeout(
324            WAIT,
325            cache.wait_for(|info| {
326                info.as_ref().and_then(|info| info.checkpoint_height) == Some(checkpoint)
327            }),
328        )
329        .await
330        .unwrap_or_else(|_| panic!("served checkpoint never reached {checkpoint}"))
331        .expect("service info sender dropped");
332        info.clone().unwrap()
333    }
334
335    fn cached_checkpoint(server: &KvRpcServer) -> Option<u64> {
336        server
337            .cache
338            .borrow()
339            .as_ref()
340            .and_then(|info| info.checkpoint_height)
341    }

354    #[test]
355    fn lowest_watermark_needs_every_pipeline_and_ignores_others() {
356        let mut watermarks = HashMap::from([
357            ("a".to_string(), watermark(10)),
358            ("b".to_string(), watermark(7)),
359            ("unserved".to_string(), watermark(3)),
360        ]);
361        assert_eq!(lowest_watermark(&watermarks, &["a", "b", "c"]), None);
362
363        watermarks.insert("c".to_string(), watermark(9));
364        assert_eq!(
365            lowest_watermark(&watermarks, &["a", "b", "c"]),
366            Some(&watermark(7))
367        );
368    }

370    #[tokio::test]
371    async fn serves_the_lowest_of_each_pipelines_highest_watermark_across_indexers() {
372        let mock = MockBigtableServer::new();
373        let (bigtable, _mock_server) = mock.start().await.expect("start mock Bigtable");
374
375        // Each indexer leads on different pipelines. Per pipeline, the higher watermark counts:
376        // objects at 30 and every other pipeline at 28, so the replica serves 28. The lower of the
377        // two indexers' own lowest watermarks would be 10, and the higher 12.
378        let a = Indexer::start(
379            local_address(),
380            served_watermarks(|pipeline| if pipeline == OBJECTS_PIPELINE { 30 } else { 10 }),
381        )
382        .await;
383        let b = Indexer::start(
384            local_address(),
385            served_watermarks(|pipeline| if pipeline == OBJECTS_PIPELINE { 12 } else { 28 }),
386        )
387        .await;
388        let (server, _replica) = start_replica(bigtable, vec![a.url.clone(), b.url.clone()]).await;
389
390        let info = wait_for_served_checkpoint(&server, 28).await;
391        assert_eq!(info.epoch, Some(2));
392        assert_eq!(info.timestamp, Some(timestamp_ms_to_proto(28_000)));
393        assert_eq!(info.lowest_available_checkpoint, Some(0));
394        assert_eq!(server.metrics.served_checkpoint.get(), 28);
395        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 28_000);
396
397        // Indexer `a` passes `b` on every pipeline but objects, so objects bounds the served
398        // checkpoint, at `a`'s 30 rather than `b`'s 12.
399        a.record(served_watermarks(|pipeline| {
400            if pipeline == OBJECTS_PIPELINE { 30 } else { 35 }
401        }));
402        let info = wait_for_served_checkpoint(&server, 30).await;
403        assert_eq!(info.epoch, Some(3));
404        assert_eq!(info.timestamp, Some(timestamp_ms_to_proto(30_000)));
405        assert_eq!(server.metrics.served_checkpoint.get(), 30);
406        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 30_000);
407
408        assert_eq!(watermark_reads(&mock).await, 0);
409    }

411    #[tokio::test]
412    async fn never_falls_back_to_bigtable_watermark_rows() {
413        let mock = MockBigtableServer::new();
414        let (bigtable, _mock_server) = mock.start().await.expect("start mock Bigtable");
415        // Bigtable's watermark rows alone would let the replica serve checkpoint 50.
416        write_bigtable_watermarks(bigtable, 50).await;
417        mock.clear_read_rows_calls().await;
418
419        // The only indexer has not advanced the epoch end pipeline since it started.
420        let address = local_address();
421        let mut progress = served_watermarks(|_| 20);
422        progress.remove(EPOCH_END_PIPELINE);
423        let indexer = Indexer::start(address, progress).await;
424        let (server, _replica) = start_replica(bigtable, vec![indexer.url.clone()]).await;
425
426        sleep(QUIET_PERIOD).await;
427        assert_eq!(cached_checkpoint(&server), None);
428        let error = tonic::Status::from(server.served_height().unwrap_err());
429        assert_eq!(error.code(), tonic::Code::Unavailable);
430        let error = get_service_info(&server).await.unwrap_err();
431        assert_eq!(error.code(), tonic::Code::Unavailable);
432        assert_eq!(server.metrics.served_checkpoint.get(), 0);
433        assert!(!server.health().is_ready());
434
435        indexer.record(served_watermarks(|_| 20));
436        wait_for_served_checkpoint(&server, 20).await;
437        assert!(server.health().is_ready());
438
439        // With every stream down, the replica keeps serving the last streamed checkpoint.
440        indexer.server.shutdown().await.unwrap();
441        sleep(QUIET_PERIOD).await;
442        let info = get_service_info(&server).await.unwrap();
443        assert_eq!(info.checkpoint_height, Some(20));
444        assert_eq!(server.metrics.served_checkpoint.get(), 20);
445        assert_eq!(watermark_reads(&mock).await, 0);
446        assert!(server.health().is_ready());
447
448        // Once the indexer is back, its stream advances the served checkpoint again.
449        let _indexer = Indexer::start(address, served_watermarks(|_| 30)).await;
450        wait_for_served_checkpoint(&server, 30).await;
451        assert_eq!(watermark_reads(&mock).await, 0);
452    }
