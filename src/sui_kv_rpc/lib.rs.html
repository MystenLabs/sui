188    /// Record the checkpoint `info` (a `GetServiceInfo` answer) says this replica serves.
189    fn observe_served_service_info(&self, info: &GetServiceInfoResponse) {
190        if let Some(checkpoint) = info.checkpoint_height {
191            self.served_checkpoint.set(checkpoint as i64);
192        }
193        if let Some(timestamp) = &info.timestamp {
194            self.served_checkpoint_timestamp_ms
195                .set(timestamp.seconds * 1_000 + i64::from(timestamp.nanos) / 1_000_000);
196        }
197    }

301/// Answers whether a replica is ready for traffic, for its `/health` endpoint.
302#[derive(Clone)]
303pub struct Health {
304    served: watch::Receiver<Option<Arc<GetServiceInfoResponse>>>,
305}
306
307impl Health {
308    /// Whether the replica is ready: once it knows the checkpoint it serves, from its first
309    /// successful watermark refresh or, following indexer watermark streams, once every served
310    /// pipeline has a streamed watermark. It stays ready after that, because its served checkpoint
311    /// is never cleared.
312    pub fn is_ready(&self) -> bool {
313        self.served.borrow().is_some()
314    }

344/// Build and start one gRPC listener serving `ledger`'s `LedgerService` and `MovePackageService`,
345/// wired with the given (shared, already-constructed) metrics/logging layers and optional reflection.
346/// `builder` carries whatever TLS config the caller wants for this listener (or none, for a plaintext
347/// listener).
348///
349/// Used once per listener -- the primary listener, and the optional second plaintext one -- so
350/// that expensive shared pieces (metrics, allowlist, request-log layer) are constructed once by
351/// the caller and passed in here, rather than rebuilt (and, for `RpcMetrics`, re-registered against
352/// the same `Registry`, which panics) per listener.
353fn spawn_listener(
354    listen_address: SocketAddr,
355    builder: Server,
356    ledger: KvRpcServer,
357    rpc_metrics: Arc<sui_rpc_api::RpcMetrics>,
358    grpc_method_allowlist: sui_rpc_api::GrpcMethodAllowlist,
359    request_log_layer: mysten_network::request_log::GrpcRequestLogLayer,
360    enable_reflection: bool,
361    file_descriptor_sets: &[&'static [u8]],
362) -> anyhow::Result<sui_futures::service::Service> {
363    use sui_http::middleware::callback::CallbackLayer;
364    use sui_rpc_api::RpcMetricsMakeCallbackHandler;
365
366    let mut router = builder
367        .layer(CallbackLayer::new(
368            RpcMetricsMakeCallbackHandler::with_grpc_method_allowlist(
369                rpc_metrics,
370                grpc_method_allowlist,
371            ),
372        ))
373        // Logs each response that ends in a server-side error; outside the admission layer so its
374        // rejections are logged too.
375        .layer(CallbackLayer::new(error_log::MakeServerErrorLog))
376        .layer(request_log_layer)
377        // Innermost, so a request held or rejected here is still counted by the metrics layer and
378        // recorded by the request log above it.
379        .layer(AdmissionLayer::new(
380            ledger.cache.subscribe(),
381            ledger.monotonic_read_wait_timeout,
382        ))
383        .add_service(move_package_service_with_response_compression(
384            ledger.clone(),
385        ))
386        .add_service(ledger_service_with_response_compression(ledger));
387
388    if enable_reflection {
389        let mut reflection_v1_builder = tonic_reflection::server::Builder::configure();
390        let mut reflection_v1alpha_builder = tonic_reflection::server::Builder::configure();
391        for &fds in file_descriptor_sets {
392            reflection_v1_builder = reflection_v1_builder.register_encoded_file_descriptor_set(fds);
393            reflection_v1alpha_builder =
394                reflection_v1alpha_builder.register_encoded_file_descriptor_set(fds);
395        }
396        router = router
397            .add_service(reflection_v1_builder.build_v1()?)
398            .add_service(reflection_v1alpha_builder.build_v1alpha()?);
399    }
400
401    // Bind here rather than inside the spawned task. `serve_with_shutdown` binds
402    // within the future it returns, so the caller would get a `Service` back while
403    // the address still refuses connections, and anything that dials immediately
404    // races the bind. Binding first also turns an unusable address into an error
405    // from `start_service` instead of a task that dies after start-up.
406    //
407    // `serve_with_incoming_shutdown` discards the builder's TCP configuration, so
408    // restate the `TCP_NODELAY` that `serve_with_shutdown` would have applied from
409    // its default. Keepalive stays off, which is also tonic's default.
410    let incoming = TcpIncoming::bind(listen_address)
411        .map_err(|e| anyhow::anyhow!("Failed to bind kv-rpc listener on {listen_address}: {e}"))?
412        .with_nodelay(Some(true));
413
414    let (tx, rx) = tokio::sync::oneshot::channel::<()>();
415    let server_future = router.serve_with_incoming_shutdown(incoming, async {
416        let _ = rx.await;
417    });
418
419    Ok(sui_futures::service::Service::new()
420        .spawn(async move {
421            server_future.await?;
422            Ok(())
423        })
424        .with_shutdown_signal(async move {
425            let _ = tx.send(());
426        }))
427}

514    fn init(
515        client: BigTableClient,
516        chain_id: ChainIdentifier,
517        server_version: Option<ServerVersion>,
518        metrics: Arc<KvRpcMetrics>,
519        ledger_history: LedgerHistoryConfig,
520        request_bigtable_concurrency: usize,
521        monotonic_read_wait_timeout: Duration,
522        stages: StagesConfig,
523        watermark_source: WatermarkSource,
524    ) -> anyhow::Result<Self> {
525        ledger_history.validate()?;
526
527        let cache = Arc::new(watch::Sender::new(None));
528
529        let package_store: Arc<dyn PackageStore> = Arc::new(PackageStoreWithLruCache::new(
530            BigTablePackageStore::new(client.clone()),
531        ));
532        let package_resolver = Arc::new(Resolver::new(package_store));
533
534        let server = Self {
535            chain_id,
536            client,
537            server_version,
538            service_info_watermark_pipelines: service_info_watermark_pipelines(
539                ledger_history.bitmap_schema(),
540            ),
541            cache,
542            watermark_source,
543            package_resolver,
544            metrics,
545            ledger_history,
546            request_bigtable_concurrency,
547            monotonic_read_wait_timeout,
548            stages,
549        };
550
551        if server.watermark_source == WatermarkSource::Bigtable {
552            let server_clone = server.clone();
553            tokio::spawn(async move {
554                loop {
555                    match v2::get_service_info(
556                        server_clone.client.clone(),
557                        server_clone.chain_id,
558                        server_clone.server_version.clone(),
559                        &server_clone.service_info_watermark_pipelines,
560                    )
561                    .await
562                    {
563                        Ok(info) => {
564                            server_clone.metrics.observe_served_service_info(&info);
565                            server_clone.cache.send_replace(Some(Arc::new(info)));
566                        }
567                        Err(e) => error!("Failed to update service info cache: {:?}", e),
568                    }
569                    sleep(Duration::from_millis(10)).await;
570                }
571            });
572        }
573
574        Ok(server)
575    }

584    /// Start this server as a tonic gRPC service on the given address, and start following the
585    /// indexer watermark streams if `watermark_source` names them.
586    /// Returns a `Service` handle for lifecycle management.
587    pub async fn start_service(
588        self,
589        listen_address: SocketAddr,
590        config: ServerConfig,
591    ) -> anyhow::Result<sui_futures::service::Service> {
592        use sui_rpc_api::{RpcMetrics, grpc_method_paths_from_file_descriptor_sets};
593
594        // Started before the listeners, so that an invalid indexer URL fails startup before any
595        // address is bound.
596        let indexer_streams = match &self.watermark_source {
597            WatermarkSource::Bigtable => None,
598            WatermarkSource::IndexerStreams(indexer_urls) => {
599                Some(indexer_streams::follow(&self, indexer_urls)?)
600            }
601        };
602
603        // Single source of truth for every encoded FileDescriptorSet that
604        // backs a gRPC service mounted below. Consumed by both the
605        // reflection services and the metrics allowlist so they cannot drift
606        // out of sync, and shared by the optional second (plaintext)
607        // listener below.
608        let file_descriptor_sets: Vec<&'static [u8]> = vec![
609            sui_rpc_api::proto::google::protobuf::FILE_DESCRIPTOR_SET,
610            sui_rpc_api::proto::google::rpc::FILE_DESCRIPTOR_SET,
611            sui_rpc::proto::sui::rpc::v2::FILE_DESCRIPTOR_SET,
612        ];
613
614        let registry = config.metrics_registry.unwrap_or_default();
615        let grpc_method_allowlist = Arc::new(grpc_method_paths_from_file_descriptor_sets(
616            &file_descriptor_sets,
617        )?);
618        // Constructed once and shared (via `Arc`/`Clone`) across both
619        // listeners below: `RpcMetrics::new` registers Prometheus collectors
620        // against `registry`, and doing that twice would panic on duplicate
621        // registration.
622        let rpc_metrics = Arc::new(RpcMetrics::new(&registry));
623        let request_log_layer =
624            mysten_network::request_log::GrpcRequestLogLayer::from_encoded_file_descriptor_sets(
625                file_descriptor_sets.iter().copied(),
626            )?;
627
628        let mut builder = Server::builder();
629        if let Some(identity) = config.tls_identity {
630            builder = builder.tls_config(ServerTlsConfig::new().identity(identity))?;
631        }
632
633        // Cloned up-front (before `self` is consumed below) only when the
634        // second listener is actually being started.
635        let ledger_for_plaintext = config.plaintext_address.is_some().then(|| self.clone());
636
637        let mut service = spawn_listener(
638            listen_address,
639            builder,
640            self,
641            rpc_metrics.clone(),
642            grpc_method_allowlist.clone(),
643            request_log_layer.clone(),
644            config.enable_reflection,
645            &file_descriptor_sets,
646        )?;
647
648        // Second, unencrypted listener for trusted internal callers (e.g.
649        // other in-cluster services) that should not need to negotiate TLS
650        // to reach this server. Serves the same `LedgerService` and `MovePackageService`.
651        if let Some(plaintext_address) = config.plaintext_address {
652            let ledger =
653                ledger_for_plaintext.expect("cloned above whenever plaintext_address is set");
654            let plaintext_service = spawn_listener(
655                plaintext_address,
656                Server::builder(),
657                ledger,
658                rpc_metrics,
659                grpc_method_allowlist,
660                request_log_layer,
661                config.enable_reflection,
662                &file_descriptor_sets,
663            )?;
664            service = service.merge(plaintext_service);
665        }
666
667        if let Some(indexer_streams) = indexer_streams {
668            service = service.attach(indexer_streams);
669        }
670
671        Ok(service)
672    }
673}

675/// Pipelines whose watermarks bound the checkpoint height reported by `GetServiceInfo`, so an
676/// instance does not advertise a height it cannot serve.
677pub fn service_info_watermark_pipelines(bitmap_schema: BitmapSchema) -> Vec<&'static str> {
678    let [transaction_bitmap, event_bitmap] = sui_kvstore::bitmap_index_pipelines(bitmap_schema);
679    vec![
680        // Point lookups.
681        CHECKPOINTS_PIPELINE,
682        CHECKPOINTS_BY_DIGEST_PIPELINE,
683        TRANSACTIONS_PIPELINE,
684        OBJECTS_PIPELINE,
685        EPOCH_START_PIPELINE,
686        EPOCH_END_PIPELINE,
687        PACKAGES_PIPELINE,
688        PACKAGES_BY_ID_PIPELINE,
689        // List APIs, which read the bitmap tables of `bitmap_schema`.
690        TX_SEQ_DIGEST_PIPELINE,
691        transaction_bitmap,
692        event_bitmap,
693    ]
694}

819    #[tokio::test]
820    async fn served_checkpoint_gauges_follow_refresh_and_survive_failed_refresh() {
821        let mock = MockBigtableServer::new();
822        let (addr, _mock_server) = mock.start().await.expect("start mock BigTable");
823        let mut client = BigTableClient::new_local(addr.to_string(), "test".to_string())
824            .await
825            .expect("connect to mock BigTable");
826
827        // The objects pipeline lags, so it bounds the served checkpoint, and the timestamp must
828        // come from its watermark rather than from a pipeline that is further ahead.
829        seed_watermarks(&mut client, |pipeline| {
830            if pipeline == OBJECTS_PIPELINE { 20 } else { 25 }
831        })
832        .await;
833
834        let server = start_v2_server(addr).await;
835        timeout(WAIT, server.cache.subscribe().wait_for(Option::is_some))
836            .await
837            .expect("service info never refreshed")
838            .expect("service info sender dropped");
839        assert_eq!(server.metrics.served_checkpoint.get(), 20);
840        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 20_000);
841
842        // Hide the objects watermark (checkpoint_hi_inclusive < reader_lo), so every refresh
843        // fails. Once a second watermark read arrives, the refresh behind the first has finished.
844        let objects_key = Bytes::from(watermarks::encode_key(OBJECTS_PIPELINE));
845        mock.insert_row(
846            watermarks::NAME,
847            objects_key.clone(),
848            [(watermarks::col::READER_LO, u64_be(21))],
849        )
850        .await;
851        mock.clear_read_rows_calls().await;
852        timeout(WAIT, async {
853            while watermark_reads(&mock).await < 2 {
854                sleep(Duration::from_millis(5)).await;
855            }
856        })
857        .await
858        .expect("service info refresh stopped polling");
859        assert_eq!(server.metrics.served_checkpoint.get(), 20);
860        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 20_000);
861        assert!(server.health().is_ready());
862
863        // Once the objects pipeline catches up past the others, they bound the served checkpoint.
864        mock.insert_row(
865            watermarks::NAME,
866            objects_key,
867            [
868                (watermarks::col::CHECKPOINT_HI, u64_be(30)),
869                (watermarks::col::TIMESTAMP_MS_HI, u64_be(30_000)),
870            ],
871        )
872        .await;
873        wait_for_served_checkpoint(&server, 25).await;
874        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 25_000);
875    }
