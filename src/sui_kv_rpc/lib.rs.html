191    /// Record the checkpoint `info` (a `GetServiceInfo` answer) says this replica serves.
192    fn observe_served_service_info(&self, info: &GetServiceInfoResponse) {
193        if let Some(checkpoint) = info.checkpoint_height {
194            self.served_checkpoint.set(checkpoint as i64);
195        }
196        if let Some(timestamp) = &info.timestamp {
197            self.served_checkpoint_timestamp_ms
198                .set(timestamp.seconds * 1_000 + i64::from(timestamp.nanos) / 1_000_000);
199        }
200    }

311/// Answers whether a replica is ready for traffic, for its `/health` endpoint.
312#[derive(Clone)]
313pub struct Health {
314    served: watch::Receiver<Option<Arc<GetServiceInfoResponse>>>,
315}
316
317impl Health {
318    /// Whether the replica is ready: once it knows the checkpoint it serves, from its first
319    /// successful watermark refresh or, following indexer watermark streams, once every served
320    /// pipeline has a streamed watermark. It stays ready after that, because its served checkpoint
321    /// is never cleared.
322    pub fn is_ready(&self) -> bool {
323        self.served.borrow().is_some()
324    }

354/// Build and start one gRPC listener serving `ledger`'s `LedgerService` and `MovePackageService`,
355/// wired with the given (shared, already-constructed) metrics/logging layers and optional reflection.
356/// `builder` carries whatever TLS config the caller wants for this listener (or none, for a plaintext
357/// listener).
358///
359/// Used once per listener -- the primary listener, and the optional second plaintext one -- so
360/// that expensive shared pieces (metrics, allowlist, request-log layer) are constructed once by
361/// the caller and passed in here, rather than rebuilt (and, for `RpcMetrics`, re-registered against
362/// the same `Registry`, which panics) per listener.
363fn spawn_listener(
364    listen_address: SocketAddr,
365    builder: Server,
366    ledger: KvRpcServer,
367    rpc_metrics: Arc<sui_rpc_api::RpcMetrics>,
368    grpc_method_allowlist: sui_rpc_api::GrpcMethodAllowlist,
369    request_log_layer: mysten_network::request_log::GrpcRequestLogLayer,
370    enable_reflection: bool,
371    file_descriptor_sets: &[&'static [u8]],
372) -> anyhow::Result<sui_futures::service::Service> {
373    use sui_http::middleware::callback::CallbackLayer;
374    use sui_rpc_api::RpcMetricsMakeCallbackHandler;
375
376    let mut router = builder
377        .layer(CallbackLayer::new(
378            RpcMetricsMakeCallbackHandler::with_grpc_method_allowlist(
379                rpc_metrics,
380                grpc_method_allowlist,
381            ),
382        ))
383        // Logs each response that ends in a server-side error; outside the admission layer so its
384        // rejections are logged too.
385        .layer(CallbackLayer::new(error_log::MakeServerErrorLog))
386        .layer(request_log_layer)
387        // Innermost, so a request held or rejected here is still counted by the metrics layer and
388        // recorded by the request log above it.
389        .layer(AdmissionLayer::new(
390            ledger.cache.subscribe(),
391            ledger.monotonic_read_wait_timeout,
392        ))
393        .add_service(move_package_service_with_response_compression(
394            ledger.clone(),
395        ))
396        .add_service(ledger_service_with_response_compression(ledger));
397
398    if enable_reflection {
399        let mut reflection_v1_builder = tonic_reflection::server::Builder::configure();
400        let mut reflection_v1alpha_builder = tonic_reflection::server::Builder::configure();
401        for &fds in file_descriptor_sets {
402            reflection_v1_builder = reflection_v1_builder.register_encoded_file_descriptor_set(fds);
403            reflection_v1alpha_builder =
404                reflection_v1alpha_builder.register_encoded_file_descriptor_set(fds);
405        }
406        router = router
407            .add_service(reflection_v1_builder.build_v1()?)
408            .add_service(reflection_v1alpha_builder.build_v1alpha()?);
409    }
410
411    // Bind here rather than inside the spawned task. `serve_with_shutdown` binds
412    // within the future it returns, so the caller would get a `Service` back while
413    // the address still refuses connections, and anything that dials immediately
414    // races the bind. Binding first also turns an unusable address into an error
415    // from `start_service` instead of a task that dies after start-up.
416    //
417    // `serve_with_incoming_shutdown` discards the builder's TCP configuration, so
418    // restate the `TCP_NODELAY` that `serve_with_shutdown` would have applied from
419    // its default. Keepalive stays off, which is also tonic's default.
420    let incoming = TcpIncoming::bind(listen_address)
421        .map_err(|e| anyhow::anyhow!("Failed to bind kv-rpc listener on {listen_address}: {e}"))?
422        .with_nodelay(Some(true));
423
424    let (tx, rx) = tokio::sync::oneshot::channel::<()>();
425    let server_future = router.serve_with_incoming_shutdown(incoming, async {
426        let _ = rx.await;
427    });
428
429    Ok(sui_futures::service::Service::new()
430        .spawn(async move {
431            server_future.await?;
432            Ok(())
433        })
434        .with_shutdown_signal(async move {
435            let _ = tx.send(());
436        }))
437}

528    fn init(
529        client: BigTableClient,
530        chain_id: ChainIdentifier,
531        server_version: Option<ServerVersion>,
532        alpha_features: AlphaFeatures,
533        metrics: Arc<KvRpcMetrics>,
534        ledger_history: LedgerHistoryConfig,
535        request_bigtable_concurrency: usize,
536        monotonic_read_wait_timeout: Duration,
537        stages: StagesConfig,
538        watermark_source: WatermarkSource,
539    ) -> anyhow::Result<Self> {
540        ledger_history.validate()?;
541
542        let cache = Arc::new(watch::Sender::new(None));
543
544        let package_store: Arc<dyn PackageStore> = Arc::new(PackageStoreWithLruCache::new(
545            BigTablePackageStore::new(client.clone()),
546        ));
547        let package_resolver = Arc::new(Resolver::new(package_store));
548
549        let server = Self {
550            chain_id,
551            client,
552            server_version,
553            service_info_watermark_pipelines: service_info_watermark_pipelines(
554                ledger_history.bitmap_schema(),
555                alpha_features,
556            ),
557            cache,
558            watermark_source,
559            package_resolver,
560            metrics,
561            ledger_history,
562            request_bigtable_concurrency,
563            monotonic_read_wait_timeout,
564            stages,
565            alpha_features,
566        };
567
568        if server.watermark_source == WatermarkSource::Bigtable {
569            let server_clone = server.clone();
570            tokio::spawn(async move {
571                loop {
572                    match v2::get_service_info(
573                        server_clone.client.clone(),
574                        server_clone.chain_id,
575                        server_clone.server_version.clone(),
576                        &server_clone.service_info_watermark_pipelines,
577                    )
578                    .await
579                    {
580                        Ok(info) => {
581                            server_clone.metrics.observe_served_service_info(&info);
582                            server_clone.cache.send_replace(Some(Arc::new(info)));
583                        }
584                        Err(e) => error!("Failed to update service info cache: {:?}", e),
585                    }
586                    sleep(Duration::from_millis(10)).await;
587                }
588            });
589        }
590
591        Ok(server)
592    }

601    /// Start this server as a tonic gRPC service on the given address, and start following the
602    /// indexer watermark streams if `watermark_source` names them.
603    /// Returns a `Service` handle for lifecycle management.
604    pub async fn start_service(
605        self,
606        listen_address: SocketAddr,
607        config: ServerConfig,
608    ) -> anyhow::Result<sui_futures::service::Service> {
609        use sui_rpc_api::{RpcMetrics, grpc_method_paths_from_file_descriptor_sets};
610
611        // Started before the listeners, so that an invalid indexer URL fails startup before any
612        // address is bound.
613        let indexer_streams = match &self.watermark_source {
614            WatermarkSource::Bigtable => None,
615            WatermarkSource::IndexerStreams(indexer_urls) => {
616                Some(indexer_streams::follow(&self, indexer_urls)?)
617            }
618        };
619
620        // Single source of truth for every encoded FileDescriptorSet that
621        // backs a gRPC service mounted below. Consumed by both the
622        // reflection services and the metrics allowlist so they cannot drift
623        // out of sync, and shared by the optional second (plaintext)
624        // listener below.
625        let file_descriptor_sets: Vec<&'static [u8]> = vec![
626            sui_rpc_api::proto::google::protobuf::FILE_DESCRIPTOR_SET,
627            sui_rpc_api::proto::google::rpc::FILE_DESCRIPTOR_SET,
628            sui_rpc::proto::sui::rpc::v2::FILE_DESCRIPTOR_SET,
629        ];
630
631        let registry = config.metrics_registry.unwrap_or_default();
632        let grpc_method_allowlist = Arc::new(grpc_method_paths_from_file_descriptor_sets(
633            &file_descriptor_sets,
634        )?);
635        // Constructed once and shared (via `Arc`/`Clone`) across both
636        // listeners below: `RpcMetrics::new` registers Prometheus collectors
637        // against `registry`, and doing that twice would panic on duplicate
638        // registration.
639        let rpc_metrics = Arc::new(RpcMetrics::new(&registry));
640        let request_log_layer =
641            mysten_network::request_log::GrpcRequestLogLayer::from_encoded_file_descriptor_sets(
642                file_descriptor_sets.iter().copied(),
643            )?;
644
645        let mut builder = Server::builder();
646        if let Some(identity) = config.tls_identity {
647            builder = builder.tls_config(ServerTlsConfig::new().identity(identity))?;
648        }
649
650        // Cloned up-front (before `self` is consumed below) only when the
651        // second listener is actually being started.
652        let ledger_for_plaintext = config.plaintext_address.is_some().then(|| self.clone());
653
654        let mut service = spawn_listener(
655            listen_address,
656            builder,
657            self,
658            rpc_metrics.clone(),
659            grpc_method_allowlist.clone(),
660            request_log_layer.clone(),
661            config.enable_reflection,
662            &file_descriptor_sets,
663        )?;
664
665        // Second, unencrypted listener for trusted internal callers (e.g.
666        // other in-cluster services) that should not need to negotiate TLS
667        // to reach this server. Serves the same `LedgerService` and `MovePackageService`.
668        if let Some(plaintext_address) = config.plaintext_address {
669            let ledger =
670                ledger_for_plaintext.expect("cloned above whenever plaintext_address is set");
671            let plaintext_service = spawn_listener(
672                plaintext_address,
673                Server::builder(),
674                ledger,
675                rpc_metrics,
676                grpc_method_allowlist,
677                request_log_layer,
678                config.enable_reflection,
679                &file_descriptor_sets,
680            )?;
681            service = service.merge(plaintext_service);
682        }
683
684        if let Some(indexer_streams) = indexer_streams {
685            service = service.attach(indexer_streams);
686        }
687
688        Ok(service)
689    }
690}

692/// Pipelines whose watermarks bound the checkpoint height reported by `GetServiceInfo`, so an
693/// instance does not advertise a height it cannot serve. They include the packages pipelines, so
694/// a package lookup bounded by the served height sees every package published by then, and the
695/// pipelines of the enabled `alpha_features`.
696pub fn service_info_watermark_pipelines(
697    bitmap_schema: BitmapSchema,
698    alpha_features: AlphaFeatures,
699) -> Vec<&'static str> {
700    let [transaction_bitmap, event_bitmap] = sui_kvstore::bitmap_index_pipelines(bitmap_schema);
701    let mut pipelines = vec![
702        // Point lookups.
703        CHECKPOINTS_PIPELINE,
704        CHECKPOINTS_BY_DIGEST_PIPELINE,
705        TRANSACTIONS_PIPELINE,
706        OBJECTS_PIPELINE,
707        EPOCH_START_PIPELINE,
708        EPOCH_END_PIPELINE,
709        PACKAGES_PIPELINE,
710        PACKAGES_BY_ID_PIPELINE,
711        PACKAGES_BY_CHECKPOINT_PIPELINE,
712        SYSTEM_PACKAGES_PIPELINE,
713        // List APIs, which read the bitmap tables of `bitmap_schema`.
714        TX_SEQ_DIGEST_PIPELINE,
715        transaction_bitmap,
716        event_bitmap,
717    ];
718    pipelines.extend(alpha_features.watermark_pipelines());
719    pipelines
720}

852    async fn served_checkpoint_gauges_follow_refresh_and_survive_failed_refresh() {
853        let mock = MockBigtableServer::new();
854        let (addr, _mock_server) = mock.start().await.expect("start mock BigTable");
855        let mut client = BigTableClient::new_local(addr.to_string(), "test".to_string())
856            .await
857            .expect("connect to mock BigTable");
858
859        // The objects pipeline lags, so it bounds the served checkpoint, and the timestamp must
860        // come from its watermark rather than from a pipeline that is further ahead.
861        let pipelines =
862            service_info_watermark_pipelines(BitmapSchema::V2, AlphaFeatures::default());
863        seed_watermarks(&mut client, &pipelines, |pipeline| {
864            if pipeline == OBJECTS_PIPELINE { 20 } else { 25 }
865        })
866        .await;
867
868        let server = start_v2_server(addr, AlphaFeatures::default()).await;
869        timeout(WAIT, server.cache.subscribe().wait_for(Option::is_some))
870            .await
871            .expect("service info never refreshed")
872            .expect("service info sender dropped");
873        assert_eq!(server.metrics.served_checkpoint.get(), 20);
874        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 20_000);
875
876        // Hide the objects watermark (checkpoint_hi_inclusive < reader_lo), so every refresh
877        // fails. Once a second watermark read arrives, the refresh behind the first has finished.
878        let objects_key = Bytes::from(watermarks::encode_key(OBJECTS_PIPELINE));
879        mock.insert_row(
880            watermarks::NAME,
881            objects_key.clone(),
882            [(watermarks::col::READER_LO, u64_be(21))],
883        )
884        .await;
885        mock.clear_read_rows_calls().await;
886        timeout(WAIT, async {
887            while watermark_reads(&mock).await < 2 {
888                sleep(Duration::from_millis(5)).await;
889            }
890        })
891        .await
892        .expect("service info refresh stopped polling");
893        assert_eq!(server.metrics.served_checkpoint.get(), 20);
894        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 20_000);
895        assert!(server.health().is_ready());
896
897        // Once the objects pipeline catches up past the others, they bound the served checkpoint.
898        mock.insert_row(
899            watermarks::NAME,
900            objects_key,
901            [
902                (watermarks::col::CHECKPOINT_HI, u64_be(30)),
903                (watermarks::col::TIMESTAMP_MS_HI, u64_be(30_000)),
904            ],
905        )
906        .await;
907        wait_for_served_checkpoint(&server, 25).await;
908        assert_eq!(server.metrics.served_checkpoint_timestamp_ms.get(), 25_000);
909    }
