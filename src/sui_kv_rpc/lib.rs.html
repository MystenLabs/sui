296/// Build and start one gRPC listener serving `ledger`'s `LedgerService`, wired with the given
297/// (shared, already-constructed) metrics/logging layers and optional reflection. `builder` carries
298/// whatever TLS config the caller wants for this listener (or none, for a plaintext listener).
299///
300/// Used once per listener -- the primary listener, and the optional second plaintext one -- so
301/// that expensive shared pieces (metrics, allowlist, request-log layer) are constructed once by
302/// the caller and passed in here, rather than rebuilt (and, for `RpcMetrics`, re-registered against
303/// the same `Registry`, which panics) per listener.
304fn spawn_listener(
305    listen_address: SocketAddr,
306    builder: Server,
307    ledger: KvRpcServer,
308    rpc_metrics: Arc<sui_rpc_api::RpcMetrics>,
309    grpc_method_allowlist: sui_rpc_api::GrpcMethodAllowlist,
310    request_log_layer: mysten_network::request_log::GrpcRequestLogLayer,
311    enable_reflection: bool,
312    file_descriptor_sets: &[&'static [u8]],
313) -> anyhow::Result<sui_futures::service::Service> {
314    use sui_http::middleware::callback::CallbackLayer;
315    use sui_rpc_api::RpcMetricsMakeCallbackHandler;
316
317    let mut router = builder
318        .layer(CallbackLayer::new(
319            RpcMetricsMakeCallbackHandler::with_grpc_method_allowlist(
320                rpc_metrics,
321                grpc_method_allowlist,
322            ),
323        ))
324        .layer(request_log_layer)
325        .add_service(ledger_service_with_response_compression(ledger));
326
327    if enable_reflection {
328        let mut reflection_v1_builder = tonic_reflection::server::Builder::configure();
329        let mut reflection_v1alpha_builder = tonic_reflection::server::Builder::configure();
330        for &fds in file_descriptor_sets {
331            reflection_v1_builder = reflection_v1_builder.register_encoded_file_descriptor_set(fds);
332            reflection_v1alpha_builder =
333                reflection_v1alpha_builder.register_encoded_file_descriptor_set(fds);
334        }
335        router = router
336            .add_service(reflection_v1_builder.build_v1()?)
337            .add_service(reflection_v1alpha_builder.build_v1alpha()?);
338    }
339
340    let (tx, rx) = tokio::sync::oneshot::channel::<()>();
341    let server_future = router.serve_with_shutdown(listen_address, async {
342        let _ = rx.await;
343    });
344
345    Ok(sui_futures::service::Service::new()
346        .spawn(async move {
347            server_future.await?;
348            Ok(())
349        })
350        .with_shutdown_signal(async move {
351            let _ = tx.send(());
352        }))
353}

452    fn init(
453        client: BigTableClient,
454        chain_id: ChainIdentifier,
455        server_version: Option<ServerVersion>,
456        enable_list_apis: bool,
457        metrics: Arc<KvRpcMetrics>,
458        ledger_history: LedgerHistoryConfig,
459        request_bigtable_concurrency: usize,
460        stages: StagesConfig,
461    ) -> anyhow::Result<Self> {
462        ledger_history.validate()?;
463
464        let cache = Arc::new(RwLock::new(None));
465
466        let package_store: Arc<dyn PackageStore> = Arc::new(PackageStoreWithLruCache::new(
467            BigTablePackageStore::new(client.clone()),
468        ));
469        let package_resolver = Arc::new(Resolver::new(package_store));
470
471        let server = Self {
472            chain_id,
473            client,
474            server_version,
475            service_info_watermark_pipelines: service_info_watermark_pipelines(enable_list_apis),
476            cache,
477            package_resolver,
478            metrics,
479            ledger_history,
480            request_bigtable_concurrency,
481            stages,
482            list_apis_enabled: enable_list_apis,
483        };
484
485        let server_clone = server.clone();
486        tokio::spawn(async move {
487            loop {
488                match v2::get_service_info(
489                    server_clone.client.clone(),
490                    server_clone.chain_id,
491                    server_clone.server_version.clone(),
492                    &server_clone.service_info_watermark_pipelines,
493                )
494                .await
495                {
496                    Ok(info) => {
497                        let mut cache = server_clone.cache.write().await;
498                        *cache = Some(info);
499                    }
500                    Err(e) => error!("Failed to update service info cache: {:?}", e),
501                }
502                sleep(Duration::from_millis(10)).await;
503            }
504        });
505
506        Ok(server)
507    }

519    /// Start this server as a tonic gRPC service on the given address.
520    /// Returns a `Service` handle for lifecycle management.
521    pub async fn start_service(
522        self,
523        listen_address: SocketAddr,
524        config: ServerConfig,
525    ) -> anyhow::Result<sui_futures::service::Service> {
526        use sui_rpc_api::{RpcMetrics, grpc_method_paths_from_file_descriptor_sets};
527
528        // Single source of truth for every encoded FileDescriptorSet that
529        // backs a gRPC service mounted below. Consumed by both the
530        // reflection services and the metrics allowlist so they cannot drift
531        // out of sync, and shared by the optional second (plaintext)
532        // listener below.
533        let file_descriptor_sets: Vec<&'static [u8]> = vec![
534            sui_rpc_api::proto::google::protobuf::FILE_DESCRIPTOR_SET,
535            sui_rpc_api::proto::google::rpc::FILE_DESCRIPTOR_SET,
536            sui_rpc::proto::sui::rpc::v2::FILE_DESCRIPTOR_SET,
537        ];
538
539        let registry = config.metrics_registry.unwrap_or_default();
540        let grpc_method_allowlist = Arc::new(grpc_method_paths_from_file_descriptor_sets(
541            &file_descriptor_sets,
542        )?);
543        // Constructed once and shared (via `Arc`/`Clone`) across both
544        // listeners below: `RpcMetrics::new` registers Prometheus collectors
545        // against `registry`, and doing that twice would panic on duplicate
546        // registration.
547        let rpc_metrics = Arc::new(RpcMetrics::new(&registry));
548        let request_log_layer =
549            mysten_network::request_log::GrpcRequestLogLayer::from_encoded_file_descriptor_sets(
550                file_descriptor_sets.iter().copied(),
551            )?;
552
553        let mut builder = Server::builder();
554        if let Some(identity) = config.tls_identity {
555            builder = builder.tls_config(ServerTlsConfig::new().identity(identity))?;
556        }
557
558        // Cloned up-front (before `self` is consumed below) only when the
559        // second listener is actually being started.
560        let ledger_for_plaintext = config.plaintext_address.is_some().then(|| self.clone());
561
562        let mut service = spawn_listener(
563            listen_address,
564            builder,
565            self,
566            rpc_metrics.clone(),
567            grpc_method_allowlist.clone(),
568            request_log_layer.clone(),
569            config.enable_reflection,
570            &file_descriptor_sets,
571        )?;
572
573        // Second, unencrypted listener for trusted internal callers (e.g.
574        // other in-cluster services) that should not need to negotiate TLS
575        // to reach this server. Serves the same `LedgerService`.
576        if let Some(plaintext_address) = config.plaintext_address {
577            let ledger =
578                ledger_for_plaintext.expect("cloned above whenever plaintext_address is set");
579            let plaintext_service = spawn_listener(
580                plaintext_address,
581                Server::builder(),
582                ledger,
583                rpc_metrics,
584                grpc_method_allowlist,
585                request_log_layer,
586                config.enable_reflection,
587                &file_descriptor_sets,
588            )?;
589            service = service.merge(plaintext_service);
590        }
591
592        Ok(service)
593    }

596/// Pipelines whose watermarks bound the checkpoint height reported by
597/// `GetServiceInfo`: the always-served set, plus the List API pipelines when
598/// those APIs are enabled.
