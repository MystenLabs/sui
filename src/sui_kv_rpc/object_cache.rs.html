4use std::collections::HashMap;
5use std::sync::Arc;
6
7use futures::StreamExt;
8use futures::stream::BoxStream;
9use sui_rpc_api::RpcError;
10use sui_types::object::Object;
11use sui_types::storage::ObjectKey;
12
13use crate::bigtable_client::BigTableClient;
14use crate::request_cache::CacheFetcher;
15
16pub(crate) type ObjectMap = Arc<HashMap<ObjectKey, Object>>;
17
18pub(crate) struct BigTableObjectFetcher {
19    client: BigTableClient,
20}
21
22impl BigTableObjectFetcher {
23    pub(crate) fn new(client: BigTableClient) -> Self {
24        Self { client }
25    }
26}
27
28#[async_trait::async_trait]
29impl CacheFetcher<ObjectKey, Object> for BigTableObjectFetcher {
30    async fn fetch(
31        &self,
32        keys: Vec<ObjectKey>,
33    ) -> Result<BoxStream<'static, Result<(ObjectKey, Object), RpcError>>, RpcError> {
34        let stream = self.client.get_objects_stream(keys).await?;
35        Ok(stream
36            .map(|row| {
37                row.map(|obj| (ObjectKey(obj.id(), obj.version()), obj))
38                    .map_err(RpcError::from)
39            })
40            .boxed())
41    }
42}
