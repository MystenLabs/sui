4//! Shared BigTable resolution layer: turns streams (or bounded sets) of
5//! checkpoint/transaction identifiers into fully-resolved entities — their
6//! transactions and objects — using the burst-planned, concurrency-limited
7//! pipeline primitives. Both the v2 point-get handlers and the list handlers
8//! build on these so read sizing, stage overlap, and object deduplication are
9//! identical across them.

37/// One transaction of a checkpoint, as `resolve_checkpoints` reads it: each
38/// checkpoint is flattened into one item per transaction digest, in contents
39/// order.
40struct CheckpointTransaction {
41    cp_seq: u64,
42    /// `None` only for the single item of a checkpoint without transactions.
43    digest: Option<TransactionDigest>,
44    /// The checkpoint itself, carried by its last item only, so that item
45    /// closes the checkpoint when the items are regrouped.
46    checkpoint: Option<CheckpointData>,
47}
48
49/// A [`CheckpointTransaction`] with its `transactions` row (`None` exactly
50/// when the item has no digest).
51type CheckpointTransactionRow = (CheckpointTransaction, Option<TransactionData>);

53/// Context of the error raised when a read of `resolve_checkpoints` returns
54/// fewer `transactions` rows than it requested.
55const MISSING_CHECKPOINT_TRANSACTIONS: &str =
56    "resolve_checkpoints: BigTable returned fewer transactions than requested";
57
58/// Context of the errors raised when a read of `resolve_checkpoints` returns a
59/// `transactions` row it did not request, or one row twice.
60const CHECKPOINT_TRANSACTION_ROW: &str = "resolve_checkpoints: transaction row";

62/// Attach a per-item `ObjectMap` to each item in `upstream`. When
63/// `needs_objects` is false every item is paired with an empty map (no
64/// BigTable read); otherwise object keys are derived per item via `keys_of`
65/// and read through `pipelined_keyed_batches`: each object key still to read
66/// in a burst of ready items is one seek, and the burst is planned into reads
67/// of at most `max_seeks_per_read` keys, split over the stage's free
68/// `concurrency` slots when `min_seeks_per_split_read` is set, so one item's
69/// objects may be divided between reads. Reads overlap with
70/// the still-arriving upstream, and an object the request has already read, or
71/// is reading, is not read again. The request-scoped cache is owned by the
72/// returned stream, so its in-flight reads abort if the consumer drops the
73/// stream.
74pub(crate) fn with_object_maps<I, E>(
75    upstream: BoxStream<'static, Result<Watermarked<I>, E>>,
76    client: BigTableClient,
77    objects_stage: ResolvedStageConfig,
78    needs_objects: bool,
79    keys_of: impl Fn(&I) -> Vec<ObjectKey> + Send + 'static,
80) -> BoxStream<'static, Result<Watermarked<(I, ObjectMap)>, E>>
81where
82    I: Send + 'static,
83    E: From<anyhow::Error> + Send + 'static,
84{
85    if needs_objects {
86        let with_keys = upstream
87            .map_ok(move |m| {
88                m.map_item(|item| {
89                    let keys = keys_of(&item);
90                    (item, keys)
91                })
92            })
93            .boxed();
94        let fetcher: Arc<dyn CacheFetcher<ObjectKey, Object>> =
95            Arc::new(BigTableObjectFetcher::new(client));
96        pipelined_keyed_batches(
97            with_keys,
98            objects_stage.max_seeks_per_read,
99            objects_stage.concurrency,
100            objects_stage.min_seeks_per_split_read,
101            fetcher,
102        )
103    } else {
104        let empty: ObjectMap = Arc::new(HashMap::new());
105        upstream
106            .map_ok(move |m| {
107                let empty = empty.clone();
108                m.map_item(move |item| (item, empty.clone()))
109            })
110            .boxed()
111    }
112}

114/// Resolve a stream of `(cp_seq, CheckpointData)` into
115/// `(cp_seq, cp_data, txs, objects)`, exactly one item per input checkpoint,
116/// in input order, with watermarks in place. Shared by `get_checkpoint` and
117/// `list_checkpoints`.
118///
119/// Reads work per transaction; only rendering needs the whole checkpoint.
120/// Each checkpoint is flattened into one item per transaction digest, so
121/// stage C plans a checkpoint's digests like any `transactions` read
122/// (`BatchGetTransactions`): split over the stage's free `concurrency` slots,
123/// at most `max_seeks_per_read` per read. Stage C emits transactions in input order,
124/// and stage D attaches each transaction's objects, so a transaction's object
125/// read can start once its row and every earlier transaction's row have
126/// arrived, without waiting for the rest of its checkpoint. The transactions
127/// are then regrouped per checkpoint.
128pub(crate) fn resolve_checkpoints<E>(
129    client: BigTableClient,
130    read_mask: &FieldMaskTree,
131    transactions_stage: ResolvedStageConfig,
132    objects_stage: ResolvedStageConfig,
133    cp_data_stream: BoxStream<'static, Result<Watermarked<(u64, CheckpointData)>, E>>,
134) -> BoxStream<'static, Result<Watermarked<ResolvedCp>, E>>
135where
136    E: From<anyhow::Error> + Send + 'static,
137{
138    let columns: Arc<[&'static str]> = list_transactions_columns(read_mask).into();
139    let needs_objects = read_mask.contains(Checkpoint::OBJECTS_FIELD);
140
141    // Stage C: one item per transaction -> + its `transactions` row, read in
142    // chunks of digests and emitted in input order.
143    let transaction_stream = pipelined_chunks(
144        flatten_checkpoints(cp_data_stream),
145        transactions_stage.max_seeks_per_read,
146        transactions_stage.concurrency,
147        transactions_stage.min_seeks_per_split_read,
148        SeekUnit::Item,
149        {
150            let client = client.clone();
151            move |items: Vec<CheckpointTransaction>| {
152                let fetch = fetch_checkpoint_transactions(client.clone(), columns.clone(), items);
153                async move {
154                    let rows = fetch.await.map_err(E::from)?;
155                    Ok::<_, E>(rows.map(|row| row.map_err(E::from)).boxed())
156                }
157            }
158        },
159    );
160
161    // Stage D: each transaction's own objects.
162    let with_objects = with_object_maps(
163        transaction_stream,
164        client,
165        objects_stage,
166        needs_objects,
167        |(_, transaction): &CheckpointTransactionRow| {
168            transaction.as_ref().map_or_else(Vec::new, |transaction| {
169                compute_object_keys(transaction).into_iter().collect()
170            })
171        },
172    );
173
174    regroup_checkpoints(with_objects)
175}

177/// Flatten each checkpoint into one [`CheckpointTransaction`] per transaction
178/// digest of its contents, in contents order; the last item carries the
179/// checkpoint. A checkpoint without transactions becomes one item with no
180/// digest. Watermarks pass through unchanged.
181fn flatten_checkpoints<E>(
182    checkpoints: BoxStream<'static, Result<Watermarked<(u64, CheckpointData)>, E>>,
183) -> BoxStream<'static, Result<Watermarked<CheckpointTransaction>, E>>
184where
185    E: From<anyhow::Error> + Send + 'static,
186{
187    async_stream::try_stream! {
188        let mut checkpoints = checkpoints;
189        while let Some(frame) = checkpoints.next().await {
190            match frame? {
191                Watermarked::Item((cp_seq, cp_data)) => {
192                    let mut digests = cp_data
193                        .contents
194                        .as_ref()
195                        .ok_or_else(|| {
196                            E::from(anyhow::anyhow!(
197                                "checkpoint {cp_seq} contents column missing"
198                            ))
199                        })?
200                        .iter()
201                        .map(|d| d.transaction)
202                        .collect::<Vec<TransactionDigest>>();
203                    let last = digests.pop();
204                    for digest in digests {
205                        yield Watermarked::Item(CheckpointTransaction {
206                            cp_seq,
207                            digest: Some(digest),
208                            checkpoint: None,
209                        });
210                    }
211                    yield Watermarked::Item(CheckpointTransaction {
212                        cp_seq,
213                        digest: last,
214                        checkpoint: Some(cp_data),
215                    });
216                }
217                Watermarked::Watermark(p) => yield Watermarked::Watermark(p),
218            }
219        }
220    }
221    .boxed()
222}

224/// Stage-C fetch of `resolve_checkpoints`: read the `transactions` rows of
225/// the items that have a digest and emit every item with its row in input
226/// order. Rows arrive in BigTable key order, so they are reordered by item
227/// index; items without a digest need no row. A row missing from the read is
228/// an error.
229async fn fetch_checkpoint_transactions(
230    client: BigTableClient,
231    columns: Arc<[&'static str]>,
232    items: Vec<CheckpointTransaction>,
233) -> Result<BoxStream<'static, Result<CheckpointTransactionRow, anyhow::Error>>, anyhow::Error> {
234    let item_count = items.len();
235    let mut digests = Vec::with_capacity(item_count);
236    let mut pending_by_digest: HashMap<TransactionDigest, (usize, CheckpointTransaction)> =
237        HashMap::with_capacity(item_count);
238    let mut without_row = Vec::new();
239    for (index, item) in items.into_iter().enumerate() {
240        match item.digest {
241            Some(digest) => {
242                digests.push(digest);
243                pending_by_digest.insert(digest, (index, item));
244            }
245            None => without_row.push((index, item)),
246        }
247    }
248    let rows = fetch_transaction_rows(client, columns, digests).await?;
249    Ok(async_stream::try_stream! {
250        // Keyed by item index so the emitter releases items in input order.
251        let mut emitter: InputOrderEmitter<usize, CheckpointTransactionRow> =
252            InputOrderEmitter::new((0..item_count).collect());
253        for (index, item) in without_row {
254            for ready in emitter.push(index, (item, None), CHECKPOINT_TRANSACTION_ROW)? {
255                yield ready;
256            }
257        }
258        let mut rows = rows;
259        while let Some(row) = rows.next().await {
260            let (digest, transaction) = row?;
261            let (index, item) = pending_by_digest.remove(&digest).ok_or_else(|| {
262                anyhow::anyhow!("resolve_checkpoints: unexpected transaction row {digest}")
263            })?;
264            for ready in emitter.push(
265                index,
266                (item, Some(transaction)),
267                CHECKPOINT_TRANSACTION_ROW,
268            )? {
269                yield ready;
270            }
271        }
272        if !pending_by_digest.is_empty() {
273            Err(missing_checkpoint_transactions(&pending_by_digest))?;
274        }
275        for ready in emitter.finish(MISSING_CHECKPOINT_TRANSACTIONS)? {
276            yield ready;
277        }
278    }
279    .boxed())
280}

282/// The error for a `resolve_checkpoints` read that returned fewer `transactions` rows than it
283/// requested. Names the checkpoints whose rows are missing and up to four of the digests, so the
284/// failing checkpoint can be found from the message alone.
285fn missing_checkpoint_transactions(
286    missing: &HashMap<TransactionDigest, (usize, CheckpointTransaction)>,
287) -> anyhow::Error {
288    let mut checkpoints: Vec<u64> = missing.values().map(|(_, item)| item.cp_seq).collect();
289    checkpoints.sort_unstable();
290    checkpoints.dedup();
291    let digests: Vec<String> = missing.keys().take(4).map(ToString::to_string).collect();
292    anyhow::anyhow!(
293        "{MISSING_CHECKPOINT_TRANSACTIONS}: missing {} row(s) of checkpoint(s) {checkpoints:?}, \
294         including {}",
295        missing.len(),
296        digests.join(", "),
297    )
298}

300/// A checkpoint whose transactions [`regroup_checkpoints`] is gathering.
301struct GatheredCheckpoint {
302    cp_seq: u64,
303    transactions: Vec<TransactionData>,
304    objects: HashMap<ObjectKey, Object>,
305}
306
307/// Fold consecutive transaction items of one checkpoint back into
308/// `(cp_seq, cp_data, txs, objects)`, emitted when the item carrying the
309/// checkpoint arrives. Each checkpoint's map holds only its own transactions'
310/// objects, since `render_full_checkpoint` folds the whole map into an
311/// `ObjectSet`.
312///
313/// A watermark, another checkpoint's item, or the end of input while a
314/// checkpoint is partly gathered means its flattening was broken: that is an
315/// internal error and the checkpoint is never emitted.
316fn regroup_checkpoints<E>(
317    transactions: BoxStream<'static, Result<Watermarked<(CheckpointTransactionRow, ObjectMap)>, E>>,
318) -> BoxStream<'static, Result<Watermarked<ResolvedCp>, E>>
319where
320    E: From<anyhow::Error> + Send + 'static,
321{
322    async_stream::try_stream! {
323        let mut transactions = transactions;
324        let mut gathering: Option<GatheredCheckpoint> = None;
325        while let Some(frame) = transactions.next().await {
326            match frame? {
327                Watermarked::Item(((item, transaction), objects)) => {
328                    let CheckpointTransaction { cp_seq, checkpoint, .. } = item;
329                    let mut gathered = gathering.take().unwrap_or_else(|| GatheredCheckpoint {
330                        cp_seq,
331                        transactions: Vec::new(),
332                        objects: HashMap::new(),
333                    });
334                    if gathered.cp_seq != cp_seq {
335                        Err::<(), E>(partly_gathered_checkpoint(
336                            gathered.cp_seq,
337                            format!("a transaction of checkpoint {cp_seq} arrived"),
338                        ))?;
339                    }
340                    if let Some(transaction) = transaction {
341                        gathered.transactions.push(transaction);
342                    }
343                    // The per-transaction map is uniquely owned unless it is
344                    // the shared empty map, so objects are moved, not cloned.
345                    let objects =
346                        Arc::try_unwrap(objects).unwrap_or_else(|shared| (*shared).clone());
347                    if gathered.objects.is_empty() {
348                        gathered.objects = objects;
349                    } else {
350                        gathered.objects.extend(objects);
351                    }
352                    match checkpoint {
353                        Some(cp_data) => {
354                            yield Watermarked::Item((
355                                cp_seq,
356                                cp_data,
357                                gathered.transactions,
358                                Arc::new(gathered.objects),
359                            ));
360                        }
361                        None => gathering = Some(gathered),
362                    }
363                }
364                Watermarked::Watermark(p) => {
365                    if let Some(gathered) = &gathering {
366                        Err::<(), E>(partly_gathered_checkpoint(
367                            gathered.cp_seq,
368                            "a watermark arrived",
369                        ))?;
370                    }
371                    yield Watermarked::Watermark(p);
372                }
373            }
374        }
375        if let Some(gathered) = &gathering {
376            Err::<(), E>(partly_gathered_checkpoint(gathered.cp_seq, "the stream ended"))?;
377        }
378    }
379    .boxed()
380}

382/// The internal error of [`regroup_checkpoints`] when `cause` interrupts
383/// checkpoint `cp_seq` before its last transaction.
384fn partly_gathered_checkpoint<E: From<anyhow::Error>>(
385    cp_seq: u64,
386    cause: impl std::fmt::Display,
387) -> E {
388    E::from(anyhow::anyhow!(
389        "resolve_checkpoints: {cause} while checkpoint {cp_seq} was partly gathered"
390    ))
391}

393/// Resolve a single already-fetched checkpoint (the point-get path): drive
394/// `resolve_checkpoints` over a one-item stream and return the sole result.
395/// `resolve_checkpoints` emits exactly one item per input checkpoint (a
396/// checkpoint without transactions reads none) or propagates a fetch error, so
397/// the stream is never empty — hence no "missing result" failure mode.
398pub(crate) async fn resolve_checkpoint(
399    client: BigTableClient,
400    read_mask: &FieldMaskTree,
401    transactions_stage: ResolvedStageConfig,
402    objects_stage: ResolvedStageConfig,
403    cp_seq: u64,
404    cp_data: CheckpointData,
405) -> Result<ResolvedCp, RpcError> {
406    let cp_stream =
407        stream::once(async move { Ok::<_, anyhow::Error>(Watermarked::Item((cp_seq, cp_data))) })
408            .boxed();
409    let stream = resolve_checkpoints(
410        client,
411        read_mask,
412        transactions_stage,
413        objects_stage,
414        cp_stream,
415    );
416    futures::pin_mut!(stream);
417    while let Some(item) = stream.next().await {
418        if let Watermarked::Item(resolved) = item.map_err(RpcError::from)? {
419            return Ok(resolved);
420        }
421    }
422    unreachable!("resolve_checkpoints emits exactly one item per input checkpoint")
423}

425/// Resolve `digests` into transactions and the object data needed to render
426/// complete `ExecutedTransaction.objects` sets or annotate effects object
427/// types, keyed by digest for request-order reconstruction. Transaction rows
428/// stream in by chunk and their object reads overlap the transaction fetch. An
429/// empty `digests` issues no BigTable read.
430///
431/// A transaction whose checkpoint is above `checkpoint_hi_inclusive` is left out
432/// of the result before its objects are read: its `transactions` row can be
433/// written before the `objects` rows it references, so reading them could fail.
434pub(crate) async fn resolve_transactions(
435    client: BigTableClient,
436    digests: Vec<TransactionDigest>,
437    checkpoint_hi_inclusive: u64,
438    read_mask: &FieldMaskTree,
439    transactions_stage: ResolvedStageConfig,
440    objects_stage: ResolvedStageConfig,
441) -> Result<HashMap<TransactionDigest, (TransactionData, ObjectMap)>, RpcError> {
442    if digests.is_empty() {
443        return Ok(HashMap::new());
444    }
445    let columns: Arc<[&'static str]> = transaction_columns(read_mask).into();
446    let needs_objects = needs_transaction_objects(read_mask);
447
448    let digest_stream = stream::iter(
449        digests
450            .into_iter()
451            .map(|d| Ok::<_, anyhow::Error>(Watermarked::Item(d))),
452    )
453    .boxed();
454
455    // Stage A: digests -> (digest, TransactionData), chunked & concurrent.
456    let tx_stream = pipelined_chunks(
457        digest_stream,
458        transactions_stage.max_seeks_per_read,
459        transactions_stage.concurrency,
460        transactions_stage.min_seeks_per_split_read,
461        SeekUnit::Item,
462        {
463            let client = client.clone();
464            let columns = columns.clone();
465            move |chunk| fetch_transaction_rows(client.clone(), columns.clone(), chunk)
466        },
467    )
468    .try_filter(move |item| {
469        std::future::ready(!matches!(
470            item,
471            Watermarked::Item((_, tx)) if tx.checkpoint_number > checkpoint_hi_inclusive
472        ))
473    })
474    .boxed();
475
476    // Stage B: attach each tx's own objects.
477    let with_objects = with_object_maps(
478        tx_stream,
479        client,
480        objects_stage,
481        needs_objects,
482        |(_, tx): &(TransactionDigest, TransactionData)| {
483            compute_object_keys(tx).into_iter().collect()
484        },
485    );
486    futures::pin_mut!(with_objects);
487
488    let mut out = HashMap::new();
489    while let Some(item) = with_objects.next().await {
490        if let Watermarked::Item(((digest, tx), objects)) = item.map_err(RpcError::from)? {
491            out.insert(digest, (tx, objects));
492        }
493    }
494    Ok(out)
495}

497/// Fetch one chunk of transaction rows as a stream, in arrival order.
498/// Used by stage A of `resolve_transactions` and stage C of
499/// `resolve_checkpoints`. An empty chunk issues no read (see the empty-RowSet
500/// full-scan trap guarded in `sui-kvstore`).
501async fn fetch_transaction_rows(
502    client: BigTableClient,
503    columns: Arc<[&'static str]>,
504    digests: Vec<TransactionDigest>,
505) -> Result<
506    BoxStream<'static, Result<(TransactionDigest, TransactionData), anyhow::Error>>,
507    anyhow::Error,
508> {
509    if digests.is_empty() {
510        return Ok(stream::empty().boxed());
511    }
512    let column_filter = BigTableClient::column_filter(&columns);
513    Ok(client
514        .get_transactions_stream(digests, Some(column_filter))
515        .await?
516        .boxed())
517}

521/// Whether the checkpoint read mask requests transactions or objects (the heavy
522/// path that needs `resolve_checkpoints`).
523pub(crate) fn needs_transactions_or_objects(mask: &FieldMaskTree) -> bool {
524    mask.contains(Checkpoint::TRANSACTIONS_FIELD) || mask.contains(Checkpoint::OBJECTS_FIELD)
525}
526
527/// Whether a transaction read mask needs object types resolved (i.e. requests
528/// `effects.changed_objects` or `effects.unchanged_consensus_objects`).
529pub(crate) fn needs_object_types(mask: &FieldMaskTree) -> bool {
530    mask.subtree(ExecutedTransaction::EFFECTS_FIELD.name)
531        .is_some_and(|submask| {
532            submask.contains(TransactionEffects::CHANGED_OBJECTS_FIELD.name)
533                || submask.contains(TransactionEffects::UNCHANGED_CONSENSUS_OBJECTS_FIELD.name)
534        })
535}
536
537/// Whether a transaction read mask needs its canonical object set or object
538/// types embedded in effects.
539pub(crate) fn needs_transaction_objects(mask: &FieldMaskTree) -> bool {
540    mask.contains(ExecutedTransaction::OBJECTS_FIELD.name) || needs_object_types(mask)
541}

543/// Canonical object keys referenced by a transaction's inputs and effects.
544/// Callers use these objects for complete transaction object sets and effects
545/// type annotation.
546pub(crate) fn compute_object_keys(source: &TransactionData) -> BTreeSet<ObjectKey> {
547    match (&source.transaction_data, &source.effects) {
548        (Some(tx_data), Some(effects)) => sui_types::storage::get_transaction_object_set(
549            tx_data,
550            effects,
551            &source.unchanged_loaded_runtime_objects,
552        ),
553        _ => BTreeSet::new(),
554    }
555}

557/// Checkpoint-table columns needed for `mask`. Always includes `s` (summary);
558/// adds `sg` (signatures) and `c` (contents) when requested.
559pub(crate) fn checkpoint_columns(mask: &FieldMaskTree) -> Vec<&'static str> {
560    use tables::checkpoints::col;
561    let mut columns = vec![col::SUMMARY];
562    if mask.contains(Checkpoint::SIGNATURE_FIELD) {
563        columns.push(col::SIGNATURES);
564    }
565    if mask.contains(Checkpoint::CONTENTS_FIELD) {
566        columns.push(col::CONTENTS);
567    }
568    columns
569}

571/// Checkpoint-table columns for the list/get checkpoint handlers. The heavy
572/// (transactions/objects) path additionally needs `signatures` (to reconstruct
573/// `CertifiedCheckpointSummary`) and `contents` (to enumerate tx digests).
574pub(crate) fn list_checkpoint_columns(mask: &FieldMaskTree, needs_full: bool) -> Vec<&'static str> {
575    use tables::checkpoints::col;
576    let mut columns = checkpoint_columns(mask);
577    if needs_full {
578        if !columns.contains(&col::CONTENTS) {
579            columns.push(col::CONTENTS);
580        }
581        if !columns.contains(&col::SIGNATURES) {
582            columns.push(col::SIGNATURES);
583        }
584    }
585    columns
586}

596fn transaction_columns_for_mask(mask: &FieldMaskTree, needs_objects: bool) -> Vec<&'static str> {
597    use tables::transactions::col;
598    let mut columns = vec![col::CHECKPOINT_NUMBER, col::TIMESTAMP];
599
600    if mask
601        .subtree(ExecutedTransaction::TRANSACTION_FIELD.name)
602        .is_some()
603    {
604        columns.push(col::DATA);
605    }
606    if mask
607        .subtree(ExecutedTransaction::SIGNATURES_FIELD.name)
608        .is_some()
609    {
610        columns.push(col::SIGNATURES);
611    }
612    if let Some(effects_submask) = mask.subtree(ExecutedTransaction::EFFECTS_FIELD.name) {
613        columns.push(col::EFFECTS);
614        if effects_submask.contains(TransactionEffects::UNCHANGED_LOADED_RUNTIME_OBJECTS_FIELD.name)
615        {
616            columns.push(col::UNCHANGED_LOADED);
617        }
618    }
619    if mask
620        .subtree(ExecutedTransaction::EVENTS_FIELD.name)
621        .is_some()
622    {
623        columns.push(col::EVENTS);
624    }
625    if mask.contains(ExecutedTransaction::BALANCE_CHANGES_FIELD.name) {
626        columns.push(col::BALANCE_CHANGES);
627    }
628    if needs_objects {
629        for column in [col::DATA, col::EFFECTS, col::UNCHANGED_LOADED] {
630            if !columns.contains(&column) {
631                columns.push(column);
632            }
633        }
634    }
635
636    columns
637}

639/// Transaction-table columns needed to resolve a checkpoint's transactions. The
640/// merge target (`full_checkpoint_content::ExecutedTransaction`) has non-Option
641/// `transaction`/`effects`, so `td`+`ef` are always fetched; object resolution
642/// also needs `ul`.
643pub(crate) fn list_transactions_columns(mask: &FieldMaskTree) -> Vec<&'static str> {
644    use tables::transactions::col;
645    let mut columns = if let Some(submask) = mask.subtree(Checkpoint::TRANSACTIONS_FIELD.name) {
646        transaction_columns_for_mask(&submask, needs_object_types(&submask))
647    } else {
648        // Baseline metadata columns even if the proto `transactions` field
649        // isn't requested; we still need the rows to compute object keys.
650        vec![col::CHECKPOINT_NUMBER, col::TIMESTAMP]
651    };
652    // Required to construct the merge target faithfully.
653    for c in [col::DATA, col::EFFECTS] {
654        if !columns.contains(&c) {
655            columns.push(c);
656        }
657    }
658    if mask.contains(Checkpoint::OBJECTS_FIELD) && !columns.contains(&col::UNCHANGED_LOADED) {
659        columns.push(col::UNCHANGED_LOADED);
660    }
661    columns
662}

686    /// Deterministic, unique 32-byte digest: `i` big-endian in the low 8 bytes.
687    fn digest(i: u64) -> TransactionDigest {
688        let mut bytes = [0u8; 32];
689        bytes[24..32].copy_from_slice(&i.to_be_bytes());
690        TransactionDigest::new(bytes)
691    }

763    /// Drive the real `resolve_checkpoints` stream over `cps` and collect the
764    /// emitted checkpoint items (dropping watermarks).
765    async fn run_resolver(
766        client: BigTableClient,
767        read_mask: &FieldMaskTree,
768        cps: Vec<(u64, CheckpointData)>,
769    ) -> Result<Vec<ResolvedCp>, RpcError> {
770        let frames = cps.into_iter().map(Watermarked::Item).collect();
771        let out =
772            run_resolver_frames(client, read_mask, transactions_stage(4, None), frames).await?;
773        Ok(out
774            .into_iter()
775            .filter_map(|frame| match frame {
776                Watermarked::Item(resolved) => Some(resolved),
777                Watermarked::Watermark(_) => None,
778            })
779            .collect())
780    }

786        read_mask: &FieldMaskTree,
787        tx_stage: ResolvedStageConfig,
788        frames: Vec<Watermarked<(u64, CheckpointData)>>,

799            .await
800    }

812    async fn resolve_checkpoints_splits_fat_checkpoint_transaction_reads() {
813        let (mock, client, _handle) = setup().await;
814        // One checkpoint with more transactions than the stage's `max_seeks_per_read`, so
815        // the pipeline must split its digests across multiple capped BigTable
816        // requests (the objects stage batches the same way). The backend
817        // `MAX_TX_DIGESTS_PER_REQUEST` clamp is a separate safety net covered by
818        // the `sui-kvstore` `get_transactions_stream_*` tests.
819        let n = TX_CHUNK_SIZE * 2 + 1;
820        let digests: Vec<_> = (0..n as u64).map(digest).collect();
821        for d in &digests {
822            insert_tx_row(&mock, *d).await;
823        }
824        let read_mask = FieldMaskTree::from(FieldMask::from_paths(["transactions.digest"]));
825        let out = run_resolver(client, &read_mask, vec![(0, checkpoint_data(&digests))])
826            .await
827            .unwrap();
828
829        assert_eq!(out.len(), 1, "exactly one checkpoint item");
830        let (_, _, txs, _) = &out[0];
831        assert_eq!(txs.len(), n, "all transactions returned");
832
833        let calls = tx_read_calls(&mock).await;
834        assert!(
835            calls.len() >= 2,
836            "fat checkpoint should split into >=2 ReadRows calls, got {}",
837            calls.len()
838        );
839        for c in &calls {
840            assert!(
841                c.row_keys.len() <= TX_CHUNK_SIZE,
842                "ReadRows call exceeded max_seeks_per_read: {} keys",
843                c.row_keys.len()
844            );
845        }
846        let total: usize = calls.iter().map(|c| c.row_keys.len()).sum();
847        assert_eq!(total, n, "every digest fetched exactly once");
848    }

1034    /// checkpoint `cp_seq` without row or objects; `last` makes it carry the

1038        digest_index: u64,
