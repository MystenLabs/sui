4//! The checkpoint height a replica serves, and the bound it puts on answers.
5//!
6//! The indexer's pipelines write their rows independently, so a checkpoint's `checkpoints` row can
7//! exist before its `transactions` and `objects` rows do. Only the pipelines' watermarks promise
8//! that every row of a checkpoint is written, so every answer that depends on how far the ledger
9//! has been indexed is bounded by the lowest of them: the height `GetServiceInfo` reports, read
10//! from the replica's cached service info. A caller that has seen a higher checkpoint from another
11//! replica asks for it with `x-sui-min-checkpoint` (see [`crate::monotonic_read`]).
12
13use sui_rpc_api::RpcError;
14
15use crate::KvRpcServer;
16
17/// The highest checkpoint served (inclusive) and the epoch it belongs to.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub(crate) struct ServedHeight {
20    pub(crate) checkpoint: u64,
21    pub(crate) epoch: u64,
22}
23
24impl KvRpcServer {
25    /// This replica's served height, from its cached service info. `Unavailable` before the
26    /// first refresh.
27    pub(crate) fn served_height(&self) -> Result<ServedHeight, RpcError> {
28        let cache = self.cache.borrow();
29        let info = cache.as_ref();
30        match (
31            info.and_then(|info| info.checkpoint_height),
32            info.and_then(|info| info.epoch),
33        ) {
34            (Some(checkpoint), Some(epoch)) => Ok(ServedHeight { checkpoint, epoch }),
35            _ => Err(RpcError::new(
36                tonic::Code::Unavailable,
37                "service info cache missing checkpoint height",
38            )),
39        }
40    }
41}
