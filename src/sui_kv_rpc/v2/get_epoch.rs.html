38/// Serve the epoch the request names, or the latest one, as of `served_height`: an epoch that
39/// starts above it is not found, and an epoch that ends above it has not ended yet, so its end
40/// fields are left out even when the indexer has already written them.
41///
42/// `system_state` is the epoch-start snapshot for an epoch below the served one. For the served
43/// epoch it is the state at the served checkpoint, read from the object versions index when
44/// `alpha_features.object_at_checkpoint`, or at the checkpoint before it when the served checkpoint
45/// is the epoch's last: the change-epoch transaction there has already turned the state into the
46/// next epoch's. Without `alpha_features.object_at_checkpoint`, a request for the served epoch's
47/// `system_state` is `Unimplemented` / `NOT_SERVED_HERE` (cause `disabled_by_config`), refused
48/// before any read. `committee` comes from the same state when `system_state` is read too, and
49/// from the epoch-start snapshot otherwise; the committee does not change within an epoch.
50///
51/// `protocol_config` comes from the `protocol_configs_v2` table with
52/// `alpha_features.protocol_configs_v2`, and otherwise from the protocol configs compiled into
53/// this binary for `chain`.
54///
55/// Every read goes through `client`, the request's limited client.
56pub async fn get_epoch(
57    client: BigTableClient,
58    alpha_features: AlphaFeatures,
59    chain: Chain,
60    served_height: ServedHeight,
61    request: GetEpochRequest,
62) -> sui_rpc_api::Result<GetEpochResponse> {
63    let read_mask = {
64        let read_mask = request
65            .read_mask
66            .unwrap_or_else(|| FieldMask::from_str(READ_MASK_DEFAULT));
67        read_mask.validate::<Epoch>().map_err(|path| {
68            FieldViolation::new("read_mask")
69                .with_description(format!("invalid read_mask path: {path}"))
70                .with_reason(ErrorReason::FieldInvalid)
71        })?;
72        FieldMaskTree::from(read_mask)
73    };
74
75    let mut message = Epoch::default();
76
77    let epoch = request.epoch.unwrap_or(served_height.epoch);
78    if epoch > served_height.epoch {
79        return Err(EpochNotFoundError::new(epoch).into());
80    }
81    // The served epoch's `system_state` is read at the served height, and `committee` comes from
82    // it, so the first read fetches the snapshot only for an epoch below the served one, or for
83    // `committee` without `system_state`.
84    let reads_state_at_served_height =
85        epoch == served_height.epoch && read_mask.contains(Epoch::SYSTEM_STATE_FIELD.name);
86    if reads_state_at_served_height && !alpha_features.object_at_checkpoint {
87        return Err(not_served_here(
88            NotServedHereCause::DisabledByConfig,
89            "the running epoch's system_state needs at_checkpoint object reads, which are not \
90             enabled on this instance",
91        ));
92    }
93    let columns = epoch_columns(&read_mask, !reads_state_at_served_height);
94    let epoch_future = async { client.get_epochs_filtered(&[epoch], Some(&columns)).await };
95    let transaction_start_future = async {
96        if !read_mask.contains(Epoch::TRANSACTION_START_INCLUSIVE_FIELD.name) {
97            return Ok(None);
98        }
99        let Some(previous) = epoch.checked_sub(1) else {
100            return Ok(Some(0));
101        };
102        // The previous epoch ended below this epoch's first checkpoint, so at or below the served
103        // height.
104        let previous = client
105            .get_epochs_filtered(&[previous], Some(&[col::EPOCH, col::TX_HI]))
106            .await?
107            .pop();
108        Ok::<_, RpcError>(previous.and_then(|previous| previous.tx_hi))
109    };
110    let (mut epochs, transaction_start_inclusive) =
111        tokio::try_join!(epoch_future, transaction_start_future)?;
112    let mut epoch_info = epochs.pop().ok_or_else(|| EpochNotFoundError::new(epoch))?;
113    if epoch_info
114        .end_checkpoint
115        .is_some_and(|end_checkpoint| end_checkpoint > served_height.checkpoint)
116    {
117        epoch_info.end_checkpoint = None;
118        epoch_info.end_timestamp_ms = None;
119    }
120    if read_mask.contains(Epoch::EPOCH_FIELD.name) {
121        message.epoch = epoch_info.epoch;
122    }
123    if read_mask.contains(Epoch::FIRST_CHECKPOINT_FIELD.name) {
124        message.first_checkpoint = epoch_info.start_checkpoint;
125    }
126    if read_mask.contains(Epoch::LAST_CHECKPOINT_FIELD.name) {
127        message.last_checkpoint = epoch_info.end_checkpoint;
128    }
129    if read_mask.contains(Epoch::START_FIELD.name) {
130        message.start = epoch_info.start_timestamp_ms.map(timestamp_ms_to_proto);
131    }
132    if read_mask.contains(Epoch::END_FIELD.name) {
133        message.end = epoch_info.end_timestamp_ms.map(timestamp_ms_to_proto);
134    }
135    if read_mask.contains(Epoch::REFERENCE_GAS_PRICE_FIELD.name) {
136        message.reference_gas_price = epoch_info.reference_gas_price;
137    }
138    if let (Some(submask), Some(version)) = (
139        read_mask.subtree(Epoch::PROTOCOL_CONFIG_FIELD.name),
140        epoch_info.protocol_version,
141    ) {
142        // The `epochs` row already holds the version, so the `protocol_configs_v2` row is read
143        // only for the settings it stores.
144        let needs_settings = [
145            RpcProtocolConfig::FEATURE_FLAGS_FIELD,
146            RpcProtocolConfig::ATTRIBUTES_FIELD,
147            RpcProtocolConfig::CONFIGS_FIELD,
148        ]
149        .iter()
150        .any(|field| submask.contains(field.name));
151        let protocol_config = if !alpha_features.protocol_configs_v2 {
152            ProtocolConfig::get_for_version_if_supported(version.into(), chain)
153                .map(protocol_config_to_proto)
154        } else if !needs_settings {
155            Some(RpcProtocolConfig::default().with_protocol_version(version))
156        } else {
157            // The epoch is served, so its version was decided at or below the served height,
158            // which the `protocol_configs_v2` pipeline's watermark bounds.
159            let Some(protocol_config) = read_protocol_config(&client, version).await? else {
160                return Err(RpcError::new(
161                    tonic::Code::Internal,
162                    format!(
163                        "epoch {epoch} ran protocol version {version}, which has no \
164                         protocol_configs_v2 row"
165                    ),
166                ));
167            };
168            Some(protocol_config)
169        };
170        message.protocol_config = protocol_config
171            .map(|protocol_config| RpcProtocolConfig::merge_from(protocol_config, &submask));
172    }
173    if epoch_info.end_checkpoint.is_some() {
174        set_epoch_end_fields(&mut message, &read_mask, &epoch_end(&epoch_info)?);
175    }
176    message.transaction_start_inclusive = transaction_start_inclusive;
177    let system_state = if reads_state_at_served_height {
178        // Epoch 0's last checkpoint is never the genesis checkpoint.
179        let checkpoint = if epoch_info.end_checkpoint == Some(served_height.checkpoint) {
180            served_height.checkpoint - 1
181        } else {
182            served_height.checkpoint
183        };
184        Some(live_system_state(&client, checkpoint).await?)
185    } else {
186        epoch_info.system_state.take()
187    };
188    if read_mask.contains(Epoch::COMMITTEE_FIELD.name) {
189        message.committee = system_state.as_ref().map(|system_state| {
190            system_state
191                .get_current_epoch_committee()
192                .committee()
193                .clone()
194                .into()
195        });
196    }
197    if read_mask.contains(Epoch::SYSTEM_STATE_FIELD.name) {
198        message.system_state = system_state.map(|system_state| Box::new(system_state.into()));
199    }
200    Ok(GetEpochResponse::new(message))
201}

203/// The end-of-epoch columns of `epoch_info`, decoded into the shape `set_epoch_end_fields` reads.
204fn epoch_end(epoch_info: &EpochData) -> Result<EpochInfo, RpcError> {
205    let epoch_commitments = epoch_info
206        .epoch_commitments
207        .as_deref()
208        .map(bcs::from_bytes::<Vec<CheckpointCommitment>>)
209        .transpose()
210        .map_err(|e| {
211            RpcError::new(
212                tonic::Code::Internal,
213                format!("decode epoch commitments: {e}"),
214            )
215        })?;
216    Ok(EpochInfo {
217        tx_hi: epoch_info.tx_hi,
218        safe_mode: epoch_info.safe_mode,
219        epoch_commitments,
220        total_stake: epoch_info.total_stake,
221        storage_fund_balance: epoch_info.storage_fund_balance,
222        storage_fund_reinvestment: epoch_info.storage_fund_reinvestment,
223        storage_charge: epoch_info.storage_charge,
224        storage_rebate: epoch_info.storage_rebate,
225        stake_subsidy_amount: epoch_info.stake_subsidy_amount,
226        total_gas_fees: epoch_info.total_gas_fees,
227        total_stake_rewards_distributed: epoch_info.total_stake_rewards_distributed,
228        leftover_storage_fund_inflow: epoch_info.leftover_storage_fund_inflow,
229        ..EpochInfo::default()
230    })
231}

233/// The system state at the end of `checkpoint`: the `0x5` wrapper and the inner object its
234/// `version` keys, each read at that checkpoint. The version is read too, not taken from the
235/// epoch-start snapshot, because the first mutable access to the system state after genesis or a
236/// framework upgrade migrates the inner object to a new version mid-epoch.
237async fn live_system_state(
238    client: &BigTableClient,
239    checkpoint: u64,
240) -> Result<SuiSystemState, RpcError> {
241    let internal = |message: String| RpcError::new(tonic::Code::Internal, message);
242    let mut objects =
243        vec![object_at_checkpoint(client, SUI_SYSTEM_STATE_OBJECT_ID, checkpoint).await?];
244    let version = get_sui_system_state_wrapper(&objects.as_slice())
245        .map_err(|e| internal(format!("decode system state wrapper: {e}")))?
246        .version;
247    objects
248        .push(object_at_checkpoint(client, sui_system_state_inner_id(version), checkpoint).await?);
249    get_sui_system_state(&objects.as_slice()).map_err(|e| {
250        internal(format!(
251            "decode system state at checkpoint {checkpoint}: {e}"
252        ))
253    })
254}

256/// The version of `id` live at the end of `checkpoint`, from the version index (a single-row
257/// forward read), read from the `objects` table. `Internal` if it is not live there: callers read
258/// only objects that always exist.
259async fn object_at_checkpoint(
260    client: &BigTableClient,
261    id: ObjectID,
262    checkpoint: u64,
263) -> Result<Object, RpcError> {
264    let internal = |message: String| RpcError::new(tonic::Code::Internal, message);
265    let version = match client
266        .get_object_version_at_checkpoint(id, checkpoint)
267        .await?
268    {
269        Some((version, ObjectStatus::Live)) => version,
270        other => {
271            return Err(internal(format!(
272                "object {id} is not live at checkpoint {checkpoint}: {other:?}"
273            )));
274        }
275    };
276    // The read runs to its end, so it finishes as completed rather than cut off.
277    let rows: Vec<Object> = client
278        .get_objects_stream(vec![ObjectKey(id, version)])
279        .await?
280        .try_collect()
281        .await?;
282    rows.into_iter().next().ok_or_else(|| {
283        internal(format!(
284            "object {id} version {version} is not in the objects table"
285        ))
286    })
287}

289/// Compute the set of BigTable columns needed for the given read mask. Always includes `ep` and
290/// `ec`: the end checkpoint decides whether the epoch has ended at the served height, which every
291/// end field and `system_state` depend on. The epoch-start snapshot is read for `committee` or
292/// `system_state` only if `reads_snapshot`.
293fn epoch_columns(mask: &FieldMaskTree, reads_snapshot: bool) -> Vec<&'static str> {
294    let mut columns = vec![col::EPOCH, col::END_CHECKPOINT];
295
296    if mask.contains(Epoch::FIRST_CHECKPOINT_FIELD.name) {
297        columns.push(col::START_CHECKPOINT);
298    }
299    let end_fields = [
300        (
301            Epoch::TRANSACTION_END_EXCLUSIVE_FIELD.name,
302            &[col::TX_HI][..],
303        ),
304        (Epoch::ENDED_IN_SAFE_MODE_FIELD.name, &[col::SAFE_MODE][..]),
305        (
306            Epoch::SYSTEM_EPOCH_INFO_FIELD.name,
307            &[
308                col::TOTAL_STAKE,
309                col::STORAGE_FUND_REINVESTMENT,
310                col::STORAGE_CHARGE,
311                col::STORAGE_REBATE,
312                col::STORAGE_FUND_BALANCE,
313                col::STAKE_SUBSIDY_AMOUNT,
314                col::TOTAL_GAS_FEES,
315                col::TOTAL_STAKE_REWARDS_DISTRIBUTED,
316                col::LEFTOVER_STORAGE_FUND_INFLOW,
317            ][..],
318        ),
319        (
320            Epoch::EPOCH_COMMITMENTS_FIELD.name,
321            &[col::EPOCH_COMMITMENTS][..],
322        ),
323    ];
324    for (field, field_columns) in end_fields {
325        if mask.contains(field) {
326            columns.extend_from_slice(field_columns);
327        }
328    }
329    if mask.contains(Epoch::START_FIELD.name) {
330        columns.push(col::START_TIMESTAMP);
331    }
332    if mask.contains(Epoch::END_FIELD.name) {
333        columns.push(col::END_TIMESTAMP);
334    }
335    if mask.contains(Epoch::REFERENCE_GAS_PRICE_FIELD.name) {
336        columns.push(col::REFERENCE_GAS_PRICE);
337    }
338    if mask.subtree(Epoch::PROTOCOL_CONFIG_FIELD.name).is_some() {
339        columns.push(col::PROTOCOL_VERSION);
340    }
341    if reads_snapshot
342        && (mask.contains(Epoch::COMMITTEE_FIELD.name)
343            || mask.contains(Epoch::SYSTEM_STATE_FIELD.name))
344    {
345        columns.push(col::SYSTEM_STATE);
346    }
347
348    columns
349}

475    /// The served epoch's `system_state` is the state at the served checkpoint, read through the
476    /// version index, never a version written above it, and following the inner object when it
477    /// migrates to a new `system_state_version` mid-epoch. At the epoch's last checkpoint it is the
478    /// state at the checkpoint before, since the change-epoch transaction there has already
479    /// written the next epoch's state. A past epoch's is its epoch-start snapshot. Without
480    /// checkpoint-pinned reads the served epoch's `system_state` is `Unimplemented` /
481    /// `NOT_SERVED_HERE` with cause `disabled_by_config`, refused before any read.
482    #[tokio::test]
483    async fn served_epoch_system_state_is_read_at_the_served_checkpoint() {
484        let mock = MockBigtableServer::new();
485        let (addr, _server) = mock.start().await.expect("start mock BigTable");
486        let client = KvStoreClient::new_local(addr.to_string(), "test".to_string())
487            .await
488            .expect("connect to mock BigTable");
489        // Epoch 0 starts on a V1 inner object with reference gas price 1 and ends at checkpoint
490        // 15. The system state changes at checkpoint 3, and at checkpoint 12 it migrates to V2:
491        // the V1 inner object is deleted and a V2 one created under the new version's key. The
492        // change-epoch transaction at checkpoint 15 writes epoch 1's state.
493        mock.insert_row(
494            epochs::NAME,
495            epochs::encode_key(0),
496            [
497                u64_cell(col::EPOCH, 0),
498                u64_cell(col::START_CHECKPOINT, 0),
499                u64_cell(col::END_CHECKPOINT, 15),
500                (
501                    col::SYSTEM_STATE,
502                    Bytes::from(bcs::to_bytes(&v1(1)).unwrap()),
503                ),
504            ],
505        )
506        .await;
507        insert_system_state(&mock, v1(100), 3).await;
508        insert_system_state(&mock, v2(200), 12).await;
509        insert_version(
510            &mock,
511            sui_system_state_inner_id(1),
512            SequenceNumber::from_u64(12),
513            ObjectStatus::Deleted,
514            12,
515        )
516        .await;
517        insert_system_state(&mock, v2(300), 15).await;
518
519        let at = |checkpoint| ServedHeight {
520            checkpoint,
521            epoch: 0,
522        };
523        let reference_gas_price = |epoch: Epoch| {
524            epoch
525                .system_state
526                .expect("system_state")
527                .reference_gas_price
528        };
529
530        for (checkpoint, expected) in [(10, 100), (12, 200), (14, 200), (15, 200)] {
531            let epoch = try_get(
532                &client,
533                OBJECT_AT_CHECKPOINT,
534                at(checkpoint),
535                None,
536                &["system_state"],
537            )
538            .await
539            .expect("get_epoch succeeds");
540            assert_eq!(
541                reference_gas_price(epoch),
542                Some(expected),
543                "served at checkpoint {checkpoint}",
544            );
545        }
546
547        // Once epoch 1 is served, epoch 0 is past: its snapshot, without touching the index, and
548        // without checkpoint-pinned reads.
549        mock.clear_read_rows_calls().await;
550        let inside_epoch_1 = ServedHeight {
551            checkpoint: 20,
552            epoch: 1,
553        };
554        let past = try_get(
555            &client,
556            OBJECT_AT_CHECKPOINT,
557            inside_epoch_1,
558            Some(0),
559            &["system_state"],
560        )
561        .await
562        .expect("get_epoch succeeds");
563        assert_eq!(reference_gas_price(past), Some(1));
564        assert!(
565            mock.read_rows_calls()
566                .await
567                .iter()
568                .all(|call| call.table != object_versions_by_checkpoint::NAME),
569            "a past epoch's system_state reads no object versions",
570        );
571        let past = get(&client, inside_epoch_1, Some(0), &["system_state"]).await;
572        assert_eq!(reference_gas_price(past), Some(1));
573
574        for checkpoint in [10, 15] {
575            mock.clear_read_rows_calls().await;
576            let status = try_get(
577                &client,
578                AlphaFeatures::default(),
579                at(checkpoint),
580                None,
581                &["system_state"],
582            )
583            .await
584            .expect_err("system_state needs checkpoint-pinned reads");
585            assert!(
586                mock.read_rows_calls().await.is_empty(),
587                "refused before any read, served at checkpoint {checkpoint}",
588            );
589            let status = tonic::Status::from(status);
590            assert_eq!(
591                status.code(),
592                tonic::Code::Unimplemented,
593                "served at checkpoint {checkpoint}",
594            );
595            let error_info =
596                <sui_rpc::proto::google::rpc::Status as prost::Message>::decode(status.details())
597                    .expect("error details decode as google.rpc.Status")
598                    .details
599                    .iter()
600                    .find_map(|detail| {
601                        detail
602                            .to_msg::<sui_rpc::proto::google::rpc::ErrorInfo>()
603                            .ok()
604                    })
605                    .unwrap_or_else(|| panic!("no ErrorInfo in {status:?}"));
606            assert_eq!(error_info.reason, ErrorReason::NotServedHere.as_str_name());
607            assert_eq!(error_info.domain, ErrorReason::DOMAIN);
608            assert_eq!(
609                error_info.metadata,
610                [("cause".to_owned(), "disabled_by_config".to_owned())].into(),
611                "served at checkpoint {checkpoint}",
612            );
613        }

666
667        // Inside epoch 0: latest is epoch 0, which has not ended yet, and epoch 1 does not exist.
668        let inside_epoch_0 = ServedHeight {
669            checkpoint: 8,
670            epoch: 0,

694        assert_eq!(end_only.end, Some(timestamp_ms_to_proto(9_000)));
695
696        // Inside epoch 1: latest is epoch 1.

698            checkpoint: 10,
699            epoch: 1,
700        };
701        let latest = get(&client, inside_epoch_1, None, BOUNDS_MASK).await;
