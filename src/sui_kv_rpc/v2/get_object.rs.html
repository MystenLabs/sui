47/// The version and status an object had at the end of a checkpoint, from the
48/// `object_versions_by_checkpoint` index: its newest version written at or before the
49/// checkpoint, including a version at which it was deleted or wrapped. `None` if the object did
50/// not exist yet.
51type VersionAtCheckpoint = Option<(SequenceNumber, ObjectStatus)>;
52
53/// Serve `GetObject`. A `max_version` read counts versions up to the served height, with or
54/// without `at_checkpoint`, so a version written above the served height, whose deletion or wrap
55/// may not be indexed yet, never answers it. A read that resolves to a deletion or wrap answers
56/// `NOT_FOUND`, with an `OBJECT_REMOVED` detail where object version history is enabled (see
57/// [`respond`]). Every BigTable read goes through the request's limiter.
58pub(crate) async fn get_object(
59    server: &KvRpcServer,
60    request: tonic::Request<GetObjectRequest>,
61) -> Result<GetObjectResponse, RpcError> {
62    let served_height = ServedHeight::of(&request);
63    let mut request = request.into_inner();
64    let read_mask = request.read_mask.take();
65    let (requests, read_mask) =
66        validate_get_object_requests(std::slice::from_ref(&request), read_mask, false)?;
67    let (address, selector) = requests[0];
68    let object_id: ObjectID = address.into();
69    let client = server.limited_client("GetObject");
70    // Both checks fail the request whatever the object's state, so they come before the reads.
71    let at_checkpoint_reads = if selector.selects_version_by_bound() {
72        Some(Arc::new(
73            server.at_checkpoint_reads(client.clone(), served_height)?,
74        ))
75    } else {
76        None
77    };
78    let objects = ObjectReads {
79        client,
80        version_only: needs_only_object_id_and_version(&read_mask),
81    };
82    let answer = match plan_lookup(object_id, selector, at_checkpoint_reads.as_ref()) {
83        LookupPlan::Resolved(answer) => answer,
84        LookupPlan::NeedsScan(scan) => scan.run(&objects).await?,
85    };
86    let rows = match answer.row_key() {
87        Some(key) => objects.rows(vec![key]).await?,
88        None => HashMap::new(),
89    };
90    let found = answer.complete(&rows).await??;
91    let message = respond(
92        found.as_ref(),
93        object_id,
94        selector.not_found(address),
95        server.alpha_features.object_at_checkpoint,
96        &read_mask,
97        &server.package_resolver,
98    )
99    .await?;
100    Ok(GetObjectResponse::new(message))
101}

103/// Serve a batch as its distinct lookups, read and rendered as they become ready:
104///
105/// 1. Each lookup first learns which `objects` row answers it, if any. An exact version names its
106///    row. A latest lookup reads its object with one reverse scan of `objects`. Every other
107///    lookup reads `object_versions_by_checkpoint` (see [`AtCheckpointReads`]): a `version`
108///    lookup bounded by `at_checkpoint` names its row at once and starts its index read as it is
109///    planned, and its row counts only if the index shows the version was written by then; an
110///    `at_checkpoint` or `max_version` lookup is answered as [`bounded_read`] decides, by a live
111///    version's row, by the tombstone row of the removal the index records, or by a reverse scan
112///    of `objects` down from `max_version`. The scans run in groups of the `objects` stage's
113///    `max-seeks-per-read`, each group one read of a [`pipelined_chunks`] stage, and every group
114///    starts at once. A group's answers move on together, so the rows they name fill at most one
115///    `objects` read.
116/// 2. The rows the lookups name are read as the lookups arrive, by [`pipelined_keyed_batches`]
117///    under the `objects` stage ([`crate::config::StagesConfig::objects`]): reads of at most
118///    `max-seeks-per-read` keys, at most `concurrency` of them at a time, and each row at most
119///    once per request. A lookup whose row is missing is `NOT_FOUND`, unless the version index
120///    named the row: then the index and `objects` disagree, and the lookup is `INTERNAL`, as it
121///    is when the row is not the live object or the tombstone the index records. A lookup whose
122///    row is a tombstone is otherwise `NOT_FOUND`, with an `OBJECT_REMOVED` detail where object
123///    version history is enabled (see [`respond`]).
124/// 3. Each lookup is rendered once, at most `DEFAULT_RENDER_AHEAD` at a time (the list endpoints'
125///    default render-ahead), and every request for it gets the result.
126///
127/// Every BigTable read of the batch, scans and `objects` reads alike, shares one request-scoped
128/// limiter, and each `(object_id, checkpoint)` of the version index is read at most once per
129/// request. Rendering is not limited by it: package loads go through the server-wide
130/// [`PackageResolver`]. A lookup whose `at_checkpoint` is above the served height fails without
131/// a read, and a batch with a lookup that the instance cannot serve fails as a whole.
132///
133/// When the read mask asks for nothing but `object_id`, `version` and `display`, which kv-rpc
134/// does not render, a lookup that the version index resolves to a live version is answered by
135/// its scans alone, and the `objects` reads fetch only row keys and tombstones, not object
136/// contents.
137pub(crate) async fn batch_get_objects(
138    server: &KvRpcServer,
139    request: tonic::Request<BatchGetObjectsRequest>,
140) -> Result<BatchGetObjectsResponse, RpcError> {
141    let served_height = ServedHeight::of(&request);
142    let BatchGetObjectsRequest {
143        requests,
144        read_mask,
145        ..
146    } = request.into_inner();
147    if requests.len() > MAX_BATCH_REQUESTS {
148        return Err(RpcError::new(
149            tonic::Code::InvalidArgument,
150            format!("number of batch requests exceed limit of {MAX_BATCH_REQUESTS}"),
151        ));
152    }
153    let (requests, read_mask) = validate_get_object_requests(&requests, read_mask, true)?;
154    let version_only = needs_only_object_id_and_version(&read_mask);
155
156    // Requests for the same object state share one lookup: `request_lookups[i]` is the index in
157    // `lookups` of request `i`'s lookup.
158    let mut lookups = Vec::new();
159    let mut lookup_indexes = HashMap::new();
160    let request_lookups: Vec<usize> = requests
161        .into_iter()
162        .map(|lookup| {
163            *lookup_indexes.entry(lookup).or_insert_with(|| {
164                lookups.push(lookup);
165                lookups.len() - 1
166            })
167        })
168        .collect();
169    let lookup_count = lookups.len();
170
171    // Every read of the batch goes through this client, under one request-wide limiter.
172    let limited_client = server.limited_client("BatchGetObjects");
173    let at_checkpoint_reads = if lookups
174        .iter()
175        .any(|(_, selector)| selector.selects_version_by_bound())
176    {
177        Some(Arc::new(server.at_checkpoint_reads(
178            limited_client.clone(),
179            served_height,
180        )?))
181    } else {
182        None
183    };
184    let objects = ObjectReads {
185        client: limited_client,
186        version_only,
187    };
188
189    let mut resolved_without_scan = Vec::new();
190    let mut scans = Vec::new();
191    for (index, &(address, selector)) in lookups.iter().enumerate() {
192        match plan_lookup(address.into(), selector, at_checkpoint_reads.as_ref()) {
193            LookupPlan::Resolved(answer) => resolved_without_scan.push((index, answer)),
194            LookupPlan::NeedsScan(scan) => scans.push((index, scan)),
195        }
196    }
197
198    // A group's drainer task runs its scans at once and to their ends, so each scan returns its
199    // permit whether or not the `objects` reads below are taking lookups. Every group starts at
200    // once: the request limiter alone bounds how many scans reach BigTable.
201    let objects_stage = server.stages.stage(PipelineStage::Objects);
202    let scans_per_group = objects_stage.max_seeks_per_read;
203    let scan_groups = scans.len().div_ceil(scans_per_group).max(1);
204    let scans: BoxStream<'static, Result<Watermarked<(usize, Scan)>, RpcError>> =
205        stream::iter(scans.into_iter().map(|scan| Ok(Watermarked::Item(scan)))).boxed();
206    let scanned = pipelined_chunks(scans, scans_per_group, scan_groups, None, SeekUnit::Item, {
207        let objects = objects.clone();
208        move |group: Vec<(usize, Scan)>| {
209            let objects = objects.clone();
210            async move {
211                let answers = future::try_join_all(group.into_iter().map(|(index, scan)| {
212                    let objects = &objects;
213                    async move { Ok::<_, RpcError>((index, scan.run(objects).await?)) }
214                }))
215                .await?;
216
217                Ok::<_, RpcError>(stream::iter(answers.into_iter().map(Ok)).boxed())
218            }
219        }
220    });
221    // Exact versions go out in key order, so an `objects` read covers one run of adjacent keys.
222    resolved_without_scan.sort_by_key(|(_, answer)| answer.row_key());
223    let resolved_without_scan = stream::iter(
224        resolved_without_scan
225            .into_iter()
226            .map(|answer| Ok(Watermarked::Item(answer))),
227    )
228    .boxed();
229    let lookups_with_keys: MarkedKeyedUpstream<(usize, ObjectsStageInput), ObjectKey, RpcError> =
230        stream::select(resolved_without_scan, scanned)
231            .map_ok(|frame| {
232                frame.map_item(|(index, answer)| {
233                    let keys = answer.row_keys();
234                    ((index, answer), keys)
235                })
236            })
237            .boxed();
238
239    let fetcher: Arc<dyn CacheFetcher<ObjectKey, Found>> = Arc::new(objects);
240    let answered: BoxStream<'static, Result<ResolvedWatermarked<LookupWithRows>, RpcError>> =
241        pipelined_keyed_batches(
242            lookups_with_keys,
243            objects_stage.max_seeks_per_read,
244            objects_stage.concurrency,
245            objects_stage.min_seeks_per_split_read,
246            fetcher,
247            MissingRow::Omit,
248        )
249        // The lookups carry no watermarks.
250        .try_filter_map(|frame| {
251            future::ready(Ok(match frame {
252                Watermarked::Item(item) => Some(ResolvedWatermarked::Item(item)),
253                Watermarked::Watermark(_) => None,
254            }))
255        })
256        .boxed();
257
258    let mut rendered = render_ahead(answered, DEFAULT_RENDER_AHEAD, {
259        let read_mask = Arc::new(read_mask);
260        let resolver = server.package_resolver.clone();
261        let removal_details = server.alpha_features.object_at_checkpoint;
262        move |((index, answer), rows)| {
263            let (address, selector) = lookups[index];
264            let read_mask = read_mask.clone();
265            let resolver = resolver.clone();
266            async move {
267                // A failed read fails the whole batch; the lookup's own answer, error or not,
268                // goes to its requests.
269                let result = match answer.complete(&rows).await? {
270                    Ok(found) => respond(
271                        found.as_ref(),
272                        address.into(),
273                        selector.not_found(address),
274                        removal_details,
275                        &read_mask,
276                        &resolver,
277                    )
278                    .await
279                    .map(GetObjectResult::new_object),
280                    Err(err) => Err(err),
281                };
282                let result = result
283                    .unwrap_or_else(|err| GetObjectResult::new_error(err.into_status_proto()));
284                Ok::<_, RpcError>((index, result))
285            }
286        }
287    });
288
289    let mut results: Vec<Option<GetObjectResult>> = vec![None; lookup_count];
290    while let Some(frame) = rendered.next().await {
291        match frame {
292            Ok(ResolvedWatermarked::Item((index, result))) => results[index] = Some(result),
293            Ok(ResolvedWatermarked::Watermark { .. }) => {}
294            Err(RenderAheadError::Upstream(err) | RenderAheadError::Render(err)) => {
295                return Err(err);
296            }
297        }
298    }
299
300    // A lookup's result goes to every request for it; the last of those requests takes it.
301    let mut requests_left = vec![0usize; lookup_count];
302    for &lookup in &request_lookups {
303        requests_left[lookup] += 1;
304    }
305    let objects = request_lookups
306        .into_iter()
307        .map(|lookup| {
308            requests_left[lookup] -= 1;
309            let result = if requests_left[lookup] == 0 {
310                results[lookup].take()
311            } else {
312                results[lookup].clone()
313            };
314            result.ok_or_else(|| {
315                RpcError::new(
316                    tonic::Code::Internal,
317                    "BatchGetObjects: a lookup ended without a result",
318                )
319            })
320        })
321        .collect::<Result<Vec<_>, _>>()?;
322    Ok(BatchGetObjectsResponse::new(objects))
323}

325/// The bounds of an `at_checkpoint` or `max_version` read: it selects the object's newest version
326/// at or below `max_version` written by the end of checkpoint `height`, and finds nothing if that
327/// version was written after `checkpoint`, or deleted or wrapped the object.
328#[derive(Clone, Copy, Debug)]
329struct Bounds {
330    max_version: u64,
331    checkpoint: u64,
332    height: u64,
333}
334
335/// How an `at_checkpoint` or `max_version` read is answered. See [`bounded_read`].
336#[derive(Clone, Copy, Debug)]
337enum BoundedRead {
338    /// No version answers it.
339    NotFound,
340    /// The object's entry at `version`, its version at the end of the checkpoint, answers it: the
341    /// object if `status` is `Live`, else the removal's tombstone row in the `objects` table.
342    Indexed {
343        version: SequenceNumber,
344        status: ObjectStatus,
345    },
346    /// The newest version at or below `from`, from an `objects` reverse scan, answers it. `from`
347    /// is below the object's version at the end of the checkpoint, so every version the scan can
348    /// find was written by then.
349    ScanBelow(SequenceNumber),
350    /// The object's entry at the end of the checkpoint is at `version`, with `status`, and the
351    /// object changed again before the height. The newest version at or below `from`, from an
352    /// `objects` reverse scan, is selected: one above `version` was written after the checkpoint,
353    /// so nothing answers; otherwise it must be the index's entry (see [`indexed_entry`]), which
354    /// answers.
355    ScanToIndexed {
356        from: SequenceNumber,
357        version: SequenceNumber,
358        status: ObjectStatus,
359    },
360}
361
362/// How a read with [`Bounds`] is answered, given the object's [`VersionAtCheckpoint`] at
363/// `checkpoint` and at `height`. An object's versions increase in checkpoint order, so a version
364/// above its version at the end of a checkpoint was written after it.
365///
366/// Below the version at the checkpoint, every version at or below `max_version` was written by
367/// the checkpoint, so a reverse scan from `max_version` answers. Otherwise, if the object did not
368/// change between the checkpoint and the height, the version index alone decides the answer: the
369/// object if it is live at its version at the checkpoint, and its removal if it was deleted or
370/// wrapped there. If it did change and `max_version` reaches its version at the height, that newer
371/// version is selected, and nothing answers. Below it, a reverse scan from `max_version` answers
372/// only if it finds the version at the checkpoint itself. Every version the scans can reach is at
373/// or below the version at the height, so versions written above the height never count.
374fn bounded_read(
375    max_version: u64,
376    version_at_checkpoint: VersionAtCheckpoint,
377    version_at_height: VersionAtCheckpoint,
378) -> BoundedRead {
379    let Some((at_checkpoint, status)) = version_at_checkpoint else {
380        // Every version of the object was written after the checkpoint.
381        return BoundedRead::NotFound;
382    };
383    let from = max_version.into();
384    if max_version < at_checkpoint.value() {
385        return BoundedRead::ScanBelow(from);
386    }
387    match version_at_height {
388        Some((at_height, _)) if at_height > at_checkpoint => {
389            if max_version >= at_height.value() {
390                BoundedRead::NotFound
391            } else {
392                BoundedRead::ScanToIndexed {
393                    from,
394                    version: at_checkpoint,
395                    status,
396                }
397            }
398        }
399        _ => BoundedRead::Indexed {
400            version: at_checkpoint,
401            status,
402        },
403    }
404}

406/// The entry the version index records at `key`, with `status`, from `row`, the `objects` row a
407/// read found for it: the live object at `key`'s version if `status` is `Live`, else the tombstone
408/// of a removal of that kind at that version. Anything else, no row included, means the index and
409/// `objects` diverged, which is `INTERNAL`.
410fn indexed_entry(
411    key: ObjectKey,
412    status: ObjectStatus,
413    row: Option<&Found>,
414) -> Result<Found, RpcError> {
415    let agrees = |found: &Found| match found {
416        Found::Object(_) | Found::Version(_) => status == ObjectStatus::Live,
417        Found::Removed(removal) => {
418            status != ObjectStatus::Live && removal.kind == removal_kind(status)
419        }
420    };
421    match row {
422        Some(found) if found.version() == key.1 && agrees(found) => Ok(found.clone()),
423        _ => Err(RpcError::new(
424            tonic::Code::Internal,
425            format!(
426                "object_versions_by_checkpoint names {key:?} as {status:?}, but objects has no \
427                 such row"
428            ),
429        )),
430    }
431}

433/// The kind of removal a tombstone or version index row with `status` records.
434pub(super) fn removal_kind(status: ObjectStatus) -> ObjectRemovalKind {
435    match status {
436        ObjectStatus::Deleted => ObjectRemovalKind::Deleted,
437        ObjectStatus::Wrapped => ObjectRemovalKind::Wrapped,
438        ObjectStatus::Live => unreachable!("removals are deleted or wrapped"),
439    }
440}

442/// An object as a read found it: the whole object, or, for a read mask that needs only
443/// `object_id` and `version`, just its version; or the removal a tombstone row records.
444#[derive(Clone)]
445enum Found {
446    Object(Arc<Object>),
447    Version(SequenceNumber),
448    Removed(ObjectRemoval),
449}
450
451impl Found {
452    /// The `objects` row at `version`.
453    fn from_row(version: SequenceNumber, row: ObjectVersionRow) -> Self {
454        match row {
455            ObjectVersionRow::Live(Some(object)) => Found::Object(Arc::new(object)),
456            ObjectVersionRow::Live(None) => Found::Version(version),
457            ObjectVersionRow::Tombstone(ObjectTombstone { status, .. }) => {
458                Found::Removed(ObjectRemoval {
459                    version: version.value(),
460                    kind: removal_kind(status),
461                })
462            }
463        }
464    }
465
466    fn version(&self) -> SequenceNumber {
467        match self {
468            Found::Object(object) => object.version(),
469            Found::Version(version) => *version,
470            Found::Removed(removal) => SequenceNumber::from_u64(removal.version),
471        }
472    }
473}

475/// The answer to a read of `object_id` that found `found`: the live version rendered with
476/// `read_mask`, or `not_found`. With `removal_details`, object version history is enabled
477/// (`enable-object-at-checkpoint`), the operator's statement that every tombstone is backfilled:
478/// a read that resolved to a deletion or wrap adds an `OBJECT_REMOVED` detail. Without it,
479/// tombstones may be missing, so no read describes a removal.
480async fn respond(
481    found: Option<&Found>,
482    object_id: ObjectID,
483    not_found: ObjectNotFoundError,
484    removal_details: bool,
485    read_mask: &FieldMaskTree,
486    resolver: &PackageResolver,
487) -> Result<ProtoObject, RpcError> {
488    match found {
489        Some(Found::Object(object)) => Ok(object_to_response(object, read_mask, resolver).await),
490        Some(Found::Version(version)) => Ok(object_version_to_response(object_id, *version)),
491        Some(Found::Removed(_)) if !removal_details => Err(not_found.into()),
492        Some(Found::Removed(removal)) => Err(not_found.removed(*removal).into()),
493        None => Err(not_found.into()),
494    }
495}

497/// A batch lookup after its `objects` reads: its index in the batch's lookups, its
498/// [`ObjectsStageInput`], and the rows of the keys it listed.
499type LookupWithRows = KeyedBatchOutput<(usize, ObjectsStageInput), ObjectKey, Found>;

501/// How a lookup is answered before any `objects` multi-get: resolved up to one `objects` row (or
502/// already failed), possibly with a version index read already running, or only after a scan.
503enum LookupPlan {
504    Resolved(ObjectsStageInput),
505    NeedsScan(Scan),
506}
507
508/// Plans the lookup of `object_id` by `selector`. A selector that picks its version by a bound
509/// reads the version index through `at_checkpoint_reads`, and fails without a read if its
510/// `at_checkpoint` is above the served height.
511fn plan_lookup(
512    object_id: ObjectID,
513    selector: ObjectSelector,
514    at_checkpoint_reads: Option<&Arc<AtCheckpointReads>>,
515) -> LookupPlan {
516    match selector {
517        ObjectSelector::Latest => LookupPlan::NeedsScan(Scan::Latest(object_id)),
518        ObjectSelector::Version {
519            version,
520            at_checkpoint: None,
521        } => LookupPlan::Resolved(ObjectsStageInput::RequestedRow(ObjectKey(
522            object_id,
523            version.into(),
524        ))),
525        ObjectSelector::Version {
526            at_checkpoint: Some(_),
527            ..
528        }
529        | ObjectSelector::MaxVersion { .. }
530        | ObjectSelector::AtCheckpoint(_) => at_checkpoint_reads
531            .ok_or_else(|| {
532                RpcError::new(
533                    tonic::Code::Internal,
534                    "a read that picks its version by a bound has no version index reads",
535                )
536            })
537            .and_then(|reads| reads.plan(object_id, selector))
538            .unwrap_or_else(|err| LookupPlan::Resolved(ObjectsStageInput::Failed(err))),
539    }
540}

542/// A scan that a lookup starts with, before any `objects` multi-get.
543enum Scan {
544    /// The newest version of the object, from one reverse scan of `objects`.
545    Latest(ObjectID),
546    /// An `at_checkpoint` or `max_version` read, answered as [`bounded_read`] decides.
547    Bounded {
548        reads: Arc<AtCheckpointReads>,
549        object_id: ObjectID,
550        bounds: Bounds,
551    },
552}
553
554impl Scan {
555    async fn run(self, objects: &ObjectReads) -> Result<ObjectsStageInput, RpcError> {
556        Ok(match self {
557            Scan::Latest(object_id) => ObjectsStageInput::Scanned(objects.latest(object_id).await?),
558            Scan::Bounded {
559                reads,
560                object_id,
561                bounds,
562            } => {
563                // Both reads of the version index start at once. The version at the height
564                // decides nothing when the object did not exist at the checkpoint or `max_version`
565                // is below its version there; reading it anyway costs a read in those cases and
566                // saves a round trip in the others.
567                let (version_at_checkpoint, version_at_height) =
568                    if bounds.height == bounds.checkpoint {
569                        let version = reads
570                            .version_at_checkpoint(object_id, bounds.checkpoint)
571                            .await?;
572                        (version, version)
573                    } else {
574                        future::try_join(
575                            reads.version_at_checkpoint(object_id, bounds.checkpoint),
576                            reads.version_at_checkpoint(object_id, bounds.height),
577                        )
578                        .await?
579                    };
580                match bounded_read(bounds.max_version, version_at_checkpoint, version_at_height) {
581                    BoundedRead::NotFound => ObjectsStageInput::Scanned(None),
582                    // A live version's row is in the objects table, so the index answers a
583                    // version-only read without touching it.
584                    BoundedRead::Indexed {
585                        version,
586                        status: ObjectStatus::Live,
587                    } if objects.version_only => {
588                        ObjectsStageInput::Scanned(Some(Found::Version(version)))
589                    }
590                    BoundedRead::Indexed { version, status } => ObjectsStageInput::Indexed {
591                        key: ObjectKey(object_id, version),
592                        status,
593                    },
594                    BoundedRead::ScanBelow(from) => ObjectsStageInput::Scanned(
595                        objects
596                            .newest(object_id, from, !objects.version_only)
597                            .await?,
598                    ),
599                    BoundedRead::ScanToIndexed {
600                        from,
601                        version,
602                        status,
603                    } => {
604                        // Only a live entry answers with contents, so a scan for a removal skips
605                        // live objects' contents.
606                        let with_contents = status == ObjectStatus::Live && !objects.version_only;
607                        match objects.newest(object_id, from, with_contents).await? {
608                            Some(found) if found.version() > version => {
609                                ObjectsStageInput::Scanned(None)
610                            }
611                            found => match indexed_entry(
612                                ObjectKey(object_id, version),
613                                status,
614                                found.as_ref(),
615                            ) {
616                                Ok(entry) => ObjectsStageInput::Scanned(Some(entry)),
617                                Err(err) => ObjectsStageInput::Failed(err),
618                            },
619                        }
620                    }
621                }
622            }
623        })
624    }

627/// What a lookup carries into its `objects` reads: its answer as far as it is known before
628/// them. A batch hands it to the `objects` stage; `GetObject` reads its row with
629/// `ObjectReads::rows`.
630enum ObjectsStageInput {
631    /// The `objects` row at the version the request asked for answers the lookup; without the
632    /// row it is `NOT_FOUND`.
633    RequestedRow(ObjectKey),
634    /// A `version` lookup bounded by a checkpoint: the `objects` row at `key` answers it if
635    /// `version_at_checkpoint`, the object's [`VersionAtCheckpoint`] there, read alongside the
636    /// row, shows that the version was written by the end of the checkpoint. When that version is
637    /// `key`'s, the row must be the index's entry (see [`indexed_entry`]).
638    WrittenBy {
639        key: ObjectKey,
640        version_at_checkpoint: CachedValue<(SequenceNumber, ObjectStatus)>,
641    },
642    /// The `objects` row at this key, which the version index names as live or as a removal, as
643    /// `status` says, answers the lookup (see [`indexed_entry`]).
644    Indexed {
645        key: ObjectKey,
646        status: ObjectStatus,
647    },
648    /// The lookup's scan answered it; `None` makes it `NOT_FOUND`.
649    Scanned(Option<Found>),
650    /// The lookup fails with this error, without a read.
651    Failed(RpcError),
652}
653
654impl ObjectsStageInput {
655    /// The `objects` row the lookup still needs, if any.
656    fn row_key(&self) -> Option<ObjectKey> {
657        match self {
658            ObjectsStageInput::RequestedRow(key)
659            | ObjectsStageInput::WrittenBy { key, .. }
660            | ObjectsStageInput::Indexed { key, .. } => Some(*key),
661            ObjectsStageInput::Scanned(_) | ObjectsStageInput::Failed(_) => None,
662        }
663    }
664
665    /// The `objects` rows the lookup still needs.
666    fn row_keys(&self) -> Vec<ObjectKey> {
667        self.row_key().into_iter().collect()
668    }
669
670    /// The lookup's answer, given the rows the `objects` reads returned for its keys: the outer
671    /// error is a failed read, which fails the request (a whole batch); the inner result is the
672    /// lookup's own answer, where `Ok(None)` means `NOT_FOUND`.
673    async fn complete(
674        self,
675        rows: &HashMap<ObjectKey, Found>,
676    ) -> Result<Result<Option<Found>, RpcError>, RpcError> {
677        Ok(match self {
678            ObjectsStageInput::RequestedRow(key) => Ok(rows.get(&key).cloned()),
679            ObjectsStageInput::WrittenBy {
680                key,
681                version_at_checkpoint,
682            } => match version_at_checkpoint.await? {
683                // The index records the requested version, so its row must agree with the index,
684                // as for an `at_checkpoint` read alone.
685                Some((version, status)) if version == key.1 => {
686                    indexed_entry(key, status, rows.get(&key)).map(Some)
687                }
688                // An object's versions increase in checkpoint order, so a version below its
689                // version at the end of the checkpoint was written by then.
690                Some((version, _)) if key.1 < version => Ok(rows.get(&key).cloned()),
691                // Written after the checkpoint, or the object did not exist yet.
692                _ => Ok(None),
693            },
694            ObjectsStageInput::Indexed { key, status } => {
695                indexed_entry(key, status, rows.get(&key)).map(Some)
696            }
697            ObjectsStageInput::Scanned(found) => Ok(found),
698            ObjectsStageInput::Failed(err) => Err(err),
699        })
700    }
701}

703/// Reads of the `objects` table through a request's gated client, live rows and tombstones
704/// alike: whole objects, or, with `version_only`, just the versions of the live ones, read
705/// without the objects' contents. A tombstone reads as the removal it records.
706#[derive(Clone)]
707struct ObjectReads {
708    client: BigTableClient,
709    version_only: bool,
710}
711
712impl ObjectReads {
713    /// The newest row of `object_id`.
714    async fn latest(&self, object_id: ObjectID) -> Result<Option<Found>, RpcError> {
715        self.newest(object_id, SequenceNumber::MAX, !self.version_only)
716            .await
717    }
718
719    /// The rows at the versions `keys` name, in one read; a key without a row has no entry.
720    async fn rows(&self, keys: Vec<ObjectKey>) -> Result<HashMap<ObjectKey, Found>, RpcError> {
721        // The read runs to its end, so it finishes as completed rather than cut off.
722        self.fetch(keys).await?.try_collect().await
723    }
724
725    /// The newest row of `object_id` at or below `version`, from one reverse scan of `objects`,
726    /// with live objects' contents if `with_contents`.
727    async fn newest(
728        &self,
729        object_id: ObjectID,
730        version: SequenceNumber,
731        with_contents: bool,
732    ) -> Result<Option<Found>, RpcError> {
733        // The scan runs to its end, so it finishes as completed rather than cut off.
734        let rows: Vec<_> = self
735            .client
736            .object_versions_stream(
737                object_id,
738                SequenceNumber::MIN..=version,
739                1,
740                true,
741                with_contents,
742            )
743            .await?
744            .try_collect()
745            .await?;
746        Ok(rows
747            .into_iter()
748            .next()
749            .map(|(version, row)| Found::from_row(version, row)))
750    }
751}

772pub(super) struct AtCheckpointReads {

777impl KvRpcServer {

796            versions: RequestCache::new(Arc::new(VersionIndexReads { client })),
797        })
798    }

801impl AtCheckpointReads {
802    /// The plan of a lookup that picks its version by a bound: a `version` bounded by
803    /// `at_checkpoint`, a `max_version`, or an `at_checkpoint` alone. Fails if the lookup's
804    /// `at_checkpoint` is above the served height.
805    ///
806    /// A `version` bounded by `at_checkpoint` names its row at once, and its version index read
807    /// starts here, so it runs alongside the row's read. A `max_version` read counts versions up
808    /// to the served height, and its `at_checkpoint` defaults to it. An `at_checkpoint` read is a
809    /// `max_version` read with no bound on the version, counting versions up to the checkpoint.
810    fn plan(
811        self: &Arc<Self>,
812        object_id: ObjectID,
813        selector: ObjectSelector,
814    ) -> Result<LookupPlan, RpcError> {
815        let served_checkpoint = self.served_height.checkpoint;
816        if let Some(checkpoint) = selector.at_checkpoint() {
817            AtCheckpointUpperBound::new(served_checkpoint).check(checkpoint)?;
818        }
819        let reads = self.clone();
820        let scan = match selector {
821            ObjectSelector::Version {
822                version,
823                at_checkpoint: Some(checkpoint),
824            } => {
825                return Ok(LookupPlan::Resolved(ObjectsStageInput::WrittenBy {
826                    key: ObjectKey(object_id, version.into()),
827                    version_at_checkpoint: self.version_at_checkpoint(object_id, checkpoint),
828                }));
829            }
830            ObjectSelector::MaxVersion {
831                max_version,
832                at_checkpoint,
833            } => Scan::Bounded {
834                reads,
835                object_id,
836                bounds: Bounds {
837                    max_version,
838                    checkpoint: at_checkpoint.unwrap_or(served_checkpoint),
839                    height: served_checkpoint,
840                },
841            },
842            ObjectSelector::AtCheckpoint(checkpoint) => Scan::Bounded {
843                reads,
844                object_id,
845                bounds: Bounds {
846                    max_version: u64::MAX,
847                    checkpoint,
848                    height: checkpoint,
849                },
850            },
851            ObjectSelector::Latest
852            | ObjectSelector::Version {
853                at_checkpoint: None,
854                ..
855            } => {
856                return Err(RpcError::new(
857                    tonic::Code::Internal,
858                    "not an at_checkpoint or max_version read",
859                ));
860            }
861        };
862        Ok(LookupPlan::NeedsScan(scan))
863    }
864
865    /// The version and status `object_id` had at the end of `checkpoint`. The first call for an
866    /// `(object_id, checkpoint)` starts reading it at once, in a task the cache owns, so the read
867    /// runs and returns its permit whether or not the returned future is polled; later calls share
868    /// that read.
869    pub(super) fn version_at_checkpoint(
870        &self,
871        object_id: ObjectID,
872        checkpoint: u64,
873    ) -> CachedValue<(SequenceNumber, ObjectStatus)> {
874        let key = (object_id, checkpoint);
875        let mut reservation = self.versions.reserve(&[key]);
876        if reservation.is_unread(&key) {
877            // The read's end needs no wait: its value arrives through the reservation.
878            let _read_end = self.versions.read(reservation.take_unread(&[key]));
879        }
880        reservation.value(&key)
881    }
882}

885/// `(object_id, checkpoint)`, because BigTable cannot return the newest row of several ranges in
886/// one `ReadRows` (`rows_limit` covers the whole request).
887struct VersionIndexReads {
888    client: BigTableClient,
889}
890
891#[async_trait::async_trait]
892impl CacheFetcher<(ObjectID, u64), (SequenceNumber, ObjectStatus)> for VersionIndexReads {
893    async fn fetch(
894        &self,
895        keys: Vec<(ObjectID, u64)>,
896    ) -> Result<
897        BoxStream<'static, Result<((ObjectID, u64), (SequenceNumber, ObjectStatus)), RpcError>>,
898        RpcError,
899    > {

908                    Ok::<_, RpcError>(version.map(|version| ((object_id, checkpoint), version)))
909                }
910            })
911            .try_filter_map(|row| future::ready(Ok(row)))
912            .boxed())
913    }
