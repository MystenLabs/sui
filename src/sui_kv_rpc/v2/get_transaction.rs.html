31pub const READ_MASK_DEFAULT: &str = sui_rpc_api::read_mask_defaults::TRANSACTION;

33pub(crate) fn validate_read_mask(read_mask: Option<FieldMask>) -> Result<FieldMaskTree, RpcError> {
34    let read_mask = read_mask.unwrap_or_else(|| FieldMask::from_str(READ_MASK_DEFAULT));
35    read_mask
36        .validate::<ExecutedTransaction>()
37        .map_err(|path| {
38            FieldViolation::new("read_mask")
39                .with_description(format!("invalid read_mask path: {path}"))
40                .with_reason(ErrorReason::FieldInvalid)
41        })?;
42    Ok(FieldMaskTree::from(read_mask))
43}

45/// Serve a transaction whose checkpoint is at or below `served_height`. One above it is `NotFound`,
46/// as if not yet written: its `transactions` row can already exist while the `objects` rows a read
47/// mask with objects joins are still being written.
48pub async fn get_transaction(
49    client: BigTableClient,
50    stages: &StagesConfig,
51    served_height: ServedHeight,
52    request: GetTransactionRequest,
53    resolver: &PackageResolver,
54) -> Result<GetTransactionResponse, RpcError> {
55    let transaction_digest = request
56        .digest
57        .ok_or_else(|| {
58            FieldViolation::new("digest")
59                .with_description("missing digest")
60                .with_reason(ErrorReason::FieldMissing)
61        })?
62        .parse::<TransactionDigest>()
63        .map_err(|e| {
64            FieldViolation::new("digest")
65                .with_description(format!("invalid digest: {e}"))
66                .with_reason(ErrorReason::FieldInvalid)
67        })?;
68
69    let read_mask = validate_read_mask(request.read_mask)?;
70
71    let transactions_stage = stages.stage(PipelineStage::Transactions);
72    let objects_stage = stages.stage(PipelineStage::Objects);
73    // Read to the end rather than stopping at the first item, so no read is dropped before
74    // BigTable ends its response, which would cancel it and reset its HTTP/2 stream.
75    let (_, transaction, objects) = resolve::resolve_transactions(
76        client,
77        vec![transaction_digest],
78        served_height.checkpoint,
79        &read_mask,
80        transactions_stage,
81        objects_stage,
82    )
83    .try_collect::<Vec<_>>()
84    .await?
85    .pop()
86    .ok_or(TransactionNotFoundError(transaction_digest.into()))?;
87    Ok(GetTransactionResponse::new(
88        transaction_to_response(transaction, &read_mask, objects.as_ref(), resolver).await?,
89    ))
90}

92/// [`get_transaction`] for each digest, answering per digest, in request order. Requests for the
93/// same digest share one lookup: each distinct digest is read once and rendered once, and every
94/// request for it gets the result. Lookups are rendered as their reads complete, at most
95/// `DEFAULT_RENDER_AHEAD` at a time (the list endpoints' default render-ahead), while later reads
96/// run. A digest without a row, or above the served height, is `NOT_FOUND`, and a render error
97/// answers that digest's requests; a failed read fails the whole batch. A malformed digest is an
98/// `INVALID_ARGUMENT` for its own request, naming the field `digests[<index>]`, and is not read.
99pub async fn batch_get_transactions(
100    client: BigTableClient,
101    stages: &StagesConfig,
102    served_height: ServedHeight,
103    BatchGetTransactionsRequest {
104        digests, read_mask, ..
105    }: BatchGetTransactionsRequest,
106    resolver: &PackageResolver,
107) -> Result<BatchGetTransactionsResponse, RpcError> {
108    let read_mask = validate_read_mask(read_mask)?;
109
110    if digests.len() > MAX_BATCH_REQUESTS {
111        return Err(RpcError::new(
112            tonic::Code::InvalidArgument,
113            format!("number of batch requests exceed limit of {MAX_BATCH_REQUESTS}"),
114        ));
115    }
116
117    let digests = digests
118        .into_iter()
119        .enumerate()
120        .map(|(index, digest)| {
121            digest.parse::<TransactionDigest>().map_err(|e| {
122                RpcError::from(
123                    FieldViolation::new_at("digests", index)
124                        .with_description(format!("invalid digest: {e}"))
125                        .with_reason(ErrorReason::FieldInvalid),
126                )
127            })
128        })
129        .collect::<Vec<_>>();
130
131    // Requests for the same digest share one lookup: `lookups` holds each distinct digest once,
132    // in order of its first request, and `requests_left` counts the requests for each.
133    let mut requests_left: HashMap<TransactionDigest, usize> = HashMap::new();
134    let mut lookups = Vec::new();
135    for &digest in digests.iter().flatten() {
136        *requests_left.entry(digest).or_insert_with(|| {
137            lookups.push(digest);
138            0
139        }) += 1;
140    }
141
142    let transactions_stage = stages.stage(PipelineStage::Transactions);
143    let objects_stage = stages.stage(PipelineStage::Objects);
144    let resolved: BoxStream<'static, Result<ResolvedWatermarked<ResolvedTransaction>, RpcError>> =
145        resolve::resolve_transactions(
146            client,
147            lookups,
148            served_height.checkpoint,
149            &read_mask,
150            transactions_stage,
151            objects_stage,
152        )
153        .map_ok(ResolvedWatermarked::Item)
154        .boxed();
155
156    let mut rendered = render_ahead(resolved, DEFAULT_RENDER_AHEAD, {
157        let read_mask = Arc::new(read_mask);
158        let resolver = resolver.clone();
159        move |(digest, transaction, objects): ResolvedTransaction| {
160            let read_mask = read_mask.clone();
161            let resolver = resolver.clone();
162            async move {
163                let result =
164                    match transaction_to_response(transaction, &read_mask, &objects, &resolver)
165                        .await
166                    {
167                        Ok(transaction) => GetTransactionResult::new_transaction(transaction),
168                        Err(err) => GetTransactionResult::new_error(err.into_status_proto()),
169                    };
170                Ok::<_, Infallible>((digest, result))
171            }
172        }
173    });
174
175    let mut results: HashMap<TransactionDigest, GetTransactionResult> = HashMap::new();
176    while let Some(frame) = rendered.next().await {
177        match frame {
178            // Rows arrive in no particular order, so results are kept by digest.
179            Ok(ResolvedWatermarked::Item((digest, result))) => {
180                if requests_left.contains_key(&digest) {
181                    results.insert(digest, result);
182                }
183            }
184            Ok(ResolvedWatermarked::Watermark { .. }) => {}
185            Err(RenderAheadError::Upstream(err)) => return Err(err),
186            Err(RenderAheadError::Render(never)) => match never {},
187        }
188    }
189
190    // A digest's result goes to every request for it; the last of those requests takes it. A
191    // digest without a result had no row, or its checkpoint is above the served height.
192    let transactions = digests
193        .into_iter()
194        .map(|digest| {
195            let digest = match digest {
196                Ok(digest) => digest,
197                Err(err) => return GetTransactionResult::new_error(err.into_status_proto()),
198            };
199            let last_request = requests_left.get_mut(&digest).is_some_and(|left| {
200                *left -= 1;
201                *left == 0
202            });
203            let result = if last_request {
204                results.remove(&digest)
205            } else {
206                results.get(&digest).cloned()
207            };
208            result.unwrap_or_else(|| {
209                let err: RpcError = TransactionNotFoundError(digest.into()).into();
210                GetTransactionResult::new_error(err.into_status_proto())
211            })
212        })
213        .collect();
214    Ok(BatchGetTransactionsResponse::new(transactions))
215}

634        .expect("one-permit BatchGetTransactions should succeed");
635        assert_batch_objects(one_permit_response, &checkpoint, false);
636
637        one_permit_server.abort();
638        server.abort();
639    }

641    /// A malformed digest is an `INVALID_ARGUMENT` for its own request, naming the field
642    /// `digests[<index>]`, and is not read; the other digests are answered in request order.
643    #[tokio::test]
644    async fn batch_get_transactions_malformed_digest_fails_only_its_request() {
645        use sui_rpc::proto::google::rpc::BadRequest;
646
647        let (ctx, _registry, mock, server) =
648            query_context_with_mock_and_registry("batch_get_transactions", 1).await;
649        let checkpoint = two_transaction_object_checkpoint();
650        insert_checkpoint_rows(&mock, &checkpoint).await;
651        let stages = StagesConfig::default();
652        let first_digest = checkpoint.transactions[0].transaction.digest();
653        let second_digest = checkpoint.transactions[1].transaction.digest();
654
655        mock.clear_read_rows_calls().await;
656        let mut batch_request = BatchGetTransactionsRequest::default();
657        batch_request.digests = vec![
658            first_digest.to_string(),
659            "not-a-digest-0OIl".to_string(),
660            second_digest.to_string(),
661        ];
662        batch_request.read_mask = Some(FieldMask::from_paths(["digest"]));
663        let results = batch_get_transactions(
664            ctx.client().clone(),
665            &stages,
666            UNBOUNDED_SERVED_HEIGHT,
667            batch_request,
668            ctx.package_resolver(),
669        )
670        .await
671        .expect("BatchGetTransactions succeeds")
672        .transactions;
673        let [first, malformed, second] = <[_; 3]>::try_from(results).expect("one per digest");
674
675        let first = first.to_result().expect("valid digest is found");
676        assert_eq!(first.digest, Some(first_digest.to_string()));
677        let second = second.to_result().expect("valid digest is found");
678        assert_eq!(second.digest, Some(second_digest.to_string()));
679
680        let status = malformed
681            .to_result()
682            .expect_err("malformed digest should return an error");
683        assert_eq!(status.code, tonic::Code::InvalidArgument as i32);
684        let violations: Vec<_> = status
685            .details
686            .iter()
687            .filter_map(|detail| detail.to_msg::<BadRequest>().ok())
688            .flat_map(|bad_request| bad_request.field_violations)
689            .collect();
690        let [violation] = <[_; 1]>::try_from(violations).expect("one field violation");
691        assert_eq!(violation.field, "digests[1]");
692        assert!(
693            violation.description.starts_with("invalid digest: "),
694            "unexpected description: {}",
695            violation.description
696        );
697
698        let mut read_transaction_rows = mock
699            .read_rows_calls()
700            .await
701            .into_iter()
702            .filter(|call| call.table == sui_kvstore::tables::transactions::NAME)
703            .flat_map(|call| call.row_keys)
704            .map(|key| key.to_vec())
705            .collect::<Vec<_>>();
706        read_transaction_rows.sort();
707        let mut expected_transaction_rows = [first_digest, second_digest]
708            .iter()
709            .map(sui_kvstore::tables::transactions::encode_key)
710            .collect::<Vec<_>>();
711        expected_transaction_rows.sort();
712        assert_eq!(
713            read_transaction_rows, expected_transaction_rows,
714            "only the well-formed digests are read"
715        );
716
717        server.abort();
718    }
