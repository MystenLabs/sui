37/// Serve `ListObjectVersions` from the `objects` table, listing live rows as live versions and
38/// tombstones as removals. Every listing is bounded by a checkpoint: `end_checkpoint - 1`, or the
39/// served height when `end_checkpoint` is unset or above it (see
40/// [`ObjectVersionsQuery::checkpoint_window`]). `objects` rows above the served height can be
41/// written out of checkpoint order, so a listing past it could list a version and skip an older
42/// one still being written, which a client paging on from the newer version's cursor would never
43/// see. The object's versions at the end of that checkpoint and of the one before
44/// `start_checkpoint`, read at once, the request's version range, and its cursors fold into one
45/// inclusive version range (see [`ObjectVersionsQuery::plan`]), read as one range scan of up to
46/// `limit` rows whose items stream to the client as they arrive. A read mask that needs only
47/// `object_id` and `version` (see [`needs_only_object_id_and_version`]) reads live rows' keys
48/// alone; any other is rendered from the objects, `render-ahead` items at a time.
49///
50/// The checkpoint bounds read the `object_versions_by_checkpoint` index, so every request needs
51/// `enable-object-at-checkpoint`, which also requires the tombstones that list removals to be
52/// backfilled from genesis. kv-rpc keeps every version, so any `start_checkpoint` and
53/// `end_checkpoint` up to the served height is served.
54pub(crate) async fn list_object_versions(
55    server: &KvRpcServer,
56    request: tonic::Request<ListObjectVersionsRequest>,
57) -> Result<ObjectVersionFrames, RpcError> {
58    let started = Instant::now();
59    let served_height = ServedHeight::of(&request);
60    let request = request.into_inner();
61    let config = server.ledger_history.list_object_versions();
62    let query =
63        ObjectVersionsQuery::new(&request, config.default_limit_items, config.max_limit_items)?;
64    let client = server.limited_client(OPERATION);
65
66    // Both checks fail the request whatever the object's history, so they come before any read.
67    let reads = server.at_checkpoint_reads(client.clone(), served_height)?;
68    let upper_bound = AtCheckpointUpperBound::new(reads.served_height.checkpoint);
69    let window = query.checkpoint_window(&upper_bound)?;
70    let ctx = server.query_context(OPERATION, client.clone(), reads.served_height)?;
71    let object_id = query.object_id;
72    let read_mask = query.read_mask.clone();
73    let lists_removal_transactions = query.lists_removal_transactions();
74    let with_contents = !needs_only_object_id_and_version(&read_mask);
75    let resolution = if with_contents { "objects" } else { "versions" };
76    let observed = |frames| observe_frames(ctx.clone(), resolution, &query, started, frames);
77    // The object's versions at the window's two edges. An empty checkpoint range lists nothing,
78    // so it reads no version at its checkpoints.
79    let version_at = async |checkpoint: Option<u64>| match checkpoint {
80        Some(checkpoint) if !window.is_empty() => Ok(reads
81            .version_at_checkpoint(object_id, checkpoint)
82            .await?
83            .map(|(version, _)| version)),
84        _ => Ok::<_, RpcError>(None),
85    };
86    let (newest_visible, newest_before_start) = futures::try_join!(
87        version_at(Some(window.bounding_checkpoint)),
88        version_at(window.below_start),
89    )?;
90    let plan = query.plan(&window, &upper_bound, newest_visible, newest_before_start)?;
91    let bounding_checkpoint = window.bounding_checkpoint;
92    let end_cursor = query.end_cursor(&plan, bounding_checkpoint);
93    let Some(versions) = plan.versions else {
94        return Ok(observed(query.frames(
95            bounding_checkpoint,
96            stream::empty().boxed(),
97            plan.end,
98            std::future::ready(end_cursor),
99        )));
100    };
101
102    let rows = client
103        .object_versions_stream(
104            object_id,
105            versions,
106            query.limit,
107            query.descending,
108            with_contents,
109        )
110        .await?
111        .map_err(RpcError::from)
112        .map_ok(ResolvedWatermarked::<(SequenceNumber, ObjectVersionRow)>::Item)
113        .boxed();
114    let resolver = server.package_resolver.clone();
115    let render_ctx = ctx.clone();
116    let listed = render_ahead(
117        rows,
118        config.render_ahead,
119        move |(version, row): (SequenceNumber, ObjectVersionRow)| {
120            let read_mask = read_mask.clone();
121            let resolver = resolver.clone();
122            let ctx = render_ctx.clone();
123            async move {
124                let render_started = Instant::now();
125                let item = match row {
126                    ObjectVersionRow::Live(Some(object)) => ObjectVersionItem::Live {
127                        object: object_to_response(&object, &read_mask, &resolver).await,
128                        version,
129                    },
130                    ObjectVersionRow::Live(None) => ObjectVersionItem::Live {
131                        object: object_version_to_response(object_id, version),
132                        version,
133                    },
134                    ObjectVersionRow::Tombstone(tombstone) => {
135                        removal(object_id, version, tombstone, lists_removal_transactions)?
136                    }
137                };
138                ctx.observe_response_render(resolution, render_started.elapsed());
139                Ok::<_, RpcError>(item)
140            }
141        },
142    )
143    .map(|frame| match frame {
144        Ok(ResolvedWatermarked::Item(listed)) => Ok(listed),
145        Ok(ResolvedWatermarked::Watermark { .. }) => {
146            unreachable!("object version reads emit no watermarks")
147        }
148        Err(RenderAheadError::Upstream(error) | RenderAheadError::Render(error)) => Err(error),
149    })
150    .boxed();
151    Ok(observed(query.frames(
152        bounding_checkpoint,
153        listed,
154        plan.end,
155        std::future::ready(end_cursor),
156    )))
157}

159/// `frames`, recording what ListCheckpoints, ListTransactions and ListEvents record for theirs:
160/// each poll's wait (`kv_rpc_final_stream_poll_wait_ms`), each item frame's encoded size
161/// (`kv_rpc_response_page_bytes`), the terminal-only frame
162/// (`kv_rpc_stream_watermark_frames_total`), the latency to the first frame, the wait after
163/// yielding each frame, and the completion log. Every frame of `frames` but the last carries an
164/// item, and the frame with `end` is the last.
165fn observe_frames(
166    ctx: QueryContext,
167    resolution: &'static str,
168    query: &ObjectVersionsQuery,
169    started: Instant,
170    mut frames: ObjectVersionFrames,
171) -> ObjectVersionFrames {
172    let limit_items = query.limit;
173    let descending = query.descending;
174    async_stream::try_stream! {
175        let mut emitted = 0usize;
176        let mut first_frame_emitted = false;
177        let terminal_reason: Option<QueryEndReason> = loop {
178            let Some(frame) = ctx.next_response_item(resolution, &mut frames).await else {
179                break None;
180            };
181            let frame = frame?;
182            if frame.item.is_some() {
183                emitted += 1;
184                ctx.observe_response_page_bytes(resolution, frame.encoded_len());
185            } else {
186                ctx.inc_stream_watermark_frames();
187            }
188            let end = frame.end.as_ref().map(|end| end.reason());
189            if !first_frame_emitted {
190                ctx.observe_stream_first_frame_latency(resolution, started.elapsed());
191                first_frame_emitted = true;
192            }
193            let yield_started = Instant::now();
194            yield frame;
195            ctx.observe_stream_frame_yield_wait(resolution, yield_started.elapsed());
196            if end.is_some() {
197                break end;
198            }
199        };
200        info!(
201            limit_items,
202            descending,
203            emitted,
204            ?terminal_reason,
205            elapsed_ms = started.elapsed().as_millis(),
206            "list_object_versions: done"
207        );
208    }
209    .boxed()
210}

212/// The removal a tombstone at `version` lists, with its transaction if `with_transaction`. A
213/// tombstone without its transaction can't serve a request that asks for it, which is `INTERNAL`:
214/// listing removals requires the tombstone backfill, which writes every tombstone's transaction.
215fn removal(
216    object_id: ObjectID,
217    version: SequenceNumber,
218    tombstone: ObjectTombstone,
219    with_transaction: bool,
220) -> Result<ObjectVersionItem, RpcError> {
221    let transaction = if with_transaction {
222        let transaction = tombstone.transaction.ok_or_else(|| {
223            RpcError::new(
224                tonic::Code::Internal,
225                format!(
226                    "the tombstone of {object_id} at version {} has no transaction",
227                    version.value()
228                ),
229            )
230        })?;
231        Some(transaction)
232    } else {
233        None
234    };
235    Ok(ObjectVersionItem::Removal {
236        version,
237        kind: removal_kind(tombstone.status),
238        transaction,
239    })
240}

415    /// Everything `req`'s stream sends on `kv_rpc`, which serves `served_checkpoint`, checking
416    /// the frames' shape: every frame carries a cursor, each item frame carries exactly one of
417    /// `object` and `removal` and, as its cursor, the cursor of its version, minted at the
418    /// listing's bounding checkpoint, each listed object is the requested one, rendered with
419    /// `digest` under the default read mask, each removal is deleted or wrapped, and `end` is set
420    /// exactly once, on the final frame, which carries an item only for `ITEM_LIMIT`.
421    async fn list_served_at(
422        kv_rpc: &KvRpcServer,
423        req: ListObjectVersionsRequest,
424        served_checkpoint: u64,
425    ) -> Result<Listing, tonic::Status> {
426        let requested = req.object_id.clone();
427        let bounding_checkpoint = bounding_checkpoint(&req, served_checkpoint);
428        let default_mask = req.read_mask.is_none();
429        let frames: Vec<ListObjectVersionsResponse> =
430            list_object_versions(kv_rpc, kv_rpc.admitted_request(req))
431                .await?
432                .try_collect()
433                .await?;
434        let id: ObjectID = requested.as_deref().unwrap().parse().unwrap();
435        let mut listing = Listing {
436            versions: vec![],
437            items: vec![],
438            cursors: vec![],
439            end_cursor: Bytes::new(),
440            end: QueryEndReason::Unknown,
441        };
442        for (index, frame) in frames.iter().enumerate() {
443            let last = index + 1 == frames.len();
444            assert_eq!(
445                frame.end.is_some(),
446                last,
447                "end on frame {index} of {frames:?}"
448            );
449            let cursor = frame
450                .cursor
451                .clone()
452                .unwrap_or_else(|| panic!("frame {index} has no cursor: {frames:?}"));
453            if last {
454                listing.end_cursor = cursor.clone();
455            }
456            let (version, status, transaction) = match &frame.item {
457                Some(Item::Object(object)) => {
458                    if let Some(object_id) = &object.object_id {
459                        assert_eq!(Some(object_id), requested.as_ref());
460                    }
461                    if default_mask {
462                        assert!(
463                            object.digest.is_some(),
464                            "the default read mask renders digest"
465                        );
466                    }
467                    let version = object.version.expect("version is set");
468                    (version, ObjectStatus::Live, None)
469                }
470                Some(Item::Removal(removal)) => {
471                    let status = match removal.kind() {
472                        object_removal::Kind::Deleted => ObjectStatus::Deleted,
473                        object_removal::Kind::Wrapped => ObjectStatus::Wrapped,
474                        kind => panic!("removal of kind {kind:?} in {frames:?}"),
475                    };
476                    let version = removal.version.expect("removal version is set");
477                    (version, status, removal.transaction.clone())
478                }
479                Some(item) => panic!("frame {index} has an unknown item {item:?}"),
480                None => {
481                    assert!(last, "{frames:?}");
482                    continue;
483                }
484            };
485            assert_eq!(
486                cursor,
487                cursor_at(id, version, bounding_checkpoint),
488                "version {version}"
489            );
490            listing.versions.push(version);
491            listing.items.push((version, status, transaction));
492            listing.cursors.push(cursor);
493        }
494        let last = frames.last().expect("a stream ends with a frame");
495        listing.end = last
496            .end
497            .and_then(|end| end.reason)
498            .map(|reason| QueryEndReason::try_from(reason).unwrap())
499            .expect("the final frame has a reason");
500        assert_eq!(
501            last.item.is_some(),
502            listing.end == QueryEndReason::ItemLimit,
503            "{frames:?}"
504        );
505        Ok(listing)
506    }

534    /// The model check of [`test_list_object_versions_intersects_every_bound`] for a fixture
535    /// object with `history`.
536    async fn assert_listings_intersect_every_bound(history: &[(u64, u64, ObjectStatus)]) {
537        let (kv_rpc, id, mock_server) =
538            start_fixture_with_history(OBJECT_AT_CHECKPOINT, history).await;
539
540        let expected = |start: Option<u64>,
541                        end: Option<u64>,
542                        after: Option<u64>,
543                        before: Option<u64>,
544                        (start_checkpoint, end_checkpoint): (Option<u64>, Option<u64>),
545                        ordering: Ordering,
546                        limit: Option<u32>| {
547            let mut items: Vec<(u64, ObjectStatus)> = history
548                .iter()
549                .filter(|&&(checkpoint, version, _)| {
550                    start.is_none_or(|start| version >= start)
551                        && end.is_none_or(|end| version < end)
552                        && after.is_none_or(|after| version > after)
553                        && before.is_none_or(|before| version < before)
554                        && start_checkpoint.is_none_or(|start| checkpoint >= start)
555                        && end_checkpoint.is_none_or(|end| checkpoint < end)
556                })
557                .map(|&(_, version, status)| (version, status))
558                .collect();
559            if ordering == Ordering::Descending {
560                items.reverse();
561            }
562            items.truncate(limit.map_or(usize::MAX, |limit| limit as usize));
563            items
564        };
565
566        // Bounds on either side of each item (in `HISTORY`, 8 is the wrap and 14 the deletion; in
567        // `RECREATED_HISTORY`, 5 is the deletion and 8 the object's next creation), and
568        // checkpoint ranges that end before creation, at it, between changes, and at version 8,
569        // that start at a change or between changes, that hold nothing, and that end at the

1966    /// The versions of `history` a listing in `ordering` lists between the positions `after`

1984            })
1985            .map(|&(_, version, _)| version)
1986            .collect();

1988            versions.reverse();
1989        }
