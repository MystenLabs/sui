78async fn get_package(
79    mut client: BigTableClient,
80    service_info_watermark_pipelines: &[&str],
81    request: GetPackageRequest,
82) -> Result<GetPackageResponse, RpcError> {
83    let package_id_str = request.package_id.as_ref().ok_or_else(|| {
84        FieldViolation::new("package_id")
85            .with_description("missing package_id")
86            .with_reason(ErrorReason::FieldMissing)
87    })?;
88    let package_id = parse_package_id(package_id_str)?;
89
90    let selector = match request.selector {
91        None => {
92            let package = load_package(client, package_id).await?;
93            return get_package_response(&package);
94        }
95        Some(selector) => selector,
96    };
97
98    let data = match selector {
99        Selector::Version(version) => {
100            let original_id = resolve_original_package_id(client.clone(), package_id).await?;
101            client
102                .get_packages_by_version(&[(original_id, version)])
103                .await
104                .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
105                .pop()
106        }
107        Selector::AtCheckpoint(at_checkpoint) => {
108            // The response does not echo the checkpoint it resolved at, so a bound above
109            // what this replica has indexed would let a caller persist a wrong answer;
110            // fail closed instead. `Version` and the bare-id path are exact lookups and
111            // need no bound.
112            let highest_indexed = client
113                .get_watermark_for_pipelines(service_info_watermark_pipelines)
114                .await
115                .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
116                .and_then(|watermark| watermark.checkpoint_hi_inclusive)
117                .ok_or_else(|| RpcError::new(tonic::Code::Unavailable, "rpc index is empty"))?;
118            if at_checkpoint > highest_indexed {
119                return Err(RpcError::new(
120                    tonic::Code::NotFound,
121                    format!(
122                        "requested checkpoint {at_checkpoint} is not yet indexed; highest \
123                         indexed checkpoint is {highest_indexed}",
124                    ),
125                ));
126            }
127            let original_id = resolve_original_package_id(client.clone(), package_id).await?;
128            client
129                .get_package_latest(original_id, at_checkpoint)
130                .await
131                .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
132        }
133        _ => {
134            return Err(FieldViolation::new("selector")
135                .with_description("unknown selector variant")
136                .with_reason(ErrorReason::FieldInvalid)
137                .into());
138        }
139    }
140    .ok_or_else(RpcError::not_found)?;
141
142    let storage_id = ObjectID::from_bytes(&data.package_id).map_err(|e| {
143        RpcError::new(
144            tonic::Code::Internal,
145            format!("invalid stored package id: {e}"),
146        )
147    })?;
148
149    let object = client
150        .get_objects(&[ObjectKey(storage_id, data.package_version.into())])
151        .await
152        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
153        .pop()
154        .ok_or_else(RpcError::not_found)?;
155    let package = into_package(object)?;
156
157    get_package_response(&package)
158}

160async fn get_datatype(
161    client: BigTableClient,
162    request: GetDatatypeRequest,
163) -> Result<GetDatatypeResponse, RpcError> {
164    let package_id_str = request.package_id.as_ref().ok_or_else(|| {
165        FieldViolation::new("package_id")

169

177        FieldViolation::new("name")
178            .with_description("missing name")
179            .with_reason(ErrorReason::FieldMissing)
180    })?;
181
182    let package = load_package(client, parse_package_id(package_id_str)?).await?;
183    get_datatype_response(&package, module_name, datatype_name)
184}

212async fn list_package_versions(
213    mut client: BigTableClient,
214    request: ListPackageVersionsRequest,
215) -> Result<ListPackageVersionsResponse, RpcError> {
216    let package_id_str = request.package_id.as_ref().ok_or_else(|| {
217        FieldViolation::new("package_id")
218            .with_description("missing package_id")
219            .with_reason(ErrorReason::FieldMissing)
220    })?;
221    let original_package_id =
222        resolve_original_package_id(client.clone(), parse_package_id(package_id_str)?).await?;
223
224    let page_size = request
225        .page_size
226        .map(|s| (s as usize).clamp(1, 10000))
227        .unwrap_or(1000);
228
229    let page_token = request
230        .page_token
231        .map(|token| PageToken::decode(&token))
232        .transpose()?;
233
234    if let Some(token) = &page_token
235        && token.original_package_id != original_package_id
236    {
237        return Err(FieldViolation::new("page_token")
238            .with_description("page token package ID does not match request package ID")
239            .with_reason(ErrorReason::FieldInvalid)
240            .into());
241    }
242
243    // The token's version is the first version of the next page (inclusive); the reader's
244    // `after_version` bound is exclusive.
245    let after_version = page_token.map(|token| token.version.saturating_sub(1));
246
247    let mut versions: Vec<_> = client
248        .get_package_versions(
249            original_package_id,
250            u64::MAX,
251            after_version,
252            None,
253            page_size + 1,
254            false,
255        )
256        .await
257        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
258        .into_iter()
259        .map(|pkg| {
260            let storage_id = ObjectID::from_bytes(&pkg.package_id)
261                .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?;
262            Ok(PackageVersion::new(&storage_id.into(), pkg.package_version))
263        })
264        .collect::<Result<_, RpcError>>()?;
265
266    let next_page_token = if versions.len() > page_size {
267        versions.pop().and_then(|next| {
268            next.version.map(|version| {
269                PageToken {
270                    original_package_id,
271                    version,
272                }
273                .encode()
274            })
275        })
276    } else {
277        None
278    };
279
280    Ok(ListPackageVersionsResponse::new(versions, next_page_token))
281}

283/// Load a package by its storage ID: a storage ID identifies exactly one immutable package
284/// version, so the latest object at that ID is the package itself.
285async fn load_package(
286    mut client: BigTableClient,
287    package_id: ObjectID,
288) -> Result<MovePackage, RpcError> {
289    let object = client
290        .get_latest_object(&package_id)
291        .await
292        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
293        .ok_or_else(RpcError::not_found)?;
294
295    into_package(object)
296}

298fn into_package(object: Object) -> Result<MovePackage, RpcError> {
299    object
300        .into_inner()
301        .data
302        .try_into_package()
303        .ok_or_else(|| RpcError::new(tonic::Code::InvalidArgument, "object is not a package"))
304}

306/// Resolve the original (first-version) package ID for a storage ID. `packages_by_id` answers
307/// with a 32-byte point lookup; on a miss the object itself is read so the error matches what a
308/// full node returns (`NotFound` vs "object is not a package"), and so a package whose
309/// `packages_by_id` row is not yet written is still served.
310async fn resolve_original_package_id(
311    mut client: BigTableClient,
312    package_id: ObjectID,
313) -> Result<ObjectID, RpcError> {
314    let pairs = client
315        .get_package_original_ids(&[package_id])
316        .await
317        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?;
318    if let Some((_, original_id)) = pairs
319        .iter()
320        .find(|(storage_id, _)| *storage_id == package_id)
321    {
322        return Ok(*original_id);
323    }
324
325    load_package(client, package_id)
326        .await
327        .map(|package| package.original_package_id())
328}

330fn parse_package_id(package_id_str: &str) -> Result<ObjectID, RpcError> {
331    package_id_str.parse::<ObjectID>().map_err(|e| {
332        FieldViolation::new("package_id")
333            .with_description(format!("invalid package_id: {}", e))
334            .with_reason(ErrorReason::FieldInvalid)
335            .into()
336    })
337}

640        let mut req = get_package_req(fixture.original_id);
641        req.selector = Some(Selector::Version(2));

644            package.storage_id,
645            Some(fixture.upgraded_id.to_canonical_string(true)),

648
649        // A checkpoint bound between the two publishes resolves v1.
650        let mut req = get_package_req(fixture.upgraded_id);
651        req.selector = Some(Selector::AtCheckpoint(19));

659        // A checkpoint bound equal to the watermark resolves the latest version.

662        let package = fetch_pkg(&fixture, req).await;

692        let (addr, _server) = mock.start().await.expect("start mock BigTable");
693        let client = BigTableClient::new_local(addr.to_string(), "test".to_string())
694            .await
695            .expect("connect to mock BigTable");
696
697        // A bare-id request is still served without any watermark rows.
698        let mut bare = GetPackageRequest::default();
699        bare.package_id = Some(ObjectID::ZERO.to_canonical_string(true));
700        let bare_status: tonic::Status =
701            get_package(client.clone(), TEST_WATERMARK_PIPELINES, bare)
702                .await
703                .expect_err("bare lookup fails on empty store")
704                .into();
705        assert_eq!(bare_status.code(), tonic::Code::NotFound);
706
707        // The checkpoint selector fails closed instead of clamping to a checkpoint
708        // this replica has not indexed.
709        let mut req = GetPackageRequest::default();
710        req.package_id = Some(ObjectID::ZERO.to_canonical_string(true));
711        req.selector = Some(Selector::AtCheckpoint(5));
712        let status: tonic::Status = get_package(client.clone(), TEST_WATERMARK_PIPELINES, req)
713            .await
714            .expect_err("checkpoint lookup fails on empty index")
715            .into();

717        assert_eq!(status.message(), "rpc index is empty");
718
719        // The version selector is not bounded, so it reaches the underlying lookup
720        // (not found here, not rejected as unavailable).
721        let mut req = GetPackageRequest::default();
722        req.package_id = Some(ObjectID::ZERO.to_canonical_string(true));
723        req.selector = Some(Selector::Version(1));

725            .await
726            .expect_err("version lookup reaches the store")
727            .into();
