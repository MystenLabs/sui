71async fn get_package(
72    client: BigTableClient,
73    request: GetPackageRequest,
74) -> Result<GetPackageResponse, RpcError> {
75    let package_id_str = request.package_id.as_ref().ok_or_else(|| {
76        FieldViolation::new("package_id")
77            .with_description("missing package_id")
78            .with_reason(ErrorReason::FieldMissing)
79    })?;
80
81    let package = load_package(client, parse_package_id(package_id_str)?).await?;
82    get_package_response(&package)
83}

137async fn list_package_versions(
138    mut client: BigTableClient,
139    request: ListPackageVersionsRequest,
140) -> Result<ListPackageVersionsResponse, RpcError> {
141    let package_id_str = request.package_id.as_ref().ok_or_else(|| {
142        FieldViolation::new("package_id")
143            .with_description("missing package_id")
144            .with_reason(ErrorReason::FieldMissing)
145    })?;
146    let original_package_id =
147        resolve_original_package_id(client.clone(), parse_package_id(package_id_str)?).await?;
148
149    let page_size = request
150        .page_size
151        .map(|s| (s as usize).clamp(1, 10000))
152        .unwrap_or(1000);
153
154    let page_token = request
155        .page_token
156        .map(|token| PageToken::decode(&token))
157        .transpose()?;
158
159    if let Some(token) = &page_token
160        && token.original_package_id != original_package_id
161    {
162        return Err(FieldViolation::new("page_token")
163            .with_description("page token package ID does not match request package ID")
164            .with_reason(ErrorReason::FieldInvalid)
165            .into());
166    }
167
168    // The token's version is the first version of the next page (inclusive); the reader's
169    // `after_version` bound is exclusive.
170    let after_version = page_token.map(|token| token.version.saturating_sub(1));
171
172    let mut versions: Vec<_> = client
173        .get_package_versions(
174            original_package_id,
175            u64::MAX,
176            after_version,
177            None,
178            page_size + 1,
179            false,
180        )
181        .await
182        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
183        .into_iter()
184        .map(|pkg| {
185            let storage_id = ObjectID::from_bytes(&pkg.package_id)
186                .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?;
187            Ok(PackageVersion::new(&storage_id.into(), pkg.package_version))
188        })
189        .collect::<Result<_, RpcError>>()?;
190
191    let next_page_token = if versions.len() > page_size {
192        versions.pop().and_then(|next| {
193            next.version.map(|version| {
194                PageToken {
195                    original_package_id,
196                    version,
197                }
198                .encode()
199            })
200        })
201    } else {
202        None
203    };
204
205    Ok(ListPackageVersionsResponse::new(versions, next_page_token))
206}

208/// Load a package by its storage ID: a storage ID identifies exactly one immutable package
209/// version, so the latest object at that ID is the package itself.
210async fn load_package(
211    mut client: BigTableClient,
212    package_id: ObjectID,
213) -> Result<MovePackage, RpcError> {
214    let object = client
215        .get_latest_object(&package_id)
216        .await
217        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
218        .ok_or_else(RpcError::not_found)?;
219
220    object
221        .into_inner()
222        .data
223        .try_into_package()
224        .ok_or_else(|| RpcError::new(tonic::Code::InvalidArgument, "object is not a package"))
225}

227/// Resolve the original (first-version) package ID for a storage ID. `packages_by_id` answers
228/// with a 32-byte point lookup; on a miss the object itself is read so the error matches what a
229/// full node returns (`NotFound` vs "object is not a package"), and so a package whose
230/// `packages_by_id` row is not yet written is still served.
231async fn resolve_original_package_id(
232    mut client: BigTableClient,
233    package_id: ObjectID,
234) -> Result<ObjectID, RpcError> {
235    let pairs = client
236        .get_package_original_ids(&[package_id])
237        .await
238        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?;
239    if let Some((_, original_id)) = pairs
240        .iter()
241        .find(|(storage_id, _)| *storage_id == package_id)
242    {
243        return Ok(*original_id);
244    }
245
246    load_package(client, package_id)
247        .await
248        .map(|package| package.original_package_id())
249}
250
251fn parse_package_id(package_id_str: &str) -> Result<ObjectID, RpcError> {
252    package_id_str.parse::<ObjectID>().map_err(|e| {
253        FieldViolation::new("package_id")
254            .with_description(format!("invalid package_id: {}", e))
255            .with_reason(ErrorReason::FieldInvalid)
256            .into()
257    })
258}

274    #[tokio::test]
275    async fn list_package_versions_falls_back_to_object_when_packages_by_id_missing() {
276        let mock = MockBigtableServer::new();
277        let (addr, _handle) = mock.start().await.unwrap();
278        let client = BigTableClient::new_local(addr.to_string(), "test".to_string())
279            .await
280            .unwrap();
281
282        let original_id = ObjectID::random();
283        let pkg = MovePackage::new(
284            original_id,
285            SequenceNumber::from(1),
286            BTreeMap::new(),
287            100_000,
288            vec![],
289            BTreeMap::new(),
290        )
291        .unwrap();
292        let obj = Object::new_package_from_data(Data::Package(pkg), TransactionDigest::ZERO);
293
294        // Insert package object into `objects` table
295        let obj_key = tables::objects::encode_key(&ObjectKey(original_id, SequenceNumber::from(1)));
296        let obj_cells = tables::objects::encode(&obj).unwrap();
297        mock.insert_row(tables::objects::NAME, obj_key, obj_cells)
298            .await;
299
300        // Insert 3 versions into `packages` table: v1 at cp 10, v2 at cp 20, v3 at cp 30
301        for (v, cp) in [(1, 10), (2, 20), (3, 30)] {
302            let row_key = tables::packages::encode_key(original_id.as_ref(), v);
303            let cells = tables::packages::encode(cp, original_id.as_ref(), false);
304            mock.insert_row(tables::packages::NAME, row_key, cells)
305                .await;
306        }
307
308        let mut req = ListPackageVersionsRequest::default();
309        req.package_id = Some(original_id.to_string());
310        req.page_size = Some(10);
311        let resp = list_package_versions(client.clone(), req).await.unwrap();
312        assert_eq!(resp.versions.len(), 3);
313        assert_eq!(resp.versions[0].version, Some(1));
314        assert_eq!(resp.versions[1].version, Some(2));
315        assert_eq!(resp.versions[2].version, Some(3));
316        assert!(resp.next_page_token.is_none());
317    }

332        mock.insert_row(
333            tables::packages_by_id::NAME,
334            tables::packages_by_id::encode_key(storage_id.as_ref()),
335            tables::packages_by_id::encode(original_id.as_ref()),
336        )
337        .await;
338        for (v, cp) in [(1, 10), (2, 20), (3, 30)] {
339            let row_key = tables::packages::encode_key(original_id.as_ref(), v);
340            let cells = tables::packages::encode(cp, original_id.as_ref(), false);
341            mock.insert_row(tables::packages::NAME, row_key, cells)
342                .await;
343        }
344
345        let mut req = ListPackageVersionsRequest::default();
346        req.package_id = Some(storage_id.to_string());
347        req.page_size = Some(10);
348        let resp = list_package_versions(client.clone(), req).await.unwrap();
349        assert_eq!(resp.versions.len(), 3);
350        assert_eq!(resp.versions[0].version, Some(1));

354
355        assert!(
356            mock.read_rows_calls()
357                .await
