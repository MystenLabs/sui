28/// Identifies which inverted-index table a `BitmapQuery` evaluates against.
29#[derive(Clone, Copy)]
30pub struct BitmapIndexSpec {
31    pub table_name: &'static str,
32    pub schema: BitmapSchema,
33    pub bucket_size: u64,
34    pub bucket_id_width: usize,
35    pub bitmap_column: &'static str,
36}
37
38impl BitmapIndexSpec {
39    /// The transaction bitmap index table of `schema`.
40    pub const fn tx(schema: BitmapSchema) -> Self {
41        Self {
42            table_name: transaction_bitmap_index::table_name(schema),
43            schema,
44            bucket_size: transaction_bitmap_index::BUCKET_SIZE,
45            bucket_id_width: transaction_bitmap_index::BUCKET_ID_WIDTH,
46            bitmap_column: transaction_bitmap_index::col::BITMAP,
47        }
48    }
49
50    /// The event bitmap index table of `schema`.
51    pub const fn event(schema: BitmapSchema) -> Self {
52        Self {
53            table_name: event_bitmap_index::table_name(schema),
54            schema,
55            bucket_size: event_bitmap_index::BUCKET_SIZE,
56            bucket_id_width: event_bitmap_index::BUCKET_ID_WIDTH,
57            bitmap_column: event_bitmap_index::col::BITMAP,
58        }
59    }
60
61    fn is_transaction_index(&self) -> bool {
62        matches!(
63            self.table_name,
64            transaction_bitmap_index::NAME | transaction_bitmap_index::NAME_V2
65        )
66    }
67
68    fn encode_row_key(&self, dimension_key: &[u8], bucket_id: u64) -> Vec<u8> {
69        let mut key = Vec::new();
70        tables::encode_bitmap_row_key_into(
71            &mut key,
72            self.schema,
73            self.bucket_id_width,
74            dimension_key,
75            bucket_id,
76        );
77        key
78    }
79}

93impl BitmapBucketSource for BigTableBitmapSource {
94    fn scan_bucket_stream(
95        &self,
96        dimension_key: Vec<u8>,
97        range: Range<u64>,
98        direction: ScanDirection,
99    ) -> BucketStream {
100        // The tx universe is dense (every tx_seq in range is real), so it is
101        // synthesized here rather than stored; see `IndexDimension::TxUniverse`.
102        // Gated on the tx table: the key is only ever produced by the tx filter
103        // layer, and in event-space a dense universe would be semantically wrong.
104        if self.spec.is_transaction_index()
105            && dimension_key.first() == Some(&IndexDimension::TxUniverse.tag_byte())
106        {
107            return futures::stream::iter(
108                dense_universe_buckets(range, self.spec.bucket_size, direction).map(Ok),
109            )
110            .boxed();
111        }
112        scan_bucket_stream(
113            self.client.clone(),
114            dimension_key,
115            range,
116            self.spec,
117            direction,
118        )
119    }
120}

122/// Stream a single bitmap-index dimension's buckets in order, one
123/// `RoaringBitmap` per bucket with **relative** bit positions.
124fn scan_bucket_stream(
125    mut client: BigTableClient,
126    dimension_key: Vec<u8>,
127    range: Range<u64>,
128    spec: BitmapIndexSpec,
129    direction: ScanDirection,
130) -> BucketStream {
131    async_stream::try_stream! {
132        if range.is_empty() {
133            return;
134        }
135        let start_bucket = range.start / spec.bucket_size;
136        let end_bucket = (range.end - 1) / spec.bucket_size;
137
138        let start_row = spec.encode_row_key(&dimension_key, start_bucket);
139        let end_row = spec.encode_row_key(&dimension_key, end_bucket);
140
141        let stream = client
142            .range_scan_stream(
143                spec.table_name,
144                Some(bytes::Bytes::from(start_row)),
145                Some(bytes::Bytes::from(end_row)),
146                0,
147                !direction.is_ascending(),
148                None,
149            )
150            .await?;
151        futures::pin_mut!(stream);
152
153        while let Some(row) = stream.next().await {
154            let (row_key, cells) = row?;
155            let Some(bitmap_bytes) = cells
156                .iter()
157                .find(|(col, _)| col.as_ref() == spec.bitmap_column.as_bytes())
158                .map(|(_, v)| v)
159            else {
160                continue;
161            };
162
163            let hash_pos = row_key
164                .iter()
165                .rposition(|&b| b == b'#')
166                .context("malformed bitmap index row key: no '#' separator")?;
167            let suffix = &row_key[hash_pos + 1..];
168            let bucket_id: u64 = std::str::from_utf8(suffix)
169                .context("non-ascii bucket_id suffix")?
170                .parse()
171                .context("invalid bucket_id in row key")?;
172
173            let bitmap = RoaringBitmap::deserialize_from(bitmap_bytes.as_ref())
174                .context("deserializing bitmap")?;
175            yield (bucket_id, bitmap);
176        }
177    }
178    .boxed()
179}
