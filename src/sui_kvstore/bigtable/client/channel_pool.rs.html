4//! Pool of BigTable channels, each one HTTP/2 connection.
5//!
6//! Each request goes to the non-quarantined channel with the lowest load: its requests in flight,
7//! plus a penalty of `max_rpcs_per_channel` for a few seconds after a request on it fails with
8//! `Unavailable`, `ResourceExhausted`, `Internal`, or a connection error, or a ReadRows on it is
9//! abandoned after its gap timeout. A channel whose requests fail fast never builds up requests
10//! in flight; the penalty makes it look a full channel's worth busier, so it gets traffic only
11//! once every other channel is that much busier. A request stays in flight until its response
12//! body ends or is dropped, and every response frame it receives counts as progress for its
13//! channel.
14//!
15//! A health checker looks for channels where callers are waiting but nothing has arrived for a
16//! while, although another channel is receiving responses. A caller waits from its request until
17//! the response headers arrive, and again from each body poll that finds no frame buffered until
18//! its next poll or until it drops the response. A caller busy with data it already has doesn't
19//! count, so a slow consumer does not by itself make its channel look stalled; its unread data
20//! still holds the connection's flow-control window, though, and can starve the waiting callers
21//! that do count. The checker confirms each suspect with a `PingAndWarm` probe on that channel;
22//! any answer, an error status included, shows the connection works. A channel whose probe times
23//! out or fails on the connection is quarantined: its open requests fail at once with a retryable
24//! `Unavailable` status (see [`is_quarantine_cancellation`]) without waiting on the stalled
25//! connection, and new requests avoid it until a re-probe succeeds. A probe whose credential
26//! fetch fails or times out never reaches the connection and leaves its channel as it was.
27//! Channels that keep failing probes, or are quarantined far more often than the rest of the
28//! pool, are replaced by a freshly connected channel. The health checker never quarantines more
29//! than half the pool: when more channels than that look stalled at once, the cause is BigTable
30//! or the network, not a channel.

357    /// Reject HTTP/2 windows outside 1..=2^31-1 bytes, the range RFC 9113 §6.9.1 allows; h2
358    /// panics on a larger one when the channel connects.
359    pub fn validate(&self) -> Result<()> {
360        const MAX_HTTP2_WINDOW_BYTES: u32 = (1 << 31) - 1;
361        for (key, bytes) in [
362            ("http2-stream-window-bytes", self.http2_stream_window_bytes),
363            (
364                "http2-connection-window-bytes",
365                self.http2_connection_window_bytes,
366            ),
367        ] {
368            anyhow::ensure!(
369                (1..=MAX_HTTP2_WINDOW_BYTES).contains(&bytes),
370                "{key} must be between 1 and {MAX_HTTP2_WINDOW_BYTES}, got {bytes}",
371            );
372        }
373        Ok(())
374    }

399                primer,

402        }
403    }
404
405    /// Create, connect, and spawn background maintenance tasks.
406    pub(crate) async fn new_connected(
407        endpoint: Endpoint,
408        config: PoolConfig,
409        primer: Option<Box<dyn ChannelPrimer>>,
410        registry: Option<&Registry>,
411    ) -> Result<Self> {
412        let pool = Self::new(endpoint, config, primer, registry);
413        pool.connect().await?;
414        pool.spawn_background_tasks();
415        Ok(pool)
416    }
417
418    /// Connect and prime `initial_pool_size` channels concurrently.
419    pub(crate) async fn connect(&self) -> Result<()> {
420        let entries = try_join_all(
421            (0..self.inner.config.initial_pool_size).map(|_| self.inner.create_primed_entry()),
422        )
423        .await?;
424        self.inner
425            .set_entries(entries.into_iter().map(Arc::new).collect());
426        Ok(())
427    }
428
429    /// Spawn the maintenance loop, which refreshes aged-out channels and resizes the pool to its
430    /// load, and the health checker, which quarantines and replaces stalled channels.
431    fn spawn_background_tasks(&self) {
432        let weak = Arc::downgrade(&self.inner);
433        let period = self.inner.config.maintenance_interval;
434        tokio::spawn(async move {
435            let mut interval = tokio::time::interval(period);
436            interval.tick().await; // first tick completes immediately
437            loop {
438                interval.tick().await;
439                let Some(inner) = weak.upgrade() else {
440                    return;
441                };
442                inner.refresh().await;
443                inner.resize().await;
444            }
445        });
446
447        let Some(health) = self.inner.config.health_check.clone() else {
448            return;
449        };
450        if self.inner.primer.is_none() {
451            return;
452        }
453        let weak = Arc::downgrade(&self.inner);
454        tokio::spawn(async move {
455            let mut interval = tokio::time::interval(health.check_interval);
456            interval.set_missed_tick_behavior(MissedTickBehavior::Delay);
457            loop {
458                interval.tick().await;
459                let Some(inner) = weak.upgrade() else {
460                    return;
461                };
462                let Some((entry, reason)) = inner.check_channel_health(&health).await else {
463                    continue;
464                };
465                // Connecting the replacement can take a while; keep checking meanwhile.
466                let replacing_pool = Arc::downgrade(&inner);
467                let probe_timeout = health.probe_timeout;
468                tokio::spawn(async move {
469                    if let Some(inner) = replacing_pool.upgrade() {
470                        inner
471                            .replace_unhealthy_channel(&entry, reason, probe_timeout)
472                            .await;
473                    }
474                });
475            }
476        });
477    }

480impl ChannelPoolInner {
481    /// Replace channels that have exceeded their refresh age. Limits replacements
482    /// per cycle to `MAX_REPLACEMENTS_PER_CYCLE` to avoid reconnection storms.
483    /// Also resets and logs per-channel error/success counters.
484    async fn refresh(&self) {
485        let _write = self.entries_write_lock.lock().await;
486        let snapshot = self.entries.load_full();
487        let mut new_entries: Vec<Arc<PoolEntry>> = (*snapshot).clone();
488        let now = Instant::now();
489        let mut replacements = 0;
490
491        for (idx, entry) in snapshot.iter().enumerate() {
492            let successes = entry.success_count.swap(0, Ordering::Relaxed);
493            let errors = entry.error_count.swap(0, Ordering::Relaxed);
494            if errors > 0 {
495                warn!(
496                    channel_idx = idx,
497                    errors, successes, "channel errors since last maintenance cycle"
498                );
499            }
500
501            if replacements >= MAX_REPLACEMENTS_PER_CYCLE {
502                continue;
503            }
504
505            if now < entry.refresh_at {
506                continue;
507            }
508
509            match self.create_primed_entry().await {
510                Ok(new_entry) => {
511                    info!(channel_idx = idx, "replacing bigtable channel");
512                    new_entries[idx] = Arc::new(new_entry);
513                    replacements += 1;
514                }
515                Err(e) => {
516                    warn!(channel_idx = idx, error = %e, "failed to create replacement channel");
517                }
518            }
519        }
520
521        if replacements > 0 {
522            *self.last_refresh_at.lock().unwrap() = Instant::now();
523            self.set_entries(new_entries);
524            if let Some(m) = &self.metrics {
525                m.channels_replaced.inc_by(replacements as u64);
526            }
527        }
528    }
529
530    /// Resize the pool toward the band of sizes that keeps the interval's peak in-flight count
531    /// between `min_rpcs_per_channel` and `max_rpcs_per_channel` per channel, by at most
532    /// `max_resize_delta` channels, never below `min_pool_size`. Inside the band the size is left
533    /// alone, so load that moves within it causes no churn. Shrinking drops quarantined channels
534    /// first.
535    async fn resize(&self) {
536        let _write = self.entries_write_lock.lock().await;
537        let entries = self.entries.load_full();
538        let current_size = entries.len();
539        let peak_in_flight = self.load.take_peak();
540        if let Some(m) = &self.metrics {
541            m.peak_in_flight.set(peak_in_flight as i64);
542        }
543        if current_size == 0 {
544            return;
545        }
546        let Some(target_size) = resize_target(peak_in_flight, current_size, &self.config) else {
547            return;
548        };
549
550        let mut new_entries = (*entries).clone();
551        if target_size > current_size {
552            let created =
553                join_all((current_size..target_size).map(|_| self.create_primed_entry())).await;
554            for entry in created {
555                match entry {
556                    Ok(entry) => new_entries.push(Arc::new(entry)),
557                    Err(e) => warn!(error = %e, "failed to create channel during pool expansion"),
558                }
559            }
560            info!(
561                current_size,
562                new_size = new_entries.len(),
563                peak_in_flight,
564                "pool resize: expanded"
565            );
566        } else {
567            let mut excess = current_size - target_size;
568            new_entries.retain(|entry| {
569                let remove = excess > 0 && entry.is_quarantined();
570                excess -= usize::from(remove);
571                !remove
572            });
573            new_entries.truncate(new_entries.len() - excess);
574            info!(
575                current_size,
576                new_size = new_entries.len(),
577                peak_in_flight,
578                "pool resize: shrunk"
579            );
580        }
581        self.set_entries(new_entries);
582    }
583
584    /// Inspect every channel once: probe channels that look stalled and quarantined channels due
585    /// for a re-probe, quarantine suspects whose probe fails, and restore quarantined channels
586    /// whose probe succeeds. Returns the channel to replace, if one is due and the pool-wide
587    /// replacement rate limit allows it.
588    async fn check_channel_health(
589        &self,
590        health: &ChannelHealthConfig,
591    ) -> Option<(Arc<PoolEntry>, ReplacementReason)> {
592        let entries = self.entries.load_full();
593        let now = Instant::now();
594        let quarantined = entries.iter().filter(|e| e.is_quarantined()).count();
595        let stalled: Vec<usize> = (0..entries.len())
596            .filter(|&idx| {
597                let entry = &entries[idx];
598                !entry.is_quarantined()
599                    && entry.waiting.load(Ordering::Relaxed) >= health.suspect_min_waiting_requests
600                    && entry.silence(now) >= health.suspect_after_silence
601            })
602            .collect();
603        // Silence only implicates a channel when another channel is still getting responses;
604        // when every channel is silent, BigTable or the network is the cause. Only responses
605        // count here: a probe answer says nothing about whether data is flowing.
606        let any_progressing = entries.iter().any(|e| {
607            !e.is_quarantined() && e.since_last_response(now) < health.suspect_after_silence
608        });
609        // Stalls on more than half the pool point at BigTable or the network too.
610        let pool_wide_stall =
611            !stalled.is_empty() && quarantined + stalled.len() > entries.len() / 2;
612        self.log_circuit_breaker(
613            pool_wide_stall,
614            any_progressing,
615            now,
616            stalled.len(),
617            quarantined,
618            entries.len(),
619        );
620
621        // (entry index, whether the entry is a suspect rather than a quarantined channel)
622        let mut probes: Vec<(usize, bool)> = Vec::new();
623        if any_progressing {
624            for idx in stalled {
625                if !entries[idx].start_probe(now, health.probe_interval) {
626                    continue;
627                }
628                if pool_wide_stall {
629                    self.count_suspicion("circuit_breaker_skipped");
630                    continue;
631                }
632                probes.push((idx, true));
633            }
634        }
635        for (idx, entry) in entries.iter().enumerate() {
636            if entry.is_quarantined() && entry.start_probe(now, health.probe_interval) {
637                probes.push((idx, false));
638            }
639        }
640        let probe_results = join_all(
641            probes
642                .iter()
643                .map(|&(idx, _)| self.probe(&entries[idx], health.probe_timeout)),
644        )
645        .await;
646
647        // Probing can take a while, during which a resize may have changed the pool; cap
648        // quarantines against the pool as it is now.
649        let now = Instant::now();
650        let live_entries = self.entries.load_full();
651        let max_quarantined = live_entries.len() / 2;
652        let mut quarantined = live_entries.iter().filter(|e| e.is_quarantined()).count();
653        let mut replacement = None;
654        for (&(idx, suspected), outcome) in probes.iter().zip_debug_eq(probe_results) {
655            let entry = &entries[idx];
656            if outcome == ProbeOutcome::NotSent {
657                // Leave the channel as it is; the next probe that gets through decides.
658                if suspected {
659                    self.count_suspicion("probe_not_sent");
660                    log_channel_event(
661                        Level::WARN,
662                        "bigtable channel looked stalled, but its probe could not fetch credentials; leaving it in service",
663                        &self.snapshot(idx, entry, now),
664                    );
665                }
666                continue;
667            }
668            let probe_ok = outcome == ProbeOutcome::Answered;
669            if !live_entries.iter().any(|live| Arc::ptr_eq(live, entry)) {
670                // A resize or refresh dropped the channel while it was probed. Outside the pool
671                // it neither counts toward the cap nor needs replacing, but a failed probe still
672                // fails the requests left on it.
673                if suspected && !probe_ok {
674                    self.quarantine(entry, now);
675                }
676                continue;
677            }
678            match (suspected, probe_ok) {
679                (true, true) => {
680                    log_channel_event(
681                        Level::DEBUG,
682                        "bigtable channel looked stalled but answered its probe; leaving it in service",
683                        &self.snapshot(idx, entry, now),
684                    );
685                    entry.restart_silence();
686                    self.count_suspicion("probe_ok");
687                }
688                (true, false) if quarantined < max_quarantined => {
689                    let snapshot = self.snapshot(idx, entry, now);
690                    self.quarantine(entry, now);
691                    quarantined += 1;
692                    self.count_suspicion("quarantined");
693                    log_channel_event(
694                        Level::WARN,
695                        "bigtable channel looked stalled and failed its probe; quarantined it and cancelled its open requests",
696                        &snapshot,
697                    );
698                    if replacement.is_none() && is_repeat_offender(&entries, idx, now, health) {
699                        replacement = Some((entry.clone(), ReplacementReason::RepeatOffender));
700                    }
701                }
702                (true, false) => {
703                    self.count_suspicion("circuit_breaker_skipped");
704                    log_channel_event(
705                        Level::WARN,
706                        "bigtable channel looked stalled and failed its probe, but half the pool is already quarantined; leaving it in service",
707                        &self.snapshot(idx, entry, now),
708                    );
709                }
710                (false, true) => {
711                    self.restore(entry);
712                    quarantined = quarantined.saturating_sub(1);
713                    log_channel_event(
714                        Level::WARN,
715                        "quarantined bigtable channel passed its probe; restored it",
716                        &self.snapshot(idx, entry, now),
717                    );
718                }
719                (false, false) => {
720                    if replacement.is_none()
721                        && entry.quarantined_for(now) >= health.replace_after_quarantined_for
722                    {
723                        replacement = Some((entry.clone(), ReplacementReason::StillFailingProbes));
724                    }
725                }
726            }
727        }
728        self.update_quarantined_gauge(&live_entries);
729
730        let replacement = replacement.filter(|_| !pool_wide_stall)?;
731        let mut checker = self.health_checker.lock().unwrap();
732        if checker
733            .last_replacement_at
734            .is_some_and(|at| now.saturating_duration_since(at) < health.min_replacement_interval)
735        {
736            return None;
737        }
738        checker.last_replacement_at = Some(now);
739        Some(replacement)
740    }

742    /// Swap a newly connected channel in for `entry`, once the new channel has passed a probe.
743    /// The old channel leaves the pool only when its replacement has answered, so the pool never
744    /// shrinks; if the replacement cannot connect or fails its probe, the old channel stays.
745    /// Returns whether the swap happened.
746    async fn replace_unhealthy_channel(
747        &self,
748        entry: &Arc<PoolEntry>,
749        reason: ReplacementReason,
750        probe_timeout: Duration,
751    ) -> bool {
752        if !self.entries.load().iter().any(|e| Arc::ptr_eq(e, entry)) {
753            return false;
754        }
755        let replacement = match self.connect_entry().await {
756            Ok(replacement) => Arc::new(replacement),
757            Err(e) => {
758                warn!(
759                    channel_id = entry.id.0,
760                    reason = reason.label(),
761                    error = %e,
762                    "failed to connect a replacement for an unhealthy bigtable channel; keeping it"
763                );
764                return false;
765            }
766        };
767        if self.probe(&replacement, probe_timeout).await != ProbeOutcome::Answered {
768            warn!(
769                channel_id = entry.id.0,
770                reason = reason.label(),
771                "replacement for an unhealthy bigtable channel failed its probe; keeping the old channel"
772            );
773            return false;
774        }
775
776        let _write = self.entries_write_lock.lock().await;
777        let mut new_entries = (**self.entries.load()).clone();
778        let Some(idx) = new_entries.iter().position(|e| Arc::ptr_eq(e, entry)) else {
779            return false;
780        };
781        let snapshot = self.snapshot(idx, entry, Instant::now());
782        new_entries[idx] = replacement;
783        self.set_entries(new_entries);
784        if let Some(m) = &self.metrics {
785            m.unhealthy_channels_replaced
786                .with_label_values(&[reason.label()])
787                .inc();
788        }
789        log_channel_event(
790            Level::WARN,
791            &format!("replaced unhealthy bigtable channel ({})", reason.label()),
792            &snapshot,
793        );
794        true
795    }

797    /// The non-quarantined channel with the lowest load, preferring channels the request's
798    /// routing does not avoid. Load is requests in flight, plus `max_rpcs_per_channel` while the
799    /// channel carries an error penalty. Go's flat penalty of 10 would still let a channel that
800    /// fails fast win most requests once the others each have more than 10 in flight, which the
801    /// load band allows. Ties go to the first channel from a rotating start, so equally loaded
802    /// channels share requests.
803    fn select_channel(&self, routing: Option<&ChannelRouting>) -> Option<Arc<PoolEntry>> {
804        let entries = self.entries.load();
805        let start = self.ticker.fetch_add(1, Ordering::Relaxed);
806        let now = Instant::now();
807        let error_penalty_load = self.config.max_rpcs_per_channel;
808        (0..entries.len())
809            .map(|offset| &entries[(start + offset) % entries.len()])
810            .min_by_key(|entry| {
811                let penalty = if entry.is_penalized(now) {
812                    error_penalty_load
813                } else {
814                    0
815                };
816                (
817                    entry.is_quarantined(),
818                    routing.is_some_and(|routing| routing.avoids(entry.id)),
819                    entry.in_flight.load(Ordering::Relaxed) + penalty,
820                )
821            })
822            .cloned()
823    }

825    /// Send a `PingAndWarm` on `entry`'s channel and report whether the connection answered within
826    /// `timeout`. Any status BigTable sends back counts as an answer, errors included: the probe
827    /// tests the connection, and an error such as a missing permission still made the round trip.
828    /// Credentials are fetched first, under a deadline of their own: a token fetch that fails or
829    /// is slow says nothing about the connection.
830    async fn probe(&self, entry: &PoolEntry, timeout: Duration) -> ProbeOutcome {
831        let Some(primer) = self.primer.as_deref() else {
832            return ProbeOutcome::NoAnswer;
833        };
834        let started_at = Instant::now();
835        let outcome = match tokio::time::timeout(timeout, primer.fetch_credentials()).await {
836            Ok(Ok(())) => match tokio::time::timeout(timeout, primer.prime(&entry.channel)).await {
837                Ok(Ok(())) => ProbeOutcome::Answered,
838                Ok(Err(e)) if is_status_from_server(&e) => ProbeOutcome::Answered,
839                Ok(Err(_)) | Err(_) => ProbeOutcome::NoAnswer,
840            },
841            Ok(Err(_)) | Err(_) => ProbeOutcome::NotSent,
842        };
843        if let Some(m) = &self.metrics {
844            let label = match outcome {
845                ProbeOutcome::Answered => "ok",
846                ProbeOutcome::NoAnswer => "failed",
847                ProbeOutcome::NotSent => "not_sent",
848            };
849            m.probe_latency_ms
850                .with_label_values(&[label])
851                .observe(started_at.elapsed().as_secs_f64() * 1000.0);
852        }
853        outcome
854    }

856    /// Take `entry` out of selection and fail its open requests.
857    fn quarantine(&self, entry: &PoolEntry, now: Instant) {
858        {
859            let mut health = entry.health.lock().unwrap();
860            health.quarantined_at = Some(now);
861            health.recent_quarantines.push_back(now);
862        }
863        entry.quarantined.store(true, Ordering::SeqCst);
864        entry.quarantine_generation.fetch_add(1, Ordering::SeqCst);
865        entry.quarantine_notify.notify_waiters();
866        if let Some(m) = &self.metrics {
867            m.channel_quarantines.inc();
868        }
869    }

871    /// Return a quarantined `entry` to selection.
872    fn restore(&self, entry: &PoolEntry) {
873        entry.health.lock().unwrap().quarantined_at = None;
874        entry.restart_silence();
875        entry.quarantined.store(false, Ordering::SeqCst);
876        if let Some(m) = &self.metrics {
877            m.channel_restores.inc();
878        }
879    }

881    /// Log when a pool-wide stall begins, and when the condition has been clear with responses
882    /// flowing for [`CIRCUIT_BREAKER_CLEAR_AFTER`]. A pool where nothing responds has not
883    /// recovered, so it never logs the end of a stall.
884    fn log_circuit_breaker(
885        &self,
886        pool_wide_stall: bool,
887        any_progressing: bool,
888        now: Instant,
889        stalled: usize,
890        quarantined: usize,
891        pool_size: usize,
892    ) {
893        let mut checker = self.health_checker.lock().unwrap();
894        if pool_wide_stall {
895            checker.circuit_breaker_clear_since = None;
896            if !checker.circuit_breaker_open {
897                checker.circuit_breaker_open = true;
898                warn!(
899                    stalled,
900                    quarantined,
901                    pool_size,
902                    "quarantined or stalled channels make up more than half the bigtable pool; BigTable or the network is the likely cause, so no channel is quarantined or replaced"
903                );
904            }
905            return;
906        }
907        if !checker.circuit_breaker_open {
908            return;
909        }
910        if !any_progressing {
911            checker.circuit_breaker_clear_since = None;
912            return;
913        }
914        let clear_since = *checker.circuit_breaker_clear_since.get_or_insert(now);
915        if now.saturating_duration_since(clear_since) >= CIRCUIT_BREAKER_CLEAR_AFTER {
916            checker.circuit_breaker_open = false;
917            checker.circuit_breaker_clear_since = None;
918            info!(
919                pool_size,
920                "bigtable channel stalls no longer span most of the pool"
921            );
922        }
923    }

977            in_flight: AtomicUsize::new(0),

980            last_response_nanos: AtomicU64::new(NO_RESPONSE),

993impl PoolEntry {
994    fn is_quarantined(&self) -> bool {
995        self.quarantined.load(Ordering::SeqCst)
996    }
997
998    /// Note response headers or a body frame.
999    fn record_response(&self) {
1000        let nanos = self.nanos_since_created(Instant::now());
1001        self.last_progress_nanos.store(nanos, Ordering::Relaxed);
1002        self.last_response_nanos.store(nanos, Ordering::Relaxed);
1003    }
1004
1005    /// Start counting silence afresh: a caller started waiting on an idle channel, or the channel
1006    /// answered a probe.
1007    fn restart_silence(&self) {
1008        let nanos = self.nanos_since_created(Instant::now());
1009        self.last_progress_nanos.store(nanos, Ordering::Relaxed);
1010    }
1011
1012    /// Give the channel an error penalty in selection for [`ERROR_PENALTY_DURATION`].
1013    fn penalize(&self) {
1014        let until = self.nanos_since_created(Instant::now() + ERROR_PENALTY_DURATION);
1015        self.error_penalty_until_nanos
1016            .store(until, Ordering::Relaxed);
1017    }
1018
1019    /// Whether the channel carries an error penalty in selection as of `now`.
1020    fn is_penalized(&self, now: Instant) -> bool {
1021        self.error_penalty_until_nanos.load(Ordering::Relaxed) > self.nanos_since_created(now)
1022    }
1023
1024    /// How long the channel has gone without progress as of `now`.
1025    fn silence(&self, now: Instant) -> Duration {
1026        let last_progress_at = self.created_at
1027            + Duration::from_nanos(self.last_progress_nanos.load(Ordering::Relaxed));
1028        now.saturating_duration_since(last_progress_at)
1029    }
1030
1031    /// How long since the channel's last response as of `now`; `Duration::MAX` if it has never
1032    /// responded.
1033    fn since_last_response(&self, now: Instant) -> Duration {
1034        match self.last_response_nanos.load(Ordering::Relaxed) {
1035            NO_RESPONSE => Duration::MAX,
1036            nanos => now.saturating_duration_since(self.created_at + Duration::from_nanos(nanos)),
1037        }
1038    }
1039
1040    fn nanos_since_created(&self, now: Instant) -> u64 {
1041        now.saturating_duration_since(self.created_at).as_nanos() as u64
1042    }
1043
1044    /// Claim a probe of this channel at `now`, unless it was probed less than `interval` ago.
1045    fn start_probe(&self, now: Instant, interval: Duration) -> bool {
1046        let mut health = self.health.lock().unwrap();
1047        if health
1048            .last_probe_at
1049            .is_some_and(|at| now.saturating_duration_since(at) < interval)
1050        {
1051            return false;
1052        }
1053        health.last_probe_at = Some(now);
1054        true
1055    }
1056
1057    fn quarantined_for(&self, now: Instant) -> Duration {
1058        self.health
1059            .lock()
1060            .unwrap()
1061            .quarantined_at
1062            .map_or(Duration::ZERO, |at| now.saturating_duration_since(at))
1063    }
1064
1065    fn recent_quarantine_count(&self, now: Instant, window: Duration) -> usize {
1066        let mut health = self.health.lock().unwrap();
1067        while health
1068            .recent_quarantines
1069            .front()
1070            .is_some_and(|&at| now.saturating_duration_since(at) > window)
1071        {
1072            health.recent_quarantines.pop_front();
1073        }
1074        health.recent_quarantines.len()
1075    }
1076}

1281impl Service<Request<Body>> for ChannelPool {
1282    type Response = Response<Body>;
1283    type Error = Box<dyn std::error::Error + Send + Sync>;
1284    type Future = Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send>>;
1285
1286    fn poll_ready(&mut self, _cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
1287        // The channel is chosen in `call`, which then waits for that channel to be ready.
1288        if self.inner.entries.load().is_empty() {
1289            return Poll::Ready(Err("no channels available".into()));
1290        }
1291        Poll::Ready(Ok(()))
1292    }
1293
1294    fn call(&mut self, request: Request<Body>) -> Self::Future {
1295        let routing = request.extensions().get::<ChannelRouting>();
1296        let Some(entry) = self.inner.select_channel(routing) else {
1297            return Box::pin(async { Err("no channels available".into()) });
1298        };
1299        if let Some(routing) = routing {
1300            let _ = routing.0.served_by.set((entry.id, Arc::downgrade(&entry)));
1301        }
1302        let mut guard = InFlightGuard::new(entry.clone());
1303        // Read the generation before checking the flag: a quarantine that lands after the check
1304        // changes the generation and cancels the request below, and one that landed before it is
1305        // caught by the check.
1306        let generation = entry.quarantine_generation.load(Ordering::SeqCst);
1307        let routed_to_quarantined = entry.is_quarantined();
1308        let metrics = self.inner.metrics.clone();
1309
1310        Box::pin(async move {
1311            let mut quarantined: QuarantineSignal =
1312                Box::pin(quarantined_after(entry.clone(), generation));
1313            let result = if routed_to_quarantined {
1314                None
1315            } else {
1316                guard.start_waiting();
1317                let result =
1318                    send_unless_quarantined(entry.channel.clone(), request, &mut quarantined).await;
1319                guard.stop_waiting();
1320                result
1321            };
1322            if let Some(m) = &metrics {
1323                m.rpcs_completed.inc();
1324            }
1325            match result {
1326                None => {
1327                    entry.error_count.fetch_add(1, Ordering::Relaxed);
1328                    if let Some(m) = &metrics {
1329                        m.requests_cancelled_by_quarantine.inc();
1330                    }
1331                    Err(quarantine_cancellation().into())
1332                }
1333                Some(Err(e)) => {
1334                    entry.error_count.fetch_add(1, Ordering::Relaxed);
1335                    entry.penalize();
1336                    Err(e.into())
1337                }
1338                Some(Ok(response)) => {
1339                    entry.success_count.fetch_add(1, Ordering::Relaxed);
1340                    entry.record_response();
1341                    if fails_with_penalized_status(&response) {
1342                        entry.penalize();
1343                    }
1344                    Ok(response.map(|body| {
1345                        Body::new(PoolResponseBody {
1346                            inner: body,
1347                            guard: Some(guard),
1348                            quarantine_generation: generation,
1349                            quarantined,
1350                            metrics,
1351                        })
1352                    }))
1353                }
1354            }
1355        })
1356    }
1357}

1361    /// reset, so nothing waits on the stalled connection.
1362    fn cancel_for_quarantine(&mut self) -> Status {

1364        self.guard = None;
1365        if let Some(m) = &self.metrics {
1366            m.requests_cancelled_by_quarantine.inc();
1367        }
1368        quarantine_cancellation()
1369    }

1372impl http_body::Body for PoolResponseBody {
1373    type Data = Bytes;
1374    type Error = Status;
1375
1376    fn poll_frame(
1377        self: Pin<&mut Self>,
1378        cx: &mut Context<'_>,
1379    ) -> Poll<Option<Result<Frame<Bytes>, Status>>> {
1380        let this = self.get_mut();
1381        let Some(guard) = this.guard.as_mut() else {
1382            return Poll::Ready(None);
1383        };
1384        if guard.entry.quarantine_generation.load(Ordering::SeqCst) != this.quarantine_generation {
1385            return Poll::Ready(Some(Err(this.cancel_for_quarantine())));
1386        }
1387        let frame = match Pin::new(&mut this.inner).poll_frame(cx) {
1388            Poll::Ready(frame) => frame,
1389            Poll::Pending => {
1390                if this.quarantined.as_mut().poll(cx).is_ready() {
1391                    return Poll::Ready(Some(Err(this.cancel_for_quarantine())));
1392                }
1393                guard.start_waiting();
1394                return Poll::Pending;
1395            }
1396        };
1397        guard.stop_waiting();
1398        match &frame {
1399            Some(Ok(frame)) => {
1400                guard.entry.record_response();
1401                // Trailers end the response and carry its status.
1402                if let Some(trailers) = frame.trailers_ref() {
1403                    if carries_penalized_status(trailers) {
1404                        guard.entry.penalize();
1405                    }
1406                    this.guard = None;
1407                }
1408            }
1409            Some(Err(_)) => {
1410                // The connection failed under the stream.
1411                guard.entry.penalize();
1412                this.guard = None;
1413            }
1414            None => this.guard = None,
1415        }
1416        Poll::Ready(frame)
1417    }
1418
1419    fn is_end_stream(&self) -> bool {
1420        self.guard.is_none() || self.inner.is_end_stream()
1421    }

1423    fn size_hint(&self) -> SizeHint {
1424        self.inner.size_hint()
1425    }

1532/// Pool size a resize moves to, or `None` when `current_size` already lies in the band of sizes
1533/// that keep `peak_in_flight` between `min_rpcs_per_channel` and `max_rpcs_per_channel` per
1534/// channel. Outside the band the size moves toward the band's midpoint by at most
1535/// `max_resize_delta`.
1536fn resize_target(peak_in_flight: usize, current_size: usize, config: &PoolConfig) -> Option<usize> {
1537    let fewest_channels = peak_in_flight
1538        .div_ceil(config.max_rpcs_per_channel.max(1))
1539        .max(config.min_pool_size)
1540        .min(config.max_pool_size);
1541    let most_channels = match config.min_rpcs_per_channel {
1542        0 => config.max_pool_size,
1543        per_channel => peak_in_flight
1544            .div_ceil(per_channel)
1545            .min(config.max_pool_size),
1546    }
1547    .max(fewest_channels);
1548    if (fewest_channels..=most_channels).contains(&current_size) {
1549        return None;
1550    }
1551    let midpoint = (fewest_channels + most_channels) / 2;
1552    let target = if midpoint > current_size {
1553        midpoint.min(current_size + config.max_resize_delta)
1554    } else {
1555        midpoint.max(current_size.saturating_sub(config.max_resize_delta))
1556    };
1557    (target != current_size).then_some(target)
1558}

1578/// Whether channel `idx` has at least `threshold` recent quarantines and at least `ratio` times
1579/// the mean count of the other channels. Comparing against the rest of the pool keeps channels
1580/// that merely shared a pool-wide bad spell from being replaced.
1581fn is_quarantine_outlier(counts: &[usize], idx: usize, threshold: usize, ratio: f64) -> bool {
1582    let own = counts[idx];
1583    if own < threshold {
1584        return false;
1585    }
1586    let others = counts.len().saturating_sub(1).max(1);
1587    let others_mean = (counts.iter().sum::<usize>() - own) as f64 / others as f64;
1588    own as f64 >= ratio * others_mean
1589}
