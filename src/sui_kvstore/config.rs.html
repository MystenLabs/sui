21/// Default maximum rows per BigTable write batch. Matches the official Google
22/// Java client default.
23pub(crate) const DEFAULT_MAX_ROWS_PER_BIGTABLE_BATCH: usize = 100;
24
25const DEFAULT_WRITE_CONCURRENCY: usize = 256;
26
27/// Minimum gap between two watermark writes of the same concurrent pipeline (sequential pipelines
28/// write their watermark with every commit). Shorter than the time between checkpoints, so at the
29/// tip every checkpoint's watermark is written as soon as it is complete.
30const DEFAULT_WATERMARK_INTERVAL_MS: u64 = 100;
31
32/// Returns the committer defaults used by sui-kvstore before applying config overrides.
33pub fn default_committer_config() -> CommitterConfig {
34    CommitterConfig {
35        write_concurrency: DEFAULT_WRITE_CONCURRENCY,
36        watermark_interval_ms: DEFAULT_WATERMARK_INTERVAL_MS,
37        ..CommitterConfig::default()
38    }
39}

41#[derive(Clone, Debug, Deserialize, Serialize)]
42#[serde(default, rename_all = "kebab-case")]
43pub struct IndexerConfig {
44    pub ingestion: IngestionConfig,
45    pub committer: CommitterLayer,
46    pub pipeline: PipelineLayer,
47    /// Global rate limit (rows per second) shared across all pipelines.
48    pub total_max_rows_per_second: Option<u64>,
49    /// Default per-pipeline rate limit (rows per second).
50    /// Individual pipelines can override via their `ConcurrentLayer`.
51    pub max_rows_per_second: Option<u64>,
52    /// Deprecated: use `bigtable-pool` section instead. If set, overrides
53    /// `bigtable-pool.initial-pool-size`. Will be removed in a future release.
54    pub bigtable_connection_pool_size: Option<usize>,
55    /// Channel-level timeout in milliseconds for BigTable gRPC calls (default: 60000).
56    pub bigtable_channel_timeout_ms: Option<u64>,
57    /// Longest a ReadRows attempt may wait for its response headers or its next response message
58    /// before it is retried on another channel, in milliseconds (default: 5000). The final
59    /// attempt has no gap timeout. A stall before the headers is retried only while
60    /// `bigtable-channel-timeout-ms`, which ends that wait on every attempt, is longer.
61    pub bigtable_read_rows_gap_timeout_ms: Option<u64>,
62    /// Attempts per ReadRows call, including the first (default: 3). 1 turns off the gap
63    /// timeout and retries.
64    pub bigtable_read_rows_max_attempts: Option<u32>,
65    /// Enable Bigtable batch write flow control by advertising the mutate-rows
66    /// rate-limit feature flags and adaptively throttling MutateRows from
67    /// `RateLimitInfo`. Requires a single-cluster-routing app profile and pairs
68    /// with Bigtable autoscaling. Enabled by default; set to false to disable.
69    pub batch_write_flow_control: bool,
70    /// Bigtable connection pool configuration.
71    pub bigtable_pool: BigtablePoolLayer,
72}

106#[derive(Clone, Default, Debug, Deserialize, Serialize)]
107#[serde(default, rename_all = "kebab-case")]
108pub struct BigtablePoolLayer {
109    /// Number of channels to create at startup (default: 10).
110    pub initial_pool_size: Option<usize>,
111    /// Minimum number of channels the pool will maintain (default: 4).
112    pub min_pool_size: Option<usize>,
113    /// Maximum number of channels the pool can scale to (default: 200).
114    pub max_pool_size: Option<usize>,
115    /// The pool shrinks when a maintenance interval's peak in-flight requests would leave
116    /// fewer than this many per channel (default: 1).
117    pub min_rpcs_per_channel: Option<usize>,
118    /// The pool grows when a maintenance interval's peak in-flight requests would put more
119    /// than this many on each channel (default: 25). Channel selection also counts a channel
120    /// carrying an error penalty as this many requests busier.
121    pub max_rpcs_per_channel: Option<usize>,
122    /// Maximum channels to add or remove in a single resize (default: 2).
123    pub max_resize_delta: Option<usize>,
124    /// Milliseconds between maintenance cycles (resize + refresh) (default: 60000).
125    pub maintenance_interval_ms: Option<u64>,
126    /// Channel age in milliseconds before it is eligible for refresh (default: 2700000 = 45 min).
127    pub refresh_age_ms: Option<u64>,
128    /// Random jitter in milliseconds added to refresh age (default: 300000 = 5 min).
129    pub refresh_jitter_ms: Option<u64>,
130    /// Detect stalled channels, quarantine them, and replace ones that stay unhealthy
131    /// (default: true).
132    pub channel_health_check: Option<bool>,
133    /// HTTP/2 receive window of each BigTable response stream, in bytes (default: 262144 =
134    /// 256 KiB).
135    pub http2_stream_window_bytes: Option<u32>,
136    /// HTTP/2 receive window of each BigTable connection, in bytes (default: 268435456 =
137    /// 256 MiB). While unread response data fills it, no request on that channel receives
138    /// anything.
139    pub http2_connection_window_bytes: Option<u32>,
140
141    /// Deprecated and ignored; setting it logs a warning. The pool sizes itself on
142    /// `min-rpcs-per-channel` and `max-rpcs-per-channel`.
143    pub downscale_threshold: Option<usize>,
144}
145
146impl BigtablePoolLayer {
147    pub fn finish(self, deprecated_pool_size: Option<usize>) -> PoolConfig {
148        if deprecated_pool_size.is_some() {
149            warn!(
150                "bigtable-connection-pool-size is deprecated; \
151                 use the [bigtable-pool] section instead"
152            );
153        }
154        if self.downscale_threshold.is_some() {
155            warn!(
156                "Config field `downscale-threshold` in [bigtable-pool] is deprecated and ignored. \
157                 Remove it from your config; the pool resizes on `min-rpcs-per-channel` and \
158                 `max-rpcs-per-channel`."
159            );
160        }
161
162        let base = PoolConfig::default();
163        PoolConfig {
164            initial_pool_size: self
165                .initial_pool_size
166                .or(deprecated_pool_size)
167                .unwrap_or(base.initial_pool_size),
168            min_pool_size: self.min_pool_size.unwrap_or(base.min_pool_size),
169            max_pool_size: self.max_pool_size.unwrap_or(base.max_pool_size),
170            min_rpcs_per_channel: self
171                .min_rpcs_per_channel
172                .unwrap_or(base.min_rpcs_per_channel),
173            max_rpcs_per_channel: self
174                .max_rpcs_per_channel
175                .unwrap_or(base.max_rpcs_per_channel),
176            max_resize_delta: self.max_resize_delta.unwrap_or(base.max_resize_delta),
177            maintenance_interval: self
178                .maintenance_interval_ms
179                .map(Duration::from_millis)
180                .unwrap_or(base.maintenance_interval),
181            refresh_age: self
182                .refresh_age_ms
183                .map(Duration::from_millis)
184                .unwrap_or(base.refresh_age),
185            refresh_jitter: self
186                .refresh_jitter_ms
187                .map(Duration::from_millis)
188                .unwrap_or(base.refresh_jitter),
189            health_check: base
190                .health_check
191                .filter(|_| self.channel_health_check.unwrap_or(true)),
192            http2_stream_window_bytes: self
193                .http2_stream_window_bytes
194                .unwrap_or(base.http2_stream_window_bytes),
195            http2_connection_window_bytes: self
196                .http2_connection_window_bytes
197                .unwrap_or(base.http2_connection_window_bytes),
198        }
199    }
200}

202#[derive(Clone, Default, Debug, Deserialize, Serialize)]
203#[serde(default, rename_all = "kebab-case")]
204pub struct CommitterLayer {
205    pub write_concurrency: Option<usize>,
206    pub watermark_interval_ms: Option<u64>,
207    pub watermark_interval_jitter_ms: Option<u64>,
208
209    /// Minimum number of pending rows the collector waits for before cutting a batch. `0` cuts a
210    /// batch as soon as the committer has room for it.
211    pub min_batch_rows: Option<usize>,
212
213    /// Longest time, in milliseconds, the oldest pending row waits for `min-batch-rows` to
214    /// accumulate before a batch is cut anyway.
215    pub max_batch_delay_ms: Option<u64>,
216
217    /// Deprecated and ignored: batches are cut as soon as the committer has room; see
218    /// `min-batch-rows` and `max-batch-delay-ms`.
219    pub collect_interval_ms: Option<u64>,
220}
221
222impl CommitterLayer {
223    pub fn finish(self, base: CommitterConfig) -> CommitterConfig {
224        if self.collect_interval_ms.is_some() {
225            warn_ignored_config_key("collect-interval-ms");
226        }
227
228        CommitterConfig {
229            write_concurrency: self.write_concurrency.unwrap_or(base.write_concurrency),
230            watermark_interval_ms: self
231                .watermark_interval_ms
232                .unwrap_or(base.watermark_interval_ms),
233            watermark_interval_jitter_ms: self
234                .watermark_interval_jitter_ms
235                .unwrap_or(base.watermark_interval_jitter_ms),
236            min_batch_rows: self.min_batch_rows.unwrap_or(base.min_batch_rows),
237            max_batch_delay_ms: self.max_batch_delay_ms.unwrap_or(base.max_batch_delay_ms),
238        }
239    }
240}

263#[derive(Clone, Default, Debug, Deserialize, Serialize)]
264#[serde(default, rename_all = "kebab-case")]
265pub struct PipelineIngestionLayer {
266    pub subscriber_channel_size: Option<usize>,
267}
268
269impl ConcurrentLayer {
270    pub(crate) fn max_rows_or_default(&self) -> usize {
271        self.max_rows.unwrap_or(DEFAULT_MAX_ROWS_PER_BIGTABLE_BATCH)
272    }
273
274    pub fn finish(self, base: ConcurrentConfig) -> ConcurrentConfig {
275        if self.min_eager_rows.is_some() {
276            warn_ignored_config_key("min-eager-rows");
277        }
278
279        ConcurrentConfig {
280            committer: if let Some(c) = self.committer {
281                c.finish(base.committer)
282            } else {
283                base.committer
284            },
285            ingestion: if let Some(i) = self.ingestion {
286                i.finish(base.ingestion)
287            } else {
288                base.ingestion
289            },
290            pruner: None,
291            fanout: self.fanout.or(base.fanout),
292            max_pending_rows: self.max_pending_rows.or(base.max_pending_rows),
293            max_watermark_updates: self.max_watermark_updates.or(base.max_watermark_updates),
294            processor_channel_size: self.processor_channel_size.or(base.processor_channel_size),
295            collector_channel_size: self.collector_channel_size.or(base.collector_channel_size),
296            committer_channel_size: self.committer_channel_size.or(base.committer_channel_size),
297        }
298    }
299}

311#[derive(Clone, Default, Debug, Deserialize, Serialize)]
312#[serde(default, rename_all = "kebab-case")]
313pub struct SequentialLayer {
314    // Framework sequential surface — mirrors the fields actually read by
315    // `sui_indexer_alt_framework::pipeline::sequential`.
316    pub committer: Option<CommitterLayer>,
317    pub ingestion: Option<PipelineIngestionLayer>,
318    pub fanout: Option<ConcurrencyConfig>,
319    /// Deprecated and ignored: batch size is controlled by the committer's `min-batch-rows` and
320    /// `max-batch-delay-ms`.
321    pub min_eager_rows: Option<usize>,
322    pub max_pending_rows: Option<usize>,
323    pub max_batch_checkpoints: Option<usize>,
324    pub processor_channel_size: Option<usize>,
325    pub pipeline_depth: Option<usize>,
326
327    // sui-kvstore-specific config extensions
328
329    // Controls the concurrency of the bitmap flushes. The framework doesn't
330    // perform commits for sequential pipelines concurrently, but our store
331    // implementation doesn't actually write to the database on commit for
332    // the bitmap pipelines. The store buffers the bitmaps for the current
333    // working "bucket" ranges internally, merges in rows from each framework
334    // batch on commit (parallelized on background tasks), then finally flushes
335    // the updated bitmaps to bigtable concurrently. Same semantic as
336    // `ConcurrentLayer::write_concurrency`.
337    pub write_concurrency: Option<usize>,
338    /// Maximum rows per in-handler BigTable write RPC. Same semantic as
339    /// `ConcurrentLayer::max_rows`.
340    pub max_rows: Option<usize>,
341    /// Per-pipeline rate limit (rows per second). Overrides the default
342    /// `IndexerConfig::max_rows_per_second` when set.
343    pub max_rows_per_second: Option<u64>,
344}
345
346impl SequentialLayer {
347    pub(crate) fn max_rows_or_default(&self) -> usize {
348        self.max_rows.unwrap_or(DEFAULT_MAX_ROWS_PER_BIGTABLE_BATCH)
349    }
350
351    pub fn finish(self, base: ConcurrentConfig) -> SequentialConfig {
352        if self.min_eager_rows.is_some() {
353            warn_ignored_config_key("min-eager-rows");
354        }
355
356        let committer = if let Some(c) = self.committer {
357            c.finish(base.committer)
358        } else {
359            base.committer
360        };
361        SequentialConfig {
362            committer,
363            ingestion: if let Some(i) = self.ingestion {
364                i.finish(base.ingestion)
365            } else {
366                base.ingestion
367            },
368            fanout: self.fanout.or(base.fanout),
369            max_pending_rows: self.max_pending_rows.or(base.max_pending_rows),
370            max_batch_checkpoints: self.max_batch_checkpoints,
371            processor_channel_size: self.processor_channel_size.or(base.processor_channel_size),
372            pipeline_depth: self.pipeline_depth,
373        }
374    }
375}

398/// This type is identical to [`framework::ingestion::IngestionConfig`], but is set-up to be
399/// serialized and deserialized by `serde`.
400#[derive(Clone, Debug, Deserialize, Serialize)]
401#[serde(default, rename_all = "kebab-case", deny_unknown_fields)]
402pub struct IngestionConfig {
403    pub ingest_concurrency: framework::config::ConcurrencyConfig,
404    pub retry_interval_ms: u64,
405    pub streaming_backoff_initial_batch_size: NonZeroUsize,
406    pub streaming_backoff_max_batch_size: usize,
407    pub streaming_connection_timeout_ms: u64,
408    pub streaming_statement_timeout_ms: u64,
409    pub min_cohort_boundary: u64,
410
411    /// Deprecated: accepted (and ignored) so old configs don't fail to parse. Replaced by
412    /// per-pipeline `ingestion.subscriber-channel-size`.
413    pub checkpoint_buffer_size: Option<usize>,
414}
415
416impl Default for IngestionConfig {
417    fn default() -> Self {
418        framework::ingestion::IngestionConfig::default().into()
419    }
420}

422impl From<framework::ingestion::IngestionConfig> for IngestionConfig {

424        Self {

426            retry_interval_ms: config.retry_interval_ms,
427            streaming_backoff_initial_batch_size: config.streaming_backoff_initial_batch_size,
428            streaming_backoff_max_batch_size: config.streaming_backoff_max_batch_size,
429            streaming_connection_timeout_ms: config.streaming_connection_timeout_ms,
430            streaming_statement_timeout_ms: config.streaming_statement_timeout_ms,
431            min_cohort_boundary: config.min_cohort_boundary,
432            checkpoint_buffer_size: None,
433        }
434    }

437impl From<IngestionConfig> for framework::ingestion::IngestionConfig {
438    fn from(config: IngestionConfig) -> Self {
439        if config.checkpoint_buffer_size.is_some() {
440            warn!(
441                "Config field `checkpoint-buffer-size` is deprecated and ignored. Remove it from \
442                 your config; set `subscriber-channel-size` under each pipeline's `ingestion` \
443                 section if you need to override the default."
444            );
445        }
446
447        framework::ingestion::IngestionConfig {
448            ingest_concurrency: config.ingest_concurrency,
449            retry_interval_ms: config.retry_interval_ms,
450            streaming_backoff_initial_batch_size: config.streaming_backoff_initial_batch_size,
451            streaming_backoff_max_batch_size: config.streaming_backoff_max_batch_size,
452            streaming_connection_timeout_ms: config.streaming_connection_timeout_ms,
453            streaming_statement_timeout_ms: config.streaming_statement_timeout_ms,
454            min_cohort_boundary: config.min_cohort_boundary,
455        }
456    }
