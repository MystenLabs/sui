15/// Pipeline that writes a tombstone to the `objects` table at `(object_id, version)` for every
16/// object a transaction deletes or wraps, so version-bounded reads stop at the removal instead
17/// of returning the object's last live version. Live rows are written by
18/// [`crate::ObjectsPipeline`].
19pub struct ObjectTombstonesPipeline;
20
21#[async_trait::async_trait]
22impl Processor for ObjectTombstonesPipeline {
23    const NAME: &'static str = "kvstore_object_tombstones";
24    type Value = Entry;
25
26    async fn process(&self, checkpoint: &Arc<Checkpoint>) -> anyhow::Result<Vec<Self::Value>> {
27        let timestamp_ms = checkpoint.summary.timestamp_ms;
28        let mut entries = vec![];
29        for txn in &checkpoint.transactions {
30            for (id, version, status) in removed_objects(&txn.effects) {
31                entries.push(tables::make_entry(
32                    tables::objects::encode_key(&ObjectKey(id, version)),
33                    tables::objects::encode_tombstone(status)?,
34                    Some(timestamp_ms),
35                ));
36            }
37        }
38        Ok(entries)
39    }
40}
41
42impl BigTableProcessor for ObjectTombstonesPipeline {
43    const TABLE: &'static str = tables::objects::NAME;
44}

46#[cfg(test)]
47mod tests {
48    use std::collections::BTreeMap;
49
50    use sui_types::base_types::SequenceNumber;
51    use sui_types::effects::TransactionEffectsAPI;
52    use sui_types::object::Object;
53    use sui_types::test_checkpoint_data_builder::TestCheckpointBuilder;
54
55    use super::*;
56    use crate::CheckpointSpan;
57    use crate::KeyValueStoreReader;
58    use crate::ObjectsPipeline;
59    use crate::bigtable::client::BigTableClient;
60    use crate::bigtable::mock_server::MockBigtableServer;
61
62    /// An object wrapped and unwrapped by different transactions of one checkpoint: the
63    /// checkpoint's final state shows it live, but reads bounded to a version between the wrap
64    /// and the unwrap must not find it.
65    #[tokio::test]
66    async fn wrap_then_unwrap_in_one_checkpoint_hides_object_between_them() {
67        let mut builder = TestCheckpointBuilder::new(0)
68            .start_transaction(0)
69            .create_owned_object(0)
70            .finish_transaction();
71        let cp0 = Arc::new(builder.build_checkpoint());
72        let cp1 = Arc::new(
73            builder
74                .start_transaction(0)
75                .transfer_object(0, 1)
76                .finish_transaction()
77                .start_transaction(0)
78                .wrap_object(0)
79                .finish_transaction()
80                .start_transaction(0)
81                .unwrap_object(0)
82                .finish_transaction()
83                .build_checkpoint(),
84        );
85        let id = TestCheckpointBuilder::derive_object_id(0);
86        let [created, transferred, wrapped, unwrapped] = [
87            &cp0.transactions[0],
88            &cp1.transactions[0],
89            &cp1.transactions[1],
90            &cp1.transactions[2],
91        ]
92        .map(|txn| txn.effects.lamport_version());
93        assert!(created < transferred && transferred < wrapped && wrapped < unwrapped);
94
95        let mock = MockBigtableServer::new();
96        let (addr, _handle) = mock.start().await.unwrap();
97        let mut client =
98            BigTableClient::new_for_host(addr.to_string(), "test".to_string(), "test", false)
99                .await
100                .unwrap();
101        for checkpoint in [&cp0, &cp1] {
102            let mut entries = ObjectsPipeline.process(checkpoint).await.unwrap();
103            entries.extend(ObjectTombstonesPipeline.process(checkpoint).await.unwrap());
104            client
105                .write_entries(
106                    tables::objects::NAME,
107                    entries,
108                    CheckpointSpan::single(checkpoint.summary.sequence_number),
109                )
110                .await
111                .unwrap();
112        }
113
114        // The object's state at each version that changed it; `None` once wrapped.
115        let changes: BTreeMap<SequenceNumber, Option<SequenceNumber>> = [
116            (created, Some(created)),
117            (transferred, Some(transferred)),
118            (wrapped, None),
119            (unwrapped, Some(unwrapped)),
120        ]
121        .into();
122        let version_of = |object: Option<Object>| object.map(|o| (o.id(), o.version()));
123        for version in 0..=unwrapped.value() + 1 {
124            let version = SequenceNumber::from_u64(version);
125            let expected = changes
126                .range(..=version)
127                .next_back()
128                .and_then(|(_, live)| live.map(|live| (id, live)));
129            let found = client
130                .get_object_lt_or_eq_version(id, version)
131                .await
132                .unwrap();
133            assert_eq!(version_of(found), expected, "at version {version}");
134        }
135
136        let latest = client.get_latest_object(&id).await.unwrap();
137        assert_eq!(version_of(latest), Some((id, unwrapped)));
138        assert!(
139            client
140                .get_objects(&[ObjectKey(id, wrapped)])
141                .await
142                .unwrap()
143                .is_empty(),
144            "the tombstone is not an object",
145        );
146
147        let live_versions: Vec<_> = client
148            .get_live_object_versions(id, SequenceNumber::MIN..=SequenceNumber::MAX, 10, false)
149            .await
150            .unwrap()
151            .into_iter()
152            .map(|o| o.version())
153            .collect();
154        assert_eq!(live_versions, vec![created, transferred, unwrapped]);
155    }
