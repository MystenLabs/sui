19/// Pipeline that writes, for every object a checkpoint changes, the object's version and status
20/// at the end of that checkpoint to the `object_versions_by_checkpoint` table.
21pub struct ObjectVersionsByCheckpointPipeline;
22
23#[async_trait::async_trait]
24impl Processor for ObjectVersionsByCheckpointPipeline {
25    const NAME: &'static str = "kvstore_object_versions_by_checkpoint";
26    type Value = Entry;
27
28    async fn process(&self, checkpoint: &Arc<Checkpoint>) -> anyhow::Result<Vec<Self::Value>> {
29        let cp_sequence_number = checkpoint.summary.sequence_number;
30        let timestamp_ms = checkpoint.summary.timestamp_ms;
31
32        // Transactions in a checkpoint are in causal order, so the last transaction to touch an
33        // object determines its state at the end of the checkpoint.
34        let mut final_states: HashMap<ObjectID, (SequenceNumber, ObjectStatus)> = HashMap::new();
35        for txn in &checkpoint.transactions {
36            for change in txn.effects.object_changes() {
37                if let Some(version) = change.output_version {
38                    final_states.insert(change.id, (version, ObjectStatus::Live));
39                }
40            }
41            for (id, version, status) in removed_objects(&txn.effects) {
42                final_states.insert(id, (version, status));
43            }
44        }
45
46        Ok(final_states
47            .into_iter()
48            .map(|(id, (version, status))| {
49                tables::make_entry(
50                    tables::object_versions_by_checkpoint::encode_key(&id, cp_sequence_number),
51                    tables::object_versions_by_checkpoint::encode(version, status),
52                    Some(timestamp_ms),
53                )
54            })
55            .collect())
56    }
57}

61}

63#[cfg(test)]
64mod tests {
65    use bytes::Bytes;
66    use sui_types::test_checkpoint_data_builder::TestCheckpointBuilder;
67
68    use super::*;
69    use crate::bigtable::proto::bigtable::v2::mutation;

71    /// Decode every entry back into `(object_id, checkpoint) -> (version, status)`.
72    fn decode_entries(
73        entries: Vec<Entry>,
74    ) -> HashMap<(ObjectID, u64), (SequenceNumber, ObjectStatus)> {
75        entries
76            .into_iter()
77            .map(|entry| {
78                let key = entry.row_key.as_ref();
79                let object_id = ObjectID::from_bytes(&key[..ObjectID::LENGTH]).unwrap();
80                let cp = u64::from_be_bytes(key[ObjectID::LENGTH..].try_into().unwrap());
81                let cells: Vec<(Bytes, Bytes)> = entry
82                    .mutations
83                    .into_iter()
84                    .map(|m| match m.mutation {
85                        Some(mutation::Mutation::SetCell(cell)) => {
86                            (cell.column_qualifier, cell.value)
87                        }
88                        other => panic!("unexpected mutation {other:?}"),
89                    })
90                    .collect();
91                let value = tables::object_versions_by_checkpoint::decode(&cells).unwrap();
92                ((object_id, cp), value)
93            })
94            .collect()
95    }
96
97    /// Each object a checkpoint touches gets exactly one row holding its state after the
98    /// checkpoint's last transaction, even when several transactions change it.
99    #[tokio::test]
100    async fn records_final_state_of_each_changed_object() {
101        let mut builder = TestCheckpointBuilder::new(0)
102            .start_transaction(0)
103            .create_owned_object(0)
104            .create_owned_object(1)
105            .create_owned_object(2)
106            .create_owned_object(3)
107            .finish_transaction();
108        builder.build_checkpoint();
109
110        // Checkpoint 1, across two transactions:
111        // - object 0 is mutated twice,
112        // - object 1 is deleted,
113        // - object 2 is mutated and then wrapped,
114        // - object 3 is wrapped and then unwrapped.
115        let cp1 = Arc::new(
116            builder
117                .start_transaction(0)
118                .transfer_object(0, 1)
119                .delete_object(1)
120                .transfer_object(2, 1)
121                .wrap_object(3)
122                .finish_transaction()
123                .start_transaction(1)
124                .transfer_object(0, 2)
125                .wrap_object(2)
126                .unwrap_object(3)
127                .finish_transaction()
128                .build_checkpoint(),
129        );
130        let first_lamport = cp1.transactions[0].effects.lamport_version();
131        let second_lamport = cp1.transactions[1].effects.lamport_version();
132        assert!(first_lamport < second_lamport);
133
134        let entries = ObjectVersionsByCheckpointPipeline
135            .process(&cp1)
136            .await
137            .unwrap();
138        let entry_count = entries.len();
139        let rows = decode_entries(entries);
140        assert_eq!(entry_count, rows.len(), "one row per object");
141
142        let object = |idx| (TestCheckpointBuilder::derive_object_id(idx), 1);
143        assert_eq!(rows[&object(0)], (second_lamport, ObjectStatus::Live));
144        assert_eq!(rows[&object(1)], (first_lamport, ObjectStatus::Deleted));
145        assert_eq!(rows[&object(2)], (second_lamport, ObjectStatus::Wrapped));
146        assert_eq!(rows[&object(3)], (second_lamport, ObjectStatus::Live));
147    }
