4//! [`ObjectStatus`], stored as one byte by the `object_versions_by_checkpoint` table and by
5//! tombstone rows in the `objects` table, and [`removed_objects`], which derives the removal
6//! statuses from transaction effects for the pipelines that write those tables.
7
8use anyhow::{Result, bail};
9use sui_types::base_types::{ObjectID, SequenceNumber};
10use sui_types::effects::{TransactionEffects, TransactionEffectsAPI};
11
12/// Whether an object exists at a version, or how it was removed at that version.
13#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
14pub enum ObjectStatus {
15    /// The object exists; its contents are in the `objects` table at that version.
16    Live,
17    /// The object was deleted at that version, including an object unwrapped and deleted by the
18    /// same transaction. The `objects` row at that version, if written, is a tombstone.
19    Deleted,
20    /// The object was wrapped into another object at that version. The `objects` row at that
21    /// version, if written, is a tombstone.
22    Wrapped,
23}
24
25impl ObjectStatus {
26    pub(crate) fn to_byte(self) -> u8 {
27        match self {
28            ObjectStatus::Live => 0,
29            ObjectStatus::Deleted => 1,
30            ObjectStatus::Wrapped => 2,
31        }
32    }
33
34    pub(crate) fn from_byte(byte: u8) -> Result<Self> {
35        Ok(match byte {
36            0 => ObjectStatus::Live,
37            1 => ObjectStatus::Deleted,
38            2 => ObjectStatus::Wrapped,
39            _ => bail!("invalid object status byte {byte}"),
40        })
41    }
42}
43
44/// The objects `effects` removed from the live set, each with the version of its removal and its
45/// status, `Deleted` or `Wrapped`.
46pub(crate) fn removed_objects(
47    effects: &TransactionEffects,
48) -> impl Iterator<Item = (ObjectID, SequenceNumber, ObjectStatus)> + use<> {
49    // Take the removal version from each ref rather than `effects.lamport_version()`: for V1
50    // effects, that recomputes the lamport version from the modified inputs only, which can be
51    // lower than the version the transaction was executed at.
52    let deleted = effects
53        .deleted()
54        .into_iter()
55        .chain(effects.unwrapped_then_deleted())
56        .map(|(id, version, _)| (id, version, ObjectStatus::Deleted));
57    let wrapped = effects
58        .wrapped()
59        .into_iter()
60        .map(|(id, version, _)| (id, version, ObjectStatus::Wrapped));
61    deleted.chain(wrapped)
62}

64#[cfg(test)]
65mod tests {
66    use std::collections::HashSet;
67
68    use sui_types::base_types::SuiAddress;
69    use sui_types::base_types::random_object_ref;
70    use sui_types::digests::ObjectDigest;
71    use sui_types::digests::TransactionDigest;
72    use sui_types::execution_status::ExecutionStatus;
73    use sui_types::gas::GasCostSummary;
74    use sui_types::object::Owner;
75
76    use super::*;
77
78    /// Every kind of removal is reported once, at the version in its ref, even where that differs
79    /// from the version `lamport_version()` recomputes for V1 effects.
80    #[test]
81    fn removed_objects_reports_each_removal_at_its_ref_version() {
82        let deleted = ObjectID::random();
83        let unwrapped_then_deleted = ObjectID::random();
84        let wrapped = ObjectID::random();
85        let removal_version = SequenceNumber::from_u64(10);
86        let effects = TransactionEffects::new_from_execution_v1(
87            ExecutionStatus::Success,
88            0,
89            GasCostSummary::default(),
90            vec![(deleted, SequenceNumber::from_u64(3))],
91            vec![],
92            TransactionDigest::random(),
93            vec![],
94            vec![],
95            vec![],
96            vec![(
97                deleted,
98                removal_version,
99                ObjectDigest::OBJECT_DIGEST_DELETED,
100            )],
101            vec![(
102                unwrapped_then_deleted,
103                removal_version,
104                ObjectDigest::OBJECT_DIGEST_DELETED,
105            )],
106            vec![(
107                wrapped,
108                removal_version,
109                ObjectDigest::OBJECT_DIGEST_WRAPPED,
110            )],
111            (random_object_ref(), Owner::AddressOwner(SuiAddress::ZERO)),
112            None,
113            vec![],
114        );
115        assert_ne!(effects.lamport_version(), removal_version);
116
117        let removed: HashSet<_> = removed_objects(&effects).collect();
118        assert_eq!(
119            removed,
120            HashSet::from([
121                (deleted, removal_version, ObjectStatus::Deleted),
122                (
123                    unwrapped_then_deleted,
124                    removal_version,
125                    ObjectStatus::Deleted
126                ),
127                (wrapped, removal_version, ObjectStatus::Wrapped),
128            ]),
129        );
130    }
