4//! Follows the `WatermarkService` streams of kvstore instances and merges their durable
5//! watermarks. A kvstore instance follows its peers to compare its own progress against them
6//! ([`crate::readiness`]), and a reader follows the instances that write the data it serves to
7//! learn how far that data goes.
8

27/// How long to wait for a TCP connection to an instance.
28const CONNECT_TIMEOUT: Duration = Duration::from_secs(5);
29
30/// How long an attempt to subscribe to an instance may take, from connecting until its first
31/// response, before it counts as failed.
32const FIRST_RESPONSE_TIMEOUT: Duration = Duration::from_secs(5);
33
34/// Time between HTTP/2 pings on a connection to an instance. An instance that disappears without
35/// closing the connection (for example when its node fails) stops acknowledging them.
36const KEEPALIVE_INTERVAL: Duration = Duration::from_secs(10);
37
38/// How long to wait for an instance to acknowledge a ping before the connection counts as failed
39/// and the instance is subscribed to again.
40const KEEPALIVE_TIMEOUT: Duration = Duration::from_secs(5);
41
42/// Time between an instance's stream ending or failing and the next attempt to subscribe.
43const RESUBSCRIBE_INTERVAL: Duration = Duration::from_secs(2);
44
45/// What the followed instances have reported.
46#[derive(Clone, Debug, Default, PartialEq)]
47pub struct StreamedWatermarks {
48    /// Instances that have neither delivered a first response nor failed a first attempt to
49    /// subscribe. Each instance's first attempt is bounded by `CONNECT_TIMEOUT` and
50    /// `FIRST_RESPONSE_TIMEOUT`, so this drops to zero shortly after startup.
51    pub unsettled_instances: usize,
52
53    /// Per pipeline, the durable watermark with the highest checkpoint any instance has reported.
54    /// Entries only move forward and outlive the stream that reported them.
55    pub watermarks: HashMap<String, CommitterWatermark>,
56}
57
58/// Subscribe to the `WatermarkService` of every kvstore instance in `urls`.
59///
60/// The returned service follows each instance, subscribing again after its stream ends or fails,
61/// until every receiver is dropped. It is aborted on shutdown. Fails if a URL is not an `http://`
62/// URL with a host.
63pub fn follow(urls: &[String]) -> anyhow::Result<(watch::Receiver<StreamedWatermarks>, Service)> {
64    let endpoints = urls
65        .iter()
66        .map(|url| {
67            instance_endpoint(url).with_context(|| format!("Invalid kvstore instance URL {url:?}"))
68        })
69        .collect::<anyhow::Result<Vec<_>>>()?;
70
71    let (sender, receiver) = watch::channel(StreamedWatermarks {
72        unsettled_instances: endpoints.len(),
73        watermarks: HashMap::new(),
74    });
75
76    let sender = Arc::new(sender);
77    let mut service = Service::new();
78    for endpoint in endpoints {
79        service = service.spawn_aborting(follow_instance(endpoint, sender.clone()));
80    }
81
82    Ok((receiver, service))
83}

85/// The endpoint for the kvstore instance at `url`. Instances are reached over plaintext HTTP/2, so
86/// `url` must be an `http://` URL with a host; anything else would fail on every attempt to
87/// connect.
88fn instance_endpoint(url: &str) -> anyhow::Result<Endpoint> {
89    let endpoint = Endpoint::from_shared(url.to_string())?;
90    let uri = endpoint.uri();
91    anyhow::ensure!(
92        uri.scheme_str() == Some("http") && uri.host().is_some(),
93        "expected an http:// URL with a host"
94    );
95
96    // With only a response stream open, hyper treats the connection as idle and sends no pings
97    // unless pings while idle are enabled.
98    Ok(endpoint
99        .connect_timeout(CONNECT_TIMEOUT)
100        .http2_keep_alive_interval(KEEPALIVE_INTERVAL)
101        .keep_alive_timeout(KEEPALIVE_TIMEOUT)
102        .keep_alive_while_idle(true))
103}

105async fn follow_instance(
106    endpoint: Endpoint,
107    sender: Arc<watch::Sender<StreamedWatermarks>>,
108) -> anyhow::Result<()> {
109    let url = endpoint.uri().to_string();
110    let mut settled = false;
111    let mut failing = false;
112
113    loop {
114        let result = tokio::select! {
115            _ = sender.closed() => return Ok(()),
116            result = subscribe_once(&endpoint, &sender, &mut settled, &mut failing) => result,
117        };
118
119        match result {
120            Ok(()) => info!(url, "Watermark stream ended"),
121            // Repeated failures (for example while an instance is restarting) are logged once.
122            Err(e) if failing => debug!(url, "Failed to follow watermark stream: {e:#}"),
123            Err(e) => {
124                failing = true;
125                warn!(url, "Failed to follow watermark stream: {e:#}");
126            }
127        }
128
129        if !settled {
130            settled = true;
131            sender.send_modify(|state| state.unsettled_instances -= 1);
132        }
133
134        tokio::select! {
135            _ = sender.closed() => return Ok(()),
136            _ = tokio::time::sleep(RESUBSCRIBE_INTERVAL) => {}
137        }
138    }
139}

141/// Subscribe to the instance at `endpoint` and merge its responses into `sender` until the stream
142/// ends. When the first response arrives, marks the instance `settled` and clears `failing`, so
143/// that the next failure is logged as a new one.
144async fn subscribe_once(
145    endpoint: &Endpoint,
146    sender: &watch::Sender<StreamedWatermarks>,
147    settled: &mut bool,
148    failing: &mut bool,
149) -> anyhow::Result<()> {
150    let (mut stream, first_response) = tokio::time::timeout(FIRST_RESPONSE_TIMEOUT, async {
151        let channel = endpoint.connect().await.context("Failed to connect")?;
152        let mut stream = WatermarkServiceClient::new(channel)
153            .subscribe_watermarks(SubscribeWatermarksRequest {})
154            .await
155            .context("Failed to subscribe")?
156            .into_inner();
157        let first_response = stream
158            .message()
159            .await
160            .context("Failed to receive the first response")?
161            .context("Stream ended before the first response")?;
162        anyhow::Ok((stream, first_response))
163    })
164    .await
165    .context("Timed out waiting for the first response")??;
166
167    *failing = false;
168    let newly_settled = !std::mem::replace(settled, true);
169    sender.send_if_modified(|state| {
170        let merged = merge_forward_only(&mut state.watermarks, first_response);
171        if newly_settled {
172            state.unsettled_instances -= 1;
173        }
174        merged || newly_settled
175    });
176
177    while let Some(response) = stream.message().await.context("Stream failed")? {
178        sender.send_if_modified(|state| merge_forward_only(&mut state.watermarks, response));
179    }
180
181    Ok(())
182}

184/// Merge the watermarks in `response` into `watermarks`, keeping the highest checkpoint per
185/// pipeline. Watermarks with missing fields are skipped. Returns whether `watermarks` changed.
186fn merge_forward_only(
187    watermarks: &mut HashMap<String, CommitterWatermark>,
188    response: SubscribeWatermarksResponse,
189) -> bool {
190    let mut changed = false;
191    for (pipeline, watermark) in response.watermarks.into_iter().filter_map(decode_watermark) {
192        if watermarks
193            .get(&pipeline)
194            .is_none_or(|known| watermark.checkpoint_hi_inclusive > known.checkpoint_hi_inclusive)
195        {
196            watermarks.insert(pipeline, watermark);
197            changed = true;
198        }
199    }
200    changed
201}

219    fn watermark(checkpoint: u64) -> CommitterWatermark {
220        CommitterWatermark {
221            checkpoint_hi_inclusive: checkpoint,
222            timestamp_ms_hi_inclusive: checkpoint * 1_000,
223            ..Default::default()
224        }
225    }

227    /// Serve `progress` through a `WatermarkService` on a free local port. Returns the server's
228    /// URL and service.
229    async fn serve(progress: &DurableProgress) -> (String, Service) {
230        let rpc_listen_address = SocketAddr::from(([127, 0, 0, 1], get_available_port()));
231        let rpc = RpcService::new(rpc_listen_address);
232        let (watermarks, relay) =
233            WatermarkService::new(&HashMap::new(), progress.subscribe(), rpc.shutdown_token());
234        let service = rpc
235            .add_service(WatermarkServiceServer::new(watermarks))
236            .run(std::future::pending())
237            .await
238            .unwrap()
239            .attach(relay);
240        (format!("http://{rpc_listen_address}"), service)
241    }

243    /// Waits until the streamed watermarks satisfy `condition`, or panics after 5 seconds.
244    async fn wait_for(
245        streamed: &mut watch::Receiver<StreamedWatermarks>,
246        condition: impl FnMut(&StreamedWatermarks) -> bool,
247    ) {
248        tokio::time::timeout(Duration::from_secs(5), streamed.wait_for(condition))
249            .await
250            .expect("timed out waiting for streamed watermarks")
251            .expect("streamed watermark sender dropped");
252    }

254    fn checkpoint(streamed: &StreamedWatermarks, pipeline: &str) -> Option<u64> {
255        streamed
256            .watermarks
257            .get(pipeline)
258            .map(|watermark| watermark.checkpoint_hi_inclusive)
259    }

262    async fn keeps_the_highest_watermark_across_instances() {
263        let progress_a = DurableProgress::new();
264        progress_a.record("p", watermark(5));
265        progress_a.record("q", watermark(1));
266        let progress_b = DurableProgress::new();
267        progress_b.record("p", watermark(3));
268        progress_b.record("q", watermark(7));
269
270        let (url_a, server_a) = serve(&progress_a).await;
271        let (url_b, server_b) = serve(&progress_b).await;
272        let (mut streamed, _follower) = follow(&[url_a, url_b]).unwrap();
273
274        wait_for(&mut streamed, |state| {
275            state.unsettled_instances == 0
276                && checkpoint(state, "p") == Some(5)
277                && checkpoint(state, "q") == Some(7)
278        })
279        .await;
280
281        // Progress recorded after subscribing is streamed too.
282        progress_b.record("p", watermark(9));
283        wait_for(&mut streamed, |state| checkpoint(state, "p") == Some(9)).await;
284
285        server_a.shutdown().await.unwrap();
286        server_b.shutdown().await.unwrap();
287    }

334    /// Forward a local port to `upstream`, and return its URL. Cancelling the returned token
335    /// freezes the first connection, as if the machine at either end had failed: its sockets stay
336    /// open but nothing more is forwarded. Later connections are forwarded normally.
337    async fn freezing_proxy(upstream_url: &str) -> (String, CancellationToken) {
338        let upstream = upstream_url.trim_start_matches("http://").to_string();
339        let proxy = TcpListener::bind("127.0.0.1:0").await.unwrap();
340        let proxy_url = format!("http://{}", proxy.local_addr().unwrap());
341        let freeze_first = CancellationToken::new();
342        let freeze = freeze_first.clone();
343        tokio::spawn(async move {
344            let mut first = true;
345            loop {
346                let (mut client, _) = proxy.accept().await.unwrap();
347                let Ok(mut server) = TcpStream::connect(&upstream).await else {
348                    continue;
349                };
350                let freeze = if std::mem::replace(&mut first, false) {
351                    freeze.clone()
352                } else {
353                    CancellationToken::new()
354                };
355                tokio::spawn(async move {
356                    tokio::select! {
357                        _ = tokio::io::copy_bidirectional(&mut client, &mut server) => {}
358                        _ = freeze.cancelled() => std::future::pending::<()>().await,
359                    }
360                    drop((client, server));
361                });
362            }
363        });
364        (proxy_url, freeze_first)
365    }

367    #[tokio::test]
368    async fn follower_leaves_an_unresponsive_instance() {
369        let progress = DurableProgress::new();
370        progress.record("p", watermark(1));
371        let (upstream_url, _server) = serve(&progress).await;
372        let (proxy_url, freeze) = freezing_proxy(&upstream_url).await;
373
374        let (mut streamed, _follower) = follow(&[proxy_url]).unwrap();
375        wait_for(&mut streamed, |state| checkpoint(state, "p") == Some(1)).await;
376
377        freeze.cancel();
378        progress.record("p", watermark(2));
379
380        // Keepalive gives up on the frozen connection, and the follower subscribes again.
381        tokio::time::timeout(
382            KEEPALIVE_INTERVAL + KEEPALIVE_TIMEOUT + Duration::from_secs(20),
383            streamed.wait_for(|state| checkpoint(state, "p") == Some(2)),
384        )
385        .await
386        .expect("follower never left the unresponsive instance")
387        .unwrap();
388    }
