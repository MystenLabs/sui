4//! Roaring-bitmap inverted index keyed by packed event sequence number.
5//!
6//! The packed event-sequence encoding is owned by
7//! [`sui_inverted_index::event_seq`]. This table stores those packed values in
8//! bitmap buckets; its row-key layout stays byte-identical across encoding
9//! refactors.
10//!
11//! The packed namespace is sparse: most transactions use only a tiny prefix of
12//! their reserved event slots. Each row spans 268,435,456 packed event positions
13//! (`BUCKET_SIZE = 2^28`), which is exactly 4,096 transactions worth of event
14//! namespace at `EVENT_BITS = 16`, while leaving bucket-relative bit positions
15//! (max `2^28 - 1`) well within RoaringBitmap's `u32` limit.
16
17use super::BitmapSchema;
18
19/// Table of [`BitmapSchema::V1`] rows.
20pub const NAME: &str = "event_bitmap_index";
21/// Table of [`BitmapSchema::V2`] rows.
22pub const NAME_V2: &str = "event_bitmap_index_v2";
23
24pub const BUCKET_ID_WIDTH: usize = 12;
25/// Number of packed `event_seq`s per bitmap bucket. Changing it requires a new table and a
26/// backfill.
27pub const BUCKET_SIZE: u64 = 268_435_456;
28
29pub mod col {
30    pub const BITMAP: &str = "b";
31}
32
33/// The table that holds the rows of `schema`.
34pub const fn table_name(schema: BitmapSchema) -> &'static str {
35    match schema {
36        BitmapSchema::V1 => NAME,
37        BitmapSchema::V2 => NAME_V2,
38    }
39}
40
41/// Encode a full row key for the event-keyed bitmap index.
42///
43/// Format: `{prefix}{dimension_key}#{bucket_id:012}`, where the prefix is `v1#` for
44/// [`BitmapSchema::V1`] and empty for [`BitmapSchema::V2`].
45///
46/// Uses 12-digit zero-padded bucket ids because the event_seq namespace is
47/// ~2^16 larger than the tx_seq namespace, overflowing the 10-digit format
48/// used by the tx-keyed index.
49pub fn encode_row_key(schema: BitmapSchema, dimension_key: &[u8], bucket_id: u64) -> Vec<u8> {
50    let mut key = Vec::new();
51    super::encode_bitmap_row_key_into(&mut key, schema, BUCKET_ID_WIDTH, dimension_key, bucket_id);
52    key
53}

55#[cfg(test)]
56mod tests {
57    use super::*;
58
59    #[test]
60    fn test_encode_row_key_format() {
61        let dim_key = vec![0x01, 0xAA];
62        assert_eq!(
63            encode_row_key(BitmapSchema::V2, &dim_key, 42),
64            b"\x01\xAA#000000000042",
65        );
66    }
67
68    #[test]
69    fn test_bucket_size_fits_u32() {
70        assert!(BUCKET_SIZE <= u32::MAX as u64);
71    }
