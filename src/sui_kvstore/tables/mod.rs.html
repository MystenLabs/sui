43/// Row layout of a bitmap index table: its row-key format and how its cells are timestamped.
44#[derive(
45    Clone, Copy, Debug, PartialEq, Eq, serde::Deserialize, serde::Serialize, schemars::JsonSchema,
46)]
47#[serde(rename_all = "lowercase")]
48pub enum BitmapSchema {
49    /// The `transaction_bitmap_index` and `event_bitmap_index` tables. Row keys start with
50    /// `v1#`, and a cell's timestamp is the timestamp of the last checkpoint that added a bit to
51    /// the row. Checkpoints can share a timestamp, so when two writes of a row land out of order,
52    /// the row can keep the older snapshot.
53    V1,
54    /// The `transaction_bitmap_index_v2` and `event_bitmap_index_v2` tables. Row keys have no
55    /// version prefix, and a cell's timestamp is `2^42 + checkpoint` milliseconds, where
56    /// `checkpoint` is the sequence number of the last checkpoint that added a bit to the row.
57    V2,
58}
59
60impl BitmapSchema {
61    /// Cell timestamp, in milliseconds, of a bitmap whose last bit came from `checkpoint`, which
62    /// has timestamp `checkpoint_timestamp_ms`.
63    pub(crate) fn cell_timestamp_ms(self, checkpoint: u64, checkpoint_timestamp_ms: u64) -> u64 {
64        match self {
65            Self::V1 => checkpoint_timestamp_ms,
66            Self::V2 => checkpoint_cell_timestamp_ms(checkpoint),
67        }
68    }
69}
70
71/// Row-key prefix of [`BitmapSchema::V1`] tables.
72const V1_BITMAP_ROW_KEY_PREFIX: &[u8] = b"v1#";
73
74/// Shared row-key encoder for bitmap indexes.
75///
76/// Format: `{prefix}{dimension_key}#{bucket_id}`, where the prefix is `v1#` for
77/// [`BitmapSchema::V1`] and empty for [`BitmapSchema::V2`], and `bucket_id` is zero-padded to
78/// the table's configured width so bucket rows for a dimension sort lexicographically by bucket.
79pub(crate) fn encode_bitmap_row_key_into(
80    out: &mut Vec<u8>,
81    schema: BitmapSchema,
82    bucket_id_width: usize,
83    dimension_key: &[u8],
84    bucket_id: u64,
85) {
86    encode_bitmap_row_key_with_dimension(
87        out,
88        schema,
89        bucket_id_width,
90        dimension_key.len(),
91        bucket_id,
92        |out| {
93            out.extend_from_slice(dimension_key);
94        },
95    );
96}
97
98pub(crate) fn encode_bitmap_row_key_parts_into(
99    out: &mut Vec<u8>,
100    schema: BitmapSchema,
101    bucket_id_width: usize,
102    dimension_tag: u8,
103    dimension_value: &[u8],
104    bucket_id: u64,
105) {
106    encode_bitmap_row_key_with_dimension(
107        out,
108        schema,
109        bucket_id_width,
110        1 + dimension_value.len(),
111        bucket_id,
112        |out| {
113            out.push(dimension_tag);
114            out.extend_from_slice(dimension_value);
115        },
116    );
117}
118
119fn encode_bitmap_row_key_with_dimension(
120    out: &mut Vec<u8>,
121    schema: BitmapSchema,
122    bucket_id_width: usize,
123    dimension_len: usize,
124    bucket_id: u64,
125    write_dimension: impl FnOnce(&mut Vec<u8>),
126) {
127    out.clear();
128    // u64 buckets are at most 20 decimal digits.
129    out.reserve(V1_BITMAP_ROW_KEY_PREFIX.len() + dimension_len + 1 + bucket_id_width.max(20));
130    if schema == BitmapSchema::V1 {
131        out.extend_from_slice(V1_BITMAP_ROW_KEY_PREFIX);
132    }
133    write_dimension(out);
134    write!(out, "#{:0width$}", bucket_id, width = bucket_id_width)
135        .expect("writing to Vec cannot fail");
136}

138/// Start of the cell timestamps written by [`checkpoint_cell_timestamp_ms`], in milliseconds.
139/// 2^42 ms is past every checkpoint timestamp and the server's time until the year 2109, so
140/// these cells sort above any cell timestamped with a checkpoint's time or the server's time.
141const CHECKPOINT_CELL_TIMESTAMP_BASE_MS: u64 = 1 << 42;
142
143/// Cell timestamp, in milliseconds, for a value that reflects every checkpoint through
144/// `checkpoint`.
145///
146/// Readers take a column's newest cell, and several writers index the same checkpoints. A cell
147/// rewritten with a value that grows as checkpoints are indexed, like a bitmap bucket, must
148/// therefore order its versions by the checkpoint each reflects. Checkpoint timestamps cannot:
149/// checkpoints split from one consensus commit share a timestamp, and of two writes at one
150/// timestamp the later one wins, whichever checkpoint it reflects.
151pub(crate) fn checkpoint_cell_timestamp_ms(checkpoint: u64) -> u64 {
152    CHECKPOINT_CELL_TIMESTAMP_BASE_MS + checkpoint
153}

155/// Build an Entry from cells. Accepts any iterator to avoid intermediate allocations.
156pub fn make_entry(
157    row_key: impl Into<Bytes>,
158    cells: impl IntoIterator<Item = (&'static str, Bytes)>,
159    timestamp_ms: Option<u64>,
160) -> Entry {
161    let timestamp_micros = timestamp_ms
162        .map(|ms| ms.checked_mul(1000).expect("timestamp overflow") as i64)
163        // default to -1 for current Bigtable server time
164        .unwrap_or(-1);
165
166    Entry {
167        row_key: row_key.into(),
168        mutations: cells
169            .into_iter()
170            .map(|(col, val)| Mutation {
171                mutation: Some(mutation::Mutation::SetCell(SetCell {
172                    family_name: FAMILY.to_string(),
173                    column_qualifier: Bytes::from(col),
174                    timestamp_micros,
175                    value: val,
176                })),
177            })
178            .collect(),
179        idempotency: None,
180    }
