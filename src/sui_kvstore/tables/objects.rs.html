4//! Objects table: every version of every object, plus tombstones at the versions where objects
5//! were deleted or wrapped.
6//!
7//! Row key: `object_id (32B) || version (8B big-endian)`
8//!
9//! A row holds exactly one of:
10//! - `""`: the BCS-encoded [`Object`] at that version (a live row), or
11//! - a tombstone, marking the version at which the object was removed from the live set:
12//!   - `s`: a one-byte [`ObjectStatus`], `Deleted` or `Wrapped`, and
13//!   - `t`: the raw 32-byte digest of the transaction that removed the object. Tombstones written
14//!     before this column existed lack it.
15//!
16//! The newest row at or below version `V` is the object's state at `V`: a live row is the
17//! object's contents, a tombstone means the object did not exist at `V`. Tombstones are written
18//! by a separate pipeline from live rows, so they are only present where that pipeline has run.
19
20use anyhow::{Context, Result, bail, ensure};
21use bytes::Bytes;
22use sui_types::digests::TransactionDigest;
23use sui_types::object::Object;
24use sui_types::storage::ObjectKey;
25
26use crate::object_status::ObjectStatus;
27
28pub const NAME: &str = "objects";
29
30pub mod col {
31    /// BCS of a live object version.
32    pub const OBJECT: &str = crate::tables::DEFAULT_COLUMN;
33    /// One-byte [`ObjectStatus`](crate::ObjectStatus) of a tombstone.
34    pub const STATUS: &str = "s";
35    /// Raw 32-byte digest of the transaction that removed the object, in a tombstone.
36    pub const TRANSACTION: &str = "t";
37}
38
39/// A decoded row of the objects table.
40#[derive(Clone, Debug, PartialEq, Eq)]
41pub enum ObjectRow {
42    /// The object's contents at the row's version.
43    Live(Object),
44    /// The object was removed at the row's version: either `Deleted` or `Wrapped`.
45    Tombstone(ObjectStatus),
46}
47
48impl ObjectRow {
49    /// The object, or `None` for a tombstone.
50    pub fn into_live(self) -> Option<Object> {
51        match self {
52            ObjectRow::Live(object) => Some(object),
53            ObjectRow::Tombstone(_) => None,
54        }
55    }
56}
57
58pub fn encode_key(object_key: &ObjectKey) -> Vec<u8> {
59    let mut raw_key = object_key.0.to_vec();
60    raw_key.extend(object_key.1.value().to_be_bytes());
61    raw_key
62}
63
64pub fn encode_key_upper_bound(object_key: &ObjectKey) -> Vec<u8> {
65    encode_key(object_key)
66}
67
68pub fn encode(object: &Object) -> Result<[(&'static str, Bytes); 1]> {
69    Ok([(col::OBJECT, Bytes::from(bcs::to_bytes(object)?))])
70}
71
72/// Cells of a tombstone row. `status` must be `Deleted` or `Wrapped`; `transaction` is the
73/// transaction that removed the object.
74pub fn encode_tombstone(
75    status: ObjectStatus,
76    transaction: &TransactionDigest,
77) -> Result<[(&'static str, Bytes); 2]> {
78    ensure!(
79        status != ObjectStatus::Live,
80        "a tombstone's status must be Deleted or Wrapped"
81    );
82    Ok([
83        (col::STATUS, Bytes::copy_from_slice(&[status.to_byte()])),
84        (
85            col::TRANSACTION,
86            Bytes::copy_from_slice(transaction.inner()),
87        ),
88    ])
89}
90
91/// Parse a row as a live object or a tombstone, selecting the cell by column.
92pub fn decode(row: &[(Bytes, Bytes)]) -> Result<ObjectRow> {
93    let mut object = None;
94    let mut status = None;
95    for (column, value) in row {
96        if column == col::OBJECT.as_bytes() {
97            object = Some(value);
98        } else if column == col::STATUS.as_bytes() {
99            status = Some(value);
100        }
101    }
102    match (object, status) {
103        (Some(value), None) => Ok(ObjectRow::Live(
104            bcs::from_bytes(value).context("invalid object BCS")?,
105        )),
106        (None, Some(value)) => {
107            let [byte] = value.as_ref() else {
108                bail!("expected 1-byte status, got {} bytes", value.len());
109            };
110            match ObjectStatus::from_byte(*byte)? {
111                ObjectStatus::Live => bail!("tombstone has status Live"),
112                status => Ok(ObjectRow::Tombstone(status)),
113            }
114        }
115        (Some(_), Some(_)) => bail!("row has both an object and a tombstone status"),
116        (None, None) => bail!("row has neither an object nor a tombstone status"),
117    }
118}

120#[cfg(test)]
121mod tests {
122    use sui_types::base_types::ObjectID;
123    use sui_types::base_types::SuiAddress;
124
125    use super::*;
126
127    fn cells<const N: usize>(encoded: [(&'static str, Bytes); N]) -> Vec<(Bytes, Bytes)> {
128        encoded
129            .into_iter()
130            .map(|(column, value)| (Bytes::from_static(column.as_bytes()), value))
131            .collect()
132    }
133
134    #[test]
135    fn decode_round_trips_live_rows_and_tombstones() {
136        let object = Object::with_id_owner_for_testing(ObjectID::random(), SuiAddress::ZERO);
137        assert_eq!(
138            decode(&cells(encode(&object).unwrap())).unwrap(),
139            ObjectRow::Live(object),
140        );
141        let transaction = TransactionDigest::random();
142        for status in [ObjectStatus::Deleted, ObjectStatus::Wrapped] {
143            assert_eq!(
144                decode(&cells(encode_tombstone(status, &transaction).unwrap())).unwrap(),
145                ObjectRow::Tombstone(status),
146            );
147        }
148    }
149
150    /// A tombstone stores the removing transaction as its raw digest in column `t`. Rows with
151    /// and without that cell decode to the same tombstone.
152    #[test]
153    fn tombstone_transaction_cell_does_not_change_decoding() {
154        let transaction = TransactionDigest::random();
155        for (byte, status) in [(1u8, ObjectStatus::Deleted), (2, ObjectStatus::Wrapped)] {
156            let with_transaction = cells(encode_tombstone(status, &transaction).unwrap());
157            assert_eq!(
158                with_transaction,
159                vec![
160                    (Bytes::from_static(b"s"), Bytes::copy_from_slice(&[byte])),
161                    (
162                        Bytes::from_static(b"t"),
163                        Bytes::copy_from_slice(transaction.inner())
164                    ),
165                ],
166            );
167            let status_only = vec![with_transaction[0].clone()];
168            for row in [with_transaction, status_only] {
169                assert_eq!(decode(&row).unwrap(), ObjectRow::Tombstone(status));
170            }
171        }
172    }
173
174    /// The cell is chosen by column, not position, and unknown columns are ignored.
175    #[test]
176    fn decode_selects_cell_by_column() {
177        let mut row = vec![(Bytes::from_static(b"other"), Bytes::from_static(b"junk"))];
178        row.extend(cells(
179            encode_tombstone(ObjectStatus::Wrapped, &TransactionDigest::random()).unwrap(),
180        ));
181        assert_eq!(
182            decode(&row).unwrap(),
183            ObjectRow::Tombstone(ObjectStatus::Wrapped)
184        );
185    }
186
187    #[test]
188    fn decode_rejects_malformed_rows() {
189        let object = Object::with_id_owner_for_testing(ObjectID::random(), SuiAddress::ZERO);
190        let transaction = TransactionDigest::random();
191        let mut both = cells(encode(&object).unwrap());
192        both.extend(cells(
193            encode_tombstone(ObjectStatus::Deleted, &transaction).unwrap(),
194        ));
195        let status = |bytes: &'static [u8]| vec![(Bytes::from_static(b"s"), Bytes::from(bytes))];
196
197        for (name, row) in [
198            ("empty", vec![]),
199            ("both", both),
200            ("live status", status(&[0])),
201            ("unknown status", status(&[3])),
202            ("long status", status(&[1, 1])),
203        ] {
204            assert!(decode(&row).is_err(), "{name} row decoded");
205        }
206        assert!(encode_tombstone(ObjectStatus::Live, &transaction).is_err());
207    }
