4//! Objects table: every version of every object, plus tombstones at the versions where objects
5//! were deleted or wrapped.
6//!
7//! Row key: `object_id (32B) || version (8B big-endian)`
8//!
9//! A row holds exactly one of:
10//! - `""`: the BCS-encoded [`Object`] at that version (a live row), or
11//! - a tombstone, marking the version at which the object was removed from the live set:
12//!   - `s`: a one-byte [`ObjectStatus`], `Deleted` or `Wrapped`, and
13//!   - `t`: the raw 32-byte digest of the transaction that removed the object. Tombstones written
14//!     before this column existed lack it.
15//!
16//! The newest row at or below version `V` is the object's state at `V`: a live row is the
17//! object's contents, a tombstone means the object did not exist at `V`. Tombstones are written
18//! by a separate pipeline from live rows, so they are only present where that pipeline has run.
19

31pub mod col {
32    /// BCS of a live object version.
33    pub const OBJECT: &str = crate::tables::DEFAULT_COLUMN;
34    /// One-byte [`ObjectStatus`](crate::ObjectStatus) of a tombstone.
35    pub const STATUS: &str = "s";
36    /// Raw 32-byte digest of the transaction that removed the object, in a tombstone.
37    pub const TRANSACTION: &str = "t";
38}
39
40/// A decoded row of the objects table.
41#[derive(Clone, Debug, PartialEq, Eq)]
42pub enum ObjectRow {
43    /// The object's contents at the row's version.
44    Live(Object),
45    /// The object was removed at the row's version.
46    Tombstone(ObjectTombstone),
47}
48
49/// A row of the objects table, read with or without live objects' contents.
50#[derive(Clone, Debug, PartialEq, Eq)]
51pub enum ObjectVersionRow {
52    /// A live version, with the object's contents if the read included them.
53    Live(Option<Object>),
54    /// The object was removed at the row's version.
55    Tombstone(ObjectTombstone),
56}
57
58/// How an object was removed at a tombstone row's version.
59#[derive(Clone, Copy, Debug, PartialEq, Eq)]
60pub struct ObjectTombstone {
61    /// `Deleted` or `Wrapped`.
62    pub status: ObjectStatus,
63    /// The transaction that removed the object. `None` for tombstones written without it.
64    pub transaction: Option<TransactionDigest>,
65}
66
67impl ObjectRow {
68    /// The object, or `None` for a tombstone.
69    pub fn into_live(self) -> Option<Object> {
70        match self {
71            ObjectRow::Live(object) => Some(object),
72            ObjectRow::Tombstone(_) => None,
73        }
74    }
75}

77pub fn encode_key(object_key: &ObjectKey) -> Vec<u8> {
78    let mut raw_key = object_key.0.to_vec();
79    raw_key.extend(object_key.1.value().to_be_bytes());
80    raw_key
81}
82
83/// The `ObjectKey` of a row key built by [`encode_key`]: the object id followed by the version,
84/// big-endian.
85pub fn decode_key(row_key: &[u8]) -> Result<ObjectKey> {
86    let (id, version) = row_key
87        .split_at_checked(ObjectID::LENGTH)
88        .context("objects row key is shorter than an object id")?;
89    let id = ObjectID::from_bytes(id).context("objects row key has an invalid object id")?;
90    let version: [u8; 8] = version
91        .try_into()
92        .context("objects row key version is not 8 bytes")?;
93    Ok(ObjectKey(
94        id,
95        SequenceNumber::from_u64(u64::from_be_bytes(version)),
96    ))
97}
98
99pub fn encode_key_upper_bound(object_key: &ObjectKey) -> Vec<u8> {
100    encode_key(object_key)
101}

103pub fn encode(object: &Object) -> Result<[(&'static str, Bytes); 1]> {
104    Ok([(col::OBJECT, Bytes::from(bcs::to_bytes(object)?))])
105}
106
107/// Cells of a tombstone row. `status` must be `Deleted` or `Wrapped`; `transaction` is the
108/// transaction that removed the object.
109pub fn encode_tombstone(
110    status: ObjectStatus,
111    transaction: &TransactionDigest,
112) -> Result<[(&'static str, Bytes); 2]> {
113    ensure!(
114        status != ObjectStatus::Live,
115        "a tombstone's status must be Deleted or Wrapped"
116    );
117    Ok([
118        (col::STATUS, Bytes::copy_from_slice(&[status.to_byte()])),
119        (
120            col::TRANSACTION,
121            Bytes::copy_from_slice(transaction.inner()),
122        ),
123    ])
124}
125
126/// Parse a row as a live object or a tombstone, selecting the cells by column.
127pub fn decode(row: &[(Bytes, Bytes)]) -> Result<ObjectRow> {
128    if let Some(tombstone) = decode_tombstone(row)? {
129        return Ok(ObjectRow::Tombstone(tombstone));
130    }
131    let object = cell(row, col::OBJECT).context("live row has no object")?;
132    Ok(ObjectRow::Live(
133        bcs::from_bytes(object).context("invalid object BCS")?,
134    ))
135}

137/// Parse a row read with live objects' contents if `with_contents`, or with them stripped
138/// otherwise, as a live version, with its object only if read with contents, or a tombstone.
139pub fn decode_version_row(row: &[(Bytes, Bytes)], with_contents: bool) -> Result<ObjectVersionRow> {
140    Ok(if with_contents {
141        match decode(row)? {
142            ObjectRow::Live(object) => ObjectVersionRow::Live(Some(object)),
143            ObjectRow::Tombstone(tombstone) => ObjectVersionRow::Tombstone(tombstone),
144        }
145    } else {
146        decode_tombstone(row)?.map_or(ObjectVersionRow::Live(None), ObjectVersionRow::Tombstone)
147    })
148}
149
150/// Parse a row as a tombstone, or `None` for a live row. Live rows are told apart by column
151/// alone, so this also works on rows read with live objects' contents stripped.
152pub fn decode_tombstone(row: &[(Bytes, Bytes)]) -> Result<Option<ObjectTombstone>> {
153    if is_live(row)? {
154        return Ok(None);
155    }
156    let status = cell(row, col::STATUS).context("tombstone has no status")?;
157    let [byte] = status.as_ref() else {
158        bail!("expected 1-byte status, got {} bytes", status.len());
159    };
160    let status = match ObjectStatus::from_byte(*byte)? {
161        ObjectStatus::Live => bail!("tombstone has status Live"),
162        status => status,
163    };
164    let transaction = cell(row, col::TRANSACTION)
165        .map(|digest| {
166            <[u8; 32]>::try_from(digest.as_ref())
167                .map(TransactionDigest::new)
168                .with_context(|| {
169                    format!(
170                        "expected 32-byte transaction digest, got {} bytes",
171                        digest.len()
172                    )
173                })
174        })
175        .transpose()?;
176    Ok(Some(ObjectTombstone {
177        status,
178        transaction,
179    }))
180}

182/// The value of `row`'s cell in `column`, if it has one.
183fn cell<'row>(row: &'row [(Bytes, Bytes)], column: &str) -> Option<&'row Bytes> {
184    row.iter()
185        .find(|(name, _)| name == column.as_bytes())
186        .map(|(_, value)| value)
187}
188
189/// Whether a row is live rather than a tombstone, told apart by column alone, so it also works on
190/// rows read with their values stripped.
191pub fn is_live(row: &[(Bytes, Bytes)]) -> Result<bool> {
192    let has_column = |column: &str| row.iter().any(|(name, _)| name == column.as_bytes());
193    match (has_column(col::OBJECT), has_column(col::STATUS)) {
194        (true, false) => Ok(true),
195        (false, true) => Ok(false),
196        (true, true) => bail!("row has both an object and a tombstone status"),
197        (false, false) => bail!("row has neither an object nor a tombstone status"),
198    }
199}

201#[cfg(test)]
202mod tests {
203    use sui_types::base_types::ObjectID;
204    use sui_types::base_types::SuiAddress;

208    fn cells<const N: usize>(encoded: [(&'static str, Bytes); N]) -> Vec<(Bytes, Bytes)> {
209        encoded
210            .into_iter()
211            .map(|(column, value)| (Bytes::from_static(column.as_bytes()), value))
212            .collect()
213    }
214
215    #[test]
216    fn decode_round_trips_live_rows_and_tombstones() {
217        let object = Object::with_id_owner_for_testing(ObjectID::random(), SuiAddress::ZERO);
218        assert_eq!(
219            decode(&cells(encode(&object).unwrap())).unwrap(),
220            ObjectRow::Live(object),
221        );
222        let transaction = TransactionDigest::random();
223        for status in [ObjectStatus::Deleted, ObjectStatus::Wrapped] {
224            assert_eq!(
225                decode(&cells(encode_tombstone(status, &transaction).unwrap())).unwrap(),
226                ObjectRow::Tombstone(ObjectTombstone {
227                    status,
228                    transaction: Some(transaction)
229                }),
230            );
231        }
232    }
233
234    /// A tombstone's transaction is its raw digest in column `t`, decoded when present, also
235    /// from rows read with live objects' contents stripped, which decode as live.
236    #[test]
237    fn decode_reads_tombstone_transactions() {
238        let transaction = TransactionDigest::random();
239        let mut row = cells(encode_tombstone(ObjectStatus::Deleted, &transaction).unwrap());
240        let tombstone = ObjectTombstone {
241            status: ObjectStatus::Deleted,
242            transaction: Some(transaction),
243        };
244        assert_eq!(decode(&row).unwrap(), ObjectRow::Tombstone(tombstone));
245        assert_eq!(decode_tombstone(&row).unwrap(), Some(tombstone));
246
247        let stripped_live = vec![(Bytes::from_static(col::OBJECT.as_bytes()), Bytes::new())];
248        assert_eq!(decode_tombstone(&stripped_live).unwrap(), None);
249
250        let short = &transaction.inner()[..31];
251        row.last_mut().unwrap().1 = Bytes::copy_from_slice(short);
252        assert!(decode(&row).is_err(), "31-byte transaction decoded");
253    }
254
255    /// A tombstone stores the removing transaction as its raw digest in column `t`. A row without
256    /// that cell, written before the column existed, decodes with no transaction.
257    #[test]
258    fn tombstone_transaction_cell() {
259        let transaction = TransactionDigest::random();
260        for (byte, status) in [(1u8, ObjectStatus::Deleted), (2, ObjectStatus::Wrapped)] {
261            let with_transaction = cells(encode_tombstone(status, &transaction).unwrap());
262            assert_eq!(
263                with_transaction,
264                vec![
265                    (Bytes::from_static(b"s"), Bytes::copy_from_slice(&[byte])),
266                    (
267                        Bytes::from_static(b"t"),
268                        Bytes::copy_from_slice(transaction.inner())
269                    ),
270                ],
271            );
272            let status_only = vec![with_transaction[0].clone()];
273            for (row, transaction) in [(with_transaction, Some(transaction)), (status_only, None)] {
274                assert_eq!(
275                    decode(&row).unwrap(),
276                    ObjectRow::Tombstone(ObjectTombstone {
277                        status,
278                        transaction
279                    })
280                );
281            }
282        }
283    }

285    /// The cell is chosen by column, not position, and unknown columns are ignored.
286    #[test]

302    #[test]
303    fn decode_rejects_malformed_rows() {
304        let object = Object::with_id_owner_for_testing(ObjectID::random(), SuiAddress::ZERO);
305        let transaction = TransactionDigest::random();
306        let mut both = cells(encode(&object).unwrap());
307        both.extend(cells(
308            encode_tombstone(ObjectStatus::Deleted, &transaction).unwrap(),
309        ));
310        let status = |bytes: &'static [u8]| vec![(Bytes::from_static(b"s"), Bytes::from(bytes))];
311
312        for (name, row) in [
313            ("empty", vec![]),
314            ("both", both),
315            ("live status", status(&[0])),
316            ("unknown status", status(&[3])),
317            ("long status", status(&[1, 1])),
318        ] {
319            assert!(decode(&row).is_err(), "{name} row decoded");
320        }
321        assert!(encode_tombstone(ObjectStatus::Live, &transaction).is_err());
322    }
