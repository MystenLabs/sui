21#[derive(Clone)]
22pub struct ClockBorrowCostParams {
23    pub clock_borrow_cost_base: Option<InternalGas>,
24}
25/***************************************************************************************************
26 * native fun native_borrow
27 * Implementation of the Move native function `fun native_borrow(): &Clock`
28 *   gas cost: clock_borrow_cost_base | fixed cost; the Clock is a fixed-size object
29 **************************************************************************************************/
30pub fn borrow(
31    context: &mut NativeContext,
32    ty_args: Vec<Type>,
33    args: VecDeque<Value>,
34) -> PartialVMResult<NativeResult> {
35    safe_assert!(ty_args.is_empty());
36    safe_assert!(args.is_empty());
37
38    let clock_borrow_cost_base = safe_unwrap!(
39        get_extension!(context, NativesCostTable)?
40            .clock_borrow_cost_params
41            .clock_borrow_cost_base
42    );
43    native_charge_gas_early_exit!(context, clock_borrow_cost_base);
44
45    let layout = context
46        .type_tag_to_type_layout(&TypeTag::Struct(Box::new(Clock::type_())))
47        .ok_or_else(|| {
48            PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR)
49                .with_message("Failed to load the Clock type layout".to_string())
50        })?;
51    let object_runtime: &mut ObjectRuntime = get_extension_mut!(context)?;
52    let clock_ref = object_runtime.borrow_clock(&layout)?;
53
54    Ok(NativeResult::ok(context.gas_used(), smallvec![clock_ref]))
55}
56
57/***************************************************************************************************
58 * native fun native_set_timestamp_ms_for_testing
59 * Implementation of the Move native function
60 * `fun native_set_timestamp_ms_for_testing(timestamp_ms: u64)`
61 **************************************************************************************************/
62pub fn set_timestamp_ms_for_testing(
63    context: &mut NativeContext,
64    ty_args: Vec<Type>,
65    mut args: VecDeque<Value>,
66) -> PartialVMResult<NativeResult> {
67    safe_assert!(ty_args.is_empty());
68    safe_assert!(args.len() == 1);
69
70    let timestamp_ms = pop_arg!(args, u64);
71    let store: &&InMemoryTestStore = get_extension!(context)?;
72    store.set_clock_timestamp_ms(timestamp_ms);
73    let object_runtime: &mut ObjectRuntime = get_extension_mut!(context)?;
74    object_runtime.clear_clock_for_testing();
75
76    Ok(NativeResult::ok(legacy_test_cost(), smallvec![]))
77}
