4//! Peer-to-peer data synchronization of checkpoints.
5//!
6//! This StateSync module is responsible for the synchronization and dissemination of checkpoints
7//! and the transactions, and their effects, contained within. This module is *not* responsible for
8//! the execution of the transactions included in a checkpoint, that process is left to another
9//! component in the system.
10//!
11//! # High-level Overview of StateSync
12//!
13//! StateSync discovers new checkpoints via a few different sources:
14//! 1. If this node is a Validator, checkpoints will be produced via consensus at which point
15//!    consensus can notify state-sync of the new checkpoint via [Handle::send_checkpoint].
16//! 2. A peer notifies us of the latest checkpoint which they have synchronized. State-Sync will
17//!    also periodically query its peers to discover what their latest checkpoint is.
18//!
19//! We keep track of two different watermarks:
20//! * highest_verified_checkpoint - This is the highest checkpoint header that we've locally
21//!   verified. This indicated that we have in our persistent store (and have verified) all
22//!   checkpoint headers up to and including this value.
23//! * highest_synced_checkpoint - This is the highest checkpoint that we've fully synchronized,
24//!   meaning we've downloaded and have in our persistent stores all of the transactions, and their
25//!   effects (but not the objects), for all checkpoints up to and including this point. This is
26//!   the watermark that is shared with other peers, either via notification or when they query for
27//!   our latest checkpoint, and is intended to be used as a guarantee of data availability.
28//!
29//! The `PeerHeights` struct is used to track the highest_synced_checkpoint watermark for all of
30//! our peers.
31//!
32//! When a new checkpoint is discovered, and we've determined that it is higher than our
33//! highest_verified_checkpoint, then StateSync will kick off a task to synchronize and verify all
34//! checkpoints between our highest_synced_checkpoint and the newly discovered checkpoint. This
35//! process is done by querying one of our peers for the checkpoints we're missing (using the
36//! `PeerHeights` struct as a way to intelligently select which peers have the data available for
37//! us to query) at which point we will locally verify the signatures on the checkpoint header with
38//! the appropriate committee (based on the epoch). As checkpoints are verified, the
39//! highest_synced_checkpoint watermark will be ratcheted up.
40//!
41//! Once we've ratcheted up our highest_verified_checkpoint, and if it is higher than
42//! highest_synced_checkpoint, StateSync will then kick off a task to synchronize the contents of
43//! all of the checkpoints from highest_synced_checkpoint..=highest_verified_checkpoint. After the
44//! contents of each checkpoint is fully downloaded, StateSync will update our
45//! highest_synced_checkpoint watermark and send out a notification on a broadcast channel
46//! indicating that a new checkpoint has been fully downloaded. Notifications on this broadcast
47//! channel will always be made in order. StateSync will also send out a notification to its peers
48//! of the newly synchronized checkpoint so that it can help other peers synchronize.

147pub(super) fn compute_adaptive_timeout(
148    tx_count: u64,
149    min_timeout: Duration,
150    max_timeout: Duration,
151) -> Duration {
152    const MAX_TRANSACTIONS_PER_CHECKPOINT: u64 = 10_000;
153    const JITTER_FRACTION: f64 = 0.1;
154
155    let max_timeout = max_timeout.max(min_timeout);
156    let ratio = (tx_count as f64 / MAX_TRANSACTIONS_PER_CHECKPOINT as f64).min(1.0);
157    let extra = Duration::from_secs_f64((max_timeout - min_timeout).as_secs_f64() * ratio);
158    let base = min_timeout + extra;
159
160    let jitter_range = base.as_secs_f64() * JITTER_FRACTION;
161    let jitter = rand::thread_rng().gen_range(-jitter_range..jitter_range);
162    Duration::from_secs_f64((base.as_secs_f64() + jitter).max(min_timeout.as_secs_f64()))
163}

175/// Near the release frontier, how a contents request that is not a hedge relates to its
176/// peer's slow flag when it is sent (see [`SLOW_PEER_PROBE_INTERVAL`]).
177#[derive(Clone, Copy, Debug, PartialEq, Eq)]
178pub(super) enum SlowPeerRequest {
179    /// The peer was not slow from this node.
180    PeerNotSlow,
181    /// The peer was slow, and the request is a slow-peer probe.
182    Probe,
183    /// The peer was slow, and the request is not a probe.
184    NotProbe,
185}

187impl SlowPeerRequest {
188    /// Whether the request's outcome is recorded as a response-time sample (see
189    /// `PeerScore::record_content_response_time`). While a peer is slow, only probes are.
190    /// Other requests to it are hedged right away once a peer that isn't slow has the
191    /// checkpoint, so their slow outcomes are cut short before the peer's slow-response
192    /// threshold while their fast answers still arrive. Recording them would make the peer look
193    /// less slow than it is.
194    pub(super) fn records_response_time(self) -> bool {
195        self != Self::NotProbe
196    }
197}

200pub(super) struct PeerScore {
201    successes: VecDeque<(Instant, u64 /* response_size_bytes */, Duration)>,
202    /// Requests cancelled because another peer delivered the same contents first, with the time
203    /// they had run. That time is a lower bound on the response time, so a sample is kept only
204    /// if it lowers the throughput estimate (see `record_cancelled`). These samples count as
205    /// neither successes nor failures.
206    cancelled: VecDeque<(Instant, u64 /* response_size_bytes */, Duration)>,
207    /// How long recent checkpoint contents requests that were not hedges took, oldest first.
208    /// Used to decide whether requests to this peer should be hedged sooner (see
209    /// `slow_peer_hedge_delay`).
210    content_response_times: VecDeque<(Instant, ContentResponseTime)>,
211    /// Contents requests that were not hedges sent to this peer while it was slow since the last
212    /// slow-peer probe (see [`SLOW_PEER_PROBE_INTERVAL`]).
213    requests_since_probe: u32,
214    failures: VecDeque<Instant>,
215    window: Duration,
216    failure_rate: f64,
217    failing_since: Option<Instant>,
218}

220impl PeerScore {
221    /// How many of a peer's most recent samples of each kind (successes, failures, cancellations,
222    /// contents response times) are kept.
223    const MAX_SAMPLES: usize = 20;
224    const MIN_SAMPLES_FOR_FAILURE: usize = 10;
225    /// A contents response is slow when it takes more than 25% (this factor) longer than the peer's
226    /// fastest recent response, and at least [`Self::SLOW_RESPONSE_MIN_MARGIN`] longer.
227    const SLOW_RESPONSE_FACTOR: f64 = 1.25;
228    /// For a peer whose fastest response is under 80 ms, 25% of it is less than this margin. The
229    /// margin keeps such a peer's jitter, for example 12 ms answers that sometimes take 23 ms, from
230    /// counting as slow.
231    const SLOW_RESPONSE_MIN_MARGIN: Duration = Duration::from_millis(20);
232    /// A peer is slow when at least this share of its recent contents responses were slow.
233    const SLOW_PEER_SHARE: f64 = 0.25;
234    /// A peer can only be slow once it has at least this many recent contents responses whose
235    /// slowness is known: answers, and cancellations after its slow-response threshold.
236    const MIN_SAMPLES_FOR_SLOW: usize = 10;
237
238    pub(super) fn new(window: Duration, failure_rate: f64) -> Self {
239        Self {
240            successes: VecDeque::new(),
241            cancelled: VecDeque::new(),
242            content_response_times: VecDeque::new(),
243            requests_since_probe: 0,
244            failures: VecDeque::new(),
245            window,
246            failure_rate,
247            failing_since: None,
248        }
249    }
250
251    pub(super) fn record_success(&mut self, size: u64, response_time: Duration) {
252        let now = Instant::now();
253        self.successes.push_back((now, size, response_time));
254        while self.successes.len() > Self::MAX_SAMPLES {
255            self.successes.pop_front();
256        }
257        // Note: a single success does NOT clear `failing_since`. Whether the peer is
258        // "consistently failing" is driven by the windowed failure RATE (see
259        // `update_failing_state`), so an occasional success while the rate stays high (e.g.
260        // a brief reconnect during connection churn) must not keep resetting the clock and
261        // prevent the peer from ever being reported.
262    }
263
264    /// Records a request that was cancelled after `elapsed` because another peer delivered `size`
265    /// bytes first. `size / elapsed` is only an upper bound on this peer's throughput, so the
266    /// sample is kept only when it is below the current estimate, which it then lowers. Without
267    /// an estimate the sample carries no usable information and is dropped.
268    pub(super) fn record_cancelled(&mut self, size: u64, elapsed: Duration) {
269        let Some(current) = self.effective_throughput() else {
270            return;
271        };
272        if elapsed.is_zero() || size as f64 / elapsed.as_secs_f64() >= current {
273            return;
274        }
275        self.cancelled.push_back((Instant::now(), size, elapsed));
276        while self.cancelled.len() > Self::MAX_SAMPLES {
277            self.cancelled.pop_front();
278        }
279    }
280
281    /// Records how long a checkpoint contents request that was not a hedge took. Hedge requests
282    /// are left out: a hedge is cancelled whenever the request it hedged answers first, however
283    /// short a time it has run, so how often hedges are cut short says nothing about the peer.
284    /// Near the release frontier, requests sent while the peer was slow that were not slow-peer
285    /// probes are left out too (see [`SlowPeerRequest::records_response_time`]).
286    /// A request that was not a hedge and is cancelled after its peer's slow-response threshold
287    /// was slow. One cancelled sooner, for example by a hedge to a closer peer, might still have
288    /// answered in time, so `slow_peer_hedge_delay` counts it neither way. Such a sample is
289    /// dropped when the peer already has a slow-response threshold, so that it doesn't take the
290    /// place of a sample that counts; without a threshold yet, it is kept, because the threshold
291    /// it is compared against is only known once the peer answers.
292    pub(super) fn record_content_response_time(&mut self, response_time: ContentResponseTime) {
293        if let ContentResponseTime::LongerThan(elapsed) = response_time
294            && self
295                .slow_response_threshold()
296                .is_some_and(|threshold| elapsed < threshold)
297        {
298            return;
299        }
300        self.content_response_times
301            .push_back((Instant::now(), response_time));
302        while self.content_response_times.len() > Self::MAX_SAMPLES {
303            self.content_response_times.pop_front();
304        }
305    }
306
307    /// The fastest answer among this peer's recent contents requests that were not hedges.
308    pub(super) fn fastest_recent_content_answer(&self) -> Option<Duration> {
309        self.recent_content_response_times()
310            .filter_map(|response_time| match response_time {
311                ContentResponseTime::Answered(elapsed) => Some(elapsed),
312                ContentResponseTime::LongerThan(_) => None,
313            })
314            .min()
315    }
316
317    fn recent_content_response_times(&self) -> impl Iterator<Item = ContentResponseTime> + '_ {
318        let now = Instant::now();
319        self.content_response_times
320            .iter()
321            .filter(move |(ts, _)| now.duration_since(*ts) < self.window)
322            .map(|(_, response_time)| *response_time)
323    }
324
325    /// The peer's fastest recent contents response times [`Self::SLOW_RESPONSE_FACTOR`], or plus
326    /// [`Self::SLOW_RESPONSE_MIN_MARGIN`], whichever is more. Responses that take longer are
327    /// slow. `None` if the peer has no recent answers.
328    fn slow_response_threshold(&self) -> Option<Duration> {
329        let fastest = self.fastest_recent_content_answer()?;
330        Some(
331            fastest
332                .mul_f64(Self::SLOW_RESPONSE_FACTOR)
333                .max(fastest + Self::SLOW_RESPONSE_MIN_MARGIN),
334        )
335    }
336
337    /// If this peer is slow, its slow-response threshold (see `slow_response_threshold`), after
338    /// which requests to it should be hedged. The peer is slow when at least
339    /// [`Self::SLOW_PEER_SHARE`] of its recent responses took longer than the threshold. Returns
340    /// `None` if the peer isn't slow, or if it has too few recent samples to tell.
341    pub(super) fn slow_peer_hedge_delay(&self) -> Option<Duration> {
342        let threshold = self.slow_response_threshold()?;
343        let mut slow = 0;
344        let mut known = 0;
345        for response_time in self.recent_content_response_times() {
346            match response_time {
347                ContentResponseTime::Answered(elapsed) => {
348                    known += 1;
349                    if elapsed > threshold {
350                        slow += 1;
351                    }
352                }
353                ContentResponseTime::LongerThan(elapsed) if elapsed >= threshold => {
354                    known += 1;
355                    slow += 1;
356                }
357                // Cancelled before the threshold: whether the response would have been slow is
358                // unknown. Such samples are only stored while the peer has no threshold yet, or
359                // under an earlier, lower threshold.
360                ContentResponseTime::LongerThan(_) => {}
361            }
362        }
363        (known >= Self::MIN_SAMPLES_FOR_SLOW && slow as f64 >= Self::SLOW_PEER_SHARE * known as f64)
364            .then_some(threshold)
365    }
366
367    /// Called when a contents request that is not a hedge is about to be sent to this peer near
368    /// the release frontier. Every [`SLOW_PEER_PROBE_INTERVAL`]th such request sent while
369    /// the peer is slow is a slow-peer probe. Requests sent while the peer isn't slow are not
370    /// counted.
371    pub(super) fn classify_new_request(&mut self) -> SlowPeerRequest {
372        if self.slow_peer_hedge_delay().is_none() {
373            return SlowPeerRequest::PeerNotSlow;
374        }
375        self.requests_since_probe += 1;
376        if self.requests_since_probe < SLOW_PEER_PROBE_INTERVAL {
377            return SlowPeerRequest::NotProbe;
378        }
379        self.requests_since_probe = 0;
380        SlowPeerRequest::Probe
381    }

383    pub(super) fn record_failure(&mut self) {
384        let now = Instant::now();
385        self.failures.push_back(now);
386        while self.failures.len() > Self::MAX_SAMPLES {
387            self.failures.pop_front();
388        }
389    }

391    pub(super) fn is_failing(&self) -> bool {
392        let now = Instant::now();
393        let recent_failures = self
394            .failures
395            .iter()
396            .filter(|ts| now.duration_since(**ts) < self.window)
397            .count();
398        let recent_successes = self
399            .successes
400            .iter()
401            .filter(|(ts, _, _)| now.duration_since(*ts) < self.window)
402            .count();
403
404        let total = recent_failures + recent_successes;
405        if total < Self::MIN_SAMPLES_FOR_FAILURE {
406            return false;
407        }
408
409        let rate = recent_failures as f64 / total as f64;
410        rate >= self.failure_rate
411    }
412
413    pub(super) fn update_failing_state(&mut self) {
414        if self.is_failing() {
415            // Start the consistently-failing clock when the windowed failure rate first
416            // crosses the threshold, and let it accumulate while the rate stays high.
417            if self.failing_since.is_none() {
418                self.failing_since = Some(Instant::now());
419            }
420        } else {
421            // The windowed failure rate dropped below the threshold: the peer is no longer
422            // failing, so clear the clock. Only an actual drop in the rate resets it.
423            self.failing_since = None;
424        }
425    }
426
427    pub(super) fn reset_failing_since(&mut self) {
428        self.failing_since = None;
429    }
430
431    pub(super) fn consistently_failing(&self, threshold: Duration) -> bool {
432        match self.failing_since {
433            Some(since) => since.elapsed() >= threshold,
434            None => false,
435        }
436    }
437
438    pub(super) fn effective_throughput(&self) -> Option<f64> {
439        let now = Instant::now();
440        let (total_size, total_time) = self
441            .successes
442            .iter()
443            .chain(&self.cancelled)
444            .filter(|(ts, _, _)| now.duration_since(*ts) < self.window)
445            .fold((0u64, Duration::ZERO), |(size, time), (_, s, d)| {
446                (size + s, time + *d)
447            });
448
449        if total_size == 0 {
450            return None;
451        }
452
453        if total_time.is_zero() {
454            return None;
455        }
456
457        Some(total_size as f64 / total_time.as_secs_f64())
458    }

510    // Returns a bool that indicates if the update was done successfully.
511    //
512    // This will return false if the given peer doesn't have an entry or is not on the same chain
513    // as us
514    #[instrument(level = "debug", skip_all, fields(peer_id=?peer_id, checkpoint=?checkpoint.sequence_number()))]
515    pub fn update_peer_info(
516        &mut self,
517        peer_id: PeerId,
518        checkpoint: Checkpoint,
519        low_watermark: Option<CheckpointSequenceNumber>,
520    ) -> bool {
521        debug!("Update peer info");
522
523        let info = match self.peers.get_mut(&peer_id) {
524            Some(info) if info.on_same_chain_as_us => info,
525            _ => return false,
526        };
527
528        info.height = std::cmp::max(*checkpoint.sequence_number(), info.height);
529        if let Some(low_watermark) = low_watermark {
530            info.lowest = low_watermark;
531        }
532        self.insert_checkpoint(checkpoint);
533
534        true
535    }

537    #[instrument(level = "debug", skip_all, fields(peer_id=?peer_id, lowest = ?info.lowest, height = ?info.height))]
538    pub fn insert_peer_info(&mut self, peer_id: PeerId, info: PeerStateSyncInfo) {
539        use std::collections::hash_map::Entry;
540        debug!("Insert peer info");
541
542        match self.peers.entry(peer_id) {
543            Entry::Occupied(mut entry) => {
544                // If there's already an entry and the genesis checkpoint digests match then update
545                // the maximum height. Otherwise we'll use the more recent one
546                let entry = entry.get_mut();
547                if entry.genesis_checkpoint_digest == info.genesis_checkpoint_digest {
548                    entry.height = std::cmp::max(entry.height, info.height);
549                } else {
550                    *entry = info;
551                }
552            }
553            Entry::Vacant(entry) => {
554                entry.insert(info);
555            }
556        }
557    }

594    /// Inserts a checkpoint into the unprocessed checkpoints store.
595    /// Only one checkpoint per sequence number is stored. If a checkpoint with the same
596    /// sequence number but different digest already exists, the new one is dropped.
597    // TODO: also record who gives this checkpoint info for peer quality measurement?
598    pub fn insert_checkpoint(&mut self, checkpoint: Checkpoint) {
599        let digest = *checkpoint.digest();
600        let sequence_number = *checkpoint.sequence_number();
601
602        // Check if we already have a checkpoint for this sequence number.
603        if let Some(existing_digest) = self.sequence_number_to_digest.get(&sequence_number) {
604            if *existing_digest == digest {
605                // Same checkpoint, nothing to do.
606                return;
607            }
608            tracing::info!(
609                ?sequence_number,
610                ?existing_digest,
611                ?digest,
612                "received checkpoint with same sequence number but different digest, dropping new checkpoint"
613            );
614            return;
615        }
616
617        self.unprocessed_checkpoints.insert(digest, checkpoint);
618        self.sequence_number_to_digest
619            .insert(sequence_number, digest);
620    }

622    pub fn remove_checkpoint(&mut self, digest: &CheckpointDigest) {
623        if let Some(checkpoint) = self.unprocessed_checkpoints.remove(digest) {
624            self.sequence_number_to_digest
625                .remove(checkpoint.sequence_number());
626        }
627    }

761// PeerBalancer selects peers using weighted random selection:
762// - Most of the time: select from known peers, weighted by throughput
763// - With configured probability: select from unknown peers (to explore)
764// - Failing peers: only as last resort
765#[derive(Clone)]
766struct PeerBalancer {
767    known_peers: Vec<(anemo::Peer, PeerStateSyncInfo, f64)>,
768    unknown_peers: Vec<(anemo::Peer, PeerStateSyncInfo, f64)>,
769    failing_peers: Vec<(anemo::Peer, PeerStateSyncInfo)>,
770    requested_checkpoint: Option<CheckpointSequenceNumber>,
771    request_type: PeerCheckpointRequestType,
772    exploration_probability: f64,
773}
774
775#[derive(Clone)]
776enum PeerCheckpointRequestType {
777    Summary,
778    Content,
779}
780
781impl PeerBalancer {
782    pub fn new(
783        network: &anemo::Network,
784        peer_heights: Arc<RwLock<PeerHeights>>,
785        request_type: PeerCheckpointRequestType,
786    ) -> Self {
787        let peer_heights_guard = peer_heights.read().unwrap();
788
789        let mut known_peers = Vec::new();
790        let mut unknown_peers = Vec::new();
791        let mut failing_peers = Vec::new();
792        let exploration_probability = peer_heights_guard.exploration_probability;
793
794        for (peer_id, info) in peer_heights_guard.peers_on_same_chain() {
795            let Some(peer) = network.peer(*peer_id) else {
796                continue;
797            };
798            let rtt_secs = peer.connection_rtt().as_secs_f64();
799
800            if peer_heights_guard.is_failing(peer_id) {
801                failing_peers.push((peer, *info));
802            } else if let Some(throughput) = peer_heights_guard.get_throughput(peer_id) {
803                known_peers.push((peer, *info, throughput));
804            } else {
805                unknown_peers.push((peer, *info, rtt_secs));
806            }
807        }
808        drop(peer_heights_guard);
809
810        unknown_peers.sort_by(|(_, _, rtt_a), (_, _, rtt_b)| {
811            rtt_a
812                .partial_cmp(rtt_b)
813                .unwrap_or(std::cmp::Ordering::Equal)
814        });
815
816        Self {
817            known_peers,
818            unknown_peers,
819            failing_peers,
820            requested_checkpoint: None,
821            request_type,
822            exploration_probability,
823        }
824    }
825
826    pub fn with_checkpoint(mut self, checkpoint: CheckpointSequenceNumber) -> Self {
827        self.requested_checkpoint = Some(checkpoint);
828        self
829    }
830
831    /// Removes `peers` from the candidates.
832    pub fn excluding(mut self, peers: &HashSet<PeerId>) -> Self {
833        self.known_peers
834            .retain(|(peer, _, _)| !peers.contains(&peer.peer_id()));
835        self.unknown_peers
836            .retain(|(peer, _, _)| !peers.contains(&peer.peer_id()));
837        self.failing_peers
838            .retain(|(peer, _)| !peers.contains(&peer.peer_id()));
839        self
840    }
841
842    fn is_eligible(&self, info: &PeerStateSyncInfo) -> bool {
843        let requested = self.requested_checkpoint.unwrap_or(0);
844        match &self.request_type {
845            PeerCheckpointRequestType::Summary => info.height >= requested,
846            PeerCheckpointRequestType::Content => info.has_contents(requested),
847        }
848    }
849
850    fn select_by_throughput(&mut self) -> Option<(anemo::Peer, PeerStateSyncInfo)> {
851        let eligible: Vec<_> = self
852            .known_peers
853            .iter()
854            .enumerate()
855            .filter(|(_, (_, info, _))| self.is_eligible(info))
856            .map(|(i, (_, _, t))| (i, *t))
857            .collect();
858
859        if eligible.is_empty() {
860            return None;
861        }
862
863        let total: f64 = eligible.iter().map(|(_, t)| t).sum();
864        let mut pick = rand::thread_rng().gen_range(0.0..total);
865
866        for (idx, throughput) in &eligible {
867            pick -= throughput;
868            if pick <= 0.0 {
869                let (peer, info, _) = self.known_peers.remove(*idx);
870                return Some((peer, info));
871            }
872        }
873
874        let (idx, _) = eligible.last().unwrap();
875        let (peer, info, _) = self.known_peers.remove(*idx);
876        Some((peer, info))
877    }
878
879    fn select_by_rtt(&mut self) -> Option<(anemo::Peer, PeerStateSyncInfo)> {
880        let pos = self
881            .unknown_peers
882            .iter()
883            .position(|(_, info, _)| self.is_eligible(info))?;
884        let (peer, info, _) = self.unknown_peers.remove(pos);
885        Some((peer, info))
886    }
887
888    fn select_failing(&mut self) -> Option<(anemo::Peer, PeerStateSyncInfo)> {
889        let pos = self
890            .failing_peers
891            .iter()
892            .position(|(_, info)| self.is_eligible(info))?;
893        Some(self.failing_peers.remove(pos))
894    }
895}

897impl Iterator for PeerBalancer {
898    type Item = StateSyncClient<anemo::Peer>;
899
900    fn next(&mut self) -> Option<Self::Item> {
901        let has_eligible_known = self
902            .known_peers
903            .iter()
904            .any(|(_, info, _)| self.is_eligible(info));
905        let has_eligible_unknown = self
906            .unknown_peers
907            .iter()
908            .any(|(_, info, _)| self.is_eligible(info));
909
910        let explore = has_eligible_unknown
911            && (!has_eligible_known || rand::thread_rng().gen_bool(self.exploration_probability));
912
913        if explore && let Some((peer, _)) = self.select_by_rtt() {
914            return Some(StateSyncClient::new(peer));
915        }
916
917        if has_eligible_known && let Some((peer, _)) = self.select_by_throughput() {
918            return Some(StateSyncClient::new(peer));
919        }
920
921        if let Some((peer, _)) = self.select_failing() {
922            return Some(StateSyncClient::new(peer));
923        }
924
925        None
926    }
927}

930enum StateSyncMessage {
931    StartSyncJob,
932    // Validators will send this to the StateSyncEventLoop in order to kick off notifying our peers
933    // of the new checkpoint.
934    VerifiedCheckpoint(Box<VerifiedCheckpoint>),
935    // Notification that the checkpoint content sync task will send to the event loop in the event
936    // it was able to successfully sync a checkpoint's contents. If multiple checkpoints were
937    // synced at the same time, only the highest checkpoint is sent.
938    SyncedCheckpoint(Box<VerifiedCheckpoint>),
939}

941struct StateSyncEventLoop<S> {
942    config: StateSyncConfig,
943
944    mailbox: mpsc::Receiver<StateSyncMessage>,
945    /// Weak reference to our own mailbox
946    weak_sender: mpsc::WeakSender<StateSyncMessage>,
947
948    tasks: JoinSet<()>,
949    sync_checkpoint_summaries_task: Option<AbortHandle>,
950    sync_checkpoint_contents_task: Option<AbortHandle>,
951    download_limit_layer: Option<CheckpointContentsDownloadLimitLayer>,
952
953    store: S,
954    peer_heights: Arc<RwLock<PeerHeights>>,
955    checkpoint_event_sender: broadcast::Sender<VerifiedCheckpoint>,
956    network: anemo::Network,
957    metrics: Metrics,
958
959    sync_checkpoint_from_archive_task: Option<AbortHandle>,
960    archive_config: Option<ArchiveReaderConfig>,
961    discovery_sender: Option<crate::discovery::Sender>,
962}
963
964impl<S> StateSyncEventLoop<S>
965where
966    S: WriteStore + Clone + Send + Sync + 'static,
967{
968    // Note: A great deal of care is taken to ensure that all event handlers are non-asynchronous
969    // and that the only "await" points are from the select macro picking which event to handle.
970    // This ensures that the event loop is able to process events at a high speed and reduce the
971    // chance for building up a backlog of events to process.
972    pub async fn start(mut self) {
973        info!("State-Synchronizer started");
974
975        self.config.pinned_checkpoints.sort();
976
977        let mut interval = tokio::time::interval(self.config.interval_period());
978        let mut peer_events = {
979            let (subscriber, peers) = self.network.subscribe().unwrap();
980            for peer_id in peers {
981                self.spawn_get_latest_from_peer(peer_id);
982            }
983            subscriber
984        };
985        let (
986            target_checkpoint_contents_sequence_sender,
987            target_checkpoint_contents_sequence_receiver,
988        ) = watch::channel(0);
989
990        // Spawn tokio task to update metrics periodically in the background
991        let (_sender, receiver) = oneshot::channel();
992        tokio::spawn(update_checkpoint_watermark_metrics(
993            receiver,
994            self.store.clone(),
995            self.metrics.clone(),
996        ));
997
998        // Start checkpoint contents sync loop.
999        let task = sync_checkpoint_contents(
1000            self.network.clone(),
1001            self.store.clone(),
1002            self.peer_heights.clone(),
1003            self.weak_sender.clone(),
1004            self.checkpoint_event_sender.clone(),
1005            self.config.checkpoint_content_download_concurrency(),
1006            self.config.checkpoint_content_download_tx_concurrency(),
1007            self.config.use_get_checkpoint_contents_v2(),
1008            self.config.checkpoint_content_hedge_delay(),
1009            self.metrics.clone(),
1010            target_checkpoint_contents_sequence_receiver,
1011        );
1012        let task_handle = self.tasks.spawn(task);
1013        self.sync_checkpoint_contents_task = Some(task_handle);
1014
1015        // Start archive based checkpoint content sync loop.
1016        // TODO: Consider switching to sync from archive only on startup.
1017        // Right now because the peer set is fixed at startup, a node may eventually
1018        // end up with peers who have all purged their local state. In such a scenario it will be
1019        // stuck until restart when it ends up with a different set of peers. Once the discovery
1020        // mechanism can dynamically identify and connect to other peers on the network, we will rely
1021        // on sync from archive as a fall back.
1022        let task = sync_checkpoint_contents_from_archive(
1023            self.network.clone(),
1024            self.archive_config.clone(),
1025            self.store.clone(),
1026            self.peer_heights.clone(),
1027            self.metrics.clone(),
1028        );
1029        let task_handle = self.tasks.spawn(task);
1030        self.sync_checkpoint_from_archive_task = Some(task_handle);
1031
1032        // Start main loop.
1033        loop {
1034            tokio::select! {
1035                now = interval.tick() => {
1036                    self.handle_tick(now.into_std());
1037                },
1038                maybe_message = self.mailbox.recv() => {
1039                    // Once all handles to our mailbox have been dropped this
1040                    // will yield `None` and we can terminate the event loop
1041                    if let Some(message) = maybe_message {
1042                        self.handle_message(message);
1043                    } else {
1044                        break;
1045                    }
1046                },
1047                peer_event = peer_events.recv() => {
1048                    self.handle_peer_event(peer_event);
1049                },
1050                Some(task_result) = self.tasks.join_next() => {
1051                    match task_result {
1052                        Ok(()) => {},
1053                        Err(e) => {
1054                            if e.is_cancelled() {
1055                                // avoid crashing on ungraceful shutdown
1056                            } else if e.is_panic() {
1057                                // propagate panics.
1058                                std::panic::resume_unwind(e.into_panic());
1059                            } else {
1060                                panic!("task failed: {e}");
1061                            }
1062                        },
1063                    };
1064
1065                    if matches!(&self.sync_checkpoint_contents_task, Some(t) if t.is_finished()) {
1066                        panic!("sync_checkpoint_contents task unexpectedly terminated")
1067                    }
1068
1069                    if matches!(&self.sync_checkpoint_summaries_task, Some(t) if t.is_finished()) {
1070                        self.sync_checkpoint_summaries_task = None;
1071                    }
1072
1073                    if matches!(&self.sync_checkpoint_from_archive_task, Some(t) if t.is_finished()) {
1074                        panic!("sync_checkpoint_from_archive task unexpectedly terminated")
1075                    }
1076                },
1077            }
1078
1079            self.maybe_start_checkpoint_summary_sync_task();
1080            self.maybe_trigger_checkpoint_contents_sync_task(
1081                &target_checkpoint_contents_sequence_sender,
1082            );
1083        }
1084
1085        info!("State-Synchronizer ended");
1086    }
1087
1088    fn handle_message(&mut self, message: StateSyncMessage) {
1089        debug!("Received message: {:?}", message);
1090        match message {
1091            StateSyncMessage::StartSyncJob => self.maybe_start_checkpoint_summary_sync_task(),
1092            StateSyncMessage::VerifiedCheckpoint(checkpoint) => {
1093                self.handle_checkpoint_from_consensus(checkpoint)
1094            }
1095            // After we've successfully synced a checkpoint we can notify our peers
1096            StateSyncMessage::SyncedCheckpoint(checkpoint) => {
1097                self.spawn_notify_peers_of_checkpoint(*checkpoint)
1098            }
1099        }
1100    }
1101
1102    // Handle a checkpoint that we received from consensus
1103    #[instrument(level = "debug", skip_all)]
1104    fn handle_checkpoint_from_consensus(&mut self, checkpoint: Box<VerifiedCheckpoint>) {
1105        // Always check previous_digest matches in case there is a gap between
1106        // state sync and consensus.
1107        let prev_digest = *self.store.get_checkpoint_by_sequence_number(checkpoint.sequence_number() - 1)
1108            .unwrap_or_else(|| panic!("Got checkpoint {} from consensus but cannot find checkpoint {} in certified_checkpoints", checkpoint.sequence_number(), checkpoint.sequence_number() - 1))
1109            .digest();
1110        if checkpoint.previous_digest != Some(prev_digest) {
1111            panic!(
1112                "Checkpoint {} from consensus has mismatched previous_digest, expected: {:?}, actual: {:?}",
1113                checkpoint.sequence_number(),
1114                Some(prev_digest),
1115                checkpoint.previous_digest
1116            );
1117        }
1118
1119        let latest_checkpoint = self
1120            .store
1121            .get_highest_verified_checkpoint()
1122            .expect("store operation should not fail");
1123
1124        // If this is an older checkpoint, just ignore it
1125        if latest_checkpoint.sequence_number() >= checkpoint.sequence_number() {
1126            return;
1127        }
1128
1129        let checkpoint = *checkpoint;
1130        let next_sequence_number = latest_checkpoint.sequence_number().checked_add(1).unwrap();
1131        if *checkpoint.sequence_number() > next_sequence_number {
1132            debug!(
1133                "consensus sent too new of a checkpoint, expecting: {}, got: {}",
1134                next_sequence_number,
1135                checkpoint.sequence_number()
1136            );
1137        }
1138
1139        // Because checkpoint from consensus sends in order, when we have checkpoint n,
1140        // we must have all of the checkpoints before n from either state sync or consensus.
1141        #[cfg(debug_assertions)]
1142        {
1143            let _ = (next_sequence_number..=*checkpoint.sequence_number())
1144                .map(|n| {
1145                    let checkpoint = self
1146                        .store
1147                        .get_checkpoint_by_sequence_number(n)
1148                        .unwrap_or_else(|| panic!("store should contain checkpoint {n}"));
1149                    self.store
1150                        .get_full_checkpoint_contents(Some(n), &checkpoint.content_digest)
1151                        .unwrap_or_else(|| {
1152                            panic!(
1153                                "store should contain checkpoint contents for {:?}",
1154                                checkpoint.content_digest
1155                            )
1156                        });
1157                })
1158                .collect::<Vec<_>>();
1159        }
1160
1161        // If this is the last checkpoint of a epoch, we need to make sure
1162        // new committee is in store before we verify newer checkpoints in next epoch.
1163        // This could happen before this validator's reconfiguration finishes, because
1164        // state sync does not reconfig.
1165        // TODO: make CheckpointAggregator use WriteStore so we don't need to do this
1166        // committee insertion in two places (only in `insert_checkpoint`).
1167        if let Some(EndOfEpochData {
1168            next_epoch_committee,
1169            ..
1170        }) = checkpoint.end_of_epoch_data.as_ref()
1171        {
1172            let next_committee = next_epoch_committee.iter().cloned().collect();
1173            let committee =
1174                Committee::new(checkpoint.epoch().checked_add(1).unwrap(), next_committee);
1175            self.store
1176                .insert_committee(committee)
1177                .expect("store operation should not fail");
1178        }
1179
1180        self.store
1181            .update_highest_verified_checkpoint(&checkpoint)
1182            .expect("store operation should not fail");
1183        self.store
1184            .update_highest_synced_checkpoint(&checkpoint)
1185            .expect("store operation should not fail");
1186
1187        // We don't care if no one is listening as this is a broadcast channel
1188        let _ = self.checkpoint_event_sender.send(checkpoint.clone());
1189
1190        self.spawn_notify_peers_of_checkpoint(checkpoint);
1191    }

1193    fn handle_peer_event(
1194        &mut self,
1195        peer_event: Result<PeerEvent, tokio::sync::broadcast::error::RecvError>,
1196    ) {
1197        use tokio::sync::broadcast::error::RecvError;
1198
1199        match peer_event {
1200            Ok(PeerEvent::NewPeer(peer_id)) => {
1201                self.spawn_get_latest_from_peer(peer_id);
1202            }
1203            Ok(PeerEvent::LostPeer(peer_id, _)) => {
1204                let mut heights = self.peer_heights.write().unwrap();
1205                heights.peers.remove(&peer_id);
1206                heights.scores.remove(&peer_id);
1207            }
1208
1209            Err(RecvError::Closed) => {

1211            }

1213            Err(RecvError::Lagged(_)) => {
1214                trace!("State-Sync fell behind processing PeerEvents");
1215            }
1216        }
1217    }

1236    fn handle_tick(&mut self, _now: std::time::Instant) {
1237        let task = query_peers_for_their_latest_checkpoint(
1238            self.network.clone(),
1239            self.peer_heights.clone(),
1240            self.weak_sender.clone(),
1241            self.config.timeout(),
1242        );
1243        self.tasks.spawn(task);
1244
1245        if let Some(layer) = self.download_limit_layer.as_ref() {
1246            layer.maybe_prune_map();
1247        }
1248
1249        self.maybe_report_failing_peer();
1250    }

1252    fn maybe_report_failing_peer(&mut self) {
1253        let threshold = self.config.peer_disconnect_threshold();
1254        if threshold.is_zero() {
1255            return;
1256        }
1257
1258        let mut peer_heights = self.peer_heights.write().unwrap();
1259        peer_heights.update_failing_states();
1260
1261        let Some(peer_id) = peer_heights.find_peer_to_report_for_failure(threshold) else {
1262            return;
1263        };
1264        // Reset the failing clock so this peer isn't re-reported on the next tick.
1265        // If the peer is still failing, update_failing_states will restart the clock
1266        // and the peer must fail for another full threshold before being reported again.
1267        // Discovery may decline to disconnect (e.g. too few peers), so this also
1268        // prevents spamming reports every tick in that case.
1269        if let Some(score) = peer_heights.scores.get_mut(&peer_id) {
1270            score.reset_failing_since();
1271        }
1272        drop(peer_heights);
1273
1274        info!("reporting peer {peer_id} for consistent state sync failures");
1275        if let Some(sender) = &self.discovery_sender {
1276            sender.report_peer_failure(peer_id);
1277        } else {
1278            let _ = self.network.disconnect(peer_id);
1279        }
1280        self.metrics.inc_peers_reported_for_failure();
1281    }

1283    fn maybe_start_checkpoint_summary_sync_task(&mut self) {
1284        // Only run one sync task at a time
1285        if self.sync_checkpoint_summaries_task.is_some() {
1286            return;
1287        }
1288
1289        let highest_processed_checkpoint = self
1290            .store
1291            .get_highest_verified_checkpoint()
1292            .expect("store operation should not fail");
1293
1294        let highest_known_sequence_number = self
1295            .peer_heights
1296            .read()
1297            .unwrap()
1298            .highest_known_checkpoint_sequence_number();
1299
1300        if let Some(target_seq) = highest_known_sequence_number
1301            && *highest_processed_checkpoint.sequence_number() < target_seq
1302        {
1303            // Limit the per-sync batch size according to config.
1304            let max_batch_size = self.config.max_checkpoint_sync_batch_size();
1305            let limited_target = std::cmp::min(
1306                target_seq,
1307                highest_processed_checkpoint
1308                    .sequence_number()
1309                    .saturating_add(max_batch_size),
1310            );
1311            let was_limited = limited_target < target_seq;
1312
1313            // Start sync job.
1314            let weak_sender = self.weak_sender.clone();
1315            let task = sync_to_checkpoint(
1316                self.network.clone(),
1317                self.store.clone(),
1318                self.peer_heights.clone(),
1319                self.metrics.clone(),
1320                self.config.pinned_checkpoints.clone(),
1321                self.config.checkpoint_header_download_concurrency(),
1322                self.config.timeout(),
1323                limited_target,
1324            )
1325            .map(move |result| match result {
1326                Ok(()) => {
1327                    // If we limited the sync range, immediately trigger
1328                    // another sync to continue catching up.
1329                    if was_limited && let Some(sender) = weak_sender.upgrade() {
1330                        let _ = sender.try_send(StateSyncMessage::StartSyncJob);
1331                    }
1332                }
1333                Err(e) => {
1334                    debug!("error syncing checkpoint {e}");
1335                }
1336            });
1337            let task_handle = self.tasks.spawn(task);
1338            self.sync_checkpoint_summaries_task = Some(task_handle);
1339        }
1340    }

1342    fn maybe_trigger_checkpoint_contents_sync_task(
1343        &mut self,
1344        target_sequence_channel: &watch::Sender<CheckpointSequenceNumber>,
1345    ) {
1346        let highest_verified_checkpoint = self
1347            .store
1348            .get_highest_verified_checkpoint()
1349            .expect("store operation should not fail");
1350        let highest_synced_checkpoint = self
1351            .store
1352            .get_highest_synced_checkpoint()
1353            .expect("store operation should not fail");
1354
1355        if highest_verified_checkpoint.sequence_number()
1356            > highest_synced_checkpoint.sequence_number()
1357            // skip if we aren't connected to any peers that can help
1358            && self
1359                .peer_heights
1360                .read()
1361                .unwrap()
1362                .highest_known_checkpoint_sequence_number()
1363                > Some(*highest_synced_checkpoint.sequence_number())
1364        {
1365            let _ = target_sequence_channel.send_if_modified(|num| {
1366                let new_num = *highest_verified_checkpoint.sequence_number();
1367                if *num == new_num {
1368                    return false;
1369                }
1370                *num = new_num;
1371                true
1372            });
1373        }
1374    }

1412async fn get_latest_from_peer(
1413    our_genesis_checkpoint_digest: CheckpointDigest,
1414    peer: anemo::Peer,
1415    peer_heights: Arc<RwLock<PeerHeights>>,
1416    timeout: Duration,
1417) {
1418    let peer_id = peer.peer_id();
1419    let mut client = StateSyncClient::new(peer);
1420
1421    let info = {
1422        let maybe_info = peer_heights.read().unwrap().peers.get(&peer_id).copied();
1423
1424        if let Some(info) = maybe_info {
1425            info
1426        } else {
1427            // TODO do we want to create a new API just for querying a node's chainid?
1428            //
1429            // We need to query this node's genesis checkpoint to see if they're on the same chain
1430            // as us
1431            let request = Request::new(GetCheckpointSummaryRequest::BySequenceNumber(0))
1432                .with_timeout(timeout);
1433            let response = client
1434                .get_checkpoint_summary(request)
1435                .await
1436                .map(Response::into_inner);
1437
1438            let info = match response {
1439                Ok(Some(checkpoint)) => {
1440                    let digest = *checkpoint.digest();
1441                    PeerStateSyncInfo {
1442                        genesis_checkpoint_digest: digest,
1443                        on_same_chain_as_us: our_genesis_checkpoint_digest == digest,
1444                        height: *checkpoint.sequence_number(),
1445                        lowest: CheckpointSequenceNumber::default(),
1446                    }
1447                }
1448                Ok(None) => PeerStateSyncInfo {
1449                    genesis_checkpoint_digest: CheckpointDigest::default(),
1450                    on_same_chain_as_us: false,
1451                    height: CheckpointSequenceNumber::default(),
1452                    lowest: CheckpointSequenceNumber::default(),
1453                },
1454                Err(status) => {
1455                    trace!("get_latest_checkpoint_summary request failed: {status:?}");
1456                    return;
1457                }
1458            };
1459            peer_heights
1460                .write()
1461                .unwrap()
1462                .insert_peer_info(peer_id, info);
1463            info
1464        }
1465    };
1466
1467    // Bail early if this node isn't on the same chain as us
1468    if !info.on_same_chain_as_us {
1469        trace!(?info, "Peer {peer_id} not on same chain as us");
1470        return;
1471    }
1472    let Ok(Some((highest_checkpoint, low_watermark))) =
1473        query_peer_for_latest_info(&mut client, timeout).await
1474    else {
1475        return;
1476    };
1477    peer_heights
1478        .write()
1479        .unwrap()
1480        .update_peer_info(peer_id, highest_checkpoint, low_watermark);
1481}

1483/// Queries a peer for their highest_synced_checkpoint and low checkpoint watermark.
1484///
1485/// Distinguishes three outcomes so callers can score the peer:
1486/// - `Ok(Some(..))`: the peer answered with usable latest-checkpoint info.
1487/// - `Ok(None)`: the peer answered but had nothing useful to report (e.g. a pre-upgrade peer
1488///   that returns no latest summary). This is not a failure.
1489/// - `Err(())`: the request itself failed (timeout / transport error). This is a failure: a
1490///   peer that consistently fails these liveness queries is consistently unreachable.
1491async fn query_peer_for_latest_info(
1492    client: &mut StateSyncClient<anemo::Peer>,
1493    timeout: Duration,
1494) -> Result<Option<(Checkpoint, Option<CheckpointSequenceNumber>)>, ()> {
1495    let request = Request::new(()).with_timeout(timeout);
1496    let response = client
1497        .get_checkpoint_availability(request)
1498        .await
1499        .map(Response::into_inner);
1500    match response {
1501        Ok(GetCheckpointAvailabilityResponse {
1502            highest_synced_checkpoint,
1503            lowest_available_checkpoint,
1504        }) => {
1505            return Ok(Some((
1506                highest_synced_checkpoint,
1507                Some(lowest_available_checkpoint),
1508            )));
1509        }
1510        Err(status) => {
1511            // If peer hasn't upgraded they would return 404 NotFound error
1512            if status.status() != anemo::types::response::StatusCode::NotFound {
1513                trace!("get_checkpoint_availability request failed: {status:?}");
1514                return Err(());
1515            }
1516        }
1517    };
1518
1519    // Then we try the old query
1520    // TODO: remove this once the new feature stabilizes
1521    let request = Request::new(GetCheckpointSummaryRequest::Latest).with_timeout(timeout);
1522    let response = client
1523        .get_checkpoint_summary(request)
1524        .await
1525        .map(Response::into_inner);
1526    match response {
1527        Ok(Some(checkpoint)) => Ok(Some((checkpoint, None))),
1528        Ok(None) => Ok(None),
1529        Err(status) => {
1530            trace!("get_checkpoint_summary (latest) request failed: {status:?}");
1531            Err(())
1532        }
1533    }
1534}

1536#[instrument(level = "debug", skip_all)]
1537async fn query_peers_for_their_latest_checkpoint(
1538    network: anemo::Network,
1539    peer_heights: Arc<RwLock<PeerHeights>>,
1540    sender: mpsc::WeakSender<StateSyncMessage>,
1541    timeout: Duration,
1542) {
1543    let peer_heights = &peer_heights;
1544    let futs = peer_heights
1545        .read()
1546        .unwrap()
1547        .peers_on_same_chain()
1548        // Filter out any peers who we aren't connected with
1549        .flat_map(|(peer_id, _info)| network.peer(*peer_id))
1550        .map(|peer| {
1551            let peer_id = peer.peer_id();
1552            let mut client = StateSyncClient::new(peer);
1553
1554            async move {
1555                let start = Instant::now();
1556                let response = query_peer_for_latest_info(&mut client, timeout).await;
1557                let elapsed = start.elapsed();
1558                match response {
1559                    Ok(Some((highest_checkpoint, low_watermark))) => {
1560                        // A peer that answers our periodic liveness/discovery query is healthy;
1561                        // record it so its success rate offsets occasional failures.
1562                        let size = bcs::serialized_size(&highest_checkpoint).unwrap_or(0) as u64;
1563                        let mut peer_heights = peer_heights.write().unwrap();
1564                        peer_heights.record_success(peer_id, size, elapsed);
1565                        peer_heights
1566                            .update_peer_info(peer_id, highest_checkpoint.clone(), low_watermark)
1567                            .then_some(highest_checkpoint)
1568                    }
1569                    // Peer responded but had nothing useful; neutral, do not score.
1570                    Ok(None) => None,
1571                    // Request failed (timeout / transport). A peer that consistently fails
1572                    // these liveness queries is consistently unreachable, so record the
1573                    // failure to feed the scorer even when we are not actively syncing from
1574                    // it. Without this, an unreachable peer is only scored while there is
1575                    // active sync traffic, so it can evade the consistently-failing-peer
1576                    // disconnect logic indefinitely.
1577                    Err(()) => {
1578                        peer_heights.write().unwrap().record_failure(peer_id);
1579                        None
1580                    }
1581                }
1582            }
1583        })
1584        .collect::<Vec<_>>();
1585
1586    debug!("Query {} peers for latest checkpoint", futs.len());
1587
1588    let checkpoints = futures::future::join_all(futs).await.into_iter().flatten();
1589
1590    let highest_checkpoint_seq = checkpoints
1591        .map(|checkpoint| *checkpoint.sequence_number())
1592        .max();
1593
1594    let our_highest_seq = peer_heights
1595        .read()
1596        .unwrap()
1597        .highest_known_checkpoint_sequence_number();
1598
1599    debug!(
1600        "Our highest checkpoint {our_highest_seq:?}, peers' highest checkpoint {highest_checkpoint_seq:?}"
1601    );
1602
1603    let _new_checkpoint = match (highest_checkpoint_seq, our_highest_seq) {
1604        (Some(theirs), None) => theirs,
1605        (Some(theirs), Some(ours)) if theirs > ours => theirs,
1606        _ => return,
1607    };
1608
1609    if let Some(sender) = sender.upgrade() {
1610        let _ = sender.send(StateSyncMessage::StartSyncJob).await;
1611    }
1612}

1614async fn sync_to_checkpoint<S>(
1615    network: anemo::Network,
1616    store: S,
1617    peer_heights: Arc<RwLock<PeerHeights>>,
1618    metrics: Metrics,
1619    pinned_checkpoints: Vec<(CheckpointSequenceNumber, CheckpointDigest)>,
1620    checkpoint_header_download_concurrency: usize,
1621    timeout: Duration,
1622    target_sequence_number: CheckpointSequenceNumber,
1623) -> Result<()>
1624where
1625    S: WriteStore,
1626{
1627    metrics.set_highest_known_checkpoint(target_sequence_number);
1628
1629    let mut current = store
1630        .get_highest_verified_checkpoint()
1631        .expect("store operation should not fail");
1632    if *current.sequence_number() >= target_sequence_number {
1633        return Err(anyhow::anyhow!(
1634            "target checkpoint {} is older than highest verified checkpoint {}",
1635            target_sequence_number,
1636            current.sequence_number(),
1637        ));
1638    }
1639
1640    let peer_balancer = PeerBalancer::new(
1641        &network,
1642        peer_heights.clone(),
1643        PeerCheckpointRequestType::Summary,
1644    );
1645    // range of the next sequence_numbers to fetch
1646    let mut request_stream = (current.sequence_number().checked_add(1).unwrap()
1647        ..=target_sequence_number)
1648        .map(|next| {
1649            let peers = peer_balancer.clone().with_checkpoint(next);
1650            let peer_heights = peer_heights.clone();
1651            let pinned_checkpoints = &pinned_checkpoints;
1652            async move {
1653                if let Some(checkpoint) = peer_heights
1654                    .read()
1655                    .unwrap()
1656                    .get_checkpoint_by_sequence_number(next)
1657                {
1658                    return (Some(checkpoint.to_owned()), next, None);
1659                }
1660
1661                // Iterate through peers trying each one in turn until we're able to
1662                // successfully get the target checkpoint.
1663                for mut peer in peers {
1664                    let peer_id = peer.inner().peer_id();
1665                    let request = Request::new(GetCheckpointSummaryRequest::BySequenceNumber(next))
1666                        .with_timeout(timeout);
1667                    let start = Instant::now();
1668                    let result = peer.get_checkpoint_summary(request).await;
1669                    let elapsed = start.elapsed();
1670
1671                    let checkpoint = match result {
1672                        Ok(response) => match response.into_inner() {
1673                            Some(cp) => Some(cp),
1674                            None => {
1675                                trace!("peer unable to help sync");
1676                                peer_heights.write().unwrap().record_failure(peer_id);
1677                                None
1678                            }
1679                        },
1680                        Err(e) => {
1681                            trace!("{e:?}");
1682                            peer_heights.write().unwrap().record_failure(peer_id);
1683                            None
1684                        }
1685                    };
1686
1687                    let Some(checkpoint) = checkpoint else {
1688                        continue;
1689                    };
1690
1691                    let size = bcs::serialized_size(&checkpoint).expect("serialization should not fail") as u64;
1692                    peer_heights.write().unwrap().record_success(peer_id, size, elapsed);
1693
1694                    // peer didn't give us a checkpoint with the height that we requested
1695                    if *checkpoint.sequence_number() != next {
1696                        tracing::debug!(
1697                            "peer returned checkpoint with wrong sequence number: expected {next}, got {}",
1698                            checkpoint.sequence_number()
1699                        );
1700                        peer_heights
1701                            .write()
1702                            .unwrap()
1703                            .mark_peer_as_not_on_same_chain(peer_id);
1704                        continue;
1705                    }
1706
1707                    // peer gave us a checkpoint whose digest does not match pinned digest
1708                    let checkpoint_digest = checkpoint.digest();
1709                    if let Ok(pinned_digest_index) = pinned_checkpoints.binary_search_by_key(
1710                        checkpoint.sequence_number(),
1711                        |(seq_num, _digest)| *seq_num
1712                    )
1713                        && pinned_checkpoints[pinned_digest_index].1 != *checkpoint_digest
1714                    {
1715                        tracing::debug!(
1716                            "peer returned checkpoint with digest that does not match pinned digest: expected {:?}, got {:?}",
1717                            pinned_checkpoints[pinned_digest_index].1,
1718                            checkpoint_digest
1719                        );
1720                        peer_heights
1721                            .write()
1722                            .unwrap()
1723                            .mark_peer_as_not_on_same_chain(peer.inner().peer_id());
1724                        continue;
1725                    }
1726
1727                    // Insert in our store in the event that things fail and we need to retry
1728                    peer_heights
1729                        .write()
1730                        .unwrap()
1731                        .insert_checkpoint(checkpoint.clone());
1732                    return (Some(checkpoint), next, Some(peer_id));
1733                }
1734                (None, next, None)
1735            }
1736        })
1737        .pipe(futures::stream::iter)
1738        .buffered(checkpoint_header_download_concurrency);
1739
1740    while let Some((maybe_checkpoint, next, maybe_peer_id)) = request_stream.next().await {
1741        assert_eq!(
1742            current
1743                .sequence_number()
1744                .checked_add(1)
1745                .expect("exhausted u64"),
1746            next
1747        );
1748
1749        // Verify the checkpoint
1750        let checkpoint = 'cp: {
1751            let checkpoint = maybe_checkpoint.ok_or_else(|| {
1752                anyhow::anyhow!("no peers were able to help sync checkpoint {next}")
1753            })?;
1754            // Skip verification for manually pinned checkpoints.
1755            if pinned_checkpoints
1756                .binary_search_by_key(checkpoint.sequence_number(), |(seq_num, _digest)| *seq_num)
1757                .is_ok()
1758            {
1759                break 'cp VerifiedCheckpoint::new_unchecked(checkpoint);
1760            }
1761            match verify_checkpoint(&current, &store, checkpoint) {
1762                Ok(verified_checkpoint) => verified_checkpoint,
1763                Err(checkpoint) => {
1764                    let mut peer_heights = peer_heights.write().unwrap();
1765                    // Remove the checkpoint from our temporary store so that we can try querying
1766                    // another peer for a different one
1767                    peer_heights.remove_checkpoint(checkpoint.digest());
1768
1769                    // Mark peer as not on the same chain as us
1770                    if let Some(peer_id) = maybe_peer_id {
1771                        peer_heights.mark_peer_as_not_on_same_chain(peer_id);
1772                    }
1773
1774                    return Err(anyhow::anyhow!(
1775                        "unable to verify checkpoint {checkpoint:?}"
1776                    ));
1777                }
1778            }
1779        };
1780
1781        debug!(checkpoint_seq = ?checkpoint.sequence_number(), "verified checkpoint summary");
1782        if let Some((checkpoint_summary_age_metric, checkpoint_summary_age_metric_deprecated)) =
1783            metrics.checkpoint_summary_age_metrics()
1784        {
1785            checkpoint.report_checkpoint_age(
1786                checkpoint_summary_age_metric,
1787                checkpoint_summary_age_metric_deprecated,
1788            );
1789        }
1790
1791        // Insert the newly verified checkpoint into our store, which will bump our highest
1792        // verified checkpoint watermark as well.
1793        store
1794            .insert_checkpoint(&checkpoint)
1795            .expect("store operation should not fail");
1796
1797        current = checkpoint;
1798    }
1799
1800    peer_heights
1801        .write()
1802        .unwrap()
1803        .cleanup_old_checkpoints(*current.sequence_number());
1804
1805    Ok(())
1806}

1844        .filter_map(|(peer_id, info)| network.peer(*peer_id).map(|peer| (peer, *info)))
1845        .collect();
1846    let lowest_checkpoint_on_peers = peers
1847        .iter()
1848        .map(|(_p, state_sync_info)| state_sync_info.lowest)
1849        .min();
1850    let highest_synced = store
1851        .get_highest_synced_checkpoint()
1852        .expect("store operation should not fail")
1853        .sequence_number;
1854    let sync_from_archive = if let Some(lowest_checkpoint_on_peers) = lowest_checkpoint_on_peers {
1855        highest_synced < lowest_checkpoint_on_peers
1856    } else {
1857        false
1858    };
1859    debug!(
1860        "Syncing checkpoint contents from archive: {sync_from_archive},  highest_synced: {highest_synced},  lowest_checkpoint_on_peers: {}",
1861        lowest_checkpoint_on_peers.map_or_else(|| "None".to_string(), |l| l.to_string())
1862    );
1863    if sync_from_archive {
1864        let start = highest_synced
1865            .checked_add(1)
1866            .expect("Checkpoint seq num overflow");
1867        let end = lowest_checkpoint_on_peers.unwrap();
1868
1869        let Some(archive_config) = archive_config else {
1870            warn!("Failed to find an archive reader to complete the state sync request");
1871            return;
1872        };
1873        let Some(ingestion_url) = &archive_config.ingestion_url else {
1874            warn!("Archival ingestion url for state sync is not configured");
1875            return;
1876        };
1877        if ingestion_url.contains("checkpoints.mainnet.sui.io") {
1878            warn!("{} can't be used as an archival fallback", ingestion_url);
1879            return;
1880        }
1881        let obj_store = build_object_store(
1882            ingestion_url,
1883            archive_config.remote_store_options.clone(),
1884            archive_config.remote_store_headers.clone(),
1885        );
1886        let mut checkpoint_stream = futures::stream::iter(start..=end)
1887            .map(|seq| {
1888                let obj_store = obj_store.clone();
1889                async move { (seq, fetch_checkpoint(&obj_store, seq).await) }
1890            })
1891            .buffered(archive_config.download_concurrency.get());
1892
1893        while let Some((seq, result)) = checkpoint_stream.next().await {
1894            let checkpoint = match result {
1895                Ok(checkpoint) => checkpoint,
1896                Err(err) => {
1897                    warn!("State sync from archive failed fetching checkpoint {seq}: {err}");
1898                    return;
1899                }
1900            };
1901            if let Err(err) = process_archive_checkpoint(&store, &checkpoint, &metrics) {
1902                warn!("State sync from archive failed processing checkpoint {seq}: {err}");
1903                return;
1904            }
1905        }
1906    }
1907}

1909async fn sync_checkpoint_contents<S>(
1910    network: anemo::Network,
1911    store: S,
1912    peer_heights: Arc<RwLock<PeerHeights>>,
1913    sender: mpsc::WeakSender<StateSyncMessage>,
1914    checkpoint_event_sender: broadcast::Sender<VerifiedCheckpoint>,
1915    checkpoint_content_download_concurrency: usize,
1916    checkpoint_content_download_tx_concurrency: u64,
1917    use_get_checkpoint_contents_v2: bool,
1918    checkpoint_content_hedge_delay: Option<Duration>,
1919    metrics: Metrics,
1920    mut target_sequence_channel: watch::Receiver<CheckpointSequenceNumber>,
1921) where
1922    S: WriteStore + Clone,
1923{
1924    let mut highest_synced = store
1925        .get_highest_synced_checkpoint()
1926        .expect("store operation should not fail");
1927
1928    let mut current_sequence = highest_synced.sequence_number().checked_add(1).unwrap();
1929    let mut target_sequence_cursor = 0;
1930    let mut highest_started_network_total_transactions = highest_synced.network_total_transactions;
1931    let mut checkpoint_contents_tasks = FuturesOrdered::new();
1932
1933    let mut tx_concurrency_remaining = checkpoint_content_download_tx_concurrency;
1934
1935    let hedge = checkpoint_content_hedge_delay.map(|delay| ContentHedge {
1936        delay,
1937        highest_synced: Arc::new(AtomicU64::new(*highest_synced.sequence_number())),
1938    });
1939
1940    loop {
1941        tokio::select! {
1942            result = target_sequence_channel.changed() => {
1943                match result {
1944                    Ok(()) => {
1945                        target_sequence_cursor = (*target_sequence_channel.borrow_and_update()).checked_add(1).unwrap();
1946                    }
1947                    Err(_) => {
1948                        // Watch channel is closed, exit loop.
1949                        return
1950                    }
1951                }
1952            },
1953            Some(maybe_checkpoint) = checkpoint_contents_tasks.next() => {
1954                match maybe_checkpoint {
1955                    Ok(checkpoint) => {
1956                        let _: &VerifiedCheckpoint = &checkpoint;  // type hint
1957
1958                        store
1959                            .update_highest_synced_checkpoint(&checkpoint)
1960                            .expect("store operation should not fail");
1961                        // We don't care if no one is listening as this is a broadcast channel
1962                        let _ = checkpoint_event_sender.send(checkpoint.clone());
1963                        tx_concurrency_remaining += checkpoint.network_total_transactions - highest_synced.network_total_transactions;
1964                        highest_synced = checkpoint;
1965                        if let Some(hedge) = &hedge {
1966                            hedge
1967                                .highest_synced
1968                                .store(*highest_synced.sequence_number(), Ordering::Relaxed);
1969                        }
1970
1971                    }
1972                    Err(checkpoint) => {
1973                        let _: &VerifiedCheckpoint = &checkpoint;  // type hint
1974                        if let Some(lowest_peer_checkpoint) =
1975                            peer_heights.read().ok().and_then(|x| x.peers.values().map(|state_sync_info| state_sync_info.lowest).min()) {
1976                            if checkpoint.sequence_number() >= &lowest_peer_checkpoint {
1977                                info!("unable to sync contents of checkpoint through state sync {} with lowest peer checkpoint: {}", checkpoint.sequence_number(), lowest_peer_checkpoint);
1978                            }
1979                        } else {
1980                            info!("unable to sync contents of checkpoint through state sync {}", checkpoint.sequence_number());
1981
1982                        }
1983                        // Calculate tx_count for retry by getting previous checkpoint
1984                        let retry_tx_count = if *checkpoint.sequence_number() == 0 {
1985                            checkpoint.network_total_transactions
1986                        } else {
1987                            let prev = store
1988                                .get_checkpoint_by_sequence_number(checkpoint.sequence_number() - 1)
1989                                .expect("previous checkpoint must exist")
1990                                .network_total_transactions;
1991                            checkpoint.network_total_transactions - prev
1992                        };
1993                        // Retry contents sync on failure.
1994                        checkpoint_contents_tasks.push_front(sync_one_checkpoint_contents(
1995                            network.clone(),
1996                            &store,
1997                            peer_heights.clone(),
1998                            use_get_checkpoint_contents_v2,
1999                            hedge.clone(),
2000                            metrics.clone(),
2001                            retry_tx_count,
2002                            checkpoint,
2003                        ));
2004                    }
2005                }
2006            },
2007        }
2008
2009        // Start new tasks up to configured concurrency limits.
2010        while current_sequence < target_sequence_cursor
2011            && checkpoint_contents_tasks.len() < checkpoint_content_download_concurrency
2012        {
2013            let next_checkpoint = store
2014                .get_checkpoint_by_sequence_number(current_sequence)
2015                .unwrap_or_else(|| panic!(
2016                    "BUG: store should have all checkpoints older than highest_verified_checkpoint (checkpoint {})",
2017                    current_sequence
2018                ));
2019
2020            // Enforce transaction count concurrency limit.
2021            let tx_count = next_checkpoint.network_total_transactions
2022                - highest_started_network_total_transactions;
2023            if tx_count > tx_concurrency_remaining {
2024                break;
2025            }
2026            tx_concurrency_remaining -= tx_count;
2027
2028            highest_started_network_total_transactions = next_checkpoint.network_total_transactions;
2029            current_sequence += 1;
2030            checkpoint_contents_tasks.push_back(sync_one_checkpoint_contents(
2031                network.clone(),
2032                &store,
2033                peer_heights.clone(),
2034                use_get_checkpoint_contents_v2,
2035                hedge.clone(),
2036                metrics.clone(),
2037                tx_count,
2038                next_checkpoint,
2039            ));
2040        }
2041
2042        if highest_synced
2043            .sequence_number()
2044            .is_multiple_of(checkpoint_content_download_concurrency as u64)
2045            || checkpoint_contents_tasks.is_empty()
2046        {
2047            // Periodically notify event loop to notify our peers that we've synced to a new checkpoint height
2048            if let Some(sender) = sender.upgrade() {
2049                let message = StateSyncMessage::SyncedCheckpoint(Box::new(highest_synced.clone()));
2050                let _ = sender.send(message).await;
2051            }
2052        }
2053    }
2054}

2056// Err hands the checkpoint back so the caller can retry it; Ok carries the same type, so boxing
2057// the error would not shrink the Result.
2058#[allow(clippy::result_large_err)]
2059#[instrument(level = "debug", skip_all, fields(sequence_number = ?checkpoint.sequence_number()))]
2060async fn sync_one_checkpoint_contents<S>(
2061    network: anemo::Network,
2062    store: S,
2063    peer_heights: Arc<RwLock<PeerHeights>>,
2064    use_get_checkpoint_contents_v2: bool,
2065    hedge: Option<ContentHedge>,
2066    metrics: Metrics,
2067    tx_count: u64,
2068    checkpoint: VerifiedCheckpoint,
2069) -> Result<VerifiedCheckpoint, VerifiedCheckpoint>
2070where
2071    S: WriteStore + Clone,
2072{
2073    let (timeout_min, timeout_max) = {
2074        let ph = peer_heights.read().unwrap();
2075        (
2076            ph.checkpoint_content_timeout_min,
2077            ph.checkpoint_content_timeout_max,
2078        )
2079    };
2080    let timeout = compute_adaptive_timeout(tx_count, timeout_min, timeout_max);
2081    debug!(
2082        "syncing checkpoint contents with adaptive timeout {:?} for {} txns",
2083        timeout, tx_count
2084    );
2085
2086    // Check if we already have produced this checkpoint locally. If so, we don't need
2087    // to get it from peers anymore.
2088    if store
2089        .get_highest_synced_checkpoint()
2090        .expect("store operation should not fail")
2091        .sequence_number()
2092        >= checkpoint.sequence_number()
2093    {
2094        debug!("checkpoint was already created via consensus output");
2095        return Ok(checkpoint);
2096    }
2097
2098    // Request checkpoint contents from peers.
2099    let now = tokio::time::Instant::now();
2100    let fetch = ContentFetch {
2101        network: &network,
2102        peer_heights: &peer_heights,
2103        checkpoint: &checkpoint,
2104        use_get_checkpoint_contents_v2,
2105        timeout,
2106        metrics: &metrics,
2107    };
2108    let Some(_contents) = get_full_checkpoint_contents(fetch, &store, hedge.as_ref()).await else {
2109        // Delay completion in case of error so we don't hammer the network with retries.
2110        let duration = peer_heights
2111            .read()
2112            .unwrap()
2113            .wait_interval_when_no_peer_to_sync_content();
2114        if now.elapsed() < duration {
2115            let duration = duration - now.elapsed();
2116            info!("retrying checkpoint sync after {:?}", duration);
2117            tokio::time::sleep(duration).await;
2118        }
2119        return Err(checkpoint);
2120    };
2121    debug!("completed checkpoint contents sync");
2122    Ok(checkpoint)
2123}

2125/// In hedged mode, the most requests for one checkpoint's contents that are in flight at once.
2126const MAX_CONCURRENT_CONTENT_REQUESTS: usize = 2;
2127
2128/// In hedged mode, only checkpoints at most this far past the highest synced checkpoint are
2129/// hedged, or keep looking for a peer after every peer that had the checkpoint failed.
2130/// Checkpoints are released in order, so a later checkpoint can't be released before the ones
2131/// ahead of it. Extra requests for it would add load (for example during catch-up, with hundreds
2132/// of downloads in flight) without reducing how long it takes to release.
2133const CONTENT_HEDGE_WINDOW: CheckpointSequenceNumber = 8;
2134
2135/// How often a download near the release frontier checks which peers have announced the
2136/// checkpoint: while a hedge is due or every request failed, and, for measuring when peers
2137/// announce it and for early hedges, for the whole download after its first request.
2138const CONTENT_CANDIDATE_POLL_INTERVAL: Duration = Duration::from_millis(10);
2139
2140/// How much sooner a peer that hasn't been asked yet must be expected to finish than the request
2141/// in flight for a hedge to go out to it right away. Keeps small differences in expected response
2142/// times, which are within their measurement noise, from sending extra requests.
2143const CLOSER_PEER_MIN_GAIN: Duration = Duration::from_millis(50);
2144
2145/// Near the release frontier, one in this many requests that are not hedges sent to a peer
2146/// while it is slow from this node is a slow-peer probe. Requests to a slow peer are otherwise
2147/// hedged right away once a peer that isn't slow has the checkpoint, so their slow outcomes are
2148/// cut short before the slow peer's slow-response threshold, where they don't count (see
2149/// `PeerScore::slow_peer_hedge_delay`), while their fast answers still arrive. Recording those
2150/// would make the peer look less slow than it is, so while a peer is slow, only probes add
2151/// response-time samples (see `SlowPeerRequest::records_response_time`). Samples from before it
2152/// became slow still count until newer ones replace them or they age out.
2153/// A probe is only hedged after its hedge delay (see `ContentFetch::hedge_delay_for`) or to a
2154/// peer expected to finish at least [`CLOSER_PEER_MIN_GAIN`] sooner, so it usually ends in a
2155/// sample that counts: an answer, or a cancellation at or after the threshold. Probes keep the
2156/// slow flag measurable and re-check the peer every so often, so that it stops being treated as
2157/// slow once it answers in time again.
2158const SLOW_PEER_PROBE_INTERVAL: u32 = 4;

2160/// Settings for hedged checkpoint contents downloads.

2163    /// How long a request to a peer that is not slow from this node may run before another peer
2164    /// is also asked. Requests to a slow peer are hedged after the peer's slow-response threshold
2165    /// if that is shorter (see [`ContentFetch::hedge_delay_for`]).
2166    delay: Duration,
2167    /// Sequence number of the highest checkpoint whose contents are synced and released.
2168    highest_synced: Arc<AtomicU64>,
2169}

2174            <= self

2406                    self.network,
2407                    peer_id,
2408                ));
2409        }
2410        slow_peer
2411    }

2413    fn request(
2414        &self,
2415        mut peer: StateSyncClient<anemo::Peer>,
2416        kind: ContentRequestKind,
2417    ) -> impl Future<Output = ContentResponse> {
2418        let peer_id = peer.inner().peer_id();
2419        let digest = self.checkpoint.content_digest;
2420        let timeout = self.timeout;
2421        let use_get_checkpoint_contents_v2 = self.use_get_checkpoint_contents_v2;
2422        debug!(
2423            ?timeout,
2424            ?kind,
2425            "requesting checkpoint contents from {}",
2426            peer_id
2427        );
2428        async move {
2429            let request = Request::new(digest).with_timeout(timeout);
2430            let start = Instant::now();
2431            let result = if use_get_checkpoint_contents_v2 {
2432                peer.get_checkpoint_contents_v2(request).await
2433            } else {
2434                peer.get_checkpoint_contents(request)
2435                    .await
2436                    .map(|r| r.map(|c| c.map(VersionedFullCheckpointContents::V1)))
2437            };
2438            let elapsed = start.elapsed();
2439            let contents = match result {
2440                Ok(response) => {
2441                    let contents = response.into_inner();
2442                    if contents.is_none() {
2443                        trace!("peer unable to help sync");
2444                    }
2445                    contents
2446                }
2447                Err(e) => {
2448                    trace!("{e:?}");
2449                    None
2450                }
2451            };
2452            ContentResponse {
2453                peer_id,
2454                elapsed,
2455                contents,
2456            }
2457        }
2458    }

2460    /// Scores the peer and records metrics for a finished request. Returns the contents and
2461    /// their serialized size if they match the checkpoint's content digest.
2462    fn record_response(
2463        &self,
2464        response: ContentResponse,
2465    ) -> Option<(VersionedFullCheckpointContents, u64)> {
2466        let ContentResponse {
2467            peer_id,
2468            elapsed,
2469            contents,
2470        } = response;
2471        let peer_label = content_metrics_peer_label(self.network, peer_id);
2472        let Some(contents) = contents else {
2473            self.peer_heights.write().unwrap().record_failure(peer_id);
2474            self.metrics.record_checkpoint_content_peer_request(
2475                &peer_label,
2476                ContentRequestOutcome::Failure,
2477                elapsed,
2478            );
2479            return None;
2480        };
2481        if contents
2482            .verify_digests(self.checkpoint.content_digest)
2483            .is_err()
2484        {
2485            self.metrics.record_checkpoint_content_peer_request(
2486                &peer_label,
2487                ContentRequestOutcome::Failure,
2488                elapsed,
2489            );
2490            return None;
2491        }
2492        let size = bcs::serialized_size(&contents).expect("serialization should not fail") as u64;
2493        self.peer_heights
2494            .write()
2495            .unwrap()
2496            .record_success(peer_id, size, elapsed);
2497        self.metrics.record_checkpoint_content_peer_request(
2498            &peer_label,
2499            ContentRequestOutcome::Success,
2500            elapsed,
2501        );
2502        Some((contents, size))
2503    }

2505    /// Tries the peers that had the checkpoint when the download started, one at a time.
2506    async fn one_peer_at_a_time(&self) -> Option<VersionedFullCheckpointContents> {
2507        let mut kind = ContentRequestKind::First;
2508        for peer in self.candidates() {
2509            if kind == ContentRequestKind::Fallback {
2510                self.metrics.inc_checkpoint_content_fallback_requests();
2511            }
2512            let response = self.request(peer, kind).await;
2513            if let Some((contents, _size)) = self.record_response(response) {
2514                return Some(contents);
2515            }
2516            kind = ContentRequestKind::Fallback;
2517        }
2518        None
2519    }

2521    /// Sends a hedge request to another peer when the latest request has run for its peer's
2522    /// hedge delay (see [`Self::hedge_delay_for`]) without a valid response, with at most
2523    /// [`MAX_CONCURRENT_CONTENT_REQUESTS`] requests in flight. If no other peer has the
2524    /// checkpoint at that point, checks again every [`CONTENT_CANDIDATE_POLL_INTERVAL`]. When
2525    /// every request has failed, asks another peer right away. Peers are picked from the peers
2526    /// that have announced the checkpoint at the time each request is sent, not when the download
2527    /// started. The first valid response wins, and the other request is cancelled.
2528    ///
2529    /// A download near the release frontier when it starts sends its first request, hedges after
2530    /// the hedge delay, and requests after every request failed to the peer with the lowest
2531    /// [`expected_response_time`], preferring peers that are not slow from this node (see
2532    /// [`Self::closest_unasked_peer`]). From its first request until it ends, it checks every
2533    /// [`CONTENT_CANDIDATE_POLL_INTERVAL`] which peers have the checkpoint (see
2534    /// [`PeerAnnouncements`]). At each check while only one request is in flight, it hedges right
2535    /// away to a peer that hasn't been asked yet if the request in flight is to a slow peer and is
2536    /// not a slow-peer probe (see [`SLOW_PEER_PROBE_INTERVAL`]) and a peer that isn't slow has the
2537    /// checkpoint, or if a peer is expected to finish at least [`CLOSER_PEER_MIN_GAIN`] before the
2538    /// request in flight (see [`Self::early_hedge`]).
2539    async fn hedged(&self, hedge: &ContentHedge) -> Option<VersionedFullCheckpointContents> {
2540        let sequence_number = *self.checkpoint.sequence_number();
2541        let near_frontier = hedge.covers(sequence_number);
2542        let wait_interval = self
2543            .peer_heights
2544            .read()
2545            .unwrap()
2546            .wait_interval_when_no_peer_to_sync_content();
2547        let download_start = tokio::time::Instant::now();
2548        let mut asked = HashSet::new();
2549        let mut in_flight = FuturesUnordered::new();
2550        // Each request still in flight, by peer.
2551        let mut in_flight_requests: HashMap<PeerId, InFlightContentRequest> = HashMap::new();
2552        let mut no_hedge_candidate_reported = false;
2553        let mut announcements = PeerAnnouncements::default();
2554        let next_hedge = tokio::time::sleep(hedge.delay);
2555        tokio::pin!(next_hedge);
2556        let next_other_peer_check = tokio::time::sleep(CONTENT_CANDIDATE_POLL_INTERVAL);
2557        tokio::pin!(next_other_peer_check);
2558
2559        loop {
2560            // The next request to send: a first or fallback request when nothing is in flight,
2561            // otherwise a hedge.
2562            let (peer, kind, slow_peer) = if in_flight.is_empty() {
2563                // Either nothing was sent yet or every request failed. Ask the next peer that has
2564                // the checkpoint now instead of waiting for the retry interval.
2565                if let Some(peer) = self.candidate(&asked, near_frontier) {
2566                    let kind = if asked.is_empty() {
2567                        ContentRequestKind::First
2568                    } else {
2569                        self.metrics.inc_checkpoint_content_fallback_requests();
2570                        ContentRequestKind::Fallback
2571                    };
2572                    let slow_peer = if near_frontier {
2573                        self.classify_new_request(peer.inner().peer_id())
2574                    } else {
2575                        SlowPeerRequest::PeerNotSlow
2576                    };
2577                    (peer, kind, slow_peer)
2578                } else if hedge.covers(sequence_number) && download_start.elapsed() < wait_interval
2579                {
2580                    // No peer that hasn't been asked has the checkpoint yet. Near the release
2581                    // frontier, check again after a short poll interval, or when the retry
2582                    // interval ends if that is sooner.
2583                    let next_check = tokio::time::Instant::now() + CONTENT_CANDIDATE_POLL_INTERVAL;
2584                    tokio::time::sleep_until(next_check.min(download_start + wait_interval)).await;
2585                    continue;
2586                } else {
2587                    return None;
2588                }
2589            } else {
2590                tokio::select! {
2591                    biased;
2592                    Some(response) = in_flight.next() => {
2593                        // Inference only learns the request type from the `push` below.
2594                        let response: ContentResponse = response;
2595                        // Every request in flight is in `in_flight_requests`: each peer is asked
2596                        // at most once per download, and a request is tracked from the moment it
2597                        // is sent.
2598                        let finished = in_flight_requests
2599                            .remove(&response.peer_id)
2600                            .expect("every request in flight is tracked");
2601                        let (peer_id, elapsed) = (response.peer_id, response.elapsed);
2602                        let Some((contents, size)) = self.record_response(response) else {
2603                            // The request still running, if any, is now the latest one: hedge it
2604                            // once it has run for its own hedge delay, which may already have
2605                            // passed.
2606                            if let Some((&running_peer_id, running)) =
2607                                in_flight_requests.iter().next()
2608                            {
2609                                next_hedge.as_mut().reset(tokio::time::Instant::from_std(
2610                                    running.start + self.hedge_delay_for(hedge, running_peer_id),
2611                                ));
2612                            }
2613                            continue;
2614                        };
2615                        if finished.kind == ContentRequestKind::Hedge {
2616                            self.metrics.inc_checkpoint_content_hedge(HedgeEvent::Won);
2617                        } else if finished.slow_peer.records_response_time() {
2618                            self.peer_heights.write().unwrap().record_content_response_time(
2619                                peer_id,
2620                                ContentResponseTime::Answered(elapsed),
2621                            );
2622                        }
2623                        // Dropping `in_flight` on return cancels the requests that lost.
2624                        for (peer_id, request) in in_flight_requests {
2625                            let cancelled_after = request.start.elapsed();
2626                            self.metrics.record_checkpoint_content_peer_request(
2627                                &content_metrics_peer_label(self.network, peer_id),
2628                                ContentRequestOutcome::Cancelled,
2629                                cancelled_after,
2630                            );
2631                            let mut peer_heights = self.peer_heights.write().unwrap();
2632                            peer_heights.record_cancelled_request(peer_id, size, cancelled_after);
2633                            if request.kind != ContentRequestKind::Hedge
2634                                && request.slow_peer.records_response_time()
2635                            {
2636                                peer_heights.record_content_response_time(
2637                                    peer_id,
2638                                    ContentResponseTime::LongerThan(cancelled_after),
2639                                );
2640                            }
2641                        }
2642                        return Some(contents);
2643                    }
2644                    () = next_hedge.as_mut(), if in_flight.len() < MAX_CONCURRENT_CONTENT_REQUESTS => {
2645                        if !hedge.covers(sequence_number) {
2646                            next_hedge
2647                                .as_mut()
2648                                .reset(tokio::time::Instant::now() + hedge.delay);
2649                            continue;
2650                        }
2651                        let Some(peer) = self.candidate(&asked, near_frontier) else {
2652                            if !no_hedge_candidate_reported {
2653                                no_hedge_candidate_reported = true;
2654                                self.metrics
2655                                    .inc_checkpoint_content_hedge(HedgeEvent::NoCandidate);
2656                            }
2657                            next_hedge
2658                                .as_mut()
2659                                .reset(tokio::time::Instant::now() + CONTENT_CANDIDATE_POLL_INTERVAL);
2660                            continue;
2661                        };
2662                        self.metrics.inc_checkpoint_content_hedge(HedgeEvent::Sent);
2663                        (peer, ContentRequestKind::Hedge, SlowPeerRequest::PeerNotSlow)
2664                    }
2665                    // Requests are in flight, so the first request has gone out.
2666                    () = next_other_peer_check.as_mut(), if near_frontier => {
2667                        next_other_peer_check
2668                            .as_mut()
2669                            .reset(tokio::time::Instant::now() + CONTENT_CANDIDATE_POLL_INTERVAL);
2670                        let peers = self.peers_with_checkpoint();
2671                        announcements.observe(&peers, self.network, self.metrics);
2672                        if in_flight.len() >= MAX_CONCURRENT_CONTENT_REQUESTS {
2673                            continue;
2674                        }
2675                        let Some((peer, event)) = in_flight_requests
2676                            .iter()
2677                            .next()
2678                            .and_then(|(&in_flight_peer_id, in_flight_request)| {
2679                                self.early_hedge(&peers, &asked, in_flight_peer_id, in_flight_request)
2680                            })
2681                        else {
2682                            continue;
2683                        };
2684                        self.metrics.inc_checkpoint_content_hedge(HedgeEvent::Sent);
2685                        self.metrics.inc_checkpoint_content_hedge(event);
2686                        (peer, ContentRequestKind::Hedge, SlowPeerRequest::PeerNotSlow)
2687                    }
2688                }
2689            };
2690
2691            let peer_id = peer.inner().peer_id();
2692            asked.insert(peer_id);
2693            in_flight_requests.insert(
2694                peer_id,
2695                InFlightContentRequest {
2696                    start: Instant::now(),
2697                    kind,
2698                    slow_peer,
2699                },
2700            );
2701            in_flight.push(self.request(peer, kind));
2702            next_hedge
2703                .as_mut()
2704                .reset(tokio::time::Instant::now() + self.new_request_hedge_delay(hedge, peer_id));
2705            if kind == ContentRequestKind::First && near_frontier {
2706                announcements.first_request_sent();
2707                next_other_peer_check
2708                    .as_mut()
2709                    .reset(tokio::time::Instant::now() + CONTENT_CANDIDATE_POLL_INTERVAL);
2710                announcements.observe(&self.peers_with_checkpoint(), self.network, self.metrics);
2711            }
2712        }
2713    }
2714}

2716/// For a download near the release frontier, tracks from its first request which peers have the
2717/// checkpoint, and records for each peer how long after the first request a check first saw it
2718/// having the checkpoint (`state_sync_checkpoint_content_peer_announce_delay`).
2719#[derive(Default)]
2720struct PeerAnnouncements {
2721    /// When the first request went out, once it has.
2722    first_request_at: Option<Instant>,
2723    /// Peers seen having the checkpoint since the first request.
2724    seen: HashSet<PeerId>,
2725}
2726
2727impl PeerAnnouncements {
2728    fn first_request_sent(&mut self) {
2729        self.first_request_at = Some(Instant::now());
2730    }
2731
2732    /// Records the announce delay of each peer in `peers`, the peers that have the checkpoint
2733    /// now, that wasn't seen having it before.
2734    fn observe(&mut self, peers: &[anemo::Peer], network: &anemo::Network, metrics: &Metrics) {
2735        let Some(first_request_at) = self.first_request_at else {
2736            return;
2737        };
2738        let delay = first_request_at.elapsed();
2739        for peer in peers {
2740            let peer_id = peer.peer_id();
2741            if self.seen.insert(peer_id) {
2742                metrics.record_checkpoint_content_peer_announce_delay(
2743                    &content_metrics_peer_label(network, peer_id),
2744                    delay,
2745                );
2746            }
2747        }
2748    }
2749}

2751/// How long a contents request to `peer` is expected to take: the longer of the connection's
2752/// round-trip time and the peer's fastest recent answer (see
2753/// `PeerHeights::fastest_recent_content_answer`), or the round-trip time alone if the peer has
2754/// no recent answers. A request needs at least one round trip, and a peer that answered in some
2755/// time recently is not expected to answer faster.
2756fn expected_response_time(peer: &anemo::Peer, peer_heights: &PeerHeights) -> Duration {
2757    let rtt = peer.connection_rtt();
2758    peer_heights
2759        .fastest_recent_content_answer(&peer.peer_id())
2760        .map_or(rtt, |fastest| fastest.max(rtt))
2761}
2762
2763/// Whether a peer with expected response time `candidate_expected`, asked at `now`, is expected to
2764/// finish at least [`CLOSER_PEER_MIN_GAIN`] before the request in flight, which was sent at
2765/// `in_flight_start` to a peer with expected response time `in_flight_expected`. A request in
2766/// flight that has run past its expected finish never qualifies.
2767fn worth_hedging_early(
2768    now: Instant,
2769    in_flight_start: Instant,
2770    in_flight_expected: Duration,
2771    candidate_expected: Duration,
2772) -> bool {
2773    now + candidate_expected + CLOSER_PEER_MIN_GAIN <= in_flight_start + in_flight_expected
2774}

2776/// Label for per-peer content request metrics. Only peers in the network's known-peer set get
2777/// their own label, which bounds the label cardinality.
2778fn content_metrics_peer_label(network: &anemo::Network, peer_id: PeerId) -> String {
2779    if network.known_peers().get(&peer_id).is_some() {
2780        peer_id.to_string()
2781    } else {
2782        "other".to_owned()
2783    }
2784}

2786#[instrument(level = "debug", skip_all)]
2787async fn get_full_checkpoint_contents<S>(
2788    fetch: ContentFetch<'_>,
2789    store: S,
2790    hedge: Option<&ContentHedge>,
2791) -> Option<VersionedFullCheckpointContents>
2792where
2793    S: WriteStore,
2794{
2795    let checkpoint = fetch.checkpoint;
2796    let sequence_number = checkpoint.sequence_number;
2797    let digest = checkpoint.content_digest;
2798    if let Some(contents) = store.get_full_checkpoint_contents(Some(sequence_number), &digest) {
2799        debug!("store already contains checkpoint contents");
2800        return Some(contents);
2801    }
2802
2803    let start = Instant::now();
2804    let contents = match hedge {
2805        None => fetch.one_peer_at_a_time().await,
2806        Some(hedge) => fetch.hedged(hedge).await,
2807    };
2808    let Some(contents) = contents else {
2809        debug!("no peers had checkpoint contents");
2810        return None;
2811    };
2812    fetch
2813        .metrics
2814        .record_checkpoint_content_download(start.elapsed());
2815    let verified_contents = VerifiedCheckpointContents::new_unchecked(contents.clone());
2816    store
2817        .insert_checkpoint_contents(checkpoint, verified_contents)
2818        .expect("store operation should not fail");
2819    Some(contents)
2820}
