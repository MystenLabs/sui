28/// Sui-Rosetta specific error types.
29/// This contains all the errors returns by the sui-rosetta server.
30#[derive(Debug, Error, EnumDiscriminants, EnumProperty)]
31#[strum_discriminants(
32    name(ErrorType),
33    derive(Display, EnumIter),
34    strum(serialize_all = "kebab-case")
35)]
36#[allow(clippy::enum_variant_names)]
37pub enum Error {
38    #[error("Unsupported blockchain: {0}")]
39    UnsupportedBlockchain(String),
40    #[error("Unsupported network: {0:?}")]
41    UnsupportedNetwork(SuiEnv),
42    #[error("Invalid input: {0}")]
43    InvalidInput(String),
44    #[error("Missing input: {0}")]
45    MissingInput(String),
46    #[error("Missing metadata")]
47    MissingMetadata,
48    #[error("{0}")]
49    MalformedOperationError(String),
50    #[error("Unsupported operation: {0:?}")]
51    UnsupportedOperation(OperationType),
52    #[error("Data error: {0}")]
53    DataError(String),
54    #[error("Block not found, index: {index:?}, hash: {hash:?}")]
55    BlockNotFound {
56        index: Option<u64>,
57        hash: Option<BlockHash>,
58    },
59    #[error("Public key deserialization error: {0:?}")]
60    PublicKeyDeserializationError(PublicKey),
61
62    #[error("Error executing transaction: {0}")]
63    TransactionExecutionError(Box<ExecutionError>),
64
65    #[error("{0}")]
66    TransactionDryRunError(Box<ExecutionError>),
67
68    #[error(transparent)]
69    InternalError(#[from] anyhow::Error),
70    #[error(transparent)]
71    BCSSerializationError(#[from] bcs::Error),
72    #[error(transparent)]
73    CryptoError(#[from] FastCryptoError),
74    #[error(transparent)]
75    SuiError(#[from] SuiError),
76    #[error(transparent)]
77    SuiRpcError(#[from] tonic::Status),
78    #[error(transparent)]
79    EncodingError(#[from] eyre::Report),
80    #[error(transparent)]
81    JsonExtractorRejection(#[from] JsonRejection),
82
83    #[error("Retries exhausted while getting balance. try again.")]
84    #[strum(props(retriable = "true"))]
85    RetryExhausted(String),
86
87    // Appended last so existing `ErrorType` discriminant positions — and thus the
88    // Rosetta wire `code` integers — stay stable.
89    #[error("Coin metadata temporarily unavailable: {0}")]
90    #[strum(props(retriable = "true"))]
91    CoinMetadataUnavailable(String),
92}

100/// The `google.rpc.Status` the fullnode encodes into the details of an `InvalidArgument` status,
101/// together with its `ErrorInfo` detail of the given `reason` in domain `ErrorReason::DOMAIN`.
102/// `None` for any other code, for details that do not decode, or when no such `ErrorInfo` exists.
103fn invalid_argument_error_info(
104    status: &tonic::Status,
105    reason: ErrorReason,
106) -> Option<(RpcStatus, ErrorInfo)> {
107    if status.code() != tonic::Code::InvalidArgument {
108        return None;
109    }
110    let rpc_status = RpcStatus::decode(status.details()).ok()?;
111    let info = rpc_status
112        .details
113        .iter()
114        .filter_map(|detail| detail.to_msg::<ErrorInfo>().ok())
115        .find(|info| info.reason == reason.as_str_name() && info.domain == ErrorReason::DOMAIN)?;
116    Some((rpc_status, info))
117}
118
119/// The `ErrorInfo` detail (reason `INSUFFICIENT_GAS_BALANCE`, domain `ErrorReason::DOMAIN`) the
120/// fullnode attaches to an `InvalidArgument` status when gas selection fails because the gas owner
121/// lacks enough SUI for the budget. `None` for any other error, including statuses whose details do
122/// not decode.
123pub(crate) fn insufficient_gas_balance(error: &Error) -> Option<ErrorInfo> {
124    let Error::SuiRpcError(status) = error else {
125        return None;
126    };
127    invalid_argument_error_info(status, ErrorReason::InsufficientGasBalance).map(|(_, info)| info)
128}
129
130/// The `requiredBudget` metadata entry, in MIST, of the `ErrorInfo` that
131/// [`insufficient_gas_balance`] returns. `None` for any other error, or when the entry is missing
132/// or does not parse.
133pub(crate) fn insufficient_gas_required_budget(error: &Error) -> Option<u64> {
134    insufficient_gas_balance(error)?
135        .metadata
136        .get("requiredBudget")?
137        .parse()
138        .ok()
139}

141/// The `ExecutionError` detail the fullnode attaches, next to an `ErrorInfo` with reason
142/// `TRANSACTION_EXECUTION_FAILED` in domain `ErrorReason::DOMAIN`, to an `InvalidArgument` status
143/// when a simulated transaction fails in execution and gas selection cannot cover its budget. It is
144/// the same error a simulation that can pay for gas reports in its effects. `None` for any other
145/// status, including statuses whose details do not decode.
146pub(crate) fn execution_failure(status: &tonic::Status) -> Option<ExecutionError> {
147    let (rpc_status, _) =
148        invalid_argument_error_info(status, ErrorReason::TransactionExecutionFailed)?;
149    rpc_status
150        .details
151        .iter()
152        .find_map(|detail| detail.to_msg::<ExecutionError>().ok())
153}

174impl ErrorType {
175    fn json(&self) -> Value {
176        let retriable = false;
177        // Safe to unwrap
178        let error_code = ErrorType::iter().position(|e| &e == self).unwrap();
179        let message = format!("{self}").replace('-', " ");
180        let message = message[0..1].to_uppercase() + &message[1..];
181
182        json![{
183            "code": error_code,
184            "message": message,
185            "retriable": retriable,
186        }]
187    }
188}
189
190impl Serialize for Error {
191    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
192    where
193        S: Serializer,
194    {
195        let type_: ErrorType = self.into();
196        let mut json = type_.json();
197        // Safe to unwrap, we know ErrorType must be an object.
198        let error = json.as_object_mut().unwrap();
199        let mut details = serde_json::Map::new();
200        details.insert("error".into(), json!(format!("{self}")));
201        // When gas selection fails for lack of SUI, include the amounts the fullnode reported so
202        // callers can prefund the account without parsing the error message.
203        if let Some(info) = insufficient_gas_balance(self) {
204            for (metadata_key, details_key) in [
205                ("requiredBudget", "required_budget"),
206                ("coinBalance", "coin_balance"),
207                ("addressBalance", "address_balance"),
208            ] {
209                if let Some(value) = info.metadata.get(metadata_key) {
210                    details.insert(details_key.into(), json!(value));
211                }
212            }
213        }
214        error.insert("details".into(), Value::Object(details));
215        error.insert("retriable".into(), json!(self.retriable()));
216        error.serialize(serializer)
217    }
218}

220impl IntoResponse for Error {
221    fn into_response(self) -> Response {
222        (StatusCode::INTERNAL_SERVER_ERROR, Json(self)).into_response()
223    }
