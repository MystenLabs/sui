38#[async_trait]
39impl TryConstructTransaction for PayCoin {
40    async fn try_fetch_needed_objects(
41        self,
42        client: &mut Client,
43        gas_price: Option<u64>,
44        budget: Option<u64>,
45    ) -> Result<TransactionObjectData, Error> {
46        let Self {
47            sender,
48            recipients,
49            amounts,
50            currency,
51        } = self;
52
53        let sdk_coin_type = SdkTypeTag::from_str(&currency.metadata.coin_type)
54            .map_err(|e| Error::DataError(format!("Invalid coin type: {}", e)))?;
55
56        let total_payment: u64 = amounts.iter().sum();
57
58        // Query address balance for the payment coin type
59        let address_balance = {
60            let request = GetBalanceRequest::default()
61                .with_owner(sender.to_string())
62                .with_coin_type(currency.metadata.coin_type.clone());
63            client
64                .state_client()
65                .get_balance(request)
66                .await?
67                .into_inner()
68                .balance()
69                .address_balance()
70        };
71
72        // Select all coin objects (up to 1500). Storage refunds from merging dust outweigh
73        // smashing costs, so we merge as many as possible.
74        let all_coins = client
75            .select_up_to_n_largest_coins(&Address::from(sender), &sdk_coin_type, 1500, &[])
76            .await?;
77
78        let coins_total: u64 = all_coins.iter().map(|c| c.balance()).sum();
79
80        // Separate party objects (ConsensusAddressOwner) from regular objects.
81        let (party_objects, non_party_objects): (Vec<_>, Vec<_>) = all_coins
82            .iter()
83            .partition(|obj| obj.owner().kind() == OwnerKind::ConsensusAddress);
84
85        let coins: Vec<ObjectRef> = non_party_objects
86            .iter()
87            .map(|obj: &&Object| obj.object_reference().try_to_object_ref())
88            .collect::<Result<Vec<_>, _>>()?;
89
90        let party_coins: Vec<(ObjectID, SequenceNumber)> = party_objects
91            .iter()
92            .map(|obj: &&Object| -> Result<_, Error> {
93                let id = ObjectID::from_str(obj.object_id())
94                    .map_err(|e| Error::DataError(format!("Invalid party object ID: {}", e)))?;
95                let start_version = SequenceNumber::from_u64(obj.owner().version());
96                Ok((id, start_version))
97            })
98            .collect::<Result<Vec<_>, _>>()?;
99
100        // Compute deficit: how much we need from address balance beyond what coins provide
101        let deficit = total_payment.saturating_sub(coins_total);
102        if deficit > address_balance {
103            return Err(Error::InvalidInput(format!(
104                "Insufficient funds: need {} but only have {} in coins + {} in address balance",
105                total_payment, coins_total, address_balance
106            )));
107        }
108
109        // Free-tier ("gasless") attempt. Build an AB-deposit PTB from the same funds the priced
110        // path would smash — owned coins, party coins, and an `deficit` top-up from the sender's
111        // address balance — but deliver via `coin::send_funds` (which credits each recipient's
112        // address balance) with any change deposited into the sender's own address balance. The
113        // node's SimulateTransaction auto-switches to gasless only for actually-whitelisted coins
114        // meeting the per-token minimum and gasless limits, so a simulate with the price left unset
115        // is a definitive eligibility oracle. On confirmation (resolved price 0 + no gas objects)
116        // we signal gasless downstream via `budget == 0`; otherwise we fall through to the priced
117        // smash path below, which delivers coin objects.
118        //
119        // We attempt this for *every* payment (not just AB-covered ones): a free-tier coin held as
120        // coin objects can be spent gaslessly by consuming the objects, so gating on AB coverage
121        // would needlessly charge gas. The speculative simulate inherits the protocol's eligibility
122        // rules for free — allowlist membership, per-token minimum, and the requirement that any
123        // change deposited back to the sender is `0` or `>= min` (a sub-min crumb makes the node
124        // decline gasless, and we fall back to smash).
125        let gasless_pt = pay_coin_gasless_pt(
126            sender,
127            recipients.clone(),
128            amounts.clone(),
129            &coins,
130            &party_coins,
131            deficit,
132            &currency,
133        )?;
134        match simulate_transaction(client, gasless_pt, sender, vec![], None, None).await {
135            Ok((0, gas_coin_objs)) if gas_coin_objs.is_empty() => {
136                return Ok(TransactionObjectData {
137                    gas_coins: vec![],
138                    objects: coins,
139                    party_objects: party_coins,
140                    total_sui_balance: 0,
141                    budget: 0,
142                    address_balance_withdrawal: deficit,
143                    fss_object_count: None,
144                    redeem_token_amount: None,
145                    redeem_plan: None,
146                    bind_epoch: None,
147                });
148            }
149            // Fall through to the smash path when gasless is ineligible (priced simulate), the
150            // probe's dry run fails, or the probe fails gas selection because the account lacks
151            // the SUI to pay for it (the priced path then reports the budget of the transaction
152            // actually sent). Other errors, such as transport failures, propagate.
153            Ok(_) | Err(Error::TransactionDryRunError(_)) => {}
154            Err(e) if insufficient_gas_balance(&e).is_some() => {}
155            Err(e) => return Err(e),
156        }
157
158        // Merge coins directly, optionally withdraw deficit from address balance,
159        // split payments and transfer coins to recipients.
160        // No GasCoin reference → simulator auto-selects SUI gas.
161        let pt = pay_coin_pt(
162            sender,
163            recipients,
164            amounts,
165            &coins,
166            &party_coins,
167            deficit,
168            &currency,
169        )?;
170        let (budget, gas_coin_objs) =
171            simulate_transaction(client, pt, sender, vec![], gas_price, budget).await?;
172
173        if gas_coin_objs.is_empty() {
174            Ok(TransactionObjectData {
175                gas_coins: vec![],
176                objects: coins,
177                party_objects: party_coins,
178                total_sui_balance: budget as i128,
179                budget,
180                address_balance_withdrawal: deficit,
181                fss_object_count: None,
182                redeem_token_amount: None,
183                redeem_plan: None,
184                bind_epoch: None,
185            })
186        } else {
187            let total_sui_balance = gas_coin_objs.iter().map(|c| c.balance()).sum::<u64>() as i128;
188            let gas_coins = gas_coin_objs
189                .iter()
190                .map(|obj: &Object| obj.object_reference().try_to_object_ref())
191                .collect::<Result<Vec<_>, _>>()?;
192
193            Ok(TransactionObjectData {
194                gas_coins,
195                objects: coins,
196                party_objects: party_coins,
197                total_sui_balance,
198                budget,
199                address_balance_withdrawal: deficit,
200                fss_object_count: None,
201                redeem_token_amount: None,
202                redeem_plan: None,
203                bind_epoch: None,
204            })
205        }
206    }
207}

209/// Merge coin objects, optionally withdraw deficit from address balance,
210/// split payments and transfer coins to each recipient.
211/// Remainder stays as a coin owned by the sender.
212/// No GasCoin reference → simulator auto-selects SUI gas.
213pub fn pay_coin_pt(
214    sender: SuiAddress,
215    recipients: Vec<SuiAddress>,
216    amounts: Vec<u64>,
217    coins: &[ObjectRef],
218    party_coins: &[(ObjectID, SequenceNumber)],
219    address_balance_withdrawal: u64,
220    currency: &Currency,
221) -> anyhow::Result<ProgrammableTransaction> {
222    let sdk_type = SdkTypeTag::from_str(&currency.metadata.coin_type)?;
223    let core_type = type_tag_sdk_to_core(sdk_type)?;
224
225    let mut builder = ProgrammableTransactionBuilder::new();
226
227    let (source, source_is_command_result) = merge_payment_source(
228        &mut builder,
229        coins,
230        party_coins,
231        address_balance_withdrawal,
232        &core_type,
233    )?;
234
235    // Split payment amounts and transfer coins to recipients
236    let amount_args: Vec<Argument> = amounts
237        .iter()
238        .map(|&v| builder.pure(v))
239        .collect::<Result<Vec<_>, _>>()?;
240    let split_result = builder.command(Command::SplitCoins(source, amount_args));
241    let Argument::Result(split_idx) = split_result else {
242        anyhow::bail!("Expected Result argument from SplitCoins");
243    };
244
245    for (i, recipient) in recipients.into_iter().enumerate() {
246        builder.transfer_arg(recipient, Argument::NestedResult(split_idx, i as u16));
247    }
248
249    // Command results (from AB withdrawal) aren't implicitly returned like input objects.
250    // Transfer the remainder back to the sender (who is always the negative-amount account).
251    if source_is_command_result {
252        builder.transfer_arg(sender, source);
253    }
254
255    Ok(builder.finish())
256}

258/// Build a gasless ("free tier") PTB paying `amounts` to `recipients`, crediting each recipient's
259/// address balance. Funds come from the same sources the priced path smashes — owned coins, party
260/// coins, and an optional `address_balance_withdrawal` top-up — but delivery uses
261/// `coin::send_funds` (deposit to the recipient's address balance) instead of `TransferObjects`,
262/// and the change is deposited into the *sender's* address balance rather than returned as a coin.
263///
264/// Every input coin is consumed (deleted), never mutated or returned, as the post-execution gasless
265/// check requires. Uses only gasless-permitted commands — notably no `TransferObjects`. If the
266/// change deposited back to the sender lands in `(0, min)` for the token, the node declines gasless
267/// and the caller falls back to the priced smash path.
268pub fn pay_coin_gasless_pt(
269    sender: SuiAddress,
270    recipients: Vec<SuiAddress>,
271    amounts: Vec<u64>,
272    coins: &[ObjectRef],
273    party_coins: &[(ObjectID, SequenceNumber)],
274    address_balance_withdrawal: u64,
275    currency: &Currency,
276) -> anyhow::Result<ProgrammableTransaction> {
277    let sdk_type = SdkTypeTag::from_str(&currency.metadata.coin_type)?;
278    let core_type = type_tag_sdk_to_core(sdk_type)?;
279
280    let mut builder = ProgrammableTransactionBuilder::new();
281
282    let (source, _source_is_command_result) = merge_payment_source(
283        &mut builder,
284        coins,
285        party_coins,
286        address_balance_withdrawal,
287        &core_type,
288    )?;
289
290    // Split every recipient amount off the source so `/parse` tracks each piece with its own value.
291    let amount_args: Vec<Argument> = amounts
292        .iter()
293        .map(|&v| builder.pure(v))
294        .collect::<Result<Vec<_>, _>>()?;
295    let split_result = builder.command(Command::SplitCoins(source, amount_args));
296    let Argument::Result(split_idx) = split_result else {
297        anyhow::bail!("Expected Result argument from SplitCoins");
298    };
299
300    // Deposit each split piece into the recipient's address balance.
301    for (i, recipient) in recipients.into_iter().enumerate() {
302        coin_send_funds(
303            &mut builder,
304            &core_type,
305            Argument::NestedResult(split_idx, i as u16),
306            recipient,
307        )?;
308    }
309
310    // Consume the source by depositing the remainder (the change) into the *sender's* address
311    // balance. This is mandatory for gasless: every input coin must be deleted, not mutated or
312    // implicitly returned. `/parse` ignores it — an input-object source isn't tracked as a transfer
313    // and a command-result source resolves to a net-zero self-payment — so it never pollutes parsed
314    // operations.
315    coin_send_funds(&mut builder, &core_type, source, sender)?;
316
317    Ok(builder.finish())
318}

320/// Merge owned coins, party (ConsensusAddress) coins, and an optional address-balance withdrawal
321/// into a single `Coin<T>` source. Returns the source argument and whether it is a command result
322/// (an AB withdrawal with no coin objects) rather than an input object — the priced path uses that
323/// to decide whether the remainder needs an explicit transfer back to the sender.
324fn merge_payment_source(
325    builder: &mut ProgrammableTransactionBuilder,
326    coins: &[ObjectRef],
327    party_coins: &[(ObjectID, SequenceNumber)],
328    address_balance_withdrawal: u64,
329    core_type: &TypeTag,
330) -> anyhow::Result<(Argument, bool)> {
331    // Merge all coin objects into a single target (input object).
332    let mut source: Option<Argument> = None;
333    // Track whether the source is a command result (vs input object). Command results need explicit
334    // transfer after SplitCoins; input objects are implicitly returned.
335    let mut source_is_command_result = false;
336
337    if !coins.is_empty() || !party_coins.is_empty() {
338        let target = if let Some(&first) = coins.first() {
339            builder.obj(ObjectArg::ImmOrOwnedObject(first))?
340        } else {
341            let &(id, initial_shared_version) = &party_coins[0];
342            builder.obj(ObjectArg::SharedObject {
343                id,
344                initial_shared_version,
345                mutability: sui_types::transaction::SharedObjectMutability::Mutable,
346            })?
347        };
348
349        if coins.len() > 1 {
350            coins[1..]
351                .chunks(MAX_COMMAND_ARGS)
352                .try_for_each(|chunk| -> anyhow::Result<()> {
353                    let to_merge = chunk
354                        .iter()
355                        .map(|&o| builder.obj(ObjectArg::ImmOrOwnedObject(o)))
356                        .collect::<Result<Vec<Argument>, _>>()?;
357                    builder.command(Command::MergeCoins(target, to_merge));
358                    Ok(())
359                })?;
360        }
361
362        let party_skip = if coins.is_empty() { 1 } else { 0 };
363        let party_slice = &party_coins[party_skip..];
364        if !party_slice.is_empty() {
365            party_slice
366                .chunks(MAX_COMMAND_ARGS)
367                .try_for_each(|chunk| -> anyhow::Result<()> {
368                    let to_merge = chunk
369                        .iter()
370                        .map(|&(id, initial_shared_version)| {
371                            builder.obj(ObjectArg::SharedObject {
372                                id,
373                                initial_shared_version,
374                                mutability: sui_types::transaction::SharedObjectMutability::Mutable,
375                            })
376                        })
377                        .collect::<Result<Vec<Argument>, _>>()?;
378                    builder.command(Command::MergeCoins(target, to_merge));
379                    Ok(())
380                })?;
381        }
382
383        source = Some(target);
384    }
385
386    // Withdraw deficit from address balance and merge into source.
387    if address_balance_withdrawal > 0 {
388        let withdrawal_coin = withdraw_coin_from_address_balance(
389            builder,
390            address_balance_withdrawal,
391            core_type.clone(),
392        )?;
393        match source {
394            Some(target) => {
395                // Merge command result into input object — input is implicitly returned.
396                builder.command(Command::MergeCoins(target, vec![withdrawal_coin]));
397            }
398            None => {
399                // No coin objects; source is entirely from AB (a command result).
400                source = Some(withdrawal_coin);
401                source_is_command_result = true;
402            }
403        }
404    }
405
406    let source =
407        source.ok_or_else(|| anyhow::anyhow!("No coins or address balance to pay from"))?;
408    Ok((source, source_is_command_result))
409}

411/// Deposit a `Coin<T>` argument into `recipient`'s address balance via `coin::send_funds<T>`
412/// (consuming the coin).
413fn coin_send_funds(
414    builder: &mut ProgrammableTransactionBuilder,
415    core_type: &TypeTag,
416    coin: Argument,
417    recipient: SuiAddress,
418) -> anyhow::Result<()> {
419    let recipient_arg = builder.pure(recipient)?;
420    builder.command(Command::move_call(
421        SUI_FRAMEWORK_PACKAGE_ID,
422        Identifier::new("coin")?,
423        Identifier::new("send_funds")?,
424        vec![core_type.clone()],
425        vec![coin, recipient_arg],
426    ));
427    Ok(())
