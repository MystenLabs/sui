40/// The fullnode reports a chunk's scan progress only when the chunk ends at its
41/// scan budget, and then reports the `ScanStop::ScanLimit` frontier; every
42/// other evaluator watermark is dropped (see
43/// [`DrainedBitmapHits::scan_limit_frontier`]). So it asks the evaluator for as
44/// few watermarks as possible.
45const FULLNODE_WATERMARK_PACING: WatermarkPacing = WatermarkPacing {
46    min_buckets: u64::MAX,
47    emit_held_at_end: false,
48};

56pub(super) struct DrainedBitmapHits {
57    pub(super) items: Vec<u64>,
58    pub(super) pending_bucket: Option<PendingBitmapBucket>,
59    pub(super) next_range: Option<Range<u64>>,
60    pub(super) buckets_scanned: usize,
61    /// `Some(frontier)` exactly when the evaluator exhausted the scan budget
62    /// supplied for this chunk (`ScanStop::ScanLimit`), carrying that
63    /// terminal's frontier in absolute member-id space (the slowest source's
64    /// frontier). Ascending frontiers are the first unscanned position
65    /// (inclusive); descending frontiers are an exclusive upper bound. The
66    /// caller decides whether the request budget is also exhausted or another
67    /// chunk should continue scanning, and resolves the frontier to a
68    /// checkpoint for the request's scan-limit terminal or, if the chunk
69    /// produced no items, for a scan watermark.
70    pub(super) scan_limit_frontier: Option<u64>,
71}

73/// Evaluated bitmap state carried between blocking chunks.
74///
75/// The inverted-index evaluator yields `(bucket_id, RoaringBitmap)`. Holding the
76/// bitmap plus an absolute remaining range lets us resume a dense bucket without
77/// flattening it into an unbounded `Vec<u64>` or rereading it from RocksDB.
78#[derive(Clone)]
79pub(super) struct PendingBitmapBucket {
80    bucket_id: u64,
81    bitmap: RoaringBitmap,
82    remaining: Range<u64>,
83}
84
85impl PendingBitmapBucket {
86    fn new(
87        bucket_id: u64,
88        bitmap: RoaringBitmap,
89        range: &Range<u64>,
90        bucket_size: u64,
91    ) -> Option<Self> {
92        let bucket_start = bucket_id * bucket_size;
93        let bucket_end = bucket_start + bucket_size;
94        let remaining = range.start.max(bucket_start)..range.end.min(bucket_end);
95        (!remaining.is_empty() && !bitmap.is_empty()).then_some(Self {
96            bucket_id,
97            bitmap,
98            remaining,
99        })
100    }
101
102    fn drain_into(
103        &mut self,
104        out: &mut Vec<u64>,
105        hit_limit: usize,
106        bucket_size: u64,
107        direction: ScanDirection,
108    ) {
109        while out.len() < hit_limit {
110            let Some(seq) = self.next_seq(bucket_size, direction) else {
111                break;
112            };
113            out.push(seq);
114            self.remaining =
115                remaining_range_after(self.remaining.clone(), seq, direction.is_ascending())
116                    .unwrap_or(0..0);
117        }
118    }
119
120    fn has_hits(&self, bucket_size: u64, direction: ScanDirection) -> bool {
121        self.next_seq(bucket_size, direction).is_some()
122    }
123
124    fn next_seq(&self, bucket_size: u64, direction: ScanDirection) -> Option<u64> {
125        if self.remaining.is_empty() {
126            return None;
127        }
128
129        let bucket_start = self.bucket_id * bucket_size;
130        let lo = (self.remaining.start - bucket_start) as u32;
131        let hi = (self.remaining.end - bucket_start) as u32;
132        let bit = if direction.is_ascending() {
133            self.bitmap.range(lo..hi).next()
134        } else {
135            self.bitmap.range(lo..hi).next_back()
136        }?;
137        Some(bucket_start + u64::from(bit))
138    }
139
140    #[cfg(test)]
141    fn remaining(&self) -> Range<u64> {
142        self.remaining.clone()
143    }
144}

170pub(super) fn drain_bitmap_hits_with_budget(
171    service: RpcService,
172    kind: LedgerBitmapKind,
173    bucket_size: u64,
174    query: BitmapQuery,
175    pending_bucket: Option<PendingBitmapBucket>,
176    range: Option<Range<u64>>,
177    direction: ScanDirection,
178    hit_limit: usize,
179    scan_budget: usize,
180    cancel: &CancellationToken,
181) -> Result<DrainedBitmapHits, RpcError> {
182    if hit_limit == 0 {
183        return Ok(DrainedBitmapHits {
184            items: Vec::new(),
185            pending_bucket,
186            next_range: range,
187            buckets_scanned: 0,
188            scan_limit_frontier: None,
189        });
190    }
191
192    // The iterator evaluator in `sui-inverted-index` is synchronous, so the
193    // borrowed RocksDB iterators stay on this blocking task for the whole chunk.
194    // That avoids a runtime dependency in this path while keeping BigTable on
195    // the async stream evaluator. The evaluator charges `scan_budget` and
196    // reports the charged bucket rows through `on_metrics` once the iterator
197    // finishes or is dropped, which happens inside `drain_watermarked_buckets`.
198    // The count stays 0 when the chunk only drains a pending bucket and never
199    // opens the iterator.
200    let buckets_evaluated = Rc::new(Cell::new(0u64));
201    let record_buckets_evaluated = buckets_evaluated.clone();
202    let source = RpcIndexesBitmapSource::new(&service, kind, bucket_size, cancel.clone());
203    let state = drain_watermarked_buckets(
204        move |scan_range| {
205            eval_bitmap_query_bucket_iter(
206                source,
207                query,
208                scan_range,
209                bucket_size,
210                direction,
211                scan_budget as u64,
212                SkipPolicy {
213                    drain_probe_rows: std::num::NonZeroU32::new(BITMAP_DRAIN_PROBE_ROWS),
214                },
215                FULLNODE_WATERMARK_PACING,
216                move |metrics| record_buckets_evaluated.set(metrics.buckets_evaluated),
217            )
218        },
219        pending_bucket,
220        range,
221        bucket_size,
222        direction,
223        hit_limit,
224        cancel,
225    )?;
226    Ok(DrainedBitmapHits {
227        items: state.items,
228        pending_bucket: state.pending_bucket,
229        next_range: state.next_range,
230        buckets_scanned: buckets_evaluated.get() as usize,
231        scan_limit_frontier: state.scan_limit_frontier,
232    })
233}

239/// Loop state from draining a watermarked-bucket iterator, independent of budget
240/// accounting (the caller adds `buckets_scanned` from the evaluator's
241/// [`sui_inverted_index::BitmapScanMetrics`]).
242struct DrainLoopState {
243    items: Vec<u64>,
244    pending_bucket: Option<PendingBitmapBucket>,
245    next_range: Option<Range<u64>>,
246    scan_limit_frontier: Option<u64>,
247}
248
249/// Drain matching member ids until `hit_limit` items, the iterator ends, or it
250/// signals [`ScanStop::ScanLimit`]. `open_iter` builds the iterator over a scan
251/// range and is invoked at most once, lazily — only when a fresh scan beyond
252/// `pending_bucket` is needed — so resuming a dense pending bucket never opens
253/// a RocksDB iterator. Ascending evaluator frontiers are the first unscanned
254/// position and remain inclusive; descending frontiers retain their existing
255/// exclusive upper-bound semantics. The caller decides whether a scan-limit
256/// stop is the per-request cap or a per-chunk cap.
257fn drain_watermarked_buckets<I>(
258    open_iter: impl FnOnce(Range<u64>) -> I,
259    mut pending_bucket: Option<PendingBitmapBucket>,
260    mut next_range: Option<Range<u64>>,
261    bucket_size: u64,
262    direction: ScanDirection,
263    hit_limit: usize,
264    cancel: &CancellationToken,
265) -> Result<DrainLoopState, RpcError>
266where
267    I: Iterator<Item = WatermarkedBucketItem>,
268{
269    let mut open_iter = Some(open_iter);
270    let mut iter: Option<I> = None;
271    let mut iter_range: Option<Range<u64>> = None;
272    let mut out = Vec::new();
273    let mut scan_limit_frontier: Option<u64> = None;
274    while out.len() < hit_limit {
275        if cancel.is_cancelled() {
276            return Err(cancelled());
277        }
278        if let Some(mut bucket) = pending_bucket.take() {
279            bucket.drain_into(&mut out, hit_limit, bucket_size, direction);
280            if bucket.has_hits(bucket_size, direction) {
281                pending_bucket = Some(bucket);
282                break;
283            }
284            continue;
285        }
286
287        let Some(scan_range) = next_range.clone() else {
288            break;
289        };
290        if iter.is_none() {
291            let open = open_iter
292                .take()
293                .expect("bitmap query iterator is only opened once");
294            iter_range = Some(scan_range.clone());
295            iter = Some(open(scan_range));
296        }
297
298        let iter_range = iter_range
299            .as_ref()
300            .expect("bitmap iterator range is set before polling");
301        match iter
302            .as_mut()
303            .expect("bitmap iterator is set before polling")
304            .next()
305        {
306            Some(Ok(Watermarked::Item((bucket_id, bitmap)))) => {
307                next_range =
308                    remaining_range_after_bucket(iter_range, bucket_id, bucket_size, direction);
309                pending_bucket =
310                    PendingBitmapBucket::new(bucket_id, bitmap, iter_range, bucket_size);
311            }
312            // The fullnode only reports the scan-limit frontier.
313            Some(Ok(Watermarked::Watermark(_))) => {}
314            Some(Err(ScanStop::ScanLimit { scan_frontier })) => {
315                // Budget stop: the terminal carries the merged resume frontier
316                // (the exact value the stopping round's beacon would have held).
317                // In ascending scans this is the first unscanned position, so
318                // resume inclusively without item-consumption's `+ 1`.
319                // Descending continuation remains strictly below the frontier.
320                scan_limit_frontier = Some(scan_frontier);
321                next_range = if direction.is_ascending() {
322                    let remaining = scan_frontier.max(iter_range.start)..iter_range.end;
323                    (!remaining.is_empty()).then_some(remaining)
324                } else {
325                    remaining_range_after(iter_range.clone(), scan_frontier, false)
326                };
327                break;
328            }
329            // Cancelled stream → gRPC Cancelled status.
330            Some(Err(ScanStop::Cancelled)) => {
331                return Err(RpcError::new(
332                    tonic::Code::Cancelled,
333                    ScanStop::Cancelled.to_string(),
334                ));
335            }
336            // Genuine fault → gRPC Internal, carrying the error unchanged.
337            Some(Err(ScanStop::Fault(inner))) => {
338                return Err(RpcError::new(tonic::Code::Internal, inner.to_string()));
339            }
340            None => {
341                next_range = None;
342                break;
343            }
344        }
345    }
346    Ok(DrainLoopState {
347        items: out,
348        pending_bucket,
349        next_range,
350        scan_limit_frontier,
351    })
352}

354fn remaining_range_after_bucket(
355    range: &Range<u64>,
356    bucket_id: u64,
357    bucket_size: u64,
358    direction: ScanDirection,
359) -> Option<Range<u64>> {
360    let remaining = if direction.is_ascending() {
361        ((bucket_id + 1) * bucket_size).max(range.start)..range.end
362    } else {
363        range.start..(bucket_id * bucket_size).min(range.end)
364    };
365    (!remaining.is_empty()).then_some(remaining)
366}

389/// Inner bucket source: a stored RocksDB scan, or the synthesized dense
390/// tx-universe sequence — `IndexDimension::TxUniverse` is query-only and
391/// never reaches storage.
392enum BitmapBucketIter<'a> {
393    Stored(LedgerBitmapBucketIterator<'a>),
394    Universe(DenseUniverseBuckets),
395}

397/// Leaf bucket iterator for one index dimension. It yields buckets as read; the
398/// evaluator charges each pulled bucket against the scan budget, including the
399/// reserved first bucket that lets every leaf report its first watermark.
400struct RpcIndexesBitmapIterator<'a> {
401    cancel: CancellationToken,
402    iter: Option<BitmapBucketIter<'a>>,
403    finished: bool,
404    initial_error: Option<LeafStop>,
405}
406
407impl<'a> RpcIndexesBitmapIterator<'a> {
408    fn new(
409        service: &'a RpcService,
410        kind: LedgerBitmapKind,
411        bucket_size: u64,
412        cancel: CancellationToken,
413        dimension_key: Vec<u8>,
414        range: Range<u64>,
415        direction: ScanDirection,
416    ) -> Self {
417        if range.is_empty() {
418            return Self {
419                cancel,
420                iter: None,
421                finished: true,
422                initial_error: None,
423            };
424        }
425
426        // The tx universe is dense (every tx_seq in range is real), so it is
427        // synthesized rather than read from storage. Gated on the tx kind: the
428        // key is only ever produced by the tx filter layer, and in event-space
429        // a dense universe would be semantically wrong.
430        if kind == LedgerBitmapKind::Transaction
431            && dimension_key.first() == Some(&IndexDimension::TxUniverse.tag_byte())
432        {
433            return Self {
434                cancel,
435                finished: false,
436                iter: Some(BitmapBucketIter::Universe(dense_universe_buckets(
437                    range,
438                    bucket_size,
439                    direction,
440                ))),
441                initial_error: None,
442            };
443        }
444
445        let start_bucket = range.start / bucket_size;
446        let end_bucket_exclusive = (range.end - 1) / bucket_size + 1;
447        let iter = service
448            .reader
449            .inner()
450            .indexes()
451            .ok_or_else(|| anyhow::anyhow!("rpc indexes are disabled"))
452            .and_then(|indexes| {
453                let descending = !direction.is_ascending();
454                let iter = match kind {
455                    LedgerBitmapKind::Transaction => indexes.transaction_bitmap_bucket_iter(
456                        dimension_key,
457                        start_bucket,
458                        end_bucket_exclusive,
459                        descending,
460                    ),
461                    LedgerBitmapKind::Event => indexes.event_bitmap_bucket_iter(
462                        dimension_key,
463                        start_bucket,
464                        end_bucket_exclusive,
465                        descending,
466                    ),
467                };
468                iter.map_err(|e| anyhow::anyhow!(e.to_string()))
469            });
470
471        let (iter, initial_error) = match iter {
472            Ok(iter) => (Some(BitmapBucketIter::Stored(iter)), None),
473            Err(e) => (None, Some(e.into())),
474        };
475        Self {
476            cancel,
477            finished: false,
478            iter,
479            initial_error,
480        }
481    }
482
483    fn read_next_bucket(&mut self) -> Option<Result<(u64, RoaringBitmap), LeafStop>> {
484        if self.cancel.is_cancelled() {
485            self.finished = true;
486            return Some(Err(LeafStop::Cancelled));
487        }
488        let Some(iter) = self.iter.as_mut() else {
489            self.finished = true;
490            return None;
491        };
492
493        let next = match iter {
494            BitmapBucketIter::Stored(iter) => match iter.next() {
495                Some(Ok(bucket)) => Some(Ok((bucket.bucket_id, bucket.bitmap))),
496                Some(Err(e)) => Some(Err(LeafStop::Fault(anyhow::anyhow!(e.to_string())))),
497                None => None,
498            },
499            BitmapBucketIter::Universe(iter) => iter.next().map(Ok),
500        };
501        if !matches!(next, Some(Ok(_))) {
502            self.finished = true;
503        }
504        next
505    }
506}
507
508impl Iterator for RpcIndexesBitmapIterator<'_> {
509    type Item = Result<(u64, RoaringBitmap), LeafStop>;
510
511    fn next(&mut self) -> Option<Self::Item> {
512        if let Some(e) = self.initial_error.take() {
513            self.finished = true;
514            return Some(Err(e));
515        }
516        if self.finished {
517            return None;
518        }
519
520        self.read_next_bucket()
521    }
522}
523impl SeekableBucketIterator for RpcIndexesBitmapIterator<'_> {
524    fn seek_bucket(&mut self, bucket: u64) {
525        match self.iter.as_mut() {
526            Some(BitmapBucketIter::Stored(iter)) => iter.seek_bucket(bucket),
527            Some(BitmapBucketIter::Universe(iter)) => iter.seek_bucket(bucket),
528            None => {}
529        }
530    }
