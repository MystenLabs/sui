4//! !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
5//! !! THIS MODULE, AND THE FLOOR SEARCH IN `retention.rs`, ARE COMPLICATED ONLY BECAUSE THE     !!
6//! !! FULLNODE HAS NO INDEX FROM AN OBJECT VERSION TO THE TRANSACTION THAT WROTE IT.           !!
7//! !!                                                                                          !!
8//! !! A tombstone stores no transaction digest, the version index is keyed by checkpoint, and  !!
9//! !! the transaction bitmap holds object IDs without versions. So the transaction behind a    !!
10//! !! removal is pieced together from all three: the item above it, a search of the version   !!
11//! !! index for its checkpoint, a walk of the bitmap, and the candidates' effects.             !!
12//! !!                                                                                          !!
13//! !! An rpc index keyed by (object ID, version), holding the sequence number of the           !!
14//! !! transaction that wrote that item, live or a removal, answers both questions with one     !!
15//! !! read and lets all of this be deleted. It needs a backfill from history, so the next time !!
16//! !! a release backfills an rpc index from history, ADD THIS INDEX IN THAT SAME RELEASE.      !!
17//! !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!

35pub(super) struct FoundRemoval {
36    pub(super) version: SequenceNumber,
37    pub(super) transaction: LedgerTxSeqDigest,
38}

40/// The transaction that deleted or wrapped `object_id` at the tombstone stored at `version`, or
41/// `None` when this node no longer has what it needs to find it. Fails with `CANCELLED` once
42/// `cancel` fires.
43///
44/// Nothing changes a deleted object until a transaction creates it again under the same ID, nor
45/// a wrapped one until a transaction unwraps or deletes it. Such a transaction writes the
46/// object's next item, unless it creates the object straight into a wrapper, which writes no
47/// item of it. So between the transaction that wrote the item right below the tombstone and the
48/// one that wrote the item right above it, the object's only transactions are the removal and
49/// such creations, and the transaction bitmap lists them. The lookup walks them from a bound it
50/// knows, passing over candidates that removed nothing:
51/// - The item above is a live version: back from the transaction that wrote it.
52/// - There is no item above up to the indexed tip: back from the end of the tip checkpoint's
53///   transactions.
54/// - The item above is a tombstone whose removal `previous` holds, which a descending listing
55///   found last: back from that removal.
56/// - The item below is a tombstone whose removal `previous` holds, which an ascending listing
57///   found last: forward from that removal.
58/// - Otherwise, inside a run of tombstones: back from the end of the checkpoint the version index
59///   places the removal in, the first one after the floor at whose end it records a version
60///   above the object's stored item right below the tombstone, stopping below that checkpoint.
61///   The comparison is with the item below because the index records a removal by a transaction
62///   with V1 effects below its tombstone (see [`super::stored_version_at`]). Later transactions
63///   of the object in that checkpoint are passed over, so a listing pays for them once, at the
64///   first tombstone it looks up in a run.
65///
66/// `None` when the transaction that wrote the item above is not indexed, when a candidate's
67/// rpc-index row or effects are pruned or it is at or below `floor_checkpoint`, or when the
68/// search or the walk finds no removal at `version`.
69pub(super) fn removal_transaction(
70    reader: &StateReader,
71    object_id: ObjectID,
72    version: SequenceNumber,
73    previous: Option<FoundRemoval>,
74    floor_checkpoint: Option<u64>,
75    tip: &IndexedTip,
76    cancel: &CancellationToken,
77) -> Result<Option<LedgerTxSeqDigest>> {
78    let after_floor = floor_checkpoint.map_or(0, |floor| floor.saturating_add(1));
79    let walk_back_from = |before, lowest_checkpoint| {
80        walk_back(
81            reader,
82            object_id,
83            version,
84            before,
85            lowest_checkpoint,
86            cancel,
87        )
88    };
89    let above = match first_item_after(reader, object_id, version, tip)? {
90        None => {
91            return walk_back_from(
92                reader.transactions_end(tip.upper_bound.highest_indexed())?,
93                after_floor,
94            );
95        }
96        Some(ObjectOrTombstone::Object(object)) => {
97            return match reader.transaction_sequence(&object.previous_transaction)? {
98                Some(sequence) => walk_back_from(sequence, after_floor),
99                None => Ok(None),
100            };
101        }
102        Some(ObjectOrTombstone::Tombstone((_, above, _))) => above,
103    };
104    if let Some(previous) = previous
105        && previous.version == above
106    {
107        return walk_back_from(previous.transaction.tx_sequence_number, after_floor);
108    }
109    let below = match version.value().checked_sub(1) {
110        Some(below) => reader
111            .object_versions_iter(
112                object_id,
113                SequenceNumber::MIN..=SequenceNumber::from_u64(below),
114                true,
115            )?
116            .next()
117            .transpose()?
118            .map(|item| stored_version(&item)),
119        None => None,
120    };
121    if let Some(previous) = previous
122        && Some(previous.version) == below
123    {
124        return walk_forward(
125            reader,
126            object_id,
127            version,
128            previous.transaction.tx_sequence_number + 1,
129            tip,
130            cancel,
131        );
132    }
133    let Some((checkpoint, _)) =
134        reader.first_checkpoint_recording_above(object_id, below, after_floor, tip)?
135    else {
136        return Ok(None);
137    };
138    walk_back_from(reader.transactions_end(checkpoint)?, checkpoint)
139}

141/// The transaction that removed `object_id` at `version`: the first of the object's
142/// transactions from sequence number `from` on, when the transactions before it that changed
143/// the object since `from` removed nothing (see [`removal_transaction`]). `None` at the first
144/// candidate that removed the object at another version, that is above the indexed tip, or
145/// whose rpc-index row or effects are pruned.
146fn walk_forward(
147    reader: &StateReader,
148    object_id: ObjectID,
149    version: SequenceNumber,
150    mut from: u64,
151    tip: &IndexedTip,
152    cancel: &CancellationToken,
153) -> Result<Option<LedgerTxSeqDigest>> {
154    loop {
155        if cancel.is_cancelled() {
156            return Err(cancelled());
157        }
158        let Some(sequence) = reader.next_object_transaction(object_id, from)? else {
159            return Ok(None);
160        };
161        let Some(transaction) = reader.ledger_transaction(sequence)? else {
162            return Ok(None);
163        };
164        if transaction.checkpoint_number > tip.upper_bound.highest_indexed() {
165            return Ok(None);
166        }
167        match reader.removal_version(&transaction.digest, object_id) {
168            Some(Some(removed)) => return Ok((removed == version).then_some(transaction)),
169            Some(None) => from = sequence + 1,
170            None => return Ok(None),
171        }
172    }
173}

175/// The transaction that removed `object_id` at `version`: the last of the object's transactions
176/// before the one with sequence number `before` whose effects show it. Candidates whose effects
177/// show no removal, or one at a later version by a later transaction of the same checkpoint, are
178/// passed over. `None` at the first candidate that removed the object below `version`, which
179/// the walk passed the removal to reach, in a checkpoint below `lowest_checkpoint`, or whose
180/// rpc-index row or effects are pruned. The indexes prune a transaction's bitmap entry no sooner
181/// than its row, so the walk never passes over a removal whose entry is gone without stopping
182/// at a pruned row first.
183fn walk_back(
184    reader: &StateReader,
185    object_id: ObjectID,
186    version: SequenceNumber,
187    mut before: u64,
188    lowest_checkpoint: u64,
189    cancel: &CancellationToken,
190) -> Result<Option<LedgerTxSeqDigest>> {
191    loop {
192        if cancel.is_cancelled() {
193            return Err(cancelled());
194        }
195        let Some(sequence) = reader.last_object_transaction(object_id, before)? else {
196            return Ok(None);
197        };
198        let Some(transaction) = reader.ledger_transaction(sequence)? else {
199            return Ok(None);
200        };
201        if transaction.checkpoint_number < lowest_checkpoint {
202            return Ok(None);
203        }
204        match reader.removal_version(&transaction.digest, object_id) {
205            Some(Some(removed)) if removed == version => return Ok(Some(transaction)),
206            Some(Some(removed)) if removed < version => return Ok(None),
207            Some(_) => before = sequence,
208            None => return Ok(None),
209        }
210    }
