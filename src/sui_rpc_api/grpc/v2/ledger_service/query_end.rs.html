8/// Effective terminal reason for a query stream, from the reason its scan terminal ends the page
9/// with ([`ScanTerminal::reason`]).
10///
11/// Hitting the requested item limit takes precedence over the underlying chunk
12/// scan result, including a terminal that would fail the stream: the page ends
13/// normally at the item limit. Below the item limit, a failing terminal's error
14/// passes through; the caller sends the terminal watermark frame, then fails
15/// the stream with it. The caller remains responsible for constructing and
16/// attaching `QueryEnd`.
17///
18/// [`ScanTerminal::reason`]: crate::ledger_history::watermark::ScanTerminal::reason
19pub(super) fn effective_terminal_reason(
20    produced: usize,
21    limit_items: usize,
22    scan_end_reason: Result<QueryEndReason, RpcError>,
23) -> Result<QueryEndReason, RpcError> {
24    if produced == limit_items {
25        Ok(QueryEndReason::ItemLimit)
26    } else {
27        scan_end_reason
28    }
29}

43    #[test]
44    fn item_limit_takes_precedence_without_overriding_earlier_terminal_reasons() {
45        assert_eq!(
46            effective_terminal_reason(3, 3, Ok(QueryEndReason::ScanLimit)).unwrap(),
47            QueryEndReason::ItemLimit
48        );
49        assert_eq!(
50            effective_terminal_reason(2, 3, Ok(QueryEndReason::LedgerTip)).unwrap(),
51            QueryEndReason::LedgerTip
52        );
53        assert_eq!(
54            effective_terminal_reason(2, 3, Ok(QueryEndReason::ScanLimit)).unwrap(),
55            QueryEndReason::ScanLimit
56        );
57    }
58
59    /// A terminal that fails the stream fails it only below the item limit; a page that fills its
60    /// item limit ends normally.
61    #[test]
62    fn item_limit_takes_precedence_over_a_failing_terminal() {
63        let serving_floor = || Err(below_serving_floor(7));
64        assert_eq!(
65            effective_terminal_reason(3, 3, serving_floor()).unwrap(),
66            QueryEndReason::ItemLimit
67        );
68        let status = effective_terminal_reason(2, 3, serving_floor())
69            .expect_err("below the item limit the terminal's error passes through")
70            .into_status_proto();
71        assert_eq!(status.code, tonic::Code::OutOfRange as i32);
72    }

74    fn assert_query_end_metric(
75        produced: usize,
76        limit_items: usize,
77        scan_end_reason: QueryEndReason,
78        expected_reason: QueryEndReason,
79        expected_label: &str,
80    ) {
81        let registry = Registry::new();
82        let metrics = ListApiMetrics::new(&registry);
83        let handles = metrics.stream_metrics("list_transactions", "digest");
84        let mut request_metrics = ListRequestMetrics::new(Some(handles), Instant::now());
85
86        let effective_reason =
87            effective_terminal_reason(produced, limit_items, Ok(scan_end_reason)).unwrap();
88        assert_eq!(effective_reason, expected_reason);
89        request_metrics.finish_success(effective_reason, None);
90
91        let families = registry.gather();
92        let query_ends = families
93            .iter()
94            .find(|family| family.name() == "list_query_ends_total")
95            .expect("list_query_ends_total metric family");
96        assert_eq!(query_ends.get_metric().len(), 1);
97        let metric = &query_ends.get_metric()[0];
98        assert_eq!(metric.get_counter().value(), 1.0);
99        let labels = metric
100            .get_label()
101            .iter()
102            .map(|label| (label.name(), label.value()))
103            .collect::<BTreeMap<_, _>>();
104        assert_eq!(labels.get("method"), Some(&"list_transactions"));
105        assert_eq!(labels.get("reason"), Some(&expected_label));
106    }

108    #[test]
109    fn query_end_metrics_use_effective_reason_and_item_limit_precedence() {
110        assert_query_end_metric(
111            3,
112            3,
113            QueryEndReason::ScanLimit,
114            QueryEndReason::ItemLimit,
115            "item_limit",
116        );
117        for (reason, label) in [
118            (QueryEndReason::ScanLimit, "scan_limit"),
119            (QueryEndReason::LedgerTip, "ledger_tip"),
120            (QueryEndReason::CheckpointBound, "checkpoint_bound"),
121            (QueryEndReason::CursorBound, "cursor_bound"),
122        ] {
123            assert_query_end_metric(2, 3, reason, reason, label);
124        }
125    }
