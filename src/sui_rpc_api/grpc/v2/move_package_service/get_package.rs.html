19#[tracing::instrument(skip(service))]
20pub fn get_package(service: &RpcService, request: GetPackageRequest) -> Result<GetPackageResponse> {
21    let package_id_str = request.package_id.as_ref().ok_or_else(|| {
22        FieldViolation::new("package_id")
23            .with_description("missing package_id")
24            .with_reason(ErrorReason::FieldMissing)
25    })?;
26
27    let package = match request.selector {
28        Some(Selector::Version(version)) => {
29            load_package_at_version(service, package_id_str, version)?
30        }
31        Some(Selector::AtCheckpoint(at_checkpoint)) => {
32            load_package_at_checkpoint(service, package_id_str, at_checkpoint)?
33        }
34        None => load_package(service, package_id_str)?,
35        _ => {
36            return Err(FieldViolation::new("selector")
37                .with_description("unknown selector variant")
38                .with_reason(ErrorReason::FieldInvalid)
39                .into());
40        }
41    };
42
43    get_package_response(&package)
44}

46/// Load the package in `package_id_str`'s upgrade lineage with exactly
47/// `version`.
48///
49/// System packages are upgraded in place: all their versions share one ID, so
50/// an earlier version is read by that ID and the version, without the
51/// package-version index. Nodes without object version history refuse that
52/// read (`OutOfRange`, `VERSION_NOT_RETAINED`) even while the version is still
53/// stored. Other packages keep each version at its own storage ID, found
54/// through the package-version index, whose rows are keyed by the lineage's
55/// original package id.
56///
57/// A version this node does not store is refused (see
58/// [`refuse_unstored_version`]). A version below the loaded package's that the
59/// package-version index has no row for is refused as not ready while the
60/// index has no row for the loaded package either: the index trails the
61/// node's live state, and the requested version may have been published after
62/// the index's tip. Any other version the index has no row for is `NotFound`,
63/// unless the index is empty or still backfilling history after a restore and
64/// the row may yet be written.
65fn load_package_at_version(
66    service: &RpcService,
67    package_id_str: &str,
68    version: u64,
69) -> Result<MovePackage> {
70    let package = load_package(service, package_id_str)?;
71    let original_id = package.original_package_id();
72    if package.version().value() == version {
73        return Ok(package);
74    }
75    // A lineage's versions run from `OBJECT_START_VERSION` up without gaps, so a version
76    // below the loaded package's existed.
77    let existed = (OBJECT_START_VERSION.value()..package.version().value()).contains(&version);
78
79    // Only a package upgraded in place keeps its original ID past its first version, so the
80    // loaded package is its lineage's latest version.
81    if package.id() == original_id && package.version() > OBJECT_START_VERSION {
82        if !existed {
83            return Err(RpcError::not_found());
84        }
85        service.reader.require_object_version_history()?;
86        return match load_package_by_id_and_version(service, original_id, version)? {
87            Some(package) => Ok(package),
88            None => refuse_unstored_version(service, original_id, version),
89        };
90    }
91
92    let indexes = service
93        .reader
94        .inner()
95        .indexes()
96        .ok_or_else(indexes_disabled)?;
97
98    let storage_id = indexes
99        .get_package_version_storage_id(original_id, version)
100        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?;
101    let stored = match storage_id {
102        Some(storage_id) => load_package_by_id_and_version(service, storage_id, version)?,
103        None if existed => {
104            let latest_storage_id = indexes
105                .get_package_version_storage_id(original_id, package.version().value())
106                .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?;
107            if latest_storage_id.is_none() {
108                check_restore_backfill_done(service)?;
109                return Err(not_ready(
110                    NotReadyCause::HistoryCatchingUp,
111                    format!(
112                        "the rpc indexes have not yet reached version {} of package {original_id}",
113                        package.version().value()
114                    ),
115                ));
116            }
117            // The index writes a lineage's rows in publish order, so with the loaded version's
118            // row present the requested version's row is too, unless it was missing because the
119            // index committed both rows between the two reads: read it again.
120            match indexes
121                .get_package_version_storage_id(original_id, version)
122                .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
123            {
124                Some(storage_id) => load_package_by_id_and_version(service, storage_id, version)?,
125                None => None,
126            }
127        }
128        // No lineage has a version below `OBJECT_START_VERSION`.
129        None if version < OBJECT_START_VERSION.value() => return Err(RpcError::not_found()),
130        None => {
131            check_restore_backfill_done(service)?;
132            return Err(RpcError::not_found());
133        }
134    };
135
136    match stored {
137        Some(package) => Ok(package),
138        None => refuse_unstored_version(service, original_id, version),
139    }
140}

142/// Refuse `version` of the package lineage `original_id`, which existed but this node does not
143/// store. The refusal is not ready while the rpc indexes are empty or backfill history after a
144/// restore, pruned when this node's object history has a floor (a node restored from a snapshot
145/// has none of the versions replaced before it), and an internal error on a node that has pruned
146/// nothing, which stores every version.
147fn refuse_unstored_version(
148    service: &RpcService,
149    original_id: ObjectID,
150    version: u64,
151) -> Result<MovePackage> {
152    check_restore_backfill_done(service)?;
153    let lower_bound = service.reader.at_checkpoint_lower_bound()?;
154    if lower_bound.pruned_floor().is_none() {
155        return Err(RpcError::new(
156            tonic::Code::Internal,
157            format!(
158                "version {version} of package {original_id} is not stored, although this node \
159                 has pruned no object history"
160            ),
161        ));
162    }
163    Err(lower_bound.pruned())
164}

166/// `UNAVAILABLE` (`NOT_READY`) while the rpc indexes backfill history after a restore (cause
167/// `restore_backfill`) or have not committed a checkpoint yet (cause `index_empty`), when an
168/// answer that a package version is missing may not hold once the indexes catch up.
169fn check_restore_backfill_done(service: &RpcService) -> Result<()> {
170    let Some(indexes) = service.reader.inner().indexes() else {
171        return Ok(());
172    };
173    if indexes
174        .is_history_backfill_pending()
175        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
176    {
177        return Err(not_ready(
178            NotReadyCause::RestoreBackfill,
179            "rpc index history is still backfilling after a restore",
180        ));
181    }
182    if indexes
183        .get_highest_indexed_checkpoint_seq_number()
184        .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
185        .is_none()
186    {
187        return Err(index_empty());
188    }
189    Ok(())
190}

192/// Load the latest package in `package_id_str`'s upgrade lineage that
193/// existed at or before `at_checkpoint`.
194///
195/// `at_checkpoint` is checked against the shared checkpoint-pinned read
196/// bounds, the upper bound before the package read and the lower bound after
197/// it (see `StateReader::at_checkpoint_upper_bound`), so a successful answer
198/// is exact as of `at_checkpoint`. Both bounds apply to every outcome of the
199/// read, `NotFound` included, so an out-of-range checkpoint is `OutOfRange`
200/// whether or not the package exists, as on kv-rpc.
201fn load_package_at_checkpoint(
202    service: &RpcService,
203    package_id_str: &str,
204    at_checkpoint: u64,
205) -> Result<MovePackage> {
206    let package_id = parse_package_id(package_id_str)?;
207
208    service
209        .reader
210        .at_checkpoint_upper_bound()?
211        .check(at_checkpoint)?;
212
213    let indexes = service
214        .reader
215        .inner()
216        .indexes()
217        .ok_or_else(indexes_disabled)?;
218
219    let package = load_package_by_id(service, package_id).and_then(|package| {
220        indexes
221            .get_package_at_checkpoint(package.original_package_id(), at_checkpoint)
222            .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
223            .ok_or_else(RpcError::not_found)
224            .and_then(|(version, storage_id)| {
225                // The lower bound below replaces any answer at a checkpoint it refuses; at the
226                // others, pruning has not reached the version current at `at_checkpoint`.
227                load_package_by_id_and_version(service, storage_id, version)?.ok_or_else(|| {
228                    RpcError::new(
229                        tonic::Code::Internal,
230                        format!(
231                            "package {storage_id} version {version} has a package-version index \
232                             row but no stored object"
233                        ),
234                    )
235                })
236            })
237    });
238
239    service
240        .reader
241        .at_checkpoint_lower_bound()?
242        .check(at_checkpoint)?;
243    package
244}

246/// Build a `GetPackage` response from an already-loaded package. Shared with other implementors
247/// of `MovePackageService` (e.g. sui-kv-rpc) that load packages from a different backend.
248pub fn get_package_response(package: &MovePackage) -> Result<GetPackageResponse> {
249    let package_id = package.id();
250
251    let resolved_package =
252        sui_package_resolver::Package::read_from_package(package).map_err(convert_error)?;
253
254    let modules: Vec<_> = resolved_package
255        .modules()
256        .iter()
257        .map(|(module_name, resolver_module)| {
258            convert_module(module_name, resolver_module, &package_id)
259        })
260        .collect::<Result<Vec<_>>>()?;
261
262    let mut message = Package::default();
263    message.storage_id = Some(package_id.to_canonical_string(true));
264    message.original_id = Some(package.original_package_id().to_canonical_string(true));
265    message.version = Some(package.version().value());
266    message.modules = modules;
267
268    Ok(GetPackageResponse::new(message))
269}

412    /// A user package lineage with one version per ID in `storage_ids`, from version 1 at the
413    /// original ID `storage_ids[0]` up.
414    fn user_package_lineage(storage_ids: &[ObjectID]) -> Vec<Object> {
415        let original_id = storage_ids[0];
416        let config = ProtocolConfig::get_for_max_version_UNSAFE();
417        let mut lineage = vec![
418            Object::new_package(
419                &[module(original_id)],
420                TransactionDigest::genesis_marker(),
421                &config,
422                [],
423            )
424            .unwrap(),
425        ];
426        for &storage_id in &storage_ids[1..] {
427            let Data::Package(previous) = &lineage.last().unwrap().data else {
428                unreachable!("packages only");
429            };
430            let upgraded = previous
431                .new_upgraded(storage_id, &[module(original_id)], &config, [])
432                .unwrap();
433            lineage.push(Object::new_package_from_data(
434                Data::Package(upgraded),
435                TransactionDigest::genesis_marker(),
436            ));
437        }
438        lineage
439    }
