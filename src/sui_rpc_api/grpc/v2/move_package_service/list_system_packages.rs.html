11/// List the system packages that existed as of the end of `at_checkpoint`, each at its latest
12/// version as of that checkpoint, ordered by package id. Without `at_checkpoint`, the request is
13/// answered exactly as if it named the highest checkpoint every rpc index has committed: each
14/// version's publish checkpoint comes from the `package_versions` index, so the answer is at a
15/// height that index has fully recorded.
16///
17/// The answer is the complete set or an error: `at_checkpoint` is checked against the shared
18/// checkpoint-pinned read bounds, the upper bound before the reads and the lower bound after
19/// them (see `StateReader::at_checkpoint_upper_bound`), so a checkpoint outside the indexed range
20/// fails the whole request. Either way, a node without rpc indexes refuses, as does one whose
21/// indexes are empty or still backfilling history after a restore.
22///
23/// The candidates are the system package ids compiled into this binary. A fullnode cannot execute
24/// past a protocol upgrade it does not know, so every system package in the data it serves is in
25/// its compiled list. Only the rpc indexes are read, never object contents.
26#[tracing::instrument(skip(service))]
27pub fn list_system_packages(
28    service: &RpcService,
29    request: ListSystemPackagesRequest,
30) -> Result<ListSystemPackagesResponse> {
31    let upper_bound = service.reader.at_checkpoint_upper_bound()?;
32    let at_checkpoint = request
33        .at_checkpoint
34        .unwrap_or_else(|| upper_bound.highest_indexed());
35    upper_bound.check(at_checkpoint)?;
36
37    let indexes = service
38        .reader
39        .inner()
40        .indexes()
41        .ok_or_else(indexes_disabled)?;
42
43    let system_packages = system_package_versions(indexes, at_checkpoint);
44
45    service
46        .reader
47        .at_checkpoint_lower_bound()?
48        .check(at_checkpoint)?;
49    system_packages.map(ListSystemPackagesResponse::new)
50}

52/// The version of each compiled system package as of the end of `at_checkpoint`, ordered by
53/// package id, omitting packages that did not exist yet. Each version carries the checkpoint it
54/// was published in, unset when the indexes' history does not reach back to it.
55fn system_package_versions(
56    indexes: &dyn RpcIndexes,
57    at_checkpoint: u64,
58) -> Result<Vec<PackageVersion>> {
59    let mut package_ids: Vec<ObjectID> = SYSTEM_PACKAGE_ADDRESSES
60        .iter()
61        .copied()
62        .map(ObjectID::from)
63        .collect();
64    package_ids.sort();
65
66    let mut system_packages = Vec::with_capacity(package_ids.len());
67    for package_id in package_ids {
68        // System packages are never deleted or wrapped, so a resolved version is never a
69        // tombstone: it is the live package.
70        let Some(version) = indexes
71            .get_object_version_at_checkpoint(package_id, at_checkpoint)
72            .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
73        else {
74            continue;
75        };
76        let version = version.value();
77
78        // A system package is its own original package, so its versions are listed under its id.
79        let published = indexes
80            .package_versions_iter(package_id, version, version.saturating_add(1), false)
81            .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
82            .next()
83            .transpose()
84            .map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?
85            .and_then(|entry| entry.checkpoint);
86
87        let mut entry = PackageVersion::new(&package_id.into(), version)
88            .with_original_id(package_id.to_canonical_string(true));
89        entry.checkpoint = published;
90        system_packages.push(entry);
91    }
92    Ok(system_packages)
93}

161        /// Stores `version` of system package `package_id` in the objects table, as executing
162        /// the transaction that wrote it does.
163        fn store_package(&self, package_id: ObjectID, version: u64) {
164            let mut module: CompiledModule = empty_module();
165            module.address_identifiers[0] = package_id.into();
166            let package = Object::new_package_from_data(
167                Data::Package(MovePackage::new_system(
168                    SequenceNumber::from_u64(version),
169                    &[module],
170                    [],
171                )),
172                TransactionDigest::genesis_marker(),
173            );
174            let key = objects::Key {
175                id: package.id(),
176                version: package.version(),
177            };
178            let mut batch = self.db.batch();
179            batch
180                .put(&self.schema.objects, &key, &objects::store(&package))
181                .unwrap();
182            batch.commit().unwrap();
183        }
184
185        /// Writes the index rows recording `version` of system package `package_id` as published
186        /// at `checkpoint`. Reads see them once the pipelines commit `checkpoint`.
187        fn index_package(&self, package_id: ObjectID, version: u64, checkpoint: u64) {
188            let (key, value) = package_versions::store(package_id, version, package_id, checkpoint);
189            let (version_key, version_value) = object_version_by_checkpoint::store(
190                package_id,
191                checkpoint,
192                SequenceNumber::from_u64(version),
193            );
194            let mut batch = self.db.batch();
195            batch
196                .put(&self.schema.package_versions, &key, &value)
197                .unwrap();
198            batch
199                .put(
200                    &self.schema.object_version_by_checkpoint,
201                    &version_key,
202                    &version_value,
203                )
204                .unwrap();
205            batch.commit().unwrap();
206        }
207
208        fn list(&self, at_checkpoint: Option<u64>) -> Result<Vec<PackageVersion>> {
209            let request = at_checkpoint.map_or_else(
210                ListSystemPackagesRequest::default,
211                ListSystemPackagesRequest::new,
212            );
213            list_system_packages(&self.service, request).map(|response| response.system_packages)
214        }
215    }

228    /// Without `at_checkpoint`, the answer is the one for the highest checkpoint every index has
229    /// committed. A newer version that execution has written, and whose index rows are written
230    /// above that checkpoint, is not listed until the indexes commit its checkpoint.
231    #[test]
232    fn omitted_checkpoint_answers_at_the_highest_indexed_checkpoint() {
233        let fixture = Fixture::new();
234        commit_every_pipeline(&fixture.db, 20);
235        let genesis = vec![
236            system_package_version(MOVE_STDLIB_PACKAGE_ID, 1, 0),
237            system_package_version(SUI_FRAMEWORK_PACKAGE_ID, 1, 0),
238            system_package_version(SUI_SYSTEM_PACKAGE_ID, 1, 0),
239        ];
240        assert_eq!(fixture.list(None).unwrap(), genesis);
241        assert_eq!(fixture.list(Some(20)).unwrap(), genesis);
242
243        // A framework upgrade at checkpoint 21: executed and partly indexed, but not committed.
244        fixture.store_package(SUI_SYSTEM_PACKAGE_ID, 2);
245        fixture.index_package(SUI_SYSTEM_PACKAGE_ID, 2, 21);
246        assert_eq!(fixture.list(None).unwrap(), genesis);
247        assert_eq!(fixture.list(Some(20)).unwrap(), genesis);
248
249        commit_every_pipeline(&fixture.db, 21);
250        let upgraded = vec![
251            system_package_version(MOVE_STDLIB_PACKAGE_ID, 1, 0),
252            system_package_version(SUI_FRAMEWORK_PACKAGE_ID, 1, 0),
253            system_package_version(SUI_SYSTEM_PACKAGE_ID, 2, 21),
254        ];
255        assert_eq!(fixture.list(None).unwrap(), upgraded);
256        assert_eq!(fixture.list(Some(21)).unwrap(), upgraded);
257        assert_eq!(fixture.list(Some(20)).unwrap(), genesis);
258    }
