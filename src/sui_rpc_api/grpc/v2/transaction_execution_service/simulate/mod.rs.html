43pub fn simulate_transaction(
44    service: &RpcService,
45    request: SimulateTransactionRequest,
46) -> Result<SimulateTransactionResponse> {
47    let executor = service
48        .executor
49        .as_ref()
50        .ok_or_else(|| RpcError::new(tonic::Code::Unimplemented, "no transaction executor"))?;
51
52    let read_mask = match request.read_mask.as_ref() {
53        Some(read_mask) => {
54            read_mask
55                .validate::<SimulateTransactionResponse>()
56                .map_err(|path| {
57                    FieldViolation::new("read_mask")
58                        .with_description(format!("invalid read_mask path: {path}"))
59                        .with_reason(ErrorReason::FieldInvalid)
60                })?;
61            FieldMaskTree::from_field_mask(read_mask)
62        }
63        None => FieldMaskTree::new_wildcard(),
64    };
65
66    let transaction_proto = request
67        .transaction
68        .as_ref()
69        .ok_or_else(|| FieldViolation::new("transaction").with_reason(ErrorReason::FieldMissing))?;
70
71    let checks = TransactionChecks::from(request.checks());
72
73    // TODO make this more efficient
74    let (reference_gas_price, protocol_config) = {
75        let system_state = service.reader.get_system_state_summary()?;
76        let protocol_config = ProtocolConfig::get_for_version_if_supported(
77            system_state.protocol_version.into(),
78            service.reader.inner().get_chain_identifier()?.chain(),
79        )
80        .ok_or_else(|| {
81            RpcError::new(
82                tonic::Code::Internal,
83                "unable to get current protocol config",
84            )
85        })?;
86
87        (system_state.reference_gas_price, protocol_config)
88    };
89
90    // Try to parse out a fully-formed transaction. If one wasn't provided then we will attempt to
91    // perform transaction resolution.
92    let mut transaction = match sui_sdk_types::Transaction::try_from(transaction_proto) {
93        Ok(transaction) => sui_types::transaction::TransactionData::try_from(transaction)?,
94
95        // If we weren't able to parse out a fully-formed transaction and the client provided BCS
96        // TransactionData, then we'll error out early since we're unable to perform resolution
97        // given a BCS payload
98        Err(e) if transaction_proto.bcs.is_some() => {
99            return Err(FieldViolation::new("transaction")
100                .with_description(format!("invalid transaction: {e}"))
101                .with_reason(ErrorReason::FieldInvalid)
102                .into());
103        }
104
105        // We weren't able to parse out a fully-formed transaction so we'll attempt to perform
106        // transaction resolution
107        _ => resolve::resolve_transaction(
108            service,
109            transaction_proto,
110            reference_gas_price,
111            &protocol_config,
112        )?,
113    };
114
115    let perform_gas_selection = request.do_gas_selection() && checks.enabled();
116    let simulation_result = 'simulate: {
117        // BCS transactions that are already in gasless shape (price=0, no payment) are simulated
118        // as-is — the caller pre-built the transaction, so gas selection is not needed and the
119        // priced flow must be skipped (it would fail with GasPriceUnderRGP for price=0).
120        let skip_gas_selection_for_bcs_gasless = perform_gas_selection
121            && request.transaction().bcs_opt().is_some()
122            && transaction.is_gasless_transaction();
123
124        if perform_gas_selection && !skip_gas_selection_for_bcs_gasless {
125            // If the caller didn't set a non-zero price and the tx passes the cheap structural +
126            // object-input gasless checks, try a gasless simulate first. Post-execution gasless
127            // requirements (all input Coins consumed, minimum transfer amounts) can only be
128            // verified by running the tx. If that fails, we discard the gasless variant and
129            // fall through to the priced flow. `payment` is already empty here, verified by
130            // is_gasless_candidate.
131            if is_gasless_candidate(&request, &transaction, &protocol_config, service)? {
132                let mut gasless_tx = transaction.clone();
133                gasless_tx.gas_data_mut().price = 0;
134                gasless_tx.gas_data_mut().budget = 0;
135                // All gasless txns must carry an epoch-scoped validity window for replay
136                // protection.
137                configure_transaction_validity(service, &protocol_config, &mut gasless_tx)?;
138
139                let simulation_result = executor
140                    .simulate_transaction(gasless_tx.clone(), checks, false)
141                    .map_err(simulation_error_to_rpc_error)?;
142
143                if !is_gasless_post_execution_failure(simulation_result.effects.status()) {
144                    transaction = gasless_tx;
145                    break 'simulate simulation_result;
146                }
147            }
148
149            // Priced-flow budget estimation and gas selection.
150            // At this point, the budget on the transaction can be set to one of the following:
151            // - The budget from the request, if specified.
152            // - The total balance of all of the gas payment coins (clamped to the protocol
153            //   MAX_GAS_BUDGET) in the request if the budget was not
154            //   specified but the gas payment coins were specified.
155            // - Protocol MAX_GAS_BUDGET if the request did not specified neither gas payment or budget.
156            //
157            // If the request did not specify a budget, then simulate the transaction to get a budget estimate and
158            // overwrite the resolved budget with the more accurate estimate.
159            // When the request didn't specify a budget, the budget computed below covers
160            // computation + storage + safe-overhead, with the synthetic gas coin's storage
161            // cost subtracted (it doesn't exist at execution time). The cost of loading
162            // any additional payment objects is added either in `estimate_gas_budget_from_gas_cost`
163            // (when payment was specified) or incrementally inside `select_gas` (when gas
164            // selection picks the coins).
165            let budget_was_estimated = request.transaction().gas_payment().budget.is_none()
166                && request.transaction().bcs_opt().is_none();
167            if budget_was_estimated {
168                let mut estimation_transaction = transaction.clone();
169                estimation_transaction.gas_data_mut().payment = Vec::new();
170                estimation_transaction.gas_data_mut().budget = protocol_config.max_tx_gas();
171
172                let simulation_result = executor
173                    .simulate_transaction(
174                        estimation_transaction,
175                        TransactionChecks::Enabled,
176                        true, /* allow mock gas coin */
177                    )
178                    .map_err(simulation_error_to_rpc_error)?;
179
180                let estimate = estimate_gas_budget_from_gas_cost(
181                    simulation_result.effects.gas_cost_summary(),
182                    reference_gas_price,
183                    request.transaction().gas_payment().objects.len(),
184                    mock_gas_storage_cost(&simulation_result),
185                    &protocol_config,
186                );
187
188                // If the request specified gas payment, then transaction.gas_data().budget should have been
189                // resolved to the cumulative balance of those coins. We don't want to return a resolved transaction
190                // where the gas payment can't satisfy the budget, so validate that balance can actually cover the
191                // estimated budget.
192                let gas_balance = transaction.gas_data().budget;
193                if gas_balance < estimate {
194                    return Err(RpcError::new(
195                        tonic::Code::InvalidArgument,
196                        format!(
197                            "Insufficient gas balance to cover estimated transaction cost. \
198                            Available gas balance: {gas_balance} MIST. Estimated gas budget required: {estimate} MIST"
199                        ),
200                    ));
201                }
202                transaction.gas_data_mut().budget = estimate;
203            }
204
205            if transaction.gas_data().payment.is_empty() {
206                select_gas(
207                    service,
208                    &mut transaction,
209                    // Only adjust the budget for actually-selected coins when we just
210                    // computed the budget from estimation. A caller-supplied budget is
211                    // taken as-is.
212                    budget_was_estimated.then_some(reference_gas_price),
213                    &protocol_config,
214                )?;
215            }
216
217            // Coin-paid transactions get a proposer restriction too, so that nobody else can
218            // amplify them into consensus. A no-op on the address-balance paths, which already
219            // set their expiration above.
220            restrict_transaction_proposers(service, &protocol_config, &mut transaction)?;
221        }
222
223        executor
224            .simulate_transaction(transaction.clone(), checks, !perform_gas_selection)
225            .map_err(simulation_error_to_rpc_error)?
226    };
227
228    let SimulateTransactionResult {
229        effects,
230        events,
231        objects,
232        execution_result,
233        mock_gas_id,
234        unchanged_loaded_runtime_objects,
235        suggested_gas_price,
236    } = simulation_result;
237
238    if perform_gas_selection && mock_gas_id.is_some() {
239        // If we don't allow for using a mock coin, but we still did, return a server error
240        return Err(RpcError::new(
241            tonic::Code::Internal,
242            "unexpected mock gas coin used",
243        ));
244    }
245
246    let transaction = if let Some(submask) = read_mask.subtree("transaction") {
247        let mut message = ExecutedTransaction::default();
248        let transaction = sui_sdk_types::Transaction::try_from(transaction)?;
249
250        message.balance_changes =
251            if submask.contains(ExecutedTransaction::BALANCE_CHANGES_FIELD.name) {
252                derive_balance_changes_2(&effects, &objects)
253                    .into_iter()
254                    .map(Into::into)
255                    .collect()
256            } else {
257                vec![]
258            };
259
260        message.effects = submask
261            .subtree(ExecutedTransaction::EFFECTS_FIELD)
262            .map(|mask| {
263                service.render_effects_to_proto(
264                    &effects,
265                    &unchanged_loaded_runtime_objects,
266                    &objects,
267                    &mask,
268                )
269            });
270
271        message.events = submask
272            .subtree(ExecutedTransaction::EVENTS_FIELD.name)
273            .and_then(|mask| {
274                events.map(|events| service.render_events_to_proto(&events, &mask, &objects))
275            });
276
277        message.transaction = submask
278            .subtree(ExecutedTransaction::TRANSACTION_FIELD.name)
279            .map(|mask| Transaction::merge_from(transaction, &mask));
280
281        message.objects = submask
282            .subtree(
283                ExecutedTransaction::path_builder()
284                    .objects()
285                    .objects()
286                    .finish(),
287            )
288            .map(|mask| {
289                ObjectSet::default().with_objects(
290                    objects
291                        .iter()
292                        .map(|o| service.render_object_to_proto(o, &mask, &objects))
293                        .collect(),
294                )
295            });
296
297        Some(message)
298    } else {
299        None
300    };
301
302    let outputs = if read_mask.contains(SimulateTransactionResponse::COMMAND_OUTPUTS_FIELD) {
303        execution_result
304            .into_iter()
305            .flatten()
306            .map(|(reference_outputs, return_values)| {
307                let mut message = CommandResult::default();
308                message.return_values = return_values
309                    .into_iter()
310                    .map(|(bcs, ty)| to_command_output(service, None, bcs, ty))
311                    .collect();
312                message.mutated_by_ref = reference_outputs
313                    .into_iter()
314                    .map(|(arg, bcs, ty)| to_command_output(service, Some(arg), bcs, ty))
315                    .collect();
316                message
317            })
318            .collect()
319    } else {
320        Vec::new()
321    };
322
323    let mut response = SimulateTransactionResponse::default();
324    response.transaction = transaction;
325    response.command_outputs = outputs;
326    if read_mask.contains(SimulateTransactionResponse::SUGGESTED_GAS_PRICE_FIELD) {
327        response.suggested_gas_price = suggested_gas_price;
328    }
329    Ok(response)
330}

332fn simulation_error_to_rpc_error(error: SuiError) -> RpcError {
333    match error.as_inner() {
334        SuiErrorKind::UserInputError { .. } => {
335            RpcError::new(tonic::Code::InvalidArgument, error.to_string())
336        }
337        SuiErrorKind::UnsupportedFeatureError { .. } => {
338            RpcError::new(tonic::Code::InvalidArgument, error.to_string())
339        }
340        _ => RpcError::new(tonic::Code::Internal, error.to_string()),
341    }
342}

344fn to_command_output(
345    service: &RpcService,
346    arg: Option<sui_types::transaction::Argument>,
347    bcs: Vec<u8>,
348    ty: sui_types::TypeTag,
349) -> CommandOutput {
350    let json = service
351        .reader
352        .inner()
353        .get_type_layout(&ty)
354        .ok()
355        .flatten()
356        .and_then(|layout| {
357            let bound = service.config.max_json_move_value_size();
358            sui_types::object::rpc_visitor::proto::ProtoVisitor::new(bound)
359                .deserialize_value(&bcs, &layout)
360                .map_err(|e| tracing::debug!("unable to convert to JSON: {e}"))
361                .ok()
362                .map(Box::new)
363        });
364
365    let mut message = CommandOutput::default();
366    message.argument = arg.map(Into::into);
367    message.value = Some(Bcs::from(bcs).with_name(ty.to_canonical_string(true)));
368    message.json = json;
369    message
370}

372/// Estimate the gas budget for a transaction based on simulation results.
373///
374/// The estimation includes:
375/// 1. Base cost from gas_cost_summary (computation + storage costs), with the synthetic gas
376///    coin's storage cost subtracted (it doesn't exist at execution time).
377/// 2. Cost of loading additional gas payment objects beyond the synthetic gas coin already
378///    in the simulation. When the request didn't specify gas payment objects, this is 0 —
379///    `select_gas` will pick the actual coins (or address balance) and adjust the budget
380///    incrementally for each one.
381/// 3. Rounding up to the protocol gas rounding step (typically 1000 MIST).
382/// 4. Adding safe overhead buffer (1000 * reference_gas_price).
383/// 5. Clamping to max_tx_gas protocol limit.
384fn estimate_gas_budget_from_gas_cost(
385    gas_cost_summary: &sui_types::gas::GasCostSummary,
386    reference_gas_price: u64,
387    num_payment_objects_on_request: usize,
388    mock_gas_storage_cost: u64,
389    protocol_config: &ProtocolConfig,
390) -> u64 {
391    const GAS_SAFE_OVERHEAD: u64 = 1000;
392
393    // The simulation always loads a synthetic gas coin so that it can produce a gas cost
394    // summary even when the caller hasn't specified a gas payment. That coin's storage
395    // write is phantom — it is not written at execution time (real address-balance gas
396    // emits an accumulator event, real coin gas writes the user-provided coin instead) —
397    // so subtract its contribution from `storage_cost` before deriving the estimate.
398    let storage_cost = gas_cost_summary
399        .storage_cost
400        .saturating_sub(mock_gas_storage_cost);
401    let gas_used = gas_cost_summary
402        .computation_cost
403        .saturating_add(storage_cost);
404    let net_gas_usage = (gas_used as i64).saturating_sub(gas_cost_summary.storage_rebate as i64);
405    let base_estimate_mist = gas_cost_summary.computation_cost.max(if net_gas_usage < 0 {
406        0
407    } else {
408        net_gas_usage as u64
409    });
410
411    // Loading cost for additional gas coins beyond the synthetic gas coin already counted
412    // by the simulation. When the request did not specify any payment objects, the loading
413    // cost is added incrementally inside `select_gas` once the actual coins are known.
414    let extra_payment_objects = (num_payment_objects_on_request as u64).saturating_sub(1);
415    let gas_loading_cost_mist =
416        compute_gas_loading_cost_mist(extra_payment_objects, reference_gas_price, protocol_config);
417
418    let safe_overhead_mist = GAS_SAFE_OVERHEAD.saturating_mul(reference_gas_price);
419
420    base_estimate_mist
421        .saturating_add(gas_loading_cost_mist)
422        .saturating_add(safe_overhead_mist)
423        .min(protocol_config.max_tx_gas())
424}

426/// Cost in MIST of loading `extra_coins` gas-payment objects beyond the synthetic gas coin
427/// already accounted for by the estimation simulation. Mirrors the protocol's per-byte
428/// read cost, rounded up to the protocol gas rounding step in gas units before being
429/// converted to MIST.
430fn compute_gas_loading_cost_mist(
431    extra_coins: u64,
432    reference_gas_price: u64,
433    protocol_config: &ProtocolConfig,
434) -> u64 {
435    let units = extra_coins
436        .saturating_mul(GAS_COIN_SIZE_BYTES)
437        .saturating_mul(protocol_config.obj_access_cost_read_per_byte());
438    let rounded = if let Some(step) = protocol_config.gas_rounding_step_as_option() {
439        round_up_to_nearest(units, step)
440    } else {
441        units
442    };
443    rounded.saturating_mul(reference_gas_price)
444}
445
446/// Round up a value to the nearest multiple of `step` using saturating arithmetic.
447fn round_up_to_nearest(value: u64, step: u64) -> u64 {
448    let remainder = value % step;
449    if remainder == 0 {
450        value
451    } else {
452        value.saturating_add(step - remainder)
453    }
454}

456/// Storage cost (in MIST) of the synthetic gas coin that the simulator wrote during the
457/// estimation pass. The new `storage_rebate` on the written object equals the storage cost
458/// (see `track_storage_mutation` / `collect_storage_and_rebate`) so we can read it directly
459/// from the simulation's object set instead of re-deriving it from protocol parameters.
460/// Returns 0 when the simulation didn't use a mock gas coin.
461fn mock_gas_storage_cost(
462    simulation_result: &sui_types::transaction_executor::SimulateTransactionResult,
463) -> u64 {
464    let Some(mock_gas_id) = simulation_result.mock_gas_id else {
465        return 0;
466    };
467    // Both the input version (rebate 0, since the mock coin is fresh) and the written version
468    // (rebate set to the storage cost) end up in `objects`. The written version always carries
469    // the larger value, so taking the max is correct and avoids depending on iteration order.
470    simulation_result
471        .objects
472        .iter()
473        .filter(|o| o.id() == mock_gas_id)
474        .map(|o| o.storage_rebate)
475        .max()
476        .unwrap_or(0)
477}

479/// How many proposers a simulated transaction is restricted to.
480///
481/// Enough that the transaction stays submittable when some of them are offline, and no more than
482/// `MAX_UNPAID_ALLOWED_PROPOSERS` so that it remains valid at the reference gas price.
483const MAX_ALLOWED_PROPOSERS: usize = MAX_UNPAID_ALLOWED_PROPOSERS as usize;
484
485/// Populate an expiration covering the current epoch and the next one.
486///
487/// When this node can name the validators it would submit to, the expiration also restricts the
488/// transaction to them, so that nobody else can amplify it into consensus.
489fn configure_transaction_validity(
490    service: &RpcService,
491    protocol_config: &ProtocolConfig,
492    transaction: &mut sui_types::transaction::TransactionData,
493) -> Result<()> {
494    // Early return if the caller already chose an expiration with a validity window.
495    if matches!(
496        transaction.expiration(),
497        TransactionExpiration::ValidDuring { .. } | TransactionExpiration::Validity { .. }
498    ) {
499        return Ok(());
500    }
501
502    let current_epoch = service.reader.inner().get_latest_checkpoint()?.epoch();
503    let min_epoch = Some(current_epoch);
504    let max_epoch = Some(current_epoch.saturating_add(1));
505    let chain = service.chain_id;
506    let nonce = rand::random();
507
508    *transaction.expiration_mut() =
509        match select_allowed_proposers(service, protocol_config, current_epoch) {
510            Some(allowed_proposers) => TransactionExpiration::Validity {
511                min_epoch,
512                max_epoch,
513                min_timestamp: None,
514                max_timestamp: None,
515                chain,
516                nonce,
517                allowed_proposers: Some(allowed_proposers),
518            },
519            None => TransactionExpiration::ValidDuring {
520                min_epoch,
521                max_epoch,
522                min_timestamp: None,
523                max_timestamp: None,
524                chain,
525                nonce,
526            },
527        };
528    Ok(())
529}

531/// Restrict the transaction to the validators this node would submit to, even though its gas
532/// coins already provide replay protection.
533///
534/// Unlike the address-balance paths there is no window the transaction needs, so the expiration
535/// is only touched when the caller left it unset and a proposer set is actually available;
536/// otherwise the transaction is returned exactly as resolved.
537fn restrict_transaction_proposers(
538    service: &RpcService,
539    protocol_config: &ProtocolConfig,
540    transaction: &mut sui_types::transaction::TransactionData,
541) -> Result<()> {
542    if !matches!(transaction.expiration(), TransactionExpiration::None) {
543        return Ok(());
544    }
545
546    let current_epoch = service.reader.inner().get_latest_checkpoint()?.epoch();
547    let Some(allowed_proposers) = select_allowed_proposers(service, protocol_config, current_epoch)
548    else {
549        return Ok(());
550    };
551
552    *transaction.expiration_mut() = TransactionExpiration::Validity {
553        min_epoch: Some(current_epoch),
554        max_epoch: Some(current_epoch.saturating_add(1)),
555        min_timestamp: None,
556        max_timestamp: None,
557        chain: service.chain_id,
558        nonce: rand::random(),
559        allowed_proposers: Some(allowed_proposers),
560    };
561    Ok(())
562}

564/// The proposer set this node would restrict a transaction to, if one can be formed.
565///
566/// Only offered where the network accepts the `Validity` variant at all — otherwise
567/// validity_check would reject the very transaction simulate just handed back. `None` on nodes
568/// without a transaction driver, or before the driver has observed validator latencies.
569///
570/// Proposer sets are resolved against the committee of the epoch they name, so one selected now
571/// is only usable while this epoch lasts; a set naming any other epoch is not emitted, and the
572/// transaction simply stays unrestricted rather than becoming unproposable.
573fn select_allowed_proposers(
574    service: &RpcService,
575    protocol_config: &ProtocolConfig,
576    current_epoch: u64,
577) -> Option<AllowedProposers> {
578    if !protocol_config.allowed_proposers() {
579        return None;
580    }
581    service
582        .proposer_selector
583        .as_ref()
584        .and_then(|selector| selector.preferred_proposers(MAX_ALLOWED_PROPOSERS))
585        .filter(|allowed| allowed.epoch == current_epoch)
586}

588fn select_gas(
589    service: &RpcService,
590    transaction: &mut sui_types::transaction::TransactionData,
591    incremental_loading_rgp: Option<u64>,
592    protocol_config: &ProtocolConfig,
593) -> Result<()> {
594    use sui_types::accumulator_root::AccumulatorValue;
595    use sui_types::balance::Balance;
596    use sui_types::base_types::SequenceNumber;
597    use sui_types::coin_reservation::CoinReservationResolver;
598    use sui_types::coin_reservation::ParsedDigest;
599    use sui_types::coin_reservation::ParsedObjectRefWithdrawal;
600    use sui_types::gas_coin::GAS;
601    use sui_types::gas_coin::GasCoin;
602    use sui_types::transaction::Command;
603    use sui_types::transaction::TransactionDataAPI;
604
605    let reader = &service.reader;
606
607    let owner = transaction.gas_data().owner;
608    let budget = transaction.gas_data().budget;
609
610    let gas_coin_used = transaction
611        .kind()
612        .iter_commands()
613        .any(Command::is_gas_coin_used);
614    let address_balance = reader
615        .lookup_address_balance(owner, GAS::type_())
616        .map(|balance| {
617            // Sum up the explicit SUI reservations (excluding the implicit gas payment) for the
618            // `owner` so that we can deduct that from the available address balance. We use the
619            // estimation variant to avoid double-counting: the gas budget is what we're trying to
620            // satisfy, not a pre-existing reservation.
621            let coin_resolver = CoinReservationResolver::new(reader.inner().clone());
622
623            let reserved_sui = transaction
624                .process_funds_withdrawals_for_estimation(service.chain_id, &coin_resolver)
625                .ok()
626                .and_then(|withdrawals| {
627                    let sui_type = Balance::type_tag(GAS::type_tag());
628                    let sui_account_id = AccumulatorValue::get_field_id(owner, &sui_type).ok()?;
629                    withdrawals
630                        .get(&sui_account_id)
631                        .map(|(amount, _, _)| *amount)
632                })
633                .unwrap_or(0);
634
635            balance.saturating_sub(reserved_sui)
636        });
637
638    // If the gas coin isn't used and there is sufficient address balance budget to satisfy the
639    // required budget then we will use the `owner`s address balance to pay for gas. Otherwise we
640    // fallback to doing coin selection
641    let selected_gas_value = if !gas_coin_used
642        && let Some(address_balance) = address_balance
643        && address_balance >= budget
644    {
645        // We probably don't need to do this, but explicitly clear out the payment to force using
646        // Address balance
647        transaction.gas_data_mut().payment.clear();
648
649        if matches!(transaction.expiration(), TransactionExpiration::None) {
650            configure_transaction_validity(service, protocol_config, transaction)?;
651        }
652
653        budget
654    } else {
655        let input_objects = transaction
656            .input_objects()
657            .map_err(anyhow::Error::from)?
658            .iter()
659            .flat_map(|obj| match obj {
660                sui_types::transaction::InputObjectKind::ImmOrOwnedMoveObject((id, _, _)) => {
661                    Some(*id)
662                }
663                _ => None,
664            })
665            .collect_vec();
666
667        let gas_coins = reader
668            .inner()
669            .indexes()
670            .ok_or_else(RpcError::not_found)?
671            .owned_objects_iter(owner, Some(GasCoin::type_()), None)?
672            .filter_ok(|info| !input_objects.contains(&info.object_id))
673            .filter_map_ok(|info| reader.inner().get_object(&info.object_id))
674            // filter for objects which are not ConsensusAddress owned,
675            // since only Address owned can be used for gas payments today
676            .filter_ok(|object| !object.is_consensus())
677            .filter_map_ok(|object| {
678                GasCoin::try_from(&object)
679                    .ok()
680                    .map(|coin| (object.compute_object_reference(), coin.value()))
681            })
682            .take(protocol_config.max_gas_payment_objects() as usize);
683
684        let mut selected_gas = vec![];
685        let mut selected_gas_value = 0;
686
687        for maybe_coin in gas_coins {
688            let (object_ref, value) =
689                maybe_coin.map_err(|e| RpcError::new(tonic::Code::Internal, e.to_string()))?;
690            selected_gas.push(object_ref);
691            selected_gas_value += value;
692        }
693
694        // When GasCoin is used and there's address balance, prepend a coin reservation
695        // to make all SUI in the account available (coins + address balance)
696        if protocol_config.enable_coin_reservation_obj_refs()
697            && gas_coin_used
698            && let Some(ab_value) = address_balance
699            && ab_value > 0
700        {
701            let current_epoch = service.reader.inner().get_latest_checkpoint()?.epoch();
702
703            let accumulator_obj_id =
704                AccumulatorValue::get_field_id(owner, &Balance::type_tag(GAS::type_tag()))
705                    .map_err(|e| {
706                        RpcError::new(
707                            tonic::Code::Internal,
708                            format!("Failed to get accumulator object ID: {e}"),
709                        )
710                    })?;
711
712            let reservation = ParsedObjectRefWithdrawal::new(
713                *accumulator_obj_id.inner(),
714                current_epoch,
715                ab_value,
716            );
717            let coin_reservation = reservation.encode(SequenceNumber::new(), service.chain_id);
718
719            // Prepend coin reservation to make address balance accessible via GasCoin
720            selected_gas.insert(0, coin_reservation);
721            selected_gas_value += ab_value;
722
723            if matches!(transaction.expiration(), TransactionExpiration::None) {
724                configure_transaction_validity(service, protocol_config, transaction)?;
725            }
726        }
727
728        transaction.gas_data_mut().payment = selected_gas;
729
730        selected_gas_value
731    };
732
733    // When the caller asked us to top up the budget for the loading cost of the just-picked
734    // payment objects, do so before the balance check. The simulation already counted the
735    // synthetic gas coin's load, so charge for the rest. Coin reservations don't load real
736    // gas-coin objects and are excluded.
737    let final_budget = if let Some(rgp) = incremental_loading_rgp {
738        let real_coins = transaction
739            .gas_data()
740            .payment
741            .iter()
742            .filter(|obj_ref| !ParsedDigest::is_coin_reservation_digest(&obj_ref.2))
743            .count() as u64;
744        let extra_loading_mist =
745            compute_gas_loading_cost_mist(real_coins.saturating_sub(1), rgp, protocol_config);
746        let new_budget = budget
747            .saturating_add(extra_loading_mist)
748            .min(protocol_config.max_tx_gas());
749        transaction.gas_data_mut().budget = new_budget;
750        new_budget
751    } else {
752        budget
753    };
754
755    if selected_gas_value >= final_budget {
756        Ok(())
757    } else {
758        Err(RpcError::new(
759            tonic::Code::InvalidArgument,
760            format!(
761                "Unable to perform gas selection due to insufficient SUI \
762                balance (in address balance or coins) for account {owner} \
763                to satisfy required budget {final_budget}."
764            ),
765        ))
766    }
767}

769/// Returns true if the simulate request is eligible for auto gas_price=0 handling.
770///
771/// Requires: gasless enabled by protocol, caller did not set price (or set it to 0) and did not
772/// set gas payment objects, tx is a PTB that passes the structural gasless checks, and all loaded
773/// Move object inputs pass the runtime gasless input check (`Coin<T>` with `T` allowlisted,
774/// AddressOwner/ConsensusAddressOwner).
775fn is_gasless_candidate(
776    request: &SimulateTransactionRequest,
777    transaction: &sui_types::transaction::TransactionData,
778    protocol_config: &ProtocolConfig,
779    service: &RpcService,
780) -> Result<bool> {
781    if !protocol_config.enable_gasless() {
782        return Ok(false);
783    }
784    // When the caller passed a full BCS TransactionData, treat it as explicit — don't second-guess
785    // their gas choice. Only auto-switch in the unresolved/proto path.
786    if request.transaction().bcs_opt().is_some() {
787        return Ok(false);
788    }
789    // An explicit non-zero price opts out of gasless. price=0 is treated the same as unset: the
790    // caller is either asserting gasless intent or echoing the suggestion from a prior simulate.
791    if request
792        .transaction()
793        .gas_payment()
794        .price
795        .is_some_and(|p| p != 0)
796    {
797        return Ok(false);
798    }
799    if !request.transaction().gas_payment().objects.is_empty() {
800        return Ok(false);
801    }
802    let TransactionKind::ProgrammableTransaction(pt) = transaction.kind() else {
803        return Ok(false);
804    };
805    if pt.validate_gasless_transaction(protocol_config).is_err() {
806        return Ok(false);
807    }
808
809    // Load Move object inputs so we can run the runtime input check. Packages need not be loaded
810    // since check_gasless_object_inputs skips them.
811    let input_object_kinds = match transaction.input_objects() {
812        Ok(kinds) => kinds,
813        Err(_) => return Ok(false),
814    };
815    let mut loaded = Vec::with_capacity(input_object_kinds.len());
816    for kind in input_object_kinds {
817        match kind {
818            InputObjectKind::MovePackage(_) => continue,
819            InputObjectKind::ImmOrOwnedMoveObject(object_ref) => {
820                let Some(object) = service.reader.inner().get_object(&object_ref.0) else {
821                    return Ok(false);
822                };
823                loaded.push(ObjectReadResult::new(kind, object.into()));
824            }
825            InputObjectKind::SharedMoveObject { id, .. } => {
826                let Some(object) = service.reader.inner().get_object(&id) else {
827                    return Ok(false);
828                };
829                loaded.push(ObjectReadResult::new(kind, object.into()));
830            }
831        }
832    }
833    let input_objects = InputObjects::new(loaded);
834    Ok(
835        sui_transaction_checks::check_gasless_object_inputs(&input_objects, protocol_config)
836            .is_ok(),
837    )
838}

840/// The executor maps a post-execution gasless-requirements failure
841/// (`TemporaryStore::check_gasless_execution_requirements`) to
842/// `ExecutionErrorKind::InsufficientGas` on the effects (see
843/// `sui-execution/latest/sui-adapter/src/execution_engine.rs`). During a gasless simulate, that's
844/// the only way InsufficientGas can surface (gasless uses a large compute cap and ignores budget),
845/// so we treat it as the fallback trigger.
846fn is_gasless_post_execution_failure(status: &ExecutionStatus) -> bool {
847    matches!(
848        status,
849        ExecutionStatus::Failure(ExecutionFailure {
850            error: sui_types::execution_status::ExecutionErrorKind::InsufficientGas,
851            ..
852        })
853    )
854}
