39pub fn resolve_transaction(
40    service: &RpcService,
41    unresolved_transaction: &Transaction,
42    reference_gas_price: u64,
43    protocol_config: &ProtocolConfig,
44) -> Result<TransactionData> {
45    let sender = unresolved_transaction.sender().parse().map_err(|e| {
46        FieldViolation::new("transaction.sender")
47            .with_description(format!("invalid sender: {e}"))
48            .with_reason(ErrorReason::FieldInvalid)
49    })?;
50
51    let ptb = unresolved_transaction
52        .kind()
53        .programmable_transaction_opt()
54        .ok_or_else(|| {
55            FieldViolation::new("transaction.kind.programmable_transaction")
56                .with_reason(ErrorReason::FieldMissing)
57        })?;
58
59    let commands = ptb
60        .commands
61        .iter()
62        .map(sui_sdk_types::Command::try_from)
63        .collect::<Result<Vec<_>, _>>()
64        .map_err(|e| {
65            FieldViolation::new("commands")
66                .with_description(format!("invalid command: {e}"))
67                .with_reason(ErrorReason::FieldInvalid)
68        })?;
69
70    // Enforce the protocol's structural limits on the PTB before any
71    // per-input scan, package fetch, or normalization runs.
72    enforce_ptb_structural_limits(protocol_config, &ptb.inputs, &commands)?;
73
74    let mut called_packages = called_packages(service, protocol_config, &commands)?;
75    resolve_unresolved_transaction(
76        service,
77        &mut called_packages,
78        reference_gas_price,
79        protocol_config.max_tx_gas(),
80        sender,
81        &ptb.inputs,
82        commands,
83        unresolved_transaction.gas_payment.as_ref(),
84        unresolved_transaction.expiration.as_ref(),
85    )
86}

88/// Reject PTBs that exceed the protocol's structural limits before doing any
89/// resolution work. These match the bounds applied later by the
90/// `ProgrammableTransaction` and `Command` validity checks; we apply them
91/// upfront so the resolver itself cannot be driven into pathological work by
92/// an unauthenticated caller.
93fn enforce_ptb_structural_limits(
94    protocol_config: &ProtocolConfig,
95    inputs: &[sui_rpc::proto::sui::rpc::v2::Input],
96    commands: &[Command],
97) -> Result<()> {
98    let max_commands = protocol_config.max_programmable_tx_commands() as usize;
99    if commands.len() >= max_commands {
100        return Err(RpcError::new(
101            tonic::Code::InvalidArgument,
102            format!(
103                "programmable transaction has too many commands: {} (limit {})",
104                commands.len(),
105                max_commands
106            ),
107        ));
108    }
109
110    let max_inputs = protocol_config.max_input_objects() as usize;
111    if inputs.len() > max_inputs {
112        return Err(RpcError::new(
113            tonic::Code::InvalidArgument,
114            format!(
115                "programmable transaction has too many inputs: {} (limit {})",
116                inputs.len(),
117                max_inputs
118            ),
119        ));
120    }
121
122    let max_args = protocol_config.max_arguments() as usize;
123    for command in commands {
124        let arg_count = match command {
125            Command::MoveCall(call) => call.arguments.len(),
126            Command::TransferObjects(t) => t.objects.len(),
127            Command::SplitCoins(s) => s.amounts.len(),
128            Command::MergeCoins(m) => m.coins_to_merge.len(),
129            Command::MakeMoveVector(v) => v.elements.len(),
130            Command::Publish(_) | Command::Upgrade(_) => 0,
131            _ => 0,
132        };
133        if arg_count >= max_args {
134            return Err(RpcError::new(
135                tonic::Code::InvalidArgument,
136                format!(
137                    "programmable transaction command has too many arguments: \
138                    {} (limit {})",
139                    arg_count, max_args
140                ),
141            ));
142        }
143    }
144
145    Ok(())
146}

153struct NormalizedPackage {
154    #[allow(unused)]
155    package: MovePackage,
156    normalized_modules: BTreeMap<String, normalized::Module<normalized::RcIdentifier>>,
157}

159pub(super) fn called_packages(
160    service: &RpcService,
161    protocol_config: &ProtocolConfig,
162    commands: &[Command],
163) -> Result<NormalizedPackages> {
164    let binary_config = protocol_config.binary_config(None);
165    let mut pool = normalized::RcPool::new();
166    let mut packages = HashMap::new();
167
168    for move_call in commands.iter().filter_map(|command| {
169        if let Command::MoveCall(move_call) = command {
170            Some(move_call)
171        } else {
172            None
173        }
174    }) {
175        // Skip packages already normalized for this request. The same package
176        // can appear in many commands (e.g. the framework `0x2`), and
177        // re-fetching it then redeserializing every module via
178        // `CompiledModule::deserialize_with_config` is unnecessary work.
179        if packages.contains_key(&move_call.package) {
180            continue;
181        }
182
183        let package = service
184            .reader
185            .inner()
186            .get_object(&(move_call.package.into()))
187            .ok_or_else(|| ObjectNotFoundError::new(move_call.package))?
188            .data
189            .try_as_package()
190            .ok_or_else(|| {
191                RpcError::new(
192                    tonic::Code::InvalidArgument,
193                    format!("object {} is not a package", move_call.package),
194                )
195            })?
196            .to_owned();
197
198        // Normalization doesn't take the linkage or type origin tables into account, which means
199        // that if you have an upgraded package that introduces a new type, then that type's
200        // package ID is going to appear incorrectly if you fetch it from its normalized module.
201        //
202        // Despite the above this is safe given we are only using the signature information (and in
203        // particular the reference kind) from the normalized package.
204        let normalized_modules = package
205            .normalize(&mut pool, &binary_config, /* include code */ true)
206            .map_err(|e| {
207                RpcError::new(
208                    tonic::Code::Internal,
209                    format!("unable to normalize package {}: {e}", move_call.package),
210                )
211            })?;
212        let package = NormalizedPackage {
213            package,
214            normalized_modules,
215        };
216
217        packages.insert(move_call.package, package);
218    }
219
220    Ok(NormalizedPackages { pool, packages })
221}

223fn resolve_unresolved_transaction(
224    service: &RpcService,
225    called_packages: &mut NormalizedPackages,
226    reference_gas_price: u64,
227    max_gas_budget: u64,
228    sender: sui_sdk_types::Address,
229    unresolved_inputs: &[sui_rpc::proto::sui::rpc::v2::Input],
230    commands: Vec<Command>,
231    gas_payment: Option<&sui_rpc::proto::sui::rpc::v2::GasPayment>,
232    expiration: Option<&sui_rpc::proto::sui::rpc::v2::TransactionExpiration>,
233) -> Result<TransactionData> {
234    let gas_data = if let Some(unresolved_gas_payment) = gas_payment {
235        let gas_coins = unresolved_gas_payment
236            .objects
237            .iter()
238            .map(|unresolved| resolve_gas_object_reference(service, unresolved.try_into()?))
239            .collect::<Result<Vec<_>>>()?;
240        let payment = gas_coins.iter().map(|(r, _)| *r).collect::<Vec<_>>();
241        let max_gas_budget = if payment.is_empty() {
242            max_gas_budget
243        } else {
244            gas_coins
245                .iter()
246                .map(|(_, value)| *value)
247                .sum::<u64>()
248                .min(max_gas_budget)
249        };
250        GasData {
251            payment,
252            owner: unresolved_gas_payment.owner().parse().map_err(|e| {
253                FieldViolation::new("owner")
254                    .with_description(format!("invalid owner: {e}"))
255                    .with_reason(ErrorReason::FieldInvalid)
256            })?,
257            price: unresolved_gas_payment.price.unwrap_or(reference_gas_price),
258            budget: unresolved_gas_payment.budget.unwrap_or(max_gas_budget),
259        }
260    } else {
261        GasData {
262            payment: vec![],
263            owner: sender.into(),
264            price: reference_gas_price,
265            budget: max_gas_budget,
266        }
267    };
268
269    let expiration = expiration
270        .map(sui_types::transaction::TransactionExpiration::try_from)
271        .transpose()
272        .map_err(|e| {
273            FieldViolation::new("expiration")
274                .with_description(e)
275                .with_reason(ErrorReason::FieldInvalid)
276        })?
277        .unwrap_or(sui_types::transaction::TransactionExpiration::None);
278    let ptb = resolve_ptb(service, called_packages, unresolved_inputs, commands)?;
279    Ok(TransactionData::V1(
280        sui_types::transaction::TransactionDataV1 {
281            kind: sui_types::transaction::TransactionKind::ProgrammableTransaction(ptb),
282            sender: sender.into(),
283            gas_data,
284            expiration,
285        },
286    ))
287}

289/// If the unresolved reference has a digest that matches the coin reservation
290/// magic, parse it into a `ParsedObjectRefWithdrawal`. Coin reservation
291/// ObjectRefs encode an address balance reservation and don't correspond to
292/// real objects in storage, so callers must pass them through without a
293/// storage lookup.
294///
295/// Returns the parsed withdrawal together with the version from the
296/// unresolved reference (which `ParsedObjectRefWithdrawal` does not store).
297fn try_parse_coin_reservation(
298    unresolved: &UnresolvedObjectReference,
299    service: &RpcService,
300) -> Option<(
301    ParsedObjectRefWithdrawal,
302    sui_types::base_types::SequenceNumber,
303)> {
304    use sui_types::coin_reservation::ParsedDigest;
305
306    let digest = unresolved.digest?;
307    let object_digest = sui_types::digests::ObjectDigest::new(*digest.inner());
308    if !ParsedDigest::is_coin_reservation_digest(&object_digest) {
309        return None;
310    }
311
312    let object_id: sui_types::base_types::ObjectID = unresolved.object_id.into();
313    let version = sui_types::base_types::SequenceNumber::from_u64(unresolved.version.unwrap_or(0));
314    let obj_ref = (object_id, version, object_digest);
315    let parsed = ParsedObjectRefWithdrawal::parse(&obj_ref, service.chain_id)?;
316    Some((parsed, version))
317}

319fn resolve_gas_object_reference(
320    service: &RpcService,
321    unresolved_object_reference: UnresolvedObjectReference,
322) -> Result<(ObjectRef, u64)> {
323    // Coin reservation ObjectRefs don't exist in storage; pass them through
324    // as-is when the digest identifies one.
325    if let Some((parsed, version)) =
326        try_parse_coin_reservation(&unresolved_object_reference, service)
327    {
328        return Ok((
329            parsed.encode(version, service.chain_id),
330            parsed.reservation_amount(),
331        ));
332    }
333
334    let object = service
335        .reader
336        .inner()
337        .get_object(&(unresolved_object_reference.object_id.into()))
338        .ok_or_else(|| ObjectNotFoundError::new(unresolved_object_reference.object_id))?;
339
340    let Ok(gas_coin) = sui_types::gas_coin::GasCoin::try_from(&object) else {
341        return Err(FieldViolation::new("payment")
342            .with_description(format!("object {} is not a valid gas coin", object.id()))
343            .with_reason(ErrorReason::FieldInvalid)
344            .into());
345    };
346
347    resolve_object_reference_with_object(&object, unresolved_object_reference)
348        .map(|r| (r, gas_coin.value()))
349}

351fn resolve_object_reference(
352    service: &RpcService,
353    unresolved_object_reference: UnresolvedObjectReference,
354) -> Result<ObjectRef> {
355    // Coin reservation ObjectRefs don't exist in storage; pass them through
356    // as-is when the digest identifies one.
357    if let Some((parsed, version)) =
358        try_parse_coin_reservation(&unresolved_object_reference, service)
359    {
360        return Ok(parsed.encode(version, service.chain_id));
361    }
362
363    let object = service
364        .reader
365        .inner()
366        .get_object(&(unresolved_object_reference.object_id.into()))
367        .ok_or_else(|| ObjectNotFoundError::new(unresolved_object_reference.object_id))?;
368    resolve_object_reference_with_object(&object, unresolved_object_reference)
369}

371// Resolve an object reference against the provided object.
372//
373// Callers should check that the object_id matches the id in the `unresolved_object_reference`
374// before calling.
375fn resolve_object_reference_with_object(
376    object: &sui_types::object::Object,
377    unresolved_object_reference: UnresolvedObjectReference,
378) -> Result<ObjectRef> {
379    let UnresolvedObjectReference {
380        object_id,
381        version,
382        digest,
383    } = unresolved_object_reference;
384
385    match object.owner() {
386        sui_types::object::Owner::AddressOwner(_) | sui_types::object::Owner::Immutable => {}
387        _ => {
388            return Err(RpcError::new(
389                tonic::Code::InvalidArgument,
390                format!("object {object_id} is not Immutable or AddressOwned"),
391            ));
392        }
393    }
394
395    let id = object.id();
396    let v = object.version();
397    let d = object.digest();
398
399    // This really should be an assert
400    if object_id.inner() != &id.into_bytes() {
401        return Err(RpcError::new(
402            tonic::Code::Internal,
403            "provided object and object_id should match",
404        ));
405    }
406
407    if let Some(version) = version.filter(|version| *version != v.value()) {
408        return Err(RpcError::new(
409            tonic::Code::InvalidArgument,
410            format!(
411                "provided version doesn't match for object {id}, provided: {version} actual: {v}"
412            ),
413        ));
414    }
415
416    if let Some(digest) = digest.filter(|digest| digest.inner() != d.inner()) {
417        return Err(RpcError::new(
418            tonic::Code::InvalidArgument,
419            format!(
420                "provided digest doesn't match for object {id}, provided: {digest} actual: {d}"
421            ),
422        ));
423    }
424
425    Ok((id, v, d))
426}

428pub(super) fn resolve_ptb(
429    service: &RpcService,
430    called_packages: &mut NormalizedPackages,
431    unresolved_inputs: &[sui_rpc::proto::sui::rpc::v2::Input],
432    commands: Vec<Command>,
433) -> Result<ProgrammableTransaction> {
434    // Precompute uses of every input argument across all commands once, so that
435    // per-input resolution is linear in the number of uses rather than scanning
436    // every command and every argument for each input. Without this, the
437    // resolver does O(inputs * commands * args/cmd) work.
438    let arg_uses = ArgUses::build(unresolved_inputs.len(), &commands);
439    let inputs = unresolved_inputs
440        .iter()
441        .enumerate()
442        .map(|(arg_idx, arg)| resolve_arg(service, called_packages, &arg_uses, arg, arg_idx))
443        .collect::<Result<_>>()?;
444
445    ProgrammableTransaction {
446        inputs,
447        commands: commands.into_iter().map(Into::into).collect(),
448    }
449    .pipe(Ok)
450}

452fn resolve_arg(
453    service: &RpcService,
454    called_packages: &mut NormalizedPackages,
455    arg_uses: &ArgUses,
456    arg: &sui_rpc::proto::sui::rpc::v2::Input,
457    arg_idx: usize,
458) -> Result<CallArg> {
459    use sui_rpc::proto::sui::rpc::v2::input::InputKind;
460
461    match UnresolvedInput::from_proto(arg)? {
462        // Pure, already prepared BCS input
463        UnresolvedInput {
464            kind: Some(InputKind::Pure),
465            pure: Some(pure),
466            object_id: None,
467            version: None,
468            digest: None,
469            mutable: None,
470            funds_withdrawal: None,
471            literal: None,
472        }
473        | UnresolvedInput {
474            kind: None,
475            pure: Some(pure),
476            object_id: None,
477            version: None,
478            digest: None,
479            mutable: None,
480            funds_withdrawal: None,
481            literal: None,
482        } => CallArg::Pure(pure.to_vec()),
483
484        // Immutable or owned
485        UnresolvedInput {
486            kind: Some(InputKind::ImmutableOrOwned),
487            pure: None,
488            object_id: Some(object_id),
489            version,
490            digest,
491            mutable: None,
492            funds_withdrawal: None,
493            literal: None,
494        } => CallArg::Object(ObjectArg::ImmOrOwnedObject(resolve_object_reference(
495            service,
496            UnresolvedObjectReference {
497                object_id,
498                version,
499                digest,
500            },
501        )?)),
502
503        // Shared object
504        UnresolvedInput {
505            kind: Some(InputKind::Shared),
506            pure: None,
507            object_id: Some(object_id),
508            version: _,
509            digest: None,
510            mutable: _,
511            funds_withdrawal: None,
512            literal: None,
513        } => CallArg::Object(resolve_shared_input(
514            service,
515            called_packages,
516            arg_uses,
517            arg_idx,
518            object_id,
519            stated_shared_mutability(arg),
520        )?),
521
522        // Receiving
523        UnresolvedInput {
524            kind: Some(InputKind::Receiving),
525            pure: None,
526            object_id: Some(object_id),
527            version,
528            digest,
529            mutable: None,
530            funds_withdrawal: None,
531            literal: None,
532        } => CallArg::Object(ObjectArg::Receiving(resolve_object_reference(
533            service,
534            UnresolvedObjectReference {
535                object_id,
536                version,
537                digest,
538            },
539        )?)),
540
541        // Object, could be Immutable, Owned, Shared, or Receiving
542        UnresolvedInput {
543            kind: None,
544            pure: None,
545            object_id: Some(object_id),
546            version,
547            digest,
548            mutable,
549            funds_withdrawal: None,
550            literal: None,
551        } => CallArg::Object(resolve_object(
552            service,
553            called_packages,
554            arg_uses,
555            arg_idx,
556            object_id,
557            version,
558            digest,
559            mutable,
560        )?),
561
562        // FundsWithdrawal
563        UnresolvedInput {
564            kind: Some(InputKind::FundsWithdrawal),
565            pure: None,
566            object_id: None,
567            version: None,
568            digest: None,
569            mutable: None,
570            funds_withdrawal: Some(w),
571            literal: None,
572        }
573        | UnresolvedInput {
574            kind: None,
575            pure: None,
576            object_id: None,
577            version: None,
578            digest: None,
579            mutable: None,
580            funds_withdrawal: Some(w),
581            literal: None,
582        } => CallArg::FundsWithdrawal(w),
583
584        // Literal, unresolved pure argument
585        UnresolvedInput {
586            kind: None, // TODO should we have a kind?
587            pure: None,
588            object_id: None,
589            version: None,
590            digest: None,
591            mutable: None,
592            funds_withdrawal: None,
593            literal: Some(literal),
594        } => CallArg::Pure(literal::resolve_literal(
595            called_packages,
596            arg_uses,
597            arg_idx,
598            literal,
599        )?),
600
601        _ => {
602            return Err(RpcError::new(
603                tonic::Code::InvalidArgument,
604                "invalid unresolved input argument",
605            ));
606        }
607    }
608    .pipe(Ok)
609}

611fn resolve_object(
612    service: &RpcService,
613    called_packages: &NormalizedPackages,
614    arg_uses: &ArgUses,
615    arg_idx: usize,
616    object_id: Address,
617    version: Option<sui_sdk_types::Version>,
618    digest: Option<sui_sdk_types::Digest>,
619    _mutable: Option<bool>,
620) -> Result<ObjectArg> {
621    // Coin reservation ObjectRefs don't exist in storage; pass them through
622    // as-is when the digest identifies one.
623    let unresolved = UnresolvedObjectReference {
624        object_id,
625        version,
626        digest,
627    };
628    if let Some((parsed, ver)) = try_parse_coin_reservation(&unresolved, service) {
629        return Ok(ObjectArg::ImmOrOwnedObject(
630            parsed.encode(ver, service.chain_id),
631        ));
632    }
633
634    let id = object_id.into();
635    let object = service
636        .reader
637        .inner()
638        .get_object(&id)
639        .ok_or_else(|| ObjectNotFoundError::new(object_id))?;
640
641    match object.owner() {
642        sui_types::object::Owner::Immutable => resolve_object_reference_with_object(
643            &object,
644            UnresolvedObjectReference {
645                object_id,
646                version,
647                digest,
648            },
649        )
650        .map(ObjectArg::ImmOrOwnedObject),
651
652        sui_types::object::Owner::AddressOwner(_) => {
653            let object_ref = resolve_object_reference_with_object(
654                &object,
655                UnresolvedObjectReference {
656                    object_id,
657                    version,
658                    digest,
659                },
660            )?;
661
662            if is_input_argument_receiving(called_packages, arg_uses, arg_idx)? {
663                ObjectArg::Receiving(object_ref)
664            } else {
665                ObjectArg::ImmOrOwnedObject(object_ref)
666            }
667            .pipe(Ok)
668        }
669        sui_types::object::Owner::Shared { .. }
670        | sui_types::object::Owner::ConsensusAddressOwner { .. }
671        | sui_types::object::Owner::Party { .. } => {
672            resolve_shared_input_with_object(called_packages, arg_uses, arg_idx, object, None)
673        }
674        sui_types::object::Owner::ObjectOwner(_) => Err(RpcError::new(
675            tonic::Code::InvalidArgument,
676            format!("object {object_id} is object owned and cannot be used as an input"),
677        )),
678    }
679}

681fn stated_shared_mutability(
682    input: &sui_rpc::proto::sui::rpc::v2::Input,
683) -> Option<SharedObjectMutability> {
684    match input.mutability() {
685        Mutability::Immutable => Some(SharedObjectMutability::Immutable),
686        Mutability::Mutable => Some(SharedObjectMutability::Mutable),
687        Mutability::NonExclusiveWrite => Some(SharedObjectMutability::NonExclusiveWrite),
688        _ => input.mutable.map(mutability_from_bool),
689    }
690}
691
692fn mutability_from_bool(mutable: bool) -> SharedObjectMutability {
693    if mutable {
694        SharedObjectMutability::Mutable
695    } else {
696        SharedObjectMutability::Immutable
697    }
698}

700fn resolve_shared_input(
701    service: &RpcService,
702    called_packages: &NormalizedPackages,
703    arg_uses: &ArgUses,
704    arg_idx: usize,
705    object_id: Address,
706    stated_mutability: Option<SharedObjectMutability>,
707) -> Result<ObjectArg> {
708    let id = object_id.into();
709    let object = service
710        .reader
711        .inner()
712        .get_object(&id)
713        .ok_or_else(|| ObjectNotFoundError::new(object_id))?;
714    resolve_shared_input_with_object(
715        called_packages,
716        arg_uses,
717        arg_idx,
718        object,
719        stated_mutability,
720    )
721}

723// Checks if the provided input argument is used as a receiving object
724fn is_input_argument_receiving(
725    called_packages: &NormalizedPackages,
726    arg_uses: &ArgUses,
727    arg_idx: usize,
728) -> Result<bool> {
729    let (receiving_package, receiving_module, receiving_struct) =
730        sui_types::transfer::RESOLVED_RECEIVING_STRUCT;
731
732    let mut receiving = false;
733    for (command, idx) in arg_uses.uses_of(arg_idx) {
734        if let (Command::MoveCall(move_call), Some(idx)) = (command, idx) {
735            let arg_type = arg_type_of_move_call_input(called_packages, move_call, idx)?;
736
737            let inner_type = match &*arg_type {
738                normalized::Type::Reference(_, inner) => inner,
739                _ => &*arg_type,
740            };
741
742            if let normalized::Type::Datatype(dt) = inner_type
743                && receiving_package == &dt.module.address
744                && receiving_module == dt.module.name.as_ref()
745                && receiving_struct == dt.name.as_ref()
746            {
747                receiving = true;
748            }
749        }
750
751        //XXX do we want to ensure its only used once as receiving?
752        if receiving {
753            break;
754        }
755    }
756
757    Ok(receiving)
758}

760// TODO still need to handle the case where a function parameter is a generic parameter and the
761// real type needs to be lookedup from the provided type args in the MoveCall itself
762fn arg_type_of_move_call_input(
763    called_packages: &NormalizedPackages,
764    move_call: &sui_sdk_types::MoveCall,
765    idx: usize,
766) -> Result<Rc<normalized::Type<normalized::RcIdentifier>>> {
767    let function = called_packages
768        .packages
769        // Find the package
770        .get(&move_call.package)
771        // Find the module
772        .and_then(|package| package.normalized_modules.get(move_call.module.as_str()))
773        // Find the function
774        .and_then(|module| module.functions.get(move_call.function.as_str()))
775        .ok_or_else(|| {
776            RpcError::new(
777                tonic::Code::InvalidArgument,
778                format!(
779                    "unable to find function {package}::{module}::{function}",
780                    package = move_call.package,
781                    module = move_call.module,
782                    function = move_call.function
783                ),
784            )
785        })?;
786    let Some(ty) = function.parameters.get(idx) else {
787        return Err(RpcError::new(
788            tonic::Code::InvalidArgument,
789            "invalid input parameter",
790        ));
791    };
792    Ok(ty.clone())
793}

795fn resolve_shared_input_with_object(
796    called_packages: &NormalizedPackages,
797    arg_uses: &ArgUses,
798    arg_idx: usize,
799    object: sui_types::object::Object,
800    stated_mutability: Option<SharedObjectMutability>,
801) -> Result<ObjectArg> {
802    let object_id = object.id();
803    let initial_shared_version = if let sui_types::object::Owner::Shared {
804        initial_shared_version,
805    }
806    | sui_types::object::Owner::ConsensusAddressOwner {
807        start_version: initial_shared_version,
808        ..
809    } = object.owner()
810    {
811        *initial_shared_version
812    } else {
813        return Err(RpcError::new(
814            tonic::Code::InvalidArgument,
815            format!("object {object_id} is not a shared or consensus object"),
816        ));
817    };
818
819    let mutability = match stated_mutability {
820        Some(mutability) => mutability,
821        None => infer_shared_mutability(called_packages, arg_uses, arg_idx)?,
822    };
823
824    Ok(ObjectArg::SharedObject {
825        id: object_id,
826        initial_shared_version,
827        mutability,
828    })
829}
830
831fn infer_shared_mutability(
832    called_packages: &NormalizedPackages,
833    arg_uses: &ArgUses,
834    arg_idx: usize,
835) -> Result<SharedObjectMutability> {
836    let mut mutable = false;
837    for (command, idx) in arg_uses.uses_of(arg_idx) {
838        match (command, idx) {
839            (Command::MoveCall(move_call), Some(idx)) => {
840                let arg_type = arg_type_of_move_call_input(called_packages, move_call, idx)?;
841                if matches!(
842                    &*arg_type,
843                    normalized::Type::Reference(/* mut */ true, _)
844                        | normalized::Type::Datatype(_)
845                        | normalized::Type::TypeParameter(_)
846                ) {
847                    mutable = true;
848                }
849            }
850            (Command::SplitCoins(_) | Command::MergeCoins(_) | Command::MakeMoveVector(_), _) => {
851                mutable = true;
852            }
853            // `None` is the recipient address.
854            (Command::TransferObjects(_), Some(_)) => {
855                mutable = true;
856            }
857            _ => {}
858        }
859        // Early break out of the loop if we've already determined that the shared object
860        // is needed to be mutable
861        if mutable {
862            break;
863        }
864    }
865
866    Ok(mutability_from_bool(mutable))
867}

869/// Precomputed map from input argument index to the commands that consume it.
870///
871/// Building this once is `O(commands * args_per_command)`. Looking up uses for a
872/// single input is then `O(uses_of_input)`, so the resolver as a whole is linear
873/// in the total number of arguments rather than quadratic.
874pub(super) struct ArgUses<'a> {
875    commands: &'a [Command],
876    /// `uses[input_idx]` is the list of `(command_idx, position)` pairs where
877    /// `Argument::Input(input_idx)` first appears within that command. The
878    /// `position` is `Some(i)` for argument lists (e.g. `MoveCall` parameters,
879    /// `TransferObjects` objects) and `None` for the "primary" slot of commands
880    /// that have one (e.g. `TransferObjects::address`, `SplitCoins::coin`).
881    /// Matches the prior `find_arg_uses` behaviour: at most one entry per
882    /// command, recording the first matching position.
883    uses: Vec<Vec<(usize, Option<usize>)>>,
884}
885
886impl<'a> ArgUses<'a> {
887    pub(super) fn build(num_inputs: usize, commands: &'a [Command]) -> Self {
888        let mut uses: Vec<Vec<(usize, Option<usize>)>> = vec![Vec::new(); num_inputs];
889
890        for (cmd_idx, command) in commands.iter().enumerate() {
891            // Track the first matching position per input within this command,
892            // matching the prior `find_arg_uses` semantics that returned only
893            // one entry per command.
894            let mut first_pos: BTreeMap<u16, Option<usize>> = BTreeMap::new();
895            let mut record = |arg: &Argument, pos: Option<usize>| {
896                if let Argument::Input(idx) = arg {
897                    first_pos.entry(*idx).or_insert(pos);
898                }
899            };
900
901            match command {
902                Command::MoveCall(move_call) => {
903                    for (i, arg) in move_call.arguments.iter().enumerate() {
904                        record(arg, Some(i));
905                    }
906                }
907                Command::TransferObjects(transfer_objects) => {
908                    record(&transfer_objects.address, None);
909                    for (i, arg) in transfer_objects.objects.iter().enumerate() {
910                        record(arg, Some(i));
911                    }
912                }
913                Command::SplitCoins(split_coins) => {
914                    record(&split_coins.coin, None);
915                    for (i, arg) in split_coins.amounts.iter().enumerate() {
916                        record(arg, Some(i));
917                    }
918                }
919                Command::MergeCoins(merge_coins) => {
920                    record(&merge_coins.coin, None);
921                    for (i, arg) in merge_coins.coins_to_merge.iter().enumerate() {
922                        record(arg, Some(i));
923                    }
924                }
925                Command::MakeMoveVector(make_move_vector) => {
926                    for (i, arg) in make_move_vector.elements.iter().enumerate() {
927                        record(arg, Some(i));
928                    }
929                }
930                Command::Upgrade(upgrade) => {
931                    record(&upgrade.ticket, None);
932                }
933                Command::Publish(_) => {}
934                _ => {}
935            }
936
937            for (input_idx, pos) in first_pos {
938                if let Some(slot) = uses.get_mut(input_idx as usize) {
939                    slot.push((cmd_idx, pos));
940                }
941            }
942        }
943
944        Self { commands, uses }
945    }
946
947    pub(super) fn uses_of(
948        &self,
949        arg_idx: usize,
950    ) -> impl Iterator<Item = (&Command, Option<usize>)> {
951        self.uses
952            .get(arg_idx)
953            .into_iter()
954            .flat_map(|entries| entries.iter())
955            .map(|(cmd_idx, pos)| (&self.commands[*cmd_idx], *pos))
956    }
957}

959struct UnresolvedObjectReference {
960    object_id: Address,
961    version: Option<sui_sdk_types::Version>,
962    digest: Option<sui_sdk_types::Digest>,
963}
964
965impl TryFrom<&sui_rpc::proto::sui::rpc::v2::ObjectReference> for UnresolvedObjectReference {
966    type Error = FieldViolation;
967
968    fn try_from(
969        value: &sui_rpc::proto::sui::rpc::v2::ObjectReference,
970    ) -> Result<Self, Self::Error> {
971        let object_id = value.object_id().parse().map_err(|e| {
972            FieldViolation::new("object_id")
973                .with_description(format!("invalid object_id: {e}"))
974                .with_reason(ErrorReason::FieldInvalid)
975        })?;
976        let version = value.version;
977        let digest = value
978            .digest
979            .as_ref()
980            .map(|digest| digest.parse())
981            .transpose()
982            .map_err(|e| {
983                FieldViolation::new("digest")
984                    .with_description(format!("invalid digest: {e}"))
985                    .with_reason(ErrorReason::FieldInvalid)
986            })?;
987
988        Ok(Self {
989            object_id,
990            version,
991            digest,
992        })
993    }
994}

996struct UnresolvedInput<'a> {
997    pub kind: Option<sui_rpc::proto::sui::rpc::v2::input::InputKind>,
998    pub pure: Option<&'a Bytes>,
999    pub object_id: Option<sui_sdk_types::Address>,
1000    pub version: Option<sui_sdk_types::Version>,
1001    pub digest: Option<sui_sdk_types::Digest>,
1002    pub mutable: Option<bool>,
1003    pub funds_withdrawal: Option<FundsWithdrawalArg>,
1004    pub literal: Option<&'a prost_types::Value>,
1005}
1006
1007impl<'a> UnresolvedInput<'a> {
1008    fn from_proto(input: &'a sui_rpc::proto::sui::rpc::v2::Input) -> Result<Self, FieldViolation> {
1009        Ok(Self {
1010            kind: input.kind.map(|_| input.kind()),
1011            literal: input.literal.as_deref(),
1012            pure: input.pure.as_ref(),
1013            object_id: input
1014                .object_id
1015                .as_ref()
1016                .map(|id| {
1017                    id.parse().map_err(|e| {
1018                        FieldViolation::new("object_id")
1019                            .with_description(format!("invalid object_id: {e}"))
1020                            .with_reason(ErrorReason::FieldInvalid)
1021                    })
1022                })
1023                .transpose()?,
1024
1025            version: input.version,
1026            digest: input
1027                .digest
1028                .as_ref()
1029                .map(|digest| {
1030                    digest.parse().map_err(|e| {
1031                        FieldViolation::new("digest")
1032                            .with_description(format!("invalid digest: {e}"))
1033                            .with_reason(ErrorReason::FieldInvalid)
1034                    })
1035                })
1036                .transpose()?,
1037            funds_withdrawal: input
1038                .funds_withdrawal_opt()
1039                .map(|w| {
1040                    Ok(FundsWithdrawalArg {
1041                        reservation: Reservation::MaxAmountU64(w.amount.ok_or_else(|| {
1042                            FieldViolation::new("amount").with_reason(ErrorReason::FieldMissing)
1043                        })?),
1044                        type_arg: WithdrawalTypeArg::Balance(
1045                            w.coin_type().parse::<sui_types::TypeTag>().map_err(|e| {
1046                                FieldViolation::new("coin_type")
1047                                    .with_description(format!("invalid coin_type: {e}"))
1048                                    .with_reason(ErrorReason::FieldInvalid)
1049                            })?,
1050                        ),
1051                        withdraw_from: match w.source() {
1052                            sui_rpc::proto::sui::rpc::v2::funds_withdrawal::Source::Sender => {
1053                                WithdrawFrom::Sender
1054                            }
1055                            sui_rpc::proto::sui::rpc::v2::funds_withdrawal::Source::Sponsor => {
1056                                WithdrawFrom::Sponsor
1057                            }
1058                            sui_rpc::proto::sui::rpc::v2::funds_withdrawal::Source::SenderAllowance => {
1059                                WithdrawFrom::SenderAllowance {
1060                                    funder: w.funder().parse().map_err(|e| {
1061                                        FieldViolation::new("funder")
1062                                            .with_description(format!("invalid funder: {e}"))
1063                                            .with_reason(ErrorReason::FieldInvalid)
1064                                    })?,
1065                                    allowance: w.allowance().parse().map_err(|e| {
1066                                        FieldViolation::new("allowance")
1067                                            .with_description(format!("invalid allowance: {e}"))
1068                                            .with_reason(ErrorReason::FieldInvalid)
1069                                    })?,
1070                                }
1071                            }
1072                            sui_rpc::proto::sui::rpc::v2::funds_withdrawal::Source::FrozenAddress => {
1073                                WithdrawFrom::FrozenAddress {
1074                                    seized_from: w.seized_from().parse().map_err(|e| {
1075                                        FieldViolation::new("seized_from")
1076                                            .with_description(format!("invalid seized_from: {e}"))
1077                                            .with_reason(ErrorReason::FieldInvalid)
1078                                    })?,
1079                                    seize_auth: w.seize_auth().parse().map_err(|e| {
1080                                        FieldViolation::new("seize_auth")
1081                                            .with_description(format!("invalid seize_auth: {e}"))
1082                                            .with_reason(ErrorReason::FieldInvalid)
1083                                    })?,
1084                                }
1085                            }
1086                            _ => WithdrawFrom::Sender,
1087                        },
1088                    })
1089                })
1090                .transpose()?,
1091            mutable: input.mutable,
1092        })
1093    }
