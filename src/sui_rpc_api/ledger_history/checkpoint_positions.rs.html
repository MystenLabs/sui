4//! Start and end positions inside the checkpoints at the edges of a `ListTransactions` /
5//! `ListEvents` request's checkpoint window.
6//!
7//! A position is a gap between items, named by checkpoint-relative offsets: `(checkpoint,
8//! transaction_index[, event_index])` is the gap just before that transaction (or that event of
9//! it). The start position `(start_checkpoint, start_transaction_index[, start_event_index])` is
10//! the first position a scan admits. The end position is exclusive and lies in the window's last
11//! checkpoint, `(end_checkpoint - 1, end_transaction_index[, end_event_index])`, so a server that
12//! ignores it returns a superset. Offsets past the end of their checkpoint (or transaction) clamp
13//! to its end instead of failing: the gap after a checkpoint's last item is `(C, n)` for its
14//! transaction count `n`, which clients cannot know.
15

27/// A request's start and end positions within its checkpoint window's edge checkpoints. An unset
28/// position leaves its edge at the checkpoint boundary.
29#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
30pub struct CheckpointPositions {
31    /// Where the scan starts in `start_checkpoint` (inclusive).
32    start: Option<PositionInCheckpoint>,
33    /// Where the scan ends in `end_checkpoint - 1` (exclusive).
34    end: Option<PositionInCheckpoint>,
35    /// The request's `end_checkpoint`, set whenever `end` is.
36    end_checkpoint: Option<u64>,
37}
38
39/// A gap between items inside one checkpoint: just before the event at `event_index` of the
40/// transaction at `transaction_index`. Event index 0 is the gap before the whole transaction.
41#[derive(Clone, Copy, Debug, Eq, PartialEq)]
42struct PositionInCheckpoint {
43    transaction_index: u64,
44    event_index: u32,
45}
46
47impl CheckpointPositions {
48    pub fn from_transactions_request(request: &ListTransactionsRequest) -> Result<Self, RpcError> {
49        type Request = ListTransactionsRequest;
50        require_field_when_set(
51            request.start_checkpoint,
52            Request::START_CHECKPOINT_FIELD.name,
53            request.start_transaction_index,
54            Request::START_TRANSACTION_INDEX_FIELD.name,
55        )?;
56        require_field_when_set(
57            request.end_checkpoint,
58            Request::END_CHECKPOINT_FIELD.name,
59            request.end_transaction_index,
60            Request::END_TRANSACTION_INDEX_FIELD.name,
61        )?;
62        Ok(Self {
63            start: PositionInCheckpoint::from_fields(request.start_transaction_index, None),
64            end: PositionInCheckpoint::from_fields(request.end_transaction_index, None),
65            end_checkpoint: request.end_checkpoint,
66        })
67    }
68
69    pub fn from_events_request(request: &ListEventsRequest) -> Result<Self, RpcError> {
70        type Request = ListEventsRequest;
71        require_field_when_set(
72            request.start_checkpoint,
73            Request::START_CHECKPOINT_FIELD.name,
74            request.start_transaction_index,
75            Request::START_TRANSACTION_INDEX_FIELD.name,
76        )?;
77        require_field_when_set(
78            request.start_transaction_index,
79            Request::START_TRANSACTION_INDEX_FIELD.name,
80            request.start_event_index,
81            Request::START_EVENT_INDEX_FIELD.name,
82        )?;
83        require_field_when_set(
84            request.end_checkpoint,
85            Request::END_CHECKPOINT_FIELD.name,
86            request.end_transaction_index,
87            Request::END_TRANSACTION_INDEX_FIELD.name,
88        )?;
89        require_field_when_set(
90            request.end_transaction_index,
91            Request::END_TRANSACTION_INDEX_FIELD.name,
92            request.end_event_index,
93            Request::END_EVENT_INDEX_FIELD.name,
94        )?;
95        Ok(Self {
96            start: PositionInCheckpoint::from_fields(
97                request.start_transaction_index,
98                request.start_event_index,
99            ),
100            end: PositionInCheckpoint::from_fields(
101                request.end_transaction_index,
102                request.end_event_index,
103            ),
104            end_checkpoint: request.end_checkpoint,
105        })
106    }

108    /// A superset of the checkpoints whose first transaction's `tx_seq`
109    /// [`Self::transaction_window`] and [`Self::event_window`] read for the checkpoint `window`,
110    /// so a backend can fetch them all in one read.
111    pub fn boundary_checkpoints(&self, window: &Range<u64>) -> Vec<u64> {
112        let mut checkpoints = vec![window.start];
113        if !window.is_empty() {
114            checkpoints.push(window.end);
115            if self.start.is_some() {
116                checkpoints.push(window.start + 1);
117            }
118            if self.end.is_some() {
119                checkpoints.push(window.end - 1);
120            }
121        }
122        checkpoints
123    }
124
125    /// The `tx_seq` scan window for the checkpoint `window` (the request's checkpoint bounds
126    /// resolved against this server's tip), narrowed to these positions. `boundary(c)` returns
127    /// the `tx_seq` of checkpoint `c`'s first transaction, which is one past the last transaction
128    /// of checkpoint `c - 1`.
129    pub fn transaction_window(
130        &self,
131        window: &Range<u64>,
132        boundary: impl FnMut(u64) -> Result<u64, RpcError>,
133    ) -> Result<ScanWindow<u64>, RpcError> {
134        self.narrow(window, boundary, |tx_seq, _event_index| tx_seq)
135    }
136
137    /// The event-coordinate scan window for the checkpoint `window`, narrowed to these positions.
138    /// See [`Self::transaction_window`] for `boundary`. An event index past its transaction's
139    /// events needs no clamp: as a start it admits the next transaction's events, and as an end
140    /// it admits all of its own transaction's.
141    pub fn event_window(
142        &self,
143        window: &Range<u64>,
144        boundary: impl FnMut(u64) -> Result<u64, RpcError>,
145    ) -> Result<ScanWindow<IntraTxCoordinate>, RpcError> {
146        self.narrow(window, boundary, |tx_seq, index| IntraTxCoordinate {
147            tx_seq,
148            index,
149        })
150    }
151
152    fn narrow<P: Copy + Eq>(
153        &self,
154        window: &Range<u64>,
155        mut boundary: impl FnMut(u64) -> Result<u64, RpcError>,
156        position: impl Fn(u64, u32) -> P,
157    ) -> Result<ScanWindow<P>, RpcError> {
158        let window_start = boundary(window.start)?;
159        let start_of_tx = |tx_seq| position(tx_seq, 0);
160        if window.is_empty() {
161            let edge = start_of_tx(window_start);
162            return Ok(ScanWindow::whole_checkpoints(edge..edge));
163        }
164        let window_end = boundary(window.end)?;
165
166        // A non-empty window starts at the requested `start_checkpoint`. A transaction index past
167        // that checkpoint's end starts at the next checkpoint's first transaction, and drops the
168        // event index, which names an event of a transaction past the checkpoint.
169        let start = match self.start {
170            None => start_of_tx(window_start),
171            Some(start) => {
172                let next_checkpoint = boundary(window.start + 1)?;
173                let tx_seq = window_start.saturating_add(start.transaction_index);
174                if tx_seq < next_checkpoint {
175                    position(tx_seq, start.event_index)
176                } else {
177                    start_of_tx(next_checkpoint)
178                }
179            }
180        };
181
182        // The end position applies only while the window still ends at the requested
183        // `end_checkpoint`: an `end_checkpoint` past this server's tip leaves the position's
184        // checkpoint unindexed, so the window ends at the tip instead. A transaction index past
185        // the last checkpoint's end ends after that whole checkpoint.
186        let end = match self.end.filter(|_| self.end_checkpoint == Some(window.end)) {
187            None => start_of_tx(window_end),
188            Some(end) => {
189                let tx_seq = boundary(window.end - 1)?.saturating_add(end.transaction_index);
190                if tx_seq < window_end {
191                    position(tx_seq, end.event_index)
192                } else {
193                    start_of_tx(window_end)
194                }
195            }
196        };
197
198        Ok(ScanWindow {
199            start_cuts_checkpoint: start != start_of_tx(window_start),
200            end_cuts_checkpoint: end != start_of_tx(window_end),
201            range: start..end,
202        })
203    }
204}

206impl PositionInCheckpoint {
207    /// The position a request's index fields name, or `None` when its transaction index is
208    /// unset. An unset event index is the gap before the whole transaction.
209    fn from_fields(transaction_index: Option<u64>, event_index: Option<u32>) -> Option<Self> {
210        transaction_index.map(|transaction_index| Self {
211            transaction_index,
212            event_index: event_index.unwrap_or(0),
213        })
214    }
215}

217/// `InvalidArgument` when `when_set` is set but `required` is not, for a field that only means
218/// something together with another: an index with the checkpoint or transaction it indexes into.
219fn require_field_when_set<R, W>(
220    required: Option<R>,
221    required_name: &'static str,
222    when_set: Option<W>,
223    when_set_name: &'static str,
224) -> Result<(), RpcError> {
225    if when_set.is_some() && required.is_none() {
226        return Err(FieldViolation::new(required_name)
227            .with_description(format!(
228                "{required_name} is required when {when_set_name} is set"
229            ))
230            .with_reason(ErrorReason::FieldMissing)
231            .into());
232    }
233    Ok(())
234}

251    /// The model ledger's first transaction of each checkpoint, and the tip. Checkpoint 0 holds
252    /// transactions 0..3, checkpoint 1 none, checkpoint 2 3..7, checkpoint 3 7..9, and checkpoint
253    /// 4 9..14.

255    const TIP: u64 = 5;

259    fn boundary(checkpoint: u64) -> Result<u64, RpcError> {
260        Ok(BOUNDARIES[checkpoint as usize])

263    /// A request over checkpoints `[start, end)` with the given start and end positions, each a
264    /// `(transaction_index, event_index)`.
265    fn events_request(
266        start: Option<u64>,
267        end: Option<u64>,
268        start_position: Option<(u64, Option<u32>)>,
269        end_position: Option<(u64, Option<u32>)>,
270    ) -> ListEventsRequest {
271        let mut request = ListEventsRequest::default();
272        request.start_checkpoint = start;
273        request.end_checkpoint = end;
274        request.start_transaction_index = start_position.map(|(tx, _)| tx);
275        request.start_event_index = start_position.and_then(|(_, event)| event);
276        request.end_transaction_index = end_position.map(|(tx, _)| tx);
277        request.end_event_index = end_position.and_then(|(_, event)| event);
278        request
279    }

295    fn options(
296        ordering: Ordering,
297        after: Option<Position>,
298        before: Option<Position>,
299    ) -> QueryOptions {
300        let cursor = |position| CursorToken::boundary(position).encode();
301        let mut proto = ProtoQueryOptions::default();
302        if ordering == Ordering::Descending {
303            proto.ordering = Some(ProtoOrdering::Descending as i32);
304        }
305        proto.after = after.map(cursor);
306        proto.before = before.map(cursor);
307        QueryOptions::transactions_from_proto(Some(&proto), 100, 100).unwrap()
308    }

310    fn tx_position(checkpoint: u64, tx_seq: u64) -> Position {
311        Position::Transactions { checkpoint, tx_seq }
312    }

314    fn resolve_transactions(
315        request: &ListTransactionsRequest,
316        options: &QueryOptions,
317    ) -> ResolvedScan<u64> {
318        let positions = CheckpointPositions::from_transactions_request(request).unwrap();
319        let cp_range = ResolvedCheckpointRange::from_request(
320            request.start_checkpoint,
321            request.end_checkpoint,
322            TIP,
323            options,
324        )
325        .unwrap();
326        let window = positions
327            .transaction_window(&cp_range.range, boundary)
328            .unwrap();
329        ResolvedScan::resolve(cp_range, window, options).unwrap()
330    }

332    fn event_window(request: &ListEventsRequest) -> ScanWindow<IntraTxCoordinate> {
333        let positions = CheckpointPositions::from_events_request(request).unwrap();
334        let window = request.start_checkpoint.unwrap_or(0)..request.end_checkpoint.unwrap_or(TIP);
335        positions.event_window(&window, boundary).unwrap()
336    }

346    /// What a resolved scan ends with: `(range, terminal cursor's checkpoint field, terminal
347    /// position, end reason, natural end claim)`.
348    fn terminal(
349        resolved: &ResolvedScan<u64>,
350        options: &QueryOptions,
351    ) -> (Range<u64>, u64, u64, RangeExhaustion, Option<u64>) {
352        (
353            resolved.range(),
354            resolved.end_checkpoint,
355            resolved.end_position,
356            resolved.exhaustion,
357            resolved.natural_end_claim(options),
358        )

387    /// A start position inside the first checkpoint narrows the scan. Ascending, it vouches for
388    /// the positions before it, so the scan still claims that checkpoint; descending, the scan
389    /// never covers that checkpoint, so its end claims the one above.

391    fn start_position_inside_checkpoint() {
392        let request = transactions_request(Some(2), Some(4), Some(1), None);

448    #[test]
449    fn event_positions() {
450        // Mid-checkpoint start and end, with event indexes.
451        let window = event_window(&events_request(
452            Some(2),
453            Some(4),
454            Some((1, Some(2))),
455            Some((0, Some(3))),
456        ));
457        assert_eq!(window.range, event(4, 2)..event(7, 3));
458
459        // An event index past its transaction's events needs no clamp: the end admits all of
460        // that transaction's events.
461        let window = event_window(&events_request(Some(2), Some(4), None, Some((1, Some(9)))));
462        assert_eq!(window.range, event(3, 0)..event(8, 9));
463
464        // A transaction index past the checkpoint drops the event index: the start is the next
465        // checkpoint's first event, and the end covers the whole checkpoint.
466        let window = event_window(&events_request(
467            Some(2),
468            Some(4),
469            Some((4, Some(5))),
470            Some((2, Some(1))),
471        ));
472        assert_eq!(window.range, event(7, 0)..event(9, 0));
473        assert!(window.start_cuts_checkpoint);
474        assert!(!window.end_cuts_checkpoint);
475
476        // In an empty checkpoint, every start position is its end, which is also its start.
477        let window = event_window(&events_request(Some(1), Some(3), Some((0, Some(2))), None));
478        assert_eq!(window.range, event(3, 0)..event(7, 0));
479        assert!(!window.start_cuts_checkpoint);
480    }

510    /// Positions and cursors bound the scan by intersection: the later start and the earlier end
511    /// win, and the end's reason and cursor follow whichever ends it.
512    #[test]
513    fn positions_intersect_cursors() {
514        // Scans transactions 4..8.
515        let request = transactions_request(Some(2), Some(4), Some(1), Some(1));
516        let cases = [
517            (
518                options(Ordering::Ascending, Some(tx_position(2, 6)), None),
519                (6..8, 3, 8, CHECKPOINT_BOUND),
520            ),
521            (
522                options(Ordering::Ascending, Some(tx_position(0, 2)), None),
523                (4..8, 3, 8, CHECKPOINT_BOUND),
524            ),
525            (
526                options(Ordering::Ascending, None, Some(tx_position(2, 5))),
527                (4..5, 2, 5, CURSOR_BOUND),
528            ),
529            (
530                options(Ordering::Ascending, None, Some(tx_position(4, 9))),
531                (4..8, 3, 8, CHECKPOINT_BOUND),
532            ),
533            (
534                options(Ordering::Descending, None, Some(tx_position(3, 6))),
535                (4..6, 3, 4, CHECKPOINT_BOUND),
536            ),
537            (
538                options(Ordering::Descending, Some(tx_position(2, 5)), None),
539                (5..8, 2, 5, CURSOR_BOUND),
540            ),
541        ];
542        for (options, expected) in cases {
543            let resolved = resolve_transactions(&request, &options);
544            assert_eq!(
545                (
546                    resolved.range(),
547                    resolved.end_checkpoint,
548                    resolved.end_position,
549                    resolved.exhaustion
550                ),
551                expected,
552                "{options:?}"
553            );
554        }
555    }
