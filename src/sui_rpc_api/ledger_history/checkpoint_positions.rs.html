4//! Start and end positions inside the checkpoints at the edges of a `ListTransactions` /
5//! `ListEvents` request's checkpoint window.
6//!
7//! A position is a gap between items, named by checkpoint-relative offsets: `(checkpoint,
8//! transaction_index[, event_index])` is the gap just before that transaction (or that event of
9//! it). The start position `(start_checkpoint, start_transaction_index[, start_event_index])` is
10//! the first position a scan admits. The end position is exclusive and lies in the window's last
11//! checkpoint, `(end_checkpoint - 1, end_transaction_index[, end_event_index])`, so a server that
12//! ignores it returns a superset. Offsets past the end of their checkpoint (or transaction) clamp
13//! to its end instead of failing: the gap after a checkpoint's last item is `(C, n)` for its
14//! transaction count `n`, which clients cannot know.
15

29/// A request's start and end positions within its checkpoint window's edge checkpoints. An unset
30/// position leaves its edge at the checkpoint boundary.
31#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
32pub struct CheckpointPositions {
33    /// Where the scan starts in `start_checkpoint` (inclusive).
34    start: Option<PositionInCheckpoint>,
35    /// Where the scan ends in `end_checkpoint - 1` (exclusive).
36    end: Option<PositionInCheckpoint>,
37    /// The request's `start_checkpoint`.
38    start_checkpoint: Option<u64>,
39    /// The request's `end_checkpoint`, set whenever `end` is.
40    end_checkpoint: Option<u64>,
41}
42
43/// A gap between items inside one checkpoint: just before the event at `event_index` of the
44/// transaction at `transaction_index`. Event index 0 is the gap before the whole transaction.
45#[derive(Clone, Copy, Debug, Eq, PartialEq)]
46struct PositionInCheckpoint {
47    transaction_index: u64,
48    event_index: u32,
49}

51impl CheckpointPositions {
52    pub fn from_transactions_request(request: &ListTransactionsRequest) -> Result<Self, RpcError> {
53        type Request = ListTransactionsRequest;
54        require_field_when_set(
55            request.start_checkpoint,
56            Request::START_CHECKPOINT_FIELD.name,
57            request.start_transaction_index,
58            Request::START_TRANSACTION_INDEX_FIELD.name,
59        )?;
60        require_field_when_set(
61            request.end_checkpoint,
62            Request::END_CHECKPOINT_FIELD.name,
63            request.end_transaction_index,
64            Request::END_TRANSACTION_INDEX_FIELD.name,
65        )?;
66        Ok(Self {
67            start: PositionInCheckpoint::from_fields(request.start_transaction_index, None),
68            end: PositionInCheckpoint::from_fields(request.end_transaction_index, None),
69            start_checkpoint: request.start_checkpoint,
70            end_checkpoint: request.end_checkpoint,
71        })
72    }

114    /// A superset of the checkpoints whose first transaction's `tx_seq`
115    /// [`Self::transaction_window`] and [`Self::event_window`] read for the checkpoint `window`,
116    /// so a backend can fetch them all in one read.
117    pub fn boundary_checkpoints(&self, window: &Range<u64>) -> Vec<u64> {
118        let mut checkpoints = vec![window.start];
119        if !window.is_empty() {
120            checkpoints.push(window.end);
121            if self.start.is_some() {
122                checkpoints.push(window.start + 1);
123            }
124            if self.end.is_some() {
125                checkpoints.push(window.end - 1);
126            }
127        }
128        checkpoints
129    }

131    /// The `tx_seq` scan window for the checkpoint `window` (the request's checkpoint bounds
132    /// resolved against this server's tip), narrowed to these positions. `boundary(c)` returns
133    /// the `tx_seq` of checkpoint `c`'s first transaction, which is one past the last transaction
134    /// of checkpoint `c - 1`.
135    pub fn transaction_window(
136        &self,
137        window: &Range<u64>,
138        boundary: impl FnMut(u64) -> Result<u64, RpcError>,
139    ) -> Result<ScanWindow<u64>, RpcError> {
140        self.narrow(window, boundary, |tx_seq, _event_index| tx_seq)
141    }
142
143    /// The event-coordinate scan window for the checkpoint `window`, narrowed to these positions.
144    /// See [`Self::transaction_window`] for `boundary`. An event index past its transaction's
145    /// events needs no clamp: as a start it admits the next transaction's events, and as an end
146    /// it admits all of its own transaction's.
147    pub fn event_window(
148        &self,
149        window: &Range<u64>,
150        boundary: impl FnMut(u64) -> Result<u64, RpcError>,
151    ) -> Result<ScanWindow<IntraTxCoordinate>, RpcError> {
152        self.narrow(window, boundary, |tx_seq, index| IntraTxCoordinate {
153            tx_seq,
154            index,
155        })
156    }
157
158    fn narrow<P: Copy + Eq>(
159        &self,
160        window: &Range<u64>,
161        mut boundary: impl FnMut(u64) -> Result<u64, RpcError>,
162        position: impl Fn(u64, u32) -> P,
163    ) -> Result<ScanWindow<P>, RpcError> {
164        let window_start = boundary(window.start)?;
165        let start_of_tx = |tx_seq| position(tx_seq, 0);
166        if window.is_empty() {
167            // A window without a checkpoint ends where the request's own fields name; its scan
168            // edge is the start of the window's checkpoint.
169            let edge = start_of_tx(window_start);
170            let edge_point = LedgerPoint::checkpoint_start(window.start);
171            return Ok(ScanWindow::whole_checkpoints(
172                edge..edge,
173                LedgerWindow {
174                    start: edge_point,
175                    end: edge_point,
176                    requested: Some((
177                        self.requested_start(),
178                        self.requested_end().unwrap_or(edge_point),
179                    )),
180                },
181            ));
182        }
183        let window_end = boundary(window.end)?;
184
185        // A non-empty window starts at the requested `start_checkpoint`. A transaction index past
186        // that checkpoint's end starts at the next checkpoint's first transaction, and drops the
187        // event index, which names an event of a transaction past the checkpoint.
188        let (start, ledger_start) = match self.start {
189            None => (
190                start_of_tx(window_start),
191                LedgerPoint::checkpoint_start(window.start),
192            ),
193            Some(start) => {
194                let next_checkpoint = boundary(window.start + 1)?;
195                let tx_seq = window_start.saturating_add(start.transaction_index);
196                if tx_seq < next_checkpoint {
197                    (
198                        position(tx_seq, start.event_index),
199                        start.point(window.start),
200                    )
201                } else {
202                    (
203                        start_of_tx(next_checkpoint),
204                        LedgerPoint::checkpoint_start(window.start + 1),
205                    )
206                }
207            }
208        };
209
210        // The end position applies only while the window still ends at the requested
211        // `end_checkpoint`: an `end_checkpoint` past this server's tip leaves the position's
212        // checkpoint unindexed, so the window ends at the tip instead. A transaction index past
213        // the last checkpoint's end ends after that whole checkpoint.
214        let (end, ledger_end) = match self.end.filter(|_| self.end_checkpoint == Some(window.end)) {
215            None => (
216                start_of_tx(window_end),
217                LedgerPoint::checkpoint_start(window.end),
218            ),
219            Some(end) => {
220                let tx_seq = boundary(window.end - 1)?.saturating_add(end.transaction_index);
221                if tx_seq < window_end {
222                    (position(tx_seq, end.event_index), end.point(window.end - 1))
223                } else {
224                    (
225                        start_of_tx(window_end),
226                        LedgerPoint::checkpoint_start(window.end),
227                    )
228                }
229            }
230        };
231
232        Ok(ScanWindow {
233            start_cuts_checkpoint: start != start_of_tx(window_start),
234            end_cuts_checkpoint: end != start_of_tx(window_end),
235            range: start..end,
236            ledger: LedgerWindow {
237                start: ledger_start,
238                end: ledger_end,
239                requested: None,
240            },
241        })
242    }
243
244    /// The point the request's start fields name.
245    fn requested_start(&self) -> LedgerPoint {
246        let checkpoint = self.start_checkpoint.unwrap_or(0);
247        match self.start {
248            Some(start) => start.point(checkpoint),
249            None => LedgerPoint::checkpoint_start(checkpoint),
250        }
251    }
252
253    /// The point the request's end fields name, if it set `end_checkpoint`: the end position in
254    /// `end_checkpoint - 1`, or the start of `end_checkpoint`.
255    fn requested_end(&self) -> Option<LedgerPoint> {
256        let end_checkpoint = self.end_checkpoint?;
257        Some(match (self.end, end_checkpoint.checked_sub(1)) {
258            (Some(end), Some(last_checkpoint)) => end.point(last_checkpoint),
259            _ => LedgerPoint::checkpoint_start(end_checkpoint),
260        })
261    }
262}

264impl PositionInCheckpoint {
265    /// The position a request's index fields name, or `None` when its transaction index is
266    /// unset. An unset event index is the gap before the whole transaction.
267    fn from_fields(transaction_index: Option<u64>, event_index: Option<u32>) -> Option<Self> {
268        transaction_index.map(|transaction_index| Self {
269            transaction_index,
270            event_index: event_index.unwrap_or(0),
271        })
272    }
273
274    /// This position in `checkpoint`, in ledger coordinates.
275    fn point(self, checkpoint: u64) -> LedgerPoint {
276        LedgerPoint::event(checkpoint, self.transaction_index, self.event_index)
277    }
278}

280/// `InvalidArgument` when `when_set` is set but `required` is not, for a field that only means
281/// something together with another: an index with the checkpoint or transaction it indexes into.
282fn require_field_when_set<R, W>(
283    required: Option<R>,
284    required_name: &'static str,
285    when_set: Option<W>,
286    when_set_name: &'static str,
287) -> Result<(), RpcError> {
288    if when_set.is_some() && required.is_none() {
289        return Err(FieldViolation::new(required_name)
290            .with_description(format!(
291                "{required_name} is required when {when_set_name} is set"
292            ))
293            .with_reason(ErrorReason::FieldMissing)
294            .into());
295    }
296    Ok(())
297}

360    fn options(
361        ordering: Ordering,
362        after: Option<Position>,
363        before: Option<Position>,
364    ) -> QueryOptions {
365        let mut proto = ProtoQueryOptions::default();
366        if ordering == Ordering::Descending {
367            proto.ordering = Some(ProtoOrdering::Descending as i32);
368        }
369        proto.after = after.map(|position| CursorToken::boundary(position).encode());
370        proto.before = before.map(|position| CursorToken::boundary(position).encode());
371        QueryOptions::transactions_from_proto(Some(&proto), 100, 100).unwrap()

374    /// A cursor position at transaction `tx_seq` of the model ledger, carrying its public
375    /// position.
376    fn tx_position(tx_seq: u64) -> Position {
377        let checkpoint = BOUNDARIES[1..]
378            .iter()
379            .position(|end| tx_seq < *end)
380            .unwrap();
381        let transaction_index = tx_seq - BOUNDARIES[checkpoint];
382        Position::Transactions {
383            tx_seq,
384            public_position: LedgerPoint::transaction(checkpoint as u64, transaction_index),
385        }
386    }

416    /// Where a transactions request's terminal frame resumes in ledger coordinates, and the
417    /// public position of its cursor.
418    fn transactions_terminal(
419        request: &ListTransactionsRequest,
420        options: &QueryOptions,
421    ) -> (LedgerPoint, LedgerPoint) {
422        let positions = CheckpointPositions::from_transactions_request(request).unwrap();

437        )
438    }

722    #[test]
723    fn event_positions() {
724        // Mid-checkpoint start and end, with event indexes.
725        let window = event_window(&events_request(
726            Some(2),
727            Some(4),
728            Some((1, Some(2))),
729            Some((0, Some(3))),
730        ));
731        assert_eq!(window.range, event(4, 2)..event(7, 3));
732
733        // An event index past its transaction's events needs no clamp: the end admits all of
734        // that transaction's events.
735        let window = event_window(&events_request(Some(2), Some(4), None, Some((1, Some(9)))));
736        assert_eq!(window.range, event(3, 0)..event(8, 9));
737
738        // A transaction index past the checkpoint drops the event index: the start is the next
739        // checkpoint's first event, and the end covers the whole checkpoint.
740        let window = event_window(&events_request(
741            Some(2),
742            Some(4),
743            Some((4, Some(5))),
744            Some((2, Some(1))),
745        ));
746        assert_eq!(window.range, event(7, 0)..event(9, 0));
747        assert!(window.start_cuts_checkpoint);
748        assert!(!window.end_cuts_checkpoint);
749
750        // In an empty checkpoint, every start position is its end, which is also its start.
751        let window = event_window(&events_request(Some(1), Some(3), Some((0, Some(2))), None));
752        assert_eq!(window.range, event(3, 0)..event(7, 0));
753        assert!(!window.start_cuts_checkpoint);
754    }

802    /// Positions and cursors bound the scan by intersection: the later start and the earlier end
803    /// win, and the end's reason and cursor follow whichever ends it.
804    #[test]
805    fn positions_intersect_cursors() {
806        // Scans transactions 4..8.
807        let request = transactions_request(Some(2), Some(4), Some(1), Some(1));
808        let cases = [
809            (
810                options(Ordering::Ascending, Some(tx_position(6)), None),
811                (6..8, LedgerPoint::transaction(3, 1), 8, CHECKPOINT_BOUND),
812            ),
813            (
814                options(Ordering::Ascending, Some(tx_position(2)), None),
815                (4..8, LedgerPoint::transaction(3, 1), 8, CHECKPOINT_BOUND),
816            ),
817            (
818                options(Ordering::Ascending, None, Some(tx_position(5))),
819                (4..5, LedgerPoint::transaction(2, 2), 5, CURSOR_BOUND),
820            ),
821            (
822                options(Ordering::Ascending, None, Some(tx_position(9))),
823                (4..8, LedgerPoint::transaction(3, 1), 8, CHECKPOINT_BOUND),
824            ),
825            (
826                options(Ordering::Descending, None, Some(tx_position(6))),
827                (4..6, LedgerPoint::transaction(2, 1), 4, CHECKPOINT_BOUND),
828            ),
829            (
830                options(Ordering::Descending, Some(tx_position(5)), None),
831                (5..8, LedgerPoint::transaction(2, 2), 5, CURSOR_BOUND),
832            ),
833        ];
834        for (options, expected) in cases {
835            let resolved = resolve_transactions(&request, &options);
836            assert_eq!(
837                (
838                    resolved.range(),
839                    resolved.end_cursor_public_position(&options),
840                    resolved.end_scan_position,
841                    resolved.exhaustion
842                ),
843                expected,
844                "{options:?}"
845            );
846        }
847    }
