221/// Set of `/package.Service/Method` paths that are safe to use as metric
222/// labels.
223///
224/// Services are mounted with the wildcard route `/{ServiceName}/{*rest}`, so
225/// any path under a registered prefix matches a route and would otherwise be
226/// taken verbatim as a `path` label. Bounding the labels to known methods
227/// prevents an unauthenticated attacker from inflating Prometheus label maps
228/// (which the prometheus crate retains for the lifetime of the process) by
229/// streaming requests with random method suffixes.
230pub type GrpcMethodAllowlist = Arc<HashSet<String>>;
231
232/// Decode one or more encoded `FileDescriptorSet` byte slices and return the
233/// set of `/package.Service/Method` paths they declare.
234///
235/// Intended to be called once at server startup with the same bytes that are
236/// registered with `tonic_reflection`, so the metrics allowlist stays in sync
237/// with the services actually exposed over gRPC.
238pub fn grpc_method_paths_from_file_descriptor_sets(
239    encoded_sets: &[&[u8]],
240) -> Result<HashSet<String>, prost::DecodeError> {
241    let mut paths = HashSet::new();
242    for bytes in encoded_sets {
243        let fds = prost_types::FileDescriptorSet::decode(*bytes)?;
244        for file in fds.file {
245            let package = file.package.unwrap_or_default();
246            for service in file.service {
247                let Some(service_name) = service.name else {
248                    continue;
249                };
250                let qualified_service = if package.is_empty() {
251                    service_name
252                } else {
253                    format!("{}.{}", package, service_name)
254                };
255                for method in service.method {
256                    let Some(method_name) = method.name else {
257                        continue;
258                    };
259                    paths.insert(format!("/{}/{}", qualified_service, method_name));
260                }
261            }
262        }
263    }
264    Ok(paths)
265}

267/// Builds the per-request metrics handlers for [`sui_http::middleware::callback::CallbackLayer`].
268///
269/// The layer must be installed outside any layer that wraps the request body
270/// with its own failure modes (body-size limits, request decompression,
271/// body-read timeouts). A request whose body errors is counted `canceled`,
272/// which is only correct while the body this layer observes is the
273/// transport's: then an error means the client reset the stream, the
274/// connection failed, or the client broke the protocol. Errors from an outer
275/// body-wrapping layer would be counted `canceled` and hide real rejections.
276#[derive(Clone)]
277pub struct RpcMetricsMakeCallbackHandler {
278    metrics: Arc<RpcMetrics>,
279    grpc_method_allowlist: GrpcMethodAllowlist,
280}
281
282impl RpcMetricsMakeCallbackHandler {
283    /// Construct a handler with no gRPC method allowlist. All gRPC requests
284    /// will be labelled with their matched route pattern (e.g.
285    /// `/sui.rpc.v2.LedgerService/{*rest}`) rather than the per-method path,
286    /// which is safe but loses per-method granularity.
287    pub fn new(metrics: Arc<RpcMetrics>) -> Self {
288        Self::with_grpc_method_allowlist(metrics, Arc::new(HashSet::new()))
289    }
290
291    /// Construct a handler that uses `allowlist` to decide which gRPC request
292    /// paths are safe to emit as Prometheus labels.
293    pub fn with_grpc_method_allowlist(
294        metrics: Arc<RpcMetrics>,
295        allowlist: GrpcMethodAllowlist,
296    ) -> Self {
297        Self {
298            metrics,
299            grpc_method_allowlist: allowlist,
300        }
301    }
302}

304impl MakeCallbackHandler for RpcMetricsMakeCallbackHandler {
305    type RequestHandler = RpcMetricsRequestHandler;
306    type ResponseHandler = RpcMetricsCallbackHandler;
307
308    fn make_handler(
309        &self,
310        request: &http::request::Parts,
311    ) -> (Self::RequestHandler, Self::ResponseHandler) {
312        let start = Instant::now();
313        let metrics = self.metrics.clone();
314
315        let matched_path = request
316            .extensions
317            .get::<axum::extract::MatchedPath>()
318            .map(|m| m.as_str());
319        let is_grpc = request
320            .headers
321            .get(&http::header::CONTENT_TYPE)
322            .is_some_and(is_grpc_content_type);
323
324        let path = compute_metric_label(
325            is_grpc,
326            request.uri.path(),
327            matched_path,
328            &self.grpc_method_allowlist,
329        );
330
331        metrics
332            .inflight_requests
333            .with_label_values(&[path.as_ref()])
334            .inc();
335
336        let request_body_failed = Arc::new(AtomicBool::new(false));
337
338        (
339            RpcMetricsRequestHandler {
340                request_body_failed: request_body_failed.clone(),
341            },
342            RpcMetricsCallbackHandler {
343                metrics,
344                path,
345                start,
346                request_body_failed,
347                counting: CountingState::AwaitingResponse,
348                counted_first_chunk: false,
349            },
350        )
351    }
352}

354/// Decide which string to use as the `path` Prometheus label for a request.
355///
356/// For gRPC traffic, prefer the per-method URI path when it is in the
357/// allowlist; otherwise fall back to the matched route pattern so unknown
358/// methods collapse into a single bounded series per service. For non-gRPC
359/// traffic the matched path is already bounded by the routes registered on
360/// the router, so it is used directly.
361fn compute_metric_label(
362    is_grpc: bool,
363    uri_path: &str,
364    matched_path: Option<&str>,
365    grpc_method_allowlist: &HashSet<String>,
366) -> Cow<'static, str> {
367    match (is_grpc, matched_path) {
368        (true, _) if grpc_method_allowlist.contains(uri_path) => Cow::Owned(uri_path.to_owned()),
369        (true, Some(matched)) => Cow::Owned(matched.to_owned()),
370        (false, Some(matched)) => Cow::Owned(matched.to_owned()),
371        (_, None) => Cow::Borrowed("unknown"),
372    }
373}

387/// Observes the request body so that a request the client abandoned is
388/// counted as `canceled` rather than with whatever status the service
389/// produced for the truncated request. This relies on the placement described
390/// on [`RpcMetricsMakeCallbackHandler`].
391pub struct RpcMetricsRequestHandler {
392    request_body_failed: Arc<AtomicBool>,
393}
394
395impl RequestHandler for RpcMetricsRequestHandler {
396    fn on_body_error<E>(&mut self, _error: &E)
397    where
398        E: std::fmt::Display + 'static,
399    {
400        self.request_body_failed.store(true, Ordering::Release);
401    }

404/// Progress of a request towards its single `rpc_requests` increment.
405enum CountingState {
406    /// The service has not produced a response yet.
407    AwaitingResponse,
408    /// The response is a native gRPC stream whose status arrives in the
409    /// trailers that end the response body.
410    AwaitingTrailers,
411    Counted,
412}
413
414pub struct RpcMetricsCallbackHandler {
415    metrics: Arc<RpcMetrics>,
416    path: Cow<'static, str>,
417    start: Instant,
418    request_body_failed: Arc<AtomicBool>,
419    // Requests that end before reaching `Counted` (the service future or the
420    // response body is dropped first) are counted as `canceled` on drop.
421    counting: CountingState,
422    counted_first_chunk: bool,
423}
424
425impl RpcMetricsCallbackHandler {
426    fn count(&mut self, status: &str) {
427        // A request body that errors means the client reset the stream or the
428        // connection failed while the request was arriving, so no response
429        // reaches the client. The service's answer to the truncated request is
430        // not the outcome: tonic, for example, answers a request whose message
431        // never arrived with `internal`.
432        let status = if self.request_body_failed.load(Ordering::Acquire) {
433            "canceled"
434        } else {
435            status
436        };
437
438        self.metrics
439            .num_requests
440            .with_label_values(&[self.path.as_ref(), status])
441            .inc();
442
443        self.counting = CountingState::Counted;
444    }
445}
446
447impl ResponseHandler for RpcMetricsCallbackHandler {
448    fn on_response(&mut self, response: &http::response::Parts) {
449        // Unlike `request_latency` (observed in `Drop`, after the response
450        // body finished streaming), this fires as soon as the handler
451        // produced a response, so it excludes client-side network latency.
452        // Skip when reading the request body failed before the handler ran.
453        if !self.request_body_failed.load(Ordering::Acquire) {
454            self.metrics
455                .request_handler_latency
456                .with_label_values(&[self.path.as_ref()])
457                .observe(self.start.elapsed().as_secs_f64());
458        }
459        let content_type = response.headers.get(&http::header::CONTENT_TYPE);
460        let status = if content_type.is_some_and(is_grpc_content_type) {
461            match response.headers.get(&GRPC_STATUS) {
462                // Trailers-only response: the status is final, and the empty
463                // body may never be polled.
464                Some(grpc_status) => code_as_str(tonic::Code::from_bytes(grpc_status.as_bytes())),
465                // grpc-web encodes the trailers into the response body, which
466                // is not parsed here.
467                None if content_type.is_some_and(is_grpc_web_content_type) => {
468                    code_as_str(tonic::Code::Ok)
469                }
470                None => {
471                    self.counting = CountingState::AwaitingTrailers;
472                    return;
473                }
474            }
475        } else {
476            response.status.as_str()
477        };
478
479        self.count(status);
480    }
481
482    fn on_body_chunk<B>(&mut self, _chunk: &B)
483    where
484        B: bytes::Buf,
485    {
486        if !self.counted_first_chunk {
487            self.metrics
488                .first_chunk_latency
489                .with_label_values(&[self.path.as_ref()])
490                .observe(self.start.elapsed().as_secs_f64());
491            self.counted_first_chunk = true;
492        }
493    }
494
495    fn on_end_of_stream(&mut self, trailers: Option<&http::HeaderMap>) {
496        if let CountingState::AwaitingTrailers = self.counting {
497            // gRPC requires `grpc-status` in the trailers; clients report a
498            // stream that ends without it as `unknown`.
499            let code = trailers
500                .and_then(|trailers| trailers.get(&GRPC_STATUS))
501                .map(|grpc_status| tonic::Code::from_bytes(grpc_status.as_bytes()))
502                .unwrap_or(tonic::Code::Unknown);
503
504            self.count(code_as_str(code));
505        }
506    }
507
508    fn on_body_error<E>(&mut self, _error: &E)
509    where
510        E: std::fmt::Display + 'static,
511    {
512        // The stream ends without trailers: hyper resets it (with
513        // `INTERNAL_ERROR` unless the error carries an HTTP/2 reason), which
514        // gRPC clients report as `internal`.
515        if let CountingState::AwaitingTrailers = self.counting {
516            self.count(code_as_str(tonic::Code::Internal));
517        }
518    }
519
520    fn on_service_error<E>(&mut self, _error: &E)
521    where
522        E: std::fmt::Display + 'static,
523    {
524        // Do nothing if the whole service errored
525        //
526        // in Axum this isn't possible since all services are required to have an error type of
527        // Infallible
528    }
529}

531impl Drop for RpcMetricsCallbackHandler {
532    fn drop(&mut self) {
533        self.metrics
534            .inflight_requests
535            .with_label_values(&[self.path.as_ref()])
536            .dec();
537
538        let latency = self.start.elapsed().as_secs_f64();
539        self.metrics
540            .request_latency
541            .with_label_values(&[self.path.as_ref()])
542            .observe(latency);
543
544        if !matches!(self.counting, CountingState::Counted) {
545            self.metrics
546                .num_requests
547                .with_label_values(&[self.path.as_ref(), "canceled"])
548                .inc();
549        }
550    }
551}

631        let Some(inner) = &mut self.inner else {
632            return;
633        };
634        if is_data {

655    /// Pair with `yield_clock`: capture immediately before `yield`, then observe as the first
656    /// statement after resumption. A stream dropped while suspended records no sample.
657    pub(crate) fn observe_yield_wait(&self, yield_started: Option<Instant>) {
658        if let (Some(inner), Some(yield_started)) = (&self.inner, yield_started) {
659            inner
660                .handles
661                .yield_wait
662                .observe(yield_started.elapsed().as_secs_f64());
663        }
664    }

666    pub(crate) fn finish_success(
667        &mut self,
668        reason: sui_rpc::proto::sui::rpc::v2::QueryEndReason,
669        bitmap_buckets_evaluated: Option<usize>,
670    ) {
671        let Some(inner) = &mut self.inner else {
672            return;
673        };
674        if inner.success_finished {
675            return;
676        }
677        inner.success_finished = true;
678        let reason = match reason {
679            sui_rpc::proto::sui::rpc::v2::QueryEndReason::ItemLimit => "item_limit",
680            sui_rpc::proto::sui::rpc::v2::QueryEndReason::ScanLimit => "scan_limit",
681            sui_rpc::proto::sui::rpc::v2::QueryEndReason::LedgerTip => "ledger_tip",
682            sui_rpc::proto::sui::rpc::v2::QueryEndReason::CheckpointBound => "checkpoint_bound",
683            sui_rpc::proto::sui::rpc::v2::QueryEndReason::CursorBound => "cursor_bound",
684            // Validation guarantees successful List streams always have a concrete end reason.
685            sui_rpc::proto::sui::rpc::v2::QueryEndReason::Unknown => {
686                unreachable!("validated successful List stream has an unspecified end reason")
687            }
688            _ => unreachable!("validated successful List stream has an unsupported end reason"),
689        };
690        inner
691            .handles
692            .query_ends
693            .with_label_values(&[inner.handles.method, reason])
694            .inc();
695        if let Some(bitmap_buckets_evaluated) = bitmap_buckets_evaluated {
696            inner
697                .handles
698                .bitmap_buckets_evaluated
699                .observe(bitmap_buckets_evaluated as f64);
700        }
701    }

1754        /// trailers-only response, otherwise from the trailers after the body.
1755        async fn client_grpc_status(response: http::Response<h2::RecvStream>) -> tonic::Code {
1756            let (parts, mut body) = response.into_parts();
1757            let grpc_status = match parts.headers.get(&GRPC_STATUS) {
1758                Some(grpc_status) => grpc_status.clone(),
1759                None => {
1760                    while let Some(chunk) = body.data().await {
1761                        let chunk = chunk.unwrap();
1762                        body.flow_control().release_capacity(chunk.len()).unwrap();
1763                    }
1764                    let trailers = body.trailers().await.unwrap().expect("missing trailers");
1765                    trailers
1766                        .get(&GRPC_STATUS)
1767                        .expect("missing grpc-status")
1768                        .clone()
1769                }
1770            };
1771            tonic::Code::from_bytes(grpc_status.as_bytes())
1772        }
