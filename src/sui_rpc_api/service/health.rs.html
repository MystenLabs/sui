12/// The largest gap, in checkpoints, between the latest executed checkpoint and
13/// the highest checkpoint the live-object index has committed while still
14/// considered healthy.
15///
16/// The embedded indexer follows the tip asynchronously, so the live-object
17/// index always trails the executed tip by a little (roughly the indexer's
18/// snapshot window). A gap larger than this means the live index has fallen
19/// behind -- e.g. the indexer has stalled -- and cannot serve current
20/// live-object reads. The ledger-history cohort backfills separately after a
21/// restore and is excluded from this gap, so by default a node is healthy as
22/// soon as its live-object reads are caught up; callers that also need
23/// history caught up opt in with `max_history_lag` (see [`Threshold`]).
24const MAX_HEALTHY_INDEX_LAG: u64 = 60;
25
26impl RpcService {
27    /// Perform a simple health check on the service.
28    ///
29    /// `threshold_seconds` is the threshold, or delta, between the server's
30    /// system time and the timestamp in the most recently executed checkpoint
31    /// for which the server is considered to be healthy. If not provided, the
32    /// server's tip is not subject to a staleness check.
33    ///
34    /// Independent of the threshold, when indexing is enabled the server is
35    /// only considered healthy once its live-object indexes have caught up to
36    /// within `MAX_HEALTHY_INDEX_LAG` checkpoints of the latest executed
37    /// checkpoint.
38    ///
39    /// `max_history_lag`, when provided, additionally requires every index
40    /// pipeline -- including the ledger-history cohort, which backfills
41    /// separately after a restore -- to have caught up to within that many
42    /// checkpoints of the live-object index. A history index with no watermark
43    /// yet is never healthy, and neither is a node without rpc indexes: its
44    /// reported tip is its executed checkpoint, which can run ahead of what
45    /// indexed replicas have reached.
46    ///
47    /// When indexing is disabled and `max_history_lag` is not provided, the
48    /// index checks are skipped.
49    pub fn health_check(
50        &self,
51        threshold_seconds: Option<u32>,
52        max_history_lag: Option<u64>,
53    ) -> Result<()> {
54        let latest = self.reader.inner().get_latest_checkpoint()?;
55
56        // If we have a provided threshold, check that it's close to the current
57        // time.
58        if let Some(threshold_seconds) = threshold_seconds {
59            let latest_chain_time = latest.timestamp();
60
61            let threshold = SystemTime::now() - Duration::from_secs(threshold_seconds as u64);
62
63            if latest_chain_time < threshold {
64                return Err(anyhow::anyhow!(
65                    "The latest checkpoint timestamp is less than the provided threshold"
66                )
67                .into());
68            }
69        }
70
71        // When indexing is enabled, the node is only healthy once its
72        // live-object indexes (owned objects, types, balances) have kept up
73        // with the executed tip. Those indexes are restored to the tip and
74        // follow it, so a healthy node's live frontier trails execution by at
75        // most the indexer's snapshot window. The ledger-history cohort
76        // backfills independently after a restore and is only checked when the
77        // caller passes `max_history_lag`: gating on it by default would report
78        // a node unhealthy for the whole backfill even though its live-object
79        // reads are already caught up. The executed tip is read unbounded
80        // (rather than via `get_latest_checkpoint`, which is itself bounded to
81        // the live frontier) so a stalled live indexer, whose frontier falls
82        // behind ongoing execution, is still detected.
83        //
84        // The reported tip is bounded to the live frontier, so a replica
85        // whose history trails it advertises checkpoints its ledger-history
86        // reads cannot serve yet. Measure the history gap against that
87        // frontier rather than the executed tip, so the live index's own
88        // (separately bounded) lag behind execution is not counted twice. A
89        // node without an index surface (indexing disabled) reports its
90        // executed checkpoint as the tip, so it fails the history check
91        // whenever `max_history_lag` is set and skips the index checks
92        // otherwise.
93        let history_ok = if let Some(indexes) = self.reader.inner().indexes() {
94            let executed = self
95                .reader
96                .inner()
97                .get_highest_executed_checkpoint_seq_number()?;
98            let highest_live_indexed = indexes.get_highest_live_indexed_checkpoint_seq_number()?;
99
100            if !index_caught_up(executed, highest_live_indexed, MAX_HEALTHY_INDEX_LAG) {
101                return Err(anyhow::anyhow!(
102                    "the live-object index is not caught up to within {MAX_HEALTHY_INDEX_LAG} \
103                     checkpoints of the latest executed checkpoint"
104                )
105                .into());
106            }
107
108            history_caught_up(
109                max_history_lag,
110                Some(IndexHeights {
111                    highest_live_indexed,
112                    read_highest_indexed: &|| indexes.get_highest_indexed_checkpoint_seq_number(),
113                }),
114            )?
115        } else {
116            history_caught_up(max_history_lag, None)?
117        };
118
119        if !history_ok {
120            return Err(anyhow::anyhow!(
121                "`max_history_lag` is set and the ledger-history indexes are absent or not \
122                 caught up to within that many checkpoints of the live-object index"
123            )
124            .into());
125        }
126
127        Ok(())
128    }
129}

131/// Whether the highest live-indexed checkpoint is close enough to the executed
132/// tip to be considered healthy.
133///
134/// `highest_live_indexed` is `None` when the live-object index has not committed
135/// any checkpoint yet, which is never healthy. The live frontier never runs
136/// ahead of execution (it indexes executed checkpoints), but an equal frontier
137/// saturates to a zero lag rather than underflowing.
138fn index_caught_up(executed_seq: u64, highest_live_indexed: Option<u64>, max_lag: u64) -> bool {
139    match highest_live_indexed {
140        Some(indexed) => executed_seq.saturating_sub(indexed) <= max_lag,
141        None => false,
142    }
143}

145/// The rpc index surface as the history check sees it.
146struct IndexHeights<'a> {
147    /// The live-object frontier; `None` before the live cohort's first commit.
148    highest_live_indexed: Option<u64>,
149    /// Reads the minimum committed checkpoint across all pipelines (live and
150    /// ledger-history); `None` while any pipeline has no watermark yet.
151    read_highest_indexed: &'a dyn Fn() -> StorageResult<Option<u64>>,
152}
153
154/// Whether every index pipeline (live and ledger-history) has caught up to
155/// within `max_lag` checkpoints of the live-object frontier.
156///
157/// With no `max_lag` the history cohort is not checked: the result is `true`
158/// and nothing is read. Otherwise `indexes` is `None` for a node without rpc
159/// indexes, which is never healthy. The all-pipelines minimum includes the
160/// live pipelines, but it is read separately from the live height, so a live
161/// cohort that advanced in between can leave it slightly above; that counts as
162/// zero lag. A `None` minimum (in practice a history pipeline after a restore,
163/// before its seed) is never healthy, and neither is a missing live frontier.
164fn history_caught_up(
165    max_lag: Option<u64>,
166    indexes: Option<IndexHeights<'_>>,
167) -> StorageResult<bool> {
168    let Some(max_lag) = max_lag else {
169        return Ok(true);
170    };
171    let Some(indexes) = indexes else {
172        return Ok(false);
173    };
174    Ok(
175        match (
176            indexes.highest_live_indexed,
177            (indexes.read_highest_indexed)()?,
178        ) {
179            (Some(live), Some(indexed)) => live.saturating_sub(indexed) <= max_lag,
180            _ => false,
181        },
182    )
183}

185#[derive(Debug, serde::Serialize, serde::Deserialize)]
186pub struct Threshold {
187    /// The threshold, or delta, between the server's system time and the timestamp in the most
188    /// recently executed checkpoint for which the server is considered to be healthy.
189    ///
190    /// If not provided, the server will be considered healthy if it can simply fetch the latest
191    /// checkpoint from its store and, when indexing is enabled, its live-object indexes have
192    /// caught up to it.
193    pub threshold_seconds: Option<u32>,
194
195    /// The largest gap, in checkpoints, between the live-object index and the slowest index
196    /// pipeline (in practice the ledger-history indexes, which backfill separately after a
197    /// restore) for which the server is considered to be healthy. A history index with no
198    /// watermark yet is unhealthy, and so is a node without rpc indexes (indexing disabled),
199    /// whose reported tip is its executed checkpoint.
200    ///
201    /// If not provided, ledger-history progress and the presence of rpc indexes do not affect
202    /// health.
203    pub max_history_lag: Option<u64>,
204}
205
206pub async fn health(
207    Query(Threshold {
208        threshold_seconds,
209        max_history_lag,
210    }): Query<Threshold>,
211    State(state): State<RpcService>,
212) -> impl axum::response::IntoResponse {
213    match state.health_check(threshold_seconds, max_history_lag) {
214        Ok(()) => (axum::http::StatusCode::OK, "up"),
215        Err(_) => (axum::http::StatusCode::SERVICE_UNAVAILABLE, "down"),
216    }
217}

219#[cfg(test)]
220mod tests {
221    use super::*;
222    use sui_types::storage::error::Error as StorageError;
223
224    // The live-object index has not committed any checkpoint yet: never
225    // healthy.
226    #[test]
227    fn not_caught_up_when_unindexed() {
228        assert!(!index_caught_up(100, None, MAX_HEALTHY_INDEX_LAG));
229        assert!(!index_caught_up(0, None, MAX_HEALTHY_INDEX_LAG));
230    }
231
232    // The live frontier is within (or at) the allowed lag of the executed tip:
233    // healthy.
234    #[test]
235    fn caught_up_within_lag() {
236        assert!(index_caught_up(100, Some(100), 60)); // no lag
237        assert!(index_caught_up(100, Some(40), 60)); // exactly at the bound
238        assert!(index_caught_up(100, Some(41), 60)); // inside the bound
239    }
240
241    // The live frontier trails the executed tip by more than the allowed lag
242    // (e.g. the live indexer stalled while execution advanced): unhealthy. A
243    // lagging ledger-history backfill does not reach this path -- it is not part
244    // of the live frontier.
245    #[test]
246    fn not_caught_up_beyond_lag() {
247        assert!(!index_caught_up(100, Some(39), 60)); // one past the bound
248        assert!(!index_caught_up(1_000, Some(0), 60)); // live index far behind
249    }
250
251    // A live frontier level with the executed tip saturates to zero lag rather
252    // than underflowing.
253    #[test]
254    fn caught_up_when_index_at_tip() {
255        assert!(index_caught_up(100, Some(100), 60));
256        assert!(index_caught_up(100, Some(200), 60));
257    }
258
259    /// Evaluate the history check on a node with rpc indexes, with a fixed
260    /// all-pipelines height.
261    fn history_ok(max_lag: Option<u64>, live: Option<u64>, all: Option<u64>) -> bool {
262        let read = || Ok(all);
263        let indexes = IndexHeights {
264            highest_live_indexed: live,
265            read_highest_indexed: &read,
266        };
267        history_caught_up(max_lag, Some(indexes)).unwrap()
268    }
269
270    // Without `max_history_lag` the history cohort does not affect health and
271    // its height is never read, so neither a mid-backfill history cohort nor a
272    // failing history read can take the node out of rotation. A node without
273    // rpc indexes is likewise unaffected.
274    #[test]
275    fn history_ignored_when_unset() {
276        let read = || -> StorageResult<Option<u64>> {
277            panic!("history height read with max_history_lag unset")
278        };
279        let indexes = IndexHeights {
280            highest_live_indexed: Some(1_000),
281            read_highest_indexed: &read,
282        };
283        assert!(history_caught_up(None, Some(indexes)).unwrap());
284        assert!(history_caught_up(None, None).unwrap());
285    }
286
287    // A node without rpc indexes reports its executed checkpoint as the tip,
288    // which can run ahead of what indexed replicas have reached: unhealthy
289    // whenever `max_history_lag` is set, however large.
290    #[test]
291    fn history_not_caught_up_without_indexes() {
292        assert!(!history_caught_up(Some(u64::MAX), None).unwrap());
293        assert!(!history_caught_up(Some(0), None).unwrap());
294    }
295
296    // Every pipeline is within (or at) the allowed lag of the live frontier:
297    // healthy.
298    #[test]
299    fn history_caught_up_within_lag() {
300        assert!(history_ok(Some(10), Some(1_000), Some(1_000))); // no lag
301        assert!(history_ok(Some(10), Some(1_000), Some(990))); // exactly at the bound
302        assert!(history_ok(Some(10), Some(1_000), Some(995))); // inside the bound
303    }
304
305    // The two heights are read separately, so the live cohort can advance
306    // between the reads and leave the all-pipelines height above the live one:
307    // that saturates to zero lag rather than underflowing.
308    #[test]
309    fn history_caught_up_when_above_live() {
310        assert!(history_ok(Some(10), Some(1_000), Some(1_005)));
311        assert!(history_ok(Some(0), Some(1_000), Some(1_005)));
312    }
313
314    // History is backfilling far behind the live frontier (e.g. after a
315    // snapshot restore): unhealthy.
316    #[test]
317    fn history_not_caught_up_beyond_lag() {
318        assert!(!history_ok(Some(10), Some(1_000), Some(989))); // one past the bound
319        assert!(!history_ok(Some(10), Some(1_000_000), Some(5))); // mid-backfill
320    }
321
322    // A history pipeline with no watermark yet makes the all-pipelines height
323    // `None`: unhealthy, however large the allowed lag.
324    #[test]
325    fn history_not_caught_up_when_uncommitted() {
326        assert!(!history_ok(Some(u64::MAX), Some(1_000), None));
327        assert!(!history_ok(Some(u64::MAX), None, None));
328    }
329
330    // A failure reading the all-pipelines height propagates rather than being
331    // read as healthy.
332    #[test]
333    fn history_read_error_propagates() {
334        let read = || -> StorageResult<Option<u64>> { Err(StorageError::custom("read failed")) };
335        let indexes = IndexHeights {
336            highest_live_indexed: Some(1_000),
337            read_highest_indexed: &read,
338        };
339        assert!(history_caught_up(Some(10), Some(indexes)).is_err());
340    }
