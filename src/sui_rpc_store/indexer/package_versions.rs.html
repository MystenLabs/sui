4//! Sequential pipeline that populates the
5//! [`schema::package_versions`](crate::schema::package_versions)
6//! CF: one row per `(original_package_id, version)` published in
7//! the checkpoint, plus the matching
8//! [`schema::packages_by_checkpoint`](crate::schema::packages_by_checkpoint)
9//! row in the same commit batch, so the checkpoint-ordered index
10//! cannot drift from `package_versions`.
11//!
12//! Each row records the storage id and the checkpoint at which the
13//! version was published. As a history-cohort member this CF is both
14//! restored and backfilled: the live-set restore writes its rows via
15//! [`store_restored`](crate::schema::package_versions::store_restored),
16//! leaving the publish checkpoint unset (a restore floor), and the
17//! embedded backfill of `(L, T]` then re-publishes every version
18//! created in that window with its real publish checkpoint, overwriting
19//! the floor. Versions that predate the available window are never
20//! re-published, so they keep their floor and read as having always
21//! existed.
22//!
23//! The live set holds only the latest version of a package that is
24//! upgraded in place under one id (the system packages), so the restore
25//! misses the version it had when the window opened if it was upgraded
26//! in the window. The backfill recovers it from the upgrade's input:
27//! an input package version with no row was published before the
28//! history this store holds, so it is written like a restore floor. An
29//! input version that already has a row (published in the window, or
30//! by a from-genesis build), or whose publish this run has staged but
31//! the store has not written yet, is left as is.
32//!
33//! Publishes are pure puts — packages are immutable once written, so a
34//! later publish at the same `(original_id, version)` (which would
35//! itself be a chain-level error) deterministically overwrites
36//! the prior storage id rather than dueling with it. The same
37//! overwrite is what lets the backfill replace a restore floor with
38//! the real publish checkpoint.
39//!
40//! The live-set restore writes no `packages_by_checkpoint` rows: a
41//! restore floor has no publish checkpoint to key it by. The backfill
42//! writes that index row when it re-publishes the version with its
43//! real checkpoint.

68/// Pipeline marker for `package_versions`.
69#[derive(Default)]
70pub struct PackageVersions {
71    /// The restore anchor `T`, or `None` for a store that was never restored. Upgrade inputs
72    /// only matter inside the backfill window `[L, T]` (see the module docs), so the processor
73    /// only scans checkpoints at or below `T` for them.
74    anchor: Option<u64>,
75    /// The `(original_id, version)` of every system package version whose publish this run has
76    /// staged. The synchronizer can still hold an earlier checkpoint's batch, unwritten, when a
77    /// later checkpoint's commit stages, so an upgrade input can miss its publish in the store;
78    /// this set keeps such an input from overwriting the publish with a restore floor. Only
79    /// packages upgraded in place are ever upgrade inputs, and only system packages are upgraded
80    /// in place, so the set holds only those and stays small. The [`Mutex`] only guards interior
81    /// mutation behind `&self`; commits are sequential, so it is never contended.
82    staged_system_publishes: Mutex<HashSet<(ObjectID, u64)>>,
83}
84
85impl PackageVersions {
86    /// Marker for the tip/backfill registration, carrying the restore anchor `T` (or `None` for a
87    /// store that was never restored) so the processor scopes its upgrade-input scan to the
88    /// backfill window `[L, T]`.
89    pub fn with_anchor(anchor: Option<u64>) -> Self {
90        Self {
91            anchor,
92            ..Self::default()
93        }
94    }
95}

97pub enum Row {
98    /// A package version published in `checkpoint`.
99    Published {
100        original_id: ObjectID,
101        version: u64,
102        storage_id: ObjectID,
103        checkpoint: u64,
104    },
105    /// A package version upgraded in place, seen as its upgrade's input. Recorded with no
106    /// publish checkpoint if the store has no row for it yet.
107    UpgradeInput {
108        original_id: ObjectID,
109        version: u64,
110        storage_id: ObjectID,
111    },
112}
113
114#[async_trait]
115impl Processor for PackageVersions {
116    const NAME: &'static str = "package_versions";
117    type Value = Row;
118
119    async fn process(&self, checkpoint: &Arc<Checkpoint>) -> anyhow::Result<Vec<Row>> {
120        let cp = checkpoint.summary.sequence_number;
121        let mut rows = Vec::new();
122        // Packages only appear as inputs when they are upgraded in place; ordinary packages are
123        // immutable and never change. An upgrade input only lacks a row for a version the
124        // restore's live set missed, which was live when the backfill window `[L, T]` opened and
125        // so is an input at or below `T`. Past `T`, or in a store that was never restored, every
126        // input version already has a row, so the scan is skipped.
127        if self.anchor.is_some_and(|t| cp <= t) {
128            for (_, (object, _)) in checkpoint_input_objects(checkpoint)? {
129                if let Data::Package(pkg) = &object.data {
130                    rows.push(Row::UpgradeInput {
131                        original_id: pkg.original_package_id(),
132                        version: pkg.version().value(),
133                        storage_id: pkg.id(),
134                    });
135                }
136            }
137        }
138        for (_, (object, _)) in checkpoint_output_objects(checkpoint)? {
139            if let Data::Package(pkg) = &object.data {
140                rows.push(Row::Published {
141                    original_id: pkg.original_package_id(),
142                    version: pkg.version().value(),
143                    storage_id: pkg.id(),
144                    checkpoint: cp,
145                });
146            }
147        }
148        Ok(rows)
149    }
150}

152impl Restore for PackageVersions {
153    type Schema = RpcStoreSchema;
154
155    fn restore(
156        &self,
157        schema: &Self::Schema,
158        object: &Object,
159        batch: &mut Batch,
160    ) -> anyhow::Result<()> {
161        // Every version of an ordinary package lives forever as its
162        // own immutable object, so the live set holds all of them. A
163        // package upgraded in place keeps only its latest version
164        // here; the backfill recovers the earlier one (see the module
165        // docs).
166        let Data::Package(pkg) = &object.data else {
167            return Ok(());
168        };
169        let (key, value) = package_versions::store_restored(
170            pkg.original_package_id(),
171            pkg.version().value(),
172            pkg.id(),
173        );
174        batch.put(&schema.package_versions, &key, &value)?;
175        Ok(())
176    }
177}

179#[async_trait]
180impl sequential::Handler for PackageVersions {
181    type Store = Store;
182    type Batch = Vec<Row>;
183
184    fn batch(&self, batch: &mut Self::Batch, values: std::vec::IntoIter<Row>) {
185        batch.extend(values);
186    }
187
188    async fn commit<'a>(
189        &self,
190        batch: &Self::Batch,
191        conn: &mut sui_consistent_store::Connection<'a, Schema>,
192    ) -> anyhow::Result<usize> {
193        let mut staged = self
194            .staged_system_publishes
195            .lock()
196            .expect("staged system publishes mutex poisoned");
197        stage_rows(conn.store.schema(), batch, &mut staged, &mut conn.batch)
198    }
199}
200
201/// Stage `rows` into `batch`, returning how many rows were written. A batch can span many
202/// checkpoints, so an upgrade input can depend on a publish staged earlier in the same batch, or
203/// on one an earlier commit staged that the store has not written yet. `staged` holds the system
204/// package publishes staged so far and gains each one as its row is staged, so an input sees
205/// every publish in the rows before it.
206///
207/// A publish also stages its `packages_by_checkpoint` row in the same batch. An upgrade input
208/// recorded with no publish checkpoint has nothing to key that index by, so it stages none.
209fn stage_rows(
210    schema: &RpcStoreSchema,
211    rows: &[Row],
212    staged: &mut HashSet<(ObjectID, u64)>,
213    batch: &mut Batch,
214) -> anyhow::Result<usize> {
215    let cf = &schema.package_versions;
216    let mut written = 0;
217    for row in rows {
218        let (key, value) = match *row {
219            Row::Published {
220                original_id,
221                version,
222                storage_id,
223                checkpoint,
224            } => {
225                if is_system_package(original_id) {
226                    staged.insert((original_id, version));
227                }
228                let (key, value) =
229                    packages_by_checkpoint::store(checkpoint, original_id, version, storage_id);
230                batch.put(&schema.packages_by_checkpoint, &key, &value)?;
231                package_versions::store(original_id, version, storage_id, checkpoint)
232            }
233            Row::UpgradeInput {
234                original_id,
235                version,
236                storage_id,
237            } => {
238                let (key, value) =
239                    package_versions::store_restored(original_id, version, storage_id);
240                if staged.contains(&(original_id, version)) || cf.get(&key)?.is_some() {
241                    continue;
242                }
243                (key, value)
244            }
245        };
246        batch.put(cf, &key, &value)?;
247        written += 1;
248    }
249    Ok(written)
250}

657    #[test]
658    fn upgrade_input_finds_a_publish_earlier_in_the_same_batch() {
659        let dir = tempfile::tempdir().unwrap();
660        let (db, schema) = Db::open::<RpcStoreSchema>(dir.path(), DbOptions::default()).unwrap();
661        let original_id = ObjectID::from_single_byte(2);
662
663        // Checkpoint 10 publishes v5, and checkpoint 11 upgrades it in place to v6.
664        let rows = [
665            Row::Published {
666                original_id,
667                version: 5,
668                storage_id: original_id,
669                checkpoint: 10,
670            },
671            Row::UpgradeInput {
672                original_id,
673                version: 5,
674                storage_id: original_id,
675            },
676            Row::Published {
677                original_id,
678                version: 6,
679                storage_id: original_id,
680                checkpoint: 11,
681            },
682        ];
683        let mut batch = db.batch();
684        assert_eq!(
685            stage_rows(&schema, &rows, &mut HashSet::new(), &mut batch).unwrap(),
686            2
687        );
688        batch.commit().unwrap();
689
690        assert_eq!(
691            stored(&schema, original_id),
692            vec![(5, Some(10)), (6, Some(11))]
693        );
694    }
