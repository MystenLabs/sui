4//! `(checkpoint, original_package_id, version)` → `storage_id`.
5//!
6//! Lists package versions across every package in the order they were
7//! published: the key is the 8-byte big-endian publish checkpoint, the
8//! 32 raw bytes of the package's original id, then the 8-byte
9//! big-endian version, so key order is checkpoint order. This is the
10//! same key layout as the Bigtable `packages_by_checkpoint` table
11//! (`sui-kvstore`), and backs
12//! [`RpcIndexes::packages_by_checkpoint_iter`](sui_types::storage::RpcIndexes::packages_by_checkpoint_iter).
13//!
14//! The value is the 32-byte storage id holding that version. Items are
15//! identified by `(original_id, version)`, never by storage id alone:
16//! system packages are upgraded in place, so every version of a system
17//! package shares one storage id.
18//!
19//! The CF is a projection of [`super::package_versions`]: one row per
20//! `package_versions` row whose publish checkpoint is known. The
21//! `PackageVersions` pipeline writes both CFs in the same commit batch,
22//! so they cannot drift. Restore-floor rows (publish checkpoint unset)
23//! have no row here. The backfill, which re-indexes checkpoints `L`
24//! through `T`, re-publishes each version published in that window with
25//! its real checkpoint and writes its row here alongside, so once the
26//! backfill reaches `T`, the floor rows left are versions published
27//! before the available window. Like `package_versions`, the CF is never
28//! pruned.
29//!
30//! A database open (`RpcStoreSchema::open`) builds this CF from
31//! `package_versions`, writing the row of every version with a known
32//! publish checkpoint, until one build completes. That builds the CF for
33//! a database created before it existed, and finishes a build a crash
34//! interrupted. The build records its completion in
35//! [`super::packages_by_checkpoint_built`], and later opens skip it; see
36//! `RpcStoreSchema::build_packages_by_checkpoint`.
37

54/// `(checkpoint, original_id, version)`, encoded big-endian so rows
55/// sort by publish checkpoint, then by package, then by version. The
56/// derived `Ord` matches the encoded byte order.
57#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
58pub struct Key {
59    pub checkpoint: u64,
60    pub original_id: ObjectID,
61    pub version: u64,
62}
63
64/// The on-chain storage id holding the keyed package version, stored
65/// as its 32 raw bytes.
66#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
67pub struct StorageId(pub ObjectID);
68
69pub type Value = StorageId;
70
71const KEY_LENGTH: usize = 8 + ObjectID::LENGTH + 8;
72
73impl Encode for Key {
74    fn encode_into<B: BufMut>(&self, buf: &mut B) -> Result<(), EncodeError> {
75        buf.put_slice(&self.checkpoint.to_be_bytes());
76        buf.put_slice(self.original_id.as_ref());
77        buf.put_slice(&self.version.to_be_bytes());
78        Ok(())
79    }
80}
81
82impl Decode for Key {
83    fn decode<B: Buf>(buf: &mut B) -> Result<Self, DecodeError> {
84        if buf.remaining() != KEY_LENGTH {
85            return Err(DecodeError::msg(format!(
86                "expected {KEY_LENGTH} bytes for {NAME} Key, got {}",
87                buf.remaining(),
88            )));
89        }
90        let checkpoint = buf.get_u64();
91        let mut id = [0u8; ObjectID::LENGTH];
92        buf.copy_to_slice(&mut id);
93        let version = buf.get_u64();
94        Ok(Key {
95            checkpoint,
96            original_id: ObjectID::new(id),
97            version,
98        })
99    }
100}

122impl Encode for StorageId {
123    fn encode_into<B: BufMut>(&self, buf: &mut B) -> Result<(), EncodeError> {
124        buf.put_slice(self.0.as_ref());
125        Ok(())
126    }
127}
128
129impl Decode for StorageId {
130    fn decode<B: Buf>(buf: &mut B) -> Result<Self, DecodeError> {
131        let bytes = buf.copy_to_bytes(buf.remaining());
132        Ok(StorageId(decode_storage_id(&bytes)?))
133    }
134}

140/// Build the `(Key, Value)` pair recording that version `version` of
141/// the package originally published at `original_id` was published in
142/// `checkpoint` and lives at the storage id `storage_id`.
143pub fn store(
144    checkpoint: u64,
145    original_id: ObjectID,
146    version: u64,
147    storage_id: ObjectID,
148) -> (Key, Value) {
149    (
150        Key {
151            checkpoint,
152            original_id,
153            version,
154        },
155        StorageId(storage_id),
156    )
157}

159/// `package_versions` rows read and written per batch by the build:
160/// bounds the build's memory and write-batch size regardless of how many
161/// package versions the chain holds.
162const BUILD_CHUNK_ROWS: usize = 10_000;
163
164impl super::RpcStoreSchema {
165    /// Write this CF's row for every [`super::package_versions`] row
166    /// with a known publish checkpoint and record that the build
167    /// completed in [`super::packages_by_checkpoint_built`], returning
168    /// the number of rows written.
169    ///
170    /// `RpcStoreSchema::open` runs it, before any pipeline can write,
171    /// until one build completes. It scans `package_versions` and
172    /// projects each row with a known publish checkpoint to its key here
173    /// (restore floors are skipped, as on the indexing path). A row
174    /// already present is written again unchanged: its key determines
175    /// its value. Work is committed in chunks of [`BUILD_CHUNK_ROWS`];
176    /// because every write is a put of a row derived from
177    /// `package_versions`, a crash between chunks leaves a subset of the
178    /// right rows, and the next open writes them all again. The
179    /// completion row is written in the last chunk's batch, so it exists
180    /// only once every row does.
181    ///
182    /// Later opens can skip the build because, once it has run, every
183    /// write of a `package_versions` row with a known publish checkpoint
184    /// writes its row here in the same batch.
185    ///
186    /// Rows here with no matching `package_versions` row are left alone.
187    /// The pipeline writes both CFs together and neither is pruned, so
188    /// such rows do not arise.
189    pub(crate) fn build_packages_by_checkpoint(&self, db: &Db) -> Result<usize, Error> {
190        self.build_packages_by_checkpoint_in_chunks(db, BUILD_CHUNK_ROWS)
191    }
192
193    fn build_packages_by_checkpoint_in_chunks(
194        &self,
195        db: &Db,
196        chunk_rows: usize,
197    ) -> Result<usize, Error> {
198        let mut written = 0;
199        let mut chunk = Vec::with_capacity(chunk_rows);
200        for row in self.package_versions.iter(..)? {
201            let (key, value) = row?;
202            let info = value.into_inner();
203            let Some(checkpoint) = info.checkpoint else {
204                continue;
205            };
206            let storage_id = decode_storage_id(&info.storage_id)?;
207            chunk.push(store(checkpoint, key.original_id, key.version, storage_id));
208            if chunk.len() == chunk_rows {
209                self.write_rows(db, &chunk, false)?;
210                written += chunk.len();
211                chunk.clear();
212            }
213        }
214        self.write_rows(db, &chunk, true)?;
215        Ok(written + chunk.len())
216    }
217
218    /// Write the rows of `chunk` in one batch. If `last` is set, the
219    /// batch also records that the build completed.
220    fn write_rows(&self, db: &Db, chunk: &[(Key, Value)], last: bool) -> Result<(), Error> {
221        let mut batch = db.batch();
222        for (key, value) in chunk {
223            batch.put(&self.packages_by_checkpoint, key, value)?;
224        }
225        if last {
226            batch.put(&self.packages_by_checkpoint_built, &UnitKey, &UnitKey)?;
227        }
228        batch.commit()?;
229        Ok(())
230    }
231}

248    fn fresh_db() -> (tempfile::TempDir, Db, RpcStoreSchema) {
249        let dir = tempfile::tempdir().unwrap();
250        let (db, schema) = Db::open::<RpcStoreSchema>(dir.path(), DbOptions::default()).unwrap();
251        (dir, db, schema)
252    }

254    fn key(checkpoint: u64, original_id: ObjectID, version: u64) -> Key {
255        Key {
256            checkpoint,
257            original_id,
258            version,
259        }
260    }

262    /// Publishes in index order: two packages in checkpoint 5, two
263    /// in-place upgrades of a system package (same storage id) in
264    /// checkpoints 7 and 9, a user-package upgrade sharing checkpoint 9,
265    /// and a last package in checkpoint 12.
266    fn publishes() -> Vec<(Key, ObjectID)> {
267        vec![
268            (key(5, A, 1), A),
269            (key(5, B, 1), B),
270            (key(7, SYSTEM, 2), SYSTEM),
271            (key(9, SYSTEM, 3), SYSTEM),
272            (key(9, A, 2), A_V2),
273            (key(12, C, 1), C),
274        ]
275    }
276
277    fn seed_index(db: &Db, schema: &RpcStoreSchema, rows: &[(Key, ObjectID)]) {
278        let mut batch = db.batch();
279        for (key, storage_id) in rows {
280            batch
281                .put(&schema.packages_by_checkpoint, key, &StorageId(*storage_id))
282                .unwrap();
283        }
284        batch.commit().unwrap();
285    }
286
287    fn seed_package_versions(
288        db: &Db,
289        schema: &RpcStoreSchema,
290        rows: &[(ObjectID, u64, ObjectID, Option<u64>)],
291    ) {
292        let mut batch = db.batch();
293        for &(original_id, version, storage_id, checkpoint) in rows {
294            let (key, value) = match checkpoint {
295                Some(checkpoint) => {
296                    package_versions::store(original_id, version, storage_id, checkpoint)
297                }
298                None => package_versions::store_restored(original_id, version, storage_id),
299            };
300            batch.put(&schema.package_versions, &key, &value).unwrap();
301        }
302        batch.commit().unwrap();
303    }
304
305    fn index_rows(schema: &RpcStoreSchema) -> Vec<(Key, ObjectID)> {
306        schema
307            .packages_by_checkpoint
308            .iter(..)
309            .unwrap()
310            .map(|row| {
311                let (key, StorageId(storage_id)) = row.unwrap();
312                (key, storage_id)
313            })
314            .collect()
315    }
316
317    /// Delete the build's completion row, as on a database last opened
318    /// by a binary that predates it or whose build a crash interrupted.
319    fn forget_build(db: &Db, schema: &RpcStoreSchema) {
320        let mut batch = db.batch();
321        batch
322            .delete(&schema.packages_by_checkpoint_built, &UnitKey)
323            .unwrap();
324        batch.commit().unwrap();
325        assert!(!schema.packages_by_checkpoint_built().unwrap());
326    }
327
328    /// `publishes()` as `package_versions` rows, each with its real
329    /// publish checkpoint.
330    fn published_package_versions() -> Vec<(ObjectID, u64, ObjectID, Option<u64>)> {
331        publishes()
332            .into_iter()
333            .map(|(key, storage_id)| {
334                (
335                    key.original_id,
336                    key.version,
337                    storage_id,
338                    Some(key.checkpoint),
339                )
340            })
341            .collect()
342    }

344    #[test]
345    fn build_indexes_published_versions_and_skips_restore_floors() {
346        let (_dir, db, schema) = fresh_db();
347        // A mid-backfill `package_versions`: `B` and the system
348        // package's first version are still restore floors; the rest
349        // have their real publish checkpoints.
350        seed_package_versions(
351            &db,
352            &schema,
353            &[
354                (SYSTEM, 1, SYSTEM, None),
355                (SYSTEM, 2, SYSTEM, Some(7)),
356                (SYSTEM, 3, SYSTEM, Some(9)),
357                (A, 1, A, Some(5)),
358                (A, 2, A_V2, Some(9)),
359                (B, 1, B, None),
360                (C, 1, C, Some(12)),
361            ],
362        );
363
364        assert_eq!(schema.build_packages_by_checkpoint(&db).unwrap(), 5);
365        let mut expected = publishes();
366        expected.retain(|(key, _)| key.original_id != B);
367        assert_eq!(index_rows(&schema), expected);
368
369        // Rerunning on a built index changes nothing.
370        schema.build_packages_by_checkpoint(&db).unwrap();
371        assert_eq!(index_rows(&schema), expected);
372    }

374    #[test]
375    fn build_completes_a_partially_built_index() {
376        let (_dir, db, schema) = fresh_db();
377        // Every `package_versions` row of the fixture has a real
378        // checkpoint, and the index holds some of them, as a crash
379        // between build chunks would leave it.
380        seed_package_versions(&db, &schema, &published_package_versions());
381        seed_index(&db, &schema, &[publishes()[1], publishes()[4]]);
382        forget_build(&db, &schema);
383
384        // Six eligible rows in chunks of four exercise both a full-chunk
385        // flush and the final partial chunk; the missing rows are
386        // written, and the build records that it completed.
387        schema
388            .build_packages_by_checkpoint_in_chunks(&db, 4)
389            .unwrap();
390        assert_eq!(index_rows(&schema), publishes());
391        assert!(schema.packages_by_checkpoint_built().unwrap());
392
393        // In chunks of three the last chunk is empty; the build still
394        // records that it completed.
395        forget_build(&db, &schema);
396        schema
397            .build_packages_by_checkpoint_in_chunks(&db, 3)
398            .unwrap();
399        assert!(schema.packages_by_checkpoint_built().unwrap());
400    }

409    #[test]
410    fn opening_builds_the_index_once() {
411        let dir = tempfile::tempdir().unwrap();
412        let open = || Db::open::<RpcStoreSchema>(dir.path(), DbOptions::default()).unwrap();
413        {
414            // `package_versions` populated without the index or its
415            // completion row (as by a binary that predates both).
416            let (db, schema) = open();
417            seed_package_versions(
418                &db,
419                &schema,
420                &[(A, 1, A, Some(5)), (A, 2, A_V2, Some(9)), (B, 1, B, None)],
421            );
422            forget_build(&db, &schema);
423            assert!(index_rows(&schema).is_empty());
424        }
425        {
426            let (db, schema) = open();
427            assert_eq!(
428                index_rows(&schema),
429                vec![(key(5, A, 1), A), (key(9, A, 2), A_V2)],
430            );
431            assert!(schema.packages_by_checkpoint_built().unwrap());
432            // Drop an index row and add a `package_versions` row with no
433            // index row. A rebuild would restore the first and add the
434            // second.
435            let mut batch = db.batch();
436            batch
437                .delete(&schema.packages_by_checkpoint, &key(9, A, 2))
438                .unwrap();
439            batch.commit().unwrap();
440            seed_package_versions(&db, &schema, &[(C, 1, C, Some(12))]);
441        }
442
443        // The completion row is present, so the open does not rebuild.
444        let (_db, schema) = open();
445        assert_eq!(index_rows(&schema), vec![(key(5, A, 1), A)]);
446        assert!(schema.packages_by_checkpoint_built().unwrap());
447    }

450    fn opening_completes_an_interrupted_build() {
451        let dir = tempfile::tempdir().unwrap();
452        let open = || Db::open::<RpcStoreSchema>(dir.path(), DbOptions::default()).unwrap();
453        {
454            // Some chunks of the build committed, and the crash came
455            // before the last chunk wrote the completion row.

464        assert!(schema.packages_by_checkpoint_built().unwrap());
