4//! `()` → `()`.
5//!
6//! Singleton row recording that the startup build of
7//! [`super::packages_by_checkpoint`] from [`super::package_versions`] has
8//! completed. The row's presence is the record; its value is empty.
9//!
10//! The build writes this row in the same batch as its last chunk, so a
11//! crash mid-build leaves no row and the next open finishes the build.
12//! `RpcStoreSchema::open` skips the build while the row exists: after the
13//! build, every writer of a `package_versions` row with a known publish
14//! checkpoint writes its `packages_by_checkpoint` row in the same batch,
15//! so rescanning would find nothing to write.
16//!
17//! The row lives in its own CF, rather than under a made-up pipeline key
18//! in the framework's `__restore` or `__watermark` CFs, because code
19//! walks every `__watermark` entry (the pruner's epoch floor) and would
20//! read such an entry as a pipeline.
21
22use sui_consistent_store::error::Error;
23use sui_consistent_store::reader::Reader;
24
25use crate::schema::primitives::UnitKey;
26
27pub const NAME: &str = "packages_by_checkpoint_built";
28
29pub type Key = UnitKey;
30pub type Value = UnitKey;
31
32pub fn options(resolver: &sui_consistent_store::CfOptionsResolver) -> rocksdb::Options {
33    resolver.options(NAME)
34}
35
36impl<R: Reader> super::RpcStoreSchema<R> {
37    /// Whether the startup build of `packages_by_checkpoint` has
38    /// completed on this database.
39    pub fn packages_by_checkpoint_built(&self) -> Result<bool, Error> {
40        Ok(self.packages_by_checkpoint_built.get(&UnitKey)?.is_some())
41    }
42}
