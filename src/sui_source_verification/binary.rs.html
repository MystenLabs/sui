19/// Resolve the path to a `sui` binary for `version`, downloading and caching it under the user's
20/// cache directory (one subdirectory per version) if necessary.
21///
22/// If `version` is the version of the running binary, the running executable is used directly
23/// (avoiding a redundant download of the version already in hand). This is also the precache /
24/// warm entry point: calling it ahead of time populates the cache so later runs need no network.
25pub fn ensure_binary(version: &str) -> Result<PathBuf, Error> {
26    if version == CURRENT_COMPILER_VERSION {
27        return std::env::current_exe().map_err(|e| Error::BinaryDownload {
28            version: version.to_string(),
29            message: format!("could not locate the running executable: {e}"),
30        });
31    }
32
33    let platform = detect_platform(version)?;
34    let binary_name = platform.binary_name();
35
36    let cache_root = binary_cache_dir();
37    let version_dir = cache_root.join(version);
38    let canonical = version_dir.join("target").join("release").join(binary_name);
39
40    if canonical.exists() {
41        touch_last_used(&version_dir);
42        return Ok(canonical);
43    }
44
45    download_and_install(
46        version,
47        platform.artifact_str(),
48        binary_name,
49        &cache_root,
50        &version_dir,
51    )
52    .map_err(|e| Error::BinaryDownload {
53        version: version.to_string(),
54        message: e.to_string(),
55    })?;
56
57    touch_last_used(&version_dir);
58    evict_stale(&cache_root, cache_limit());
59
60    Ok(canonical)
61}

63/// Name of the per-version marker file recording when a cached binary was last used, for LRU
64/// eviction.
65const LAST_USED_FILE: &str = ".last_used";
66
67/// The number of downloaded `sui` binaries to keep cached. After each install the least-recently-used
68/// versions beyond this are evicted, so the cache stays small in disk-scarce environments (enclaves).
69/// Overridable with `SUI_BINARY_CACHE_LIMIT`.
70const DEFAULT_CACHE_LIMIT: usize = 5;
71
72fn cache_limit() -> usize {
73    std::env::var("SUI_BINARY_CACHE_LIMIT")
74        .ok()
75        .and_then(|v| v.parse().ok())
76        .unwrap_or(DEFAULT_CACHE_LIMIT)
77}

79/// Record that the binary under `version_dir` was just used. Best-effort: a failure here (for
80/// example, a read-only cache) must not fail verification.
81fn touch_last_used(version_dir: &Path) {
82    let now = std::time::SystemTime::now()
83        .duration_since(std::time::UNIX_EPOCH)
84        .map(|d| d.as_secs())
85        .unwrap_or(0);
86    let _ = fs::write(version_dir.join(LAST_USED_FILE), now.to_string());
87}
88
89/// The recorded last-used timestamp for a cached version, or `0` when it has none (a directory
90/// predating this bookkeeping), which sorts it as the oldest.
91fn last_used(version_dir: &Path) -> u64 {
92    fs::read_to_string(version_dir.join(LAST_USED_FILE))
93        .ok()
94        .and_then(|s| s.trim().parse().ok())
95        .unwrap_or(0)
96}

98/// Evict least-recently-used cached binaries, keeping at most `limit`. Best-effort: eviction failures
99/// do not fail verification. Only whole other-version directories are removed; the version in hand
100/// keeps its own path, and on unix an executing binary survives removal of its directory. In-progress
101/// `.tmp-*` installs (hidden entries) are never touched.
102fn evict_stale(cache_root: &Path, limit: usize) {
103    let Ok(entries) = fs::read_dir(cache_root) else {
104        return;
105    };
106    let mut versions: Vec<(u64, PathBuf)> = entries
107        .flatten()
108        .map(|entry| entry.path())
109        .filter(|path| path.is_dir() && !is_hidden(path))
110        .map(|dir| (last_used(&dir), dir))
111        .collect();
112
113    // Newest first; keep the first `limit`, remove the rest.
114    versions.sort_by_key(|(last_used, _)| std::cmp::Reverse(*last_used));
115    for (_, dir) in versions.into_iter().skip(limit) {
116        let _ = fs::remove_dir_all(&dir);
117    }
118}

120/// Whether a cache entry is hidden (dot-prefixed) — an in-progress `.tmp-*` install rather than a
121/// cached version, so not a candidate for eviction.
122fn is_hidden(path: &Path) -> bool {
123    path.file_name()
124        .and_then(|name| name.to_str())
125        .is_some_and(|name| name.starts_with('.'))
126}

128/// Download the `sui` release tarball for `version`, streaming out just the `sui` binary, and install
129/// it atomically into `version_dir`. The binary is written under a temporary directory that is renamed
130/// into place only once complete, so concurrent installs of the same version cannot observe a partial
131/// tree, and nothing but the `sui` binary is ever written to disk.
132fn download_and_install(
133    version: &str,
134    platform: &str,
135    binary_name: &str,
136    cache_root: &Path,
137    version_dir: &Path,
138) -> anyhow::Result<()> {
139    fs::create_dir_all(cache_root).context("creating binary cache directory")?;
140
141    // Temp dir on the same filesystem as the cache so the final rename is atomic.
142    let tmp = cache_root.join(format!(".tmp-{version}-{}", std::process::id()));
143    let _ = fs::remove_dir_all(&tmp);
144    fs::create_dir_all(&tmp).context("creating temporary install directory")?;
145
146    let result = (|| {
147        // Stream the release archive into the temp tree, writing only the `sui` binary. A release
148        // archive carries every shipped binary (`sui-debug`, `sui-node`, ...), well over a gigabyte,
149        // so streaming keeps everything but `sui` off disk. The completed tree is renamed into place
150        // below, so the install stays atomic.
151        let release_dir = tmp.join("target").join("release");
152        fs::create_dir_all(&release_dir).context("creating release directory")?;
153        let staged_binary = release_dir.join(binary_name);
154        stream_sui_binary(version, platform, &staged_binary)?;
155        set_executable_permission(staged_binary.as_os_str())?;
156        Ok(())
157    })();
158
159    if let Err(e) = result {
160        let _ = fs::remove_dir_all(&tmp);
161        return Err(e);
162    }
163
164    match fs::rename(&tmp, version_dir) {
165        Ok(()) => Ok(()),
166        // Another process installed the same version first; use theirs and drop ours.
167        Err(_)
168            if version_dir
169                .join("target")
170                .join("release")
171                .join(binary_name)
172                .exists() =>
173        {
174            let _ = fs::remove_dir_all(&tmp);
175            Ok(())
176        }
177        Err(e) => {
178            let _ = fs::remove_dir_all(&tmp);
179            Err(anyhow!("installing downloaded binary: {e}"))
180        }
181    }
182}

184/// Download the `sui` release tarball for `version` and stream it, extracting only the `sui` binary
185/// to `dest`.
186fn stream_sui_binary(version: &str, platform: &str, dest: &Path) -> anyhow::Result<()> {
187    if let Some(tarball) = suiup_cached_tarball(version, platform) {
188        debug!("reusing suiup's cached release at {}", tarball.display());
189        let reader = fs::File::open(&tarball).context("opening suiup's cached release")?;
190        return extract_sui_from_stream(reader, version, platform, dest);
191    }
192    let reader = download_reader(version, platform)?;
193    extract_sui_from_stream(reader, version, platform, dest)
194}

196/// The path to a `sui` release tarball already cached by `suiup`, if one exists. `suiup` keeps
197/// release archives under `<cache-dir>/suiup/releases/sui-<net>-v<version>-<platform>.tgz`; reusing
198/// one avoids re-downloading a release the user already has.
199fn suiup_cached_tarball(version: &str, platform: &str) -> Option<PathBuf> {
200    let releases = dirs::cache_dir()?.join("suiup").join("releases");
201    suiup_tarball_in(&releases, version, platform)
202}

204/// The path to a matching `suiup` release tarball under `releases` (checking the mainnet and testnet
205/// naming), or `None` if neither is present.
206fn suiup_tarball_in(releases: &Path, version: &str, platform: &str) -> Option<PathBuf> {
207    ["mainnet", "testnet"]
208        .into_iter()
209        .map(|net| releases.join(format!("sui-{net}-v{version}-{platform}.tgz")))
210        .find(|path| path.exists())
211}

213/// Open a streaming reader over the `sui` release tarball for `version`, trying the mainnet release
214/// first and falling back to the testnet release on a 404.
215fn download_reader(version: &str, platform: &str) -> anyhow::Result<impl io::Read> {
216    let mainnet_url = format!(
217        "https://github.com/MystenLabs/sui/releases/download/mainnet-v{version}/sui-mainnet-v{version}-{platform}.tgz",
218    );
219
220    // Progress goes to stderr so it does not corrupt a `--json` verification result on stdout.
221    eprintln!(
222        "{} sui compiler @ {} (this may take a while)",
223        "DOWNLOADING".bold().green(),
224        version.yellow()
225    );
226
227    let response = match ureq::get(&mainnet_url).call() {
228        Ok(response) => response,
229        Err(ureq::Error::Status(404, _)) => {
230            debug!("no mainnet release for {version}, trying testnet");
231            let testnet_url = format!(
232                "https://github.com/MystenLabs/sui/releases/download/testnet-v{version}/sui-testnet-v{version}-{platform}.tgz",
233            );
234            ureq::get(&testnet_url).call()?
235        }
236        Err(e) => return Err(e.into()),
237    };
238    Ok(response.into_reader())
239}

241/// Read a gzipped tar archive from `reader` and unpack only the `sui` binary to `dest`, discarding
242/// every other entry as it streams. Errors if the archive for `version` contains no `sui` binary.
243fn extract_sui_from_stream(
244    reader: impl io::Read,
245    version: &str,
246    platform: &str,
247    dest: &Path,
248) -> anyhow::Result<()> {
249    let tar = flate2::read::GzDecoder::new(reader);
250    let mut archive = Archive::new(tar);
251    let entries = archive.entries().context("reading release archive")?;
252
253    for entry in entries {
254        let mut entry = entry.context("reading release archive entry")?;
255        let is_sui = {
256            let path = entry.path().context("reading archive entry path")?;
257            matches_sui(&path, platform)
258        };
259        if is_sui {
260            entry.unpack(dest).context("unpacking the sui binary")?;
261            return Ok(());
262        }
263    }
264    Err(anyhow!(
265        "no sui binary found in the {version} release archive"
266    ))
267}

269/// Whether `path`, the path of an entry in a release archive, is the `sui` executable — either the
270/// modern root-level `sui` or the older `target/release/sui-<platform>`. Other shipped binaries
271/// (`sui-node`, `sui-tool`, ...) do not match.
272fn matches_sui(path: &Path, platform: &str) -> bool {
273    let suffix = if platform == "windows-x86_64" {
274        ".exe"
275    } else {
276        ""
277    };
278    let names = [format!("sui{suffix}"), format!("sui-{platform}{suffix}")];
279    path.file_name()
280        .and_then(|n| n.to_str())
281        .is_some_and(|n| names.iter().any(|candidate| candidate == n))
282}

284/// A platform for which `sui` release binaries are published.
285enum Platform {
286    MacosArm64,
287    MacosX86_64,
288    UbuntuX86_64,
289    WindowsX86_64,
290}
291
292impl Platform {
293    /// The platform string used in release download URLs.
294    fn artifact_str(&self) -> &'static str {
295        match self {
296            Platform::MacosArm64 => "macos-arm64",
297            Platform::MacosX86_64 => "macos-x86_64",
298            Platform::UbuntuX86_64 => "ubuntu-x86_64",
299            Platform::WindowsX86_64 => "windows-x86_64",
300        }
301    }
302
303    /// The name of the `sui` executable on this platform.
304    fn binary_name(&self) -> &'static str {
305        match self {
306            Platform::WindowsX86_64 => "sui.exe",
307            _ => "sui",
308        }
309    }
310}

312/// The [`Platform`] for the current OS/architecture, or an error explaining how to sideload a binary
313/// if there is no downloadable release for this platform.
314fn detect_platform(version: &str) -> Result<Platform, Error> {
315    match (std::env::consts::OS, std::env::consts::ARCH) {
316        ("macos", "aarch64") => Ok(Platform::MacosArm64),
317        ("macos", "x86_64") => Ok(Platform::MacosX86_64),
318        ("linux", "x86_64") => Ok(Platform::UbuntuX86_64),
319        ("windows", "x86_64") => Ok(Platform::WindowsX86_64),
320        (os, arch) => Err(Error::BinaryDownload {
321            version: version.to_string(),
322            message: format!(
323                "no downloadable sui {version} release for your platform \
324                 (OS: {os}, architecture: {arch}); pass --toolchain <path> to build with a local binary"
325            ),
326        }),
327    }
328}

330/// Directory under which downloaded `sui` binaries are cached, one subdirectory per version. Uses the
331/// platform cache directory (as other Sui tooling does), falling back to `$MOVE_HOME` if it cannot be

333fn binary_cache_dir() -> PathBuf {

341#[cfg(unix)]
342fn set_executable_permission(path: &OsStr) -> anyhow::Result<()> {
343    use std::os::unix::prelude::PermissionsExt;
344    let mut perms = fs::metadata(path)?.permissions();
345    perms.set_mode(0o755);
346    fs::set_permissions(path, perms)?;
347    Ok(())
348}
349
350#[cfg(not(unix))]
351fn set_executable_permission(_path: &OsStr) -> anyhow::Result<()> {
352    // On Windows an executable is runnable by virtue of its extension, and the freshly-written file
353    // is already owned by the current user, so there is no permission bit to set.
354    Ok(())
355}

363    /// Build a gzipped tar archive from `(path, contents)` entries, as the release download stream
364    /// would deliver it.
365    fn make_tgz(entries: &[(&str, &[u8])]) -> Vec<u8> {
366        let mut tar = Vec::new();
367        {
368            let mut builder = tar::Builder::new(&mut tar);
369            for (name, data) in entries {
370                let mut header = tar::Header::new_gnu();
371                header.set_size(data.len() as u64);
372                header.set_mode(0o755);
373                builder.append_data(&mut header, name, *data).unwrap();
374            }
375            builder.finish().unwrap();
376        }
377        let mut gz = flate2::write::GzEncoder::new(Vec::new(), flate2::Compression::fast());
378        gz.write_all(&tar).unwrap();
379        gz.finish().unwrap()
380    }

431    /// Eviction keeps the most-recently-used versions and leaves an in-progress install untouched.
432    #[test]
433    fn evict_keeps_most_recently_used() {
434        let cache = tempfile::tempdir().unwrap();
435        let seed = |name: &str, last_used: u64| {
436            let dir = cache.path().join(name);
437            fs::create_dir_all(&dir).unwrap();
438            fs::write(dir.join(LAST_USED_FILE), last_used.to_string()).unwrap();
439        };
440        seed("1.0.0", 100);
441        seed("1.1.0", 300);
442        seed("1.2.0", 200);
443        seed("1.3.0", 50);
444        // A concurrent install's temp directory must survive eviction despite having no marker.
445        fs::create_dir_all(cache.path().join(".tmp-9.9.9-1")).unwrap();
446
447        evict_stale(cache.path(), 2);
448
449        // The two most recently used remain; the two oldest are gone.
450        assert!(cache.path().join("1.1.0").exists());
451        assert!(cache.path().join("1.2.0").exists());
452        assert!(!cache.path().join("1.0.0").exists());
453        assert!(!cache.path().join("1.3.0").exists());
454        // The in-progress install is never evicted.
455        assert!(cache.path().join(".tmp-9.9.9-1").exists());
456    }
