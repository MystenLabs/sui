26/// The publication metadata a successful [`verify_source`] run relied on: the addresses it checked
27/// against, the toolchain it rebuilt with, and the `sui` binary it used.
28#[derive(Debug, Serialize)]
29#[serde(rename_all = "camelCase")]
30pub struct VerifiedMetadata {
31    /// The package's original (first-published) id.
32    pub original_id: ObjectID,
33    /// The on-chain address whose bytecode the rebuild was compared against.
34    pub published_at: ObjectID,
35    /// The `sui` toolchain version the source was rebuilt with, or `None` when a caller-supplied
36    /// binary was used (its version is not known).
37    pub toolchain_version: Option<String>,
38    /// The path to the `sui` binary used for the rebuild.
39    pub binary_path: PathBuf,
40}
41
42/// How to obtain the `sui` toolchain that rebuilds a package.
43pub enum ToolchainSource {
44    /// Download and cache a release: the publication's recorded version, or `Some(version)` to
45    /// override it.
46    Version(Option<String>),
47    /// Use the `sui` binary already at this path, skipping version resolution and download.
48    Binary(PathBuf),
49}
50
51/// Verify that the Move source package at `source_path` compiles to the on-chain package described
52/// by `publication`, matching both its module bytecode and its linkage. On success, returns the
53/// [`VerifiedMetadata`] the verification relied on.
54///
55/// `toolchain` selects the `sui` binary to rebuild with: a downloaded release (the publication's
56/// recorded version, or an override) or a caller-supplied binary. The package is rebuilt against
57/// `env`, the resulting `0x0` root address is rewritten to the publication's original id, and the
58/// modules and linkage are compared against the package fetched from the publication's published-at
59/// address. `client_config` locates the wallet for releases whose build contacts the network.
60pub async fn verify_source(
61    source_path: &Path,
62    publication: &Publication<SuiFlavor>,
63    toolchain: ToolchainSource,
64    env: &Environment,
65    client: &Client,
66    client_config: Option<&Path>,
67) -> Result<VerifiedMetadata, AggregateError> {
68    // Resolve the binary to rebuild with. A downloaded release has a known version to report; a
69    // caller-supplied binary does not.
70    let (binary, toolchain_version) = match toolchain {
71        ToolchainSource::Binary(path) => (path, None),
72        ToolchainSource::Version(override_) => {
73            let version = resolve_toolchain(
74                publication.metadata.toolchain_version.clone(),
75                source_path,
76                override_,
77            )?;
78            check_toolchain_version(&version)?;
79            let binary = ensure_binary(&version)?;
80            (binary, Some(version))
81        }
82    };
83
84    // Verification is attempted even for packages whose dependencies are not pinned to commit
85    // hashes; only if that attempt fails is the lack of pinning reported, since it explains why the
86    // rebuild could not reproduce what was published. The package is built against the environment
87    // it is being verified on.
88    let generated = build::dump(&binary, source_path, env.name(), client_config)
89        .map_err(|e| explain_unpinned_dependencies(source_path, e.into()))?;
90
91    let published_at = ObjectID::from_address(publication.addresses.published_at.0);
92    let original_id = publication.addresses.original_id.0;
93    let onchain = onchain::fetch(client, published_at).await?;
94
95    // The package the source claims to be (its recorded original id) must be the one actually at
96    // `published_at`, otherwise a source could be verified against an unrelated on-chain package.
97    if onchain.original_id != original_id {
98        return Err(Error::OriginalIdMismatch {
99            recorded: original_id,
100            on_chain: onchain.original_id,
101        }
102        .into());
103    }
104
105    compare::check(client, generated, onchain)
106        .await
107        .map_err(|e| explain_unpinned_dependencies(source_path, e))?;
108
109    Ok(VerifiedMetadata {
110        original_id: ObjectID::from_address(original_id),
111        published_at,
112        toolchain_version,
113        binary_path: binary,
114    })
115}

117/// Verify that the modules already compiled under `source_path/build` match the on-chain package at
118/// `on_chain_id`, without invoking the compiler. Only module bytecode is compared, not linkage. The
119/// caller supplies `on_chain_id` directly (there is no publication metadata to read), so this is a
120/// local-build-versus-chain diff rather than an authenticity check on recorded source.
121pub async fn verify_built(
122    source_path: &Path,
123    on_chain_id: ObjectID,
124    client: &Client,
125) -> Result<(), AggregateError> {
126    let modules = prebuilt::read_modules(source_path)?;
127    let generated = build::GeneratedPackage {
128        modules,
129        dependencies: vec![],
130    };
131    let onchain = onchain::fetch(client, on_chain_id).await?;
132    compare::check(client, generated, onchain).await
133}

135/// Determine which toolchain version to rebuild with.
136///
137/// `recorded` is the version from the package's publication metadata, if any. When it is absent, the
138/// legacy `Move.lock` is consulted (older packages record the version only there). `override_` is
139/// the user's `--toolchain-version`: it is used when nothing is recorded, and otherwise takes
140/// precedence with a warning, so a package whose recorded version cannot be built can still be
141/// rebuilt with a working one.
142fn resolve_toolchain(
143    recorded: Option<String>,
144    source_path: &Path,
145    override_: Option<String>,
146) -> Result<String, Error> {
147    let recorded = recorded.or_else(|| toolchain_version::legacy_move_lock_version(source_path));
148
149    match (override_, recorded) {
150        (Some(override_), Some(recorded)) if override_ != recorded => {
151            eprintln!(
152                "{} rebuilding with toolchain {} instead of the recorded {}",
153                "WARNING".bold().yellow(),
154                override_.yellow(),
155                recorded.yellow(),
156            );
157            Ok(override_)
158        }
159        (Some(version), _) | (None, Some(version)) => Ok(version),
160        (None, None) => Err(Error::ToolchainVersionNotFound),
161    }
162}

164/// Append an explanation for each dependency that is not pinned to a commit hash. Such a package
165/// resolves its dependencies to whatever they point at now rather than at publish time, which is
166/// the usual reason a rebuild neither compiles nor matches.
167fn explain_unpinned_dependencies(source_path: &Path, mut error: AggregateError) -> AggregateError {
168    for moving in pinning::moving_revisions(source_path) {
169        error.0.push(Error::NonReproducibleDependency {
170            dependency: moving.dependency,
171            rev: moving.rev,
172        });
173    }
174    error
175}

177/// Fail up front for the handful of releases known not to work for verification, naming a nearby
178/// release that does — so a package recording such a toolchain gets a precise, actionable error
179/// rather than an opaque download or build failure. Unparseable versions (e.g. a nightly) are let
180/// through to be attempted.
181fn check_toolchain_version(version: &str) -> Result<(), Error> {
182    let parse = || -> Option<(u32, u32, u32)> {
183        let mut parts = version.split('.');
184        let mut next = || parts.next()?.parse::<u32>().ok();
185        Some((next()?, next()?, next()?))
186    };
187    let Some(v) = parse() else { return Ok(()) };
188
189    let (reason, alternative) = if v <= (1, 8, 1) {
190        // No release publishes a binary for this platform this far back.
191        (
192            "no release publishes a binary for this platform at v1.8.1 or earlier",
193            "1.9.0",
194        )
195    } else if (v.0, v.1) == (1, 64) {
196        // v1.64.x pins a framework revision (protocol 108) that is no longer in the repository.
197        (
198            "v1.64.x pins a framework revision that is no longer available",
199            "1.65.2",
200        )
201    } else {
202        return Ok(());
203    };
204
205    Err(Error::UnsupportedToolchain {
206        version: version.to_string(),
207        reason: reason.to_string(),
208        alternative: alternative.to_string(),
209    })
210}

212#[cfg(test)]
213mod tests {
214    use super::check_toolchain_version;
215
216    fn suggestion(version: &str) -> Option<String> {
217        check_toolchain_version(version)
218            .err()
219            .map(|e| e.to_string())
220    }
221
222    #[test]
223    fn flags_known_unusable_versions() {
224        assert!(
225            suggestion("1.8.1")
226                .unwrap()
227                .contains("--toolchain-version 1.9.0")
228        );
229        assert!(
230            suggestion("1.5.0")
231                .unwrap()
232                .contains("--toolchain-version 1.9.0")
233        );
234        assert!(
235            suggestion("1.64.1")
236                .unwrap()
237                .contains("--toolchain-version 1.65.2")
238        );
239        assert!(
240            suggestion("1.64.2")
241                .unwrap()
242                .contains("--toolchain-version 1.65.2")
243        );
244        // Usable and unparseable versions are let through.
245        assert!(suggestion("1.65.2").is_none());
246        assert!(suggestion("1.50.0").is_none());
247        assert!(suggestion("nightly").is_none());
248    }
