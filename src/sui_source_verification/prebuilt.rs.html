12/// Read the root package's already-compiled modules from `package_path/build/<name>/bytecode_modules`
13/// without invoking the compiler, where `<name>` is the package's name from its `Move.toml`. Only the
14/// root package's modules are read; the `dependencies` subdirectory is skipped. Returns an error if
15/// the build directory is missing or empty (the package has not been built).
16pub(crate) fn read_modules(package_path: &Path) -> Result<Vec<CompiledModule>, Error> {
17    let dir = package_path
18        .join("build")
19        .join(package_name(package_path)?)
20        .join("bytecode_modules");
21
22    let entries = fs::read_dir(&dir).map_err(|e| Error::BuildOutputParse {
23        message: format!("could not read build directory {}: {e}", dir.display()),
24    })?;
25
26    let mut modules = vec![];
27    for entry in entries.flatten() {
28        let path = entry.path();
29        // Skips the `dependencies` subdirectory and any non-module files.
30        if path.extension().and_then(|e| e.to_str()) != Some("mv") {
31            continue;
32        }
33        let bytes = fs::read(&path).map_err(|e| Error::BuildOutputParse {
34            message: format!("could not read {}: {e}", path.display()),
35        })?;
36        let module = CompiledModule::deserialize_with_defaults(&bytes).map_err(|e| {
37            Error::BuildOutputParse {
38                message: format!("could not deserialize {}: {e}", path.display()),
39            }
40        })?;
41        modules.push(module);
42    }
43
44    if modules.is_empty() {
45        return Err(Error::BuildOutputParse {
46            message: format!(
47                "no compiled modules in {}; build the package first",
48                dir.display()
49            ),
50        });
51    }
52    Ok(modules)
53}

55/// The root package's name, read from `Move.toml`; it names the `build/<name>` subdirectory.
56fn package_name(package_path: &Path) -> Result<String, Error> {
57    let manifest_path = package_path.join("Move.toml");
58    let contents = fs::read_to_string(&manifest_path).map_err(|e| Error::BuildOutputParse {
59        message: format!("could not read {}: {e}", manifest_path.display()),
60    })?;
61
62    #[derive(Deserialize)]
63    struct Package {
64        name: String,
65    }
66    #[derive(Deserialize)]
67    struct Manifest {
68        package: Package,
69    }
70
71    let manifest: Manifest = toml::from_str(&contents).map_err(|e| Error::BuildOutputParse {
72        message: format!("could not parse {}: {e}", manifest_path.display()),
73    })?;
74    Ok(manifest.package.name)
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80    use move_binary_format::file_format::empty_module;
81    use move_binary_format::file_format_common::VERSION_MAX;
82
83    fn write_module(path: &Path) {
84        let mut bytes = vec![];
85        empty_module()
86            .serialize_with_version(VERSION_MAX, &mut bytes)
87            .unwrap();
88        fs::write(path, bytes).unwrap();
89    }
90
91    /// Reads the root package's `.mv` modules and ignores the `dependencies` subdirectory.
92    #[test]
93    fn reads_root_modules_only() {
94        let tmp = tempfile::tempdir().unwrap();
95        let pkg = tmp.path();
96        fs::write(pkg.join("Move.toml"), "[package]\nname = \"demo\"\n").unwrap();
97
98        let modules = pkg.join("build").join("demo").join("bytecode_modules");
99        fs::create_dir_all(modules.join("dependencies")).unwrap();
100        write_module(&modules.join("demo.mv"));
101        write_module(&modules.join("dependencies").join("dep.mv"));
102
103        assert_eq!(read_modules(pkg).unwrap().len(), 1);
104    }
105
106    /// Errors when the package has not been built.
107    #[test]
108    fn errors_when_unbuilt() {
109        let tmp = tempfile::tempdir().unwrap();
110        fs::write(tmp.path().join("Move.toml"), "[package]\nname = \"demo\"\n").unwrap();
111        assert!(read_modules(tmp.path()).is_err());
112    }
113}
