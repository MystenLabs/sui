46pub type WeightUnit = u8;
47pub type ThresholdUnit = u16;

49pub const MAX_SIGNER_IN_MULTISIG: usize = 10;

51/// The struct that contains signatures and public keys necessary for authenticating a MultiSig.
52#[serde_as]
53#[derive(Debug, Serialize, Deserialize, Clone, JsonSchema)]
54pub struct MultiSig {
55    /// The plain signature encoded with signature scheme.
56    sigs: Vec<CompressedSignature>,
57    /// A bitmap that indicates the position of which public key the signature should be authenticated with.
58    bitmap: BitmapUnit,
59    /// The public key encoded with each public key with its signature scheme used along with the corresponding weight.
60    multisig_pk: MultiSigPublicKey,
61    /// A bytes representation of [struct MultiSig]. This helps with implementing [trait AsRef<[u8]>].
62    #[serde(skip)]
63    bytes: OnceCell<Vec<u8>>,
64}
65
66/// Necessary trait for [struct SenderSignedData].
67impl PartialEq for MultiSig {
68    fn eq(&self, other: &Self) -> bool {
69        self.sigs == other.sigs
70            && self.bitmap == other.bitmap
71            && self.multisig_pk == other.multisig_pk
72    }
73}

85impl AuthenticatorTrait for MultiSig {
86    fn verify_user_authenticator_epoch(
87        &self,
88        epoch_id: EpochId,
89        max_epoch_upper_bound_delta: Option<u64>,
90    ) -> Result<(), SuiError> {
91        // If there is any zkLogin signatures, filter and check epoch for each.
92        // TODO: call this on all sigs to avoid future lapses
93        self.get_zklogin_sigs()?.iter().try_for_each(|s| {
94            s.verify_user_authenticator_epoch(epoch_id, max_epoch_upper_bound_delta)
95        })
96    }
97
98    fn verify_claims<T>(
99        &self,
100        value: &IntentMessage<T>,
101        multisig_address: SuiAddress,
102        verify_params: &VerifyParams,
103        zklogin_inputs_cache: Arc<VerifiedDigestCache<ZKLoginInputsDigest>>,
104    ) -> Result<(), SuiError>
105    where
106        T: Serialize,
107    {
108        self.multisig_pk
109            .validate()
110            .map_err(|_| SuiErrorKind::InvalidSignature {
111                error: "Invalid multisig pubkey".to_string(),
112            })?;
113
114        // Additional validation on zkLogin public identifier struct if flag is enabled.
115        if verify_params.validate_zklogin_public_identifier {
116            for (pk, _weight) in &self.multisig_pk.pk_map {
117                if let PublicKey::ZkLogin(zklogin_pk) = pk {
118                    zklogin_pk
119                        .validate()
120                        .map_err(|e| SuiErrorKind::InvalidSignature {
121                            error: format!("Invalid zkLogin pk in multisig: {}", e),
122                        })?;
123                }
124            }
125        }
126
127        if SuiAddress::from(&self.multisig_pk) != multisig_address {
128            return Err(SuiErrorKind::InvalidSignature {
129                error: "Invalid address derived from pks".to_string(),
130            }
131            .into());
132        }
133
134        if self.has_zklogin_sigs() && !verify_params.accept_zklogin_in_multisig {
135            return Err(SuiErrorKind::InvalidSignature {
136                error: "zkLogin sig not supported inside multisig".to_string(),
137            }
138            .into());
139        }
140
141        if self.has_passkey_sigs() && !verify_params.accept_passkey_in_multisig {
142            return Err(SuiErrorKind::InvalidSignature {
143                error: "Passkey sig not supported inside multisig".to_string(),
144            }
145            .into());
146        }
147
148        // Same shape as the passkey/zkLogin checks: gates the member *signing*,
149        // not membership, so hybrid committees keep working on their classical
150        // members until the flag turns on.
151        if self.has_mldsa65_sigs() && !verify_params.accept_mldsa65_in_multisig {
152            return Err(SuiErrorKind::InvalidSignature {
153                error: "ML-DSA-65 sig not supported inside multisig".to_string(),
154            }
155            .into());
156        }
157
158        let mut weight_sum: u16 = 0;
159        let mut hasher = DefaultHash::default();
160        bcs::serialize_into(&mut hasher, &value).expect("Message serialization should not fail");
161        let digest = hasher.finalize().digest;
162        // Verify each signature against its corresponding signature scheme and public key.
163        // TODO: further optimization can be done because multiple Ed25519 signatures can be batch verified.
164        for (sig, i) in self.sigs.iter().zip_debug_eq(as_indices(self.bitmap)?) {
165            let (subsig_pubkey, weight) =
166                self.multisig_pk
167                    .pk_map
168                    .get(i as usize)
169                    .ok_or(SuiErrorKind::InvalidSignature {
170                        error: "Invalid public keys index".to_string(),
171                    })?;
172            let res = match sig {
173                CompressedSignature::Ed25519(s) => {
174                    if verify_params.additional_multisig_checks
175                        && !matches!(subsig_pubkey.scheme(), SignatureScheme::ED25519)
176                    {
177                        return Err(SuiErrorKind::InvalidSignature {
178                            error: format!(
179                                "Invalid sig for pk={} address={:?} error=signature/pubkey type mismatch",
180                                subsig_pubkey.encode_base64(),
181                                SuiAddress::from(subsig_pubkey)
182                            ),
183                        }.into());
184                    }
185                    let pk =
186                        Ed25519PublicKey::from_bytes(subsig_pubkey.as_ref()).map_err(|_| {
187                            SuiErrorKind::InvalidSignature {
188                                error: "Invalid ed25519 pk bytes".to_string(),
189                            }
190                        })?;
191                    pk.verify(
192                        &digest,
193                        &s.try_into().map_err(|_| SuiErrorKind::InvalidSignature {
194                            error: "Invalid ed25519 signature bytes".to_string(),
195                        })?,
196                    )
197                }
198                CompressedSignature::Secp256k1(s) => {
199                    if verify_params.additional_multisig_checks
200                        && !matches!(subsig_pubkey.scheme(), SignatureScheme::Secp256k1)
201                    {
202                        return Err(SuiErrorKind::InvalidSignature {
203                            error: format!(
204                                "Invalid sig for pk={} address={:?} error=signature/pubkey type mismatch",
205                                subsig_pubkey.encode_base64(),
206                                SuiAddress::from(subsig_pubkey)
207                            ),
208                        }.into());
209                    }
210                    let pk =
211                        Secp256k1PublicKey::from_bytes(subsig_pubkey.as_ref()).map_err(|_| {
212                            SuiErrorKind::InvalidSignature {
213                                error: "Invalid k1 pk bytes".to_string(),
214                            }
215                        })?;
216                    pk.verify(
217                        &digest,
218                        &s.try_into().map_err(|_| SuiErrorKind::InvalidSignature {
219                            error: "Invalid k1 signature bytes".to_string(),
220                        })?,
221                    )
222                }
223                CompressedSignature::Secp256r1(s) => {
224                    if verify_params.additional_multisig_checks
225                        && !matches!(subsig_pubkey.scheme(), SignatureScheme::Secp256r1)
226                    {
227                        return Err(SuiErrorKind::InvalidSignature {
228                            error: format!(
229                                "Invalid sig for pk={} address={:?} error=signature/pubkey type mismatch",
230                                subsig_pubkey.encode_base64(),
231                                SuiAddress::from(subsig_pubkey)
232                            ),
233                        }.into());
234                    }
235                    let pk =
236                        Secp256r1PublicKey::from_bytes(subsig_pubkey.as_ref()).map_err(|_| {
237                            SuiErrorKind::InvalidSignature {
238                                error: "Invalid r1 pk bytes".to_string(),
239                            }
240                        })?;
241                    pk.verify(
242                        &digest,
243                        &s.try_into().map_err(|_| SuiErrorKind::InvalidSignature {
244                            error: "Invalid r1 signature bytes".to_string(),
245                        })?,
246                    )
247                }
248                CompressedSignature::MLDSA65(s) => {
249                    if verify_params.additional_multisig_checks
250                        && !matches!(subsig_pubkey.scheme(), SignatureScheme::MLDSA65)
251                    {
252                        return Err(SuiErrorKind::InvalidSignature {
253                            error: format!(
254                                "Invalid sig for pk={} address={:?} error=signature/pubkey type mismatch",
255                                subsig_pubkey.encode_base64(),
256                                SuiAddress::from(subsig_pubkey)
257                            ),
258                        }.into());
259                    }
260                    let pk =
261                        MLDSA65PublicKey::from_bytes(subsig_pubkey.as_ref()).map_err(|_| {
262                            SuiErrorKind::InvalidSignature {
263                                error: "Invalid mldsa65 pk bytes".to_string(),
264                            }
265                        })?;
266                    pk.verify(
267                        &digest,
268                        &MLDSA65Signature::from_bytes(&s.0).map_err(|_| {
269                            SuiErrorKind::InvalidSignature {
270                                error: "Invalid mldsa65 signature bytes".to_string(),
271                            }
272                        })?,
273                    )
274                }
275                CompressedSignature::ZkLogin(z) => {
276                    if verify_params.additional_multisig_checks
277                        && !matches!(
278                            subsig_pubkey.scheme(),
279                            SignatureScheme::ZkLoginAuthenticator
280                        )
281                    {
282                        return Err(SuiErrorKind::InvalidSignature {
283                            error: format!(
284                                "Invalid sig for pk={} address={:?} error=signature/pubkey type mismatch",
285                                subsig_pubkey.encode_base64(),
286                                SuiAddress::from(subsig_pubkey)
287                            ),
288                        }.into());
289                    }
290                    let authenticator = ZkLoginAuthenticator::from_bytes(&z.0).map_err(|_| {
291                        SuiErrorKind::InvalidSignature {
292                            error: "Invalid zklogin authenticator bytes".to_string(),
293                        }
294                    })?;
295                    authenticator
296                        .verify_claims(
297                            value,
298                            SuiAddress::from(subsig_pubkey),
299                            verify_params,
300                            zklogin_inputs_cache.clone(),
301                        )
302                        .map_err(|e| FastCryptoError::GeneralError(e.to_string()))
303                }
304                CompressedSignature::Passkey(bytes) => {
305                    let authenticator =
306                        PasskeyAuthenticator::from_bytes(&bytes.0).map_err(|_| {
307                            SuiErrorKind::InvalidSignature {
308                                error: "Invalid passkey authenticator bytes".to_string(),
309                            }
310                        })?;
311                    authenticator
312                        .verify_claims(
313                            value,
314                            SuiAddress::from(subsig_pubkey),
315                            verify_params,
316                            zklogin_inputs_cache.clone(),
317                        )
318                        .map_err(|e| FastCryptoError::GeneralError(e.to_string()))
319                }
320            };
321            if res.is_ok() {
322                weight_sum += *weight as u16;
323            } else {
324                return res.map_err(|e| {
325                    SuiErrorKind::InvalidSignature {
326                        error: format!(
327                            "Invalid sig for pk={} address={:?} error={:?}",
328                            subsig_pubkey.encode_base64(),
329                            SuiAddress::from(subsig_pubkey),
330                            e.to_string()
331                        ),
332                    }
333                    .into()
334                });
335            }
336        }
337        if weight_sum >= self.multisig_pk.threshold {
338            Ok(())
339        } else {
340            Err(SuiErrorKind::InvalidSignature {
341                error: format!(
342                    "Insufficient weight={:?} threshold={:?}",
343                    weight_sum, self.multisig_pk.threshold
344                ),
345            }
346            .into())
347        }
348    }
349}

351/// Interpret a bitmap of 01s as a list of indices that is set to 1s.
352/// e.g. 22 = 0b10110, then the result is [1, 2, 4].
353pub fn as_indices(bitmap: u16) -> Result<Vec<u8>, SuiError> {
354    if bitmap > MAX_BITMAP_VALUE {
355        return Err(SuiErrorKind::InvalidSignature {
356            error: "Invalid bitmap".to_string(),
357        }
358        .into());
359    }
360    let mut res = Vec::new();
361    for i in 0..10 {
362        if bitmap & (1 << i) != 0 {
363            res.push(i as u8);
364        }
365    }
366    Ok(res)
367}

369impl MultiSig {
370    /// Create MultiSig from its fields without validation
371    pub fn insecure_new(
372        sigs: Vec<CompressedSignature>,
373        bitmap: BitmapUnit,
374        multisig_pk: MultiSigPublicKey,
375    ) -> Self {
376        Self {
377            sigs,
378            bitmap,
379            multisig_pk,
380            bytes: OnceCell::new(),
381        }
382    }
383    /// This combines a list of [enum Signature] `flag || signature || pk` to a MultiSig.
384    /// The order of full_sigs must be the same as the order of public keys in
385    /// [enum MultiSigPublicKey]. e.g. for [pk1, pk2, pk3, pk4, pk5],
386    /// [sig1, sig2, sig5] is valid, but [sig2, sig1, sig5] is invalid.
387    pub fn combine(
388        full_sigs: Vec<GenericSignature>,
389        multisig_pk: MultiSigPublicKey,
390    ) -> Result<Self, SuiError> {
391        multisig_pk
392            .validate()
393            .map_err(|_| SuiErrorKind::InvalidSignature {
394                error: "Invalid multisig public key".to_string(),
395            })?;
396
397        if full_sigs.len() > multisig_pk.pk_map.len() || full_sigs.is_empty() {
398            return Err(SuiErrorKind::InvalidSignature {
399                error: "Invalid number of signatures".to_string(),
400            }
401            .into());
402        }
403        let mut bitmap = 0;
404        let mut sigs = Vec::with_capacity(full_sigs.len());
405        for s in full_sigs {
406            let pk = s.to_public_key()?;
407            let index = multisig_pk
408                .get_index(&pk)
409                .ok_or(SuiErrorKind::IncorrectSigner {
410                    error: format!("pk does not exist: {:?}", pk),
411                })?;
412            if bitmap & (1 << index) != 0 {
413                return Err(SuiErrorKind::InvalidSignature {
414                    error: "Duplicate public key".to_string(),
415                }
416                .into());
417            }
418            bitmap |= 1 << index;
419            sigs.push(s.to_compressed()?);
420        }
421
422        Ok(MultiSig {
423            sigs,
424            bitmap,
425            multisig_pk,
426            bytes: OnceCell::new(),
427        })
428    }
429
430    pub fn init_and_validate(&mut self) -> Result<Self, FastCryptoError> {
431        if self.sigs.len() > self.multisig_pk.pk_map.len()
432            || self.sigs.is_empty()
433            || self.bitmap > MAX_BITMAP_VALUE
434        {
435            return Err(FastCryptoError::InvalidInput);
436        }
437        self.multisig_pk.validate()?;
438        Ok(self.to_owned())
439    }

453    pub fn get_zklogin_sigs(&self) -> Result<Vec<ZkLoginAuthenticator>, SuiError> {
454        let authenticator_as_bytes: Vec<_> = self
455            .sigs
456            .iter()
457            .filter_map(|s| match s {
458                CompressedSignature::ZkLogin(z) => Some(z),
459                _ => None,
460            })
461            .collect();
462        authenticator_as_bytes
463            .iter()
464            .map(|z| {
465                ZkLoginAuthenticator::from_bytes(&z.0).map_err(|_| {
466                    SuiErrorKind::InvalidSignature {
467                        error: "Invalid zklogin authenticator bytes".to_string(),
468                    }
469                    .into()
470                })
471            })
472            .collect()
473    }

498impl ToFromBytes for MultiSig {
499    fn from_bytes(bytes: &[u8]) -> Result<MultiSig, FastCryptoError> {
500        // The first byte matches the flag of MultiSig.
501        if bytes.first().ok_or(FastCryptoError::InvalidInput)? != &SignatureScheme::MultiSig.flag()
502        {
503            return Err(FastCryptoError::InvalidInput);
504        }
505        let mut multisig: MultiSig =
506            bcs::from_bytes(&bytes[1..]).map_err(|_| FastCryptoError::InvalidSignature)?;
507        multisig.init_and_validate()
508    }
509}

525/// This initialize the underlying bytes representation of MultiSig. It encodes
526/// [struct MultiSig] as the MultiSig flag (0x03) concat with the bcs bytes
527/// of [struct MultiSig] i.e. `flag || bcs_bytes(MultiSig)`.
528impl AsRef<[u8]> for MultiSig {
529    fn as_ref(&self) -> &[u8] {
530        self.bytes
531            .get_or_try_init::<_, eyre::Report>(|| {
532                let as_bytes = bcs::to_bytes(self).expect("BCS serialization should not fail");
533                let mut bytes = Vec::with_capacity(1 + as_bytes.len());
534                bytes.push(SignatureScheme::MultiSig.flag());
535                bytes.extend_from_slice(as_bytes.as_slice());
536                Ok(bytes)
537            })
538            .expect("OnceCell invariant violated")
539    }

542/// The struct that contains the public key used for authenticating a MultiSig.
543#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
544pub struct MultiSigPublicKey {
545    /// A list of public key and its corresponding weight.
546    pk_map: Vec<(PublicKey, WeightUnit)>,
547    /// If the total weight of the public keys corresponding to verified signatures is larger than threshold, the MultiSig is verified.
548    threshold: ThresholdUnit,
549}
550
551impl MultiSigPublicKey {
552    /// Construct MultiSigPublicKey without validation.
553    pub fn insecure_new(pk_map: Vec<(PublicKey, WeightUnit)>, threshold: ThresholdUnit) -> Self {
554        Self { pk_map, threshold }
555    }
556
557    pub fn new(
558        pks: Vec<PublicKey>,
559        weights: Vec<WeightUnit>,
560        threshold: ThresholdUnit,
561    ) -> Result<Self, SuiError> {
562        if pks.is_empty()
563            || weights.is_empty()
564            || threshold == 0
565            || pks.len() != weights.len()
566            || pks.len() > MAX_SIGNER_IN_MULTISIG
567            || weights.contains(&0)
568            || weights
569                .iter()
570                .map(|w| *w as ThresholdUnit)
571                .sum::<ThresholdUnit>()
572                < threshold
573            || pks
574                .iter()
575                .enumerate()
576                .any(|(i, pk)| pks.iter().skip(i + 1).any(|other_pk| *pk == *other_pk))
577        {
578            return Err(SuiErrorKind::InvalidSignature {
579                error: "Invalid multisig public key construction".to_string(),
580            }
581            .into());
582        }
583
584        Ok(MultiSigPublicKey {
585            pk_map: pks.into_iter().zip_debug_eq(weights).collect(),
586            threshold,
587        })
588    }
589
590    pub fn get_index(&self, pk: &PublicKey) -> Option<u8> {
591        self.pk_map.iter().position(|x| &x.0 == pk).map(|x| x as u8)
592    }
593
594    pub fn threshold(&self) -> &ThresholdUnit {
595        &self.threshold
596    }
597
598    pub fn pubkeys(&self) -> &Vec<(PublicKey, WeightUnit)> {
599        &self.pk_map
600    }
601
602    pub fn validate(&self) -> Result<MultiSigPublicKey, FastCryptoError> {
603        let pk_map = self.pubkeys();
604        if self.threshold == 0
605            || pk_map.is_empty()
606            || pk_map.len() > MAX_SIGNER_IN_MULTISIG
607            || pk_map.iter().any(|(_pk, weight)| *weight == 0)
608            || pk_map
609                .iter()
610                .map(|(_pk, weight)| *weight as ThresholdUnit)
611                .sum::<ThresholdUnit>()
612                < self.threshold
613            || pk_map.iter().enumerate().any(|(i, (pk, _weight))| {
614                pk_map
615                    .iter()
616                    .skip(i + 1)
617                    .any(|(other_pk, _weight)| *pk == *other_pk)
618            })
619        {
620            return Err(FastCryptoError::InvalidInput);
621        }
622        Ok(self.to_owned())
623    }
