18/// Checks valid rules for entry points, both for module initialization and transactions
19///
20/// For module initialization
21/// - The existence of the function is optional
22/// - The function must have the name specified by `INIT_FN_NAME`
23/// - The function must have `Visibility::Private`
24/// - The function can have at most two parameters:
25///   - mandatory &mut TxContext or &TxContext (see `is_tx_context`) in the last position
26///   - optional one-time witness type (see one_time_witness verifier pass) passed by value in the
27///     first position
28///
29/// For transaction entry points
30/// - The function must have `is_entry` true
31/// - The function may have a &mut TxContext or &TxContext (see `is_tx_context`) parameter
32///   - The transaction context parameter must be the last parameter
33/// - The function cannot have any return values
34pub fn verify_module(
35    module: &CompiledModule,
36    fn_info_map: &FnInfoMap,
37) -> Result<(), ExecutionError> {
38    // When verifying test functions, a check preventing explicit calls to init functions is
39    // disabled.
40
41    for func_def in &module.function_defs {
42        let handle = module.function_handle_at(func_def.function);
43        let name = module.identifier_at(handle.name);
44
45        // allow calling init function in the test code
46        if !is_test_fun(name, module, fn_info_map) {
47            verify_init_not_called(module, func_def).map_err(verification_failure)?;
48        }
49
50        if name == INIT_FN_NAME {
51            verify_init_function(module, func_def).map_err(verification_failure)?;
52            continue;
53        }
54    }
55    Ok(())
56}
57
58fn verify_init_not_called(
59    module: &CompiledModule,
60    fdef: &FunctionDefinition,
61) -> Result<(), String> {
62    let code = match &fdef.code {
63        None => return Ok(()),
64        Some(code) => code,
65    };
66    code.code
67        .iter()
68        .enumerate()
69        .filter_map(|(idx, instr)| match instr {
70            Bytecode::Call(fhandle_idx) => Some((idx, module.function_handle_at(*fhandle_idx))),
71            Bytecode::CallGeneric(finst_idx) => {
72                let finst = module.function_instantiation_at(*finst_idx);
73                Some((idx, module.function_handle_at(finst.handle)))
74            }
75            _ => None,
76        })
77        .try_for_each(|(idx, fhandle)| {
78            let name = module.identifier_at(fhandle.name);
79            if name == INIT_FN_NAME {
80                Err(format!(
81                    "{}::{} at offset {}. Cannot call a module's '{}' function from another Move function",
82                    module.self_id(),
83                    name,
84                    idx,
85                    INIT_FN_NAME
86                ))
87            } else {
88                Ok(())
89            }
90        })
91}

93/// Checks if this module has a conformant `init`
94fn verify_init_function(module: &CompiledModule, fdef: &FunctionDefinition) -> Result<(), String> {
95    if fdef.visibility != Visibility::Private {
96        return Err(format!(
97            "{}. '{}' function must be private",
98            module.self_id(),
99            INIT_FN_NAME
100        ));
101    }
102
103    if fdef.is_entry {
104        return Err(format!(
105            "{}. '{}' cannot be 'entry'",
106            module.self_id(),
107            INIT_FN_NAME
108        ));
109    }
110
111    let fhandle = module.function_handle_at(fdef.function);
112    if !fhandle.type_parameters.is_empty() {
113        return Err(format!(
114            "{}. '{}' function cannot have type parameters",
115            module.self_id(),
116            INIT_FN_NAME
117        ));
118    }
119
120    if !module.signature_at(fhandle.return_).is_empty() {
121        return Err(format!(
122            "{}, '{}' function cannot have return values",
123            module.self_id(),
124            INIT_FN_NAME
125        ));
126    }
127
128    let parameters = &module.signature_at(fhandle.parameters).0;
129    if parameters.is_empty() || parameters.len() > 2 {
130        return Err(format!(
131            "Expected at least one and at most two parameters for {}::{}",
132            module.self_id(),
133            INIT_FN_NAME,
134        ));
135    }
136
137    // Checking only the last (and possibly the only) parameter here. If there are two parameters,
138    // then the first parameter must be of a one-time witness type and must be passed by value. This
139    // is checked by the verifier for pass one-time witness value (one_time_witness_verifier) -
140    // please see the description of this pass for additional details.
141    if TxContext::kind(module, &parameters[parameters.len() - 1]) != TxContextKind::None {
142        Ok(())
143    } else {
144        Err(format!(
145            "Expected last parameter for {0}::{1} to be &mut {2}::{3}::{4} or &{2}::{3}::{4}, \
146            but found {5}",
147            module.self_id(),
148            INIT_FN_NAME,
149            SUI_FRAMEWORK_ADDRESS,
150            TX_CONTEXT_MODULE_NAME,
151            TX_CONTEXT_STRUCT_NAME,
152            format_signature_token(module, &parameters[0]),
153        ))
154    }
155}
